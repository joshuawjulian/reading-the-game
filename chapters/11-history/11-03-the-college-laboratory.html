def O(key, pos, d, w, label):
    return Player(key, pos, "off", d, w, label=label)


def D(key, pos, d, w, label):

def line5():
    """Five offensive linemen, unlabelled (in this chapter 'C' and 'CB' always mean a cornerback)."""
    return [q.moved(label="") for q in offense("gun_2x2") if q.key in OL]


def spread10(rb_w=-2.4, qb_d=-5.0, rb_d=-5.4):
    """Shotgun, four receivers (10 personnel): X and Z on the line, the slots H and F a step off it.
    The back is offset 2.4 yards and a little deeper, so his marker never touches the quarterback's."""
    return line5() + [O("QB", "QB", qb_d, 0.0, "Q"), O("RB", "RB", rb_d, rb_w, "R"),
                      O("X", "WR", -0.7, -16.0, "X"), O("H", "WR", -1.5, -9.0, "H"),
                      O("F", "WR", -1.5, 9.0, "F"), O("Z", "WR", -0.7, 16.0, "Z")]


def check_legal(players):
    """Seven offensive players on the line, and only the two at its ends may be eligible receivers."""
    off = [q for q in players if q.side == "off"]
    on = sorted([q for q in off if q.d > -1.0], key=lambda q: q.w)
    assert len(on) == 7, [q.key for q in on]
    for q in on[1:-1]:
        assert q.key in OL, f"{q.key} is covered up but not a lineman"


def go_to(p, key, end, kind="block", delay=0.0, speed=None, via=()):
    """Move `key` through absolute (d, w) points `via` to `end`, as a block, path, run, route or rush."""
    start = np.array(p._cur(key))
    pts = [tuple(np.array(q) - start) for q in (*via, end)]
    if kind == "block":
        return p.block(key, pts=pts, delay=delay, speed=speed)
    if kind == "route":
        return p.route(key, pts, delay=delay, speed=speed, frame="field")
    return {"path": p.path, "run": p.run, "pull": p.pull, "rush": p.rush}[kind](key, pts, delay=delay, speed=speed)


def _P(fld, p, d, w):
    return fld.xy(*p.to_abs(d, w))


def branch(fld, p, pts, color=style.THIRD, label=None, label_at=None, lw=2.0, fontsize=7.5,
           dash=(0, (3, 2.2)), end="arrow"):
    """A dashed 'option' path through absolute (d, w) points, with an arrowhead (or block tee)."""
    xs, ys = _P(fld, p, np.array([q[0] for q in pts]), np.array([q[1] for q in pts]))
    if end == "arrow":
        fld.ax.plot(xs[:-1], ys[:-1], color=color, lw=lw, ls=dash, zorder=6, solid_capstyle="round")
        fld.ax.add_patch(FancyArrowPatch((xs[-2], ys[-2]), (xs[-1], ys[-1]), arrowstyle="-|>",
                                         mutation_scale=11, color=color, lw=lw, zorder=6))
    else:
        fld.ax.plot(xs, ys, color=color, lw=lw, ls=dash, zorder=6)
    if label:
        x, y = _P(fld, p, *(label_at or pts[-1]))
        fld.ax.text(x, y, label, fontsize=fontsize, color=color, fontweight="bold", ha="center",
                    va="center", zorder=9,
                    bbox=dict(boxstyle="round,pad=0.2", fc=style.SURFACE, ec=color, lw=0.8))


def pitch_arc(fld, p, a, b, color=style.BALL, rad=0.25):
    """A pitch or a throw: a dashed brown arc from (d, w) a to (d, w) b."""
    (x0, y0), (x1, y1) = _P(fld, p, *a), _P(fld, p, *b)
    fld.ax.add_patch(FancyArrowPatch((float(x0), float(y0)), (float(x1), float(y1)), arrowstyle="-|>",

                                     connectionstyle=f"arc3,rad={rad}"))


def read_line(fld, p, a, b, color=GREEN, num=None, at=0.5):

def box_shade(fld, p, d0, d1, w0, w1, color=GREEN, alpha=0.07):
    """Shade a rectangle of the field (the 'box') between depths d0-d1 and widths w0-w1."""
    (x0, y0), (x1, y1) = _P(fld, p, d0, w0), _P(fld, p, d1, w1)
    fld.ax.add_patch(Rectangle((min(x0, x1), min(y0, y1)), abs(x1 - x0), abs(y1 - y0), fc=color, alpha=alpha,

def clean(ax, numbers=True, hashes=True):
    """Drop hash ticks and/or painted yard numbers (box-sized windows), clip everything to the window."""
    for ln in list(ax.lines):
        xs, ys = ln.get_data()
        if hashes and len(xs) == 2 and ln.get_linewidth() == 0.5 and np.hypot(xs[1] - xs[0], ys[1] - ys[0]) < 1.0:
            ln.remove()
    for t in list(ax.texts):
        if numbers and t.get_text().isdigit() and t.get_alpha() == 0.55:
            t.remove()
        else:
            t.set_clip_on(True)


def draw(p, width_in=5.8, ax=None, numbers=True, hashes=True, **kw):
    """Play.draw, sized for the page, cleaned for a box-sized window."""
    fld = p.draw(width_in=width_in, ax=ax, **kw)
    clean(fld.ax, numbers=numbers, hashes=hashes)
    return fld


def panel_title(ax, text, size=9):
    ax.set_title(text, fontsize=size, loc="left", color=style.INK, fontweight="bold")


class hashes_at:
    """Temporarily draw gridiron Fields with college hash marks (60 feet, 20 yards, in from each sideline).
    gridiron.field reads HASH_NEAR / HASH_FAR when it draws, so we swap them and put them back."""

    def __init__(self, from_sideline=20.0):
        self.near, self.far = from_sideline, FIELD_WIDTH - from_sideline

    def __enter__(self):
        self.saved = (gfield.HASH_NEAR, gfield.HASH_FAR)
        gfield.HASH_NEAR, gfield.HASH_FAR = self.near, self.far
        return self

    def __exit__(self, *exc):
        gfield.HASH_NEAR, gfield.HASH_FAR = self.saved
        return False

MARK_R = 0.72


def tracking(play):
    df = play.to_tracking()
    df["_id"] = df["displayName"].astype(str)
    return df


def carrier(play, t):
    """Who holds the ball at time t (QB until the first handoff, pitch or catch)."""
    holder = "QB"
    for kind, to, te in sorted(play.ball_events, key=lambda e: e[2]):
        if t >= te + (0.35 if kind in ("pitch", "pass") else 0.0):
            holder = to
    return holder


def panel(ax, play, df, t, lateral, window, rings=()):
    """One moment of a simulated play: markers, trails, the ball, rings (and a ring on the ball carrier)."""
    tmap = df.groupby("frameId")["time"].first()
    fid = int((tmap - t).abs().idxmin())
    ylo, yhi = sorted((play.ball_y - lateral[0], play.ball_y - lateral[1]))
    fld = Field(ax=ax, orient="vertical", los=play.los, window=window, numbers=False, hashes=False,
                lateral=(ylo, yhi), to_go=play.to_go)
    cur = df[df["frameId"] == fid]
    shown = cur[cur["x"].between(play.los + window[0] + 0.4, play.los + window[1] - 0.4)
                & cur["y"].between(ylo + 0.4, yhi - 0.4)]
    ids = set(shown["_id"])
    trails = df[(df["frameId"] <= fid) & df["_id"].isin(ids) & (df["side"] != "ball")
                & df["x"].between(play.los + window[0], play.los + window[1])]
    for (_, side), g in trails.groupby(["_id", "side"]):
        xs, ys = fld.xy(g["x"].values, g["y"].values)
        ax.plot(xs, ys, color=style.OFFENSE if side == "off" else style.DEFENSE, lw=0.9, alpha=0.4, zorder=3)
    ppy = fld.pts_per_yard()
    for _, r in shown.iterrows():
        x, y = (float(v) for v in fld.xy(r["x"], r["y"]))
        if r["side"] == "ball":
            ax.add_patch(Ellipse((x, y + 0.5), 0.75, 0.45, fc=style.BALL, ec="white", lw=0.5, zorder=11))
            continue
        lab = str(r.get("label", "") or "")
        if r["side"] == "off":
            ax.add_patch(Circle((x, y), MARK_R, fc=style.OFFENSE, ec="white", lw=0.6, zorder=8))
            tc = "white"
        else:
            ax.add_patch(RegularPolygon((x, y), 4, radius=MARK_R * 1.3, orientation=np.pi / 4,
                                        fc=style.DEFENSE_TINT, ec=style.DEFENSE, lw=1.0, zorder=8))
            tc = style.INK
        fs = max(4.0, min(8.0, ppy * (0.8 if len(lab) <= 1 else 0.6)))
        ax.text(x, y, lab, ha="center", va="center", fontsize=fs, color=tc, fontweight="bold", zorder=10)
    for key, col in list(rings) + [(carrier(play, t), style.THIRD)]:
        rr = cur[cur["displayName"] == key]
        if len(rr):
            x, y = fld.xy(rr["x"].iloc[0], rr["y"].iloc[0])
            ax.add_patch(Circle((float(x), float(y)), 1.25, fc="none", ec=col, lw=1.9, zorder=7))
    return fld


def strip(plays, times, lateral, window, notes=None, rows=None, rings=(), width_in=6.5, ncols=None):
    """Frame strip: one row per play (an option's branches), or one play wrapped over rows."""
    dfs = [tracking(p) for p in plays]
    if len(plays) > 1:
        cells = [(i, t) for i in range(len(plays)) for t in times]
        ncols = len(times)
    else:
        cells = [(0, t) for t in times]
        ncols = ncols or min(3, len(times))
    nrows = int(np.ceil(len(cells) / ncols))
    pw = width_in / ncols
    span_w, span_d = lateral[1] - lateral[0], window[1] - window[0]
    fig, axes = plt.subplots(nrows, ncols, squeeze=False,
                             figsize=(width_in, nrows * (pw * span_d / span_w + (0.55 if notes else 0.25))))
    for ax in axes.flat[len(cells):]:
        ax.axis("off")
    for k, (ax, (i, t)) in enumerate(zip(axes.flat, cells)):
        panel(ax, plays[i], dfs[i], t, lateral, window, rings)
        head = f"{k % len(times) + 1 if len(plays) > 1 else k + 1}.  " + ("snap" if abs(t) < 1e-6 else f"{t:+.1f} s")
        if rows and k % ncols == 0:
            head = rows[i] + "   " + head
        ax.set_title(head, fontsize=8, color=style.INK_2 if not (rows and k % ncols == 0) else style.INK,
                     loc="left", fontweight="bold" if rows and k % ncols == 0 else "normal")
        if notes:
            ax.text(0.5, -0.03, notes[k], transform=ax.transAxes, ha="center", va="top",
                    fontsize=7.2, color=style.INK)
    fig.tight_layout(h_pad=1.4 if notes else 0.6, w_pad=0.5)
    return fig


def video(plays, times, lateral, window, notes=None, rows=None, rings=(), width_in=6.4):
    """Web version: the same panels drawn at every tenth of a second, side by side for branches."""
    import shutil
    from matplotlib import animation
    from IPython.display import HTML, display
    dfs = [tracking(p) for p in plays]
    t0 = min(float(df["time"].min()) for df in dfs)
    t1 = max(float(df["time"].max()) for df in dfs)
    grid = np.round(np.arange(t0, t1 + 1e-9, 0.1), 2)
    n = len(plays)
    span_w, span_d = lateral[1] - lateral[0], window[1] - window[0]
    pw = (width_in if n == 1 else width_in / n) * (0.85 if n == 1 else 1.0)
    fig, axes = plt.subplots(1, n, squeeze=False, figsize=(pw * n, pw * span_d / span_w + 0.6))
    axes = axes.flat

    def update(t):
        for i, ax in enumerate(axes):
            ax.clear()
            panel(ax, plays[i], dfs[i], t, lateral, window, rings)
            txt = "snap" if abs(t) < 0.05 else f"{t:+.1f} s"
            if notes:
                per = len(times)
                mine = notes[i * per:(i + 1) * per] if n > 1 else notes
                cur = [m for tt, m in zip(times, mine) if tt <= t + 1e-6]
                txt += ("\n" + cur[-1].replace("\n", " ")) if cur else ""
            head = (rows[i] + ": ") if rows else ""
            ax.set_title(head + txt, fontsize=7.5, color=style.INK, loc="left")
        return []

    anim = animation.FuncAnimation(fig, update, frames=list(grid), interval=100)
    html = (anim.to_html5_video(embed_limit=40).replace("<video ", '<video style="max-width:100%;height:auto" ')
            if shutil.which("ffmpeg") else anim.to_jshtml(default_mode="once"))
    plt.close(fig)
    display(HTML(html))


def option_animation(plays, times, lateral, window, notes=None, rows=None, rings=(), **kw):
    """Video on the website; a numbered, captioned frame strip in the PDF."""
    if pdf_build():
        strip(plays, times, lateral, window, notes=notes, rows=rows, rings=rings, **kw)
        plt.show()
    else:
        video(plays, times, lateral, window, notes=notes, rows=rows, rings=rings)

Saturday night, third-and-6. The offense snaps the ball eleven seconds after the last play ended. Four receivers stand spread across a field whose hash marks seem to sit almost on the sideline; the ball is on the left one, and the whole offense seems to have drifted to the wide side of the field. The quarterback, five yards behind the center, puts the ball in his running back’s belly, watches a defensive end, pulls the ball, takes two steps, and flips it to a slot receiver bowing backward toward the sideline. Twelve yards. The next afternoon you watch an NFL team run almost the same play: same read, same flip, same receiver. The NFL version is a little quicker, the line looks a little different, and the announcer calls it “a college concept.” He is right, and he could have said the same thing about half the plays in the game. Where did they come from, why does college football keep inventing them, and why do some make the jump to Sunday while others never do?

You’ll be able to:

  • Explain the four forces that make college football the sport’s laboratory: talent gaps across hundreds of programs, wide hash marks, rules that tolerate the run-pass blur, and roster churn and tempo.
  • Trace the main offensive lineages (split-T and veer, wishbone and flexbone, Wing-T, BYU’s passing game, the Air Raid, the spread offense, the spread option, the pistol, tempo and the RPO) and point to their DNA in an NFL play.
  • Trace the defensive answers (the 3-3-5 stack, pattern-matching quarters, tite and mint fronts, and the three-high safety defense) and say which problem each was built to solve.
  • Say when and how each idea crossed into the NFL, and why some never did.
  • Read a college broadcast with the rulebook differences in mind (hash width, one foot in bounds, three yards downfield, a 15-yard pass-interference cap, the college clock, overtime, targeting and more), and predict how each one changes what the teams do.

  • Option Football: the read key, give and keep, the zone read, the triple option’s dive, keep and pitch, and the count (#1 dive key, #2 pitch key, #3 alley, #4 deep). This chapter traces where those plays came from rather than re-teaching them.

  • Quarters and Match Coverage: two deep safeties who read the receivers and fit the run from depth; the read safety; pattern matching; Cover 6.

Why college invents and the NFL borrows

Watch enough football and a pattern appears. Almost every offensive idea that changed the NFL in the last thirty years, the shotgun spread run game, the zone read, the pistol, the RPO, the spread no-huddle, the Air Raid’s handful of pass concepts, arrived from college (the NFL had flirted with the no-huddle in the K-Gun era; college made it an identity), often a decade or more after it was born there. The traffic runs the other way much less often. That isn’t because college coaches are smarter. It is because college football is a different environment, and four features of that environment reward invention.

1. More teams, and bigger talent gaps

The NFL has 32 teams drawing from one talent pool, with a draft and a salary cap built to keep them roughly equal (1978 and the Passing Revolution tells how the cap arrived). College football has more than 130 teams in its top division alone, the Football Bowl Subdivision (FBS), and hundreds more below it, and no mechanism at all to keep them equal.1 Every year a few programs sign most of the best high-school players, and everyone else has to play them.

That gap is the mother of invention. A coach who can’t recruit the bigger, faster players has two choices: lose, or change the problem. Almost every lineage in this chapter starts with a coach in the second group: Bill Yeoman at Houston with an offensive line that couldn’t block, LaVell Edwards at BYU with no chance of out-recruiting the Southwest’s powers, Hal Mumme and Mike Leach at tiny Iowa Wesleyan, Rich Rodriguez at Glenville State, a Division II school in West Virginia. Their inventions all do the same thing in different ways: they change the arithmetic so that the defense’s better players have more to defend than they have bodies, or have to make a decision they aren’t trained for.

The NFL has talent gaps too, but small ones. When the other team’s players are as good as yours, a gimmick that exploits a mismatch is worth less, and a scheme that gets your quarterback hit is worth much less.

2. A wider field

College hash marks are 40 feet apart, 60 feet in from each sideline; NFL hash marks are 18 feet 6 inches apart, the width of the goal posts. The field is the same size. What changes is where the ball can start, and that changes everything about spacing. With the ball on a college hash, the short (boundary) side is 20 yards wide and the wide (field) side 33. Before 1993 the college hashes sat even wider, a third of the way across (53 feet 4 inches from each sideline, the high-school spot), so the wishbone and veer offenses of this chapter played with a boundary under 18 yards wide.2 The Game in Fifteen Minutes drew the same four-receiver formation on both fields.

A wide field is a gift to a horizontal offense, one that stretches a defense from sideline to sideline. With 33 yards to the field side, a defense that wants to cover four receivers spread across it has to send defenders a long way from the ball, out of the box (the area near the ball where runs are defended). With 20 yards to the boundary, the offense can put a single receiver into the short side and dare a cornerback to cover him alone, with no room to help. Option and spread offenses were made for that geometry. The NFL’s narrow hashes put the ball near the middle almost every snap, which makes the field more symmetrical and takes away some of the leverage.

3. Rules that tolerate the run-pass blur

College rules have been friendlier to offenses that mix run and pass on the same snap. The biggest difference is the ineligible downfield rule from Run-Pass Options: in college an offensive lineman may be up to three yards past the line when a pass that crosses it is thrown, and the rule doesn’t apply at all to passes caught behind the line; in the NFL, the limit is one yard on any forward pass.3 Three yards is room to run-block for a beat and still throw. It is the reason college RPOs can be “run first, throw later,” and the reason the college game grew so many plays that are a run and a pass at once. Other college rules nudge the same way: one foot in bounds makes a sideline catch easier, and a defensive pass interference foul costs at most 15 yards, which changes the math of a deep throw (the rules section near the end puts them all side by side).

4. Roster churn, practice limits and tempo

A college roster turns over every four or five years, and the best players leave sooner. NCAA rules cap the hours a team can make players practice in season.4 A system that has to be taught to a new quarterback every two or three years, in limited hours, rewards simplicity: a small number of plays repeated until they are automatic, which is exactly what the Air Raid and the spread option were. Simplicity, in turn, makes tempo possible. A team that runs twelve concepts can call them with one word or one hand signal and snap the ball before the defense has substituted or even lined up. College football’s clock (for decades it stopped after every first down) gave fast teams more snaps to use it on.

The filter: what the NFL keeps

Those same four forces explain what doesn’t cross. The NFL is a filter, and an idea has to survive four tests to get through it.

  • The quarterback test. NFL quarterbacks are the most expensive players in the sport, and every hit is a risk the team can’t insure. Ideas that hit the quarterback on most snaps (the triple option, the wishbone) stop at the border; ideas that hit him occasionally (the zone read) cross as a change-up.
  • The talent test. NFL defenders are fast enough, and evenly enough spread across the league, that a scheme built on a mismatch has to work against the best defense on the schedule, not the average one.
  • The rules test. The one-yard rule, two feet in bounds and the spot-foul DPI all change how an idea plays on Sunday. Some ideas cross in modified form (the RPO, with quicker throws); some don’t.
  • The people test. Ideas reach the NFL most easily inside players. A quarterback who ran the zone read for four years arrives knowing how; a coach who designed it has to convince an NFL building to change.

The option line: split-T, veer, wishbone, flexbone

Option Football taught how the option works: leave a defender unblocked, read him, and give the ball to whoever he isn’t playing. The history of that idea is a history of college coaches finding new backfields from which to run it.

It begins in 1941 with Don Faurot at Missouri, who widened the T-formation’s line splits and built the split-T, a series of dives, quarterback keeps and pitches that worked like a modern option. Bud Wilkinson took it to Oklahoma, and Oklahoma’s split-T teams won 47 straight games in the 1950s (03-04 has the details). The split-T never became an NFL offense, but it fixed the pattern for the next seventy years: option football is what you run when the other team’s linemen are bigger than yours.

The veer (Houston, 1965). Bill Yeoman’s Houston teams of the mid-1960s couldn’t block the defensive tackles, so Yeoman stopped trying: the play-side tackle released inside and the quarterback read the first defender outside him, giving to a diving halfback or keeping. That is the veer, and with the other halfback trailing as a pitch man it is the triple option. It made Houston one of the highest-scoring teams in the country: in 1968 the Cougars beat Tulsa 100–6.5

The wishbone (Texas, 1968). Two years later, Texas offensive coordinator Emory Bellard moved the split-T’s fullback up close behind the quarterback and set the two halfbacks deeper and wider behind him: the wishbone of Classic Formations. Darrell Royal’s Longhorns won 30 straight games with it; Oklahoma and Alabama copied it; and through the 1970s it was the most feared offense in college football.6

The wishbone’s genius was that every back had a job on every play, to either side: dive, lead or pitch. The defense had to assign someone to each of three runners on both sides of the formation, with every assignment perfect, and do it without knowing which way the play was going until the snap. In 1971 Oklahoma’s wishbone averaged 472.4 rushing yards a game, still the major-college record.7 Here is the play to one side.

Why did it fade? The same forces that made it, in reverse. Defenses got faster through the 1970s and 1980s, and fast defenders who play their assignments shrink the margins the triple option lives on. The best high-school quarterbacks wanted to pass, because passers were what the NFL drafted, and the programs that could recruit them stopped needing a talent equalizer. And it was hard to come from behind with an offense that threw a dozen times a game. Oklahoma won its last national title with the wishbone in 1985; by the 1990s the formation survived mostly at the service academies (Army, Navy and Air Force).8

The flexbone (late 1970s and 1980s). The wishbone’s last descendant fixed its biggest weakness, width. By standing the two halfbacks just off the line as slotbacks (A-backs), the flexbone spread the defense like a passing formation while keeping three runners near the ball; one A-back’s pre-snap motion became the pitch. Its invention is disputed. The name is usually traced to Fisher DeBerry’s Air Force teams of the 1980s (the wishbone, he is said to have remarked, needed to be more flexible), and Ken Hatfield, DeBerry’s predecessor there, ran a version at Arkansas later in the decade. Paul Johnson, Georgia Southern’s offensive coordinator from 1985, is the coach most credited with building it into a full system, and the formation itself is the double-slot set of the run-and-shoot (more on that offense below) with the option added. Johnson’s Navy and Georgia Tech teams are its modern face (02-03).9

The Wing-T and BYU: misdirection and the passing equalizer

Not every college lineage was an option. Two others, one running and one passing, fed the NFL more quietly.

The Wing-T (Maine and Delaware, about 1950). Dave Nelson crossed the T-formation with the single wing’s wingback to make the Wing-T, and his successor at Delaware, Tubby Raymond, ran it for 36 seasons.11 Its logic, taught in Perimeter Runs and Misdirection, is series football: several plays that start with the same backfield action and end differently, so a defense that flows to the first thing it sees is punished. The buck sweep, the trap, the counter and the waggle (the quarterback fakes the sweep and rolls out the other way to pass) all start the same way. The Wing-T itself never became an NFL offense, but its logic did, twice: through high-school coaches such as Gus Malzahn, who ran its series from the shotgun at Auburn (below), and through the NFL’s jet-motion era of the late 2010s. Sean McVay’s Rams and Kyle Shanahan’s 49ers ran what is, in effect, a Wing-T series with a receiver in the wingback’s role: the jet sweep one way, then the same motion with a handoff the other way, then the same motion with a bootleg pass off it. The defense that flows to the motion is punished, which is the Wing-T’s whole idea (03-05 has the film rooms). That is why the timeline gives the Wing-T an NFL diamond in 2017, while admitting it is the idea, not the formation, that crossed.

BYU (1972 to 2000). LaVell Edwards took over a Brigham Young program in Provo, Utah, that had no hope of out-recruiting the big programs, and decided to throw. Through the 1970s and 1980s BYU passed when almost nobody else in college football did, spreading the ball to backs and tight ends with timing routes and crossing patterns, and produced a line of record-setting quarterbacks: Gifford Nielsen, Marc Wilson, Jim McMahon, Steve Young, Robbie Bosco and Ty Detmer, who won the Heisman Trophy in 1990. In 1984 BYU went 13–0 and won the national title.12

BYU’s passing game didn’t cross into the NFL as a system. It crossed in people, which is the first example of a pattern this chapter keeps finding. McMahon won a Super Bowl with the 1985 Bears and Young two MVP awards with the 49ers; Mike Holmgren, BYU’s quarterbacks coach in the early 1980s, went to San Francisco and later built the Packers’ West Coast offense; Andy Reid played offensive line for Edwards; and Norm Chow, Edwards’s assistant for 27 years and his play-caller from 1982, became one of college football’s most celebrated offensive coordinators at USC and, briefly, the Tennessee Titans’.13 And BYU’s spacing and crossing routes became the raw material of the next lineage.

The run-and-shoot (Portland State, 1975 to 1980). The NFL side of this offense is in 11-02; its roots are a textbook small-school story. Glenn “Tiger” Ellison designed it as a high-school coach in Middletown, Ohio, and Mouse Davis, after 15 seasons coaching high school in Oregon, refined it as Portland State’s head coach, where June Jones and then Neil Lomax set passing records. Davis carried it to the USFL’s Houston Gamblers in 1984 and the Detroit Lions in 1988.14

ImportantWhy it exists

Every lineage so far answers the same problem: the other team has better players. The option answers it by reading a defender instead of blocking him (one fewer man to beat), the Wing-T by making the defense guess wrong about where the ball is going (defenders who flow the wrong way don’t count), and BYU’s passing game by spreading the ball to five receivers so a bigger, slower defense has to cover all of them. Each trade is real. The option gives up the passing game and puts the quarterback in harm’s way; the Wing-T gives up width and demands huge practice time for its fakes; the passing game gives up the clock (every incompletion stops it, so a passing team can’t sit on a lead) and asks a quarterback to be very good. College coaches could make those trades because their alternative was losing. NFL coaches, whose players are evenly matched, usually take the idea and leave the trade.

The Air Raid: subtraction as a strategy

Hal Mumme was the head coach at Iowa Wesleyan, an NAIA school (a small-college association outside the NCAA’s divisions) in Mount Pleasant, Iowa, when he hired a law-school graduate turned coach named Mike Leach as his offensive coordinator in 1989. Their plan, borrowed openly from BYU and then pushed much further, was to throw on almost every down from four- and five-receiver sets, with very wide line splits, and to run very few plays practised an enormous number of times. Leach gave it its name, helped by an old air-raid siren that a fan sounded after touchdowns (accounts differ on whether that began at Iowa Wesleyan or at their next stop). The pair took it to Valdosta State (1992–1996) and then to Kentucky, where in 1998 quarterback Tim Couch became a consensus All-American and, the next spring, the first pick in the NFL draft.15

Offensive Systems explains the Air Raid’s language (one-word calls, a slim playbook) and Dropback Concepts draws its signature concepts: mesh, the shallow cross, Y-cross and four verticals. What matters for this chapter is why a small menu was a weapon:

  • Repetition beats variety when your players are worse. A quarterback who has read the same mesh concept against every coverage ten thousand times throws it before the defender has finished deciding.
  • A small menu travels. It can be taught quickly to a new quarterback every two or three years, and it fits on a wristband or in a hand signal, which makes the no-huddle easy.
  • Spreading five receivers shrinks the box, so even a pass-first team can run the ball against a light front (more on that arithmetic in the next section).

Leach became Texas Tech’s head coach in 2000 and spent a decade there, then Washington State and Mississippi State, until his death in December 2022. His tree became the Air Raid’s distribution network: Dana Holgorsen, Sonny Dykes, Kliff Kingsbury and Lincoln Riley among many others.16 Riley’s Oklahoma and USC offenses produced three Heisman-winning quarterbacks who went first in the NFL draft: Baker Mayfield (2017 Heisman), Kyler Murray (2018) and Caleb Williams (2022).17

Leach’s best team, quarterback Graham Harrell and receiver Michael Crabtree, beat top-ranked Texas 39–33 in Lubbock when Crabtree caught a pass near the sideline, kept his feet and scored with one second left.18

What to notice: count the plays. Texas Tech’s offense ran a handful of concepts all night from spread formations with very wide line splits. Wide splits widen the gaps between the defensive linemen too, which makes stunts harder to run, forces the defense to show where its pressure is coming from, and opens clear throwing windows for a quarterback on a quick three- or five-step drop. And watch where the ball went: short and quick to the sideline, again and again, to receivers who caught it and ran, until the defense crept up and the vertical route behind them came open.

The Air Raid’s route into the NFL ran through its quarterbacks and coaches. Patrick Mahomes played for Kingsbury at Texas Tech; Kingsbury became the Cardinals’ head coach in 2019, hired straight from college with no NFL coaching experience to bring the Air Raid with him,19 and later Washington’s offensive coordinator with Jayden Daniels (04-08 has that film room). Mesh, the shallow cross and four verticals are now in every NFL playbook. What the NFL left at the border was the pure system: the very wide line splits that NFL pass rushers would knife through, the near-total absence of a power run game, and the refusal to go under center.

The spread offense: a philosophy, not a formation

By the late 1990s these lineages were converging on one idea, and it’s worth naming exactly, because the word gets used loosely. A spread offense lines up in the shotgun with three or four receivers spread across the field so that the defense must send defenders out of the box to cover them, then attacks the space that leaves, with quick passes, with option reads, or with both. Spread and Modern Formations drew the distinction: a spread formation is a picture every NFL team uses; a spread offense is the whole system built on that picture.

The core of the philosophy is arithmetic, the hat math of the run game (count blockers against defenders; the side with more wins) applied to the whole field.

That is the whole idea in one picture. Spreading out is a way to run the ball, not only a way to throw it: four receivers take five defenders out of the box, and a defense that brings one back in leaves a receiver alone. The spread then needs one more thing to finish the job, because six defenders against five blockers is still one free man. Two answers emerged, and they split the spread into two families:

  • Spread to pass (the Air Raid, and Joe Tiller’s Purdue offense with Drew Brees, which Tiller himself called “basketball on grass”):20 accept the free man and throw quickly to whoever the defense left alone.
  • Spread to run (Rich Rodriguez, Urban Meyer and their heirs): make the quarterback a runner, so the read takes care of the free man. That is the spread option, next.

The NFL had plenty of the first family of its own (Air Coryell, the K-Gun, later the 2007 Patriots). What it imported from college was mostly the second, the spread run game, as the shotgun data near the end of the chapter shows.

CautionCommon misconception

“The spread is a passing offense.” Some spread offenses are, but the most influential ones were run-first. Rich Rodriguez’s West Virginia, Urban Meyer’s Florida and Gus Malzahn’s Auburn won with the run game, using the spread to empty the box and the quarterback’s legs to win the numbers.

The spread option: Rodriguez, Meyer and the quarterback’s legs

The spread option is a spread offense whose run game is built on option reads from the shotgun. Its origin story is one of the best in football, and 03-04 tells it: at Glenville State in 1991, Rich Rodriguez’s quarterback fumbled a zone-play handoff in practice, picked the ball up, saw the backside end crashing down on the running back, and ran where the end had been.21 Rodriguez saw the mistake was a play. The zone read married the option’s read to the most common run in football, inside zone, and to the shotgun, where the quarterback could see the read key and the receivers at once.

Rodriguez took it to Tulane and Clemson as an offensive coordinator, then to West Virginia as head coach (2001–2007), where quarterback Pat White and running back Steve Slaton made it the most dangerous running offense in the country. Urban Meyer, at Bowling Green, Utah and Florida, built his own spread option on the same principles, with bigger quarterbacks who could run between the tackles: Alex Smith, whose unbeaten 2004 Utah team won the Fiesta Bowl before Smith went first in the 2005 draft, and Tim Tebow, the 2007 Heisman winner, whose Florida teams won national titles in the 2006 and 2008 seasons.22

What made the spread option more than the zone read was what Rodriguez attached to it. With four receivers spread out, the defense’s apex defenders (the ones ringed in Figure 56.4) are stuck between the box and the slots. Rodriguez’s quarterbacks read two defenders on the same side: the end, to give or keep, and then the apex defender, to keep or throw a bubble screen to the slot. It is the triple option’s logic with the pitch replaced by a pass.

Pistol, tempo and the RPO: the 2000s laboratory

Three more inventions finished the modern college offense, and each one came from a coach looking for an edge the others didn’t have.

The pistol (Nevada, 2005). Chris Ault put his quarterback about four yards behind the center, between under-center and shotgun depth, with the running back directly behind him. The pistol (02-02) kept the shotgun’s view for the quarterback’s read while letting the back run downhill to either side, so the defense couldn’t tell the play’s direction from his alignment. In 2009 Nevada had three 1,000-yard rushers, one of them the quarterback, Colin Kaepernick.24

Tempo (Oregon, Auburn and many others). Chip Kelly, Oregon’s offensive coordinator from 2007 and head coach from 2009 to 2012, ran the spread option at a pace college football had never seen: plays signalled from the sideline with picture boards (big posters whose pictures code the play), the ball snapped as soon as the officials allowed. Kelly went 46–7 as Oregon’s head coach.25 Gus Malzahn, a former Arkansas high-school coach who had learned offense from the Delaware Wing-T book, combined tempo, the shotgun and Wing-T misdirection; as Auburn’s offensive coordinator he won the 2010 national title with quarterback Cam Newton (the play we now call the inverted veer was one of their signatures), and as head coach he won the 2013 Iron Bowl on the “Kick Six.”26 In-Game Adjustments and Tempo explains what tempo does to a defense: it freezes the defense’s personnel and its calls, and wears it down.

Why did tempo work so much better on Saturday? Partly the clock, which (as the first section said) stopped after every first down until 2023, handing a fast team even more snaps. Partly the roster: a college team’s backups are much weaker than its starters, so a defense that can’t substitute, or has to play its second string because the first is exhausted, loses more than an NFL defense would. And partly the menu: tempo needs the small, signal-sized playbook the spread already had.

Defenses fought back with the only tool they had left, the rulebook. In October 2012 Nick Saban, whose Alabama defenses were built to substitute by situation, complained about no-huddle offenses and asked, “Is this what we want football to be?”27 In 2014 the NCAA’s rules committee proposed making offenses wait until 10 seconds had run off the play clock before snapping, so defenses could substitute; after an outcry from coaches, the proposal was tabled before it could be voted on.28 The offense had won the argument. The best evidence came from the critic himself. Saban’s Alabama, the most dominant defensive program of the 2010s, lost games to up-tempo spread teams (including Malzahn’s Auburn on the Kick Six and Ohio State’s spread offense in the playoff that ended the 2014 season), and Saban answered by hiring offensive coordinators who brought spread and up-tempo ideas, starting with Lane Kiffin in 2014. By 2020 his national champions were an up-tempo, RPO-heavy spread offense that set a school scoring record.29

The RPO (everyone, 2000s–2010s). Put a quick throw on a running play and you have the RPO. Run-Pass Options tells its tangled origin story (Air Raid tags, Rodriguez’s bubbles, Meyer’s accidental discovery at Utah, Brett Favre’s claim to have done it on his own) and why no single inventor can be named. The laboratory conditions explain why it was college that made it routine: the three-yard rule from the first section let linemen run-block while the quarterback decided, and a bubble or now screen thrown behind the line wasn’t covered by it at all. Art Briles’s Baylor teams took the RPO to an extreme, with receivers split almost to the sideline and the quarterback, Robert Griffin III in 2011, reading defenders on the run and the throw on the same snap.30

The defense strikes back

Every offensive idea in this chapter created a problem for defenses, and college defenses, facing a spread offense almost every week, solved those problems first. Four answers matter.

The 3-3-5 stack: speed and mystery

The first answer was simply to match the offense’s speed with defensive backs. The 3-3-5 stack has three down linemen, three linebackers stacked directly behind them, and five defensive backs, two of whom are hybrids, safeties who play like linebackers, aligned outside the box over the slots. Joe Lee Dunn is usually credited with its first modern version, at Memphis in 1991, born of the familiar college reason: he didn’t have enough big defensive linemen, so he swapped one for a defensive back.31 (Be careful with the name. Odd and Hybrid Fronts explains that NFL coaches use “3-3-5” for something else: a five-man line in nickel, not this stack.)

The stack’s weapons are speed (an extra defensive back against an extra receiver) and mystery: with eight potential rushers and only three hands in the dirt, a protection scheme that counts the defenders on the line before the snap counts only three. (The hybrid safeties’ names are a dialect: spur and bandit are the names West Virginia’s stack made familiar; other staffs say star, rover, viper, or simply strong and weak safety, so expect a broadcast to use any of them.) Its price is size. Three linemen against five offensive linemen can be pushed around, which is why the stack thrived in conferences full of spread passing offenses and struggled against teams that lined up with tight ends and ran straight at it: double-team a down lineman, climb to a stacked linebacker who has no one in front of him, and the light front has nothing left. West Virginia, under Rodriguez’s defensive coordinator Jeff Casteel, paired the stack with the spread option, so the Mountaineers practised against the offense they defended.32 The NFL never adopted it as a base defense; NFL offensive lines are too good to give away a lineman every down.

Quarters and pattern matching: the college two-high

The second answer was in the secondary. Against spread offenses that ran option and play-action from the same looks on wide college hashes, a defense that put a safety in the box got beaten deep, and one that kept two safeties deep got run over. Quarters with pattern matching let it do both: two safeties 10 to 12 yards deep who read the receivers in front of them, carry the vertical routes and fit the run from depth when it shows. Quarters and Match Coverage tells this history in full: Nick Saban’s pattern-match rules, developed as Bill Belichick’s coordinator in Cleveland and refined at Michigan State, LSU and Alabama; Mark Dantonio and Pat Narduzzi’s Michigan State; Iowa’s long-running split-field quarters; Gary Patterson’s 4-2-5 at TCU.

This is the one lineage that made a round trip. Its roots were in an NFL building (Cleveland in the early 1990s), it matured on college fields against the spread, and when NFL offenses went spread in the late 2010s, NFL defenses went back to the college quarters coaches for the answer. NFL Next Gen Stats measured split-safety coverage on 32.9% of dropbacks in 2018 and 42.0% in 2025, the highest since tracking began.33

Tite and mint: walling off the middle

The third answer was a front built for the spread’s favourite run. With five or six defenders in the box, a defense can’t afford to let inside zone’s double teams climb to the linebackers. The tite front puts a nose on the center and two 4i-techniques on the tackles’ inside shoulders, so every interior lineman is covered and nobody climbs cleanly (05-03 draws it). Kirby Smart’s Georgia calls the same picture the mint front, and Georgia’s defenses won national titles in the 2021 and 2022 seasons with it as a base.34 It is also a case of parallel evolution rather than borrowing: Vic Fangio’s NFL defenses were running their own tite against the same spread run games, and the two traditions met in the middle.

Three-high: Iowa State’s answer to the Air Raid

The newest answer came from the conference where the Air Raid lived. By 2017 the Big 12 (Texas, Oklahoma and their neighbours on the Plains) was a league of spread passing offenses, and Iowa State, under head coach Matt Campbell and defensive coordinator Jon Heacock, was losing shootouts. After a 44–41 overtime loss to Iowa in the third game of 2017, Heacock and Campbell used the bye week to throw out the playbook, and against Texas they lined up in a 3-3-5 with three safeties deep. In the very next game Iowa State, a 31-point underdog with a walk-on backup quarterback, went to Norman and beat third-ranked Oklahoma, Baker Mayfield’s team, 38–31.35

A three-high safety defense shows three safeties deep before the snap, usually from a 3-3-5. It is built on one bet: make the offense run the ball against a light box, and fill the box late.

Why does that work against the Air Raid in particular? Because the Air Raid’s whole menu is built to find the hole in a two-high or one-high shell: the seam against Cover 3, the deep middle against Cover 2, the crossers underneath against man. Against five deep defenders and three underneath, there is no deep hole, and the offense’s best answer, running the ball against a light box, is something many pure passing offenses were least prepared to do. The cost is the run defense if the safety is late: count four in the box and two apex players who may or may not get there, against five linemen and a running back with a head start. Iowa State’s version spread through the Big 12 and beyond (Kansas’s offensive coordinator called Iowa State “the originators”), and three-safety looks became a regular part of college defense.36 NFL defenses have mostly borrowed the personnel (a third safety on the field, the big nickel of A First Look at Defense) and the habit of disguising the shell before the snap (06-06), rather than the full college package. The likely reason is the same filter as always: the bet only pays if a safety arrives on time against NFL backs running behind NFL linemen, and a three-man rush gives an NFL quarterback all day to find the soft spot underneath. The margins that make three-high work against a Big 12 Air Raid are too thin on Sunday.

Crossing over: how the ideas reached Sunday

Every idea arrives a little smaller.

The crossing has a rhythm. Ideas arrive in waves when a generation of players makes them unavoidable, and then the NFL’s defenses answer and the wave recedes. Read Figure 56.9 as two stories. The passing line is the spread formation arriving: from about 19% of dropbacks in 1999 to 55% in 2009, as offenses learned what college coaches had learned, that a quarterback who starts five yards deep sees the defense better and gets the ball out faster. That shift needed no college import at all; the 2007 Patriots, the most famous spread passing offense of its era, were run by NFL lifers. The running line is the college story. In 2011, only 14% of NFL runs came from the shotgun: the NFL used the gun to pass and went under center to run. That was a reasoned choice, and it answers the question the timeline left open. Formation Rules listed what the shotgun costs a run game: the back beside the quarterback takes the handoff moving sideways rather than downhill, a five-yard snap can be fumbled, and play-action is less convincing when the quarterback never turns his back. For a quarterback who never kept the ball, those costs bought nothing. The zone read changed the arithmetic: once the quarterback’s read made a shotgun run plus-one, the costs were worth paying. Then the read-option wave and Chip Kelly’s Eagles arrived, and shotgun runs jumped to 32% in 2013. The spread run game, the zone read, the pistol and then the RPO, had crossed. By 2023 the shotgun accounted for 50% of runs and 72% of all plays. In 2025 both lines fell back (to 66% of all plays), as NFL offenses went back under center to run and play-action, the latest counterpunch, told in The Modern Era. Two cautions: the play-by-play’s shotgun flag comes from the official gamebook descriptions, which probably recorded it less consistently in the early seasons, and it can’t tell a spread-option shotgun run from a plain shotgun handoff.38

The borrowing hasn’t stopped. Pre-snap motion, the Wing-T’s oldest tool, was on 63.9% of NFL plays in 2025 by PFF’s count, up from 37.6% in 2014,39 and The Modern Era follows what is crossing now.

Most rows explain themselves; four deserve a sentence on why.

The hash marks are the one you can see from your couch. Offenses put their strength into the wide field, defenses set their coverage by field and boundary rather than by formation (hence split-field quarters, 06-06), and college kickers sometimes waste a down centering the ball before a field goal.41

The three-yard rule, the one-foot catch and the capped DPI all favour offenses that throw in rhythm off the run and to the sideline; Drill 3 below puts the first two on one play.42 The NFL’s spot-foul DPI pushes the other way, making a deep throw into tight coverage a reasonable gamble (Contact Rules and Penalties measures it).

The clock and the headsets are where college has moved toward the NFL. The 2023 clock change took away roughly seven or eight snaps a game, and the two-minute timeout followed for 2024;43 helmet radios arrived in 2024 too, one player per side, cut off at 15 seconds as in the NFL.44 Before that, an offense that already signalled every play from the sideline lost nothing by going fast, one more reason tempo grew up on Saturday.

Overtime, targeting and kickoffs change the late game. College overtime since 2021 is a series of possessions from the 25 that turns into alternating two-point plays from the third overtime45 (the NFL’s version is in Clock and Game Management). A targeting foul ejects the player; for 2026, in a one-year trial, a first foul no longer carries into the next game.46 And while the NFL rebuilt its kickoff in 2024 to bring the return back (Kickoffs and Returns), college has let a fair catch inside the 25 come out to the 25 since 2018,47 and for 2026 adopted the fair-catch kick, the old NFL rule that allows a free field goal try from the spot of a fair catch.48

NoteMadden vs. real life

In a video game, every college team has roughly the same playbook menus. Real college football is closer to a hundred different sports: on one Saturday a service academy runs the flexbone triple option, an Air Raid team throws 60 times from five-receiver sets, and an SEC power lines up under center with two tight ends. NFL teams, drawing from one talent pool under one set of rules, are far more alike. That variety is the laboratory, and it makes a college game the best place to see a concept in its purest form.

College: a legal 17-yard completion. The guard is inside the three-yard limit, and a pass that crosses the neutral zone is exactly what the college rule allows him to be up to three yards beyond. One foot in bounds is a catch in college.

NFL: no gain. One foot in bounds isn’t a catch in the NFL, so the pass is incomplete. There is also a flag: the guard is more than a yard downfield when a forward pass is thrown, which is illegal man downfield (04-06). The defense chooses: five yards and replay the down, or decline and take the incompletion. Note the planted detail: had the guard still been blocking the man he engaged at the line, the NFL would have let him be carried past a yard, but he had left that block to climb and wasn’t touching anyone.

That is why NFL RPOs look different, and it closes this chapter’s opening scene: the Sunday version of the college play was a little quicker, and its line looked a little different, because the throw has to come out in about a second, before any lineman can drift a yard. The deep seam a college team runs off the same action shows up on Sunday only as play-action, with the linemen pass-blocking.

Takeaways

  • College football is the sport’s laboratory because of its environment: huge talent gaps across hundreds of programs, wide hash marks, rules like the three-yard ineligible limit that tolerate the run-pass blur, and roster churn and tempo that reward small, repeatable systems.
  • The offensive lineages share one aim, changing the arithmetic for the team with lesser players: the option (split-T, veer, wishbone, flexbone, spread option) reads a defender instead of blocking him; the Wing-T makes him guess; BYU and the Air Raid spread the ball to five receivers and repeat a few plays endlessly; the spread offense empties the box so the run game has numbers.
  • The spread option is the triple option rebuilt from the shotgun, with the pitch replaced by a throw; add the three-yard rule and tempo and you have the modern college RPO offense.
  • Defenses answered with speed and disguise (the 3-3-5 stack), two-high match coverage (quarters, which made a round trip back to the NFL), fronts that wall off inside zone (tite and mint) and three-high safeties that concede the light-box run and fill it late.
  • The NFL is a filter: ideas that hit the quarterback, rely on a talent gap or need college rules arrive smaller or not at all. Players carry ideas across better than coaches, which is why the big crossings (2012’s read option, 2017’s RPOs, the Air Raid’s concepts) came with young quarterbacks.
