def go_to(p, key, end, kind="block", delay=0.0, speed=None, via=()):
    """Move `key` through absolute (d, w) points `via` to `end`, as a block, pull, path, run, rush or route."""
    start = np.array(p._cur(key))
    pts = [tuple(np.array(q) - start) for q in (*via, end)]
    if kind == "block":
        return p.block(key, pts=pts, delay=delay, speed=speed)
    if kind == "pull":
        return p.pull(key, pts, delay=delay, speed=speed)
    if kind == "rush":
        return p.rush(key, pts=pts, delay=delay, speed=speed)
    if kind == "route":
        return p.route(key, [tuple(q) for q in (*via, end)], absolute=True, delay=delay, speed=speed)
    return {"path": p.path, "run": p.run}[kind](key, pts, delay=delay, speed=speed)

def clean(ax, numbers=True):
    """Drop hash ticks (and optionally yard numbers); clip text to the window."""
    for ln in list(ax.lines):
        xs, ys = ln.get_data()
        if len(xs) == 2 and ln.get_linewidth() == 0.5 and np.hypot(xs[1] - xs[0], ys[1] - ys[0]) < 1.0:
            ln.remove()
    for t in list(ax.texts):
        if numbers and t.get_text().isdigit() and t.get_alpha() == 0.55:
            t.remove()
        else:
            t.set_clip_on(True)

MARK_R = 0.62


def tracking(play):
    df = play.to_tracking()
    df["_id"] = df["displayName"].astype(str)
    return df


def panel(ax, play, df, t, lateral, window, rings=(), ring_r=0.85):
    """One moment of a simulated play on `ax`: markers, trails, rings."""
    tmap = df.groupby("frameId")["time"].first()
    fid = int((tmap - t).abs().idxmin())
    ylo, yhi = sorted((play.ball_y - lateral[0], play.ball_y - lateral[1]))
    fld = Field(ax=ax, orient="vertical", los=play.los, window=window, numbers=False, hashes=False,
                lateral=(ylo, yhi), to_go=play.to_go)
    cur = df[df["frameId"] == fid]
    shown = cur[cur["x"].between(play.los + window[0] + 0.5, play.los + window[1] - 0.5)
                & cur["y"].between(ylo + 0.5, yhi - 0.5) | (cur["side"] == "ball")]
    ids = set(shown["_id"])
    trails = df[(df["frameId"] <= fid) & df["_id"].isin(ids) & (df["side"] != "ball")]
    for (_, side), g in trails.groupby(["_id", "side"]):
        xs, ys = fld.xy(g["x"].values, g["y"].values)
        ax.plot(xs, ys, color=style.OFFENSE if side == "off" else style.DEFENSE, lw=0.9, alpha=0.4, zorder=3)
    ppy = fld.pts_per_yard()
    for _, r_ in shown.iterrows():
        x, y = (float(v) for v in fld.xy(r_["x"], r_["y"]))
        if r_["side"] == "ball":
            ax.add_patch(Ellipse((x, y + 0.45), 0.7, 0.42, fc=style.BALL, ec="white", lw=0.5, zorder=9))
            continue
        lab = str(r_.get("label", "") or "")
        if r_["side"] == "off":
            ax.add_patch(Circle((x, y), MARK_R, fc=style.OFFENSE, ec="white", lw=0.6, zorder=8))
            tc = "white"
        else:
            ax.add_patch(RegularPolygon((x, y), 4, radius=MARK_R * 1.3, orientation=np.pi / 4,
                                        fc=style.DEFENSE_TINT, ec=style.DEFENSE, lw=1.0, zorder=8))
            tc = style.INK
        fs = max(4.0, min(8.0, ppy * (0.85 if len(lab) <= 1 else 0.62)))
        ax.text(x, y, lab, ha="center", va="center", fontsize=fs, color=tc, fontweight="bold", zorder=10)
    for key, col in rings:
        r_ = cur[cur["displayName"] == key].iloc[0]
        x, y = fld.xy(r_["x"], r_["y"])
        ax.add_patch(Circle((float(x), float(y)), ring_r, fc="none", ec=col, lw=1.6, zorder=7))
    for patch in ax.patches:
        patch.set_clip_on(True)
    return fld


def strip(moments, lateral, window, notes=None, rings=(), ncols=2, width_in=6.2, after=None):
    """Numbered frame strip. `moments` is a list of (play, t, tag)."""
    cache = {}
    nrows = int(np.ceil(len(moments) / ncols))
    pw = width_in / ncols
    span_w, span_d = lateral[1] - lateral[0], window[1] - window[0]
    fig, axes = plt.subplots(nrows, ncols, squeeze=False,
                             figsize=(width_in, nrows * (pw * span_d / span_w + (0.75 if notes else 0.25))))
    for ax in axes.flat[len(moments):]:
        ax.axis("off")
    for i, (ax, (play, t, tag)) in enumerate(zip(axes.flat, moments)):
        if id(play) not in cache:
            cache[id(play)] = tracking(play)
        fld = panel(ax, play, cache[id(play)], t, lateral, window, rings)
        if after:
            after(fld, play, t, i)
        lab = "snap" if abs(t) < 1e-6 else f"{t:+.1f} s"
        ax.set_title(f"{i + 1}.  {lab}" + (f"   {tag}" if tag else ""), fontsize=8.3, color=style.INK_2, loc="left")
        if notes:
            ax.text(0.5, -0.03, notes[i], transform=ax.transAxes, ha="center", va="top",
                    fontsize=7.1, color=style.INK)
    fig.tight_layout(h_pad=1.3 if notes else 0.6, w_pad=0.6)
    return fig


def video(play, notes, lateral, window, rings=(), width_in=4.8, after=None):
    """Web version: the play frame by frame, with each moment's note shown as the caption."""
    import shutil
    from matplotlib import animation
    from IPython.display import HTML, display
    span_w, span_d = lateral[1] - lateral[0], window[1] - window[0]
    fig, ax = plt.subplots(figsize=(width_in, width_in * span_d / span_w + 0.5))
    df = tracking(play)
    tmap = df.groupby("frameId")["time"].first()
    frames = [float(tmap[f]) for f in sorted(df["frameId"].unique())]

    def update(k):
        t = frames[k]
        ax.clear()
        fld = panel(ax, play, df, t, lateral, window, rings)
        if after:
            after(fld, play, t, -1)
        cur = [n for tt, n in notes if tt <= t + 1e-6]
        ax.set_title(("snap" if abs(t) < 0.05 else f"{t:+.1f} s") + (("\n" + cur[-1].replace("\n", " ")) if cur else ""),
                     fontsize=8.2, color=style.INK, loc="left")
        return []

    anim = animation.FuncAnimation(fig, update, frames=range(len(frames)), interval=100)
    html = (anim.to_html5_video(embed_limit=40).replace("<video ", '<video style="max-width:100%;height:auto" ')
            if shutil.which("ffmpeg") else anim.to_jshtml(default_mode="once"))
    plt.close(fig)
    display(HTML(html))


def animation_or_strip(play, moments, lateral, window, rings=(), after=None, ncols=2, width_in=6.2):

Put on a game from the early 1970s. The picture looks familiar: two wide receivers, a tight end, two backs, a quarterback under center. Then watch a receiver run a route. A linebacker hits him as he leaves the line, a safety hits him again eight yards downfield, and nobody throws a flag, because none of it is illegal. Watch an offensive tackle pass-block: his fists are balled against his chest and his elbows are out, like a man shielding himself in a crowd. The league trimmed that downfield contact twice in the mid-1970s and it made little difference: in 1977 NFL teams averaged 17.2 points a game, the fewest since 1942. Twenty-one years later a single team, the 1998 Minnesota Vikings, scored 556 points, a league record at the time. What happened in between? The answer starts with two short changes to the rulebook in March 1978, and with the twenty years of move and counter-move they set off, a story in which every offensive idea you now see on Sundays met a defensive answer, and every answer started the next round.

Where the story starts: the 1977 problem

The previous chapter ended with the defenses on top, and the recap is short. Until 1974 a defender could hit a receiver anywhere on the field until the ball was in the air, and a pass blocker had to keep his hands in to his chest and block with his forearms and shoulders. The league trimmed downfield contact in 1974, and in 1977 it allowed each receiver to be contacted only once and banned the head slap that pass rushers like Deacon Jones had used to stun blockers.1 None of it worked. In 1977 teams scored 17.2 points a game; only 40 percent of offensive plays were passes; and 5.7 percent of passes were intercepted, more than one in eighteen.

Why didn’t “only once” fix it? Because the defense chose where to spend its one hit. A pass play is a timed contract: the receiver promises to be at a spot at a moment, and the quarterback throws to the spot. One collision at 8 yards, from a linebacker the receiver had to run past anyway, was enough to make him half a second late, and a late receiver on a timed throw means an incompletion or an interception. The 1977 rule cut the number of hits but left the one that mattered. And for the pass blockers nothing had changed at all.

The league had reasons to care beyond taste. In October 1977 Pete Rozelle signed television contracts with all three networks that would carry every regular-season game from 1978, then the largest television package ever negotiated, and an expanded 16-game schedule was coming with them.2 A sport selling itself to a national audience every week needed points, and points in the 1970s came mostly from long, grinding drives that defenses were winning more often every season.

Spring 1978: two rules, one direction

In March 1978 the owners passed a package of changes. Two of them did almost all the work.3

The receiver goes free: the Mel Blount rule

The first rule drew a line five yards past the line of scrimmage. Inside that line, a defender squared up in front of a receiver may still contact him, jamming him, rerouting him, knocking him off the line, and he may stay on him (the rule’s word was maintain contact) until the receiver gets level with him or crosses the five-yard line. It is not a one-hit rule; it is a five-yard zone. Beyond it, while the quarterback still has the ball, a defender may not initiate contact that impedes the receiver: no shove, no chuck, no collision. Once the ball is thrown, any contact that interferes is pass interference, as before. (Today’s rulebook adds one condition worth knowing for old and new film alike: the restriction holds only while the quarterback is in the pocket with the ball. Once he scrambles out of it, defenders may chuck receivers downfield again; see Contact Rules and Penalties.) That five-yard zone is the Mel Blount rule (1978): a defender may contact a receiver only within five yards of the line, before the throw. The name honors, if that is the word, Pittsburgh’s 6-foot-3 cornerback Mel Blount, the 1975 defensive player of the year, whose bump-and-run coverage was so physical that the league’s answer was named after him. Some accounts attach the nickname to the 1977 one-chuck rule instead; the five-yard rule of 1978 is the one most people mean.4

Figure 55.1 shows what changed for the defense. Before, every defender between the receiver and his landmark was a potential speed bump. A zone defense could re-route receivers: knock crossers off their paths, steer a tight end toward a waiting safety, and destroy the timing a passing game depends on. After 1978, only the defenders near the line got a shot, and only once the receiver was past them was he free. A linebacker dropping to 8 yards could no longer hit the tight end running past him. He had to cover him, with his feet, and linebackers built to take on guards are not built to run with tight ends.

The deepest effect was on timing. A route that can be disrupted anywhere can’t be thrown on a count, because the quarterback never knows when his receiver will arrive. A route that can be disrupted only in the first five yards can be. If the receiver gets a clean release, he will be at 12 yards, breaking toward the sideline, at the same moment on Sunday that he was in Wednesday’s practice. The quarterback can throw to the spot before the receiver turns around. Every timing route you have met in this book, from the quick game to the dropback concepts, assumes that.

The blocker gets his hands: open-hand pass protection

The second rule was about the other end of the play. From 1978 an offensive lineman pass-blocking could extend his arms and open his hands. That sounds like a technicality; it changed the geometry of the pocket. A blocker restricted to his chest and forearms meets the rusher body to body, and a rusher who gets his shoulder past the blocker’s hip is past him. A blocker who can extend his arms meets the rusher at arm’s length and can steer him: feel which way he is going and push him further that way, past the quarterback, while staying between him and the passer. That is the punch of Pass Protection, and it doesn’t exist without the 1978 rule. Holding (grabbing and restricting the rusher) stayed illegal. What changed is that a lineman could now meet the rusher with extended arms and open hands instead of his chest and forearms.

More time in the pocket matters most to the throws that take the longest. A three-step drop and a quick slant need little protection; a seven-step drop and a 20-yard post need a lot. So the second rule pointed the same way as the first: toward the deep and intermediate passing game the 1970s defenses had strangled.

ImportantWhy it exists

The two rules attacked the 1970s defensive advantage from both ends at once. The Blount rule took away the defense’s best way to wreck a route (hitting the receiver), and the hands rule took away its best way to wreck the quarterback’s clock (beating a blocker who couldn’t use his arms). The league’s stated aim, in the Hall of Fame’s words, was to “open up the game.” What the rules gave up was a style of defense. A team built on big, physical defensive backs who won by contact, like the Steelers who gave the rule its name, now had to win with speed and position instead. And the Steelers are the first example of the chapter’s theme, that the team that adapts first wins: Pittsburgh won the next two Super Bowls with a passing game. Terry Bradshaw was the league’s MVP in 1978 and threw for 318 yards and four touchdowns in the 35–31 win over Dallas in Super Bowl XIII.5

And the rest of the package

Three smaller changes rode along, and each fits the same story. The schedule grew from 14 games to 16 and a second wild card was added in each conference, which meant more games to televise. The league added a seventh official, the side judge, partly to rule on what happens downfield: a new contact rule needs someone standing where the contact happens. And the next spring, in 1979, officials were told to blow the whistle as soon as a quarterback was “clearly in the grasp” of a tackler, the first of many rules protecting the passer that the league’s investment in passing would require.6

CautionCommon misconception

“The Mel Blount rule outlawed bump-and-run coverage.” It didn’t. Inside five yards a cornerback can still jam a receiver at the line, and teams have played press coverage ever since, from the 1980s Raiders to Seattle’s press Cover 3 of the 2010s. What the rule removed was the second, third and fourth hit downfield, the ones that let a zone defense steer receivers wherever it wanted. It was also not the last word. Officials let a good deal of downfield contact go in the years that followed, the league renewed its emphasis on the rule in its 1994 package (later in this chapter), and it was called strictly only from 2004, the “Ty Law rule” of Coverage Fundamentals. On old film, expect to see more contact past five yards than the rulebook allowed.

NoteMadden vs. real life

In a video game the rulebook is fixed, and the best plays are the best plays forever. In the NFL the rulebook is rewritten every spring, and schemes chase the rewrites. The 1978 changes are the clearest example in the sport’s history: no coach invented a play that ended the 1970s scoring drought. The owners did, by changing what a defender was allowed to do with his hands, and then the coaches spent twenty years finding out what that made possible. When a rule changes, ask the question this chapter asks over and over: what did this take away, and from whom?

Offense’s first answer: Air Coryell stretches the field deep

The coach most ready for 1978 was already in the league. Don Coryell had built pass-first offenses at San Diego State in the 1960s and with the St. Louis Cardinals from 1973, and on September 25, 1978, four games into the season, he took over the San Diego Chargers. Coryell’s offense, built on Sid Gillman’s idea that a defense should be made to cover every blade of grass, put receivers in motion, lined them up all over the formation, and made the deep throw the first option on most passing plays. His quarterback, Dan Fouts, had spent five unremarkable seasons in San Diego. In 1979 he threw for 4,082 yards, breaking Joe Namath’s single-season record of 4,007; in 1981 he threw for 4,802, and he became the first quarterback to pass for 4,000 yards in three straight seasons, and San Diego led the league in passing yards every season from 1978 to 1983.8

The offense needed a different kind of tight end, and Coryell had one. Kellen Winslow, a first-round pick in 1979, was big enough to line up next to the tackle and fast enough to run with safeties. Coryell split him out, put him in motion, and threw to him down the seams. A tight end who can’t be hit past five yards and can outrun a linebacker is exactly the matchup the new rules created.

Three receivers attack three spots deep, and against the two deep safeties of Cover 2 the tight end’s seam splits them. Why it works, and what it costs. Air Coryell answers the 1978 rules with depth. Receivers who can’t be rerouted can be sent long, and blockers who can use their hands can hold up long enough for them to get there. The cost is everything a deep passing game always costs: long drops, longer protection, fewer receivers in routes, and interceptions when the throw is late. Coryell’s Chargers had famously poor defenses and never reached a Super Bowl. His ideas lasted anyway. They ran through his assistants (Joe Gibbs, Ernie Zampese) to Norv Turner’s Dallas offenses of the 1990s and to Mike Martz’s 1999 St. Louis Rams, the “Greatest Show on Turf.”

Walsh’s offense treats the short pass as a run. Figure 55.4 is the idea in its plainest form: a three-step drop, and the ball out in about a second to X on a quick slant, while the halfback runs to the flat outside him and the tight end crosses underneath. The Will linebacker can’t cover both the slant and the flat, a horizontal stretch, so whichever he chooses, the quarterback throws the other. The receiver catches the ball in stride, already running, and the yards come after the catch.

The quick slant is only the front door. The heart of Walsh’s offense was the five-step drop thrown on time, with high-low reads (one receiver deep and one short in the same area, so a defender who takes one leaves the other) and with the backs as real receivers rather than last-resort outlets. In 1985 Roger Craig became the first player to run for 1,000 yards and catch passes for 1,000 more in the same season.11

Why the 1978 rules made it work, and which rule did what. Be careful here, because the slant and the flat in Figure 55.4 live mostly inside the five-yard zone, where contact stayed legal. A corner can still jam X at the line, which is why pressing the receivers is still a defense’s first answer to a quick passing game. What the Blount rule gave Walsh was everything just past that line. X breaks inside at about 5 yards and catches the ball at 7, and the Will, who used to be allowed to knock him off his path as he crossed, can now only try to beat him to the spot; on a timed throw he never will. And most of Walsh’s offense lived further out still: crossing routes, digs and hooks at 8 to 15 yards, thrown off the five-step drop, running straight through the linebackers’ zones where re-routing used to happen. The hands rule did the rest. A timing offense needs the pocket to hold for exactly as long as the drop, every time, and blockers who could extend their arms could promise that. Walsh’s offense is the timing game at its purest: drops matched to route depths, the ball thrown to the spot before the receiver turns, and a progression the quarterback can run in a second and a half. It needs less protection than Coryell’s, throws fewer interceptions, and turns a 5-yard completion into a 15-yard gain. The cost is that it needs an extremely accurate quarterback and receivers who run every route to the inch, and that it is long and hard to learn: the West Coast play calls you met in Offensive Systems spell out every receiver’s job because the timing has no slack.

Walsh’s 49ers won Super Bowls after the 1981, 1984 and 1988 seasons, and his assistants carried the system everywhere (the coaching tree is in Case Study: Bill Walsh’s 49ers and the West Coast Offense). Between them, Coryell and Walsh defined the two ways to pass after 1978: stretch the defense vertically, or stretch it horizontally. Nearly every passing game since mixes the two.

Defense’s first answer: more defensive backs

Defenses answered the new passing game first with personnel. Offenses that meant to throw on half their plays wanted more receivers out in their routes, and they found them two ways. Some split a tight end or a back out wide, as Coryell split Winslow. Others took the second back off the field and sent on a third wide receiver, a set Sid Gillman had used since the 1960s, though teams didn’t use three-wide sets routinely, on any down, until the 1990s (the run-and-shoot and the K-Gun, later in this chapter).12 Either way, a linebacker covering that third receiver was a losing matchup. The fix was to take a linebacker off and put a fifth defensive back on: the nickel. The idea was older than 1978 (it is usually traced to Philadelphia assistant Jerry Williams around 1960, named by George Allen, and popularized by Don Shula and Bill Arnsparger’s Miami Dolphins in the 1970s),13 but after 1978 it went from a special package to an everyday one. In long-yardage situations defenses went further, to the dime: six defensive backs.

Each substitution is a trade. A nickel defense has one fewer player built to take on blockers, so it is softer against the run. And substitution became a game of its own: offenses changed their personnel to draw a defense into the wrong package, the substitution matching of Personnel Groupings. A defense in nickel against two tight ends is too light; a defense in base against three receivers is too slow. Hold on to that trade; it decides the K-Gun story later in this chapter.

Defense’s second answer: Arnsparger’s 53 and the 3-4

The second answer was about the front, and it had started in Miami before 1978. Bill Arnsparger, Don Shula’s defensive coordinator, asked the Dolphins to trade for a Cleveland linebacker named Bob Matheson before the 1971 season, and built a package around him. Matheson lined up at the end of the line on the tight end’s side, standing up. On some snaps he rushed the passer like a defensive end; on others he dropped into coverage like a linebacker. The package was named for his jersey number: the 53 defense, Arnsparger’s front in which one hybrid player decided, snap by snap, whether the defense was a 4-3 or a 3-4. The package dates from the start of the 1972 season, and the 1972 Dolphins, Arnsparger’s “No-Name Defense” (Tom Landry’s name for a unit with no stars), went 17–0.14

Look at Figure 55.5 from the quarterback’s side. Before the snap there are three linemen with a hand on the ground and a fourth man standing at the end of the line. Is it four rushers or three? The quarterback and his blockers have to decide where Matheson is going before he goes, and they have no way to know. If they block him and he drops, a blocker is wasted and a linebacker is covering a receiver they thought was open. If they ignore him and he rushes, he is a free rusher. Arnsparger’s own phrase for what the 53 gave him, in a later interview, was the ability to “rush five guys and cover with six” from a look that showed neither. He called it “safe pressure,” and it is the seed of the zone blitz.

What made the 53 special was where it hid the choice. Matheson came on the field as part of an ordinary-looking group, and he was the only standing end, so the offense had one question to answer and no substitution to warn it. The rest of the league took the alignment more than the trick. Through the mid-1970s, teams stood up the two ends of the Oklahoma 5-2 (five linemen and two linebackers, the classic college front) and called it a 3-4. Chuck Fairbanks’s 1974 Patriots and Bum Phillips’s Oilers were among the first to make it their base defense (the competing origin claims are laid out in Odd and Hybrid Fronts), and by the late 1970s and early 1980s it was the league’s most common base defense.

The 3-4 fitted the post-1978 game for a reason that had nothing to do with the nose tackle. A 3-4 has two standing players at the ends of the line, its outside linebackers, and either one can rush or drop, so the 53’s question gets asked on both edges on every snap. What the 3-4 gives up is the 53’s surprise: the offense sees three linemen and four linebackers come on the field and knows what it faces, even if it doesn’t know who is coming. And the 3-4 creates a place on the field for a particular kind of athlete, a linebacker who is really a pass rusher, the stand-up edge. In 1981 the most famous one of all arrived.

Lawrence Taylor and the blind side

The New York Giants took Lawrence Taylor from North Carolina with the second pick of the 1981 draft and played him at outside linebacker in their 3-4. He was the league’s defensive player of the year as a rookie, again in 1982, and a third time in 1986, when he had 20.5 sacks and became the first defensive player voted the league’s MVP unanimously. Bill Parcells, his defensive coordinator and then head coach, and Bill Belichick, the Giants’ defensive coordinator from 1985, built the defense around the fact that no one could block him.15

Taylor’s speed broke the way offenses protected the passer. Through the 1970s, a blitzing linebacker was a running back’s problem: the back stayed in to block whichever linebacker came. A running back blocking Lawrence Taylor was a mismatch, and teams spent the 1980s inventing better answers. San Francisco’s Walsh, in a January 1982 playoff game against the Giants, assigned his best blocker, guard John Ayers, to him, which other teams copied; then coaches moved the job to the tackle, and in the end to whole protection schemes built around where Taylor was (the slide, the chip and the help side of Pass Protection).

Where Taylor lined up mattered as much as how fast he was. He rushed most often from the defense’s right side, the offense’s left (he was the Giants’ right outside linebacker), and that is where a right-handed quarterback is most exposed. Nearly every starting quarterback threw right-handed, so a defense could count on it.

Figure 55.6 shows why. A right-handed quarterback dropping back from under center turns his body to the right as he drops, and when he sets to throw, his left shoulder points at his target and his eyes go downfield. Everything coming from behind his left shoulder is outside his field of view. A rusher from the right side is in front of him or at the edge of his vision, and he can step up or slide away from it. A rusher from the left arrives unseen, at full speed, at a quarterback holding the ball away from his body. That produces fumbles and injuries, the most famous of them Taylor’s 1985 sack that ended Joe Theismann’s career, told in The Twenty-Two.

The result was a new idea of the offensive line. The blind-side tackle is the tackle who protects the side the quarterback can’t see: the left tackle for a right-handed passer. Teams began to put their best pass blocker there, to draft him early and, once a salary cap made every dollar a choice (later in this chapter), to pay him like it. Two caveats keep the idea honest. For a left-handed quarterback (Steve Young in San Francisco, Mark Brunell in Jacksonville in the 1990s) the blind side is the right. And the side matters less once defenses learned to move their best rusher around, as they all do now, and once quarterbacks played from the shotgun and turned less on the drop; today the right tackle is paid almost as well. The blind side still pulls: when a team has to choose, its best pass protector usually plays on the left.

Gibbs: one back, two tight ends, and the counter trey

Taylor’s division rival had the most direct answer. Joe Gibbs, Coryell’s former offensive coordinator in San Diego, became Washington’s head coach in 1981 and played the Giants twice a year. Gibbs’s solution, in his own words, was a game plan built for one man: “We had to try in some way have a special game plan just for Lawrence Taylor… he’s one person that we learned the lesson the hard way. We lost ball games.”16 He took the second back out of the backfield and put him beside the tackle as an extra blocker. Playbooks of the time labelled the backs Q, H and F, so the new player kept his letter: the H-back, a tight end or fullback who lines up a step off the line, outside the tackle, where he can block an edge rusher, chip him and release into a route, or go in motion to the other side. With a second tight end on the line, the formation became the one-back, two-tight-end set (12 personnel) that Washington made its base: the ace of Classic Formations.

The one-back formation answered more than Taylor. With one back, the defense couldn’t key a fullback to find the run. With two tight ends and an H-back, the offense had extra blockers at the line who could also go out for passes, so the same formation threatened both. And it fitted the run game Washington built behind its huge offensive line, the “Hogs”: the counter trey, the guard-and-tackle counter of Gap Schemes, in which the back steps one way and two backside linemen pull the other way to lead him (the language of those runs is in The Language of the Run Game).

Gibbs won three Super Bowls, after the 1982, 1987 and 1991 seasons, with three different starting quarterbacks (Joe Theismann, Doug Williams and Mark Rypien), which says something about how much of the offense was the system.

Buddy Ryan’s 46, and how the one-back attacked it

While Gibbs was building his offense, Chicago’s defensive coordinator Buddy Ryan was building the defense meant to destroy offenses like it. The 46 put eight men near the line, covered the center and both guards with a nose and two 3-techniques (the bear front), stacked both outside linebackers on the tight end’s side, and usually played man coverage behind a blitz. It was designed against the two-back offenses of the early 1980s, and against them it was devastating. A two-back, one-tight-end offense has seven blockers at most (five linemen, the tight end and the fullback) and an obvious strong side, the tight end’s; the 46 put eight defenders near the line and aimed its stack at that side. The arithmetic was simply against the offense: the 1985 Bears allowed the fewest points in the league and beat New England 46–10 in Super Bowl XX (the full plate and Ryan’s story are in Even Fronts).17

The 46’s strength was its aggression and its alignment rules: both outside linebackers go to the tight end. Against a one-back team with two tight ends, the rule becomes a question: which tight end? Whichever side the defense chooses, the other side has fewer defenders, and a counter run pulls two extra blockers to exactly that side.

Follow Figure 55.7. The defense sets its stack to the in-line tight end, Y, on the right: two outside linebackers (W and S) out there, plus the Mike over the right tackle. On the left the offense faces only the end, a tackle, the nose and the strong safety (SS), who in the 46 plays like a linebacker. The H-back lines up on that side. At the snap the back jabs right, toward the stack, and the Mike, reading him, steps that way too. Then the right guard and right tackle pull left. Count the left side now: the guard kicks out the end, the left tackle and left guard block down on the tackle and the nose, the H-back takes the safety, and the pulling tackle leads through the hole onto the Mike. Five blockers for five defenders, and the back has a hole. The two linebackers stacked on the tight end, the players who make the 46 so dangerous against plays toward them, are out of the play: Y’s block walls off the Will at the line, and the Sam, the one man the play leaves unblocked, can only chase from behind. After the handoff the quarterback usually carries out a fake the other way, a bootleg toward the stack, to make that unblocked man hesitate.

Notice what the offense gives up to do this. The center has to block back on the 3-technique the right guard leaves, the hardest block on the play: in the 46 that tackle is coached to slant inside and penetrate, and a pulling guard leaves him a gap to do it, so if the center doesn’t cut him off, the play dies from behind before it starts. The bear front is hard on every gap scheme for that reason, since it covers all three interior linemen, and teams that ran counter at it often pulled the guard and the H-back instead of the guard and the tackle (GH counter, in Gap Schemes), which keeps the tackle home to help on the backside. And the play only works if the defense commits to one side. The 46 had answers of its own: it could set its stack to the H-back, or flip it when he went in motion, or blitz the side the counter was headed for. The more famous way to beat it was not a run at all but spreading out and throwing quickly, which is how Miami handed the 1985 Bears their only loss (see Even Fronts). That is the pattern of this whole era. The offense found a formation that forced the defense to declare, and the defense found a way to declare late.

The rivalry went on after Ryan left Chicago to coach the Philadelphia Eagles (1986–1990): his Eagles knocked nine Washington players out of a November 1990 game, the “Body Bag Game,” and Washington beat them in the playoffs eight weeks later, after which the Eagles fired Ryan.19

Spreading the field: the run-and-shoot and the K-Gun

By the late 1980s the next offensive answer was simply more receivers. If the 46 and the blitz-heavy defenses of the decade were built to win with numbers near the line, an offense could take the numbers away by spreading out.

The run-and-shoot went furthest: four wide receivers, one back, no tight end, and routes chosen after the snap from what the coverage did. Mouse Davis and June Jones, its two great teachers, brought it to the NFL, and between 1989 and the mid-1990s it was the base offense of the Detroit Lions, the Houston Oilers and the Atlanta Falcons. Warren Moon led the league in passing yards for Houston in 1990 and 1991. The defenses’ answer was the one Ryan always preached: hit the quarterback. Four receivers meant only five or six blockers, and blitzes that brought more rushers than that forced the quarterback to throw before the routes had sorted themselves out. Ryan, by then Houston’s defensive coordinator, called his own team’s run-and-shoot the “chuck and duck,” and the feud ended with him swinging at its coordinator, Kevin Gilbride, on the sideline in January 1994.20 The language and the reads of the run-and-shoot are in Offensive Systems.

Buffalo’s version was more durable. The K-Gun was a no-huddle offense run with three wide receivers, one tight end and one back (11 personnel), most often with quarterback Jim Kelly in the shotgun a few yards behind the center, calling the plays at the line.21 Its weapon was the clock between plays. A defense matches the offense’s personnel by substituting when the offense does; when the offense doesn’t substitute and doesn’t huddle, the defense has no time to change, and is stuck with whatever it had on the field when the drive started (the history, including the 1989 attempt to penalize the no-huddle, is in In-Game Adjustments and Tempo). The 1990 Bills went 13–3, scored a league-high 428 points and beat the Raiders 51–3 in the AFC Championship Game. They went on to play in four straight Super Bowls, and lost all four.

Safe pressure: the zone blitz meets the spread

The spread offenses beat man-coverage blitzes with hot throws: when more rushers come than the protection can block, the quarterback throws immediately to the receiver the blitzer left, and with four or five receivers spread wide, someone is always open for a moment. The 1990s defensive answer went back to Arnsparger’s idea of safe pressure: rush five, but from places the protection doesn’t expect, and drop a lineman into the throwing lane the quarterback expects to be open. The zone blitz gave a defense the pressure of a blitz without the man coverage that hot throws punish.

Who invented it is disputed. Some histories credit Arnsparger’s Miami defenses, in the 1970s or in his second stint in the 1980s with the “Killer B’s” (named for how many of its players’ surnames began with B); others credit Dick LeBeau, Cincinnati’s coordinator from 1984, who studied Arnsparger and turned the idea into a system. This book gives Arnsparger the idea and LeBeau the system (the case is laid out in Zone Blitzes). It became famous in Pittsburgh, where Dom Capers and LeBeau built “Blitzburgh”: the 1994 Steelers led the league with 55 sacks. Capers then took it to the expansion Carolina Panthers in 1995, and in their second season those Panthers went 12–4 and reached the NFC Championship Game.23

1994: the league tilts the board again

Look back at the points panel of Figure 55.2. By 1992 and 1993 scoring had fallen to 18.7 points a game, the lowest since 1978, as the zone blitz, better pass rushers and faster defenses caught up. The owners responded as they had in 1978: in March 1994 they adopted a package of changes “to increase offensive production.” It modified line play and the roughing-the-passer rule, adopted the neutral zone infraction (a defender who enters the neutral zone and draws a flinch stops the play dead), and, in the part that matters most to this chapter, renewed the emphasis on “downfield chucking”: the 1978 five-yard rule was still on the books, and officials were told to call it. It also made two changes that are easy to see on any broadcast.24

The two-point conversion. After a touchdown, a team may now try to score two points by running or passing the ball into the end zone from the 2-yard line, instead of kicking for one. College football had used it since 1958 and the American Football League all through the 1960s; the NFL waited until 1994, 24 seasons after the merger, to adopt it. The first one in NFL history was scored by Cleveland’s punter and holder Tom Tupa, running in a faked kick in the opening week of 1994.25

The decision is a small expected-points problem, the kind Expected Points teaches. Kicking was worth almost exactly one point: from 1999 to 2014 (the seasons in nflverse data before the extra point was moved back in 2015), NFL kickers made 98.8 percent of extra points. Going for two succeeded 46.9 percent of the time, which is worth 2 × 0.469 ≈ 0.94 points.26 So on average the kick was slightly better, and teams went for two only a few times a season each.

The two-point try is a situational weapon, because late in a game what matters is not points but the chance of winning. Suppose you trail by 8 and score a touchdown with no time left. Kick, and you lose by 1: a near-certain point that is worth nothing. Go for two, and about 47 times in 100 you tie and go to overtime, roughly a coin flip, so about a 23 percent chance to win instead of none. The same logic runs the other way: score to go up by 1 late, and a kick puts you up 2, where a field goal still beats you, while a two-pointer puts you up 3, where a field goal only ties. Coaches carry a two-point chart that works through every score in this way; it is in Clock and Game Management.

Kickoffs from the 30. The kicking spot moved back five yards, from the 35 to the 30. A kickoff from five yards further back lands five yards shorter, so fewer kicks reach the end zone and more are returned: in 1993 only 68 percent of kickoffs were returned; in 1994, 88 percent. Why does a return help the offense? A touchback then put the ball at the 20, but the average return did better: in the seasons from 1999 to 2010, when the league still kicked off from the 30, a returned kickoff started the offense at about its own 28, eight yards better, and now and then a return went the distance. More returns meant better starting field position, which is what the package was for. (The kickoff spot went back to the 35 in 2011, and the kickoff has since been rebuilt entirely; see Kickoffs and Returns.)27

Scoring rose to 20.3 points a game in 1994 and 21.5 in 1995. The 1978 pattern again: the rules open a gap, and offenses move in.

Free agency and the salary cap: changing how teams are built

The last two changes of the era weren’t about the field at all, and they may have changed the game more than any rule about hands. Until 1993, an NFL player whose contract ended could not simply sign with another team. Through the 1980s the league’s rules made movement rare, and from 1989 to 1992 a system called Plan B let each team protect 37 players and leave only the rest free to sign elsewhere. Players sued, under the antitrust laws, which forbid competing businesses to agree among themselves not to compete; the players’ case was that the teams had done exactly that, agreeing not to bid for one another’s players. In 1992 a federal jury in a case brought by eight players, led by the Jets’ Freeman McNeil, agreed that Plan B violated those laws, and in January 1993 the league and the players settled the remaining suits with a seven-year agreement that rebuilt the system.28

Free agency (1993)

Free agency (1993) is the system, in force since March 1, 1993, that lets a veteran whose contract has expired sign with any team. What counts is accrued seasons: a player earns one for each season he spends at least six regular-season games on full pay status. Today a player with four or more accrued seasons and an expired contract is an unrestricted free agent, free to sign anywhere. With three, he is a restricted free agent: other teams may bid, but his old team can keep him by matching the offer. With fewer than three, he is an exclusive-rights free agent: if his old team offers him a one-year minimum contract, he can’t negotiate with anyone else.29 The first great free-agent signing showed what it could do. Reggie White, the best defensive end of his era, left Philadelphia in 1993 for a four-year, $17 million contract with Green Bay that made him the third-highest-paid player in the league, and after the 1996 season he set a Super Bowl record with three sacks as the Packers won Super Bowl XXXI.30

The salary cap (1994)

In exchange, the owners got the salary cap: a hard limit, the same for every team, on what a club may count against its player payroll in a season, set each year from a share of league revenue. It began in 1994 at $34.6 million per team, higher than the expected $32 million because a new television deal with a new network, Fox, raised league revenue. It is a hard cap: no team may exceed it for any reason.31

ImportantWhy it exists

The two came as a pair, and each answers the other’s problem. Free agency gave players the right to sell their services to the highest bidder; the cap kept the richest owners from simply buying every good player, which the league feared would end competitive balance. The trade the players made was a fixed share of revenue in return for movement; the trade the teams made was freedom to spend in return for a ceiling that made every contract a choice against every other.

The cap counts money by rules of its own, and one of them shapes every big contract. A signing bonus is paid when the contract is signed, but it counts against the cap spread evenly over the years of the contract (today, over at most five). Without that rule a team could pay a huge bonus in a year it had room and show only small salaries afterwards. With it, the same rule became the tool teams use to push cap charges into future years: turn this year’s salary into a bonus and most of its cap charge moves into later seasons. Under a hard cap the fights are not over whether money counts but when. Denver learned that the hard way: in 2004 the league fined the Broncos $950,000 and took a third-round pick for agreements, in 1996 to 1998, that deferred players’ salaries with interest and, the league ruled, got around the cap.32

How the cap changed football strategy. You will see three of its effects in almost every chapter of this book.

  1. Positions got prices. Under a cap, paying one player means not paying another, so teams had to decide which positions are worth the most. The answers that emerged in the 1990s still hold: quarterback first, then the players who decide whether the quarterback has time, the pass rusher and the blind-side tackle. The cap is what turned Lawrence Taylor’s effect into a salary structure. Building a Roster follows that logic to today’s contracts.
  2. Rosters churned, so systems had to travel. With players moving every spring, a team could no longer assume its starters had spent five years learning its playbook. Offenses and defenses had to be teachable to a newcomer in one off-season, which rewarded concept-based languages (one word for a whole pattern, the Erhardt-Perkins idea) and coaches who could install fast. Coaches moved too, and took their systems with them; the coaching trees of Part 12 are a product of the same mobility.
  3. Depth got thinner, and cheap players got valuable. A team at the cap can’t stockpile veterans, so the draft, where players are cheap for their first contract, became the most valuable source of talent, and specialists who play only in certain packages (a third-down pass rusher, a nickel corner) are bought cheaply rather than kept on the roster as insurance.

Where the era landed: 1998 and 1999

By the end of the 1990s, each of the era’s big ideas had a winner. In 1998 the Denver Broncos went 14–2 behind Mike Shanahan’s offense, which married Walsh’s play-action passing to the outside-zone running Alex Gibbs had brought to Denver in 1995 (the story is in Zone Running). Terrell Davis ran for 2,008 yards, and John Elway, in his last game, was the MVP of a 34–19 win over Atlanta in Super Bowl XXXIII.33 The same season the Minnesota Vikings, coached by Dennis Green, a former Walsh assistant, and with a rookie receiver named Randy Moss, scored 556 points, then the most in league history, before losing the NFC Championship Game in overtime.34 And in 1999 the St. Louis Rams, running Mike Martz’s Coryell-descended offense, the “Greatest Show on Turf,” beat the Tennessee Titans 23–16 in Super Bowl XXXIV.35

Two champions and a record-setter, and every line of descent from 1978 is in them: Walsh’s timing game married to Alex Gibbs’s zone running (which came down a different line altogether) in Denver, a Walsh-tree offense with a vertical threat in Minnesota, and Coryell’s vertical game in St. Louis. Meanwhile the defenses had their nickel and dime packages, their 3-4 edge rushers and their zone blitzes, and were waiting for the next rule change. The next round of the story, the spread offenses that came up from college and the defenses built to stop them, is in The College Laboratory and Dynasties and Counterpunches.

It’s 1986, and your right-handed quarterback is about to face the Giants. They are in their 3-4, and their best pass rusher, number 56, is standing at the end of the line on your left. You have one running back, a tight end on the right, and an H-back you can line up on either side. Where do you put the H-back, and what does he do at the snap? And if 56 walks across before the snap to stand outside your tight end, on the side your quarterback faces, what do you do then?

TipAnswer

On the left, next to the left tackle. Number 56 is lined up over your quarterback’s blind side, so that is where you need an extra man. At the snap the H-back chips him: hits him as he comes off the edge, then releases into a short route if nothing else comes. That gives the left tackle help against a rusher he can’t block alone, and gives the quarterback a check-down when the H-back releases. This is the job Joe Gibbs created the position for.

If 56 shifts right, the H-back goes with him, in motion across the formation, and sets up beside the tight end. Now your quarterback can see 56 coming, but seeing him isn’t the same as blocking him: he is still the rusher who beats one blocker, and the right tackle with a tight end and an H-back beside him can chip him twice. The left tackle is left alone with the other outside linebacker (B), which is a fair fight. Defenses learned to move their best rusher around for exactly this reason, to make the offense show where its help goes. The rule is the one from Pass Protection: find the best rusher first and put help where he is. The blind side matters because that is where he usually lines up, not for its own sake.

The only player to cover him in a base defense is a linebacker, the highlighted B, and a wide receiver against a linebacker in space is the matchup the K-Gun was built to create. Base 3-4 personnel is built to stop the run, which is exactly why the K-Gun threw against it. Had the defense been in nickel instead, the call flips: Kelly hands to Thurman Thomas, on a draw or a trap, against a lighter box. That is why there was no right package to start the drive in.

The fix had to come before the drive, because once the offense stops huddling, the defense can only substitute when the offense does, and the offense won’t wait. A defense facing the K-Gun had to pick its package before the drive and live with it, and nickel or dime meant accepting the trade: lighter against the run. That is exactly what Belichick’s Giants did in Super Bowl XXV, with extra defensive backs on the field and a promise to let Thurman Thomas run. And the Giants’ offense took away the other ingredient tempo needs: the ball. You can’t hurry the next snap if you aren’t on the field.

It’s 1979, and the defense calls the Cover 2 it played in 1977: two deep safeties and five underneath. The tight end, Y, gets off the line clean and releases up the seam between the two safeties, and the Mike drops straight back to about 8 yards in the middle. In 1977 the Mike would have spent the defense’s one hit on Y right there (the previous chapter’s drill showed why). What can he do now, where is the ball going, and what does the defense have to change?

TipAnswer

He can’t touch him. Past five yards the Mike may not initiate contact with Y. He has to turn and run with Y, a linebacker chasing a tight end up the field, or let him go and hope a safety gets there. Neither safety can, because each is responsible for his own half of the field, and the seam is exactly between them. The ball goes to Y at 18 to 20 yards, in the hole that every Cover 2 has in the deep middle.

What the defense has to change: it needs someone in the deep middle. It can play a single safety there (Cover 3, or man coverage with a free safety), put a faster defensive back on the tight end, or, the answer that came to define a style of defense, make the Mike run: in Tampa 2 he sprints to the deep middle at the snap, turning Cover 2 into three deep in the middle of the field (Cover 2 and Tampa 2). That linebacker needed a sprinter’s speed, which is one more way the 1978 rules changed what a defensive player looks like.

Takeaways

  • Two 1978 rules opened the passing game from both ends. The Mel Blount rule stopped defenders from hitting receivers past five yards, which made timed routes possible; open-hand pass blocking gave the quarterback more time. Passes went from 40 percent of plays in 1977 to half of them by the early 1980s, and interceptions fell for twenty years.
  • The offense split into two answers. Air Coryell stretched the field deep with protection and big receivers; Walsh’s West Coast offense stretched it sideways with short, timed throws that worked like runs, living in the 5-to-15-yard band the Blount rule had cleared and behind blockers who could now use their hands. Gibbs’s one-back, two-tight-end offense, the run-and-shoot and the K-Gun built on both.
  • Every defensive answer answered something specific: nickel and dime for the third receiver; the 53 and the 3-4 for disguising who rushes; the 46 for two-back offenses; the zone blitz for hot throws. The 3-4 also made room for a new player, the stand-up edge rusher, and Lawrence Taylor broke protections that blocked linebackers with running backs, until the H-back and slide protections answered him. Each new answer created the next question.
  • The blind side made the left tackle a premium position, because a right-handed quarterback can’t see what comes from behind his left shoulder, and because the best rushers lined up there.
  • The league rewrote the rules again when defenses caught up: the 1994 package added the two-point conversion and moved kickoffs back to the 30.
  • Free agency (1993) and the salary cap (1994) changed how teams are built: positions got prices, systems had to be teachable fast, and cheap drafted players became the most valuable kind.
