Seventy years of action and reaction: how the forward pass went from a gamble to a weapon, how the defense answered every new offense, and why by 1977 almost nobody could score.

def check_legal(players):
    """Seven offensive players on the line, and only the two at its ends may be receivers."""
    off = [q for q in players if q.side == "off"]
    on = sorted([q for q in off if q.d > -1.0], key=lambda q: q.w)
    assert len(on) == 7, [q.key for q in on]
    for q in on[1:-1]:
        assert q.pos == "OL", f"covered receiver {q.key}"
    assert len(off) == 11, len(off)
    return players


def hide_yard_numbers(ax):

def XY(fld, p, d, w):

def draw_play(p, lateral, window, width_in=6.5):
    """Draw a play edge to edge at `width_in` (Play.draw's own figure leaves wide empty margins)."""
    span_w, span_d = lateral[1] - lateral[0], window[1] - window[0]
    fig, ax = plt.subplots(figsize=(width_in, width_in * span_d / span_w))

MARK_R = 0.62


def pdf_build():
    return os.environ.get("RTG_FORMAT", "").lower() in ("pdf", "typst", "print")


def tracking(play):
    df = play.to_tracking()
    df["_id"] = df["displayName"].astype(str)
    return df


def panel(ax, play, df, t, lateral, window, rings=(), mark_r=MARK_R):
    """One moment of a simulated play: markers, trails, optional rings (key, color)."""
    tmap = df.groupby("frameId")["time"].first()
    fid = int((tmap - t).abs().idxmin())
    ylo, yhi = sorted((play.ball_y - lateral[0], play.ball_y - lateral[1]))
    fld = Field(ax=ax, orient="vertical", los=play.los, window=window, numbers=False, hashes=False,
                lateral=(ylo, yhi), to_go=play.to_go)
    cur = df[df["frameId"] == fid]
    shown = cur[cur["x"].between(play.los + window[0] + 0.8, play.los + window[1] - 0.8)
                & cur["y"].between(ylo + 0.8, yhi - 0.8)]
    ids = set(shown["_id"])
    trails = df[(df["frameId"] <= fid) & df["_id"].isin(ids) & (df["side"] != "ball")]
    for (_, side), g in trails.groupby(["_id", "side"]):
        xs, ys = fld.xy(g["x"].values, g["y"].values)
        ax.plot(xs, ys, color=style.OFFENSE if side == "off" else style.DEFENSE, lw=0.9, alpha=0.4, zorder=3)
    ppy = fld.pts_per_yard()
    for _, r in shown.iterrows():
        x, y = (float(v) for v in fld.xy(r["x"], r["y"]))
        if r["side"] == "ball":
            ax.add_patch(Ellipse((x + 0.75, y + 0.55), 0.6, 0.36, fc=style.BALL, ec="white", lw=0.5, zorder=11))
            continue
        lab = str(r.get("label", "") or "")
        if r["side"] == "off":
            ax.add_patch(Circle((x, y), mark_r, fc=style.OFFENSE, ec="white", lw=0.6, zorder=8))
            tc = "white"
        else:
            ax.add_patch(RegularPolygon((x, y), 4, radius=mark_r * 1.3, orientation=np.pi / 4,
                                        fc=style.DEFENSE_TINT, ec=style.DEFENSE, lw=1.0, zorder=8))
            tc = style.INK
        fs = max(4.0, min(8.0, ppy * (mark_r / MARK_R) * (0.85 if len(lab) <= 1 else 0.62)))
        ax.text(x, y, lab, ha="center", va="center", fontsize=fs, color=tc, fontweight="bold", zorder=10)
    for key, col in rings:
        r = cur[cur["displayName"] == key].iloc[0]
        x, y = fld.xy(r["x"], r["y"])
        ax.add_patch(Circle((float(x), float(y)), 1.7 * mark_r, fc="none", ec=col, lw=1.7, zorder=9))
    return fld


def strip(play, times, lateral, window, notes=None, rings=(), ncols=2, width_in=6.5, mark_r=MARK_R, extra=None):
    """Numbered frame strip with a fixed window and a note under each frame.
    `extra(fld, i)`, if given, draws on frame i (0-based) after the players."""
    df = tracking(play)
    nrows = int(np.ceil(len(times) / ncols))
    pw = width_in / ncols
    span_w, span_d = lateral[1] - lateral[0], window[1] - window[0]
    fig, axes = plt.subplots(nrows, ncols, squeeze=False,
                             figsize=(width_in, nrows * (pw * span_d / span_w + (0.62 if notes else 0.25))))
    for ax in axes.flat[len(times):]:
        ax.axis("off")
    for i, (ax, t) in enumerate(zip(axes.flat, times)):
        fld = panel(ax, play, df, t, lateral, window, rings, mark_r)
        if extra is not None:
            extra(fld, i)
        ax.set_title(f"{i + 1}.  " + ("snap" if abs(t) < 1e-6 else f"{t:+.1f} s"),
                     fontsize=8.5, color=style.INK_2, loc="left")
        if notes:
            ax.text(0.5, -0.03, notes[i], transform=ax.transAxes, ha="center", va="top", fontsize=7.2,
                    color=style.INK)
    fig.tight_layout(h_pad=1.4 if notes else 0.6, w_pad=0.6)
    return fig


def video(play, times, lateral, window, notes=None, rings=(), width_in=5.2, mark_r=MARK_R, extra=None):
    """Web version of the strip: the same panel at every frame, with the strip's notes as captions."""
    from matplotlib import animation
    from IPython.display import HTML, display
    df = tracking(play)
    span_w, span_d = lateral[1] - lateral[0], window[1] - window[0]
    fig, ax = plt.subplots(figsize=(width_in, width_in * span_d / span_w + 0.4))
    tmap = df.groupby("frameId")["time"].first()

    def update(fid):
        ax.clear()
        t = float(tmap[fid])
        panel(ax, play, df, t, lateral, window, rings, mark_r)
        txt = "snap" if abs(t) < 0.05 else f"{t:+.1f} s"
        if notes:
            cur = [n for tt, n in zip(times, notes) if tt <= t + 1e-6]
            txt += ("   " + cur[-1].replace("\n", " ")) if cur else ""
        ax.set_title(txt, fontsize=8.5, color=style.INK, loc="left")
        return []

    anim = animation.FuncAnimation(fig, update, frames=sorted(df["frameId"].unique()), interval=100)
    html = (anim.to_html5_video(embed_limit=40).replace("<video ", '<video style="max-width:100%;height:auto" ')
            if shutil.which("ffmpeg") else anim.to_jshtml(default_mode="once"))
    plt.close(fig)
    display(HTML(html))

Watch an NFL game today and sooner or later a running back walks up behind the center, five yards deep, with the quarterback split out wide like a receiver. The center snaps the ball through the air straight to the back, an extra lineman stands on one side of the line, and a second back is already sprinting across the formation. The broadcaster calls it the wildcat and treats it as a gimmick. A fan from 1935 would call it Saturday: it is the single wing, the formation football was built on. Now jump to 1977. NFL teams scored 17.2 points a game that season, the fewest since 1942, and not one receiver in the league reached 1,000 yards. How did football get from the single wing to the T-formation to the pro passing games of the 1960s, and then, with all that passing knowledge, grind to a near-halt? This chapter is the first half of that story, told the way it happened: every offensive idea, the defense’s answer, and the rules that tilted the board.

Reading this part: action and reaction

Every chapter so far has met a scheme in its present form and given it a short history sidebar. This part of the book connects those sidebars into one story, and the story has a shape: an offense finds an advantage, the defense builds an answer, and when the defense gets too far ahead, the rule-makers step in. The single wing was football’s answer to a game with almost no passing. The T answered the single wing’s slowness. Platoons and Paul Brown made the T a specialist’s offense. The umbrella and the 4-3 answered Brown’s passes; the Flex answered Lombardi’s runs. The AFL (the American Football League, a rival league of the 1960s) went looking for points in the passing game, and by the 1970s the defenses had caught up with everyone. Hold on to that rhythm and the dates take care of themselves.

1906: a pass nobody wanted to throw

Football in 1905 was a shoving match. The ball was snapped, and the offense’s whole team crashed into the defense in a mass, with ball carriers pushed and pulled forward by their own teammates. That season, by the count usually cited, 19 players died nationwide, and President Theodore Roosevelt, a fan who wanted the game saved rather than banned, pressed the colleges to reform it. Among the changes the rules committee adopted on April 6, 1906, was one meant to spread the players out: the forward pass became legal.1

It was legal, but nobody wanted to throw it, and the reason was the rulebook. The early rules made a pass a gamble with lopsided odds. A pass that hit the ground untouched could cost the offense the ball (from 1907, a 15-yard penalty instead), it had to cross the line of scrimmage at least five yards to either side of the center, and a pass caught in the end zone was not a touchdown but a touchback. Some early versions also limited how far downfield a pass could be caught. Think about that as a coach: a run that fails costs you a down; a pass that fails can cost you the ball. Most coaches ran.

The rules eased slowly. In 1912 the colleges added the 10-yard end zone, a fourth down and the six-point touchdown, and dropped the limit on how far a pass could travel. A year later a small Catholic school from Indiana, Notre Dame, beat Army at West Point by throwing the ball around, with Gus Dorais passing to an end named Knute Rockne; the two had practiced it on a beach the summer before, while working as lifeguards. Newspapers loved it. Coaches mostly didn’t change.2

The professional league that became the NFL was founded in 1920 and took its rules from the colleges, and for its first dozen years it played the same cautious, running game. Two rule changes in 1933 and 1934 finally made the pass a normal play. The trigger was a controversial touchdown in the 1932 playoff game, played indoors on a shortened field because of a blizzard: Chicago’s Bronko Nagurski faked a run, backed up a step and threw a short touchdown pass, and Portsmouth argued that he hadn’t been the required five yards behind the line. For 1933 the NFL broke with the college rulebook and allowed a forward pass from anywhere behind the line of scrimmage. The same off-season it added hash marks (lines in the field from which every play starts, so the ball was no longer spotted beside the sideline) and moved the goalposts up to the goal line to encourage field goals. And from 1934, a second incomplete pass in the same set of downs no longer cost five yards.3 Meanwhile the ball itself got slimmer through the 1920s and ’30s, easier to grip and spiral.

You can see the effect in the scoring. In 1932 an NFL team scored 8.2 points a game; by 1948 it was 23.2, close to three times as many (the points chart later in this chapter shows the whole curve). The pass was not the only cause, since the T-formation and free substitution arrived in the same years (both are coming up), and the season-by-season passing numbers for the 1930s are too patchy to split the credit. But the direction is clear: by the peak years of the 1950s and ’60s, NFL teams threw on about half their plays.4 Offenses needed a player who could throw.

The single wing: power and deception from a direct snap

For the first decades of the forward pass, the formation that ran football was the single wing. It is usually credited to Glenn “Pop” Warner at the Carlisle Indian Industrial School around 1907, where Jim Thorpe played tailback for him; Warner called it the Carlisle formation, and the name single wing came later from the one “wing” back lined up outside the end.5

Read it from the back, the way Classic Formations taught you to read any formation. There’s no one under center. The tailback, the star, stands four or five yards deep and takes a direct snap, a long toss between the center’s legs. Beside him is the fullback, a power runner and faker. Up near the line is the blocking back, who was called the quarterback but almost never threw: his job was to lead the play and call the signals. And just outside the end is the wingback, a runner on reverses and a receiver on passes. The line is unbalanced: one lineman from the weak side moves over to the strong side, so four players stand to the right of the center and only two, a tackle and an end, to its left. In the picture it’s a guard (both guards, a tackle and the end on the right); many teams moved a tackle over instead, which made the strong side guard, tackle, tackle, end. Either way the point was the same.

Why it existed

The single wing was built for a game in which the run was everything and the pass was an occasional surprise, and it solved that game’s problems well.

More blockers than the defense had defenders at the point of attack. Look at the right side of the line in the picture: both guards, a tackle and the end, the wingback and the blocking back, plus the fullback, all on one side. The signature play, the off-tackle power, sent nearly all of them at one spot, and the formation was built so that the key defender there could be double-teamed.

A runner who could also throw. Because the tailback caught the snap standing up, five yards deep, every run started as a possible pass, and every pass as a possible run. The great tailbacks were triple-threat players: they could run, pass and punt, which in the low-scoring game of the era was as valuable as anything else they did.

Deception from the backfield. With the ball snapped to either deep back, the backs could spin, fake and hand off behind the line. The single wing’s spinner plays (the fullback takes the snap, turns his back to the defense and either keeps it or hands to a back circling behind him) and its reverses to the wingback were built to make the defense guess.

What it gave up

Everything in the single wing was slow. The snap traveled five yards through the air before anything could happen, the tailback started from a standstill, and the play took a long time to reach the line. The defense could see who had the ball; it couldn’t always see what he would do with it, but it knew where to look. And the formation demanded one extraordinary athlete: a tailback who ran, passed, punted and often played defense too. A team without one had no offense.

The defenses of the era matched it man for man. Six or seven linemen stood shoulder to shoulder, with one or two linebackers behind them and three defensive backs, as in the picture: a run defense with a little pass insurance.

1940: 73–0 and the T’s takeover

The T-formation was older than the single wing. What was new in 1940 was a version of it that worked. Classic Formations drew the T-formation and explained its logic: the quarterback’s hands under the center, an instant snap, backs who hit the line at full speed, and a ball handler who can fake to one back and give to another with his back to the defense. Here is how it took over.

The difference is all in the first second. The single wing’s tailback has to catch a five-yard toss before he can do anything; the T’s quarterback has the ball the instant the center moves it, and he can hand it to a back who is already moving toward the line.

The Chicago Bears’ coach and owner, George Halas, had stuck with a version of the T through the single-wing years, and his coach of 1930–32, Ralph Jones, is usually credited with adding a back in motion before the snap and spreading the ends and the backs to give it room. Clark Shaughnessy, a bookish, obsessive coach who worked with Halas as a consultant, refined it further. In 1940 Shaughnessy installed his version at Stanford, which had gone 1–7–1 the year before; Stanford went undefeated and won the Rose Bowl. In Chicago, Halas had found the quarterback the system needed: Sid Luckman, a single-wing tailback at Columbia, whom the Bears turned into a T quarterback who could take the snap, turn, fake, and drop back to throw.6

The rout settled the argument. Within a decade almost every college and pro team had switched to the T, and the Pittsburgh Steelers, the NFL’s last single-wing team, gave it up after the 1952 season.8

CautionCommon misconception

“Halas and Shaughnessy invented the T-formation.” They didn’t, and they never claimed to. A tight T with the quarterback under center was one of football’s oldest formations, and the Bears had used one all along. What Jones, Halas and Shaughnessy built was the modern T: man-in-motion, wider spacing, quick-hitting plays designed to make a defense commit, and a quarterback trained to fake and throw. The lesson repeats through this whole part of the book: formations are rarely invented; they’re re-equipped for a new problem.

Why the T won

The T’s advantages were speed and deception, and both came from the snap. The single wing’s tailback needed a second just to receive the ball; the T’s quarterback had it instantly, and his backs could be at the line in the time it took a single-wing snap to arrive. But the deepest reason was the quarterback. In the single wing, the man who threw also had to run, block and often punt. In the T, one player did nothing but handle the ball and throw it, from a few yards behind his line, facing his receivers after a short drop. That made him a specialist, and specialists improve faster than generalists. Every modern quarterback descends from Luckman more than from any single-wing tailback.

What the T gave up, and what football did about it

The T as the Bears ran it had two tight ends and three backs: lots of runners and almost no width. Over the next decade coaches pulled its players outward into the pro set of two backs, a tight end and two wide receivers (Classic Formations has the family tree, including Shaughnessy’s 1949 Rams flanking Elroy Hirsch wide9). The other thing the T needed was a different kind of roster, and that required a change in the rules.

Platoon football: the specialist arrives

Until the 1940s the rules made substitution hard: a player who left the game generally couldn’t come back in the same quarter, so the same eleven men played offense and defense. The star tailback was also a defensive back. The quarterback tackled. Linemen had to be big enough to block and quick enough to chase.

Free substitution changed all of that. The colleges allowed it in 1941, and the NFL in 1943, both partly because the war had thinned rosters. In 1945 Michigan’s Fritz Crisler, facing a much stronger Army team, used it to send in separate units for offense and defense, and the idea spread. The NFL made unlimited substitution permanent in 1950. The colleges went the other way for a while: in 1953 they went back to limited substitution, mostly over cost, because fielding two full units had pushed dozens of small schools to drop football altogether, and they didn’t restore free substitution until 1965. That gap is part of why the pros pulled ahead in specialization.10

The two-way player didn’t vanish overnight. In 1960 injuries pushed the Philadelphia Eagles to use Chuck Bednarik, a 12-year veteran, on both sides of the line again, at center and linebacker. In that season’s championship game he played 58 of the 60 minutes and made the game-saving tackle, wrestling Green Bay’s Jim Taylor to the ground as time ran out. He is remembered as one of the last of the NFL’s two-way stars.11

That is platoon football: separate groups of players for offense and defense, and eventually for kicking plays and particular situations.

Why it mattered

Platooning sounds like a roster rule. It was really a revolution in what a player could be.

  • Bodies could specialize. An offensive lineman no longer needed to chase a halfback; he could get bigger and stronger and learn to pass block. A defensive back no longer needed to run the ball; he could be the fastest man on the field and learn to cover.
  • Skills could specialize. A T quarterback could spend his whole practice throwing. A defensive end could spend his rushing. The pass-catching end became a receiver who ran routes for a living.
  • Coaching could specialize. Two units meant two sets of schemes, and soon two sets of coaches: the ancestors of today’s offensive and defensive coordinators.

Everything after this point in the chapter is an argument between specialists.

NoteMadden vs. real life

In Madden you never think about who plays defense when your offense is on the field: there are 53 players and the game substitutes for you. For football’s first half-century that roster didn’t exist. NFL roster limits ran from about 20 players in the early 1930s to the low and mid-30s in the ’40s, and the best players stayed on the field for both halves of every possession.12 The whole modern idea of personnel groupings (Part 2) depends on free substitution. Before 1943 there were no groupings to choose between: to borrow the phrase in a different sense, a team’s “11 personnel” was simply its eleven best players, and they were the defense too.

Paul Brown: football becomes a profession

The coach who did the most with those specialists was Paul Brown. He had built winners at Massillon (Ohio) High School and at Ohio State, where his 1942 team won the national championship, and in 1946 he became head coach of a new professional team named after him, the Cleveland Browns of the All-America Football Conference (AAFC), a rival league. The Browns won all four AAFC championships (1946–49). When the AAFC folded and the Browns joined the NFL in 1950, the established league assumed they would be humbled. In their first NFL game, against the two-time defending champion Philadelphia Eagles, the Browns won 35–10. They won the NFL title that season and again in 1954 and 1955, and they played in the championship game in each of their first ten seasons (AAFC and NFL) with Otto Graham at quarterback.13

Brown’s innovations matter more than his trophies, because almost all of them are still standard:

  • A full-time, year-round coaching staff, at a time when many assistants were part-timers.
  • The playbook as a textbook. Players got detailed books of formations and plays, attended classroom sessions, and took written tests on them. Brown also gave them intelligence tests.
  • Film study. Brown is credited with being the first coach to use game film to scout opponents, breaking down what the other team did before playing it.
  • Measurable athletes. He timed players over 40 yards, a distance he adopted at Ohio State in 1941, by the usual account because it was about how far a player ran on a punt cover (his son Mike remembers it as simply the longest sprint a player usually had to make). The 40-yard dash is still the most famous number at the NFL Combine.
  • The draw play, which Brown dated to an accident in 1946: Otto Graham, about to be sacked, handed the ball to his fullback, Marion Motley, who ran through a defense that had dropped back for a pass. (A draw play is a run disguised as a pass: the linemen show pass protection, then the ball goes to a back.)
  • Integration. In 1946 Brown signed Motley and guard Bill Willis, two of the first Black players in modern pro football.

The coach calls the plays

The innovation that changed the game most directly was who called the plays. Before Brown, the quarterback called them in the huddle. Brown called them from the sideline, and to get his calls in, he rotated two guards on every play: each messenger guard came in with the call and the other went out. Guards were the natural men to shuttle. Two guards of similar size and skill could swap every play without changing what the offense could do, whereas swapping a back or a receiver would have changed the personnel on the field, and the quarterback had to stay in to run the play. One of them, from 1953 to 1959, was a smart, undersized lineman named Chuck Noll, who, Brown said, soon could have called the plays himself; Noll later coached the 1970s Steelers. In 1956 Brown went further and tested a radio receiver in his quarterback’s helmet. The Detroit Lions spotted the transmitter during a preseason game, and Commissioner Bert Bell banned the device after a handful of games. The NFL legalized helmet radios in 1994.14

A coach choosing every play from the sideline is so normal now (see What a Play Really Is) that it’s easy to miss how radical it was. It moved the brain of the offense from the field to the staff, and it made the game plan, built from film during the week, the center of football. Every coordinator with a laminated call sheet works in a system Paul Brown designed.

NoteFilm room: Cleveland Browns vs. Philadelphia Eagles, 1950

The Browns’ first NFL game, on September 16, 1950, was the AAFC champion against the NFL champion, and Otto Graham passed for 346 yards and three touchdowns in a 35–10 win. The Eagles’ coach, Greasy Neale, whose five-man Eagle defense had won the previous two NFL titles, grumbled that the Browns were like a basketball team, all they did was throw. When the teams met again on December 3, in rain in Cleveland, Brown answered him: the Browns won 13–7 without a single official pass attempt. (The two passes Graham threw were both called back by penalties.) No NFL team since has played a whole game without attempting a pass.15

What to notice: the point Brown was making. A team whose opponent can only stop one thing doesn’t have to do only one thing. The pass had become a weapon because it was no longer the only weapon.

The defense answers: Steve Owen’s umbrella

First, what had the defense done about the T? Through the 1940s, most teams’ answer was more men on the line and more men deep. The most successful version was Greasy Neale’s Eagle defense in Philadelphia, a 5-2-4 from 1947: five linemen, with a middle guard head-up on the center, two linebackers behind them, and four defensive backs instead of the three of the single-wing era, the extra one for the T’s passes. The Eagles won the 1948 and 1949 titles with it. Odd and Hybrid Fronts follows the five-man line from there to the 3-4. Its soft spot was the middle: with no linebacker over the center, a quick pass over the middle guard’s head had room.16 Keep that in mind; it comes back in a few pages.

Brown’s passing game created a problem defenses of 1950 had never had to solve. Most NFL teams still lined up with five or six linemen and only three or four defensive backs, and the Browns sent out receivers like Dante Lavelli and Mac Speedie, plus a flanker, plus backs who ran routes. There were more receivers than defenders who could cover them.

The first team to solve it was the New York Giants. Their head coach, Steve Owen, a heavyset Oklahoman who had been coaching the Giants since 1930 and had won the NFL title in 1934 and 1938, built a defense for the 1950 Browns that looked like a run defense and played like a pass defense. Owen called it a 6-1-4, and in his own words, on many plays it became “a 4-1-6 in reality.” The secret was the two defensive ends: they lined up on the line of scrimmage like linemen, and at the snap they could drop back into pass coverage. The four defensive backs behind them, spread in an arc like the ribs of an umbrella, gave the defense its name: the umbrella defense. One of those defensive backs, a young Texan named Tom Landry, explained it to the others and taught it on the field.17

From the umbrella to the 4-3

The 4-3 came from two different parents, each of which gave up a lineman.

The first parent is the umbrella. Its dropping ends were really linebackers who started on the line. Stand them up for good and the 6-1-4 becomes four linemen, three linebackers (the two ends and the one linebacker) and four defensive backs.

The second parent is the five-man front of the Eagle family. Its weak spot, you’ll remember, was the middle. In 1954 the Bears’ middle guard, Bill George, kept getting beaten by quick passes thrown over his head, so he started standing up and backing off the line at the snap. Do that in a 5-2-4 and it becomes four linemen, three linebackers (the two and George) and four defensive backs. George is usually named as the first true middle linebacker, though the Hall of Fame itself says no one can be sure which middle guard first dropped back.18

Landry took both moves at once: the umbrella’s idea of ends who could play off the line became his outside linebackers, and George’s stand-up middle guard became his middle linebacker. Four linemen, three linebackers and four defensive backs: the 4-3. Landry, by then the Giants’ defensive coach, made it the Giants’ base defense in 1956, built around a rookie middle linebacker from West Virginia, Sam Huff. The Giants won the 1956 championship, much of the league copied the defense the following season, and Huff became so famous that CBS made a television documentary about him, The Violent World of Sam Huff (1960, an episode of Walter Cronkite’s The Twentieth Century). Why did the 4-3 win? Because it was the first defense balanced for both threats. Four linemen are enough to rush a passer; three linebackers can stop a run or drop into coverage; four defensive backs can cover the receivers that platoon football had produced. It was the defense equivalent of the pro set: good at everything, great at nothing in particular.

His Packers won five NFL championships in seven seasons, 1961–67, including the first two Super Bowls, and the play everyone associated with them was the power sweep. The sweep’s genius was that it didn’t need to fool anyone. Lombardi drilled it until every block was automatic, and the runner was given a lane, not a hole. A hole is a fixed spot the play is designed to hit (“between the guard and the tackle”), and the back runs there whatever happens. A lane is wherever the defense isn’t once the blocks are made: the blockers take defenders the way they want to go, and the back finds the opening the blocks create. Lombardi’s own phrase for it, and the title of his 1963 book, was run to daylight. The defense knew the sweep was coming, and the Packers ran it anyway.

Underneath the sweep was a blocking idea that defenses found maddening, which coaches of the era called option blocking (not to be confused with the option offense, where the quarterback reads a defender). Here the blocker has the choice. Say the tight end’s job is the Sam linebacker. If the Sam steps outside, the tight end doesn’t fight to move him back in; he rides him farther outside, and the runner cuts inside. If the Sam steps inside, the tight end seals him there, and the runner goes around. Block the man where he goes, not to a spot, and let the runner read the result. The pulling guards read the same block: the right guard logs the cornerback inside or kicks him out, depending on which side of him he can reach, and the left guard turns up through whichever seam the tight end’s block has opened. A defense built on linemen charging hard into the backfield, each guessing his own way, helped option blocking. The harder a lineman charged, the easier he was to push past the hole, and the daylight appeared behind him.

In the Flex, two of the four defensive linemen line up about a yard off the ball while the other two stay on the line. The flexed linemen are in no hurry to charge: they read the blocks in front of them, then step into the gap they’re assigned. The linemen on the line attack their gaps. Linebackers fill the rest. The result is a defense in which every gap has an owner who is already standing in it before the runner arrives, which is exactly what run-to-daylight blocking can’t handle: there is no daylight to run to.

Why does a single yard matter so much? Think about it from the blocker’s side. Option blocking works on a defender who is already moving: you take his momentum and add to it. A flexed lineman isn’t moving. The yard of space means the blocker has to travel to reach him, which takes a moment, and in that moment the defender sees where the block is coming from. And because he hasn’t committed to a direction, there is no momentum for the blocker to use. A blocker trying to push a flexed defender “the way he wants to go” finds that he doesn’t want to go anywhere until he knows where the ball is.

But the two linemen on the ball do charge, so why weren’t they option-blocked? Because their charge is into an assigned gap, not a free-lance guess at where the play is going. A tackle who knows he owns the A gap gets there and holds it; if he is pushed past it, the flexed man behind him is reading that very block and steps into the space he left. The on-ball men, in other words, are there to occupy blockers and keep their own gaps, and the flexed men clean up behind them. Landry built it partly around the quickness of his great defensive tackle, Bob Lilly, who could play on the ball and get into his gap before a blocker touched him.

Why flex only two, and not all four? Because somebody still has to take on the blockers at the snap and threaten the quarterback. A line with all four men standing a yard off the ball reading would give the offense a free first step everywhere and no pass rush at all. Which two flexed changed with the offense’s formation and with where Dallas expected the run.

The Flex was Landry’s style in one picture: precise, rule-bound, analytical. Players learned exactly which step to take with which foot, which some of them hated. Its price was the pass rush from the Flex itself: linemen standing back reading blocks aren’t rushing the quarterback. So Dallas used it as a run-down defense, almost always on first down but mostly only in running situations on second and third, and on clear passing downs the linemen lined up on the ball and rushed. They could rush, too: the Cowboys’ defense of the 1970s, nicknamed the Doomsday Defense, had Harvey Martin, credited unofficially with 23 sacks in 1977.21 Dallas reached two straight NFL Championship Games against Green Bay, and later won Super Bowls after the 1971 and 1977 seasons with the Flex behind them. You have met its descendant: the idea that every defender owns a gap is the core of gap integrity and run fits.

Sid Gillman and the vertical passing game

While Lombardi and Landry argued about the run, Sid Gillman was building the passing game you watch today. As a young man in Minneapolis, Gillman worked in his family’s movie theater and cut the football clips out of the newsreels so he could study them at home on a projector. He became the coach most associated with film study, and he used it to take passing apart. He coached the Los Angeles Rams (1955–59) and then the Chargers of the new AFL (1960–69), who won the 1963 AFL championship 51–10 over Boston.23

Gillman’s idea, in one sentence: use all of the field, deep and wide, so that the defense can’t protect all of it. Defenses of the 1950s and early ’60s were happy to let an offense throw short in front of them, and for a good reason: a short completion was a few yards and a tackle. To score that way an offense had to string together a long series of plays without an incompletion on a key down, a penalty or an interception (and interceptions were far more common then than now), so the defense’s first job was to stop the long ball and wait for the drive to die. Gillman threw deep first, with fast receivers like Lance Alworth, to push the safeties back, and then attacked the room that left underneath: intermediate routes, and backs released into the flats. As Al Davis, who coached under him, put it, people thought of the passing game as vertical, the length of the field, “but Sid also thought of it as horizontal.”

The mechanism under the slogan was timing. Gillman tied each route to the quarterback’s drop: on a five-step drop, the first receiver’s route was built to break just as the quarterback finished his last step, so the ball could leave before the receiver turned around, and the next receiver’s route came open a beat later for the next read.24 A defense can’t cover a throw that arrives before it has seen the receiver’s break. That idea, routes and drops on one clock, is what Don Coryell and Bill Walsh inherited.

He was building it, mostly, against man-to-man coverage. Much of the era’s pass defense, and especially the AFL’s, was man-to-man with a free safety behind it; the AFL is where the bump-and-run, a cornerback pressing his receiver and jamming him at the line, took hold in the 1960s, with Oakland’s Willie Brown its best-known practitioner. Beating man coverage is about getting one receiver open against one defender, and the whole-field spacing gave Gillman’s receivers room to do it. Remember that when you reach the 1970s: the defenses’ answer to the passing decade was to stop playing man and play zones as a team.

The AFL: a passing laboratory

In August 1959 a young Texan named Lamar Hunt, who had been refused an NFL expansion team, founded a league of his own. The AFL (American Football League) began play in 1960 with eight teams, and it needed fans, so it needed points.

The AFL was looser in almost every way. It added the two-point conversion (the NFL would not adopt it until 1994), put players’ names on their jerseys (a habit the NFL later copied), and shared its television money equally, including, from 1965, a five-year NBC contract worth $36 million that kept the league alive. Because it was scrambling for talent, it took players the NFL had overlooked: quarterbacks cut by NFL teams, small-college stars, and many players from historically Black colleges that NFL scouts had largely ignored.26

Why call it a passing laboratory? Because its best offenses built their identity on throwing deep to wide receivers. Gillman’s Chargers, the plate above, were the model: Alworth deep, the tight end and the backs underneath, every throw timed to the drop. Al Davis took the same idea to Oakland in 1963 and made the deep ball the Raiders’ trademark for decades: stretch the field vertically and use every eligible receiver.27 Quarterbacks like John Hadl, Len Dawson and Daryle Lamonica threw deep and often. The points chart below shows the result: in each of its first five seasons AFL teams outscored NFL teams, by close to three points a game in 1960 and 1961, and by about one point a game in 1962–64. And the AFL’s defenses answered in kind, with the bump-and-run corners you met above. The arms race of the next decade was rehearsed here.

Hank Stram’s Chiefs

The most inventive AFL coach after Gillman was Hank Stram, who led the Dallas Texans and then the Kansas City Chiefs from 1960 to 1974 and won three AFL titles (1962, 1966 and 1969). Stram’s contributions to the passing game and to everything around it:

  • The moving pocket. The protection shifted with a quarterback who set up somewhere other than straight behind the center, so the rushers had a moving target and the quarterback got a clear throwing lane. Part 4’s bootlegs and sprint-outs are its heirs.
  • New formations. He is credited with being the first pro coach to use both the I-formation and the two-tight-end offense.
  • The triple stack on defense. Linebackers hid directly behind defensive linemen, so the offense couldn’t tell which gap each would attack.
  • Talent from where others weren’t looking. He drafted heavily from historically Black colleges: defensive tackle Buck Buchanan (Grambling) and linebacker Willie Lanier (Morgan State) are both in the Hall of Fame.

In Super Bowl IV Stram wore a microphone for NFL Films, usually described as a first for a pro coach, and it caught him gleefully telling his team to keep “matriculating the ball down the field,” and the Chiefs beat Minnesota 23–7.28

The merger

The two leagues spent the early 1960s bidding against each other for players, and when the Jets signed Alabama quarterback Joe Namath on January 2, 1965, for a reported $427,000, the NFL owners decided the war was too expensive. The leagues announced a merger on June 8, 1966: a common draft, a championship game between them (the first, in January 1967, became known as Super Bowl I), and a single league from 1970. The NFL’s champions won the first two Super Bowls easily. Then the AFL won the next two: Namath’s Jets beat the Baltimore Colts 16–7 in Super Bowl III, and Stram’s Chiefs beat the Vikings in Super Bowl IV. When the merger took effect in 1970, the AFL’s ten teams (eight originals plus two later additions, Miami in 1966 and Cincinnati in 1968) and three NFL teams (the Colts, Browns and Steelers) formed the American Football Conference.

The AFL’s lasting gift was proof. Wide-open passing offenses could win championships, not just entertain, and the merged league inherited the coaches, quarterbacks and ideas to play that way. That makes what happened next all the stranger.

Meanwhile in the NFL: the 1960 shotgun

The NFL had its own experiment. On November 27, 1960, San Francisco’s coach, Red Hickey, lined his quarterback up about seven yards behind the center, a long direct snap away, with the backs and receivers spread around him. It was “really just a new version of an old formation,” the single wing’s direct snap with a passer instead of a runner, and Hickey called it the shotgun. As a heavy underdog the 49ers upset the Colts 30–22. In 1961 Hickey rotated three quarterbacks (John Brodie, Billy Kilmer and Bobby Waters) through it and started 4–1, including 49–0 over Detroit and 35–0 over the Rams. On October 22, 1961, the Bears found an answer, beat San Francisco 31–0, and Hickey shelved the formation.29 (What the Bears did is the second Predict-the-play drill below; try it before you look it up.) The shotgun returned in Dallas in 1975 as a passing-down weapon, and Formation Rules tells the rest; today’s NFL offenses spend most of their snaps in it.

The passing numbers tell the same story from the other side. In 1977 NFL teams threw just 25 passes a game, against 29.4 in 1954; only 42 percent of their plays were passes, and they ran 37.4 times a game, a modern-era record. A pass gained 6.5 yards per attempt, down from about 7.4 in 1956–65, and from 1970 through 1977 more than five of every hundred passes were intercepted.31 Throwing had become both less rewarding and riskier, so teams did less of it.

Why the defense got ahead

No single cause, but five that reinforced each other.

1. Receivers could be hit all the way down the field. Until 1974 a defender could chuck a receiver (hit or shove him with his hands) anywhere on the field, as often as he liked, until the ball was thrown. A receiver who has to fight through a linebacker at 5 yards, a safety at 12 and a cornerback all the way can’t run his route on time, and a timing passing game depends on routes run on time.

2. Pass blockers couldn’t use their hands. Offensive linemen of the era had to keep their hands close to their chests while pass blocking, and couldn’t extend their arms to push a rusher away, while defensive linemen could use their hands freely, including the head slap, a swat to the side of a blocker’s helmet (legal until 1977). Pass protection was a fight with one side’s hands tied.

3. The great four-man lines. The 4-3 had matured, and the decade’s best teams were built on their defensive lines: the Steelers’ “Steel Curtain,” the Vikings’ “Purple People Eaters,” Dallas’s Doomsday, and early in the decade the Rams’ “Fearsome Foursome” (Deacon Jones, its most famous member, was traded after the 1971 season). Four rushers who could win without help let seven defenders cover.32

4. Zone coverage grew up. Defenses had learned to play zones as a team. The best-known version was the two-deep shell Bud Carson had begun developing as Georgia Tech’s head coach and brought to Pittsburgh in 1972 (Cover 2): corners jamming receivers at the line, safeties over the top, and, in Pittsburgh, a middle linebacker (Jack Lambert, from 1974) who ran deep down the middle, the ancestor of Tampa 2. The 1976 Steelers allowed 28 points in their last nine games, with five shutouts.33 Zones that are combined with unlimited chucking are close to airtight.

5. Offenses played it safe. When passing is that hard, the sensible strategy is to run, punt and win the field-position battle, and many coaches did. The game turned into a contest of whose defense would blink first.

Why then? What changed around 1970

Notice something odd about that list. Causes 1 and 2 are old rules. Defenders had been allowed to chuck receivers downfield, and blockers had been forbidden to extend their arms, all through the 1950s and ’60s, when the NFL scored 21 or 22 points a game and the AFL more. Rules that didn’t change can’t by themselves explain a fall. What they can explain is why the fall, once it started, was so hard to reverse. So what was new?

  • Zones met the chuck rule. Against man coverage, a receiver had one defender to beat, and the man hitting him was the man covering him. In a team zone, the receiver runs through several defenders’ areas, and under the old rules each of them could hit him on the way. The rule was the same; its value to the defense multiplied as defenses moved from man toward zone. That shift is the late-1960s and early-‘70s story of cause 4 (Carson’s Cover 2 reached the NFL in 1972), and it was the defenses’ answer to the Gillman-style passing that had beaten man coverage in the 1960s.
  • The defensive talent caught up. The AFL’s early scoring edge probably owed something to thin defenses in a new league. By 1969 its scoring had fallen back to the NFL’s level (the orange line on the chart), and after the common draft (from 1967) and the merger, every team drew from the same talent pool. The great four-man lines of cause 3 were built in exactly those years.
  • Coaches took the hint. Once the first two changes made passing riskier, the safe, run-first strategy of cause 5 won games (the 1972 Dolphins went 17–0 with it), and winning strategies get copied.

The constants (the chuck rule and the blockers’ hands) were the fuel; the new zones and the new defensive talent were the spark. That’s why the cure, when it came, had to change the constants.

New defenses kept coming, too. In 1972 Miami’s Bill Arnsparger moved a linebacker around his front in the package that became the 53 defense, and in 1974 the New England Patriots and Houston Oilers began using the 3-4 as a base defense, a story Odd and Hybrid Fronts tells in full.

The league saw the problem early and tried to fix it, three times.

A ball spotted on a hash used to leave the offense a cramped short side and a vast wide side; now the field was almost symmetrical.

1974: the package. With a rival league (the short-lived World Football League) advertising a more exciting game, the NFL’s competition committee passed its biggest set of changes in decades, aimed, in its own words, at making long touchdown drives more achievable:36

  • The goalposts moved from the goal line to the back of the end zone, ten yards farther away, so a drive that stalled at the 20 was no longer an easy field goal. That sounds like a way to cut scoring, and at first it was. The idea was that fewer cheap field-goal tries would mean more fourth-down attempts and more drives played out for touchdowns (the same logic, half a century early, as the go-for-it argument in The Clock, the Two-Minute Drill, and Fourth Down), and more room in the end zone for passes now that the posts weren’t standing in it.
  • Kickoffs moved back from the 40 to the 35, so more of them could be returned.
  • A missed field goal went back to the line of scrimmage (or to the 20 if the kick came from inside it), so a long attempt was no longer free.
  • Offensive holding dropped from 15 yards to 10, and receivers lined up wide could no longer be cut at the knees.
  • The first chuck rule: once a receiver was 3 yards past the line of scrimmage, a defender could contact him only once.

That last item introduces this chapter’s last term. A chuck rule is any rule limiting how a defender may hit or shove a receiver running his route. The 1974 version was sometimes called the “Isaac Curtis rule,” after the fast, slender Cincinnati receiver whom Miami’s defense had knocked down as often as it could in a 1973 playoff game; the Bengals’ owner, Paul Brown, sat on the rules committee and lobbied for the change. In 1977 the league tightened it so that a defender could contact a receiver only once in all, and banned the head slap.37

Look at the chart again and you’ll see that 1974, the package’s first season, is itself a dip, from 19.4 to 18.2. The goalposts explain it. Field-goal attempts fell from more than 860 in 1973 to 553 in 1974.38 If about three in five of those lost kicks would have gone in, that is something like 180 field goals, or roughly 1.5 points per team per game, more than the whole 1974 drop. The touchdowns the committee hoped would replace them came a year later: scoring rose to 20.6 in 1975. Then it fell again, to 19.2 in 1976 and 17.2 in 1977.

Why the fixes failed

Look at what the 1972 and 1974 changes touched: where the ball was spotted, where the goalposts stood, how kicks were returned, how far a penalty walked off. Each was sensible. None changed the basic arithmetic of a pass play. Under the 1974 rule a defender could still jam a receiver as he left the line and hit him again once he crossed 3 yards; an offensive lineman still couldn’t extend his arms to protect his quarterback. As long as the receiver couldn’t get open on time and the quarterback couldn’t stand in the pocket long enough to wait for him, the passing game that the AFL had proved could win stayed too risky to build an offense around. The rules had nibbled at the edges of the defense’s advantage without touching its center.

The 1977 rule looks, at first, like the real fix: one contact and no more, plus no head slap. Why did scoring hit bottom that very season? Because where the contact came was still up to the defender. He could spend it at the line, or he could save it for the moment that hurt most: 15 yards downfield, just as the receiver planted to make his break. One well-timed hit there wrecks a timed route as surely as five scattered ones. And the other half of the problem was untouched: banning the head slap took one weapon from the rushers, but the blockers still couldn’t extend their arms.

The changes that finally worked, in 1978, went straight at both halves. Contact with receivers was allowed only within 5 yards of the line (the Mel Blount rule), so past that point a receiver could run his route untouched and arrive on time; and pass blockers were allowed to extend their arms and open their hands. That’s the next chapter, and the scoring chart turns upward from there.

ImportantWhy it exists: the rules are the third player

This chapter’s history has three actors, not two. Offenses innovate, defenses answer, and the rules decide which side the board tilts toward. The 1906 pass rules held the forward pass back for a generation; the 1933 and 1934 changes released it; free substitution made the T a specialist’s offense; and the drought of the 1970s ended only when the league changed the rules of contact. Whenever you read about a scheme “taking over” the NFL, ask what rule made it possible. You’ll ask it again in Contact Rules and Penalties as Strategy.

What survives

Not one of this chapter’s formations is a base offense in today’s NFL, and almost all of their ideas are still on the field.

  • The single wing’s direct snap became the shotgun, now the formation of most NFL snaps, and lives on in its purest form in the wildcat and in every direct snap to a running back on the goal line. Its spinners and reverses are the ancestors of today’s jet sweeps and misdirection.
  • The T’s under-center quarterback never left: it is the base of play-action, and under-center snaps rose to about a third of NFL plays in 2025.
  • Paul Brown’s organization (staffs, playbooks, film, sideline play-calling) is simply how football is run.
  • The umbrella’s idea, showing one structure and playing another after the snap, is the heart of modern disguise and simulated pressure.
  • The 4-3 remains one of the two families of base front, and the Flex’s gap ownership is how every defense now teaches run fits.
  • Gillman’s deep-and-wide passing is the grammar of every passing concept in Part 4.
  • The AFL’s two-point conversion arrived in the NFL in 1994, and its spirit, that you win by scoring, won the argument in 1978.

What is the Bears’ plan, and why does it work against this formation in particular?

TipAnswer

Rush the center. In the shotgun the center has the hardest job on the field: he must look between his legs, throw a seven-yard snap accurately, and only then lift his head to block. A defender lined up directly over him gets a head start every time. With George on the center, either the center loses the one-on-one, or a guard has to help, and then someone else is free.

Why it works here: the 1961 shotgun had almost no running threat to punish an aggressive rush. A quarterback seven yards deep with a single back beside him can’t hand off quickly or threaten a quick dive, so the defense could commit to the pass rush. With pressure up the middle arriving before receivers spread across a wide field could get open, the 49ers had nowhere to go: the Bears won 31–0, and Hickey retired the formation. The modern shotgun survives because offenses solved both problems: quarterbacks set up about five yards deep, not seven, and there is a real run game (and the RPO) from it, so a defense that sends everyone at the center pays for it.

What are the defense’s advantages under the 1977 rules, and what is the offense’s most sensible call?

TipAnswer

The defense’s advantages are the rulebook’s. Those pressed corners can jam the receivers at the line, and under the 1977 rule they get one legal contact on each, used right where it hurts the timing most. Seven defenders are in coverage behind them in a two-deep zone, the safeties over the top. Up front, four rushers can use their hands freely (the head slap is gone in 1977, but blockers still can’t extend their arms), so the quarterback doesn’t have long.

The offense’s sensible call is something short and safe: a quick pass to a back or the tight end underneath the zone, or a draw that punishes the four rushers for charging, and then a punt if it comes up short.

If you’ve read Cover 2 and Tampa 2, your first instinct may have been the classic Cover 2 beaters: smash (a corner route over the squatting cornerback) or the hole shot down the sideline between the corner and the safety. In 1977 both die before they start. The corner route needs the receiver to get off the line clean and break on time, and here he is jammed at the snap by a corner who is spending his one legal contact right where it hurts. The hole shot needs the quarterback to hold the ball while a deep route develops, behind blockers who can’t extend their arms. A 15-yard throw against this defense in 1977 asks a jammed receiver to get open in time behind a protection that can’t hold up. That’s the drought in one snap: on third-and-8, the most rational play was to try for a first down without risking an interception, and punt. A year later, with contact limited to 5 yards and blockers allowed to extend their arms, the same third-and-8 became a throwing down again.

Takeaways

  • Football’s history is action, reaction and rules. Offenses innovate, defenses answer, and the rule-makers tilt the board. The forward pass, legal from 1906, didn’t matter until the rules stopped punishing it (1933–34).
  • The single wing was built for power and deception from a direct snap; the T beat it with speed and a specialist passer. The 1940 Bears’ 73–0 made the switch, and platoon football (permanent in the NFL from 1950) made the specialist possible.
  • Paul Brown turned football into a profession: full-time staffs, playbooks, film, measurable athletes, and plays called from the sideline. Every modern staff runs on his model.
  • The 4-3 was the defense’s answer to the pro passing game, and it had two parents. Owen’s umbrella showed a 6-1-4 and played a 4-1-6 (its ends became outside linebackers); Bill George stood up the five-man front’s middle guard (the middle linebacker); Landry combined them in 1956, then built the Flex in the 1960s to take away Lombardi’s daylight by giving every gap an owner.
  • Gillman and the AFL proved the passing game could win, using the whole field, deep and wide, and the merged league inherited their ideas.
  • The 1970s drought came from old rules meeting new defenses. Defenders could hit receivers downfield and pass blockers couldn’t extend their arms, as they always had; what was new was team zone coverage (which let several defenders each take a shot at a receiver) and defensive talent spread across one merged league. The 1972 hash marks, the 1974 package and even 1977’s one-contact rule left the arithmetic of a pass play alone, so scoring fell to 17.2 in 1977. Only 1978’s five-yard limit and open-handed pass blocking changed it.
