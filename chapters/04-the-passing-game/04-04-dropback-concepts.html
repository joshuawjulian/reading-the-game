def O(key, pos, d, w, label=None):
    return Player(key, pos, "off", d, w, label=key if label is None else label)


def D(key, pos, d, w, label):

def line5():
    """The five offensive linemen, 1.6 yards apart (only the center is lettered)."""
    return [O(k, k, LINE_D, w, "C" if k == "C" else "") for k, w in
            (("LT", -3.2), ("LG", -1.6), ("C", 0.0), ("RG", 1.6), ("RT", 3.2))]


def spread(rb_w=-1.5, x_w=-15.0, h_w=-9.0, y_w=9.0, z_w=15.0):
    """11 personnel, shotgun 2x2. X and Z on the line (with the five linemen: seven on the line);
    H in the left slot and Y, a tight end split out, in the right slot, both a step off the ball."""
    return line5() + [O("QB", "QB", GUN, 0.0, "Q"), O("RB", "RB", GUN, rb_w, "R"),
                      O("X", "WR", LINE_D, x_w), O("H", "WR", OFF_BALL, h_w),
                      O("Y", "TE", OFF_BALL, y_w), O("Z", "WR", LINE_D, z_w)]


def dline(s=1):
    return [D("WDE", "DE", 1.0, -5.4 * s, "E"), D("WDT", "DT", 1.0, -0.8 * s, "T"),
            D("SDT", "DT", 1.0, 2.4 * s, "T"), D("SDE", "DE", 1.0, 5.4 * s, "E")]


def _apply(players, extra):
    extra = extra or {}
    return [q.moved(d=extra[q.key][0], w=extra[q.key][1]) if q.key in extra else q for q in players]


def cover3(nb_side=1, extra=None):
    """4-2-5 nickel, Cover 3 against 2x2: corners off with outside leverage, one deep safety, and a
    curl-flat defender over each slot: the nickel ($) on side nb_side (+1 = offense's right), the
    strong safety on the other."""
    s = nb_side
    return _apply(dline() + [D("MIKE", "MIKE", 4.5, 1.8, "M"), D("WILL", "WILL", 4.5, -2.2, "W"),
                             D("LCB", "CB", 7.0, -16.0, "CB"), D("RCB", "CB", 7.0, 16.0, "CB"),
                             D("NB", "NB", 5.0, s * 7.2, "$"), D("SS", "SS", 5.5, -s * 7.2, "SS"),
                             D("FS", "FS", 13.0, 0.0, "FS")], extra)


def cover2(extra=None):
    """4-2-5 nickel, Cover 2: corners squatting at 5 yards with outside leverage, two deep safeties."""
    return _apply(dline() + [D("MIKE", "MIKE", 4.5, 1.6, "M"), D("WILL", "WILL", 4.5, -4.2, "W"),
                             D("NB", "NB", 4.5, 6.6, "$"),
                             D("LCB", "CB", 5.0, -16.3, "CB"), D("RCB", "CB", 5.0, 16.3, "CB"),
                             D("FS", "FS", 12.5, -8.5, "FS"), D("SS", "SS", 12.5, 8.5, "SS")], extra)


def quarters(extra=None):
    """4-2-5 nickel, Quarters (Cover 4): corners off, two safeties at about 10 yards reading the slots."""
    return _apply(dline() + [D("MIKE", "MIKE", 4.5, 1.6, "M"), D("WILL", "WILL", 4.5, -4.2, "W"),

def cover1(extra=None):

    on the back, the Mike free to rush or sit in the hole, one deep safety."""
    return _apply(dline() + [D("MIKE", "MIKE", 4.2, -1.8, "M"), D("WILL", "WILL", 4.2, 1.4, "W"),

def protect(p, rushers=("WDE", "WDT", "SDT", "SDE"), depth=1.2, rush=1.6, linemen=("LT", "LG", "C", "RG", "RT")):
    """Linemen set, four rushers engage them: enough to say 'pass', no more."""
    for k in linemen:
        p.block(k, pts=[(-depth, 0.0)])
    for k in rushers:
        if k in p.players:
            p.rush(k, pts=[(-rush, 0.0)], speed=2.0)


def go_to(p, key, end, kind="path", delay=0.0, speed=None, via=()):
    """Move `key` through absolute (d, w) points `via` to `end`."""
    start = np.array(p._cur(key))
    pts = [tuple(np.array(q) - start) for q in (*via, end)]
    if kind == "rush":
        return p.rush(key, pts=pts, delay=delay, speed=speed)
    if kind == "route":
        return p.route(key, pts, frame="field", delay=delay, speed=speed)
    if kind == "block":
        return p.block(key, pts=pts, delay=delay, speed=speed)
    return {"path": p.path, "run": p.run}[kind](key, pts, delay=delay, speed=speed)


def route_to(p, key, *pts, delay=0.0, speed=None):
    """A route through absolute (d, w) points."""
    return go_to(p, key, pts[-1], kind="route", via=pts[:-1], delay=delay, speed=speed)


SWING_NUM = {-1: (-4.4, -14.6), 1: (-4.4, 14.6)}


def swing(p, side=-1, key="RB"):
    """The back checks for a blitzer, then swings wide behind the slot: the checkdown.
    Returns where to put his progression number."""
    route_to(p, key, (-4.8, side * 4.5), (-4.2, side * 10.5), (-3.0, side * 13.0), delay=0.4)
    return SWING_NUM[side]


# ---------------------------------------------------------------- drawing helpers
def _P(fld, p, d, w):
    return fld.xy(*p.to_abs(d, w))


def clean(ax):
    """Drop the painted yard numbers (they sit where receivers stand); clip all text to the window."""
    for t in list(ax.texts):
        if t.get_text().isdigit() and t.get_alpha() == 0.55:
            t.remove()
        else:
            t.set_clip_on(True)


def ruler(fld, p, fs=6.0, step=5, top=30):
    """Depth labels (yards past the line of scrimmage) just inside the left edge of the window."""
    xl = (160 / 3) - fld.y1 + 0.4
    for d in range(step, top + 1, step):
        y = p.los + d
        if fld.x0 + 1.5 <= y <= fld.x1 - 1.5:
            fld.ax.text(xl, y, f"{d}" + (" yd" if d == step else ""), fontsize=fs, color=style.INK_2,
                        ha="left", va="center", zorder=12,
                        bbox=dict(boxstyle="round,pad=0.12", fc=style.SURFACE, ec="none", alpha=0.85))


def draw(p, width_in=8.3, ax=None, show_ruler=True, **kw):
    """Draw a play without a title. A stand-alone field gets ~6.4 in of drawing width after the
    default subplot margins are cropped (width_in is the figure width before cropping)."""
    fld = p.draw(width_in=width_in, ax=ax, title=False, **kw)
    clean(fld.ax)
    if show_ruler:
        ruler(fld, p)
    return fld


def say(fld, p, d, w, text, color=style.INK_2, fontsize=7.0, ha="center", weight="normal", box=True):
    x, y = _P(fld, p, d, w)
    fld.ax.text(float(x), float(y), text, fontsize=fontsize, color=color, ha=ha, va="center", zorder=13,
                fontweight=weight, linespacing=1.15,
                bbox=dict(boxstyle="round,pad=0.18", fc=style.SURFACE, ec="none", alpha=0.9) if box else None)


def point_to(fld, p, text_at, target, text, color=style.INK_2, fontsize=7, ha="center"):
    (xt, yt), (xp, yp) = _P(fld, p, *text_at), _P(fld, p, *target)
    fld.ax.annotate(text, (float(xp), float(yp)), xytext=(float(xt), float(yt)), fontsize=fontsize,
                    color=color, ha=ha, va="center", zorder=13, linespacing=1.15,
                    arrowprops=dict(arrowstyle="-", color=color, lw=0.7, shrinkA=2, shrinkB=7),
                    bbox=dict(boxstyle="round,pad=0.18", fc=style.SURFACE, ec="none", alpha=0.9))


def branch(fld, p, pts, color=READ_GREEN, lw=1.7, dash=(0, (3, 2.2)), end="arrow", z=5):
    """A dashed path through absolute (d, w) points: a defender's reaction, or the other branch of a read."""
    xs, ys = _P(fld, p, np.array([q[0] for q in pts]), np.array([q[1] for q in pts]))
    fld.ax.plot(xs, ys, color=color, lw=lw, ls=dash, zorder=z, solid_capstyle="round")
    if end == "arrow":
        fld.ax.add_patch(FancyArrowPatch((xs[-2], ys[-2]), (xs[-1], ys[-1]), arrowstyle="-|>",
                                         mutation_scale=10, color=color, lw=lw, zorder=z))


def ghost(fld, p, pts):
    """A route that is part of the call but not part of the read: drawn faint grey."""
    branch(fld, p, pts, color="#a9a8a1", lw=1.4, dash="-", z=3.6)


def taken(fld, p, pts, lw=1.4):
    """A route the defender has taken away: grey, dashed, with an X at the end."""
    branch(fld, p, pts, color=GREY, lw=lw, dash=(0, (2, 2)), end="none")
    x, y = _P(fld, p, *pts[-1])
    fld.ax.plot([float(x)], [float(y)], marker="X", ms=8, color=DD, mec="white", mew=0.7, zorder=9)


def opening(fld, p, center, size, color=style.THIRD):
    """A soft green oval: the throwing window that opens."""
    x, y = _P(fld, p, *center)
    fld.ax.add_patch(Ellipse((float(x), float(y)), size[1], size[0], fc=color, alpha=0.2, ec=color, lw=1.0,

def triangle(fld, p, pts, color=style.THIRD):
    xy = [tuple(float(v) for v in _P(fld, p, *q)) for q in pts]
    fld.ax.add_patch(Polygon(xy, closed=True, fc=color, alpha=0.15, ec=color, lw=1.3, ls=(0, (4, 2)), zorder=1.7))


def bracket(fld, p, d, w0, w1, text, color=style.INK_2, fontsize=7, dy=1.2):
    """A horizontal dimension line at depth d from w0 to w1, labelled above."""
    (x0, y0), (x1, y1) = _P(fld, p, d, w0), _P(fld, p, d, w1)
    fld.ax.annotate("", (float(x1), float(y1)), xytext=(float(x0), float(y0)),
                    arrowprops=dict(arrowstyle="<->", color=color, lw=0.8), zorder=6)
    if text:
        say(fld, p, d + dy, (w0 + w1) / 2, text, color=color, fontsize=fontsize)


def num(fld, p, n, at, color=READ_GREEN, fs=7.6):

def plate(p, nums=(), ghosts=(), stretch=None, window=(-8.5, 26.5), lateral="full", keys=None, width_in=6.9):
    """A concept plate: the call, the read and the progression, in one consistent style."""
    fld = draw(p, lateral=lateral, window=window, width_in=width_in)
    for pts in ghosts:
        ghost(fld, p, pts)
    for n, at in nums:
        num(fld, p, n, at)
    if stretch:
        tag(fld, p, stretch)
    field_legend(fld.ax, keys or PLATE_KEYS)
    return fld

MARK_R = 0.7


def tracking(play):
    df = play.to_tracking()
    df["_id"] = df["displayName"].astype(str)
    return df


def frame_panel(ax, play, df, t, lateral, window_, rings=(), throw_t=None, deco=None):
    """One moment of a simulated play, with faint trails, optional rings and the ball's flight so far."""
    tmap = df.groupby("frameId")["time"].first()
    fid = int((tmap - t).abs().idxmin())
    ylo, yhi = sorted((play.ball_y - lateral[0], play.ball_y - lateral[1]))
    fld = Field(ax=ax, orient="vertical", los=play.los, window=window_, numbers=False, hashes=True,
                lateral=(ylo, yhi), to_go=play.to_go)
    cur = df[df["frameId"] == fid]
    shown = cur[cur["x"].between(play.los + window_[0] + 0.4, play.los + window_[1] - 0.4)
                & cur["y"].between(ylo + 0.4, yhi - 0.4)]
    ids = set(shown["_id"])
    trails = df[(df["frameId"] <= fid) & df["_id"].isin(ids) & (df["side"] != "ball")]
    for (_, side), g in trails.groupby(["_id", "side"]):
        xs, ys = fld.xy(g["x"].values, g["y"].values)
        ax.plot(xs, ys, color=style.OFFENSE if side == "off" else style.DEFENSE, lw=0.9, alpha=0.45, zorder=3)
    if throw_t is not None and t > throw_t:
        b = df[(df["side"] == "ball") & (df["frameId"] <= fid)]
        b = b[b["time"] >= throw_t - 0.05]
        xs, ys = fld.xy(b["x"].values, b["y"].values)
        ax.plot(xs, ys, color=style.BALL, lw=1.3, ls=(0, (2, 1.5)), zorder=9)
    ppy = fld.pts_per_yard()
    for _, rr in shown.iterrows():
        x, y = (float(v) for v in fld.xy(rr["x"], rr["y"]))
        if rr["side"] == "ball":
            if throw_t is None or t <= throw_t:
                # still in the quarterback's hands: draw it at his throwing shoulder so his letter stays readable
                x, y = (float(v) for v in fld.xy(rr["x"], rr["y"] - 1.0))
            elif rr["x"] > play.los + 2:
                # in flight or caught: nudge it beside the receiver so his letter stays readable
                x, y = (float(v) for v in fld.xy(rr["x"] - 0.9, rr["y"] - 0.9))
            ax.add_patch(Ellipse((x, y), 1.0, 0.62, fc=style.BALL, ec="white", lw=0.5, zorder=11))
            continue
        lab = str(rr.get("label", "") or "")
        if rr["side"] == "off":
            ax.add_patch(Circle((x, y), MARK_R, fc=style.OFFENSE, ec="white", lw=0.6, zorder=8))
            tc = "white"
        else:
            ax.add_patch(RegularPolygon((x, y), 4, radius=MARK_R * 1.3, orientation=np.pi / 4,
                                        fc=style.DEFENSE_TINT, ec=style.DEFENSE, lw=1.0, zorder=8))
            tc = style.INK
        fs = max(4.0, min(8.0, ppy * (0.85 if len(lab) <= 1 else 0.62)))
        ax.text(x, y, lab, ha="center", va="center", fontsize=fs, color=tc, fontweight="bold", zorder=10)
    for key, col in rings:
        rr = cur[cur["displayName"] == key].iloc[0]
        x, y = fld.xy(rr["x"], rr["y"])
        ax.add_patch(Circle((float(x), float(y)), MARK_R * 1.6, fc="none", ec=col, lw=1.7, zorder=7))
    if deco is not None:
        deco(ax, fld, cur, t)
    return fld


def strip(play, times, notes=None, lateral=(-24, 4), window=(-8, 14), ncols=2, rings=(), throw_t=None,
          width_in=6.5, deco=None):
    df = tracking(play)
    nrows = int(np.ceil(len(times) / ncols))
    pw = width_in / ncols
    span_w, span_d = abs(lateral[1] - lateral[0]), window[1] - window[0]
    fig, axes = plt.subplots(nrows, ncols, squeeze=False,
                             figsize=(width_in, nrows * (pw * span_d / span_w + (0.6 if notes else 0.25))))
    for ax in axes.flat[len(times):]:

def set_track(p, key, rows, t_end=None):
    """Hand-made keyframes [(t, d, w), ...] for one player (gridiron's man() ends on top of the
    receiver, and we sometimes need a defender who detours around traffic). Call last."""
    kf, t0, t1 = p._compile()
    kf = dict(kf)
    q = p.players[key]
    rows = [(min(t0, -1.0), q.d, q.w), (0.0, q.d, q.w)] + [tuple(r_) for r_ in rows]
    kf[key] = np.array(rows, dtype=float)
    p._compiled = (kf, t0, max(t1, t_end or rows[-1][0]))


def chase(p, key, target, lag=0.3, off=(-1.0, 0.0), t_start=0.15, t_on=0.8, t_end=4.0, detour=None,
          lag_after=None):
    """Keyframes for a man defender trailing `target`: he starts where he is, then settles `off` =
    (down, toward the offense's right) yards from where the receiver was `lag` seconds earlier.
    detour = (t0, t1, (d, w)): between t0 and t1 he is pulled toward that point (traffic)."""
    q = p.players[key]
    rows = []
    for tt in np.round(np.arange(t_start, t_end, 0.05), 3):
        f = min(1.0, max(0.0, (tt - t_start) / (t_on - t_start)))
        f = f * f * (3 - 2 * f)
        lg = lag
        if detour is not None and lag_after is not None and tt > detour[0]:
            k = min(1.0, (tt - detour[0]) / (detour[1] - detour[0]))
            lg = lag + (lag_after - lag) * k
        tgt = p.position(target, max(0.0, tt - lg)) + np.array(off)
        pos = np.array([q.d, q.w]) * (1 - f) + tgt * f
        if detour is not None:
            a, b, pt = detour
            if a <= tt <= b:
                g = np.sin(np.pi * (tt - a) / (b - a))
                pos = pos * (1 - g) + np.array(pt) * g
        rows.append((tt, *pos))
    return rows

What a pass concept is

A pass concept is a set of routes designed to work together on one play, usually by putting one or two defenders in a position where whatever they do leaves a receiver open. You met the idea in The Quick Game, where slant-flat and stick each pinned one defender between two receivers. This chapter takes the same idea deeper down the field, where the routes take longer to develop and the defenders in conflict are cornerbacks, linebackers and safeties instead of a single slot defender.

Why do offenses think in concepts rather than routes? Three reasons, and they explain a lot about how football is coached.

A concept is a sentence; routes are only words. A corner route on its own means nothing. A corner route run over a hitch, against a cornerback who has to choose between them, is a plan. The quarterback doesn’t read five receivers one by one; he reads one or two defenders and lets the concept tell him where the open man will be. That’s why concepts are drawn and taught as units, and why a receiver learns not only his route but his partner’s.

A name is a shortcut. “Smash” tells two receivers their routes in one syllable, from any formation, on either side of the field. Offenses that name concepts this way can teach a playbook faster and call plays faster, which matters in a two-minute drive or a no-huddle attack. Not every system names its plays like this; some spell out each receiver’s route in a long string of words or numbers, and the trade-offs between those styles are the subject of Offensive Systems and Languages, where concept-based terminology gets its full story.

A concept travels. Smash works from a 2x2 formation, from trips, with a tight end as the inside receiver or a running back. Because the conflict is about space (the gap between a cornerback sitting short and a safety playing deep), the same concept can be dressed up in a dozen formations, which makes it hard for a defense to recognize from the alignment alone.

CautionCommon misconception

“Every pass play has one intended receiver.” Almost none do. A concept has a first look, the route it was designed around, but it’s built so that whichever choice the key defender makes, somebody else is open. When an announcer says the quarterback “went to his second read,” that isn’t a failure of the play. It is often the play working as drawn.

What changes when the quarterback drops back

The quick game lives on throws that leave the quarterback’s hand in about two seconds, aimed at receivers five or six yards away. Dropback concepts give themselves more time and aim farther, and the numbers show what that buys and what it costs. In the 2025 regular season, throws that traveled zero to nine yards past the line of scrimmage left the quarterback’s hand after a median of 2.3 seconds; throws of 10 to 19 yards, the home of most of the concepts in this chapter, took 2.8 seconds, and throws of 20 yards or more took 3.1 seconds.1 That extra half-second is the price of the deeper drop (five steps from under center, or three from the shotgun, in the matched pairs of Passing Fundamentals) and of routes that need ten or fifteen yards to reach the spot where they break.

What does the half-second buy? Value. Those 10-to-19-yard throws averaged about +0.39 expected points added (EPA) per attempt, the course’s measure of how much a play changed the offense’s scoring chances, against about +0.10 for those zero-to-nine-yard throws. But they were completed only 54% of the time, against 72%, and they were intercepted twice as often (about 3.2% of attempts against 1.6%).

ImportantWhy it exists, and what it costs

Dropback concepts trade time and risk for yards. The offense asks its line to protect for three seconds or more, accepts a lower completion rate and more interceptions, and gets in return the throws that move the ball in big pieces: the 15-yard dig on third-and-8, the corner route that turns a drive into a scoring chance. The quick game keeps an offense on schedule; dropback concepts are how it gets ahead of it. The bill comes in sacks, hits on the quarterback and a pass rush with time to get home, which is why protection (Pass Protection) and dropback concepts are planned together.

Four ways to put a defender in a bind

Every concept in this chapter does one of four things to the defense. Learn the four and you can name the purpose of a concept you have never seen, which is most of what “reading” a pass play from the couch means.

  1. Vertical stretch (high-low). Two receivers in one defender’s area, one above him and one below. He can sink under the high route or come up on the low one, not both. This is the high-low read of Passing Fundamentals, and it is the most common family by far: smash, flood, levels, drive, shallow cross, dagger, Y-cross and yankee are all built on it. Notice that several of these (drive, shallow cross, Y-cross) use crossing routes: what makes them vertical stretches is that, at the moment of the throw, one route is above the key defender and one below him.
  2. Horizontal stretch. Receivers on both sides of a defender at about the same depth, the horizontal stretch of Passing Fundamentals. Underneath, that’s the quick game’s spacing; deep, it is four verticals, post-wheel and scissors, which stretch the deep defenders side to side: two routes split one deep defender’s area (or two defenders’ leverage) inside and outside.
  3. Triangle. Three receivers arranged around two defenders. A triangle read is a three-receiver read in which the routes form a triangle, one high and two low (or two high and one low), so the quarterback has a high-low and an inside-outside choice at once. Snag, from the quick game, is the classic; curl-flat with a third route inside is the dropback version.
  4. Man-beater. A man-beater is a concept designed to beat man coverage. Stretches beat zone defenders, who guard areas; a man defender follows one receiver wherever he goes, so there is no area to overload. Man-beaters win instead with routes that cross close together (so defenders run into traffic), routes that break away from a defender’s leverage, and mismatches like a running back against a linebacker. Mesh and the angle route are the classics. Crossing alone doesn’t make a man-beater; what makes one is that it wins by making a man defender run through traffic or lose his leverage, not by stretching a zone.

Most real concepts are hybrids (a flood is a vertical stretch that also stretches the flat defender sideways; mesh sits in zone holes when it isn’t running away from man defenders), so the family names describe a concept’s main idea. That idea also tells you what coverage the concept is built to beat, and therefore when an offense will call it. We’ll come back to that matchup at the end of the chapter, and What Beats What builds it into a full table.

How to read the plates

The fourteen concepts below are drawn as plates, the way a playbook draws them, in one consistent style:

  • The offense is in shotgun with 11 personnel (one back, one tight end, three receivers) unless the caption says otherwise: X and Z are the outside receivers, H a slot receiver, Y a tight end or second slot, R the running back.
  • Blue routes are the concept. Grey routes are part of the call but outside the read: they are there to hold defenders in place. They aren’t forbidden; if a grey route is plainly open (a backside receiver left alone, say), a quarterback may take it, and many teams let him peek at a one-on-one on the backside before the snap. But he doesn’t read his way to them.
  • Green numbers are the progression, the order in which the quarterback looks. These are typical teaching orders; every team writes its own, and many change the order by coverage.
  • The defense is drawn in the coverage the concept is built to beat. Dashed orange arrows are zone drops; the ringed defender is the one in conflict, the man the quarterback reads. In the man-coverage plates, dotted lines join each defender to his man; which linebacker takes the back varies by team and call, and here it is always the Will, with the Mike free to rush or to sit in the middle.
  • The thick orange line is the line to gain; most plates assume ten yards to go, and the caption says so when it doesn’t.
  • The label at the top right names the concept’s family.

Two more things the plates simplify. First, receivers run to landmarks, spots on the field rather than bare depths, and you can see them on a broadcast: seam routes about two yards outside the hash marks (wider against one deep safety), go routes on or outside the painted numbers so there are five or six yards left for a back-shoulder throw, the smash corner route toward a spot about six yards from the sideline, and the Y-cross to the far numbers (Coverage Fundamentals teaches the defense’s landmarks). Second, the ball is drawn in the middle of the field to keep the picture symmetric. In a game it usually sits on a hash, and concepts get aimed: flood, Y-cross and post-wheel usually go toward the wide side, where the routes have room to finish; smash is often run into the short side against Cover 2, because the corner route is a shorter throw there.

High to low, or low to high? Each plate’s progression follows one of two rhythms, and knowing which saves you memorizing fourteen orders. Most vertical stretches are read from the top down: look at the deep route first, because if it’s open it’s the bigger gain, then come down to the short one, which will still be there a beat later. Some are read from the bottom up: a shallow crosser against man coverage wins immediately or not at all, so it gets the first look. And on a money down the progression often starts with whichever route reaches the line to gain. Each plate says which rhythm it uses.

On the sideline: smash, flood and the curl-flat triangle

The first three concepts attack the area near one sideline. Two of them (smash and flood) are vertical stretches; the third (curl-flat) is the dropback game’s triangle. Coaches like to say the sideline is an extra defender, for the defense, because a receiver can’t run past it and a throw that drifts lands out of bounds. Offenses attack it anyway because of arithmetic: out there a defense has only two or three players guarding a wide strip of grass, against five or more crowded between the hashes.

Smash: the corner over the hitch

Smash pairs two receivers on one side. The outside receiver runs a hitch at five or six yards and turns back to the quarterback. The inside receiver runs a corner route: ten or twelve yards straight up the field, then an angle toward the sideline and the back of the end zone. The defender in conflict is the cornerback on that side, and the read is high to low: the corner route first.

Why does it work against Cover 2? Cover 2 splits the deep field between two safeties and asks the cornerbacks to stay short, guarding the flats, so that nothing quick gets thrown to the outside. That leaves a seam of open grass along each sideline between the corner’s short zone and the safety’s deep half, roughly 12 to 20 yards deep, and the safety has to cover it from his spot near the hash, ten or fifteen yards away. The corner route aims at that seam. Whatever the cornerback does, one of the two routes is on the wrong side of him, so the quarterback only has to watch one man.

The corner route has to be thrown with care. Too flat and the cornerback can recover under it; too high and the safety arrives. Many teams build a rule into the route (a route conversion, which we’ll get to). Against two deep safeties, whose cornerbacks squat, the receiver takes it to depth, 15 to 20 yards along the sideline, over the cornerback and away from the safety. Against one deep safety, whose cornerback bails into a deep third, a corner route to that depth runs straight into him, so the receiver flattens it at about 12 to 14 yards, underneath the bailing cornerback, and some teams convert it to a post.

On third-and-8 there is one more wrinkle a coach would point out: a six-yard hitch is short of the sticks. It needs yards after the catch, which is one more reason the quarterback wants the corner route, and why some teams push the hitch to the line to gain on long downs.

What smash gives up. Against Cover 3, whose cornerbacks drop into deep thirds instead of squatting, a corner route run to full depth runs into a defender standing right on top of it, so the read flips: the cornerback bails, and the hitch underneath (or the flattened corner route) is the throw. Against man coverage, smash becomes two one-on-one routes, a hitch that has to beat a cornerback and a corner route that has to beat a slot defender, which is fine if the receivers are better than the defenders and not otherwise. Smash is a Cover 2 beater first, a decent answer to Cover 3 second, and only a hope against man.

Flood: three levels on one sideline

Flood (many teams call it sail) puts three receivers at three depths near one sideline: one deep, one in between and one shallow. On the strong side of a Cover 3 defense there are only two defenders in that area, the cornerback, who has the deep third, and the curl-flat defender underneath. Three receivers, two defenders.

Flood is a vertical stretch with three floors instead of two. The go route occupies the top floor and takes the cornerback with it, because a Cover 3 cornerback is taught never to let anyone get behind him. That leaves the curl-flat defender alone with the sail and the flat: the same high-low as the curl-flat read of Passing Fundamentals and slant-flat in the quick game, but with the high route farther outside and deeper, where he has to run farther to get under it. If he comes up for the flat, the sail settles in the space he left, in front of the cornerback, for 12 or 14 yards. If he sinks, the back catches the ball in the flat with open grass in front of him.

Flood’s cost is the other side of the field. It sends three receivers to one sideline, so the backside routes are usually just there to hold defenders, and a defense that rotates its coverage toward the flood (a safety rolling down to the flat side at the snap) can outnumber it. That is why flood is so often run off a fake run or with the quarterback rolling toward the flood side; the run fake and the bootleg, which move the defenders before the routes even arrive, are in Play-Action, Bootlegs, and Screens.

Curl-flat: the West Coast triangle

Curl-flat is the high-low you met in Passing Fundamentals: a curl at 10 to 12 yards over a flat route underneath, both in the area of the one defender who guards the short outside zone. As a dropback concept it usually comes with a third route, a back or tight end who settles in a hook route just inside the curl at five or six yards. That third route turns a two-man read into a triangle.

The triangle is what makes the concept durable. The nickel can’t cover the curl and the flat (that’s the high-low), and if he splits the difference, the Mike has to choose between widening to help on the curl and staying home on the hook (that’s the in-out). The quarterback reads it with the same rhythm on almost every snap: eyes to the curl as his back foot lands, then down to the flat, then inside to the hook. The routes are short enough that the ball comes out in under three seconds, which is why curl-flat sits at the border between the quick game and the dropback game.

Curl-flat is a West Coast staple, the family of offenses descended from Bill Walsh’s 49ers, which were built on high-percentage throws with a third receiver underneath as the safety valve. Its weakness is its depth. Every route in the triangle is within 12 yards of the line of scrimmage, so a defense that squeezes the short zones and tackles well can hold it to a five-yard gain; and against two-high coverages, whose cornerbacks often sit short and outside, the curl can find a defender standing right on top of it.

Vertical stretches in the middle: levels, drive, shallow cross and dagger

The next four concepts carry the high-low into the middle of the field, where the linebackers live. The middle is harder to throw into (there are more defenders there, and they are closer together) but it’s also the most valuable area on the field, as the chart at the end of this section shows. The defender in conflict in each of these concepts is a linebacker, or a safety standing behind the linebackers, and the routes are mostly in-breakers: digs, shallow crosses and in routes that run across his face.

Levels: two in-breakers, two depths

Levels sends two receivers from the same side across the field on in-breaking routes at two depths: in the most common version the outside receiver runs a short in at about five yards and the inside receiver a dig at about twelve. The defender between those levels is the linebacker guarding the hook area on that side.

Why in-breakers? Because the linebacker in the middle is usually watching the quarterback, not the receivers, and he drops backward, which makes him slow to react to a route crossing in front of him or behind him. The short in route arrives first; the dig arrives a beat later, behind him. Coaches teach the quarterback to look high first: if the linebacker comes up on the in route, the dig behind him is the bigger gain. Peyton Manning’s Indianapolis offenses (where the play was called “Dig”) ran levels so often, by Chris Brown’s count, that ten times a game was not unusual, and its logic, two in-breakers on two floors, runs through many of the middle-of-the-field concepts that follow.2

Drive: the shallow and the dig, together

The next two concepts are built on one route, the shallow: a receiver crossing the whole field just past the line of scrimmage, two to five yards deep (some teams call it a drag). Each concept pairs it with a deeper in-breaker; what differs is where the two come from.

Drive is levels stretched across the whole field. One receiver runs a shallow, two to four yards deep, all the way to the far side; a teammate from the same side runs a dig at about twelve yards behind him, following him across. The pair passes through the hook zones of both linebackers, but the quarterback reads the first one they reach, the Mike: the ball comes out as the shallow clears him, long before the routes get to the Will. Unlike levels, drive is usually read low to high, because the shallow is open early or not at all.

The shallow is the engine. The moment one does, the dig is open behind. That makes drive one of the most useful concepts in football: a man-beater and a zone-beater in the same call. Drive is in a great many NFL playbooks, often under that name, and it’s a favorite of offenses that like to make the defense tackle a receiver who catches the ball running full speed across the field.

Shallow cross: the Air Raid’s bread and butter

Shallow cross uses the same two routes as drive, a shallow under a deeper in-breaker, but sends them from opposite sides. One receiver runs the shallow at three to five yards; the receiver from the other side runs a dig or a “sit” route at 10 to 12 yards, settling over the ball. (A sit route runs up to a depth and simply stops in the open space between zone defenders, facing the quarterback; against man it keeps running, as a dig.) Like drive, it is read low to high. The two routes cross the middle of the field going in opposite directions, so the middle linebacker has a receiver coming from each side at once.

Why would two-high coverages be vulnerable to this? In Quarters and its relatives, each safety reads the inside receiver on his side: if that receiver goes vertical, the safety takes him deep; if he runs short, the safety is free to help on the outside receiver, usually by jumping an in-cut. That rule is what makes Quarters so good against the run and the quick game, because the safety can come up hard on anything short. The cornerback, meanwhile, plays the outside receiver almost man-to-man, from outside leverage.

Dagger turns both rules against them. The seam takes the safety, the one defender whose job was to rob the dig, out of the picture. The underneath defenders don’t come up on a vertical route, as you might expect; they are taught to run with it for a few steps, because a seam thrown over their heads is the play that beats them. So the nickel carries the seam and the Mike drops to his hook zone, eight or nine yards deep. That leaves the cornerback alone on the dig, and he is on the wrong side of it: playing from outside, he can only trail a receiver who breaks inside. The dig, timed to arrive a beat after the seam, settles in the empty middle between the levels.

Against one-high coverage the read is similar but the threat is different: the seam attacks the single safety, and if he stays in the middle, the dig crosses under him. Dagger needs time, because a 15-yard dig takes a deep drop and three seconds or more, which is why it’s usually called with six-man protection, or with five and the back checking for a blitzer before he releases, as drawn.

Y-cross: a flood built from across the field

Y-cross combines the two ideas so far, the three-level flood and the in-breaker through the middle. A receiver from one side (in the Air Raid’s lettering, the Y, the inside receiver) runs a crossing route that climbs to 15 to 18 yards as it goes and finishes at the far numbers. On that far side, the outside receiver runs a go route to clear the cornerback away and another receiver runs to the flat. The result is a flood whose middle level arrives from the other side of the field, at full speed, where no defender on that side was looking for it.

NoteGo deeper: the Air Raid’s small menu

The Air Raid was built at small colleges, not in the NFL. Hal Mumme, a former high school coach and college assistant, and Mike Leach, a law-school graduate who had never played college football, put it together at Iowa Wesleyan around 1989, took it to Valdosta State and then, in 1997, to the University of Kentucky, where quarterback Tim Couch became the first pick of the 1999 NFL draft. Leach went on to run it as a head coach at Texas Tech, Washington State and Mississippi State.4

Their starting point was the passing game of LaVell Edwards’s BYU teams, which they studied in person, and their central idea was subtraction: run a handful of concepts (mesh, shallow cross, Y-cross, four verticals, a few quick-game calls) from a handful of formations, and practice them so many times that the quarterback and receivers react instead of think. Where an NFL playbook might carry hundreds of pass plays, the Air Raid lived on a dozen or so. Its lesson for this chapter is that a concept’s power comes as much from repetition as from design: a quarterback who has read the same curl-flat defender on the same Y-cross ten thousand times throws the ball before the defender has finished deciding. The full story of the Air Raid’s spread through college and then the NFL is told in The College Laboratory.

Man-beaters: mesh, the rub and the angle route

Everything so far is built to beat zone defenders, who guard areas and can be put between two receivers. Man coverage changes the problem. Each defender has one receiver and follows him, so there is no area to overload and no defender in two minds; the defense is betting its players can stay with yours. The most common version in the NFL is Cover 1: a man on every receiver, one safety deep in the middle, and often one more defender free to rush or to sit in the middle as a helper (Man Coverage builds it from the defense’s side). The answer is to make staying with a receiver hard: send receivers across the field, where their defenders have to chase them through traffic; break away from a defender’s leverage; or find a defender who is badly matched against the man he has to cover.

Mesh

Mesh is the purest man-beater in football. Two receivers from opposite sides run shallow crossing routes, five or six yards deep, that pass within a yard or two of each other over the middle of the field. One runs his cross a step higher than the other, and Air Raid coaches tell them to pass close enough to slap hands.

Against man coverage the two crossing paths do the work. The nickel chasing H has to get past Y coming the other way, and the safety chasing Y has to get past H. One of them almost always loses a step, and a receiver crossing the field with a one-step lead is nearly impossible to catch from behind. Against zone the same routes simply settle: each crosser finds the space between two underneath defenders and sits down facing the quarterback, so the concept is useful even when the defense guesses right. The corner routes give the quarterback a high-low against the cornerbacks, and keep the safety from crowding the middle.

Why would the Air Raid’s progression start with the corner route in the purest man-beater? Timing and payoff. The crossers need a beat to get through the traffic and come out the other side; the corner route over the back’s flat is ready first, and it is a high-low on whoever has the flat, so if it’s open it’s the biggest gain on the play. By the time the quarterback comes off it, one of the crossers has usually come free. Against a defense he knows is in man, many teams skip straight to the crossers, the low-to-high rhythm of drive.

Mesh came into the Air Raid from BYU, which, as Leach put it, ran the crossing pattern “for two decades” under LaVell Edwards, and it became one of the Air Raid’s signature plays;5 versions of it now appear in nearly every NFL playbook, often as a third-down call, when defenses play the most man coverage.

The rub, and where it becomes a pick

Mesh works because of a specific kind of route. A rub route (also called a pick route) is a route designed to cross close to a teammate’s route, so a defender chasing one receiver has to go around the other. Crossers, stacked receivers who release in opposite directions, and a slot receiver running a quick out under an outside receiver’s slant are all rubs. They are among the most effective plays in football against man coverage, especially near the goal line, where the field is short and defenders play tight.

They are also the most argued-about. The rules allow a receiver to run his route even if a defender runs into him. They don’t allow him to block a defender downfield until the pass is touched. In the NFL, a receiver who deliberately makes contact with a defender more than one yard past the line of scrimmage to get a teammate open, from the snap until the pass is touched, has committed offensive pass interference, a 10-yard penalty; contact within a yard of the line isn’t pass interference.6 That is the line between a rub and an illegal pick.

So is a pick inside that first yard legal? As far as pass interference goes, yes, and offenses use it: stacked and bunched receivers who release into each other right at the line are designed to make the contact happen there, before the one-yard limit. A receiver who grabs or shoves a defender can still be flagged for holding or illegal use of hands, but bumping him at the line on the way into a route is not a pick.

The angle route: a running back against a linebacker

The other classic man-beater uses a mismatch instead of traffic. In man coverage, somebody has to cover the running back, and it is usually a linebacker (in these pictures, the Will): a player chosen for stopping runs, often thirty pounds heavier than the back and a step slower in space. The angle route (many playbooks call it the Texas route) is built for that matchup. The back releases toward the flat as if running a swing, and when the linebacker widens to chase him, he plants and snaps back inside, angling upfield under the middle of the field.

The angle route is a cousin of the option route of Passing Fundamentals: the back is breaking away from his defender’s leverage. Deep shots: four verticals, post-wheel, scissors and yankee

The last four concepts go deep. They exist because a defense that never has to worry about the deep ball will crowd every short and intermediate route in this chapter; somebody has to make the deepest defenders defend something. Each of them stretches the deep defenders, the safeties and the cornerbacks who play over the top, rather than the linebackers.

Four verticals

Four verticals is the simplest idea in the passing game: four receivers run straight up the field at once. The two outside receivers stay near the sideline, the two inside receivers run up the seams, a few yards outside the hashes. The defense has only two or three deep defenders, so somebody is one-on-one, or nobody is there at all.

The arithmetic is the point. Against Cover 3, three deep defenders divide the field into thirds; four vertical routes put two receivers in the middle third, where there is only one defender. If he leans toward one seam, the quarterback throws the other. Against Cover 2, with two deep safeties each guarding a half, the seams split the safeties and the outside receivers attack the gap between the squatting cornerbacks and the safeties along the sideline, the same hole smash attacks. The concept forces the deep defenders to show where they are, and the quarterback reads them, not the underneath ones. The defense’s best answer is to have the curl-flat defenders run with the seams for the first several yards, so the safety only has to help on one of them; how Cover 3 defenses do that is a large part of The Cover 3 Family.

Notice what the quarterback did with his eyes. He looked toward Y as he dropped, and the safety, who reads the quarterback, drifted that way. Moving a defender with your eyes is called a look-off, and it’s an essential skill for any concept that reads a single deep defender (Quarterback Play has more).

Four verticals has no single inventor. Its roots are usually traced to the vertical passing games of Sid Gillman and Don Coryell, to LaVell Edwards’s BYU and to the run-and-shoot offenses of Mouse Davis, all of which built their attacks around receivers running up the field; the Air Raid turned it into a signature call, with receivers taught to stay in their lanes but get open in them.7 In the modern NFL it rarely looks like four straight lines. The seam runners usually carry a read, holding straight against one deep safety and bending inside between two (a route conversion, as below), and offenses with speed outside, like Andy Reid’s Chiefs with Tyreek Hill, use the outside go routes to push the cornerbacks off.

Post-wheel

Post-wheel splits the defenders on one side, inside and out. The outside receiver runs a post, angling toward the middle at about 12 yards; the inside receiver (or a back) runs a wheel, out toward the flat and then straight up the sideline, into the space the post receiver just left. It is read high and outside first: the wheel.

Why does it work? Because each defender on that side has a sensible rule, and the two routes make the rules collide. Many NFL defenses now play match Cover 3: instead of dropping to a spot, each defender picks up a receiver once the routes declare. The cornerback’s rule is to carry the outside receiver if he goes deep, so when Z pushes vertical and bends to the post, the cornerback goes inside with him. The curl-flat defender’s rule is to take the second receiver if he goes to the flat. The wheel starts as exactly that, a harmless flat route, and then turns up the sideline, where a nickel who started five yards deep, facing the quarterback, has to turn and run with a receiver who already has a head of steam. The wheel is behind him and outside the cornerback, and nobody’s rule covers it in time.

If the cornerback refuses to be dragged inside (some teams teach him to pass the post on to the free safety and stay outside for anything up the sideline), the post is the throw: now it’s a one-on-one inside against the free safety, whom the backside seam has held in the middle. We file post-wheel as a horizontal stretch because that is its shape: two deep routes, one breaking inside and one outside, splitting the defenders’ areas side to side. Against man coverage it is better still: the wheel puts a slot receiver or a running back against a nickel or a linebacker who has to turn and run up the sideline with him.

Scissors beats Quarters by turning its leverage inside out. In Quarters, as you saw with dagger, the safety reads the inside receiver from inside leverage and carries him if he goes vertical, and the cornerback plays the outside receiver almost man-to-man from outside leverage. Each defender starts on the side of his receiver that the receiver is least likely to break toward. Scissors sends both receivers the other way. Y goes vertical, so the safety carries him, and then Y breaks outside: the safety has to cross the field toward the sideline, through Z’s path, from the wrong side. Z breaks inside on the post: the cornerback has to cross inside, through Y’s path, from the wrong side. Both defenders lose their leverage at once, and the crossing paths add a rub at depth. Against man coverage the same crossing makes it a man-beater, too.

Yankee

Yankee is a two-man deep shot. One receiver runs a post from one side; the other runs a deep over route, a crosser at about 18 to 22 yards, from the other side. Both routes finish in the deep middle, one above the single deep safety and one in front of him. It is usually run off play-action, with a fake handoff to freeze the linebackers, and with max protection, seven or eight blockers, because both routes take three seconds or more to develop.

Why does it work? The run fake does half the job. Linebackers are taught to stop the run first, so they step forward when they see a handoff, and that step takes them out of the path of the deep over, which crosses behind them at a depth no underneath defender can reach. The other half is a high-low on the one man left: the free safety in the middle of the field can sink to take away the post or come forward to take away the over, not both. The post keeps him honest, and the over is thrown where nobody else can reach it: in front of the deep-third cornerback, who can’t leave his deep outside area, and behind linebackers who have just stepped toward a handoff. Sean McVay’s Rams made yankee one of their signature shot plays of the late 2010s (the Rams often ran it from three-receiver sets with jet motion), Kyle Shanahan’s 49ers ran it too, and it now appears in most NFL playbooks that come from the Shanahan and McVay family of offenses.8 The price is the protection: with seven or eight blockers, only two or three receivers go out, so if the safety plays it perfectly the quarterback has very little to throw to. Tags and route conversions

Fourteen concepts sounds like a lot, but a real playbook has many more, and most of the extras aren’t new concepts at all. They are the same concepts with a word added. A tag is a word added to a play call that changes one player’s assignment and leaves everyone else’s alone. “Mesh” tells all five receivers what to do; “Mesh, Z-wheel” keeps mesh intact except for Z, who runs a wheel instead of his corner route. Tags let a play-caller adjust a concept to a defense’s habits (a cornerback who jumps corner routes, a safety who cheats toward the crossers) without teaching a new play.

The second tool is subtler, and it explains how one call can be right against several coverages. A route conversion is a rule built into a route that changes it according to the coverage the receiver sees. The receiver doesn’t wait for the quarterback to tell him; he reads the defense himself, usually the safeties, and the quarterback, reading the same thing, knows where he will be.

You’ve already met the most important conversion. The option route of Passing Fundamentals, which some systems call the choice route, is a conversion with two branches: against man, break away from the defender’s leverage; against zone, sit in the hole. Others are everywhere in this chapter. The smash corner route goes to depth over a squatting Cover 2 cornerback and flattens under a bailing Cover 3 one. The levels and mesh crossers keep running against man and settle against zone. The dagger dig sits down in a window against zone and keeps crossing against man. And the Y-cross, the Air Raid’s most conversion-heavy route, keeps running toward the far sideline against Cover 3 and man, and sits down in the window against two deep safeties or a robber lurking in the middle.

One caution about names while we’re here. Teams name the same concept differently (you’ve already met flood and sail, and levels as the Colts’ “Dig”), and some reuse a name for a different concept. The shape is what to learn; a name only tells you what one team calls it.

The Patriots of the Tom Brady era were the model of an offense built on conversions. Their Erhardt-Perkins system (named for Ron Erhardt and Ray Perkins, who coached together on New England’s staff in the 1970s, and carried forward by Bill Parcells to Bill Belichick and Josh McDaniels) named whole concepts with short words, so a quarterback could call them at the line.9 It also gave its slot receivers wide freedom to read and adjust their routes after the snap: the option routes of Troy Brown, Deion Branch, Wes Welker and Julian Edelman, which broke one way against man and sat down against zone, were a staple of the Brady years.10 That’s why New England’s passing game seemed to have an answer for every defense without a thousand plays: the same few concepts, each with branches. The system’s history and its rivals are in Offensive Systems and Languages.

NoteMadden vs. real life

In a video game, the receiver runs the route drawn on the play art, whatever the defense does, and you press a button for the one you want. In the NFL, a large share of routes are conditional: the receiver converts the route against the coverage, the quarterback has to read the same thing and throw to where the receiver will be, and a mismatch between their reads is one of the commonest causes of an interception that looks, on the broadcast, like the quarterback threw to nobody. When an announcer says a receiver and quarterback “weren’t on the same page,” that’s usually what happened: one of them saw man and the other saw zone.

The quarterback’s side: one clock for every concept

Each plate in this chapter carried a progression, and they all follow the same pattern. That pattern is the real skill of dropback passing, so it is worth spelling out.

Before the snap, the quarterback identifies the shell (one high safety or two), checks the leverage of the defenders over the receivers in the concept, and asks the question every concept poses: who is the defender in conflict, and where is he? If the concept is called to both sides, as smash often is, this is when he picks the side. And if what he sees is wrong for the call (a zone-beater against a defense that is plainly in man, say), many offenses let him change it: the play comes in as a pair of concepts, or with a check-with-me, and he picks at the line (The Pre-Snap Phase). The pre-snap picture is still only a hypothesis; the first step of the safeties after the snap confirms or breaks it (Disguise, Rotation, and Split-Field Coverage).

If the defense shows a blitz, he checks the protection (Pass Protection). Every dropback concept carries its own pressure answer: if the defense brings one more rusher than the protection can block, the concept’s quickest route becomes the hot route, the throw that has to come out before the free rusher arrives. On drive and mesh that is a shallow crosser, which is open almost at once against a blitz; on smash, flood or dagger it is usually the shortest route (the hitch, the flat) or the back releasing from the backfield.

At the snap, his eyes go to the conflict defender, not to a receiver. His drop is timed so that the first read is ready as his back foot lands: on smash, the corner route breaks as he hits the bottom of his drop; on dagger, the dig arrives a beat later, which is why dagger takes a deeper drop or a hitch step.

After the first look, he moves to the next read with his feet as well as his eyes. Each hitch step, a short step up into the pocket, resets his feet for the next throw. A typical dropback progression gets two or three looks before the rush arrives; the last is the checkdown, the short throw to a back that keeps the play from losing yards. As Passing Fundamentals showed with FTN’s charting, more than half of all throws go to the first read, and those are the quickest and most valuable: the concept is at its best when its first answer is right.

The order of the looks follows the two rhythms from the start of the chapter: high to low on most vertical stretches, low to high when a shallow crosser wins early or not at all, and on a long third down, the route that reaches the line to gain first.

Two honest limits on this table. First, you usually can’t tell these concepts apart before the snap, and that is by design: smash, flood and post-wheel all start as a slot and an outside receiver on one side, and an offense that tipped which one was coming would be easy to defend. What you can predict before the snap is the family, from the shell (two-high invites smash, dagger and scissors; one-high invites flood, four verticals, post-wheel and yankee; man invites mesh and the angle route) and from the down and distance. The concept itself reveals itself in the first two seconds, when the routes break. Second, nobody publishes how often each concept is called. Public play-by-play records where the ball went, not what was drawn; the Big Data Bowl tracking data at the end of this chapter charts each receiver’s route, which is enough to find concepts by their route combinations if you want to count them yourself.

Film room

The concepts above are in almost every NFL playbook. A few teams built an identity on them, and watching those teams is the fastest way to learn to see the concepts at game speed.

NoteFilm room: Texas Tech 2007–2008

Mesh and Y-cross, rep after rep. Mike Leach’s Texas Tech teams were the Air Raid at full volume. In 2007, quarterback Graham Harrell threw for 5,705 yards and redshirt freshman receiver Michael Crabtree caught 134 passes for 1,962 yards and 22 touchdowns; Crabtree won the Biletnikoff Award as the nation’s best receiver that season and again in 2008.11 Nearly all of it came out of the same small menu of concepts drawn in this chapter: mesh, shallow cross, Y-cross and four verticals, from wide 2x2 and trips formations with enormous splits between the offensive linemen.

What to notice on the tape: the mesh point, two crossers passing over the center close enough to touch, and a defender losing a step as he goes around; Harrell’s eyes on the curl-flat defender on Y-cross, and the ball arriving as the crosser came into the window; and how few different plays there were. The defense knew what was coming, and it didn’t matter, because the offense had run each concept thousands of times against every look.

One game to find: Texas Tech’s 39–33 win over No. 1 Texas on November 1, 2008, the school’s first win over a top-ranked team. Trailing 33–32 with 1:29 left, Harrell drove the Red Raiders 62 yards in six plays, and Crabtree caught the winning touchdown at the sideline with one second left.12 The final drive is famous; the rest of the game is the small menu at work.

Leach’s former quarterback Kliff Kingsbury later brought the same concepts to the NFL as the Arizona Cardinals’ head coach (2019–2022), and in 2026 joined Sean McVay’s Rams staff as assistant head coach.13

NoteFilm room: Los Angeles Rams 2018

Yankee and the play-action shot. In Sean McVay’s second season, the Rams went 13–3 and reached Super Bowl LIII, where they lost to New England 13–3. Jared Goff threw for 4,688 yards and 32 touchdowns.14 Their passing game ran on a single foundation: line up in the same formation with three receivers, run outside-zone handoffs out of it, and throw play-action shots from the same look, yankee among them. On throws of 20 yards or more in the air, the Rams averaged about +0.77 EPA per attempt, third-best in the league, against a league average of about +0.37.15

What to notice on the tape: the linebackers’ first step on the run fake, toward the line; the deep over crossing behind them at 18 to 20 yards; and the free safety in the middle, who has to decide between the over in front of him and the post behind him. Goff often had only two routes to choose from, and his decision was made by one defender.

One game to find: the Rams’ 54–51 win over the Chiefs on Monday night, November 19, 2018, in Los Angeles, when Goff threw for 413 yards and four touchdowns and Mahomes for 478 yards and six.16 Both offenses’ play-action and deep shots are on display; it is the companion game for the next film room too.

NoteFilm room: Kansas City Chiefs 2018

Four verticals with a 50-touchdown quarterback. In Patrick Mahomes’s first season as the starter he threw for 5,097 yards and 50 touchdowns and was named the league’s MVP. Andy Reid’s offense threw deep more than almost anyone: 15% of the Chiefs’ attempts traveled 20 yards or more in the air (fifth in the league; the league average was 12%), and those throws were worth about +0.75 EPA each, fourth best.17 With Tyreek Hill’s speed outside and Travis Kelce running up the seam, vertical routes made every deep defender declare early, which is the idea four verticals is built on.

What to notice on the tape: the deep defenders’ first steps. With Hill on the outside, cornerbacks bailed at the snap, which opened everything underneath; with Kelce on a seam, the middle safety had to choose. Watch the seam routes for a conversion, bending or staying straight as the receiver reads the safeties, and Mahomes moving the safety with his eyes before throwing the other way. The November 19 game against the Rams (above) is the place to start: Mahomes threw six touchdowns and three interceptions, a good lesson in what deep shots cost as well as what they pay.

TipWatch for it: name the concept

Before the snap

  • Find the shell. One deep safety or two? Smash, dagger and scissors are two-high beaters; flood, post-wheel and yankee feast on one-high; four verticals is built for Cover 3 and still splits Cover 2’s safeties. (Can’t see the safeties on the TV angle? See How to Watch a Broadcast.)
  • Man or zone? Pressed corners, a defender following a motion man across the formation: think man, and think mesh, rubs and the angle route.
  • Check the situation. On third-and-long, expect a concept with a route past the line to gain; on third-and-short against man, expect crossers and rubs. You are predicting a family, not a name: from the same formation, smash, flood and post-wheel look identical until the routes break.

On the replay (the end-zone camera or the skycam is best)

  • Find the defender in conflict first. Who stood between the routes on the side the ball went to? What did he do? The ball usually went to the receiver he didn’t take.
  • Then name the stretch by where the routes were around him when the ball came out. One above and one below him: a vertical stretch (even if the routes crossed on the way there, as in drive or Y-cross). One on each side: horizontal. Three around two defenders: a triangle. No zone defender to stretch at all, with receivers running away from men chasing them through traffic: a man-beater. Then name the concept.
  • Count the time. A dropback concept takes about “one-thousand-one, one-thousand-two, one-thousand-three.” If the ball is out before “two,” it was the quick game.

For your prediction log (How to Watch a Broadcast): on each third-and-medium, write the shell you see and the concept family you expect (two-high: smash or dagger; one-high: flood or a deep shot; man: mesh or rubs). Check yourself on the replay.

A vertical stretch (a high-low) of the free safety, built to beat one-high coverage, read high to low: the post, then the dig, then the back. Takeaways

  • A pass concept is a set of routes built to work together, usually by putting one or two defenders in a bind; it has a name so the whole offense can learn and call it as a unit, from any formation.
  • Concepts come in four families, named by where the routes are around the key defender when the ball comes out: vertical stretches (one above him, one below: smash, flood, levels, drive, shallow cross, dagger, Y-cross, yankee), horizontal stretches (one on each side, deep: four verticals, post-wheel, scissors), triangles (three receivers around two defenders: curl-flat, snag) and man-beaters (traffic and leverage against man defenders: mesh, the angle route).
  • Each concept is built to beat certain coverages: smash, dagger and scissors attack two-high; flood, post-wheel and yankee attack one-high; four verticals is built for Cover 3 and still splits Cover 2; mesh and the angle route attack man. That is why the quarterback’s pre-snap read starts with the safeties.
  • The quarterback reads the defender in conflict, not the receivers, usually high to low (low to high when a shallow crosser leads), and his progression runs on his feet: the first read as his back foot lands, the next on a hitch step, the checkdown last.
  • Rub routes beat man coverage by making defenders run through traffic; a receiver who deliberately blocks a defender more than a yard downfield, until the pass is touched, commits offensive pass interference.
  • Tags change one route in a concept; route conversions let a route change itself by coverage, so one call can be right against several defenses. The intermediate middle of the field is the most valuable place to throw, and most of these concepts exist to get the ball there.
