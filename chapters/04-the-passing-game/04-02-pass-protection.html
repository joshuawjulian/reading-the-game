def to(p, key, end, kind="block", via=(), delay=0.0, speed=None):
    """Move `key` through absolute (d, w) points `via` to `end` as a block, rush, path, route or run."""
    start = np.array(p._cur(key))
    pts = [tuple(np.array(q) - start) for q in (*via, end)]
    if kind == "block":
        return p.block(key, pts=pts, delay=delay, speed=speed)
    if kind == "rush":
        return p.rush(key, pts=pts, delay=delay, speed=speed)
    if kind == "route":
        return p.route(key, pts, frame="field", delay=delay, speed=speed)
    return {"path": p.path, "run": p.run}[kind](key, pts, delay=delay, speed=speed)


def timed(p, key, way, kind="path"):
    """Script a player for the frame strips: `way` = [(t, d, w), ...] absolute spots reached at time t
    (seconds after the snap). Standing still between two times is allowed."""
    cur = np.array(p._cur(key), dtype=float)
    tc, wait = 0.0, 0.0
    for t, d, w in way:
        nxt = np.array([d, w], dtype=float)
        dist = float(np.hypot(*(nxt - cur)))
        if dist < 1e-3:
            wait += t - tc
        else:
            p._add(kind, key, [tuple(cur), tuple(nxt)], speed=dist / (t - tc - 0.0), delay=wait)
            wait = 0.0
            cur = nxt
        tc = t
    return p

def clean(ax):
    """Box-zoomed diagrams: drop hash ticks and yard numbers, clip text to the window."""
    for ln in list(ax.lines):
        xs, ys = ln.get_data()
        if len(xs) == 2 and ln.get_linewidth() == 0.5 and np.hypot(xs[1] - xs[0], ys[1] - ys[0]) < 1.0:
            ln.remove()
    for t in list(ax.texts):
        if t.get_text().isdigit() and t.get_alpha() == 0.55:
            t.remove()
        else:
            t.set_clip_on(True)

def panel(ax, play, df, t, lateral, window, rings=(), hull=False, links=()):
    """One moment of a simulated play: markers, faint trails, rings, and optionally the pocket outline."""
    tmap = df.groupby("frameId")["time"].first()
    fid = int((tmap - t).abs().idxmin())
    ylo, yhi = sorted((play.ball_y - lateral[0], play.ball_y - lateral[1]))
    fld = Field(ax=ax, orient="vertical", los=play.los, window=window, numbers=False, hashes=False,
                lateral=(ylo, yhi), to_go=play.to_go)
    cur = df[df["frameId"] == fid]
    shown = cur[cur["x"].between(play.los + window[0] + 0.4, play.los + window[1] - 0.4)
                & cur["y"].between(ylo + 0.4, yhi - 0.4)]
    ids = set(shown["_id"])
    trails = df[(df["frameId"] <= fid) & df["_id"].isin(ids) & (df["side"] != "ball")]
    for (_, side), g in trails.groupby(["_id", "side"]):
        xs, ys = fld.xy(g["x"].values, g["y"].values)
        ax.plot(xs, ys, color=style.OFFENSE if side == "off" else style.DEFENSE, lw=0.9, alpha=0.4, zorder=3)
    if hull:
        draw_hull(fld, cur)
    ppy = fld.pts_per_yard()
    for _, r in shown.iterrows():
        x, y = (float(v) for v in fld.xy(r["x"], r["y"]))
        if r["side"] == "ball":
            continue
        lab = str(r.get("label", "") or "")
        if r["side"] == "off":
            ax.add_patch(Circle((x, y), MARK_R, fc=style.OFFENSE, ec="white", lw=0.6, zorder=8))
            tc = "white"
        else:
            ax.add_patch(RegularPolygon((x, y), 4, radius=MARK_R * 1.3, orientation=np.pi / 4,
                                        fc=style.DEFENSE_TINT, ec=style.DEFENSE, lw=1.0, zorder=8))
            tc = style.INK
        fs = max(4.0, min(7.5, ppy * (0.8 if len(lab) <= 1 else 0.55)))
        ax.text(x, y, lab, ha="center", va="center", fontsize=fs, color=tc, fontweight="bold", zorder=10)
    for key, col in rings:
        r = cur[cur["displayName"] == key].iloc[0]
        x, y = fld.xy(r["x"], r["y"])
        ax.add_patch(Circle((float(x), float(y)), RING_R, fc="none", ec=col, lw=1.7, zorder=7))
    for a, b in links:                      # dotted 'who has whom' lines between current positions
        pa = cur[cur["displayName"] == a].iloc[0]
        pb = cur[cur["displayName"] == b].iloc[0]
        xs, ys = fld.xy(np.array([pa["x"], pb["x"]]), np.array([pa["y"], pb["y"]]))
        ax.plot(xs, ys, color=GREEN, lw=1.5, ls=(0, (1, 1.4)), zorder=9)
    return fld


def strip(play, times, notes=None, lateral=(-9, 9), window=(-9, 6), ncols=4, rings=(), title=None,
          width_in=6.5, hull=False, final_links=()):
    """Numbered, captioned frame strip with a fixed box-sized window, so markers stay legible in print."""
    df = tracking(play)
    nrows = int(np.ceil(len(times) / ncols))
    pw = width_in / ncols
    span_w, span_d = lateral[1] - lateral[0], window[1] - window[0]
    fig, axes = plt.subplots(nrows, ncols, squeeze=False,
                             figsize=(width_in, nrows * (pw * span_d / span_w + (0.55 if notes else 0.25))))
    for ax in axes.flat[len(times):]:
        ax.axis("off")
    for i, (ax, t) in enumerate(zip(axes.flat, times)):
        panel(ax, play, df, t, lateral, window, rings, hull, final_links if i == len(times) - 1 else ())
        ax.set_title(f"{i + 1}.  " + ("snap" if abs(t) < 1e-6 else f"{t:+.1f} s"),
                     fontsize=8.2, color=style.INK_2, loc="left")
        if notes:
            ax.text(0.5, -0.02, notes[i], transform=ax.transAxes, ha="center", va="top",
                    fontsize=7.0, color=style.INK, linespacing=1.15)
    if title:
        fig.suptitle(title, x=0.01, ha="left", fontsize=11, fontweight="bold", color=style.INK)
    fig.tight_layout(h_pad=1.0 if notes else 0.5, w_pad=0.5)
    return fig


def video(play, times, notes=None, lateral=(-9, 9), window=(-9, 6), rings=(), width_in=5.4, hull=False):
    """Web version of the strip: every frame, with the strip's notes as running captions."""
    import shutil
    from matplotlib import animation
    from IPython.display import HTML, display
    df = tracking(play)
    span_w, span_d = lateral[1] - lateral[0], window[1] - window[0]
    fig, ax = plt.subplots(figsize=(width_in, width_in * span_d / span_w + 0.4))
    tmap = df.groupby("frameId")["time"].first()

    def update(fid):
        ax.clear()
        t = float(tmap[fid])
        panel(ax, play, df, t, lateral, window, rings, hull)
        txt = "snap" if abs(t) < 0.05 else f"{t:+.1f} s"
        if notes:
            cur = [n.replace("\n", " ") for tt, n in zip(times, notes) if tt <= t + 1e-6]
            txt += ("   " + cur[-1]) if cur else ""
        ax.set_title(txt, fontsize=8.5, color=style.INK, loc="left")
        return []

    frames = [f for f in sorted(df["frameId"].unique()) if float(tmap[f]) >= -0.3]
    anim = animation.FuncAnimation(fig, update, frames=frames, interval=100)
    html = (anim.to_html5_video(embed_limit=40).replace("<video ", '<video style="max-width:100%;height:auto" ')
            if shutil.which("ffmpeg") else anim.to_jshtml(default_mode="once"))
    plt.close(fig)
    display(HTML(html))

# --- the pocket as a shape: convex hull of the five linemen and the quarterback ---------------
def convex_hull(pts):
    """Monotone-chain convex hull of (x, y) points, counter-clockwise."""
    pts = sorted(set(map(tuple, np.round(np.asarray(pts, dtype=float), 4))))
    if len(pts) < 3:
        return np.array(pts)

    def cross(o, a, b):
        return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])
    lower, upper = [], []
    for q in pts:
        while len(lower) >= 2 and cross(lower[-2], lower[-1], q) <= 0:
            lower.pop()
        lower.append(q)
    for q in reversed(pts):
        while len(upper) >= 2 and cross(upper[-2], upper[-1], q) <= 0:
            upper.pop()
        upper.append(q)
    return np.array(lower[:-1] + upper[:-1])


def hull_area(h):
    x, y = h[:, 0], h[:, 1]
    return 0.5 * abs(np.dot(x, np.roll(y, -1)) - np.dot(y, np.roll(x, -1)))

Third-and-7, late in the first half. The offense lines up with five receivers spread across the field and nobody beside the quarterback. The defense walks two linebackers right up to the line, one on each side of the center, and a slot cornerback creeps toward the edge. You count seven defenders who might rush; the offense has five blockers. The center points and shouts a number. The quarterback points somewhere else and shouts a different word. Then the ball is snapped, the slot cornerback comes free off the edge, untouched, and the ball is already gone: a quick slant to the slot receiver, caught in the space where the cornerback used to be. Nobody blocked the man who came free, and that was the plan. This chapter is about why.

You’ll be able to:

  • Count rushers against blockers right after the snap, and say whether anyone was supposed to be free and who should have taken him.
  • Name the protection from what the backs and tight ends do: slide or man, five, six, seven or eight blockers, check-release or chip.
  • Explain why the offense names a “Mike” before the snap and what goes wrong when two blockers count from different men.
  • Read the pocket: which linemen give ground and which must not, what a vertical or jump set is for, and why pressure up the middle hurts most.
  • Explain how two linemen pass off a stunt, and why “stay with your man” loses to it.
  • Read protection numbers (pressure rate, time to throw, sack rate) and say why a sack is not just the offensive line’s fault.

How to read the diagrams. Offense is blue circles, defense orange squares, and left and right are always the offense’s. The five offensive linemen are labelled LT, LG, C, RG and RT, Q is the quarterback and R the running back; X, Z and H are wide receivers and Y (and U) tight ends. On defense, E is an end, and the two defensive tackles are labelled by technique: 1 for the one shaded on the center, 3 for the one on a guard’s outside shoulder. A line ending in a short bar is a block or a pass set (the spot a blocker moves to at the snap to meet his rusher; more on sets later), an arrow a path, a dotted green line “this blocker is responsible for that defender,” and a shaded green oval a gap a lineman owns. Rushers’ paths are orange; a red dashed arrow is a rusher nobody blocks. Protection is counting

Start with the arithmetic, because every protection in this chapter is an answer to it.

On a pass, the offense always has five blockers: the five offensive linemen, who are not allowed to go out for passes (they are ineligible) and so have nothing else to do. It may add more. A running back or a tight end can stay in to block, making six, seven or even eight. Every extra blocker is one fewer receiver, and that is the trade at the heart of the whole subject.

The defense, meanwhile, can rush anyone it likes. Most of the time it rushes four: in the 2024 and 2025 regular seasons about two dropbacks in three faced exactly four rushers, about one in five faced five, and fewer than one in ten faced six or more.1 Four is the default for the reason A First Look at Defense gave: four rushers leave seven defenders to cover at most five receivers. A blitz (sending a fifth, sixth or seventh rusher) flips that trade, giving up coverage to try to create a rusher the offense cannot block.

So the offense’s question on every pass is: how many might come, how many can I block, and if the answer is “not enough,” who is left over? Three ideas follow from it, and they organise the rest of the chapter.

1. The protection does not have to win forever, only long enough. A quarterback in today’s NFL gets the ball out in a little under three seconds on average (you will see the numbers at the end of the chapter). A rusher who arrives at 3.5 seconds has not hurt anyone. Protection is a race against a clock, not a wrestling match to the finish.

2. Block inside-out. Rushers from the middle of the line have the shortest path to the quarterback; rushers from the edges have to run around the blockers. So when an offense must choose who to block and who to leave, it blocks the inside man and leaves the widest one, because he has the longest way to go and the ball can be out before he gets there.

3. Someone left unblocked on purpose is the quarterback’s. If six come and the offense has five blockers, one rusher is unblocked by design. That man is called a free rusher: a rusher nobody is assigned to block. The protection does not fail when there is a free rusher; it hands him to the quarterback, who must throw before he arrives. A free rusher who was supposed to be blocked is a different thing: a bust, and you will learn to tell the two apart.

ImportantWhy it exists

Pass protection exists because the defense chooses its numbers after the offense has chosen its own. The offense commits its blockers in the huddle; the defense can show four rushers and bring six, or show six and bring three. Every protection scheme is a set of rules that decides, after the snap, which blocker takes which rusher, so that the most dangerous rushers are always blocked and any man left over is one the quarterback can beat with a quick throw.

The pocket: a cup, not a wall

Before we can say who blocks whom, we need the shape of what they are protecting. The pocket is the protected space the blockers form around the quarterback, and the best way to picture it is a cup, not a wall.

Three facts about the cup explain most of pass protection.

The quarterback throws from a spot, and the linemen know where it is. The launch point is where the quarterback sets his feet to throw at the end of his drop. From the shotgun he starts five yards deep and takes a short drop, so on a typical dropback he throws from about seven or eight yards behind the line; from under center a five-step drop ends at a similar depth and a seven-step drop deeper.2 Linemen set their depth to that spot. The aim is not to stop every rusher at the line; it is to keep every rusher off that spot until the ball is gone. When a play-action pass or a rollout moves the launch point, the whole line’s geometry changes with it, which is why those plays have their own protections (Play-Action, Bootlegs and Screens).

The interior must not give ground. Look at the distances in Figure 21.1. A defensive tackle rushing straight up the middle is about nine yards from the launch point. An end rushing around the tackle has to run an arc of about thirteen. So the guards and center set only a step or two back and then hold, because every yard they give is a yard straight into the quarterback’s face. Pressure up the middle is the most destructive kind: the quarterback cannot step up into it, cannot see over it, and cannot escape it by moving forward.

The tackles give ground on purpose. The tackles retreat several yards, backward and outward, keeping themselves between the end and the quarterback. If the end keeps running his arc, the tackle runs with him and pushes him past the launch point, deeper than the quarterback, where he is no threat. That is why a quarterback with a good pocket steps up: he moves forward into the cup as the edge rushers are pushed behind him. Stepping up is a skill the quarterback chapter builds on; for now, notice that it only works if the interior held. A quarterback who drifts backward instead runs into the ends the tackles were pushing past him.

CautionCommon misconception

“The offensive line should form a wall.” A flat wall at the line of scrimmage would be beaten at both ends in a second, because the edge rushers would simply run around it. The pocket is deliberately bent: firm in the middle, deep at the edges, with the quarterback inside the bend. When you see a tackle five yards behind the line with a defensive end running past him, that is usually the protection working, not failing.

Slide protection. In slide protection every lineman takes his first step the same way and owns the gap on that side, whoever comes through it. In the left panel of Figure 21.2 the line slides right: the left tackle owns the gap between himself and the left guard, the left guard the gap between himself and the center, and so on across to the right tackle, who owns the space outside himself. Notice that nobody on the line has to know in advance which defender will rush where. If the 1-technique rushes, he is in the left guard’s gap; if instead the Will linebacker blitzes and the 1-technique drops into coverage (defenses do this; see Zone Blitzes and Simulated Pressure), the left tackle’s gap gets the Will and the left guard’s gap gets nobody, and the left guard helps a neighbor.

The price of sliding one way is the man on the other end. Five linemen sliding right cover five gaps, from the left B gap to outside the right tackle, and that leaves the left end, outside everyone, uncovered. Someone else has to take him: the running back in the picture, or a tight end. That is the slide’s weak spot. The man it leaves behind is usually a defensive end, often the defense’s best pass rusher, and now a 215-pound back has him.

So the direction of the slide is a decision, and it follows one rule of thumb: the slide goes where the trouble is inside, and never leaves a star alone with a back. Offenses slide toward the most dangerous inside rusher, or toward the side with more possible rushers (an overload). If the defense’s best player is an edge rusher, they slide away from him and block him with the tackle man to man (the half-slide coming up next does exactly this), then add a chip (later in this chapter) on top. Help is always in addition to the tackle, never instead of him.

Man protection. In man protection, each blocker is given a specific defender before the snap. The usual rule is the one in the right panel: the five linemen take the four defensive linemen plus one linebacker, and the back takes the other linebacker. Line coaches call it “big on big” (BOB, or big-on-big), because the big men block the big men, and the back is left with a linebacker, a fairer fight than a defensive end.

Man protection matches strength to strength, and it handles a four-man rush cleanly. Its weakness is movement. If two defenders cross paths after the snap (a stunt, coming up later in this chapter), each man-blocker must chase his man through traffic. If a linebacker the line was counting on drops out and a different one rushes, the assignments are wrong for the rusher who actually came. Slide protection shrugs off both, because a gap does not move. That is the trade, and it is the same one zone and gap blocking make in the run game: man blocking wins the one-on-one matchups it planned for; area blocking survives the ones it did not.

Half-slide: the NFL’s everyday six-man protection

Most real protections split the difference. In a half-slide the center and the two linemen on one side slide that way and block gaps, the two linemen on the other side block man-to-man, and the running back takes the first extra rusher on the man side. This is the “60” protection from the play call you decoded in What a Play Really Is, and some version of it is the most common six-man protection in football. (Many playbooks number their protections in families, “50s,” “60s,” “90s,” with the number saying which way the slide goes and who blocks; others use words. What each number means differs from team to team.)

The half-slide is popular because it gives each part of the line the job it is best at. The slide side handles the movement defenses love to run inside (stunts, linebackers looping through gaps). The man side keeps the offense’s tackle on the defensive end, big on big, rather than handing that end to the back. And the back gets the defender with the longest path, an off-the-ball linebacker, whom he can meet in the hole.

Against the four-man rush in the picture, both linebackers dropped. The protection does not care: the center had nobody in his gap, so he “looks for work” and helps a neighbor (here the left guard on the 1-technique), and the back, whose man did not come, runs a route. Six blockers turned into five plus a receiver. That last move is called check-release, and it is one of the most important things a back does; it gets its own section below.

Why would anyone still call a full slide, then? Because it puts all five linemen on gaps, and gaps are the answer to movement. Against a front that loves stunts, or that stacks linebackers to come through the A and B gaps, a full slide toward the pressure means no lineman has to chase anybody; the offense accepts the cost (a back or tight end on the backside end) because the inside threat is the bigger one.

How to tell which one you are watching. On the broadcast, look at the linemen’s first step after the snap. All five step the same way: a slide. The center and the two linemen on one side step together while the other two set straight back at the man in front of them: a half-slide, with the slide toward the center’s step. Each lineman sets to a different defender, some stepping left and some right: man.

Mike identification: counting from the same man

Every protection above depends on everyone agreeing which linebacker is which. The half-slide said “the back takes the Will,” and the man protection “the linemen take the four linemen and the Mike.” But the defense does not wear name tags, and linebackers move. So before the snap one player makes the call you heard in The Pre-Snap Phase: “54’s the Mike!” That call is Mike identification, often written MIKE ID. In most offenses the quarterback has the final word, because he sees the whole defense; in others the center, who is closest to the front and runs the line’s communication, makes it.3

How the count works

The “Mike” in the call is a reference point, not necessarily the defense’s real middle linebacker. Every assignment is written relative to him, so naming him turns six separate judgments (“which one is mine?”) into one decision that everyone follows. Systems differ in the details, but one common way to build a six-man protection around the call, the one in Figure 21.4, works like this.

  1. The center works toward the Mike. He and the two linemen on that side slide that way and own the gaps there, so the Mike (call him 0) and anyone on his side belong to the slide.
  2. The other two linemen count outward from the Mike on their side. The guard takes the first defender on the line (1), the tackle the second (2). This is the man side.
  3. The back takes the next threat beyond the line’s count (3), usually the linebacker on the man side. If he rushes, the back blocks him; if he drops, the back releases. In many playbooks the back has two men to check in order, inside first (a dual read), and takes whichever comes.

The cost of all this is time and noise: every call has to be heard by five or six men, on the road, with the play clock running.

The right panel also answers a question about the back. In most half-slides he lines up on the man side, the side he will check, so he has a short trip. Figure 21.5 shows what happens when the count is not shared: the defense rushed only five men against six blockers and still got a man free, because two blockers counted from different starting points. This is the classic bust, and when you see a linebacker run untouched through the middle against a normal-looking rush, it is the first thing to suspect. Some quarterbacks make this part of their game. Tom Brady was known for it: his Mike calls are audible on broadcasts, and teammates describe him overruling the line’s identification at the line of scrimmage (“no, he’s the Mike”) to move the protection toward a matchup he did not like, then getting the ball out to the hot receiver before a free rusher arrived.4 The skill is invisible on most broadcasts, and it is part of why analysts treat sack rate as partly a quarterback statistic, not only a line statistic.

Defenses know this, which is why they move after the call. Linebackers walk up into the A gaps at the last second, safeties rotate down, a lineman stands up and drops back. Each move is an attempt to make the offense’s count wrong after it is too late to fix. The two-linebacker version, both standing in the A gaps on either side of the center, is the “double A-gap mug” of Blitz and Man Pressure, and it exists almost entirely to attack the Mike call: with both linebackers threatening the middle, which one is the Mike? Here, in miniature, is why the pre-snap phase looks like the busiest few seconds in sport.

NoteMadden vs. real life

In Madden you can tap a button to “slide protection left” or tell the back to block, and the computer linemen then know exactly who to block. Real linemen have to agree on it, out loud, in a few seconds, in a stadium built to drown them out, while the defense rearranges itself to make their agreement wrong. Most of what makes an NFL offensive line “good” in the eyes of coaches is invisible on a broadcast: not strength, but five players reaching the same answer at the same time, snap after snap.

Three details in Figure 21.6 are worth noticing, because they are what coaches grade backs on.

His eyes go to the linebacker, not the quarterback. At the snap the back is looking at the Will. He has a fraction of a second to decide, and a back who looks at the quarterback first, or at the ball, will be late.

He steps up to meet the blitzer. The back attacks forward into the gap rather than waiting for the linebacker to arrive at the quarterback’s feet. Meeting him in the hole, a yard or two behind the line and well in front of the quarterback, keeps the collision away from the launch point and gives the linebacker less room to run around him.

He takes on a bigger, faster-moving man. A linebacker on a blitz may outweigh the back by 20 pounds and is running at full speed; the back is standing nearly still. That is why “can he pass protect?” is one of the first questions asked about a rookie running back, and why a back who can’t is often taken off the field on third down, however well he runs.

The other half of the rule, the release, is what makes the six-man protection efficient. A back who stays in to block nobody has wasted a receiver. With check-release, the offense gets a blocker when it needs one and a receiver when it doesn’t, and it finds out which only after the defense has committed. Tight ends play by the same rule when they are part of the protection.

Two refinements, both about order. When a back has more than one man to check (the dual read above), he looks inside first, because the inside blitzer has the shorter path. And when the play also asks him to chip, the order is fixed: check, then chip, then release. The blitz comes first, because a blitzer is a free man and an end already has a tackle on him.

The chip: help on the way out

Between “block” and “release” there is a third option. A chip is a quick shove by a tight end or back on a pass rusher, usually an edge rusher, delivered on the way out into his route. It is not a block. It is a hit, a second at most, aimed at knocking the rusher off his path or breaking his rhythm, so that the tackle can catch up with him.

The chip exists because of the slide’s weak spot and the defense’s best player. When a defense has one edge rusher the tackle cannot handle alone, the offense has three choices: leave the tackle on an island and hope, keep a tight end in to block the rusher full-time (and lose a receiver), or chip him (and lose a beat on the route). The chip is the cheap version of help. Its cost is timing: a receiver who chips is late into his route, so chips go with routes that do not need to be on time, like a back’s checkdown to the flat (the short, safe throw a quarterback makes when nothing downfield is open) or a tight end’s route underneath. And the chip comes from the inside, knocking the end wider and deeper, the direction the tackle is already pushing him; a chip that drives him inside would push him straight at the quarterback.

More blockers, fewer receivers: max protect

Add a tight end to a six-man protection and you have a seven-man protection: the five linemen, the back and a tight end. It usually stays flexible. The tight end and back check, chip and release, so three receivers are out on time and one or two more arrive late, and its most common job is the one you just saw: putting help on a dangerous edge rusher.

At the far end of the counting spectrum is max protect. The difference is not only the number but the commitment: seven or eight blockers who stay in to block whatever the defense does, no check-releases, with only two or three receivers in the pattern. It is the offense’s way of buying the most time it can, and it nearly always travels with a deep throw.

Max protect exists because deep routes take time. A 20-yard post needs the quarterback to hold the ball for three seconds or more, and against a good rush that time has to be bought with bodies. It pairs naturally with play-action (a fake handoff before the pass), because the heavy formation and the extra blockers look exactly like a run, which freezes the linebackers and safeties for the half-second the deep routes need.

The trade is stark, and Figure 21.8 puts numbers on it. With eight blocking and five rushing, the quarterback has all the time he wants. But two receivers against six defenders in coverage means double coverage somewhere, and nowhere to go if both are covered except the sideline, a scramble or a sack. Max protect is a bet that one of two receivers will beat his man deep. It is a good bet against a defense that has to respect the run and a bad one against a defense that just drops eight.

Fewer blockers, more receivers: scat and empty

At the other end are protections that block with five, or as close to five as possible, so that more players can run routes.

In empty protection, the offense lines up with nobody in the backfield except the quarterback (the “empty” formations of Spread and Modern Formations), so the five linemen are the whole protection. Many teams block it man-to-man, the four defensive linemen plus the Mike; others run it as a five-man slide or half-slide. Either way, any sixth rusher is the quarterback’s problem.

In scat protection, the back lines up in the backfield, so the defense has to respect him as a blocker, but he releases into his route at the snap instead of checking a blitzer first. The line blocks, and as in empty, an extra rusher is the quarterback’s.5 Teams use the word in different ways (in some playbooks “scat” is one numbered protection, in others a tag that can be added to several), so treat it as “six in the picture, five in the protection.”

Both exist for the same reason: five receivers stretch a defense further than four, and the quickest throws come out before an extra rusher could arrive anyway. The quick game of 04-03 is built on this: a ball out in a second and a half makes a sixth rusher irrelevant. And both carry the same obligation. If the defense sends one more than the protection can block, somebody has to get the ball out, now.

When rushers outnumber blockers: hot routes and sight adjusts

That obligation is the last piece of the protection puzzle. When six rush and five block, the protection has done its job if the man left over is the right man, the one with the longest path, and the rest of the answer belongs to the quarterback and his receivers.

A hot route is the throw built into a play for exactly this moment: when a rusher comes that the protection cannot block, the quarterback throws at once to a receiver running into the area that rusher just left. The logic is simple and a little beautiful. A defender who rushes is a defender who is not covering, so the space he came from is, for a second or so, empty. In Figure 21.9 the nickel back was the man over H; when he blitzes, H’s quick inside break runs straight into the hole. The quarterback does not have to read the coverage; he has to recognise the blitz and be on time.

That is why “who’s hot?” is part of the design of most dropback passes, not something made up on the field. The play call and the protection together decide it: once the offense knows which rusher it will leave unblocked if everyone comes, it knows which receiver’s route has to be ready early. Usually the hot receiver is the man whose defender is most likely to blitz (a slot receiver against a slot-cornerback blitz, a tight end against a linebacker blitz). The quarterback’s pre-snap job is to see whether the hot is live on this snap; a play whose protection can block everyone who might come needs no hot at all.

Coaches decide where the leftover man comes from, too, not only how many. Inside-out settles the count; the direction of the slide settles the side. Given the choice, they would rather leave the free rusher where the quarterback can see him, on the side he is looking to throw, than coming from behind his back.

A sight adjust is the receiver’s half of the same idea. The receiver sees the blitz too, and changes his route after the snap: a receiver running a deep route against a defender who has just blitzed cuts it off short, stopping and turning back to the quarterback or breaking inside, so the ball can come out immediately. The quarterback, reading the same blitz, expects it. Teams use the two words differently. Some call any blitz-beater “hot,” some reserve “hot” for a designated route and “sight” for a converted one. What matters on the broadcast is the shape: a quick throw, often to a slot receiver or tight end, into the space a blitzer just vacated.

NoteMadden vs. real life

In Madden a “hot route” is something you do before the snap: you press a button and change one receiver’s route to a slant or a streak. Real offenses do change individual routes before the snap, with tags and signals, but that is not what coaches mean by “hot.” In football a hot route is the answer to an unblocked rusher: a throw the quarterback makes because someone is coming free, to the space that rusher left. It is decided by the defense’s choice, after the snap, not by a menu.

Hot throws explain a strange-looking fact you will start to notice: some of the most dangerous blitzes are the ones that produce quick completions, not sacks. A six-man blitz that the offense has blocked correctly, inside-out, leaves the quarterback a free rusher and a hot receiver. If he sees it, he completes a slant for eight yards; if he doesn’t, he is sacked from his blind side and the announcer blames the tackle, who did nothing wrong. The lineman’s half: sets, punch and anchor

Schemes decide who blocks whom. Technique decides whether he can. Pass blocking is a strange athletic task: a 315-pound man moving backward, against a man who is moving forward, faster and with a head start, who knows where he is going. The offensive lineman’s advantages are that he knows where the quarterback will be and that he only has to stay in the way for about three seconds. Everything in his technique is built on those two facts.

The set: kick slide, vertical, 45-degree, jump

At the snap, a pass blocker sets: he rises out of his stance and moves to the spot from which he will meet his rusher. For a tackle facing a fast edge rusher, the footwork is the kick slide. The outside foot kicks back and out, the inside foot slides after it, and the feet never cross, so he can move backward and sideways while his shoulders stay square to the line and he can still change direction. A tackle who crosses his feet, or turns his hips to run, has handed the rusher the inside path to the quarterback.

Where the kick slide takes him is a choice, and the choice is a trade between time and space.

A vertical set kick-slides almost straight back, giving ground fast. It is the answer to a speed rusher: the deeper the tackle gets, the longer the arc the rusher must run around him, and the more likely the rusher ends up behind the quarterback. Its risk is the inside. A tackle who has bailed straight back has left room between himself and his guard, and a good rusher will fake the arc and come back underneath into that space.

A 45-degree set goes back and out at an angle. It meets a wide rusher sooner and with less ground given, at the cost of a little depth. Many tackles mix the two on purpose, so the rusher cannot time his moves to one picture.

A jump set is the aggressive one: instead of retreating, the blocker moves forward or flat to meet the rusher at or near the line of scrimmage. It takes away the rusher’s running start, the few yards he uses to build up speed before his move, and it gives the quarterback clean throwing lanes on quick passes. It is also a gamble: a lineman who lunges and misses has nobody behind him, and the rusher’s path to the quarterback is now short. That is why jump sets travel with the quick game, when the ball will be gone in a second and a half anyway, and why good tackles sprinkle them into longer dropbacks too, so rushers never know what is coming.

Interior linemen set too, but differently: they have no room to give, so their sets are short, a step or two, and their battles are decided almost immediately by hands and leverage.

Punch and anchor

Once he has set, the blocker has to win the contact. The punch is his weapon: a short, violent, two-handed strike into the rusher’s chest, inside the rusher’s arms, to stop his momentum and take control of his body. Timing is everything. Punch too early and the rusher swats the hands away and runs past; too late and the rusher’s hands are already on the blocker’s chest. That is why many NFL tackles no longer throw one textbook two-handed punch against every rusher: against men who swipe and chop at the hands, they “catch” the rusher and strike late, or use one hand at a time, so there is nothing early to swat. Hands inside matter for a second reason: the rules. A blocker may first contact a rusher on or outside his frame (his body below the neck, facing the blocker), but he must work immediately to get his hands inside it, and using hands or arms to grab, hook or turn the rusher, restricting his path, is holding, wherever the hands are.6 Hands outside the frame are where a restriction is easiest to commit and easiest for an official to see. That is why linemen spend so much practice time on “inside hands”: it is the safest place to win from.

The punch is only possible because of a rule change. Until 1978, pass blockers in the NFL had to keep their hands close to their bodies; that season the league allowed them to extend their arms and open their hands, one of a pair of rule changes meant to open up passing.7 Before it, linemen pass-blocked with their forearms and elbows, more like a wall of shoulders than a pair of arms. The change made the modern pass set, and the modern passing game, possible. The 1978 Liberation tells that story.

The anchor is the answer to power. A bull rush (the rusher driving straight through the blocker’s chest) does not try to get around the blocker; it tries to push him backward into the quarterback. To anchor, the blocker drops his hips, widens his base and resets his feet so that the push stops. A guard who cannot anchor is the most dangerous weakness a line can have, because the pocket collapses from the middle, exactly where Figure 21.1 said it must not.

CautionCommon misconception

“Pass blocking is about strength.” Strength matters most in the anchor, but most pass-blocking losses are losses of position or timing: a set too shallow or too deep, a punch thrown too early, feet that stopped moving. That is why some of the best pass-blocking tackles are not the strongest men on their line, and why a rusher’s best moves, the ones The Pass Rush describes, are aimed at the blocker’s hands and feet, not his strength.

Stunts: passing them off

The defense has its own way to beat protection without adding rushers: make two of them swap jobs. In a stunt (also called a twist or a game), two rushers exchange gaps after the snap. The most common is the T-E stunt (tackle first, end second): the defensive tackle, here the 3-technique, goes first, crashing outward into the offensive tackle, and the end loops around the back of him, on the defense’s side of the line, into the space he left inside. (Its mirror, the E-T stunt, sends the end inside first and loops the tackle around the outside.) The first man is the penetrator; his job is to occupy, or even knock over, the blocker the looper needs to get past. The second is the looper.

The offense’s answer is the pass-off: two adjacent blockers handing the rushers to each other. There are two ways to do it, switch and stay, and the animation shows the one coaches usually teach against a clean twist.

    p = Play(line() + gun(None), four_down(lbs=False), los=35, to_go=None)
    timed(p, "QB", [(0.9, -7.4, 0.0)])

    timed(p, "NT", [(0.5, 0.0, -0.9), (1.0, -0.5, -1.0), (1.6, -0.8, -1.0)], "rush")
    timed(p, "LG", [(0.5, -1.3, -1.6), (1.0, -1.7, -1.4), (1.6, -2.0, -1.3)], "block")
    timed(p, "C", [(0.5, -1.4, 0.1), (1.0, -1.8, 0.2), (1.6, -2.1, 0.3)], "block")

    timed(p, "DT", [(0.4, 0.0, 3.0), (0.8, -0.7, 3.6), (1.1, -1.1, 3.9), (1.4, -1.35, 4.0), (1.8, -1.45, 4.0)], "rush")

    loop = [(0.4, 0.1, 5.7), (0.8, 0.7, 4.7), (1.1, 0.5, 3.1), (1.4, -0.4, 2.2)]

        timed(p, "RG", [(0.4, -1.2, 1.9), (0.8, -1.6, 2.0), (1.1, -1.9, 2.0), (1.4, -2.1, 1.9), (1.8, -2.3, 1.9)], "block")

    else:

        timed(p, "RG", [(0.4, -1.2, 2.1), (0.8, -1.8, 2.6), (1.1, -2.1, 2.9), (1.4, -2.3, 3.0), (1.8, -2.4, 3.0)], "block")

Switch means each blocker keeps his area and takes whoever comes into it. In Figure 21.11 the guard starts on the 3-technique, feels him crash outside, and lets him go to the tackle; the tackle starts on the end, sees him loop away inside, and lets him go to the guard. Each lineman ends up on the other’s man. The key coaching point is that the two blockers must be on the same level (the same depth), “hip to hip.” If the guard has set deeper than the tackle, or shallower, the penetrator slips between them before the hand-off happens. Line coaches add two cues: the tackle squeezes the penetrator, flattening down to meet him so he cannot split the seam, and the guard gives him a push toward the tackle as he leaves and turns his eyes inside at once, because the looper is coming next.

Stay means each blocker follows the man he started on, wherever he goes. Against a stunt, that usually loses, and the static picture shows why.

The stunt is designed to beat “stay.” The penetrator’s crash outside drags the guard with him and puts two bodies in the tackle’s path, and the looper simply walks around the traffic. That is why most line coaches teach switching against a clean two-man twist. Stay is not useless: if one blocker is already locked onto his man and winning, letting go to switch can free the rusher he was beating, and some games are built to punish a switch, with the looper waiting a beat (a delayed loop) so that the blockers trade too early. The decision is made in a split second by feel, and it is a large part of why linemen who have played together for years protect better than five talented strangers.

Notice, too, which scheme makes this easier. In a slide, every lineman already owns a gap, so a switch is the default: the looper comes into the guard’s gap, so he is the guard’s. In man protection the blockers have to recognise the stunt and decide to switch. That is one reason offenses facing stunt-heavy fronts lean on slide protection. Where a stunt aims matters too. The T-E game in the figures attacks the guard-tackle seam; its mirror, the E-T, sends the looper toward the guard-center seam, which is where a slide is strongest, because the center and guards are all already stepping the same way. So defenses choose their games against the protection they expect, and stunting is one of the main tools a four-man rush has for creating pressure without blitzing.

Linemen do talk through all of this. Before the snap they warn each other about alignments that suggest a stunt, and during the rush a switch is often called out loud. On a broadcast you will almost never hear it; you see it in the hand-off.

The pocket over time

Everything so far has been about the first second or two. A real dropback lasts longer, and the pocket is not a fixed shape; it breathes. It widens as the linemen set, holds for a while, and then shrinks as the rushers win ground. Tracking data lets us draw that as a shape: join the five linemen and the quarterback with the smallest polygon that contains them all (a convex hull), and you have a rough outline of the pocket, frame by frame.

The Big Data Bowl files this course uses (13-05) record every player’s position ten times a second. We have not downloaded those files, so the figure below uses a simulated play from the gridiron library, written out in exactly the Big Data Bowl format; the code to run the same measurement on a real play follows it.

Measuring protection: pressure, time to throw and sacks

Now the numbers you will see on broadcasts and stat sites, and what each of them can and cannot tell you.

A sack (defined in Downs, Distance, and Field Position) is the number everyone knows: the quarterback tackled behind the line before he throws. Myles Garrett set the single-season record with 23 of them in 2025.8 But sacks are rare (about six or seven per hundred dropbacks) and they depend on many things besides the blocking, so analysts lean on broader measures.

Pressure is any rusher forcing the quarterback to change what he wanted to do. It comes in three kinds: the sack; the QB hit, contact that knocks the quarterback down or hits him as he throws; and the hurry, which forces him to throw early or move off his spot without contact. Pressure rate is pressures divided by dropbacks. It is the better measure of a protection (or a pass rush) because there are four or five pressures for every sack, so it settles down faster and is less a matter of luck: whether a pressure becomes a sack depends a great deal on the quarterback (The Pass Rush looks at it from the defense’s side).

There is one catch. Pressure is a judgment, and different data providers judge it differently. Some chart it by eye from video; others use tracking data and a distance rule. The nflverse data this course uses carries a pressure flag charted by FTN for recent seasons and by Next Gen Stats for earlier ones, and the two are not the same series.9 (FTN is a football-data company whose staff chart every play from video; Next Gen Stats is the league’s own tracking-data service.) Compare pressure rates only within one source.

Time to throw is the seconds from the snap to the release of a pass, as measured by Next Gen Stats; their average leaves sacks out.10 It tells you how long a quarterback holds the ball, which is partly the protection, partly the play call, and partly the quarterback. Pressure rate and time to throw are tied together, and the next chart shows how tightly.

The ball is coming out a little later than it used to. The league-wide average time to throw rose from 2.64 seconds in 2016 to 2.84 in 2025, while the league sack rate stayed within a point or so (5.6% of dropbacks in 2016, 6.5% in 2025). Quarterbacks hold the ball a fifth of a second longer than they did a decade ago and are sacked only a little more often, which says something about how protection, and quarterbacks’ pocket movement, have kept up.

Holding the ball explains only a little of who gets sacked. Across all team-seasons the correlation between time to throw and sack rate is 0.29, positive but weak; the trend line says each extra tenth of a second goes with about 0.3 more sacks per hundred dropbacks. Within 2025 alone it is 0.05, essentially nothing. Two teams can hold the ball equally long and be sacked at rates three times apart. What separates them is the quality of the protection, the play calls, and above all the quarterback: his sense of when to throw it away, when to step up, and when to run.

Sacks are shared. Follow the arrow. In 2024 the Bears’ quarterback, Caleb Williams, was sacked 68 times, a franchise record, on 9.9% of dropbacks. In 2025, with a new head coach and three new interior linemen, he was sacked 24 times (3.7%) while holding the ball longer: 3.19 seconds on average, the longest of any team that season, up from 2.92.11 The new linemen were the guards Joe Thuney (traded from Kansas City) and Jonah Jackson (from the Rams) and the center Drew Dalman (signed in free agency), all in the interior, the part of the pocket Figure 21.1 said must never give ground. Same quarterback, more time, a third as many sacks. Some of that is the line, some the scheme (Ben Johnson was the new head coach), some a second-year quarterback learning when to get rid of the ball. The data cannot split the credit, and neither can anyone who blames every sack on the left tackle. If you rewatch a 2025 Bears dropback next to a 2024 one, watch the interior: do the guards and center hold their ground, so that Williams can step up and throw, or step up and run?

CautionCommon misconception

“Sacks allowed measures the offensive line.” It measures the whole passing operation. A quarterback who holds the ball waiting for a deep route, refuses to throw it away, or drifts into the rush creates sacks no line could prevent, and a quarterback with a quick trigger hides a weak one. Analysts who want to grade the line alone use measures built from individual matchups, such as pass-block win rate or charted pressures allowed per blocker, and even those are noisy. When you hear “the line gave up five sacks,” ask how many of them came after three seconds.

Where protection came from

Pass protection as a system, with named schemes, counted assignments and a back with a rule, is younger than you might think, and two moments explain most of its shape.

The first is the 1978 rule change you met above. Before it, linemen could not extend their arms on a pass block, so protection was a matter of bodies in the way: a pass rusher who got his hands on a lineman had most of the advantages, and passing on long downs was dangerous. Letting blockers use their hands, together with new limits on how defenders could contact receivers, tilted the game toward the pass for good.12

The second is a player. Lawrence Taylor, the New York Giants’ outside linebacker from 1981, rushed the passer from a standing position off the edge, faster than any tackle could handle alone, and from a spot that standard protections assigned to a running back. Offenses had to invent ways to put a bigger man on him. Joe Gibbs, whose Washington teams played the Giants twice a year, is credited with building his one-back, two-tight-end formations partly to do exactly that: an extra tight end or an “H-back” lined up beside the tackle to block or chip Taylor, so that the back was never left alone with him.13 The chip, the help side and the seven-man protection, and the idea that the left tackle protecting a right-handed quarterback’s blind side is a premium position, all trace back to offenses deciding that one great rusher had to be accounted for before anyone else.

Since then the arms race has run in both directions. Defenses answered slide protections with stunts and with pressure that showed six and rushed four; offenses answered blitzes with hot routes and the quick game, and with quarterbacks and centers who could reset a protection at the line. The Mike call you hear on every broadcast is the current front of that race.

NoteFilm room: Cincinnati Bengals 2022 — rebuilding a line, and three backups in Buffalo

The problem. In 2021 Joe Burrow was sacked a league-high 51 times in the regular season and another 19 times in the playoffs, including nine in the divisional round against Tennessee and seven by the Rams in Super Bowl LVI, which tied the Super Bowl record.14 The Bengals went to the Super Bowl anyway, and lost it with their quarterback on the ground.

The rebuild. That offseason Cincinnati signed center Ted Karras, guard Alex Cappa and tackle La’el Collins, and drafted guard Cordell Volson: four new starters around Burrow.15 The team’s sack rate fell from 8.7% of dropbacks in 2021 to 6.4% in 2022.

The test. By the 2022 playoffs, injuries had taken three of the five starters, and in the divisional round at Buffalo on January 22, 2023, the Bengals started three backups: Jackson Carman, a second-year player whose six previous starts had all come at right guard, at left tackle; Max Scharping at right guard; and Hakeem Adeniji at right tackle. They won 27–10 and Burrow was sacked once and hit only three times.16 It is a good game to rewatch with this chapter’s tools: look for help on the backup tackles from tight ends and backs, and for how quickly the ball comes out.

The limit. A week later, in the AFC Championship Game at Kansas City (January 29, 2023), the same patched line gave up five sacks, two of them by defensive tackle Chris Jones, the second on third down on Cincinnati’s last possession, which forced the punt before the Chiefs’ winning drive; rookie end George Karlaftis had the first, on the game’s fifth play. Kansas City won 23–20.17 Schemes can hide a weak spot for a while. A great rusher, given enough snaps, finds it.

NoteFilm room: Philadelphia Eagles 2013–2025 — the Stoutland line

Jeff Stoutland coached the Eagles’ offensive line from 2013 through the 2025 season, a run that included both of the franchise’s Super Bowl titles (LII and LIX), and he was regarded as one of the best offensive line coaches in the league.18 His lines were known for athletic, technically precise linemen (Lane Johnson at right tackle, Jason Kelce and then Cam Jurgens at center, Jordan Mailata at left tackle) and, in Kelce’s years, for a center who made most of the protection calls; after Kelce retired, Jurgens and quarterback Jalen Hurts shared that job.19 The Eagles are also a reminder that a great line and a great run game go together: the same men who pass protect also block for the backs and for the tush push. Watch for it on Sunday

TipWatch for it

Before the snap.

  • Count the blockers you can see. Five linemen, plus a back beside the quarterback, plus any tight end on the line. Empty backfield means five-man protection: someone is hot.
  • Count the threats. How many defenders are standing within a couple of yards of the line? Six or seven at the line against five or six blockers means a hot throw is in the plan, or the defense is bluffing.
  • Listen and look for the Mike call. The center or quarterback points at a linebacker. If a defender moves after that, watch for a second call.
  • Find the help. A tight end next to a tackle may mean that side’s edge rusher is getting a chip. A back offset to one side more often shows his check side, the side he will watch for a blitzer (usually the man side, away from the slide), and maybe chip after. Neither is a guarantee: a late call can send him across the quarterback’s face.

After the snap: rushers versus blockers.

  • Watch the linemen’s first step. All five the same way: slide. The center and his two neighbours one way, the other two straight back at the man in front of them: half-slide. Each to a different defender: man.
  • Count, in the first second, how many rushed and how many blocked. Four rushers against six blockers? Look for the spare men helping. Six against six? Everyone has a man.
  • Watch the back’s first step. Forward into a gap: he has a blitzer. Out to the flat: his man dropped (check-release). A shove on an end and then out: a chip.
  • Watch the tackles’ sets. Deep and backward (vertical), angled out (45-degree), or forward at the line (jump, and expect a quick throw).
  • If a rusher comes free, guess who owned him. Was he the widest man, unblocked on purpose, with the ball out fast to the space he left? That was the plan. Was he an inside man, a linebacker through an A gap, with two blockers on someone else? That was a bust, probably a bad count. Check the replay. And look at the ball: a free man by design should be met by a quick throw. If the widest man came free and the ball was not out, the quarterback or the hot receiver missed it; if an inside man came free, suspect the protection first.
  • Note the time. Count “one-Mississippi” from the snap. Pressure inside about two and a half seconds is almost always the protection’s problem: the average throw is out by then. Past three seconds, where the EPA chart above peaked and started to fall, the quarterback owns part of it. These are rules of thumb from that chart, not laws.

For your prediction log: for one quarter, write down rushers and blockers on every dropback (for example, “4 v 6”) and whether anyone came free. By the end you will know whether that defense wins by numbers or by stunts.

How many can the offense block, how many might come, and where should the ball go if they all do?

TipAnswer

Five blockers, up to seven rushers, so the ball must come out fast, with a hot answer on each side. Count it: four defensive linemen, two linebackers in the A gaps and the walked-up nickel make seven possible rushers against an empty protection’s five. If all seven come, two are free; there is no protection that blocks seven with five.

Which two is the offense’s choice. Inside-out, the five linemen take the two A-gap linebackers and the two 3-techniques first, because their paths are shortest, and the fifth blocker goes to one of the ends. The offense picks the end by working the protection toward the side with more threats. The left has four possible rushers (the Will, a 3-technique, the end and the nickel), the right only three, so the line works left and the left tackle takes the left end. The two men left over are then the widest on each side: the nickel on the left and the right end on the right, each with the longest way to go.

Two free rushers need two answers. On the left, H is hot: if the nickel comes, H runs into the space he left. On the right, count the coverage as well as the rush, which is the first thing a quarterback coach asks. If seven rush, only four defenders are left to cover five receivers, so someone is uncovered; as drawn, nobody is over R. The quarterback’s throw depends on who actually comes, and it has to be out in about a second.

Two cautions. First, this is exactly the look that attacks the Mike call: with two linebackers in the A gaps, the offense must decide before the snap which one the line will count, and the defense will try to make him wrong. Second, the defense may not bring them all. Showing seven and rushing four, with the A-gap linebackers dropping out, is a staple of simulated pressure. Then the hot throw is thrown into a linebacker who never left. Good quarterbacks check where the threat goes at the snap, not where it stood.

(b) Now a post-snap read. Suppose that instead the Mike comes through the right A gap untouched, while the center has turned left to help the left guard on the 1-technique. Was the Mike free by design, or was it a bust? And whose was he?

TipAnswer

(a) The back takes the Will; the nickel is the quarterback’s, and the throw goes to H. On the left, the man side, the left tackle has the end and the left guard has the 1-technique. That leaves two potential rushers, the Will and the nickel, and one back. Block inside-out: the Will is closer to the quarterback’s path, so he is the back’s; the nickel, wider and deeper, has the longer way to go and is left free by design. His blitz empties the area over H, so H is the hot receiver, and the quarterback has to see the nickel come and throw at once.

The offense had other options before the snap: the quarterback could have re-declared the protection, sliding the line left toward the threat and sending the back right, or kept the tight end in. Notice why it might not bother: the right side has only three rushers for three sliding linemen, and a quick throw to H against a blitzing nickel is often the best play on the field. If the nickel doesn’t come, the back’s man (the Will) still might, and if neither comes, the back releases and five receivers are out.

(b) A bust, and the Mike was the center’s. In a half-slide right, the center owns the right A gap, whoever comes through it. He may “look for work” and help a neighbour only once nobody has come into his gap; turning left first left his own gap empty. Every tell in the “Watch for it” list points the same way: the free man came from the inside, through an A gap, the shortest path, while two blockers were on one defender. A free man by design is the widest man, and the ball is already on its way to the space he left.

Where is the protection’s help going, and what does the offense give up to send it?

TipAnswer

99 belongs to the right tackle, and the help comes on top: chips from the tight end and the back. The price is receivers and time. Everything about the offense’s alignment points at the right edge. Expect a half-slide left: the center, left guard and left tackle slide left and own the gaps on that side (the 1-technique, the Will and the left end), while on the man side the right guard takes the 3-technique and the right tackle takes 99 man to man, big on big. Then the help. The tight end chips 99 from the inside as he releases. The back, offset to that side, checks his linebacker (the Mike) first; if the Mike doesn’t come, he chips 99 too and releases (check, chip, release). If the offense wants a full-time double team instead, it is the tackle and the tight end together, a seven-man protection.

What the offense will not do is leave 99 to the tight end and the back. You never leave a star alone with a back; help means in addition to the tackle.

The cost is visible too. With two players spending time on 99, only three receivers (X, H and Z) are out on time, and the tight end and back arrive late, so expect a throw to the receivers who are on time, or a two- or three-man route combination with lots of protection behind it. The defense’s answer is just as predictable: move 99 around (inside, to the other side, standing up over a guard) or run stunts with him, so that the offense’s help is in the wrong place. That cat-and-mouse is where The Pass Rush picks up the story.

Takeaways

  • Protection is counting. The offense always blocks with five and may add a back or tight ends; every added blocker is one fewer receiver. Block inside-out, and if someone must go unblocked, make it the widest man and make him the quarterback’s.
  • Two ways to assign blockers. Slide protection blocks gaps, so it absorbs stunts and surprises but leaves an end for the back. Man protection (BOB) blocks men, so it matches big on big but has to chase movement. The half-slide, the everyday six-man protection, does both. Slide toward the inside trouble, and never leave a star edge rusher alone with a back.
  • The Mike call makes six blockers one mind. Naming a reference defender sets every assignment (the center works toward him, the man side counts out from him, the back takes the next man); two blockers counting from different men produce a free rusher without the defense sending an extra man. Defenses move after the call to make it wrong.
  • Backs and tight ends decide after the snap. Check-release turns a blocker into a receiver when his man doesn’t come; a chip gives help on the way out. Max protect buys time with up to eight blockers; empty and scat buy receivers with five, and a blitz means a hot throw or a sight adjust into the space the blitzer left.
  • The pocket is a cup. The interior holds firm; the tackles give ground with a kick slide (vertical, 45-degree, or a jump set for quick throws) and push the edge past the quarterback, who steps up. Punch wins control, the anchor stops power, and switching beats stunts.
  • Measure pressure, not just sacks, and share the blame. Pressure rate is steadier than sack rate; pressure climbs with time to throw; and sack rate depends on the quarterback and the play call as much as on the line.
