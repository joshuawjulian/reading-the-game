def O(key, pos, d, w, label=None):
    return Player(key, pos, "off", d, w, label=label or key)


def D(key, pos, d, w, label):
    return Player(key, pos, "def", d, w, label=label)


def line5():
    """The five offensive linemen, 1.6 yards apart, each labelled so the reader can follow him."""
    return [O(k, k, LINE_D, w, k) for k, w in zip(OL_NAMES, (-3.2, -1.6, 0.0, 1.6, 3.2))]


def trips(rb_w=-1.6, x_w=-15.0, y_w=4.8, h_w=10.0, z_w=16.0):
    """11 personnel, shotgun, 3x1: X alone on the left (on the line), the tight end Y attached on the right
    (on the line), H in the slot and Z wide on the right, both a step off the ball. Seven on the line:
    the five linemen plus X and Y, so X and Y are the ends of the line and everyone else is a back."""
    return line5() + [O("QB", "QB", GUN, 0.0, "Q"), O("RB", "RB", GUN, rb_w, "R"),
                      O("X", "WR", LINE_D, x_w), O("Y", "TE", LINE_D, y_w),
                      O("H", "WR", OFF_BALL, h_w), O("Z", "WR", OFF_BALL, z_w)]


def nickel(shell="two_high", **moves):
    """A 4-2-5 nickel against trips right. Defensive linemen are labelled by technique: a 5 on the left
    tackle's outside shoulder, a 3 on the left guard's outside shoulder, a 1 shading the center, and a 7
    on the tight end's inside shoulder. Mike and Will linebackers 4 1/2 yards deep; the nickel back ($)
    apexed between the tight end and H. shell="two_high": two safeties 12 yards deep.
    shell="single_high": the free safety in the deep middle and the strong safety walked down as the
    overhang defender on the single-receiver side. moves = {key: (d, w)} adjusts anyone."""
    out = [D("BE", "DE", 1.0, -4.0, "5"), D("BT", "DT", 1.0, -2.35, "3"),
           D("PT", "DT", 1.0, 0.7, "1"), D("PE", "DE", 1.0, 4.2, "7"),
           D("MIKE", "MIKE", 4.5, 1.8, "M"), D("WILL", "WILL", 4.5, -2.2, "W"),
           D("NB", "NB", 5.0, 7.4, "$"), D("LCB", "CB", 7.0, -15.8, "CB"), D("RCB", "CB", 7.0, 16.8, "CB")]
    if shell == "two_high":
        out += [D("SS", "SS", 12.0, 9.0, "SS"), D("FS", "FS", 12.5, -8.5, "FS")]
    else:
        out += [D("SS", "SS", 5.5, -9.0, "SS"), D("FS", "FS", 13.0, -0.5, "FS")]
    return [q.moved(d=moves[q.key][0], w=moves[q.key][1]) if q.key in moves else q for q in out]


def only(players, lo, hi, dmax=30.0):
    """Keep the players inside a lateral window (w from lo to hi) for a zoomed static diagram."""
    return [q for q in players if lo <= q.w <= hi and q.d <= dmax]


def check_legal(players):
    """Seven offensive players on the line, and only the two at its ends may be eligible receivers."""
    on = sorted([q for q in players if q.side == "off" and q.d > -1.0], key=lambda q: q.w)
    assert len(on) == 7, [q.key for q in on]
    for q in on[1:-1]:
        assert q.key in OL_NAMES, f"{q.key} is covered up but not a lineman"


check_legal(trips())


def go_to(p, key, end, kind="block", delay=0.0, speed=None, via=()):
    """Move `key` through absolute (d, w) points `via` to `end`, as a block, path, run or route."""
    start = np.array(p._cur(key))
    pts = [tuple(np.array(q) - start) for q in (*via, end)]
    if kind == "block":
        return p.block(key, pts=pts, delay=delay, speed=speed)
    if kind == "route":
        return p.route(key, pts, frame="field", delay=delay, speed=speed)
    return {"path": p.path, "run": p.run, "pull": p.pull}[kind](key, pts, delay=delay, speed=speed)


# ---------------------------------------------------------------- drawing helpers
def _P(fld, p, d, w):
    return fld.xy(*p.to_abs(d, w))


def clean(ax):
    """Zoomed diagrams: drop hash ticks and painted yard numbers; clip text to the window."""
    for ln in list(ax.lines):
        xs, ys = ln.get_data()
        if len(xs) == 2 and ln.get_linewidth() == 0.5 and np.hypot(xs[1] - xs[0], ys[1] - ys[0]) < 1.0:
            ln.remove()
    for t in list(ax.texts):
        if t.get_text().isdigit() and t.get_alpha() == 0.55:
            t.remove()
        else:
            t.set_clip_on(True)

# ---------------------------------------------------------------- frame strips (print) and video (web)
MARK_R = 0.72


def tracking(play):
    df = play.to_tracking()
    df["_id"] = df["displayName"].astype(str)
    return df


def catch_time(play):
    """Seconds after the snap at which the first pass arrives."""
    for kind, to, te in play.ball_events:
        if kind == "pass":
            a, b = play.pass_target_spot()
            return te + float(np.hypot(*(b - a))) / 18.0
    return None


def carrier(play, t):
    """Who holds the ball at time t (None while a pass is in the air)."""
    holder = "QB"
    for kind, to, te in sorted(play.ball_events, key=lambda e: e[2]):
        if kind == "handoff" and t >= te:
            holder = to
        if kind == "pass" and t >= te:
            holder = to if t >= catch_time(play) else None
    return holder


def panel(ax, play, df, t, lateral, window, rings=()):
    """One moment of a simulated play: markers, faint trails, the ball (and its flight), rings."""
    tmap = df.groupby("frameId")["time"].first()
    fid = int((tmap - t).abs().idxmin())
    ylo, yhi = sorted((play.ball_y - lateral[0], play.ball_y - lateral[1]))
    fld = Field(ax=ax, orient="vertical", los=play.los, window=window, numbers=False, hashes=False,
                lateral=(ylo, yhi), to_go=play.to_go)
    cur = df[df["frameId"] == fid]
    shown = cur[cur["x"].between(play.los + window[0] + 0.4, play.los + window[1] - 0.4)
                & cur["y"].between(ylo + 0.4, yhi - 0.4)]
    ids = set(shown["_id"])
    trails = df[(df["frameId"] <= fid) & df["_id"].isin(ids) & (df["side"] != "ball")]
    who_now = carrier(play, t)
    for (pid, side), g in trails.groupby(["_id", "side"]):
        xs, ys = fld.xy(g["x"].values, g["y"].values)
        if pid == who_now and pid != "QB":       # the ball carrier's path, drawn so it reads in print
            ax.plot(xs, ys, color=style.OFFENSE_DARK, lw=1.8, alpha=0.9, zorder=3.5)
            continue
        ax.plot(xs, ys, color=style.OFFENSE if side == "off" else style.DEFENSE, lw=0.9, alpha=0.4, zorder=3)
    passes = [te for kind, _, te in play.ball_events if kind == "pass"]
    if passes and t > passes[0]:
        b = df[(df["side"] == "ball") & (df["frameId"] <= fid) & (df["time"] >= passes[0] - 0.05)]
        xs, ys = fld.xy(b["x"].values, b["y"].values)
        ax.plot(xs, ys, color=style.BALL, lw=1.3, ls=(0, (2, 1.5)), zorder=9)
    ppy = fld.pts_per_yard()
    for _, rr in shown.iterrows():
        x, y = (float(v) for v in fld.xy(rr["x"], rr["y"]))
        if rr["side"] == "ball":
            ax.add_patch(Ellipse((x, y + 0.45), 0.8, 0.5, fc=style.BALL, ec="white", lw=0.5, zorder=11))
            continue
        lab = str(rr.get("label", "") or "")
        if lab in OL_NAMES and lab != "C":
            lab = ""                       # too small to read in a strip; the center keeps his letter
        if rr["side"] == "off":
            ax.add_patch(Circle((x, y), MARK_R, fc=style.OFFENSE, ec="white", lw=0.6, zorder=8))
            tc = "white"
        else:
            ax.add_patch(RegularPolygon((x, y), 4, radius=MARK_R * 1.3, orientation=np.pi / 4,
                                        fc=style.DEFENSE_TINT, ec=style.DEFENSE, lw=1.0, zorder=8))
            tc = style.INK
        fs = max(4.2, min(7.5, ppy * (0.95 if len(lab) <= 1 else 0.68)))
        ax.text(x, y, lab, ha="center", va="center", fontsize=fs, color=tc, fontweight="bold", zorder=10)
    who = carrier(play, t)
    for key, col in list(rings) + ([(who, style.THIRD)] if who else []):
        rr = cur[cur["displayName"] == key]
        if len(rr):
            x, y = fld.xy(rr["x"].iloc[0], rr["y"].iloc[0])
            ax.add_patch(Circle((float(x), float(y)), 1.05, fc="none", ec=col, lw=1.7, zorder=7))
    return fld


def strip(plays, times, lateral, window, notes=None, rows=None, rings=(), width_in=6.5):
    """Frame strip: one row per branch of the read, the same moments in each row."""
    dfs = [tracking(p) for p in plays]
    cells = [(i, t) for i in range(len(plays)) for t in times]
    ncols = len(times)
    nrows = len(plays)
    pw = width_in / ncols
    span_w, span_d = abs(lateral[1] - lateral[0]), window[1] - window[0]
    fig, axes = plt.subplots(nrows, ncols, squeeze=False,
                             figsize=(width_in, nrows * (pw * span_d / span_w + (0.75 if notes else 0.3))))
    for k, (ax, (i, t)) in enumerate(zip(axes.flat, cells)):
        panel(ax, plays[i], dfs[i], t, lateral, window, rings)
        head = f"{k % ncols + 1}.  " + ("snap" if abs(t) < 1e-6 else f"{t:+.1f} s")
        if rows and k % ncols == 0:
            head = rows[i] + "   " + head
        ax.set_title(head, fontsize=8, color=style.INK, loc="left",
                     fontweight="bold" if rows and k % ncols == 0 else "normal")
        if notes:
            ax.text(0.5, -0.03, notes[k], transform=ax.transAxes, ha="center", va="top", fontsize=7.0,
                    color=style.INK, linespacing=1.2)
    fig.tight_layout(h_pad=1.6 if notes else 0.6, w_pad=0.5)
    return fig


def video(plays, times, lateral, window, notes=None, rows=None, rings=(), width_in=6.4):
    """Web version: the branches side by side, drawn every tenth of a second."""
    from matplotlib import animation
    from IPython.display import HTML, display
    dfs = [tracking(p) for p in plays]
    t0 = min(float(df["time"].min()) for df in dfs)
    t1 = max(float(df["time"].max()) for df in dfs)
    grid = np.round(np.arange(max(t0, -0.5), t1 + 1e-9, 0.1), 2)
    n = len(plays)
    span_w, span_d = abs(lateral[1] - lateral[0]), window[1] - window[0]
    pw = width_in / n
    fig, axes = plt.subplots(1, n, squeeze=False, figsize=(pw * n, pw * span_d / span_w + 0.7))
    axes = list(axes.flat)

    def update(t):
        for i, ax in enumerate(axes):
            ax.clear()
            panel(ax, plays[i], dfs[i], t, lateral, window, rings)
            txt = "snap" if abs(t) < 0.05 else f"{t:+.1f} s"
            if notes:
                mine = notes[i * len(times):(i + 1) * len(times)]
                cur = [m for tt, m in zip(times, mine) if tt <= t + 1e-6]
                txt += ("\n" + cur[-1].replace("\n", " ")) if cur else ""
            ax.set_title((rows[i] + ": " if rows else "") + txt, fontsize=7.5, color=style.INK, loc="left")
        return []

    anim = animation.FuncAnimation(fig, update, frames=list(grid), interval=100)
    html = (anim.to_html5_video(embed_limit=40).replace("<video ", '<video style="max-width:100%;height:auto" ')
            if shutil.which("ffmpeg") else anim.to_jshtml(default_mode="once"))
    plt.close(fig)
    display(HTML(html))


def rpo_animation(plays, times, lateral, window, notes=None, rows=None, rings=(), **kw):
    """Video on the website; a numbered, captioned frame strip in the PDF."""
    if pdf_build():
        strip(plays, times, lateral, window, notes=notes, rows=rows, rings=rings, **kw)
        plt.show()
    else:
        video(plays, times, lateral, window, notes=notes, rows=rows, rings=rings)

First-and-10 on the opponent’s 40, early in the second quarter. The offense is in the shotgun with three receivers to the right and one alone on the left. At the snap the whole offensive line lurches to the right, run blocking, and the quarterback sticks the ball into the running back’s stomach. Everything about it says inside run, and the linebacker on the left side believes it: he takes two quick steps forward to fill his gap. Then the quarterback pulls the ball back out, turns, and throws to the receiver on the left, who has run upfield and bent inside, and is standing ten yards deep in the space behind where that linebacker was a second ago. Twelve yards, first down. The broadcast analyst says “RPO” the way people say “algorithm,” as if the word explained itself. So: was that a run or a pass? It was both, on purpose. The offense called a run and a pass on the same snap and let one defender’s two steps decide which one it would run. This chapter is about plays that are two plays: what they are, why they work, why the NFL’s version looks different from college’s, and how to see the decision coming.

You’ll be able to…

  • Say what makes a play an RPO, and tell it apart from play-action (the run is fake) and from the option (both choices are runs).
  • Find the defender an RPO is built to put in conflict, usually the apex or overhang defender or a linebacker, and predict give or throw from what he does.
  • Tell a pre-snap RPO (decided by counting the box) from a post-snap RPO (decided by reading a defender), and draw the bubble, glance, slant and stick RPOs and the pop pass.
  • Explain the NFL’s one-yard ineligible-downfield rule, how it differs from college’s three yards, and why it forces NFL RPO throws to be quick and short.
  • Name the defense’s main answers, and say what each one gives back.
NoteYou’ll need

One snap, two plays

A run-pass option, or RPO, is a play on which the offensive line blocks a run while one or more receivers run pass routes, and the quarterback decides which of the two to use. He decides either before the snap, by counting defenders, or just after it, by watching one defender while he holds the ball in the running back’s belly. If the defense has set itself to stop the run, he throws. If it has set itself to stop the pass, he hands off.

That sounds like every play: of course you run against a pass defense and throw against a run defense. The difference is when and by whom the choice is made. On an ordinary play the coordinator chooses, from the sideline, some forty seconds before the snap, on a guess about what the defense will do. On an RPO the coordinator calls both plays and hands the choice to the one person on the field who can see what the defense actually did, at the last possible moment. Coaches call a call like this a packaged play: one call that bundles a run with one or more passes, plus rules for picking. Every RPO is a packaged play; the older, broader term also covers packages with a run, a quarterback keep and a pass, which you will meet near the end of the chapter.

An RPO sits between two plays you already know, and the easiest way to understand it is to put the three side by side.

Read the table’s middle column next to the right one. The option reads a defender whose only job is the run: the end can tackle the back or the quarterback, so the offense makes him pick one. The RPO reads a defender who has a run job and a pass job, and makes him pick one of those. That is why it is called an option at all. It is the same arithmetic as the zone read, moved out of the backfield and into the space between the box and the receivers.

And read the left column against the right. Play-action also wants a linebacker to step up, but it needs him to believe the fake, and if he doesn’t, the quarterback is holding the ball in a pocket with no plan B except a throw into coverage. The RPO doesn’t need anyone to be fooled. If the linebacker steps up, throw. If he doesn’t, the run is real, and the run is now going against a defender who is standing in the wrong place to stop it.

ImportantWhy it exists

RPOs exist because modern defenses ask a few players to do two jobs at once, and an offense that waits to see which job they choose can always take the other one. Spread formations pulled defenses out of the box, and defenses answered with hybrid players (the nickel back, the walked-down safety, the apexed linebacker) who stand halfway between the box and a slot receiver and promise to help with both. That promise is the target. The trade: an RPO commits the offense to a simple run and a simple throw (complicated plays don’t fit in half a second), its linemen must obey a rule that run blocking normally ignores, and the quarterback has to make a clean decision while turning his body away from half the field. Most wrong reads are not a disaster, which is the beauty of it: a handoff that should have been a throw is still a run into a blocked box. The one costly mistake is the opposite, a throw into a defender who only pretended to play the run, and that is why, as you’ll see, the run is the default.

NoteMadden vs. real life

In the game, RPOs live in their own corner of the playbook, and you, the player, decide by pressing a button. That much is real: the quarterback really does decide. What the game hides is how little time he has and how little he is looking at. On a real post-snap RPO the quarterback is not scanning the field. The play call tells him which kind of defender to read (“the backside linebacker,” “the man over the slot”); before the snap he finds who that is today, then watches only him for about two-thirds of a second at the mesh, and either lets go of the ball into the back’s belly or pulls it. Most of the time, he hands it off. An NFL RPO is mostly a run play with a safety valve, not a pass play that sometimes runs.

Nobody else on the offense knows which branch is coming, and nobody needs to. The receivers run their routes every time (on a handoff they are decoys, pulling defenders away from the run), and the linemen block the run every time. A receiver who blocks a cornerback on a bubble RPO that turns into a handoff is simply blocking on a running play, and blocking downfield on a run is allowed: the downfield limits you are about to meet apply only when a forward pass is thrown.

Who gets read: the conflict defender

Every RPO is built around the defender with two jobs, the conflict defender of Passing Fundamentals. On an RPO his two jobs are a run gap and a pass area.

The usual suspect is the defender you met in A First Look at Defense: the overhang defender, standing in an apex alignment halfway between the last man in the box and a slot receiver. On a run toward him, he has to come in and fill the edge of the box; on a pass, he has to widen and take the slot’s short route. The apex is a promise to do both a step late, and the RPO calls the bluff. In a nickel defense he is usually the nickel back; against a single receiver, he may be a safety walked down or an outside linebacker.

He is the most common target, not the only one. Any defender whose run job and pass job pull in opposite directions will do:

  • An inside linebacker. On a run, he steps forward to fill a gap; on a pass, he drops back into a zone (or turns to cover a back or tight end in man). Two steps forward for the run open the space behind him.
  • A safety who fits the run. Against two-high shells, defenses often bring a safety down after the snap as the extra run defender. If he comes, the deep area he left is open.
  • Nobody in particular: the count. On a pre-snap RPO, the quarterback reads no single man. He counts how many defenders are in the box and how many are out over the receivers, and goes where the numbers are.

The run half isn’t always inside zone, though this chapter uses it throughout because you know it best. Duo and split zone carry RPO tags (a tag is a word added to a play call that changes one player’s job) too, and so, less often, do gap runs like counter and power (Gap and Power Runs). Those are rarer because a pulling lineman runs along the line and turns upfield through the hole, and a lineman on the move is the one most likely to be too far downfield if the ball is thrown.

What does the quarterback look for in a post-snap read? The same two things an option quarterback watches in a read key: direction and eyes. A defender who steps forward (or inside) with his eyes in the backfield has chosen the run. A defender who backpedals, widens, or turns his eyes to a receiver has chosen the pass. And the rule for doubt is the same as on most zone reads: the run is the default. The quarterback throws only when the defender has clearly taken himself out of the pass; if he’s unsure, he hands off, because the run was blocked and a hesitant throw is the worst outcome on the field.

Pre-snap RPOs: count, then choose

The simplest RPO is decided before anyone moves. A pre-snap RPO gives the quarterback a run and a quick throw and tells him to pick one from the defense’s alignment (some staffs call these check-with-me plays or alerts). No defender is read after the snap. The quarterback catches the ball and either hands it off or throws it, immediately, to the receiver the numbers favor.

The classic version is the bubble RPO: a run paired with a bubble screen to a slot receiver, the quick screen from The Quick Game in which the receiver bows back and out and catches the ball already running toward the sideline while the receiver outside him blocks. The quarterback counts twice:

  1. The box. How many defenders are in the box, against how many blockers the offense has? With five linemen and a tight end, the offense has six.
  2. Outside. On the bubble side, how many defenders are close enough to stop a bubble (count the ones within about seven yards of the line; a safety twelve yards deep arrives too late), against how many receivers the offense has there?

The defense has to put its players somewhere, so the two counts move against each other. Every defender who walks into the box to stop the run is one fewer outside, and every defender who stays outside over the slot is one fewer in the box.

That’s the whole logic, and it’s worth saying why it’s so hard to defend. The defense wants to put the nickel back in the box against a run-heavy offense: it’s the extra man, the hat math of 02-05. And it wants to put him out over the slot against a passing offense. The bubble RPO says: pick one, and we’ll run the other. The only escape is to have both, which means either bringing a safety down (and playing with one deep defender, which the offense’s next play can attack) or moving after the snap so the count the quarterback made is wrong by the time the ball is out.

The offense pays a little, too. The bubble is a horizontal throw, so its ceiling is low; it gains yards only if the blocker outside wins and the receiver makes the first man miss. It is also a throw that, if it is dropped behind the line, becomes a backward pass and a live ball (The Quick Game has the rule). And it is predictable: an offense that throws the bubble every time the nickel walks in teaches the nickel to walk in and then sprint back out at the snap, which a pre-snap count cannot see. That is the opening for the post-snap RPO.

CautionCommon misconception

“A pre-snap RPO is just an audible.” Not quite. An audible changes the play: the quarterback calls out a new play at the line, and all eleven players have to hear it and change what they do. On a pre-snap RPO nobody changes anything. The line blocks the run, the back runs his track, the receivers run their routes or their screen, and only the quarterback’s last action, hand off or throw, depends on the count. That’s why RPOs are so fast to execute and so hard for a defense to read from the line: there is no new call to hear and no shuffle of feet when the check is made.

Post-snap RPOs: read a defender

A post-snap RPO moves the decision to after the snap. The line blocks the run, the back takes the ball into the mesh, and the quarterback rides him for a beat while he watches one defender. It is the zone read’s mesh from 03-04, with the keep replaced by a throw. If the read defender plays the run, the quarterback pulls the ball and throws behind him. If he plays the pass, the quarterback lets go and the back runs.

One of the NFL’s most common versions, and the flagship of this chapter, is the glance RPO. The run is inside zone (03-02). The pass is a glance (many coaches simply call it a skinny post): the receiver on the side away from the run takes a few hard steps upfield, then bends inside at a shallow angle, a skinny post that ends eight to twelve yards deep. The read is the linebacker on that side, the Will, the one who would fill the backside gaps against inside zone. His run job is to step forward; his pass job, in most zone coverages, is to drop into the window the glance runs into (a passer’s word for the open lane between defenders where a ball can be completed).1

If the Will plays the run (top row of Figure 25.4), he has done what inside zone teaches him: see the line step, see the back coming, fill the gap. Against a plain handoff, that fill is how linebackers make tackles for no gain. Against the glance RPO it empties the zone behind him. The quarterback pulls the ball, the back carries out his fake (which still holds the other linebackers for a moment), and the throw goes into the space between where the Will now is and where the safety is. The receiver catches it eight to twelve yards downfield, often running.

If the Will plays the pass (bottom row), he sits or sinks to take the glance away. Fine: now he isn’t filling his gap, and inside zone is running against five box defenders with six blockers. Count the hats again. The defense started with six in the box against six blockers, an even count. The read took one of those six out of the run without anyone blocking him, so the left guard, who on a plain inside zone would come off the 3-technique and climb to the Will, can simply stay on his block.

That last point is the quiet genius of the post-snap RPO, and it’s the same as the option’s: the read is a block. On plain inside zone, the backside linebacker is somebody’s job, usually a guard who must come off his first block at just the right moment and travel four or five yards to find him. On the glance RPO, nobody has to go get the Will. Whatever he does, he has taken himself out of one of the two plays.

What can go wrong? Three things, and the defense will try all of them:

  • The defender lies. A linebacker who takes one hard step forward and then drops sells “run” to the quarterback and still takes the glance away. Coaches call this a fake fill or a bluff (on the edge, it’s the slow-play of 03-04); a quarterback reading a lie throws into a defender. The cure is the run-as-default rule: throw only on a defender who has clearly committed, two steps and shoulders turned, not one step.
  • Someone else takes the window. If the Will fills the run and a safety rotates down (shows deep before the snap, then comes down toward the line after it) into the space he left, the read was right and the throw is still covered. This is the RPO version of the scrape exchange (03-04), and it’s coming in the defense section below.
  • The clock. The decision has to be made at the mesh, inside about two-thirds of a second, and the ball has to be out right after, for a reason that is about rules rather than football. Hold that thought until The one-yard rule below.

The safety in Figure 25.5 is in the worst seat in football. The defense put him there on purpose. With three receivers to the other side, it has rotated weak: instead of sending extra defenders toward the trips, it walked a safety down on the single-receiver side, so that it still has a spare run defender there (the “+1” of 02-05: one run defender nobody blocks), and accepted three-on-three against the trips with the free safety over the top. The walked-down safety has to be that spare run defender and also take away the quick throw to X. Watch how the offense has arranged the field so that his two jobs point in opposite directions: the run’s cutback lane is inside and forward of him, the slant’s landing spot is at him, and he can’t be in both places at the moment the ball is decided.

The cornerback matters too. The slant is a throw the corner can take away by lining up with inside leverage (on the receiver’s inside shoulder, the term from The Quick Game). If he does, the quarterback’s pre-snap look tells him the slant is dead, and good RPO offenses carry a second answer to that side (a bubble or a quick out paired with the slant) so the read still has two branches. This is where pre-snap and post-snap RPOs meet: the quarterback uses the pre-snap look to choose which pass is live and the post-snap read to choose whether to throw it.

Stick and the pop pass: two more throws behind the run

The same idea works with almost any quick throw that lands behind a run defender. Two you’ll see every weekend:

  • The stick RPO. Stick from The Quick Game puts a receiver at about six yards, settling in front of the quarterback, with a second receiver in the flat. As an RPO the flat route is usually dropped, and the outside receiver clears deep instead: the run replaces the flat as the second option. The quarterback reads the defender over the slot (the nickel or the overhang). If he triggers (commits hard, downhill, to the run), the stick is open behind him: throw. If he widens or sinks under the stick, hand off. Notice that “widens” now means the opposite of what it meant in 04-03. There, a nickel who widened was chasing the flat route and left the stick open; here there is no flat route to chase, so a nickel who widens is covering the stick, and the run is the open answer.
  • The pop pass. A pop pass uses a player who looks like a run blocker. The tight end steps as if to block the end or a linebacker, then releases (“pops”) straight up the field into the space behind linebackers who have stepped up to stop the run. The read is usually the linebacker over him. The pop is the RPO’s cleverest trick, because linebackers are taught to read the tight end: a tight end who blocks means run. On the pop pass, the tight end shows them exactly that, for one beat, and they come.

The one-yard rule: why NFL RPOs look different

In football, only some players may catch a forward pass. The five interior linemen are ineligible receivers (The Twenty-Two), and the rules don’t only stop them catching the ball, they stop them running downfield before it is thrown. Imagine the game without that limit: every pass play could begin with the whole line charging out and blocking linebackers, exactly as on a run. A linebacker could never tell run from pass, so he could never safely drop into coverage, and the receivers behind him would be catching passes in space that had been blocked for them. The rule keeps a pass play looking like a pass play. The foul is called illegal man downfield, or ineligible downfield (the referee’s announcement is often “ineligible man downfield”).

Here is the NFL’s version, as written in the 2026 rulebook:2

  • On a play on which a legal forward pass is thrown, it is a foul if the entire body of an ineligible player is more than one yard beyond the line of scrimmage before the pass is thrown.
  • Exception: a lineman who makes contact with a defender within a yard of the line, on his initial charge, may be driven or may drive his man further downfield while still blocking him. If they come apart beyond the yard, he may stay where he is, move sideways or back off until the ball is thrown (some line coaches call it freezing). What he may not do is let go of that defender and keep moving forward, or go past a yard without having touched anyone.
  • Penalty: five yards from the previous spot. A completed pass is wiped out; there is no loss of down.

Now set that against the run game. On inside zone, linemen double-team a defensive lineman and then one of them climbs: he lets go and runs to a linebacker four or five yards deep. That climb is the heart of zone blocking, and on a pass play it is exactly what the rule forbids: letting go of a defender inside one yard and then advancing.

College football draws the line somewhere else. Under the NCAA rule, no originally ineligible player may be more than three yards beyond the neutral zone (the strip the length of the ball between the two lines) until a legal forward pass that crosses the neutral zone has been thrown, also a five-yard penalty. (College judges any part of the body, the NFL the entire body.)3 Three yards is room for a guard to release toward a linebacker who has stepped up. Actually blocking him that far downfield before a pass that crosses the neutral zone is still offensive pass interference in college, unless the block began on the lineman’s first charge within a yard of the neutral zone. And the “crosses the neutral zone” clause means a pass caught behind the line (a bubble, a now screen) doesn’t trigger either college rule at all.

The NFL rule has no such clause: it applies to any legal forward pass, including one caught behind the line, which is how Play-Action, Bootlegs, and Screens described NFL screens too. The one throw it doesn’t touch is a backward pass. A bubble thrown even slightly backward is a lateral, not a forward pass, so nobody can be “ineligible downfield” on it; the price, as The Quick Game warned, is that a dropped lateral is a live ball.

That difference is the single biggest reason an NFL RPO looks different from a college one:

  • NFL RPO throws are quick, and most are short. The decision is made at the mesh and the ball is out within about a second of the snap, before any lineman has had time to drift past a yard. And a ball that must leave in a second can’t go far, because a receiver can’t get far in a second: running full speed from a standing start, he is only about five yards downfield when it’s thrown. A deep route needs two and a half seconds or more to develop, and by then a run-blocking line is well past a yard. So NFL RPOs throw to routes that are open early: slants, glances and sticks a few yards past the line, and bubbles and now screens that come out at once (the bubble, thrown slightly backward, isn’t a forward pass at all). The data below shows how strong this is.
  • NFL RPO blocking stays at the line. Instead of climbing, linemen hold their first blocks, or the offense uses blocking that naturally stays on the line (down blocks, double teams that don’t release) and lets the read handle the linebacker. The glance RPO is popular in the NFL for exactly this reason: the read takes care of the man the climbing guard would have blocked.
  • College RPOs can be “run first, throw later.” With three yards and no limit at all on throws behind the line, college linemen can run-block for longer and drift up to three yards downfield, the quarterback can ride the mesh longer, and the pass can go farther downfield: seams and posts over linebackers who have stepped up to the run.

Why hasn’t college changed? It nearly did. In 2015 the NCAA’s rules committee proposed cutting its limit to one yard, and the oversight panel tabled the proposal; the rule is still three yards for 2026.4 The NFL has never had college’s three-yard cushion: its rulebook still says one yard in 2026. That leaves officials judging a one-yard line at full speed on every RPO, and RPOs put the rule under strain on every snap: an RPO whose throw comes out a beat late is exactly the play on which a run-blocking lineman has drifted too far. You will see in the numbers that RPOs draw far more than their share of the flags.

Receivers who block too early. Receivers are eligible, so the one-yard limit doesn’t bind them, but a cousin rule does: in the NFL, blocking a defender more than a yard past the line “clearly prior” to a pass is offensive pass interference, and pass interference can occur “regardless of whether [the pass] crosses the line.”5 So on a bubble RPO the receiver blocking the cornerback wants his block to land as the ball leaves the quarterback’s hand, one more reason the bubble comes out at once. Judging “clearly prior” at full speed is hard; Bill Belichick, asked about RPOs in 2019, called the downfield-blocking question “a tough call for the officials.”6

Three choices on one snap

Nothing stops an offense from putting the zone read and an RPO on the same snap. The quarterback reads two defenders in order, exactly like the triple option of 03-04, with a pass in the pitch man’s role:

  1. At the mesh, read the backside end: if he stays wide, give to the back; if he crashes on the back, pull the ball.
  2. Now moving sideways along the line with the ball, read the overhang defender: if he comes up to tackle the quarterback, throw the bubble to the receiver outside him, before crossing the line (a forward pass thrown from beyond the line of scrimmage is illegal); if he stays with the receiver, tuck the ball and run.

In practice the throw on a triple RPO is almost always a bubble or a flare, a ball thrown sideways or slightly backward, not a slant, for the reason of the previous section: by the time the quarterback has pulled the ball and taken a few steps, a second and a half has gone by, and a run-blocking line is past a yard. College offenses called these triple RPOs or simply packaged plays, and they were common in the shotgun spread offenses of the 2000s and 2010s, where the three-yard rule left more room. They are rarer in the NFL, for two reasons: the quarterback carries the ball into the second level on the keep, where he takes hits, and every extra beat of reading is a beat in which the linemen are run blocking further from the line.

Where RPOs came from

Nobody invented the RPO in one moment, and several people say they did. What is not disputed is the ingredients: the option’s habit of reading a defender instead of blocking him, and the spread offense’s habit of giving the quarterback a quick throw to whichever side the defense left light. Put a quick throw on a running play and you have it.

  • Air Raid tags and the 1990s. Coaches in Hal Mumme and Mike Leach’s Air Raid tradition (a spread, pass-first offense built on a handful of plays repeated endlessly; 04-08 has it), at Kentucky in the late 1990s and Leach’s Oklahoma in 1999, tagged quick screens onto runs so the quarterback could throw if the defense overloaded the box. Rich Rodriguez, whose zone read grew out of a fumbled practice handoff at Glenville State (03-04), built quick slants and bubbles into his zone-read offense at Tulane and Clemson.7
  • Claims to the invention. Urban Meyer says his Utah teams found it by accident in the early 2000s, when a receiver missed a block and ran a bubble instead;8 Brett Favre says he started doing it on his own, bored with handing off, before he had told his coach.9 Treat both as what they are: stories coaches tell about a thing that was in the air.
  • The college spread, 2000s–2010s. Spread offenses made packaged plays routine. Chip Kelly’s Oregon ran them fast and often enough that when Kelly took over the Eagles in 2013, Washington’s defensive coordinator Jim Haslett prepared to face Kelly’s new team by watching 23 or 24 Oregon films.10 At Auburn, Gus Malzahn’s “still-emerging RPO game,” in CBS writer Dennis Dodd’s words, helped beat Alabama in the 2013 “Kick Six” game; Dodd credits Malzahn with having “refined and perfected the modern run-pass option.”11 And when Robert Griffin III arrived in Washington in 2012 from Art Briles’s Baylor, Mike Shanahan borrowed Baylor’s packaged plays. After Griffin’s first game, Shanahan put it this way: “we had running plays called and all of a sudden they gave us certain looks to take the running game away, and we threw a couple of bubbles early.”12 The full lineage, from the Wing-T and the wishbone through the spread option, is in The College Laboratory.
  • The NFL boom, 2016–2018. The tree that made RPOs a fixture of NFL offense is Andy Reid’s. By Pro Football Focus’s charting, the share of NFL plays that were RPOs roughly doubled from about 4% in 2016 to about 8% in 2018; Reid’s Chiefs led the league in RPO usage in 2017 (17.1% of their plays) and 2018 (25.0%), and Doug Pederson, who had been Reid’s offensive coordinator in Kansas City from 2013 to 2015, ranked second with the Eagles in 2017 and won a Super Bowl with them (see the film room). Matt Nagy, Pederson’s successor in Kansas City, took the idea to Chicago, where the Bears went from 2.7% RPOs in 2017 to 19.2% in his first season, 2018.13
  • Since then. PFF’s league-wide rate kept climbing to a peak of about 10.5% of plays in 2023, then fell to about 9.0% in 2024 and 8.0% in 2025.14 RPOs stopped being a trend and became furniture: almost every NFL offense has a handful in every game plan.

What the numbers say

The NFL’s play-by-play doesn’t say whether a play was an RPO: a handoff is a handoff and a pass is a pass. But FTN’s charters, who watch every NFL play, flag RPOs, and the flag is in the nflverse data from 2022 on. The numbers come with a caveat that is worth learning once: charting services disagree. PFF’s rates for the same seasons run well above FTN’s (about 8 to 10.5 percent of plays against FTN’s 4 to 6), because deciding from film whether a handoff “was” an RPO (did a receiver run a real route? was the quarterback reading someone?) is a judgment, and charters judge differently. Five things to read from Figure 25.8 and the numbers behind it.

RPOs are a small, steady slice of NFL offense. FTN charted 5.0% of plays as RPOs in 2022, 5.3% in 2023, 3.6% in 2024 and 6.2% in 2025 (the 2024 dip is larger than PFF’s and may partly be charting). The leaders are the teams this chapter has already met: PHI in 2022 (13.5%), IND in 2023 (18.3%), and Kansas City in 2025 (14.4%), more than twice the league rate. At the other end, the RPO is plainly a play-caller’s choice, not a law of modern football. The 49ers and Rams, who run mostly from under center with wide-zone blocking, are the bottom two in 2025 (0.7% and 0.5%): 96% of the league’s charted RPOs came from the shotgun, and an offense that lives under center gets the same “make the linebacker step up, then throw behind him” effect from play-action instead (04-05). Yet the 2025 Packers, whose head coach Matt LaFleur comes from the same Shanahan–McVay tree, ranked 5th. And Baltimore, with Lamar Jackson, one of the league’s most dangerous running quarterbacks, sits mid-table (5.6%): a zone read or a designed quarterback run is an option run, and FTN doesn’t count it as an RPO.

Most RPOs are runs. On 69% of the charted RPOs from 2022 to 2025 the ball was handed off (79% in 2025). That is the “run with a safety valve” idea from the Madden callout, in data. It also means the RPO’s main effect is invisible in a box score: its product is the defender who wasn’t in the run fit.

NFL RPO throws are short, and very often behind the line. 57% of RPO throws were caught (or aimed) at or behind the line of scrimmage, against 21% of other passes; only 11% travelled more than ten yards in the air, against 30% of other passes. That shape is the clock at work: the bubbles and now screens come out about half a second after the snap, before any lineman can drift a yard, and a throw beyond the line has to leave so fast that it can’t be far beyond it.

The rule bites. From 2022 to 2025 the NFL called 246 ineligible-downfield fouls on passes in the regular season. 24% of them came on plays FTN charted as RPOs, though RPOs were only 5.0% of plays: per snap, an RPO is many times more likely to draw the flag than any other play.

Per play, RPOs are about average, and steadier. Split by branch, the numbers look dramatic: RPO throws averaged +0.057 EPA with a 79% completion rate, against +0.001 for all other passes; RPO handoffs averaged -0.053, against -0.015 for all other runs. Don’t read that as “the throws are great and the runs are bad.” It is selection, as with the quarterback keeps of 03-04: the quarterback throws precisely when the throw is open and hands off when it isn’t, so the throws are the good half of the reads by construction. Put the halves back together and RPOs come out close to ordinary plays: -0.019 EPA per play against -0.005, and an EPA success rate of 42.8% against 43.5%. What does differ is the spread. The standard deviation of EPA on RPOs is 1.00, against 1.40 on other plays: fewer disasters and fewer explosions, which is what you’d expect from a play that throws only short and only when it’s open.20

The box count shows up, weakly, too: when FTN counted six defenders in the box, 28% of RPOs were thrown; with seven, 33%; with eight or more, 40%. More defenders near the ball, more throws: the counting logic of the bubble RPO, visible in charting that sees only the alignment at the snap.

PFF’s charting, with its larger RPO count, tells a slightly different story: from 2021 to 2025, NFL RPOs gained fewer yards per play than other plays (4.53 against 5.43) but were “successful” more often (52.3% against 48.1%).21 The two series don’t contradict each other so much as measure different things. PFF uses its own success definition, not the EPA-based one above, and its RPO count is about twice FTN’s, so it includes many snaps FTN called plain runs or passes. What both agree on is the shape: RPOs are not big-play machines. They are a low-variance way to gain modest yardage with a run that is always blocked, and whether they beat an ordinary play on efficiency depends on whose charting you trust.

  • Read the line, not the back. The most useful key a linebacker has against an RPO team is the uncovered guard, the one with no defensive lineman in front of him (03-01). If he climbs hard to the second level, it’s a true run: he couldn’t legally do that on a pass. If he stays at the line, sits, or blocks whoever is in front of him without climbing, it may be an RPO, and the linebacker plays his pass job first. The cost: the key is only as good as the offense’s honesty, and offenses answer with true runs whose guards climb, to punish linebackers who wait.
  • Make the read lie. Give the read defender a rule (always play the pass) and put someone else in his run fit, usually a safety rotating down after the snap (showing deep, then dropping toward the line), as in Figure 25.9. It is the scrape exchange of 03-04 with a safety instead of a linebacker, and it’s built on the post-snap rotations of Disguise and Rotation. It works because the safety’s run job is called: he triggers at the snap without reading anything, so he arrives in time. The cost is the deep defender who left, and a play-action shot over him beats it.
  • Play two-high and fit the run from depth. Two-high shells, especially the quarters family of 06-05 (a two-deep zone in which the safeties also help against the run), keep the overhang defender honest on the pass and let a safety who starts ten yards deep become the extra run defender once he sees a handoff. Against a pre-snap count, this is a defense that shows six in the box and plays like seven. Its weakness is the run itself. Unlike the rotating safety above, this one reads first (run or pass?) and only then comes down, so he arrives late, often after the back is already past the line.
  • Take away the conflict with man coverage. In man coverage, the defender over the slot isn’t torn between a gap and an area: he has a man, and he goes with him. Nobody is in conflict, so there’s nothing to read. The price is the one man coverage always pays: defenders with their backs to the ball, and a box whose count the run can attack.
  • Squeeze the clock. Hold the read defender’s decision as long as possible (a bluff, a slow-play) so the quarterback rides the mesh a beat longer and the linemen, run blocking, drift past a yard. The flag is a defensive weapon too, and a defense that has watched the tape knows which linemen climb early. The cost: a defender who is still deciding is in neither job, so against an offense that just hands off on time, the run gets a defender who arrives a step late.
  • Hit the quarterback. Every RPO exposes him for a moment with his eyes on one defender and his body turned toward the back. While he still holds the ball at the mesh he can be tackled like any ball carrier, and a rusher who arrives just as he throws may still finish the hit within a step of the release (later than that is roughing the passer).22 Pressure off the edge makes him hurry the read. The cost: the extra rusher comes from somewhere, usually the coverage, and the quick throws an RPO carries go to exactly the space he left.

Offenses answer back, as always: with motion that changes who the conflict defender is just before the snap, with RPOs that read the safety instead of the overhang, and with play-action that looks exactly like the RPO for the first second and then throws deep over the safety who has learned to come down. That conversation, how one play makes the next one work, is the subject of Constraint Theory, where the RPO appears as a constraint play: a play that punishes a defense for overplaying the offense’s base run.

TipWatch for it

On Sunday: “Is he reading you?” Every RPO has a defender being read. Find him before the snap and you’ll see the decision before the replay does.

Before the snap

  • Shotgun or pistol, back beside the quarterback, receivers spread. That’s the RPO look. Not every snap from it is an RPO, so treat it as “possible.”
  • Count the box, then count outside. Six or seven in the box? How many defenders are within seven yards of the line over the slot receivers? If the box is heavy and the slots are light, expect a throw (a bubble or now screen) the instant the ball is snapped.
  • Find the apex. Who stands halfway between the box and a slot receiver? On the single-receiver side, is a safety walked down? That’s the likely read. Say it out loud: “he’s being read.”

At the snap

  • RPO or play-action? Watch the line for half a second. On an RPO the linemen fire forward, low, and stay on their blocks at the line; nobody climbs. On play-action they make one hard run step, then their helmets pop up and they settle into pass protection, often a step behind the line. And on a true run, the uncovered guard climbs to a linebacker. Forward and staying put means RPO.
  • Where the quarterback’s head turns. On a plain handoff, he looks at the back. On a zone read, his eyes are on a defensive lineman at the line. On a post-snap RPO they go to the second level or out over a slot: his eyes are on a defender. On the live sideline shot you can usually see his helmet turn; for his actual eyes and the defender he’s watching, wait for the end-zone replay (01-06).
  • The backside receiver. On a pure run he jogs or blocks his cornerback. On an RPO he runs a full-speed slant or glance, because he might get the ball.
  • Watch the read defender’s first two steps, not the ball. Forward and inside: predict throw, behind him. Backward, widening, or eyes on a receiver: predict give.
  • Count to one. If the ball isn’t out of the quarterback’s hand by about a second, it’s probably a handoff. (Play-action’s throw comes later, two and a half seconds or more, after a full fake and a drop.)

After the play

  • On the replay, freeze the frame when the ball leaves the quarterback’s hand and look at the linemen. Is anyone more than a yard past the line without a defender in his hands? That’s the flag the umpire is looking for. Did one get thrown?
  • On the next snap, did the read defender do the opposite? Defenses adjust within a drive.

For your prediction log (01-06): on every RPO look, write the defender you think is being read and your call (G or T) from his first steps; afterward, write what happened.

Takeaways

  • An RPO blocks a run and carries a pass on the same snap, and the quarterback picks one. Unlike play-action, the run is real; unlike the option, one branch is a pass. Every RPO is a packaged play.
  • It attacks a conflict defender with a run job and a pass job, most often the apex or overhang defender, sometimes a linebacker or a safety. If he plays the run, throw behind him; if he plays the pass, hand off. The run is the default.
  • Pre-snap RPOs (the bubble RPO) are decided by counting the box against the defenders outside; post-snap RPOs (the glance, slant and stick RPOs and the pop pass) are decided by watching one defender at the mesh. The read does a blocker’s job.
  • The NFL’s illegal man downfield rule (one yard, for any forward pass, with an exception for linemen still engaged with a defender they met at the line) makes NFL RPOs quick and short and keeps NFL linemen near the line; college’s three yards past the neutral zone, only for passes that cross it, lets college RPOs run-block longer and throw farther.
  • RPOs grew out of Air Raid tags, the zone read and the college spread, and the Reid tree carried them to the NFL (the 2017 Chiefs and Eagles). In FTN’s charting they are a few percent of NFL plays, mostly handoffs, with throws that cluster at or behind the line, and about as efficient per play as anything else, with less variance.
  • Defenses answer by reading the line instead of the back (a guard who climbs means run), making the read lie (a safety called down into the vacated run fit), playing two-high and fitting from depth, playing man to remove the conflict, stretching the clock to draw flags, and hitting the quarterback. Each one gives something back: a deep defender, a step in the run fit, or a rusher’s worth of coverage.

From diagrams to tracking data. The animation in this chapter is a gridiron simulation written in NFL Big Data Bowl coordinates (simulated, BDB-format), so real tracking data drops into the same drawing code. The 2025 Big Data Bowl’s plays.csv carries a PFF run-pass-option flag (pff_runPassOption), so finding real RPOs is a filter, not a hunt. With the tracking you can measure what this chapter only describes: how far past the line each lineman is at the frame of the pass_forward event, and how long the quarterback held the mesh. The cell below shows the shape of that code (it is not run, because the course does not download the Big Data Bowl files).

Show the Python (real tracking data; not run) 
import pandas as pd
from gridiron import bdb, show_animation

trk = bdb.load_tracking("data/bdb/tracking_week_*.csv")
plays = pd.read_csv("data/bdb/plays.csv")
rpos = plays[plays["pff_runPassOption"] == 1]                    # PFF's RPO flag (BDB 2025)
g, pl = rpos.iloc[0][["gameId", "playId"]]
one = bdb.prepare(trk, plays, game_id=g, play_id=pl)
show_animation(one, times=[0.0, 0.6, 1.0, 1.8])

# How far past the line is each offensive lineman when the ball is thrown?
los = one.loc[(one["frameType"] == "SNAP") & (one["side"] == "ball"), "x"].iloc[0]
throw_f = one.loc[one["event"] == "pass_forward", "frameId"].min()
linemen = one[(one["frameId"] == throw_f) & one["position"].isin(["T", "G", "C"])]
print((linemen["x"] - los).round(1).tolist())                  # yards past the line
