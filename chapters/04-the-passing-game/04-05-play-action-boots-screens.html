# ---------------------------------------------------------------- players and alignments
def O(key, pos, d, w, label=None):
    return Player(key, pos, "off", d, w, label=label or key)


def D(key, pos, d, w, label):
    return Player(key, pos, "def", d, w, label=label)


def line5():
    """The five offensive linemen, 1.6 yards apart, each lettered so the reader can follow him."""
    return [O(k, k, LINE_D, w, k) for k, w in OL]


def ace_tight():
    """12 personnel under center: tight ends U (left) and Y (right) on the line, receivers X and Z
    a step off the ball in tight splits, the back seven yards deep. Seven on the line: five
    linemen plus the two tight ends, who are the ends and so eligible."""
    return line5() + [O("QB", "QB", UC, 0.0, "Q"), O("RB", "RB", -7.0, 0.0, "R"),
                      O("U", "TE", LINE_D, -4.8), O("Y", "TE", LINE_D, 4.8),
                      O("X", "WR", OFF_BALL, -9.0), O("Z", "WR", OFF_BALL, 9.0)]


def i_form():
    """21 personnel I formation: X on the line wide left, Y in-line right, Z off the ball wide right."""
    return line5() + [O("QB", "QB", UC, 0.0, "Q"), O("FB", "FB", -4.5, 0.0, "F"), O("RB", "RB", -7.0, 0.0, "R"),
                      O("Y", "TE", LINE_D, 4.8), O("X", "WR", LINE_D, -14.0), O("Z", "WR", OFF_BALL, 14.0)]


def spread(rb_w=-1.5, y_inline=False):
    """11 personnel, shotgun 2x2: X and Z on the line wide, H in the left slot and Y (a tight end)
    in the right slot, both a step off the ball. y_inline=True attaches Y to the line instead and
    moves Z off the ball, so there are still exactly seven on the line."""
    out = line5() + [O("QB", "QB", GUN, 0.0, "Q"), O("RB", "RB", GUN, rb_w, "R"),
                     O("X", "WR", LINE_D, -15.0), O("H", "WR", OFF_BALL, -9.0)]
    if y_inline:
        return out + [O("Y", "TE", LINE_D, 4.8), O("Z", "WR", OFF_BALL, 14.0)]
    return out + [O("Y", "TE", OFF_BALL, 9.0), O("Z", "WR", LINE_D, 15.0)]


def even_front(cb_w=10.0, shell="two"):
    """A four-man even front (named in 05-02) with linemen labelled by technique number, three
    linebackers and either two deep safeties (shell='two') or one, with the strong safety
    walked down into the box (shell='one': eight in the box)."""
    out = [D("PE", "DE", 1.0, 4.0, "5"), D("PT", "DT", 1.0, 0.7, "1"),
           D("BT", "DT", 1.0, -2.35, "3"), D("BE", "DE", 1.0, -4.0, "5"),
           D("SAM", "SAM", 4.0, 6.6, "S"), D("MIKE", "MIKE", 4.5, 1.6, "M"), D("WILL", "WILL", 4.5, -2.6, "W"),
           D("LCB", "CB", 6.5, -cb_w - 0.5, "CB"), D("RCB", "CB", 6.5, cb_w + 0.5, "CB")]
    if shell == "two":
        out += [D("FS", "FS", 11.5, -8.0, "FS"), D("SS", "SS", 11.5, 8.0, "SS")]
    else:
        out += [D("FS", "FS", 13.0, 0.0, "FS"), D("SS", "SS", 6.5, 8.5, "SS")]
    return out


def nickel(extra=None, shell="three"):
    """A 4-2-5 nickel against the 2x2: corners off the outside receivers, the strong safety over H
    (left slot), the nickel ($) over Y (right slot), one deep safety. shell='zero' pulls the free
    safety down to eight yards over the back (no deep help) for all-out pressure looks.
    extra = {key: (d, w)} moves anyone."""
    out = [D("WDE", "DE", 1.0, -5.4, "E"), D("WDT", "DT", 1.0, -0.8, "T"),
           D("SDT", "DT", 1.0, 2.4, "T"), D("SDE", "DE", 1.0, 5.4, "E"),
           D("MIKE", "MIKE", 4.5, 1.8, "M"), D("WILL", "WILL", 4.5, -2.2, "W"),
           D("LCB", "CB", 7.0, -16.0, "CB"), D("RCB", "CB", 7.0, 16.0, "CB"),
           D("SS", "SS", 5.5, -8.6, "SS"), D("NB", "NB", 5.0, 8.6, "$"),
           D("FS", "FS", 13.0, 0.0, "FS") if shell == "three" else D("FS", "FS", 8.0, -2.0, "FS")]
    extra = extra or {}
    return [q.moved(d=extra[q.key][0], w=extra[q.key][1]) if q.key in extra else q for q in out]


# ---------------------------------------------------------------- moving people
def go_to(p, key, end, kind="path", delay=0.0, speed=None, via=()):
    """Move `key` through absolute (d, w) points `via` to `end`, as a path, run, route, block, pull or rush."""
    start = np.array(p._cur(key))
    pts = [tuple(np.array(q) - start) for q in (*via, end)]
    if kind == "rush":
        return p.rush(key, pts=pts, delay=delay, speed=speed)
    if kind == "route":
        return p.route(key, pts, frame="field", delay=delay, speed=speed)
    if kind == "block":
        return p.block(key, pts=pts, delay=delay, speed=speed)
    if kind == "pull":
        return p.pull(key, pts, delay=delay, speed=speed)
    return {"path": p.path, "run": p.run}[kind](key, pts, delay=delay, speed=speed)


def pass_set(p, keys=("LT", "LG", "C", "RG", "RT"), depth=1.2, speed=None):
    """Linemen step back into their pass sets."""
    for k in keys:
        p.block(k, pts=[(-depth, 0.0)], speed=speed)


def zone_steps(p, keys=("LT", "LG", "C", "RG", "RT"), step=(0.15, 1.3)):
    """Linemen take outside-zone steps to the right but stay at the line (the one-yard limit)."""
    for k in keys:
        p.block(k, pts=[step])


# ---------------------------------------------------------------- drawing helpers
def _P(fld, p, d, w):
    return fld.xy(*p.to_abs(d, w))


def clean(ax):
    """Drop the painted yard numbers (they sit where players stand); clip all text to the window."""
    for t in list(ax.texts):
        if t.get_text().isdigit() and t.get_alpha() == 0.55:
            t.remove()
        else:
            t.set_clip_on(True)


def draw(p, width_in=6.2, ax=None, **kw):
    fld = p.draw(width_in=width_in, ax=ax, title=False, **kw)
    clean(fld.ax)
    return fld

DASH = (0, (3.6, 3.0))     # long enough to read as dashed at print size


def branch(fld, p, pts, color=GREEN, lw=1.7, dash=DASH, end="arrow"):
    """A dashed path through absolute (d, w) points: a defender's reaction, or the other branch of a read."""
    xs, ys = _P(fld, p, np.array([q[0] for q in pts]), np.array([q[1] for q in pts]))
    fld.ax.plot(xs, ys, color=color, lw=lw, ls=dash, zorder=5, solid_capstyle="round")
    if end == "arrow":   # the head sits on the last yard only, so it never paints over the dashes
        seg = float(np.hypot(xs[-1] - xs[-2], ys[-1] - ys[-2]))
        f = min(1.0, 0.9 / seg) if seg > 0 else 1.0
        x0, y0 = xs[-1] - f * (xs[-1] - xs[-2]), ys[-1] - f * (ys[-1] - ys[-2])
        fld.ax.add_patch(FancyArrowPatch((x0, y0), (xs[-1], ys[-1]), arrowstyle="-|>",
                                         mutation_scale=10, color=color, lw=lw, ls="-", zorder=5))
    elif end == "tee":
        v = np.array([xs[-1] - xs[-2], ys[-1] - ys[-2]])
        perp = np.array([-v[1], v[0]]) / np.linalg.norm(v) * 0.75
        fld.ax.plot([xs[-1] - perp[0], xs[-1] + perp[0]], [ys[-1] - perp[1], ys[-1] + perp[1]],
                    color=color, lw=lw + 0.6, zorder=5)


def after_throw(fld, p, key, target):
    """A released lineman's climb to his block once the ball is in the air (dashed: it happens after the throw)."""
    kf, _, _ = p._compile()
    d0, w0 = kf[key][-1, 1:]
    branch(fld, p, [(d0, w0), target], color=style.OFFENSE_DARK, lw=1.5, end="tee")

# ---------------------------------------------------------------- frame strips (print) and video (web)
MARK_R = 0.7


def tracking(play):
    df = play.to_tracking()
    df["_id"] = df["displayName"].astype(str)
    return df


def frame_panel(ax, play, df, t, lateral, window_, rings=(), throw_t=None, mark_r=MARK_R):
    """One moment of a simulated play: markers, faint trails, optional rings and the ball's flight so far."""
    tmap = df.groupby("frameId")["time"].first()
    fid = int((tmap - t).abs().idxmin())
    ylo, yhi = sorted((play.ball_y - lateral[0], play.ball_y - lateral[1]))
    fld = Field(ax=ax, orient="vertical", los=play.los, window=window_, numbers=False, hashes=True,
                lateral=(ylo, yhi), to_go=play.to_go)
    cur = df[df["frameId"] == fid]
    shown = cur[cur["x"].between(play.los + window_[0] + 0.4, play.los + window_[1] - 0.4)
                & cur["y"].between(ylo + 0.4, yhi - 0.4)]
    ids = set(shown["_id"])
    trails = df[(df["frameId"] <= fid) & df["_id"].isin(ids) & (df["side"] != "ball")]
    for (_, side), g in trails.groupby(["_id", "side"]):
        xs, ys = fld.xy(g["x"].values, g["y"].values)
        ax.plot(xs, ys, color=style.OFFENSE if side == "off" else style.DEFENSE, lw=0.9, alpha=0.45, zorder=3)
    if throw_t is not None and t > throw_t:
        b = df[(df["side"] == "ball") & (df["frameId"] <= fid)]
        b = b[b["time"] >= throw_t - 0.05]
        xs, ys = fld.xy(b["x"].values, b["y"].values)
        ax.plot(xs, ys, color=style.BALL, lw=1.3, ls=(0, (2, 1.5)), zorder=9)
    ppy = fld.pts_per_yard()
    for _, rr in shown.iterrows():
        x, y = (float(v) for v in fld.xy(rr["x"], rr["y"]))
        if rr["side"] == "ball":     # drawn beside the carrier's marker so it never hides his label
            ax.add_patch(Ellipse((x + mark_r * 1.15, y + mark_r * 0.55), 0.95, 0.6, fc=style.BALL, ec="white",
                                 lw=0.5, zorder=11))
            continue
        lab = str(rr.get("label", "") or "")
        if rr["side"] == "off":
            ax.add_patch(Circle((x, y), mark_r, fc=style.OFFENSE, ec="white", lw=0.6, zorder=8))
            tc = "white"
        else:
            ax.add_patch(RegularPolygon((x, y), 4, radius=mark_r * 1.3, orientation=np.pi / 4,
                                        fc=style.DEFENSE_TINT, ec=style.DEFENSE, lw=1.0, zorder=8))
            tc = style.INK
        fs = max(4.0, min(8.0, ppy * (0.85 if len(lab) <= 1 else 0.62) * mark_r / 0.7))
        ax.text(x, y, lab, ha="center", va="center", fontsize=fs, color=tc, fontweight="bold", zorder=10)
    for key, col in rings:
        rr = cur[cur["displayName"] == key].iloc[0]
        x, y = fld.xy(rr["x"], rr["y"])
        ax.add_patch(Circle((float(x), float(y)), mark_r * 1.7, fc="none", ec=col, lw=1.7, zorder=7))
    return fld


def plan(fld, p, key, t0, color=None, lw=1.5, dashed=False):
    """The rest of a player's path from time t0 on (where he is going), ending in an arrowhead."""
    kf, _, _ = p._compile()
    t1 = float(kf[key][-1, 0])
    if t1 <= t0 + 0.05:
        return
    pts = np.array([p.position(key, tt) for tt in np.linspace(t0, t1, 50)])
    xs, ys = _P(fld, p, pts[:, 0], pts[:, 1])
    side = p.players[key].side
    col = color or (style.OFFENSE if side == "off" else style.DEFENSE)
    ls = (0, (3, 2)) if dashed else "-"
    fld.ax.plot(xs[:-3], ys[:-3], color=col, lw=lw, ls=ls, zorder=4, solid_capstyle="round")
    fld.ax.add_patch(FancyArrowPatch((xs[-4], ys[-4]), (xs[-1], ys[-1]), arrowstyle="-|>", mutation_scale=9,
                                     color=col, lw=lw, zorder=4))


def moment(ax, p, t, lateral, window, rings=(), plans=(), mark_r=0.8):
    """The play frozen at time t (faint trails show where everyone came from); `plans` adds
    each listed player's remaining path."""
    fld = frame_panel(ax, p, tracking(p), t, lateral, window, rings, mark_r=mark_r)
    for key in plans:
        plan(fld, p, key, t)
    return fld


def throw_arrow(fld, p, a, b):
    """The throw, drawn by hand from (d, w) a to (d, w) b."""
    (x0, y0), (x1, y1) = _P(fld, p, *a), _P(fld, p, *b)
    fld.ax.add_patch(FancyArrowPatch((float(x0), float(y0)), (float(x1), float(y1)), arrowstyle="-|>",
                                     mutation_scale=9, color=style.BALL, lw=1.3, ls=(0, (2, 2)), zorder=9,
                                     connectionstyle="arc3,rad=-0.08"))


def throw_at(p, key, t_catch):
    """Schedule the pass so the ball reaches `key` at about t_catch (ball speed 18 yd/s)."""
    t = t_catch - 0.6
    for _ in range(5):
        a, b = p.position("QB", t), p.position(key, t_catch)
        t = t_catch - float(np.hypot(*(b - a))) / 18.0
    p.pass_to(key, t=round(t, 2))
    return round(t, 2)

Why a fake works: the linebacker’s dilemma

Start with the defender who is being fooled, because play-action is really a play on him.

A linebacker lines up about four or five yards off the ball, and on every snap he has two jobs that pull in opposite directions. Against a run he has a gap to fill (a run fit, the defense’s plan for who fills which gap, built in Run Fits and Box Math), and he has to get into it fast: a running back hits the line about a second after the snap, and a linebacker who waits to be sure arrives after the back has gone by. Against a pass he has a zone or a man to cover, which usually means running backward, to a spot ten yards deep or to a receiver.

He cannot wait to find out which job it is. So he is taught to decide from a key, one or two players whose first step tells him run or pass (key reading is the subject of Linebackers). The usual keys are the offensive guard in front of him and the backfield. A guard who fires forward with his pads low means run. A guard who pops up and steps backward into a pass set means pass. (This is the “high hat, low hat” read you met in How to Watch a Broadcast, seen from the linebacker’s side.) A running back heading for the line with the quarterback’s hands reaching toward him means run. Those first-step signals are the linebacker’s run-fit triggers: the moment he sees them, he attacks his gap. Coaches call a linebacker who reacts fast to them a “downhill” player, and they love him, because against the run he is right.

Play-action, a pass play that begins with a fake run, is built to pull those triggers on purpose. The guard fires out as if run-blocking. The running back takes the same path he takes on the real run. The quarterback reaches the ball out to him and pulls it back. For about a second, every signal the linebacker has been trained to trust says run. By the time he sees the quarterback still holding the ball, he has taken two or three steps forward, and he now has to stop, turn around and run back to a spot ten yards behind him. That second is the play.

The fake has rules too

Two rules shape what the fake can look like, and both come from the offensive line.

First, the linemen cannot go downfield. On any play with a forward pass, an offensive lineman may not be more than one yard past the line of scrimmage when the ball is thrown; if he is, it is a foul (illegal man downfield, taught in Run-Pass Options).1 So a play-action line cannot really run-block the way it would on a run, driving defenders five yards back. It has to look like run blocking while staying at the line: aggressive, low first steps, then holding its ground. Coaches call this a “run-action” protection, and the best ones are nearly indistinguishable from the run for the first step or two, which is all the linebacker reads.

Second, the defensive line is fooled less, and for less time, than the linebackers. A defensive lineman who has a blocker in front of him fights that blocker on every snap, and he feels pass protection almost the instant the blocker stops driving forward. (The exception is a lineman nobody blocks, usually the end on the far side from the run. With no blocker to feel, he reads the backfield the way a linebacker does, and he can be fooled the same way. The bootleg, below, is built on him.) That is why play-action does not buy the quarterback extra protection; it buys him space behind the linebackers. If anything the quarterback is a little more exposed: he has turned his back or spent half a second faking, and he still has to set up and throw. The sack numbers agree. Comparing like with like (first and second down, 2022–2025), play-action dropbacks ended in a sack 5.7% of the time, against 5.2% for other dropbacks.2

ImportantWhy it exists

Defenses have to stop the run quickly, so the defenders closest to the line are trained to react to the first signs of a run. Play-action turns that training into a weakness: it shows them those signs on a pass, gets them moving forward, and throws into the space they left. The price is time and protection. The fake takes half a second, the quarterback’s back is often to the defense, the linemen are limited to the line, and the blocked defensive linemen are barely fooled. Play-action trades a little safety at the quarterback’s spot for a lot of open grass downfield.

“You have to establish the run.” Do you?

Every broadcast you have ever watched has an announcer who says play-action works because the team has been running the ball well: the linebackers have been punished by the run, so they bite on the fake. It sounds like common sense. It is, mostly, not true.

The cleanest test came from Ben Baldwin, an analyst who later co-created nflfastR, the play-by-play package behind the data this course uses. In early 2018 he looked at whether a team’s play-action passes did better when it had been running the ball often, or running it well, before them, and found essentially no relationship: play-action passes beat ordinary dropbacks by about the same margin whether the run game had been good, bad or barely used.3 Later that year Josh Hermsmeyer, writing for FiveThirtyEight, added the mechanism. With tracking data he measured how far middle linebackers moved the wrong way on a fake, and found they bit just as much on an offense’s twelfth play-action pass of a game as on its first: linebackers react to run action because reacting is their job, not because they have been convinced by a running back’s stats.

You can rerun the test with the data this course uses. FTN’s charters mark every dropback as play-action or not, and nflverse joins it to the play-by-play. Over 2022–2025, play-action dropbacks averaged +0.115 EPA against +0.003 for other dropbacks (essentially zero), and the gap was there in every one of the four seasons: 2022 +0.09 vs -0.01, 2023 +0.10 vs -0.03, 2024 +0.14 vs +0.03, 2025 +0.12 vs +0.02.4 Figure 24.2 asks Baldwin’s question of every team-season.

A correlation measures how closely two numbers move together, on a scale from −1 to +1: +1 means they rise in perfect step, 0 means knowing one tells you nothing about the other. Between a team’s rushing efficiency and its play-action boost it was +0.09, which is about as close to nothing as real data gets. (Square it and you get the share of the team-to-team differences the run game “explains”: under 1%.) Play-action beat the team’s other dropbacks in 78% of team-seasons, good running teams and bad ones alike.

One thing did matter a little. Teams that ran more often saw a somewhat bigger boost (correlation +0.24: small, but clearly not zero). That makes sense: a team that runs a lot shows the defense its run look a lot, and its fakes match what the linebackers have been reacting to all day. So the true part of “establish the run” is about showing the run, not about succeeding with it. Being good at running was not what mattered.

So what does matter? Coaches who build play-action games answer in one sentence: the fake has to look exactly like your real run. Same formation, same line steps, same back’s path, same quarterback footwork. A linebacker is not reading the season’s rushing stats. He is reading a guard’s helmet and a running back’s first two steps, and if those match the run he has seen all day, he moves.

CautionCommon misconception

“Play-action only works once the run game is working.” A team with a poor run game can still have an excellent play-action game, and many have. The flip side: a fake that looks different from the real run (a sloppy handoff motion, linemen who pop up into pass sets) gets little, however many yards the back has.

Under center, the quarterback takes the snap, turns and carries out the same footwork as a real handoff, often with his back fully to the defense, and the running back’s path is identical to the run. It is the most convincing fake in football, because it is the run for the first second. The cost is that the quarterback loses sight of the defense, then has to whip around, find his receivers and throw, often with a defensive lineman closing. Plays from under center are therefore usually built with a simple read the quarterback can find quickly after he turns.

From the shotgun, the back runs across the quarterback’s face and the quarterback “rides” the ball into his belly for a beat, the same handoff exchange (the mesh) as a shotgun run. The quarterback never stops seeing the defense, and the throw can come sooner. But a shotgun fake looks less like a run, partly because defenses know that most shotgun plays are passes.

The numbers make the split vivid. In 2025, 82% of all dropbacks from under center used play-action, against 10% from the shotgun. Under center has become, in effect, the play-action formation, which is a big reason it came back into fashion.5 And play-action from under center was the most productive kind of dropback in football. NoteMadden vs. real life

In Madden, play-action is a menu category: pick a play with “PA” in the name and the game handles the fake. In real football the fake is a skill that the quarterback, the back and the linemen rehearse every week, and it degrades if any one of them gets lazy. Quarterbacks are graded on their fakes. Running backs are coached to carry out the fake for several steps after the quarterback has pulled the ball, often running into the line with an empty hand, because a linebacker who sees the back stop knows instantly.

Where the fake sends the ball: three windows

What does the quarterback do with the second the fake bought? Look at where play-action completions end up. Over 2022–2025, play-action passes traveled about as far in the air as other passes (7.8 yards on average, against 7.7), but receivers gained 6.4 yards after the catch on them, against 4.9 on other passes.6 The fake’s main gift is not depth. It is catching the ball with nobody in front of you, because the people who would have been in front of you stepped up.

That points to the three places a play-action pass usually goes:

  1. Behind the linebackers, in the middle. The window in Figure 24.1: crossing routes, digs and shallow crosses that cross the field 5–15 yards deep through the space the linebackers just vacated.
  2. Behind the safeties. If a safety reads run and comes down to help, the deep post or the corner route behind him is open. This is the shot play, below.
  3. To the side the defense left. Move the quarterback the other way with the ball, and the whole defense’s flow is pointing at the wrong sideline. This is the bootleg.

The bootleg: move the launch point away from the flow

You met the bootleg in Zone Running as outside zone’s partner, and its ancestor, the Wing-T’s waggle, in Perimeter Runs and Misdirection. Here is the full play.

A bootleg is a play-action pass in which the quarterback fakes a run one way and then rolls out the other way with the ball, toward a group of receivers set up at three depths on that side. Nobody recorded who named it, but the usual story is that the quarterback hides the ball against his hip or thigh as he rolls away, the way a Prohibition bootlegger hid a bottle.7 Two things make it work at once:

  • The run fake pulls the linebackers and safeties toward the run side, and it pulls the one defensive lineman nobody blocks, the end on the far side, down the line after the back.
  • The rollout moves the quarterback’s launch point away from that side, so the pass rush, which is aimed at where he was, has to turn around and chase.

The classic route combination is a flood of the boot side at three depths (the flood idea of Dropback Concepts): a short route to the flat (the area near the sideline just past the line), a crosser coming all the way across the field from the other side at about 6–10 yards, and a deep route, often a deep crosser called the over route, 15–20 yards down. A receiver on the boot side runs deep straight away to clear out the cornerback. That gives the quarterback a simple read on the move: deep if it is there, crosser if the linebackers are trailing, flat if anyone comes at him.

Look at frame 2 of Figure 24.4. Every defender’s momentum is going one way and the quarterback is going the other. The Y has already slipped past the two linebackers who stepped up; they are still facing the line, and by frame 3 they are chasing him from behind. The cornerback on the boot side has turned and run with the clear-out, and the backside end, the one man whose job was to stay home, chased the fake and is behind the play.

Now count the quarterback’s options. All three receivers are on his side of the field, at three depths, and he is moving toward them, which shortens every throw. The read is usually high to low. First the over route, if nobody deep is on it: the cornerback has gone with the clear-out, so the man to check is the safety on that side (in Figure 24.4 the free safety has stayed over it, so the quarterback moves on). Then the crosser, if the linebackers are trailing him. Then the flat, if a defender comes up to stop the quarterback. If nobody comes up at all, he can run.

Why the defense struggles. It is not that the defenders are fooled for long. It is that the linebackers and the unblocked end are fooled at the start, when the decisive steps happen, and then the play moves away from them. A defense that has been flowing hard to stop outside zone has been flowing correctly, every time, for three quarters. The bootleg makes the correct reaction to the run the wrong reaction to the play.

What the bootleg gives up. The quarterback is throwing on the run, to only half the field, with fewer receivers than a dropback (three, usually, sometimes four). Moving the launch point also moves it away from most of the protection: on most boots, nobody is blocking the man the quarterback is running toward. And the direction matters. Every boot in this chapter goes left, which for a right-handed quarterback is the harder way: rolling right, a right-hander’s body is already turned the way he throws, but rolling left he has to open his hips and shoulders back toward the receiver before he can throw. Quarterbacks drill it (“flip the hips”) because outside zone goes both ways, and so must its boot.

The naked bootleg: nobody blocks the end

That last point is the bootleg’s central bet, and the naked version makes it explicitly. A naked bootleg assigns no one to block the defender on the boot side, usually the backside end. On the real outside zone run from this formation, the U cuts that end off (03-02). On the naked boot the U shows that block for a beat, then lets the end go and releases to the flat, so nobody blocks him. The whole line and the back go with the run fake. The play is betting that the end, seeing a run and no blocker, will chase the back down the line, as he does on the real run.

When the bet wins (left panel of Figure 24.5), the naked boot is one of the easiest plays in football: the quarterback is alone on the edge with three receivers and a defense running the wrong way. When it loses (right panel), the quarterback has turned his back on a free defender. That is why the flat route is always there and always first if the end stays home: it is the boot’s built-in hot throw, and the quarterback learns to check the end before anything else. It is also why offenses run naked boots after the end has been chasing outside zone all day, not before.

Keepers, protected boots and leaks

Three variations answer the obvious questions a defense asks of the bootleg.

What if nobody covers the quarterback? Then he runs. A keeper is a quarterback run off a run fake: he shows the handoff, keeps the ball and goes around the end the fake drew inside. On a bootleg it is built in (the green branch in Figure 24.5): if the end chased and the flat defender has run with the flat route, the quarterback’s best option is often the empty grass in front of him. Some teams also call keepers as designed runs, and some playbooks simply call the whole bootleg a “keep.” Either way, a quarterback who will run makes the defense pay for chasing the receivers.

What if the end stops chasing? Then the offense blocks him. A protected bootleg sends a blocker to meet the end on the boot side: a guard pulling from the line, as the Wing-T’s waggle did (Figure 24.6, left), or, very often in today’s NFL, a fullback, H-back or tight end coming across behind the line, the same block that kicks out the end on split zone. The price is a tell: a blocker running the “wrong” way, away from the run fake, is the first thing a linebacker who keys the guards and the backfield will see.

What if the whole defense starts flowing to the boot? Then throw back the other way. A leak is a late release by a tight end or back who first sells a block, then slips into the space the defense has abandoned. On a bootleg, the classic leak is the tight end on the run-fake side, who blocks the end for a count as if the run were real, then releases into the flat on his side, the side the quarterback just ran away from. It is a change-up, not a staple, because the throw is hard: the quarterback has to stop his roll, reset his feet and throw back across the field, against his own momentum, to the side he has been moving away from.

The shot play: the fake that goes deep

The third window in the list above is behind the safeties. A shot play is a play built to throw deep, usually 20 or more yards downfield, and called in a situation the offense picked for it. Most shot plays share three features, and each has a reason.

  • Play-action, because the deepest defender that matters is a safety, and a safety who reads run takes a step or two forward. A deep throw needs only that. A safety who starts 13 yards deep normally backpedals at the snap and is 15 or 16 yards deep when the receivers get to him. One who steps up on the fake is at 11 or 12 and has to stop and turn before he can run. That is about four yards of cushion gone, and a receiver at full speed covers four yards in under half a second: the difference between a breakup and a touchdown.
  • Extra blockers, often max protect with seven or eight men in protection, because a deep route takes three seconds or more to develop.
  • Only two or three receivers, which is the price of the extra blockers. They are paired so that the one safety cannot cover both: in the classic yankee concept, a deep post from one side and a deep over route from the other.

The defense in Figure 24.7 is the picture offenses want for a shot: one deep safety, the other safety down near the box, and linebackers who attack the run. Against Cover 1 (man coverage with one deep safety, built in Man Coverage), the corners are in man coverage and the yankee routes run away from them, so the free safety alone has to help on two deep routes. (Against Cover 3, where the cornerback on the over’s side bails into a deep third, the over attacks that corner’s zone and the read shifts to him, but the idea is the same: make one deep defender choose.) That defense is built to stop the run, and the offense has usually spent the game showing it run looks from this exact formation. Situations matter as much as looks. Coaches like shots on second-and-short (an incompletion leaves a manageable third-and-short), right after a turnover or a big play (the defense is rattled and expects conservatism), and when the defense has just moved a safety down to stop the run.

Downs, Distance, and Field Position showed that NFL offenses mostly run on second-and-short. That is one reason the shot works when it comes: a defense that has seen second-and-short handed off fifty times believes the fake.

What the shot play gives up. Two receivers against at least three deep defenders (both corners and the free safety), with every linebacker whose man stays in to block free to drop or to rush, is a long-odds play on any single snap, and it has no answer if both routes are covered except a throwaway, a scramble or a sack. Coaches accept a low completion rate for the size of the prize: one completion can be worth more expected points than a whole series of short gains.

Screens: let them come

Now the opposite trick. Play-action pretends to be a run so the defense steps forward. A screen pretends to be a normal pass so the defense rushes in, then throws the ball to a receiver at or behind the line with blockers already set up in front of him. The rushers who came after the quarterback are now behind the play. The defenders left in front are outnumbered by linemen.

The quick wide receiver screens, the now screen and the bubble screen, were in The Quick Game: the ball comes out at the snap and the blockers are other receivers. This section is about the screens that use the offensive linemen as blockers, which changes everything about their timing.

The one-yard rule from the start of the chapter shapes how they are blocked. In the NFL it covers every forward pass, including one caught behind the line, so screen linemen release flat, sideways along the line, set up a wall where the receiver will catch the ball, and turn upfield to block only once the ball is in the air. (College is different: its limit is three yards and covers only passes that cross the line, so college screen blockers can be well downfield before the throw.)8 Every screen in this section follows that pattern: in the diagrams, a dashed block is the climb that happens after the throw.

The slow screen

The original is the running back’s screen, the slow screen (also called the RB screen). It is “slow” because it is built on patience: everyone on the offense pretends, for a count or two, that this is an ordinary dropback.

Sell the dropback. For the first second and a half, nothing looks unusual. The linemen pass-set, the back steps up as if to pick up a blitzer, and the receivers run their routes, and those routes have a job: pulling the cornerback and safeties deep and away from where the ball is going.

Let them go. Then the three linemen on the screen side deliberately lose. Each gives his rusher a punch, then lets him run past toward the quarterback, and releases to the side. Line coaches teach the release with a count (it varies by team, but roughly “one-thousand-one, one-thousand-two”), because a lineman who leaves too early tips the play off and one who leaves too late is still in the backfield when the ball arrives. They release flat, along the line (the one-yard rule), each to a spot a few yards outside where the back will catch it, with a rule for whom to block: the first one out takes the widest threat, the next one the next defender inside, the center the first defender chasing from behind.

Pull the rush deep, then throw over it. The quarterback keeps drifting back, eight, nine, ten yards, dragging the rushers with him, then floats the ball over their heads to the back, who has slid out behind the released linemen. The deeper the rushers chase, the farther they are from the catch.

What the slow screen gives up. It is a timing play with a lot of moving parts and a low margin for error. If a rusher is not fooled (a defensive end who sees the tackle let go and stops to sit in the throwing lane, which coaches call “smelling” the screen), the ball can be batted or intercepted near the line. If the released linemen miss their spots, the back is running into a crowd. And it gives up the whole downfield passing game for one snap: the quarterback has one place to throw. When the screen is sniffed out, the quarterback’s best play is usually to throw the ball into the ground at the back’s feet. That is legal, because the back is an eligible receiver right there (intentional grounding, the foul for throwing the ball away under pressure, applies only to a pass that is not in the direction and vicinity of an eligible receiver), and an incompletion is much better than a sack or a tipped interception.9

(For the stat sheet: a screen thrown forward, even from behind the line, is an ordinary pass, with an attempt, passing yards and receiving yards, which is how this chapter’s play-by-play data can count screens at all. One thrown backward is a lateral, as The Quick Game showed.)

A swing route is a running back’s route that bows backward and then out toward the sideline, so he catches the ball on the move behind or at the line. A flare is a quicker, flatter release toward the sideline that bends slightly upfield; many playbooks use the two words for the same thing. Either way, it is a pass route: the line keeps blocking and nobody releases in front of the back. Its purposes are different from a screen’s. It is an outlet when the downfield routes are covered (a checkdown the quarterback can always find), a way to pull a linebacker wide and open the middle for another route, and a way to get a fast back the ball in space against a slower linebacker.

The tell on a broadcast is the offensive line. If linemen leave their rushers and slide out to one side before the ball is thrown, it is a screen (or a run). If the line is still pass-blocking when the back catches it, it was a route. The second tell is the back’s first steps. A swing back leaves at the snap and runs his path. A screen back waits: he sets as if to block, then slips out and settles behind the released linemen. Don’t rely on timing: a swing can be thrown at once or late as a checkdown.

A tunnel screen goes to an outside receiver who comes back inside, toward the ball. The receiver’s first steps upfield matter: the cornerback, seeing a receiver attack him, backs up, and that cushion is the space the receiver catches the ball in. Two linemen show a quick pass set, then release outward along the line to wall off the defenders inside, and a teammate (here the Y, a tight end in the slot, a natural blocker) blocks the cornerback from the outside. The receiver catches it in the gap between them and runs up the “tunnel.”

A jailbreak screen gets its name from what the line does: at the snap, the linemen on the screen side do not pass-block at all, they break out and sprint toward the screen side, leaving their rushers free. That makes it a quick screen, thrown in about a second, because nobody is blocking those rushers. Its advantage over the now and bubble screens of 04-03 is size: instead of a receiver trying to block a defender, a 310-pound lineman arrives in space against a 190-pound cornerback.

Both screens obey the one-yard rule the same way the slow screen does. The linemen sprint sideways, along the line, and the ball comes out about a second after the snap, so they are barely a yard past the line when it is thrown; only then do they turn upfield onto the linebackers and safeties. That is why the receiver’s two steps upfield matter on both: they buy the second the linemen need to get across.

The tight end screen is the slow screen run with a tight end: he blocks the end for a count, which is exactly what defenders expect a tight end to do, then slides out behind the line on his side. It is a good call against defenses that bring the edge rusher hard, because the rusher is the man the tight end just let go. The middle screen puts the back in the middle of the field instead of the flat. The tackles push the ends wide, as they do on any dropback; the guards and center let the defensive tackles through, set up a wall at the line, and climb to the linebackers as the ball is thrown; the back slips into the space the defensive tackles just left. It punishes interior pass rushers who get upfield fast, and linebackers who drop deep on passing downs.

ImportantWhy screens exist

Every good pass rush has the same weakness: its rushers are trying to get upfield as fast as possible, toward the quarterback. A screen uses that against them. It turns the rushers’ speed into distance from the ball, and it turns the five biggest people on the field, who are usually stuck at the line, into blockers in open space. The price is that a screen gives up the rest of the passing game for one snap, needs precise timing from seven or more players, and turns into a loss, or worse, if the defense recognizes it.

Screens punish pressure

The logic of the screen predicts something you can check: screens should get better the more defenders rush. Every extra rusher is one more defender running away from the ball, and one fewer defender in front of it.

Coverage matters too. Against man coverage, the defender assigned to the back (usually a linebacker or, as in Figure 24.12, a safety) has to chase him, which means running straight into the released linemen. That is why screens pair so well against Cover 0 blitzes (man coverage with nobody deep, as in Figure 24.12) and Cover 1 blitzes. Against zone, the danger is an underneath defender who simply sits in the flat where the screen is going and waits for it.

Two cautions about the chart. First, selection: offenses call screens when they expect pressure, and a screen called into a blitz is likely to be the right call at the right moment, so these averages flatter screens a little. Second, the typical screen is not a great play: across all rush counts, screens averaged -0.07 EPA per attempt, below ordinary passes. Screens are not an engine of an offense. They are a tax on pressure: a play that makes blitzing expensive, so that the defense rushes four more often, which helps every other pass the offense throws. Screens made up about 8% of pass attempts in 2025, and 38% of screens over 2022–2025 went to running backs.

How defenses fight back

None of this is free for the offense, because defenses coach against every one of these plays. The answers belong to the defensive chapters, but here is the shape of each, so you can spot it on Sunday.

  • Against the fake: linebackers are taught to read through their key. A guard who fires out but stops at the line, or a guard pulling the “wrong” way, says pass or boot even when the back says run; some teams have their linebackers cross-key, each watching the guard on the far side as well as his own. On obvious passing downs, many simply give ground first and react to the run late. (Linebackers builds this.)
  • Against the boot: the backside end is assigned to stay home (his checklist is boot, counter, reverse), and the flat defender on the boot side sinks under the over route instead of jumping the flat. Coverages with a middle-of-the-field robber or match rules that pass the crosser from one defender to the next take away the boot’s middle read. (Run Fits and Box Math and Man Coverage.)
  • Against the screen: a defensive lineman who feels his blocker let go is taught to stop rushing and turn to find the back (“smell” it), and a rusher who sees a back slip out can peel off and chase him. Defenses that expect a screen on third-and-long call a coverage that keeps an underneath defender home rather than blitzing. (Blitz and Man Pressure.)

Each of those answers costs the defense something on another play: an end who stays home is not chasing the run, and a lineman who stops to smell the screen is not rushing the passer. The offense’s reply to the reply is the subject of Constraint Theory.

A short history

Play-action is as old as the legal forward pass (1906): as soon as a team could throw, faking a run first was the obvious way to make a defense hesitate.10 What changed over the twentieth century was how systematically offenses built it in. The Wing-T of the 1950s made the run fake and the bootleg (its waggle) part of a series in which every play looked like the others, the idea you met in 03-05.

The modern NFL bootleg game runs through Mike Shanahan and Gary Kubiak in Denver. Kubiak was the Broncos’ offensive coordinator from 1995 through 2005 under Shanahan, alongside Alex Gibbs’s outside zone run game of 03-02, and John Elway, in the last four seasons of his career, made faking outside zone and rolling away from it a staple. The pairing became a coaching tree’s signature. Kyle Shanahan, Mike’s son, was the Atlanta Falcons’ offensive coordinator in 2016, when Matt Ryan, throwing off play-action and boots, was named the league’s MVP. Kyle took the system to San Francisco as head coach in 2017, and Sean McVay and Matt LaFleur, who had coached on his offensive staffs in Washington (and, for LaFleur, Atlanta), carried it to the Rams and the Packers.11

Then the data caught up with the coaches. In 2018 Sean McVay’s Rams went 13–3 and reached the Super Bowl with Jared Goff throwing off play-action more than any quarterback in the league, on about 35 percent of his dropbacks.12 In the same year Baldwin and Hermsmeyer published the analyses described above, showing that play-action’s value did not depend on the run game. The league was already moving that way: by Football Outsiders’ count, play-action rose from 18 percent of dropbacks in 2016 to 24 percent in 2018.13 (Other charting services count fakes differently, so their percentages are not comparable year to year.) In FTN’s charting the league used play-action on 21% of dropbacks in 2022 and 23% in 2025, with the return of the under-center formation (see Classic Formations) helping push it up.

Screens are just as old, and no one owns them. The screen game most associated with the modern NFL is Andy Reid’s. Reid learned the West Coast offense under Mike Holmgren in Green Bay, and from his first years as a head coach in Philadelphia (from 1999) his teams threw running back screens as a core play, not a gimmick, with Brian Westbrook as their best-known weapon in the 2000s. He took the same habits to Kansas City in 2013.14

Three things stand out in Figure 24.14.

Most teams were better with the fake. 22 of 32 offenses averaged more EPA on play-action dropbacks than on their others. That is a single season, and some teams’ play-action samples are only a hundred-odd plays, so individual gaps bounce around from year to year. The league-wide pattern does not.

The heaviest users build their passing game on their run game. McVay’s Rams led the league at 35% and Ben Johnson’s Bears were next at 31%. Several other offenses from the Shanahan family tree sat in the top half (McDaniel’s Dolphins, LaFleur’s Packers, Kubiak’s Seahawks), though Shanahan’s own 49ers ranked only 21st in 2025. Some offenses barely fake, and not all of them suffer for it. The lowest rates belong to offenses built around a shotgun quarterback who wins from the pocket, where a fake looks less like a run. The Chiefs ranked 32nd in play-action rate but 13th in EPA per dropback, above the league’s middle. Low use is a style, not a mistake: a fake is worth most when it looks like your real run, and some offenses’ real runs are not the ones defenses fear.

Why doesn’t everyone fake on every snap, if it is worth that much? Because a fake only works while the defense must respect the run (part of the measured boost belongs to the friendly early-down situations offenses pick for it). On third-and-9, nobody bites.

Watch for it on Sunday

TipWatch for it: the linebackers’ feet

Before the snap

  • Under center? About four of every five under-center dropbacks begin with a run fake. When the quarterback lines up under the center, think run or play-action, not dropback.
  • The run look. Whatever formation this offense runs its main run from (often 12 personnel, or 11 personnel with the receivers in tight splits) is the formation it fakes from.
  • The down and distance. First-and-10 and second-and-short are play-action downs; third-and-long is not.

At the snap: watch the linebackers’ feet, not the ball

  • Did they step forward? Count the steps. Two steps up on a fake is a win for the offense. The main broadcast shot is usually framed on the box, so the linebackers are in it; the replay from the end zone, behind the offense, shows their first steps best.
  • Did the guard in front of them fire out low, or pop up? If he popped up and they stayed home, the fake did not work, and the throw will be harder.

In the first two seconds

  • Did the quarterback turn his back to the defense (under-center fake) or ride the back (shotgun)?
  • Did he roll away from the fake? Find the backside end: did he chase (the boot wins) or stay home (watch for the quick throw to the flat, or a protected boot with a pulling guard or a back coming across)?
  • Only two receivers out, eight in to block? Count the receivers who release. Two out with a hard fake on an early down is a shot play: look deep, at the single safety.
  • A tight end who blocks on the run-fake side, then slips out late while everyone else floods the other way: that’s the leak.
  • Linemen leaving their rushers and sliding out to one side before the throw? It’s a screen. Which ones released, and to which side? Did the rushers run past the back?

After the play (on the replay)

  • How many yards did the crosser gain after the catch? That is the space the linebackers left.
  • On a screen, count the blockers in front of the catch and the rushers behind it.

For your prediction log (How to Watch a Broadcast): on every under-center snap, write run, PA or boot before the snap. Then, when the defense blitzes on a passing down, write screen? and see how often you are right.

What has the end taken away, what has he given up, and what are two good calls from this look?

TipAnswer

He has taken away the naked boot and given up some of his run defense. He is playing his textbook rule: the backside end on outside zone is taught to check boot, counter, reverse before he chases (Perimeter Runs and Misdirection). An end who stays home and waits for the quarterback is the one defender the naked bootleg cannot survive (Figure 24.5, right).

What he has given up depends on one thing: he cannot tell the boot from the run until it is too late, because for the first second they look the same (that is the point of the boot). So if he has decided to stay home on the boot, he is staying home on the run too, and he is no longer chasing the outside zone run from behind. The backside of the run, where the back cuts back, is softer than it was.

Two good calls:

  1. Outside zone right, and let the back cut back. The end’s new habit has taken him out of the play’s backside pursuit. (Or split zone, from 03-02: a tight end or H comes across behind the line and kicks out the end, who is now sitting still and easy to find.)
  2. A protected bootleg (Figure 24.6, left): the same fake and the same boot, but a guard pulls, or a tight end or fullback comes across, to kick the end out. The end is now a blocked defender, and the quarterback gets the edge he wanted. The risk is the tell of a blocker going the “wrong” way, which is why it is a call to make after the defense has adjusted, not before.

And if the end still wins on a boot, the quarterback has a legal escape: once he is outside the tackles he may throw the ball away with no receiver near it, as long as it reaches the line of scrimmage, without being called for intentional grounding (the rule is in the screens section’s note and in Quarterback Play). That is why “throw it away” was the right answer on the last two boots, not a sack.

Which snap is the screen, and to whom? What is the other one, and what told you?

TipAnswer

A is a tight end screen to the right; B is a swing route to the back. Look at the line first, not the ball.

  • In A, two linemen have left their rushers. The right guard and right tackle set, gave a punch and are now sliding out to the right, level with the line, while the end and tackle they were blocking run free toward the quarterback. Linemen leave rushers on purpose only for a screen (or a run), and the side they slide to is the screen side. The Y, who blocked the end for a count, is slipping out behind them. The quarterback is drifting back to pull the rush deeper. The linemen are still level with the line because the ball has not been thrown yet: in the NFL they may not be more than a yard downfield until it is (the rule from earlier in the chapter).
  • In B, all five linemen are still pass-blocking, and the back left at the snap and is already bowing out toward the left sideline on his own. Nobody is releasing in front of him. That is a route: a swing, an outlet the quarterback can throw whenever he wants.

The timing tells you less than you might think: a swing can be thrown at once or late as a checkdown. The line, and whether the back waited before leaving, tell you every time.

Takeaways

  • Play-action works on the linebackers’ run-fit triggers. Defenders near the line must react to run signals fast; a fake that shows those signals (low line steps, the back’s path, the handoff) moves them forward and opens the space behind them. It buys space, not protection.
  • It does not need a good run game. Baldwin’s 2018 analysis and this chapter’s 2022–2025 rerun agree: a team’s rushing efficiency barely predicts its play-action boost. What matters is that the fake looks exactly like the real run. Under-center play-action, the most convincing fake, was the most productive dropback in football over 2022–2025.
  • The fake sends the ball to three windows: behind the linebackers (crossers, most of the value, much of it after the catch), behind the safeties (the shot play, with extra blockers and two or three receivers), and to the side the defense left (the bootleg).
  • The bootleg moves the launch point away from the defense’s flow, with three receivers at three depths. The naked boot bets that the end chases; the keeper, the protected boot and the leak answer the defense’s adjustments to it.
  • A screen invites the rush, releases linemen in front of a receiver behind the line (in the NFL they slide out along the line and turn upfield only after the throw) and throws over the rushers. Slow, tunnel, jailbreak, tight end and middle screens vary who catches it and where the blockers go. A swing or flare is a route, not a screen: nobody releases to block.
  • Screens punish pressure. Their value climbs with every extra rusher; against a soft rush they are among the worst plays in football. They are a tax that keeps defenses from blitzing freely.
