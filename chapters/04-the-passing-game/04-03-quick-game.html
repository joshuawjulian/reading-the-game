def O(key, pos, d, w, label=None):
    return Player(key, pos, "off", d, w, label=label or key)


def D(key, pos, d, w, label):
    return Player(key, pos, "def", d, w, label=label)


def line5():
    """The five offensive linemen, 1.6 yards apart (only the center is lettered)."""
    return [O(k, k, LINE_D, w, "C" if k == "C" else "") for k, w in
            (("LT", -3.2), ("LG", -1.6), ("C", 0.0), ("RG", 1.6), ("RT", 3.2))]


def spread(rb_w=-1.5, x_w=-15.0, h_w=-9.0, y_w=9.0, z_w=15.0):
    """11 personnel, shotgun 2x2. X and Z on the line (with the five linemen: seven on the line);
    H in the left slot and Y, a tight end split out, in the right slot, both a step off the ball."""
    return line5() + [O("QB", "QB", GUN, 0.0, "Q"), O("RB", "RB", GUN, rb_w, "R"),
                      O("X", "WR", LINE_D, x_w), O("H", "WR", OFF_BALL, h_w),
                      O("Y", "TE", OFF_BALL, y_w), O("Z", "WR", LINE_D, z_w)]


def cover3(nb_side=1, cb_d=7.0, cb_out=1.0, x_w=-15.0, z_w=15.0, extra=None):
    """A 4-2-5 nickel showing Cover 3 against 2x2: corners off with outside leverage, one deep
    safety, and a curl-flat defender walked down over each slot: the nickel ($) on side
    `nb_side` (+1 = offense's right) and the strong safety on the other. extra = {key: (d, w)}."""
    s = nb_side
    out = [D("WDE", "DE", 1.0, -5.4, "E"), D("WDT", "DT", 1.0, -0.8, "T"),
           D("SDT", "DT", 1.0, 2.4, "T"), D("SDE", "DE", 1.0, 5.4, "E"),
           D("MIKE", "MIKE", 4.5, 1.8, "M"), D("WILL", "WILL", 4.5, -2.2, "W"),
           D("LCB", "CB", cb_d, x_w - cb_out, "CB"), D("RCB", "CB", cb_d, z_w + cb_out, "CB"),
           D("NB", "NB", 5.0, s * 7.2, "$"), D("SS", "SS", 5.5, -s * 7.2, "SS"),
           D("FS", "FS", 13.0, 0.0, "FS")]
    extra = extra or {}
    return [q.moved(d=extra[q.key][0], w=extra[q.key][1]) if q.key in extra else q for q in out]


def protect(p, rushers=("WDE", "WDT", "SDT", "SDE"), depth=1.2, rush=1.6):
    """Linemen set, four rushers engage them: enough to say 'pass', no more."""
    for k in ("LT", "LG", "C", "RG", "RT"):
        p.block(k, pts=[(-depth, 0.0)])
    for k in rushers:
        if k in p.players:
            p.rush(k, pts=[(-rush, 0.0)], speed=2.0)


def go_to(p, key, end, kind="path", delay=0.0, speed=None, via=()):
    """Move `key` through absolute (d, w) points `via` to `end`."""
    start = np.array(p._cur(key))
    pts = [tuple(np.array(q) - start) for q in (*via, end)]
    if kind == "rush":
        return p.rush(key, pts=pts, delay=delay, speed=speed)
    if kind == "route":
        return p.route(key, pts, frame="field", delay=delay, speed=speed)
    if kind == "block":
        return p.block(key, pts=pts, delay=delay, speed=speed)
    return {"path": p.path, "run": p.run}[kind](key, pts, delay=delay, speed=speed)


# ---------------------------------------------------------------- drawing helpers
def _P(fld, p, d, w):
    return fld.xy(*p.to_abs(d, w))


def clean(ax):
    """Drop the painted yard numbers (they sit where receivers stand); clip all text to the window."""
    for t in list(ax.texts):
        if t.get_text().isdigit() and t.get_alpha() == 0.55:
            t.remove()
        else:
            t.set_clip_on(True)


def ruler(fld, p, fs=6.0, step=5, top=30):
    """Depth labels (yards past the line of scrimmage) just inside the left edge of the window."""
    xl = (160 / 3) - fld.y1 + 0.4
    for d in range(step, top + 1, step):
        y = p.los + d
        if fld.x0 + 1.5 <= y <= fld.x1 - 1.5:
            fld.ax.text(xl, y, f"{d}" + (" yd" if d == step else ""), fontsize=fs, color=style.INK_2,
                        ha="left", va="center", zorder=12,
                        bbox=dict(boxstyle="round,pad=0.12", fc=style.SURFACE, ec="none", alpha=0.85))


def draw(p, width_in=6.2, ax=None, show_ruler=True, **kw):
    fld = p.draw(width_in=width_in, ax=ax, title=False, **kw)
    clean(fld.ax)
    if show_ruler:
        ruler(fld, p)
    return fld

def branch(fld, p, pts, color=READ_GREEN, lw=1.7, dash=(0, (3, 2.2)), end="arrow"):
    """A dashed path through absolute (d, w) points: the other branch of a read, or a defender's reaction."""
    xs, ys = _P(fld, p, np.array([q[0] for q in pts]), np.array([q[1] for q in pts]))
    fld.ax.plot(xs, ys, color=color, lw=lw, ls=dash, zorder=5, solid_capstyle="round")
    if end == "arrow":
        fld.ax.add_patch(FancyArrowPatch((xs[-2], ys[-2]), (xs[-1], ys[-1]), arrowstyle="-|>",
                                         mutation_scale=10, color=color, lw=lw, zorder=5))


def taken(fld, p, pts, lw=1.4):

def bracket(fld, p, d, w0, w1, text, color=style.INK_2, fontsize=7, dy=1.2):
    """A horizontal dimension line at depth d from w0 to w1, labelled above."""
    (x0, y0), (x1, y1) = _P(fld, p, d, w0), _P(fld, p, d, w1)

MARK_R = 0.7


def tracking(play):
    df = play.to_tracking()
    df["_id"] = df["displayName"].astype(str)
    return df


def frame_panel(ax, play, df, t, lateral, window_, rings=(), throw_t=None):
    """One moment of a simulated play, with faint trails, optional rings and the ball's flight so far."""
    tmap = df.groupby("frameId")["time"].first()
    fid = int((tmap - t).abs().idxmin())
    ylo, yhi = sorted((play.ball_y - lateral[0], play.ball_y - lateral[1]))
    fld = Field(ax=ax, orient="vertical", los=play.los, window=window_, numbers=False, hashes=True,
                lateral=(ylo, yhi), to_go=play.to_go)
    cur = df[df["frameId"] == fid]
    shown = cur[cur["x"].between(play.los + window_[0] + 0.4, play.los + window_[1] - 0.4)
                & cur["y"].between(ylo + 0.4, yhi - 0.4)]
    ids = set(shown["_id"])
    trails = df[(df["frameId"] <= fid) & df["_id"].isin(ids) & (df["side"] != "ball")]
    for (_, side), g in trails.groupby(["_id", "side"]):
        xs, ys = fld.xy(g["x"].values, g["y"].values)
        ax.plot(xs, ys, color=style.OFFENSE if side == "off" else style.DEFENSE, lw=0.9, alpha=0.45, zorder=3)
    if throw_t is not None and t > throw_t:
        b = df[(df["side"] == "ball") & (df["frameId"] <= fid)]
        b = b[b["time"] >= throw_t - 0.05]
        xs, ys = fld.xy(b["x"].values, b["y"].values)
        ax.plot(xs, ys, color=style.BALL, lw=1.3, ls=(0, (2, 1.5)), zorder=9)
    ppy = fld.pts_per_yard()
    for _, rr in shown.iterrows():
        x, y = (float(v) for v in fld.xy(rr["x"], rr["y"]))
        if rr["side"] == "ball":
            if throw_t is None or t <= throw_t:
                # still in the quarterback's hands: draw it at his throwing shoulder so his letter stays readable
                x, y = (float(v) for v in fld.xy(rr["x"], rr["y"] - 1.0))
            ax.add_patch(Ellipse((x, y), 0.9, 0.56, fc=style.BALL, ec="white", lw=0.5, zorder=11))
            continue
        lab = str(rr.get("label", "") or "")
        if rr["side"] == "off":
            ax.add_patch(Circle((x, y), MARK_R, fc=style.OFFENSE, ec="white", lw=0.6, zorder=8))
            tc = "white"
        else:
            ax.add_patch(RegularPolygon((x, y), 4, radius=MARK_R * 1.3, orientation=np.pi / 4,
                                        fc=style.DEFENSE_TINT, ec=style.DEFENSE, lw=1.0, zorder=8))
            tc = style.INK
        fs = max(4.0, min(8.0, ppy * (0.85 if len(lab) <= 1 else 0.62)))
        ax.text(x, y, lab, ha="center", va="center", fontsize=fs, color=tc, fontweight="bold", zorder=10)
    for key, col in rings:
        rr = cur[cur["displayName"] == key].iloc[0]
        x, y = fld.xy(rr["x"], rr["y"])
        ax.add_patch(Circle((float(x), float(y)), MARK_R * 1.6, fc="none", ec=col, lw=1.7, zorder=7))
    return fld


def strip(play, times, notes=None, lateral=(-24, 4), window=(-8, 14), ncols=2, rings=(), throw_t=None,
          width_in=6.5):
    df = tracking(play)
    nrows = int(np.ceil(len(times) / ncols))
    pw = width_in / ncols
    span_w, span_d = abs(lateral[1] - lateral[0]), window[1] - window[0]
    fig, axes = plt.subplots(nrows, ncols, squeeze=False,
                             figsize=(width_in, nrows * (pw * span_d / span_w + (0.6 if notes else 0.25))))
    for ax in axes.flat[len(times):]:
        ax.axis("off")
    for i, (ax, t) in enumerate(zip(axes.flat, times)):
        frame_panel(ax, play, df, t, lateral, window, rings, throw_t)
        lab = "snap" if abs(t) < 1e-6 else f"{t:+.1f} s"
        ax.set_title(f"{i + 1}.  {lab}", fontsize=8.5, color=style.INK, loc="left")
        if notes:
            ax.text(0.5, -0.03, notes[i], transform=ax.transAxes, ha="center", va="top", fontsize=7.3,
                    color=style.INK, linespacing=1.2)
    fig.tight_layout(h_pad=2.2 if notes else 0.6, w_pad=0.6)
    return fig


def video(play, times, notes=None, lateral=(-24, 4), window=(-8, 14), rings=(), throw_t=None, width_in=5.6):
    from matplotlib import animation
    from IPython.display import HTML, display
    df = tracking(play)
    span_w, span_d = abs(lateral[1] - lateral[0]), window[1] - window[0]
    fig, ax = plt.subplots(figsize=(width_in, width_in * span_d / span_w + 0.4))
    tmap = df.groupby("frameId")["time"].first()

    def update(fid):
        ax.clear()
        t = float(tmap[fid])
        frame_panel(ax, play, df, t, lateral, window, rings, throw_t)
        txt = "snap" if abs(t) < 0.05 else f"{t:+.1f} s"
        if notes:
            cur = [n for tt, n in zip(times, notes) if tt <= t + 1e-6]
            txt += ("   " + cur[-1].replace("\n", " ")) if cur else ""
        ax.set_title(txt, fontsize=8, color=style.INK, loc="left")
        return []

    anim = animation.FuncAnimation(fig, update, frames=sorted(df["frameId"].unique()), interval=100)
    html = (anim.to_html5_video(embed_limit=40).replace("<video ", '<video style="max-width:100%;height:auto" ')
            if shutil.which("ffmpeg") else anim.to_jshtml(default_mode="once"))
    plt.close(fig)
    display(HTML(html))


def play_movie(play, times, notes=None, **kw):
    """Video on the website; a numbered, captioned frame strip in the PDF."""
    if PDF:
        strip(play, times, notes=notes, **kw)
        plt.show()
    else:
        kw.pop("ncols", None)
        kw.pop("width_in", None)
        video(play, times, notes=notes, **kw)


def trail(p, key, target, jab, lag=0.35, off=(-1.1, -0.9), t_open=0.35, t_on=0.9, t_end=4.0):
    """Hand-made keyframes for a defender chasing `target` from behind: he steps to `jab` (a
    reaction to the release), then settles `off` = (down, toward the offense's right) yards from
    where the receiver was `lag` seconds earlier. (gridiron's man() ends on top of the receiver.)
    His depth is floored so he never ends up in the offense's backfield: while the receiver is near the
    line the defender stays hip to hip with him, then the floor relaxes and he trails a step behind."""
    kf, t0, t1 = p._compile()
    kf = dict(kf)
    q = p.players[key]
    rows = [(min(t0, -1.0), q.d, q.w), (0.0, q.d, q.w), (t_open, *jab)]
    for tt in np.round(np.arange(t_open + 0.05, t_end, 0.05), 3):
        f = min(1.0, (tt - t_open) / (t_on - t_open))
        f = f * f * (3 - 2 * f)
        chase = p.position(target, max(0.0, tt - lag)) + np.array(off)
        now_d = float(p.position(target, tt)[0])
        chase[0] = max(chase[0], 0.3, now_d - 0.6 - 0.5 * max(0.0, now_d - 1.5))
        rows.append((tt, *(np.array(jab) * (1 - f) + chase * f)))
    kf[key] = np.array(rows)
    p._compiled = (kf, t0, max(t1, t_end))


def catch_time(p):
    """Seconds after the snap at which the first pass arrives (rounded up to a tenth)."""
    for kind, to, te in p.ball_events:
        if kind == "pass":
            a, b = p.pass_target_spot()
            return float(np.ceil((te + np.hypot(*(b - a)) / gp.BALL_SPEED) * 10) / 10)

What the quick game is

The quick game is the set of pass plays built to get the ball out in roughly 2.5 seconds or less. Three things make a pass play “quick,” and they go together:

  1. A short drop. From under center, the quarterback takes three steps back (a three-step drop); from the shotgun, already five yards deep, he takes one step (some teams teach three quick, short steps that cover the same ground). There’s no time to drift, reset or scan.
  2. Short routes. Most receivers break at five or six yards or less: slants, hitches, flats, short outs. One receiver often runs deep, but mainly to drag a defender out of the way.
  3. A throw timed to the break. The ball leaves as the quarterback’s back foot lands, aimed at the spot where the receiver will be when he makes his cut. If the throw is late, the play is usually dead.

Nothing in the rulebook defines it, and teams draw the line in different places, but analysts usually treat throws out in under 2.5 seconds as “quick.” In 2025 about half of all NFL pass attempts came out that fast; the median attempt left the quarterback’s hand at 2.5 seconds.1 That’s the first thing to understand about the quick game: it is not a gadget or a change-up. It is a large share of everything an NFL offense throws.

Why it exists: three jobs

The quick game survives in every playbook, from the old 49ers to the modern Chiefs, because it does three jobs no other part of the offense does as cheaply.

Job 1: beat the pass rush with the clock. A pass rush is a race. Four rushers against five blockers usually need well over two and a half seconds to get home, if they get there at all; a blitzer nobody blocks can arrive in about two. The quarterback can win that race in two ways: block long enough for the deep routes to develop, or throw before the rush matters. The quick game is the second way. It can’t be sacked if the ball is already gone, and it doesn’t need the offensive line to win, only to hold for a heartbeat. The linemen block it differently, too. A quick throw is released low and early, from a quarterback only a step or two behind them, so they meet the rushers aggressively at the line (the jump set of Pass Protection) to keep the defenders’ hands down and out of the throwing lanes. A defensive lineman who gets his arms up in the gap the slant is thrown through is the quick game’s most common way to lose a ball: the batted pass.

Pass Protection showed one half of the clock’s bargain: the longer the ball stays in the quarterback’s hand, the more often the rush gets to him (under one throw in ten is pressured when it comes out in under two seconds; about half when it is held three and a half seconds or more). Here is the other half, the price of throwing early.

Fast and short come as a package: the quick throws went only three to seven yards past the line of scrimmage on average, against eleven or twelve for the slow ones.

Job 2: keep the offense on schedule. A first down takes ten yards in three plays, and the plays that kill drives are the ones that lose yards or gain none: sacks, incompletions, and the third-and-long that follows them. Coaches call staying ahead of the chains being on schedule: roughly, gaining four or more on first down and half the remaining distance on second. Quick throws are completed far more often (in 2025, about 72% of attempts out in under 2.5 seconds were caught, against 57% of the rest) and almost never end in a sack, so they keep the offense in second-and-5 instead of second-and-10.2 By success rate (the share of plays with positive EPA, meaning plays that left the offense better off than before the snap), quick throws succeeded on about 51% of attempts, slower throws on about 45%.

Job 3: stretch the run game sideways. Bill Walsh, who built the most influential short passing offense in NFL history, started from a team that couldn’t run the ball well enough to control a game, so he used short, high-percentage throws to do the running game’s job; his throws have often been described as extended handoffs.3 A five-yard pass to a receiver near the sideline gets the ball to a fast player in space, away from the big defenders in the middle, and lets him run. In 2025 about 59% of the receiving yards on quick completions came after the catch, against about 38% on slower ones.4 That is the defining fact of the quick game: the throw is short and the run after it does most of the work. It also means that a defense that wants to stop the quick game has to tackle in space, on the edges of the field, where it has the fewest defenders.

ImportantWhy it exists, and what it costs

The quick game trades depth for certainty: a high chance of a small gain, a near-zero chance of a sack and a receiver running before the defense is set, in place of the big gain through the air. The bill comes in three parts. First, the defense can squeeze it. If an offense never threatens deep, defenders creep closer and the short windows shrink; quick game works best when the defense still has to respect the deep ball. Second, it needs yards after the catch: a five-yard throw that is tackled at five yards on third-and-7 is a punt. Third, it gets harder as the field shrinks: near the goal line there is less room to run after the catch. The rest of this chapter is about how good offenses keep the bargain on their side.

A short history: the rest of the Walsh story

You met the start of this story in Passing Fundamentals: Bill Walsh, an assistant on Paul Brown’s Cincinnati Bengals in 1970, building a passing game of short, timed throws around Virgil Carter, a quarterback who was accurate but had no deep arm, and Carter leading the NFL in completion percentage in 1971.5 Two details sharpen it. Carter was never supposed to be the starter: the Bengals’ brilliant young quarterback, Greg Cook, had wrecked his throwing shoulder as a rookie in 1969, so Walsh built the offense around the quarterback he had, not the one he wanted. And Carter, who had studied statistics in college and took an MBA during his playing career, co-wrote one of the first academic papers to estimate the expected points value of field position: the same idea this book uses as its scorecard.6

In San Francisco from 1979, with Joe Montana, the approach became the backbone of three Super Bowl champions (after the 1981, 1984 and 1988 seasons).7 Its signature was the slant to Jerry Rice, thrown on rhythm before the cornerback could react and turned into a long touchdown by Rice’s speed: Rice caught a record 22 touchdown passes in just 12 games in 1987.8 The family of offenses descended from Walsh’s ideas came to be called the West Coast offense, a label with a tangled history that Offensive Systems and Languages untangles.

A rule change helped. In 1978 the NFL barred defenders from contacting receivers more than five yards past the line of scrimmage; until then a defensive back could keep bumping a receiver well down the field.9 Receivers who could no longer be mugged at eight and ten yards made short timing routes far more reliable, and passing numbers rose across the league. Walsh arrived in San Francisco a year later.

Rhythm: the drop and the throw are one motion

Watch a quick throw from the end-zone camera and you’ll notice that the quarterback never seems to decide anything. He catches the snap, takes his steps, and the ball is gone on the last one. That is a rhythm throw: a pass thrown on the last step of the drop, with no hitch or hesitation, so the ball arrives just as the receiver breaks.

The rhythm is the whole point. The quarterback’s drop and the receiver’s route are designed together, step for step: a three-step drop under center, or a one-step drop from the shotgun, is timed to a receiver who breaks at about five or six yards (Passing Fundamentals lays out the matched pairs). When the quarterback’s back foot hits, the receiver is making his cut, and the ball leaves. If the quarterback has to hitch (take an extra reset step) or look somewhere else, the timing is gone and a different, slower play has begun.

You’ll meet three different clocks in this chapter, and they measure different things:

  • About 1 to 1.5 seconds is the design: from the shotgun, a rhythm throw against a receiver who gets a clean release leaves the hand about then.
  • 2.5 seconds is the analysts’ line, the cut-off the data in this chapter uses to call a throw “quick.” It sits well past the design time because real plays lose a beat here and there: a receiver delayed by a jam, a quarterback who waits half a second for a defender to commit, a snap that comes late in his hands. A 2.1-second slant against press is still a quick-game throw; it just shows you the press cost the receiver a step.
  • “One-thousand-two” is the couch rule in the “Watch for it” box at the end: if the ball is gone before you finish saying it, you saw the quick game working as designed.

What does a late quick throw look like? The quarterback’s last step lands and the ball doesn’t come out: he hitches, or his shoulders turn to another receiver, and the slant runner is already two or three steps past his break, drifting into the linebackers. When the throw finally comes, the defender who was a step behind is now at the catch point. That is when you see the batted balls, the contested catches and the interceptions on short routes.

That has a consequence that shapes the whole quick game: the quarterback has almost no time to read the defense after the snap. He gets one look, at one defender, and has to know in advance what that defender’s answer means. Most of the “reading” therefore happens before the snap. To see how, we need one more piece of vocabulary.

Look at any cornerback before the snap and ask a simple question: is he lined up inside the receiver, outside him, or straight in front of him? That position is his leverage, and it is the most important thing a quarterback looks at before a quick throw.

Why would a defender give anything away on purpose? Because he can’t take everything away, so he takes away what his teammates can’t help him with. A defender usually lines up on the side where he has no help, and funnels the receiver toward the side where he does. A cornerback with a safety deep in the middle and linebackers underneath often plays outside leverage: his teammates can help on routes inside. A corner whose help is outside plays inside leverage, and that help can be a teammate sitting in the flat in front of him or simply the sideline. Coaches call the sideline the cornerback’s twelfth man, because a receiver near it has nowhere to go but inside or straight up. So leverage is not just one man’s position. It is a clue to the whole coverage behind him.

The receiver’s split, how far from the ball he lines up, feeds the same read from the other side. Split wide, near the sideline, he has little room to break outside, so the corner can shade him inside and let the sideline do the rest. A receiver who reduces his split, lining up closer to the ball, has room both ways, and a defender who sees a reduced split starts expecting an outside-breaking route.

The quarterback’s rule is short: throw away from leverage. A route that breaks toward the side the defender isn’t on gains a step at the break, because the defender has to cross the receiver’s path, or run around him, to get there. That’s the core of the pre-snap read: a look at the defense’s alignment before the snap (leverage, the cushion, meaning how far off the receiver the defender stands, the safeties, the box) that tells the quarterback which side, or which receiver, a play should go to. Many quick-game calls are drawn with a concept on each side of the formation for exactly this reason: the quarterback picks the side whose leverage he likes before the ball is snapped, then reads one defender on that side after it. The coordinator calls the play; the quarterback chooses the side, at the line, inside the rules the offense has practiced all week (many teams pre-set a default, such as “work the side with the extra receiver,” for when both look equal).

Three numbers belong in every pre-snap read of a single receiver:

  • Inside, outside or head-up? Head-up (straight in front of the receiver) is a defender who hasn’t given a side away. It is common in press coverage, and the release section below shows how a receiver deals with it.
  • How deep? A defender seven or eight yards off can’t stop a short throw from being caught; he can only tackle the catch. One pressed up at the line (in press coverage: a yard or so away, ready to jam the receiver at the snap) can disrupt the route itself.
  • Where is his help? A deep safety over the top lets a cornerback sit on short routes. No safety at all means the cornerback has to protect against the deep ball first.
CautionCommon misconception

“The quarterback reads the coverage after the snap.” For deep dropback passes, mostly yes. For the quick game, the important decision is made before the snap and the post-snap read is a single glance at a single defender. When you see a quarterback throw the instant his foot lands, he isn’t guessing; he made the decision ten seconds earlier, from the alignment, and the snap merely confirmed it.

Slant-flat: the high-low

Now the concepts. A pass concept is a set of routes designed to work together, usually to put one defender in a position where he can’t be right. Every quick concept in this chapter is built on one of the two stretches from Passing Fundamentals: a high-low (one receiver above a defender, one below him) or a horizontal stretch (one inside him, one outside him). The defender being stretched is the one the quarterback watches.

The simplest quick concept in football is slant-flat. The outside receiver runs a slant: three hard steps upfield, then a sharp break inside at about 45 degrees. The inside receiver runs to the flat, the area near the sideline just past the line of scrimmage. Between them stands one defender, usually the slot defender or an outside linebacker, whose job in most zone coverages is to guard the short area outside: the flat defender (or, in Cover 3 language, the curl-flat defender). He is the read.

The quarterback’s thought process is one sentence long: “If the flat defender widens, throw the slant; if he sits, throw the flat.” He doesn’t have to diagnose the coverage, find the safeties or work through five receivers. He has to watch one man take one step. That’s why slant-flat is often the first pass concept a young quarterback learns, and why it survives at every level of the sport.

Notice what each route is doing to the defender. The flat route is a low threat: it pulls him toward the sideline and forward. The slant is a high threat: it attacks the space behind him and inside him. It is the same high-low as the curl-flat read in Passing Fundamentals, squeezed into three steps. He can’t be in both places, and he has to commit in about half a second. If he widens, he has opened the door the slant runs through. If he stays put, or sinks into the slant’s path, the inside receiver catches the ball in the flat with no one within five yards, which on most downs is a fine result: a “long handoff” to the edge.

Stick: the in-out

Stick is one of the most-called quick concepts in the NFL, and you met it in What a Play Really Is as that chapter’s one example play. Here is why it is so popular. Slant-flat stretches the flat defender vertically; stick stretches him horizontally, inside and outside, at a depth that usually reaches the line to gain on second or third and medium.

Three receivers on one side. The outside man (#1) runs off, deep: a go route or a fade that carries the cornerback away. One man runs to about six yards, at or just past the line to gain, and sticks: he settles in the open space and turns to face the quarterback, or, against man coverage, breaks away from his defender. One man runs to the flat, getting to the numbers or wider. The stick sits inside the flat defender; the flat route goes outside him.

Which man runs which route depends on the formation, and you’ll see both versions on Sundays. In the trips version you met in What a Play Really Is (sometimes called Y-stick), the tight end is #3, the innermost receiver, and runs the stick, while #2 runs an arrow to the flat outside him. In the 2x2 version drawn below, the classic NFL drawing, the tight end is #2 and runs the stick, and the running back comes out of the backfield to the flat. The rule underneath is the same: the stick runner settles inside the flat defender, and the flat comes from whoever is outside or under him.

The stick route itself is the clever part. It isn’t a hitch with a fixed landmark; it’s a rule. Against zone, the receiver finds the soft spot between the flat defender and the next defender inside (the Mike linebacker here) and sits in it, facing the quarterback, so the ball can be thrown the moment he settles. Against man coverage, there is no soft spot to find, because a defender is following him, so he breaks away from that defender’s leverage, in or out. That’s an option route in miniature, and it is why the quarterback and the stick runner rehearse it endlessly: both of them have to see the same thing.

Stick is everywhere because it does three things at once: it puts the ball at or near the line to gain on the downs where five or six yards matter most; it works against nearly every zone, because nearly all of them have one defender responsible for both the area inside and the area outside at that depth; and the clear-out by #1 keeps a deep threat on the field, so the cornerback can’t come up and sit on everything short.

Snag is read against the coverage the quarterback identified before the snap. Against Cover 3, the look drawn here, the cornerback has a deep third and bails, so the corner route is covered before it starts. The quarterback’s eyes go straight to the flat defender, the $, exactly as in stick: if he widens to the flat, throw the snag inside him; if he sits on the snag, throw the flat. Against Cover 2, the cornerback squats in the flat with a safety deep behind him, and the read moves to the top of the triangle: if the corner squats on the flat route, the corner route goes over his head into the hole in front of the safety; if he sinks under the corner route, the flat is open.

So why call snag instead of stick? Against Cover 3 they are close cousins that read the same man. Snag’s extra is the corner route. Stick’s flat route runs straight into a Cover 2 cornerback squatting in the flat; snag throws over him. A quarterback who isn’t sure before the snap whether he will see Cover 3 or Cover 2 has an answer to both in one call.

Because the triangle combines both kinds of stretch, a single adjustment can’t take it away; the triangle read becomes a whole family in the next chapter, with deeper routes and more time.

Spacing: five windows

Every concept so far stretches one or two defenders. Spacing stretches the whole underneath coverage: four or five receivers settle at about the same depth, five or six yards, at even intervals across the field, each in the gap between two zone defenders. You saw the geometry in Passing Fundamentals’s horizontal stretch (five receivers against Cover 3’s four underneath defenders). As a called concept it adds two things.

First, one route goes underneath. Almost every spacing call sends one receiver, often the back or the innermost slot, on an arrow or flat route to the sideline at two or three yards. Without it, the curl-flat defender can sit at five yards on the hitch and nobody punishes him; with it, he is in a high-low as well. Second, the quarterback works one half of the field. He can’t watch five receivers in a second and a half, so before the snap he picks a side, usually the one where the offense has more receivers than the defense has underneath defenders, and after it he reads one man on that side.

Speed out: beating a soft corner to the sideline

Not every quick throw is a two-man read. The speed out is a single receiver’s route, thrown on timing against one particular look: a cornerback playing well off, with inside leverage, who is backing up at the snap. (Names vary by system: some playbooks call this five-yard rounded cut the quick out and save “speed out” for a deeper rounded out off a longer drop. The route here is the short one.)

Why would a cornerback line up inside in the first place? The leverage rule answers it: he lines up where he has no help. Against a two-high look nobody sits in the deep middle, so the routes that hurt him are the post and the dig, breaking inside toward all that open grass, while the sideline protects his outside. (In a quarters-style coverage there’s a second reason: the safety on that side is reading the slot, so the corner is effectively alone on Z, backing up to stay on top of him.)

Why does the speed out beat that? Because the cornerback’s alignment has already made a promise. Lined up inside and eight yards off, he is protecting the middle of the field and the deep ball, and the sideline in front of him is the one area he has conceded. A receiver who runs at him hard and rounds his break toward the sideline, without the throttle-down of a square cut, gets there first. The ball has to beat the corner’s break, which is why the speed out is a timing route: the quarterback lets it go before the receiver turns, to a spot, and trusts him to be there.

It is also the quick throw that most exposes a quarterback’s arm. The ball travels from the middle of the field to the sideline, the longest flight in the quick game, and a throw that floats lets a defender close on it. The dangerous defender is usually not the corner on top but one underneath: a Cover 2 cornerback squatting in the flat, or a Cover 3 curl-flat defender who sinks and widens under the out (the $ in the figure, if he reads it). An interception there has an open field in front of it, which is why coaches call the out a risky throw for a weak arm and why the speed out is a pre-snap call against one look, not a throw into any coverage.

Two more pairs: double slants and hitch-seam

Two last concepts round out the core of the quick game. Each pairs two receivers on one side.

Double slants: both receivers run slants, one shallower and quicker, the other a step deeper, so their paths stack one above the other as they cross in front of the linebackers. (Teams differ on which one goes underneath; many NFL playbooks have the inside man run the deeper slant instead.) Against zone, that is a high-low on the hook defender in the middle. Against man coverage, the version drawn below, it is even better. Two receivers break inside at the same moment, each defender is trying to keep his own man from getting across his face, and the one defender man coverage usually keeps in the middle as a helper (the “hole” or “rat” player, built in Man Coverage) faces the same high-low as a zone hook defender: he can drive on the low slant or sink under the high one, not both. The two paths also cross close to each other, and the defenders have to fight through that traffic; deliberately running into a defender to block him is a foul once it happens more than a yard past the line, the rub route question settled in the next chapter.11

Hitch-seam: the outside receiver runs a hitch (five or six yards, stop, turn back to the quarterback) and the inside receiver runs a seam straight up the field. It is a high-low on the flat defender. Against Cover 3 that defender is usually coached to carry a slot receiver who goes vertical, running up the field with him until the deep defender takes over, which is exactly why the hitch underneath him is usually the throw. The seam is the punishment if he doesn’t carry: it also threatens the deep middle safety, who can’t take a seam on both sides of the field. Offenses keep him honest by running seams from both slots, or by having the quarterback look the safety off (stare at one seam to pull him there) before throwing.

Seven concepts, and notice how few ideas they really contain. Five of them (slant-flat, stick, snag, double slants and hitch-seam) pick one defender, put two receivers on opposite sides of him (above and below, or inside and outside), and ask the quarterback to watch that one man and throw to the one he doesn’t take. Spacing does the same thing to the whole underneath coverage at once, with the quarterback choosing a half of the field before the snap. The speed out skips the post-snap read entirely: it beats one cornerback’s alignment. In every case the quarterback’s eyes are on one player or one alignment, which is why he can run the quick game at full speed.

Beating press: the receiver’s side

Everything so far assumed the receivers get into their routes on time. The defense’s most direct answer to the quick game is to make sure they don’t. In press coverage, a cornerback lines up right on the receiver, a yard or so away, and tries to jam him at the snap: punch him, reroute him, knock him off his path. Its opposite, off coverage, is a defender who starts five to ten yards away and gives ground (Coverage Fundamentals covers both from the defender’s side).

Press attacks the quick game where it lives: the timing. A slant that should break at three steps breaks at five because the receiver spent two of them fighting through a jam, and the quarterback, whose back foot has already landed, is left holding the ball with a rush closing in. That is why press and the quick game are a matched pair in every game plan: the defense presses to wreck the rhythm, and the offense answers with receivers who can beat press quickly and routes that use the press defender’s own position against him.

The rules frame the fight. Within five yards of the line of scrimmage, an NFL defender may contact a receiver in front of him; beyond five yards, until the ball is thrown, contact that impedes the receiver is illegal contact or defensive holding.12 Either foul costs the defense five yards and gives the offense an automatic first down, which on third-and-8 is a gift. So press is a five-yard fight, and the receiver’s goal is to get through those five yards on his path and on time. Here is how he does it, from the receiver’s side of the ball.

Release: getting off the line

A release is a receiver’s first moves off the line against a pressing defender: footwork to make the defender commit, and hands to keep the defender’s hands off him. The defender wants to get his hands on the receiver’s chest and mirror him; the receiver wants to make him move first, then go the other way.

The inside and outside releases are mirror images of one idea: make the defender declare first. Attack one side hard enough that he has to move his feet or reach to stop it, then go to the other side. The hands matter as much as the feet: receivers swipe the defender’s hands down, chop at his wrists or rip an arm through under his, to keep him off the chest. A defender who lands a two-hand jam controls the receiver; one who whiffs has to turn and chase.

Which release a receiver uses is not his whim. On a quick route, the route decides. A three-step slant needs an inside release, a fade or go route an outside one, because there is no time to go one way and come back. (On longer routes there is, which is why panel 3 shows a receiver releasing outside and breaking inside: the stem buys the time.) That’s why the pre-snap read includes the press corner’s leverage: a corner pressing with inside leverage is daring the receiver to win inside, so the quarterback may prefer the fade or a quick out to that side, and the slant to the other.

A corner pressing head-up, square in front of the receiver, hasn’t given a side away; he plans to mirror whichever way the receiver goes. The receiver’s answer is to make him declare anyway: a quick flurry of short, choppy steps in place, or a jab one way and then the other, until the corner’s hips or hands commit, and then a release to the side he left. Against a patient head-up corner that takes an extra beat, so quarterbacks often treat that matchup as a coin flip and look for a better one on the other side.

Stem and stack: setting up the break

Once he is off the line, the receiver still has to get open at the top of the route. The stem is the part of the route between the release and the break, and good receivers treat it as a weapon rather than a run-up. By aiming the stem at one of the defender’s shoulders, the receiver tells him what the route “is”: run at his outside shoulder at full speed and the defender has to worry about the go route and turn his hips to run, which makes a break inside much harder for him to match.

Against a defender trailing in man coverage, the receiver also wants to stack him: get directly in front of him, on his line, so the defender is behind the receiver and can’t see or reach the ball without going through him. A stacked defender can’t undercut a throw, can’t play the receiver’s hands, and has to guess which way the break will go. The best route runners win most of their battles here, in the yard or two between “released” and “broken,” and that’s why coaches say a route is won before the break, not at it.

The break and the catch point

The break is the cut itself, and its quality decides how much separation the receiver has when the ball arrives. A sharp break (sink the hips, plant, explode the other way) creates space against a defender who is mirroring. A rounded break keeps speed, as in the speed out, and works against a defender who is backing up. Either way, on a quick throw the ball is usually already in the air.

Then comes the catch point: the moment the ball and the receiver’s hands meet. If the release, stem and break were good, it’s an easy catch with the defender a step or two away. If not, it’s a contested catch: a catch made with a defender close enough to touch the ball or the receiver’s hands. FTN’s charters flagged about 14% of 2025 targets as contested; only about 42% of those were caught, against about 71% of the rest.13 The quick game is designed to avoid contested catches: a well-run slant-flat or stick throws to the receiver who is not covered. When a quick throw becomes a contested catch, usually against press-man coverage with no open receiver, the offense is relying on a receiver who can win at the catch point, and on a quarterback willing to throw him the ball anyway. Throws placed where only the receiver can reach them, like the back-shoulder throw, are how good pairs make those catches more likely.

The double move: punishing the defender who guesses right

The quick game creates its own counter. A cornerback who has watched film of an offense throwing slant after slant and hitch after hitch learns to sit on those routes, and to jump them the instant the receiver breaks. That is exactly the defender a double move is built for. The receiver fakes a quick route, the quarterback often pump-fakes (starts his throwing motion without letting go) as if throwing it, and when the defender drives on the fake, the receiver turns up the field behind him. The common forms are named for their two parts: slant-and-go (the sluggo), out-and-up, hitch-and-go, the stutter-go (a fake stop), and the stick-nod, in which the stick runner fakes settling and then breaks up the field, the counter to a flat defender who has learned to drive on the stick.

The double move is the quick game’s insurance policy: without it, a defense could squeeze the short routes harder every week; with it, every corner who jumps a slant risks a long touchdown. The cost is time. A double move takes three seconds or more, so offenses tend to call it with an extra blocker, after they have thrown the short version often enough to bait the defender.

Quick screens: now and bubble

The quick game has two close cousins that aren’t really passes in the usual sense. Both are screens, throws to a receiver behind or at the line of scrimmage with teammates blocking in front of him, and both are thrown the instant the ball is snapped. (Slower screens to running backs and tight ends, with linemen releasing to block, are in Play-Action, Bootlegs, and Screens.)

A now screen goes to an outside receiver. He takes a step back (or none) and catches the ball a second after the snap, while the receiver inside him blocks the nearest defender. Teams call versions of it a smoke screen or a quick screen; the names vary by playbook (some use “smoke” for the version with no blocker at all), but the idea is the same.

The now screen exists to punish one thing: a defender playing far off the ball. An eight-yard cushion is eight yards the defender has to cover before he can make a tackle, and a receiver who catches the ball at the snap gets to spend those yards running instead. Many offenses attach the now screen to ordinary plays as an extra option (coaches say they tag it on), so that the quarterback, seeing the cornerback backed up eight yards, can throw it as a free five yards instead of running the called play. The ball comes out at once for a rules reason too: the receivers who block for it may not block a defender more than a yard past the line before the pass is thrown, the same one-yard limit that governs rubs.

The bubble screen is the inside version. A slot receiver drifts a step or two back and then out toward the sideline, a path that looks like a bubble when you draw it, and catches the ball already running, while the receivers outside him turn and block.

Coaches who throw these screens on almost every snap, as many college offenses do and some NFL offenses do, often teach the decision with three words: numbers, leverage, grass.

  • Numbers: count the defenders outside the box on one side and compare them with the offense’s players there. Three against two is a screen. This is the box count seen from the other side: every defender a defense puts in the box to stop the run is one fewer outside it.
  • Leverage: can the blockers get between their defenders and the ball carrier? A defender lined up inside, near the ball, is easy to block from the outside; one lined up wide is not.
  • Grass: how much space is there between the ball carrier and the nearest unblocked defender? A safety twelve yards deep is twelve yards of grass.

That is why quick screens are often thought of as part of the run game, and why they are so often packaged with runs. If the defense puts a seventh or eighth man in the box to stop the run, someone outside the box has left, and the screen to that side gets better. When the quarterback decides between the run and the screen after the snap, by reading a defender, the play becomes a run-pass option, and the bubble attached to a run is one of the most common RPOs in football (bubble RPO).

One rule detail makes the bubble unusual. Because it is often thrown sideways or slightly backward, a bubble that is thrown behind the receiver is a backward pass, and a backward pass that hits the ground is a live ball, not an incompletion: either team can recover it.14 That’s why you sometimes see a defender scoop up a dropped bubble and run the other way, and why offenses want the throw to be clearly forward or clearly caught.

The answer to press and blitz

Now put the chapter together. The two most aggressive things a defense can do against the pass are to blitz (send more rushers than the offense can comfortably block) and to press (jam receivers to destroy the timing). Both are bets that the quarterback won’t get the ball out in time. The quick game is the offense taking that bet.

The arithmetic of a blitz explains why. A five-man pressure sends five rushers. Keep six men in to block (the five linemen and the back) and the offense can pick up all five with a man to spare. Keep only five and it can still block five rushers on paper, but only if the protection is aimed at the right five. Protections are set before the snap by counting the defenders who might rush (Pass Protection’s slide and Mike identification). If the line slides toward two linebackers and one of them drops into coverage while a defender from the other edge comes, one blocker is left with nobody and one rusher has no blocker at all: a free rusher. Send six against five and somebody is free no matter what. Either way the protection’s rule is that the free man belongs to the quarterback. He is “hot”: the ball has to come out before he arrives, on a hot route or a sight adjustment aimed at the area the blitzer just left.

Behind the blitz, the coverage is thin. It is often man coverage, pressed, with one safety deep or none, or a fire zone (three deep defenders and three underneath, built in Zone Blitz and Simulated Pressure). In either case the blitzer’s old spot is empty. A quick throw attacks every part of that: it beats the free rusher to the quarterback, and it goes to a receiver who is alone in coverage or running into the hole the blitzer left.

Three details in that play are worth slowing down for. First, the quarterback knew before the snap where the ball was likely to go. Pressed corners with a single safety deep is a classic pressure look, and the slot defender creeping toward the line told him which area might empty. His pre-snap plan was “if the $ comes, the ball goes to X, inside, right now.” Second, he confirmed it with one look after the snap. The $ might have been bluffing, showing a blitz and then backing out into coverage; the moment he crossed the line of scrimmage, the plan was on. Had he backed out, the quarterback would simply have run the play’s normal read. Third, the slant was the right route for the press cornerback’s leverage: he lined up on X’s outside, so an inside release was the release X could win. That release cost X a beat (against off coverage the same slant would come out around 1.2 to 1.5 seconds), which is why this ball left at 2.1: still on time for X, still ahead of the blitz.

The chart puts a number on the trade.15 Against a standard four-man rush, a quick throw was worth about an eighth of a point more than a slower outcome. Against five or more rushers, the gap roughly doubles: quick throws gained about +0.23 expected points per dropback, while everything slower, with the sacks that a blitz produces, averaged about -0.10. A blitz is a coin flip the defense is offering, and a quarterback who gets the ball out before the extra rusher arrives wins the flip far more often than he loses it.

So why do defenses blitz at all, on about one dropback in four? Because pooled over every outcome, the bet is roughly even. In 2025, dropbacks against five or more rushers averaged about +0.05 expected points for the offense, against +0.04 for four or fewer: a hair worse for the defense, within the noise, but with more sacks (about 7.8% of dropbacks against 6.0%). And blitzes come more often on third down (about 30% of dropbacks, against 25% on first and second), where forcing a short throw that falls short of the line to gain is a win in itself. The blitz is a variance play: it produces sacks and hurried incompletions, and it gives up the quick completion into the space it vacated. That is why blitz-happy defenses spend so much effort disguising when and where the pressure will come from (Blitz and Man Pressure does the math from the defense’s side; Zone Blitz and Simulated Pressure the disguises).

NoteMadden vs. real life

In a video game, the blitz is a button you press to make the other player panic. In the NFL, a quarterback who sees six defenders at the line is usually not panicking; his offense has spent the week on what to do when that happens. Every quick concept is installed with its blitz answer: which receiver becomes hot (the ball goes to him at once, often on a shortened version of his route) if a particular defender comes, which routes become sight adjustments, and which rusher, if any, the protection call leaves unblocked. The real contest is whether the defense can hide the blitz until it’s too late to adjust, or show one and not bring it. That’s why the pre-snap dance of defenders walking up and backing out, covered in The Pre-Snap Phase, matters so much.

Film room: three offenses that lived on it

The quick game is in every playbook, but a few offenses built their identity on it. Each of the three below shows a different reason to.

NoteFilm room: New Orleans Saints 2017–2019

The quick game as accuracy. Late in Drew Brees’s career, the engine of Sean Payton’s offense (Payton was head coach 2006–2021, Brees the quarterback 2006–2020) was the short, on-time throw. Brees broke the NFL’s single-season completion-percentage record in 2017 (72.0%) and broke it again in 2018 (74.4%).16 In 2019, among quarterbacks with 300 or more attempts, he had the second-fastest average time to throw and one of the shortest average throws (about 6.5 yards), and Michael Thomas set the single-season receptions record with 149, about seven in ten of them on throws that traveled less than ten yards in the air.17

What to notice on the tape: the ball leaving on Brees’s last step, again and again, and Thomas getting inside the cornerback on the slant, then shielding the ball with his body at the catch point.

NoteFilm room: New England Patriots 2007–2012

The quick game as an option tree. The Tom Brady–era Patriots ran it through a small, quick slot receiver. Wes Welker caught 112 passes in 2007, 123 in 2009 and 122 in 2011, leading the league three times in five years,18 most of them within a few yards of the line, many on option routes in which he read his defender’s leverage and broke away from it, the same rule the stick route uses. Brady’s clock stayed fast long after: in 2022, at 45, he had the fastest average time to throw of any regular starter.19

What to notice on the tape: Welker inside, a defender over him trying to guess his break, and Brady throwing as Welker turned his head, before the break had finished.

NoteFilm room: Kansas City Chiefs 2022–2024

The quick game as an adaptation. In 2018, Patrick Mahomes’s first season as the starter, the Chiefs threw deep more than almost anyone, with Tyreek Hill as the threat. Defenses adjusted first. Their throws were already getting shorter in 2019, 2020 and 2021, with Hill still on the roster, as opponents played two deep safeties to take away the big play: by 2021, Kansas City saw a two-high shell on about 46% of its dropbacks, third-most in the league.20 Then Hill was traded to Miami in March 2022,21 and over the next three seasons Andy Reid rebuilt the offense around shorter throws to Travis Kelce and a rotating cast of receivers. By 2023 and 2024 the Chiefs’ average throw was the second-shortest in the league (the chart after this box shows the whole slide). They won the Super Bowl after the 2022 and 2023 seasons anyway.

What to notice on the tape: Kelce, lined up in the slot or as the innermost receiver of trips, running a stick or an option route at five to eight yards and settling between two zone defenders, with Mahomes throwing as he turns. Watch for it on Sunday

TipWatch for it: one-two-throw

Before the snap

  • The corners. Off or pressed? Inside or outside the receiver? A corner seven yards off with outside leverage is inviting a slant or a now screen; a pressed corner is daring the receiver to beat him off the line.
  • The slot defender. Where is the man between the slot receiver and the box? He is the read on almost every quick concept to that side. If he walks up toward the line, think blitz, and think quick throw.
  • Count outside the box. Three receivers against two defenders on one side, with the safety far away, is a screen waiting to happen.

At the snap: the stopwatch. Count “one-thousand-one, one-thousand-two” from the snap. If the ball is gone before you finish “two,” you just watched the quick game working as designed. Count a dozen dropbacks in a game and you’ll find that roughly half of them are out before “two-and-a-half.”

At the snap: watch the linemen. Three plays come out in about a second, and the linemen tell them apart. On a quick throw they set to pass-block, short and aggressive, and stay at the line. On a run-pass option they fire forward as if blocking a run. On a now or bubble screen the receivers near the ball turn and block.

After the throw

  • On the replay, find the one defender between the two receivers on the side the ball went. Did he widen, sink or sit? The ball went to the receiver he didn’t take.
  • When the defense blitzes, watch where the ball goes. Usually it goes quickly to the area the blitzer came from.

For your prediction log (How to Watch a Broadcast): before each third-and-short-to-medium, write quick or not quick, and the side you expect, based on the corners’ leverage. Then time the throw.

Count the threat. What kind of throw does the offense want, how fast, and where?

TipAnswer

A hot throw, out in about a second and a half, and away from the cornerbacks’ leverage. Count first. Nobody is deeper than about six yards: this is a “zero” look, with no deep safety at all (Cover 0). Zero lets the defense put a man on every receiver and send everyone else. With five receivers out, that is six rushers against five blockers. If the back stays in to block, his defender can come too: seven rushers against six blockers. Either way one rusher is free, and nobody can hold the ball. That is the Cover 0 blitz bet: the defense is wagering that the ball can’t come out in time.

Now look at the cornerbacks. Both are pressed and both are a yard inside their receivers. That is the leverage rule at work: with no safety anywhere, the corner has no help inside, so he protects the middle and uses the sideline as his help. The slant, the quick game’s favorite throw, is the route he is sitting on. So the ball goes away from his leverage, outside. Two answers fit:

  • The outside receivers’ fade or back-shoulder throw. With no safety behind the corner, a receiver who wins outside off the line is one catch from a touchdown, and the quarterback can throw it as soon as the receiver clears the jam. It’s not a high-percentage throw (a pressed corner is right on the receiver), which is why it isn’t the answer every time.
  • The hot route into the space a blitzer leaves. Some of the defenders crowding the line will cover and some will rush. The receiver whose nearest defender rushes is hot: he breaks into the space that defender just left, and the ball goes there at once.

Which side? The side the protection call says the free rusher is coming from, because that is where the hot receiver is and where the quarterback has the least time. A now screen is a poor choice: the cornerback is standing a yard away, so there is no cushion to run into.

For a real one, see the Super Bowl LVII touchdown to Skyy Moore in the Chiefs film room above: seven rushers, Cover 0, ball out in 1.7 seconds to the flat.

The defense might be bluffing, of course: show zero, bring four, drop two into the throwing lanes. That is exactly why the throw is a read of where the extra defenders went, not just a reflex (Zone Blitz and Simulated Pressure).

The defense has loaded the box to stop the run. What quick throw does that invite, and why?

TipAnswer

A quick screen to the trips side: a bubble to H, or a now screen to Z. Start by counting. The box (roughly the area between the offensive tackles, out to about five yards deep) holds seven defenders: four linemen, two linebackers and a safety. Outside it on the right there are only the cornerback and the $. Then run the three questions:

  • Numbers: on the right, outside the box, the offense has three players (Y, H and Z) and the defense two. The safety who would usually be the third defender out there is standing beside the linebackers, in the box. That’s the trade the defense made: an extra run defender inside, one fewer pass defender outside.
  • Leverage: the cornerback and the $ are both off and roughly over their receivers, so the blockers start in the right place: Z can turn the cornerback out toward the sideline, and Y, starting inside the $, can block him from the inside while H runs between them.
  • Grass: both are about seven yards off, and the nearest help, the deep safety, is fourteen yards away in the middle of the field. A bubble caught at the snap is five yards gained before anyone can touch it.

A run into the middle would face seven defenders with five linemen to block them (six if a receiver comes in to help): the defense has the numbers inside. The quick screen is how the offense “runs” outside the box instead, and it’s why so many offenses attach a bubble to their running plays (Run-Pass Options).

Takeaways

  • The quick game is the passing game’s fast lane: a short drop, short routes and a rhythm throw, out in about 2.5 seconds or less. It is roughly half of what NFL offenses throw.
  • It exists for three jobs: beat the rush with the clock, keep the offense on schedule with high-percentage throws, and stretch the run game sideways by getting the ball to fast players in space. It trades depth for certainty and needs yards after the catch and a deep threat to keep the bargain.
  • Most quick concepts put two receivers on opposite sides of one defender, high and low (slant-flat, hitch-seam, double slants) or inside and outside (stick), or both (snag’s triangle), and the quarterback throws to the one that defender doesn’t take. Spacing stretches the whole underneath coverage; the speed out beats one cornerback’s alignment.
  • Most of the decision is made before the snap: read the defender’s leverage, cushion and help, and throw away from leverage.
  • Press is the defense’s answer to the quick game’s timing; the receiver’s answer is the release, stem, stack and break, and the double move punishes a defender who jumps the short route.
  • Now and bubble screens punish soft and outnumbered alignments (numbers, leverage, grass). Against the blitz, the free rusher is the quarterback’s, and the ball that comes out fast, into the space the blitzer left, is worth the most.
