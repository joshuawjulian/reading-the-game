def no_numbers(ax):
    """Remove the painted yard-line numbers (they sit where receivers stand and crowd the labels)."""
    for t in list(ax.texts):
        if t.get_alpha() == 0.55:
            t.remove()


def ruler(fld, play, top=40, fs=6.5):
    """Depth labels (yards past the line of scrimmage) down the left edge of the window, so the
    faint 5-yard lines read as a scale. gridiron has no ruler, so it is drawn here."""
    x_left = FIELD_W - fld.y1 + 0.4
    for d in range(5, top + 1, 5):
        y = play.los + d
        if fld.x0 + 0.8 <= y <= fld.x1 - 0.8:
            fld.ax.text(x_left, y, f"{d}" + (" yd" if d == 5 else ""), fontsize=fs, color=style.INK_2,
                        ha="left", va="center", zorder=12,
                        bbox=dict(boxstyle="round,pad=0.12", fc=style.SURFACE, ec="none", alpha=0.85))


def tag(fld, play, text, at, ha="center", fs=7.0, ec=style.BASELINE, color=style.INK, bold=False):

def protect(p, rushers=("SDE", "SDT", "WDT", "WDE")):
    """Five linemen set and four rushers engage them: enough to say 'pass', no more."""
    for k in ("LT", "LG", "C", "RG", "RT"):
        if k in p.players:
            p.block(k, pts=[(-1.2, 0.0)])
    for k in rushers:
        if k in p.players:
            p.rush(k, pts=[(-1.6, 0.0)], speed=2.0)

def snapshot(play, t, ax=None, lateral=(-6, 26.6), window=(-9, 20), width_in=6.5, notes=(), rings=(),
             show_ruler=False, numbers_off=True):
    """Everyone where he is at time t, faint trails of where they came from, and the ball in flight."""
    pos = {k: play.position(k, t) for k in play.players}
    lo, hi = sorted(lateral)
    moved = [pl.moved(d=float(pos[k][0]), w=float(pos[k][1])) for k, pl in play.players.items()
             if lo + 0.8 <= float(pos[k][1]) <= hi - 0.8]          # drop anyone outside the window
    q = Play([m for m in moved if m.side == "off"], [m for m in moved if m.side == "def"],
             los=play.los, ball_y=play.ball_y, to_go=play.to_go)
    q.highlights = dict(play.highlights)
    for k in rings:
        q.highlight(k)
    fld = q.draw(ax=ax, lateral=lateral, window=window, width_in=width_in, title=False)
    ax, ppy = fld.ax, fld.pts_per_yard()
    if numbers_off:
        no_numbers(ax)
    if show_ruler:
        ruler(fld, play, fs=5.5)
    P = lambda d, w: fld.xy(*play.to_abs(d, w))  # noqa: E731
    t0 = play._compile()[1]
    for k, pl in play.players.items():
        if pl.pos in ("LT", "LG", "C", "RG", "RT", "DE", "DT", "NT"):
            continue
        pts = np.array([play.position(k, tt) for tt in np.linspace(t0, t, 40)])
        if np.ptp(pts[:, 0]) + np.ptp(pts[:, 1]) < 0.3:
            continue
        xs, ys = P(pts[:, 0], pts[:, 1])
        if pl.side == "off":
            ax.plot(xs, ys, color=style.OFFENSE, lw=1.2, alpha=0.5, zorder=3)
        else:
            ax.plot(xs, ys, color=style.DEFENSE, lw=1.3, ls=(0, (3, 2)), zorder=3)
    bx, by = P(*play.ball_position(t))
    thrown = False
    for kind, _, t_rel in play.ball_events:
        if kind == "pass" and t > t_rel:
            thrown = True
            sx, sy = P(*play.ball_position(t_rel))
            ax.add_patch(FancyArrowPatch((float(sx), float(sy)), (float(bx), float(by)), arrowstyle="-|>",
                                         mutation_scale=8, color=style.BALL, lw=1.4, ls=(0, (2, 1.5)), zorder=9))
    if not thrown and t > 0 and "QB" in play.players:
        # still in the quarterback's hands: draw it beside him (on the emptier side), not on his "Q"
        qd = play.position("QB", t)
        others = [pos[k] for k, pl in play.players.items() if pl.side == "off" and k != "QB"]
        cands = [qd + np.array([0.0, 1.3]), qd + np.array([0.0, -1.3])]
        best = max(cands, key=lambda c: min(float(np.linalg.norm(c - o)) for o in others) if others else 0)
        bx, by = P(*best)
    draw_marker(ax, float(bx), float(by), "ball", "", ppy)
    for text, at in notes:
        tag(fld, play, text, at, fs=6.3)
    return fld


def strip(plays, times, labels, lateral=(-6, 26.6), window=(-9, 20), notes=None, rings=(), width_in=6.5,
          ncols=3, legend=True, rulers=True):
    """Panels at chosen moments. `plays` is one Play (used for every panel) or one per row."""
    n = len(times)
    nrows = int(np.ceil(n / ncols))
    if isinstance(plays, Play):
        plays = [plays] * nrows
    pw = width_in / ncols
    h = nrows * (pw * (window[1] - window[0]) / abs(lateral[1] - lateral[0]) + 0.34)
    fig, axes = plt.subplots(nrows, ncols, figsize=(width_in, h), squeeze=False)
    for ax in axes.flat[n:]:
        ax.axis("off")
    for i, (ax, t, lab) in enumerate(zip(axes.flat, times, labels)):
        snapshot(plays[i // ncols], t, ax=ax, lateral=lateral, window=window,
                 notes=(notes or {}).get(i, ()), rings=rings, show_ruler=rulers and (i % ncols == 0))
        ax.set_title(lab, fontsize=7.8, color=style.INK, loc="left")
    fig.tight_layout(h_pad=0.5, w_pad=0.3)
    if legend:
        h_ = [Line2D([], [], color=style.OFFENSE, lw=1.2, alpha=0.6, label="Receiver's path so far"),
              Line2D([], [], color=style.DEFENSE, lw=1.3, ls=(0, (3, 2)), label="Defender's path so far"),
              Line2D([], [], color=style.BALL, lw=1.4, ls=(0, (2, 1.5)), label="Ball in flight")]
        fig.legend(handles=h_, loc="upper center", bbox_to_anchor=(0.5, 0.0), ncol=3, fontsize=7.5, frameon=False)
    return fig

Second-and-8 near midfield. The quarterback takes the snap under center, takes five quick steps backward, plants, hops forward once, and throws toward the sideline, to a patch of grass where nobody is standing. The receiver he is throwing to is still sprinting straight up the field, with the cornerback backpedalling in front of him. A beat later the receiver snaps his head around, cuts hard toward the sideline, and the ball arrives at that patch of grass at the same moment he does, a step ahead of the cornerback. The analyst says “that’s timing.” How did the quarterback know where to throw before the receiver got there? Why five steps and not three or seven? And when the cornerback does get there first, how does he decide to throw somewhere else? This chapter is about the three things that make every pass play work: the routes (where the receivers go), the drop (when the ball comes out), and the read (who gets it).

You’ll be able to…

  • Name every route on a replay, by its common name and its number in the route tree, including the routes that live off the tree: seam, wheel, whip, option and sluggo.
  • Count a quarterback’s drop, from under center or from the shotgun, and say roughly how deep the throw is meant to be and when it should come out.
  • Tell a progression read from a defender read, and a half-field read from a full-field read.
  • Find the conflict defender in a high-low (vertical) stretch or a horizontal stretch, and say which receiver he left open.
  • Read air yards, YAC and aDOT in a box score or a chart, and say what each one tells you about an offense.

The passing problem: a race against the rush

A pass play has a clock on it, and the defense controls it: four or more rushers start running at the quarterback at the snap, and his linemen can’t hold them off forever. In the 2025 regular season the median pass left the quarterback’s hand 2.5 seconds after the snap.1 Be careful what that number means. It is not how long the line can hold; plenty of deep throws leave after three seconds, so a good pocket often lasts longer, and Pass Protection measures it. It is how long quarterbacks choose to hold the ball, and they choose short because every extra half-second is a gamble that the rush hasn’t arrived. Two and a half seconds is the working budget: the time for receivers to get somewhere useful, for the quarterback to decide, and for the ball to leave.

Inside that budget the offense solves three problems at once. Where: each receiver needs a path to open grass, or to a spot where his defender is at a disadvantage; that is his route. When: the quarterback must be ready to throw the moment the receiver gets there, not before (the defender is still close) and not after (the defender has recovered, or the rush has arrived); his footwork, the drop, synchronises the two. Who: up to five players can go out, and the quarterback needs a plan for choosing one in a second or two; that plan is the read. Every pass concept in the rest of Part 4 is a particular answer to those three questions.

NoteMadden vs. real life

In the video game a pass play is five fixed lines, and you choose a receiver by pressing his button whenever you like. In a real offense a route is a set of instructions that can change with the coverage and the down and distance, and the quarterback can’t wait until a receiver looks open. By the time a receiver looks open on a television screen, it is usually too late to throw it to him. Real quarterbacks throw to where a receiver is about to be open, on a count their feet keep.

Anatomy of a route

A route has four parts.

  • The release. His first steps off the line, getting past the man in front of him. Against a press cornerback standing nose to nose with him, who tries to jam him (hit him with both hands and knock him off his path), the release is a fight of its own; The Quick Game covers the release techniques.
  • The stem. The run upfield before the cut. A good stem looks the same for every route, so the defender can’t tell from it which way the receiver will break.
  • The break. The cut. Its depth is set by rule (“break at 12 yards”, measured from the line of scrimmage, not from where he lined up), and its direction names the route. Breaks come in two kinds. A speed break is rounded, with no plant, to keep the receiver’s speed; it suits quick routes against defenders guarding areas. A sink-and-plant break drops the hips and stops hard, to shake a defender running with him; deep outs and comebacks use it, and many coaches have the 12-yard out break slightly back toward the quarterback rather than dead flat.
  • The finish. What happens after the cut: keep running away from the defender, stop and face the quarterback, or settle in a gap between defenders.

The break is the moment that matters. A receiver can’t outrun a defender who is backpedalling in front of him, but he can make that defender guess. At the break the receiver knows where he is going and the defender doesn’t, so for a fraction of a second the receiver is ahead. That gap, a yard or two of separation, closes fast. Everything about timing is about getting the ball there before it does.

The route tree: a shared language

Receivers and quarterbacks need to talk about routes quickly, in a huddle or with a hand signal at the line. Most offenses therefore teach a route tree: a numbered menu of the basic routes, drawn as branches off one straight stem. Two patterns make the tree easy to remember. Odd numbers break toward the sideline and even numbers break toward the middle, and the numbers roughly grow with depth. A receiver on the left simply mirrors the tree, so “breaks outside” always means “toward his own sideline.” Each branch exists for a reason.

  • 1, the flat route. A quick release to the flat, the strip near the sideline within a few yards of the line. Its job is often to pull a defender down and out with it, opening the space above him.
  • 2, the slant. One to three steps upfield, then a sharp cut inside at about 45 degrees, between the defenders near the line and the ones behind them, with the receiver’s body shielding the ball from a defender outside him. It is one of the quickest routes in football, which makes it a favourite hot route: when the defense sends more rushers than the line can block, the quarterback throws at once to a receiver running into the space a rusher just left (Pass Protection).
  • 3, the comeback. A hard push for about 15 yards to sell a deep route, then a stop and a turn back toward the sideline. The more convincing the push, the further a cornerback afraid of the deep ball retreats, and the more room there is underneath.
  • 0 and 4, the hitch. Push upfield, stop, and turn back to face the quarterback. At 5–6 yards it is a hitch (some trees number it 0, many not at all); at 10–14 yards, turning back slightly toward the middle, it is a curl, the 4. Both attack a defender who is backing up: against a cornerback 7 or 8 yards off, a 5-yard hitch is open almost by definition.
  • 5, the out. Straight up the field, then a cut toward the sideline: a quick out at about 5 yards, the deeper out at 10–14. It attacks the area outside the defenders who guard the middle, and the sideline stops the clock.
  • 6, the dig. Up the field 10–15 yards, then a flat cut toward the middle, behind the linebackers; shallower versions are often called an in route.
  • 7, the corner route. Up 10–12 yards, then an angle toward the sideline and the pylon at the corner of the end zone (hence its other name, the flag), above a cornerback who stays low and outside a safety who stays in the middle.
  • 8, the post. Up the field, then an angle deep toward the goalposts in the middle, between deep defenders or behind a safety who has come forward. One that angles only slightly inside is a skinny post.
  • 9, the go route. A straight sprint, also called a fly or a 9 route, to outrun the man covering him. Even unthrown it does a job: a cornerback running with it is no longer near the short routes. A go that drifts toward the sideline is a fade, and a ball thrown short to his sideline shoulder, so he turns back for it while the defender keeps running, is a back-shoulder throw (Red Zone, Goal Line, Two-Point Plays, and Short Yardage).

Where the ball sits changes the menu. The ball is spotted on or between the hash marks, so one side of the formation is usually wider (the field side) and the other shorter (the boundary side). Outs and comebacks usually go to the boundary, where the throw is shorter. A deep out from the far hash to the wide side travels some 30 yards in the air before the cornerback can come down on it, the classic test of a quarterback’s arm. Receivers also keep a sideline rule, finishing their routes about 5–6 yards inside the sideline so an out, a comeback or a fade has room to land, which is one reason they move their alignment from play to play, out to the numbers or in to a reduced split (Formation Rules and the Language of Alignment).

Read the tree in pairs. A receiver runs one route per play, but his defender has to defend all of them, and many routes are built to look like another for as long as possible. The slant and the sluggo (a slant fake, then up the field) start the same way; so do the hitch and the hitch-and-go, and the post and the post-corner. A cornerback who jumps the first route all afternoon gets the second. Pairs work across a game, as calls; inside one route, as a double move; and across two receivers, as the stretches later in this chapter. The idea is the same: make a defender choose, then throw where he didn’t go.

ImportantWhy it exists

The tree makes calls short and practice efficient. If every route has a number and a fixed depth, one digit tells a receiver where to go, and a quarterback and receiver who have run the 5 a thousand times know, without talking, where the ball will be and when. The cost is rigidity: a number says nothing about how two routes work together, and a fixed depth is easy to study. So modern playbooks name combinations of routes (concepts), and teach many routes with rules that change them against different coverages (Offensive Systems and Play-Calling Languages).

Who numbered the routes, and what a call sounds like

The numbered tree is usually traced to Don Coryell, whose “Air Coryell” San Diego Chargers (1978–1986) built on Sid Gillman’s principle of making a defense cover every yard of the field, and led the NFL in passing yards six straight seasons from 1978 to 1983 and again in 1985. Numbers let a play-caller give three receivers their routes in three digits. In the classic version the digits go to the split end, the tight end and the flanker, in that order, so “896” sends X on an 8, Y on a 9 and Z on a 6: with this chapter’s tree, a post, a go up the seam and a dig.2 It never became a standard; some trees reverse the odd-even rule or swap the 3 and the 5. “A 9 route” almost always means a go; most other numbers depend on who is talking. Offensive Systems and Play-Calling Languages builds the digit system into full calls.

The names travel better, with dialects: a go is also a fly, a streak or a 9; a dig a square-in; a hitch a stop; a corner a flag; a skinny post a glance or a bang-8; an option route a choice; a high-low a hi-lo or “levels”. Some systems split the hitch family into a hook, which turns straight back, and a curl, which rounds inside.

CautionCommon misconception

“The receiver runs the route as drawn.” Routes are drawn as lines but taught as rules, and the depths in the tree are targets, not laws. A 12-yard out on third-and-14 is run at 14 yards, past the line to gain. A receiver who sees a defender sitting on his spot may be taught to adjust, and on some routes the adjustment is the whole point (the option routes below). The quarterback has to know the rule as well as the receiver does.

Off the tree: seam, wheel, whip and sluggo

The tree describes what one wide receiver can do from the outside, with the sideline on one side of him and the whole field on the other. A receiver inside, in the slot or at tight end, lives in a different place: he has room to break either way, he is usually covered by a slower linebacker or a nickel back rather than a cornerback, and he is closest to the deep middle. So inside receivers, and backs who start in the backfield, have routes of their own, and so do receivers trying to fool a defender rather than simply get past him. Four come up constantly.

Option routes: the receiver gets a vote

The tree assumes the receiver knows his route before the snap. On an option route (also called a choice route) he doesn’t: he picks the finish after the snap, from what the defense does. The rule is usually something like this: “Push to six yards. If one man is following you, break away from him, out if he’s inside, in if he’s outside. If the defenders are guarding areas, find the hole between them and sit down, facing the quarterback.”

Why give the receiver a choice? Because the right route depends on the coverage, which isn’t settled until after the snap. Against man coverage, a defender on the receiver’s inside shoulder is taking away the middle, so the open break is outside, and vice versa. That position is the defender’s leverage, and the receiver, with the two-way go of Spread Formations, breaks away from it. Against zone coverage, the underneath defenders watch the quarterback and nobody follows the receiver, so running away from someone is pointless; he stops in the space between two of them. One call, right against both. Two refinements are common. Against a man defender lined up straight in front of him, the receiver stems right at him to make him commit to one side, then breaks to the other. And many systems tie the choice to the safeties: with one deep safety in the middle (MOFC, “middle of the field closed”) he breaks away from his defender, and with the middle empty (MOFO, “open”) he sits; Coverage Fundamentals explains why.

The cost is that two people must make the same decision in a split second. If the receiver sees man and breaks out while the quarterback sees zone and throws to where he should have sat down, the ball goes to nobody, or to a defender, so option routes go to the receivers a quarterback trusts most. The Patriots of the Tom Brady era were famous for building their passing game around option routes for slot receivers such as Wes Welker and Julian Edelman.3 Many other routes carry smaller rules of the same kind (“against two deep safeties, bend the seam into the middle”), which Dropback Concepts collects as route conversions.

The drop: the quarterback’s feet are a clock

Now the “when.” On a dropback (any called pass play on which the quarterback sets up behind his line to throw, including play-action and the ones that end in a sack or a scramble), his feet follow a prescribed pattern called a drop, counted in steps, and each drop is paired with a depth of route.

From under center, where he takes the snap with his hands under the center and starts about a yard and a half behind the line, the classic menu is:

  • Three-step drop: about 5 yards deep, finished a little under a second after the snap. It goes with the quick routes that break at 5–6 yards: hitches, slants, quick outs. The ball should leave as his third step lands, with no extra step, more or less as the receiver breaks.
  • Five-step drop: about 7 yards deep, finished in roughly a second and a half. It goes with the intermediate routes at 10–14 yards: outs, curls, digs. The quarterback usually adds a short step forward at the end of the drop, a hitch step (not to be confused with the hitch route), to gather himself, and throws.
  • Seven-step drop: about 9 yards deep, finished in about two seconds. It goes with the deep routes at 15–20 yards and beyond: comebacks, posts, corners, deep digs. It is the slowest drop and the one that most needs the line to hold, which is a big reason it is rarer now than in the 1970s and 80s.

From the shotgun, the quarterback starts about five yards deep, so he needs fewer steps to reach the same depths. The pairs become one, three and five steps: a one-step drop (catch the snap, one step back, throw) for the quick game, three steps for intermediate routes and five for deep ones. Catching the snap is not a step; count from his first step back. The pairing of depth and time stays the same.4 On the standard broadcast angle, high behind the quarterback, watch his feet rather than the ball: count the steps back, look for the plant of the last one, and then for the small hop forward, the hitch step.

Why pair them? Because both are clocks, and the offense wants them to ring together. A receiver starting from a standstill, with a defender in his way, reaches 5 yards in about a second and 12 in about two; a quarterback backing up covers his drop on about the same schedule. Matched, the quarterback’s foot lands as the receiver cuts, and the ball leaves without a pause. Mismatched, one of them waits: a seven-step drop for a hitch hands the cornerback two free seconds to close, and a three-step drop for a post leaves the quarterback holding the ball while the rush arrives.

The black ticks are a reality check: in 2023–2025 the median throw 10–14 yards downfield left about 2.7 seconds after the snap, and the median deep throw about 3.0. Real throws run later because the medians include second and third reads, scrambles and backs released late. The textbook is the plan; the progression data below show what happens when the plan doesn’t work.

ImportantWhy it exists

Drops turn a judgment (“is he open yet?”) into a rhythm (“throw on the fifth step”). A quarterback who waits to see a receiver open is late, because a receiver who looks open has usually been open for half a second already, and the ball needs another second to get there. A quarterback whose feet tell him when the receiver will be at his spot can throw before the defender has reacted to the break. The cost is that a rhythm can be disrupted: a pass rusher in his face, a receiver jammed off his path at the line, or a defender sitting on the spot breaks the timing. Then the footwork carries him to his next option: the hitch step that ends the drop is also the moment he moves his eyes to the second receiver (more under Reads, below).

Three complications. The shotgun dominates: in 2025 about 82% of NFL dropbacks started from the shotgun or the pistol,5 where the quarterback sees the defense better and reaches his depth sooner. Play-action changes the footwork: a handoff fake comes first, so the routes are timed to a longer clock (Play-Action, Bootlegs, and Screens). Not every throw comes off a drop: quarterbacks roll out and escape a collapsing pocket, and the routes then follow the scramble rules of Quarterback Play.

At 1.85 seconds the quarterback’s hitch step lands and he throws, to a spot near the sideline 12 yards downfield. Z still hasn’t cut. Why doesn’t he break when the ball leaves the quarterback’s hand? Because he isn’t watching the quarterback. A defender guarding a deep zone has one job above all, not to be beaten deep, so his eyes are on the receiver who can beat him, and until Z breaks, an out and a go look the same. (The underneath zone defenders, with no deep job, are the ones who watch the quarterback.) A cornerback who guesses early and jumps the out gets beaten by the go: the route tree’s pair logic again. Against this coverage the out usually comes with a flat route from the next receiver inside, to hold the short-outside defender down; here Y stays in to block, so that defender is free to widen, one more reason the ball must be on time.

The throw is to a spot, and it leaves before the receiver gets there, because it has a long way to go: some 29 yards from the quarterback’s hand to the catch, more than a second in the air even at the firm 22 yards a second this simulation uses. That makes it hard to defend and hard to execute: a receiver who is jammed and arrives late, or rounds his cut and arrives a yard short, turns a completion into an incompletion, or an interception. Coaches have related words: a rhythm throw comes out on the last step of the drop, with no hitch step (The Quick Game), and an anticipation throw is any throw released before the receiver is open (Quarterback Play).

A short history: Walsh and the three-step game

The modern emphasis on timing owes a lot to Bill Walsh. In 1970 he was an assistant on Paul Brown’s Cincinnati Bengals, and his quarterback, Virgil Carter, was mobile and accurate but had no deep arm. Rather than ask Carter to throw deep, Walsh built a passing game of short, precisely timed throws, many off three-step drops, spread across the whole width of the field, so that a pass could do the work of a running play. Carter led the NFL in completion percentage in 1971.6 Walsh took the approach, much refined, to San Francisco as head coach in 1979 and won three Super Bowls with it; its descendants became the West Coast offense (Offensive Systems and Play-Calling Languages; Case Study: Bill Walsh’s 49ers and the West Coast Offense).

Reads: what the quarterback is looking at

The routes put receivers in places and the drop sets the moment; the quarterback still has to choose. He does it with a read, a decision made after the snap by watching something specific, in the form “if this, then that.” Pass plays use two kinds of read.

  • A progression read asks which receiver is open, looking at receivers one at a time, in order.
  • A defender read asks what is this defender doing, watching one defender and throwing to the receiver he can’t cover.

Most plays mix them. A typical call starts with a defender read on one side (“watch the defender guarding the short area outside; throw the receiver he leaves”), and if both of those receivers are covered, the quarterback moves on to the next receiver in a progression. One check comes before either. If the defense sends more rushers than the line can block, the first answer is the hot route or a receiver’s sight adjust (cutting his route short because he sees the blitz too), thrown at once, before any progression starts.

Progressions

A progression is the order in which the quarterback looks at his receivers. The play call gives him a first read, usually the route the play was designed around; if that receiver is covered, he moves his eyes to the second, then the third, and finally to the checkdown, the short, safe throw to a back or tight end (the outlet) that keeps the play from losing yards. Coaches tie each read to the footwork: the first is thrown on the last step of the drop, the second on the hitch step, the third after a second hitch, so the progression runs on the drop’s clock.

Progressions come in two sizes.

  • Half-field reads. The quarterback reads only one side of the field, usually two or three routes built to work together there. The other side’s routes keep defenders busy, or are an emergency answer if the defense blitzes. Because he looks at one side only, the read is quick, which is why it fits a short drop. He picks the side before the snap from what he sees: a receiver alone against a cornerback playing well off him, or the side with fewer defenders than receivers. Most of the quick passing game works this way.
  • Full-field reads. The quarterback starts on one side, often with a deep route, and works across the whole field to the checkdown. A full-field progression gives him more chances to find the one mistake the defense made, but moving his eyes and feet across the field takes a deeper drop and time the rush may not allow.

The trade is speed versus flexibility. A play-caller picks between them partly on the situation and partly on the quarterback: a rookie gets more half-field reads; a veteran with a good line gets the whole field.

Defender reads and the conflict defender

The other kind of read watches one defender instead of a sequence of receivers. The idea is to find a conflict defender, a defender who is responsible for two things at once, and put two receivers in his area. Whatever he does, he is wrong about one of them. You met this idea in What a Play Really Is, where the quarterback on Stick reads the slot defender, and the run game uses it on a defender responsible for both a run gap and a pass zone (Run-Pass Options). Defender reads are fast, because the quarterback looks at one man, not five, and they don’t depend on the receivers beating anyone, only on geometry: a defender can only be in one place. The offense creates the conflict up and down or side to side.

The vertical stretch: high and low

A high-low read (also called a vertical stretch) puts two receivers in one defender’s area at two depths, one above him and one below. The animation below shows the most common version, a curl over a flat route, against the defender who guards both in a Cover 3 zone, the curl-flat defender, who guards the short area outside, from the flat up to the depth of a curl. (Cover 3 has three deep defenders, each guarding a deep third of the field’s width, and four underneath; you’ll meet it properly in Cover 3. For now all that matters is the one defender.)

The ringed nickel’s zone runs from the sideline in toward the hash and from the line out to about 12 yards: two receivers’ worth of field. If he gets depth to stand under the curl, the flat route is alone in front of him, and the ball goes there for a few easy yards with room to run. If he comes forward to cover the flat route, the curl settles in the hole behind him, in front of the cornerback, who is busy guarding the deep third, and the ball goes there for 10 or 12. The quarterback doesn’t need the whole coverage, only one man.

Why does this work? A zone defender is taught to stay between the ball and the deepest threat in his area, so one who sinks under the curl is doing his job; he is just losing the flat. The defense has three good answers. It can change who covers what: in man coverage, each receiver has his own defender, and there’s no one to put in conflict. That is why offenses want a hint of man or zone before they choose a read, and the defense gives some: the safeties’ shell from A First Look at Defense, and cornerbacks who line up tight, eyes on a receiver, rather than deep and watching the quarterback (Coverage Fundamentals and What Beats What build this out). It can pattern-match: zone defenders pick up a receiver who enters their area and stay with him, so the curl-flat defender takes the flat and a linebacker gets under the curl (Cover 3). Or it can bait: show a soft flat, then drive on it once the ball is in the air.

The high-low idea is everywhere. In Cover 2, two safeties split the deep field in halves and the cornerbacks stay short, guarding the flats; a corner route over a hitch puts that cornerback in the same bind (that is the concept called smash), and three receivers at three depths on one sideline make it a three-level stretch (the flood concept). Both are built in full in Dropback Concepts.

The arithmetic is the heart of it. Eleven defenders, minus four rushers, minus the deep defenders, leaves the underneath ones: 11 − 4 − 3 = 4 in Cover 3, 11 − 4 − 2 = 5 in Cover 2. Put five receivers across the field at the same depth against four underneath defenders, and at least one receiver has nobody in front of him while every defender has a receiver on each side. Horizontal stretches work underneath, in front of the deep defenders, so they are short, reliable and quick: the backbone of the quick game’s spacing and stick concepts (The Quick Game). What they give up is depth; the big gains have to come after the catch. Combine one with a vertical stretch and you get three receivers around one defender, the triangle read of Dropback Concepts.

ImportantWhy it exists

Stretches beat zone coverage without winning a one-on-one race. Zone defenders guard areas, and areas have edges: put two receivers in one defender’s area, above and below him or on either side, and he has to choose. It is Gillman’s principle of making the defense cover the whole field, applied one defender at a time. Against man coverage there is no area to overload, so the answers are different: routes that win one-on-one, double moves, and crossing routes that make defenders run into each other (the man-beaters of Dropback Concepts). That is why “man or zone?” sits behind every read.

Measuring a throw: air yards, YAC and aDOT

A completion’s yards come from two places, the ball’s flight and the receiver’s run after the catch. Analytics splits them with three numbers you’ll meet on broadcasts and in almost every article about a quarterback.

  • Air yards measure how far past the line of scrimmage the ball traveled in the air, to the spot where it was caught, or where the target was when it fell incomplete. Only the distance downfield counts, not the sideways distance. A screen caught 3 yards behind the line has −3 air yards.
  • YAC, yards after catch, is what the receiver adds from the catch to the end of the play.
  • aDOT, average depth of target, is the average air yards per pass, for a quarterback (per throw) or a receiver (per target), completed or not. It describes where a quarterback throws or a receiver is used, not how well.

An example. A slant caught 6 yards past the line, and run 14 more: a 20-yard gain, 6 air yards and 14 YAC. An incomplete deep out aimed 14 yards downfield: no gain, but 14 air yards toward the quarterback’s aDOT. Across the NFL in 2025 the average throw traveled about 7.8 yards past the line, and about 47% of all receiving yards came after the catch, so the two halves are close to equal.8

The chart is the passing game’s central trade. Short throws are reliable but don’t gain much in the air, so they live on YAC. Throws behind the line of scrimmage, mostly screens and flips to backs, are completed four times in five, and a completed one gains about 6 yards (about three lost in the air, nine gained after the catch). Yet as a group they lose expected points, for two reasons. The one in five that falls incomplete wastes a down, and a six-yard gain is only modest progress: on first-and-10, a four-yard gain averages -0.16 EPA and it takes about five yards to break even (+0.02).9 That is stricter than the old yardage rule of thumb you met in Downs, Distance, and Field Position, which counts four yards (40% of the ten needed) as a success; the EPA version says second-and-6 is, on average, a little worse than first-and-10. Throws of 5–9 yards are the most dependable winners: completed about two times in three, they have the highest success rate of any depth (59%). Deep throws fail most of the time and pay anyway. Throws of 20 yards or more are completed less than half the time and are intercepted on about 6% of attempts, against 2% for everything shorter, yet each attempt is worth about as much as a throw of 10–19 yards, because a completion moves the ball so far. The bottom-right panel shows a surprisingly steady 3–6 yards after the catch on anything thrown past the line.

CautionCommon misconception

“High aDOT good, low aDOT bad.” aDOT describes a style, not a level of play. A quarterback in a quick, horizontal offense built on YAC can have a low aDOT and an excellent season; a quarterback with a high aDOT may be taking shots because his team is always behind. YAC has the same problem. It depends on the play design (screens and crossing routes produce YAC; comebacks and corners rarely do), on the receiver, and on the throw itself, since a ball placed in stride lets a receiver keep running. To judge a passer, the course uses measures built on top of these, such as CPOE (completion percentage over expected, which accounts for depth) in Quarterback Play, and EPA per play.

January 22, 1989, Miami. The Bengals led the 49ers 16–13 with a little over three minutes left, and San Francisco took over at its own 8-yard line. Joe Montana drove them 92 yards in 11 plays, mostly on short and intermediate throws: Walsh’s timing game, under pressure. What to notice: the winning throw went to the receiver the play was not designed for. That is what a progression is for: the first read was taken away, and the quarterback had a pre-set order of where to look next and the time to look there. It was Walsh’s last game as head coach, and the best-known demonstration of his offense: short, precise, on time, and with an answer when the first answer was covered.

What to notice: two answers to the same clock. Brady beat the rush by being done before it arrived. Mahomes beat it by keeping the play alive after the design ran out, throwing on the run, off one foot or sidearm to receivers taught where to go when the play broke down (the scramble drill of Quarterback Play). Watch a replay of either and count: on a Brady throw you can usually match the release to the end of the drop; on many Mahomes throws, the drop is long over.

TipWatch for it on Sunday: be the route caller

Pick one drive and call every pass play three times: before the snap, at the snap, and on the replay.

  • Before the snap: count and number. How many receivers on each side? Number them from the sideline in: #1, #2, #3 (Spread Formations). Note the splits and the hash: a receiver tight to the formation, or on the wide side, has room to break outside (corners, outs); one near the sideline mostly breaks inside (slants, digs, posts).
  • At the snap: watch the quarterback’s feet. From under center, count three, five or seven steps; from the shotgun, one, three or five, starting after he catches the snap. Three (or one) means the ball should be out at once to a 5-yard route; five (or three) means a 10–14-yard route; seven (or five), or extra hitch steps, means deep routes or a quarterback working past his first read.
  • Then his head. If it stays on one side and the ball goes to one of two receivers near the same defender, that was a defender read. If it snaps from receiver to receiver, often with his feet resetting each time, and the ball goes somewhere he wasn’t looking first, he moved through a progression.
  • On the replay: name every route. Use the wide angle that shows the receivers and the deep defenders. Name and number each route (“Z ran a 9, H a 5 at twelve, Y a flat, X a slant”), then find the stretch: which defender had two receivers in his area, which way did he go, and where did the ball go?
  • In the stats: lots of yards per catch could mean deep targets or big YAC; the box score can’t tell you which. Advanced stat pages (Pro Football Reference, Next Gen Stats) list aDOT and YAC separately, and many broadcasts show Next Gen Stats graphics (the NFL’s player-tracking numbers) with air yards and time to throw.

TipAnswer

Z: a go route (9). H: an out (5) at about 12 yards. Y: a flat route (1). X, on the backside: a slant (2). The three routes on the right are a three-level vertical stretch along one sideline. Z’s go carries the cornerback deep (if he stays, Z is open deep). That leaves the out and the flat as a high-low of the curl-flat defender, often the nickel: he is the conflict defender. Takeaways

  • A route has a stem and a break, and the break is where it wins. The route tree (1 flat, 2 slant, 3 comeback, 4 curl, 5 out, 6 dig, 7 corner, 8 post, 9 go; odd out, even in) is a shared vocabulary whose numbers vary by team; the seam, wheel, whip, sluggo and option route live off it.
  • The drop is a clock. Three, five and seven steps from under center (one, three and five from the shotgun) end at about 5, 7 and 9 yards and pair with routes at 5–6, 10–14 and 15–20 yards. Matched, the ball leaves as the receiver breaks; a timing route is thrown to a spot before he gets there.
  • Reads come in two kinds. A progression checks receivers in order, over half the field (fast) or the whole field (flexible, slower); a defender read watches one conflict defender and throws where he isn’t. First reads are the quickest and most valuable throws.
  • Stretch the defender, up and down or side to side. A high-low puts receivers above and below one defender; a horizontal stretch puts them on either side. Both beat zone by geometry; against man the offense needs different answers.
  • Short throws live on YAC, deep throws on air yards. Air yards, YAC and aDOT describe where an offense gains its yards, not how good it is. Throws behind the line lose value on average, 5–9-yard throws succeed most often, and deep throws fail more but pay.
