def O(key, pos, d, w, label=None):

def D(key, pos, d, w, label):
    return Player(key, pos, "def", d, w, label=label)


def line():
    """The five offensive linemen (only the center is lettered, to keep the box readable)."""
    return [O(k, k, -0.7, (i - 2) * SPLIT, "C" if k == "C" else "") for i, k in enumerate(OL_KEYS)]


def four_rush():
    """A generic four-man rush: two ends outside the tackles, two tackles inside."""
    return [D("LE", "DE", 1.0, -5.4, "E"), D("NT", "DT", 1.0, -0.7, "T"), D("DT", "DT", 1.0, 2.35, "T"),

def timed(p, key, way, kind="path"):
    """Script a player exactly: `way` = [(t, d, w), ...] absolute spots reached at t seconds after the
    snap. Repeating a spot means 'stand there until t'. (gridiron's route helpers run at a fixed speed;
    the quarterback's clock is the subject of this chapter, so here every move is timed by hand.)"""
    cur = np.array(p._cur(key), dtype=float)
    tc, wait = 0.0, 0.0
    if any(a.key == key for a in p.actions):          # continue from where his earlier moves end
        tc = max(0.0, float(p._compile()[0][key][-1, 0]))
    for t, d, w in way:
        nxt = np.array([d, w], dtype=float)
        dist = float(np.hypot(*(nxt - cur)))
        if dist < 1e-3:
            wait += t - tc
        else:
            p._add(kind, key, [tuple(cur), tuple(nxt)], speed=dist / (t - tc), delay=wait)
            wait = 0.0
            cur = nxt
        tc = t
    return p


def frame_field(ax, p, lateral, window):
    ylo, yhi = sorted((p.ball_y - lateral[0], p.ball_y - lateral[1]))
    return Field(ax=ax, orient="vertical", los=p.los, window=window, numbers=False, hashes=False,
                 lateral=(ylo, yhi), to_go=p.to_go)


def XY(fld, p, d, w):
    x, y = fld.xy(*p.to_abs(d, w))
    return float(x), float(y)


def dot(ax, x, y, side, label, ppy, r=0.9, ring=None, alpha=1.0, z=8):
    """One player: offense a filled blue circle, defense an outlined orange square."""
    if side == "off":
        ax.add_patch(Circle((x, y), r, fc=style.OFFENSE, ec="white", lw=0.6, zorder=z, alpha=alpha))
        tc = "white"
    else:
        ax.add_patch(RegularPolygon((x, y), 4, radius=r * 1.25, orientation=np.pi / 4, fc=style.DEFENSE_TINT,
                                    ec=style.DEFENSE, lw=1.0, zorder=z, alpha=alpha))
        tc = style.INK
    if ring:
        ax.add_patch(Circle((x, y), r * 1.65, fc="none", ec=ring, lw=1.7, zorder=z - 0.5))
    if label:
        fs = max(4.6, min(8.5, ppy * (1.15 if len(label) <= 1 else 0.85)))
        ax.text(x, y, label, ha="center", va="center", fontsize=fs, color=tc, fontweight="bold", zorder=z + 2)


def place_labels(ax, xs, ys, labels, fs=6.8, offs=None):
    """Greedy label placement for scatter charts: try offsets around each point until the label hits no
    point, no earlier label and stays inside the axes."""
    fig = ax.figure
    fig.canvas.draw()
    ren = fig.canvas.get_renderer()
    pts = ax.transData.transform(np.column_stack([xs, ys]))
    box = ax.get_window_extent(ren)
    offs = offs or [(5, 3), (5, -9), (-5, 3), (-5, -9), (6, -3), (-6, -3), (0, 6), (0, -12), (9, 8), (-9, 8),
                    (9, -13), (-9, -13), (12, 0), (-12, 0)]
    placed = [t.get_window_extent(ren) for t in ax.texts]          # never cover text already on the chart
    if ax.get_legend() is not None:
        placed.append(ax.get_legend().get_window_extent(ren))
    for x, y, lab in zip(xs, ys, labels):
        for dx, dy in offs:
            t = ax.annotate(lab, (x, y), xytext=(dx, dy), textcoords="offset points", fontsize=fs, color=style.INK,
                            ha="left" if dx > 0 else ("right" if dx < 0 else "center"), zorder=4)
            bb = t.get_window_extent(ren).expanded(1.04, 1.1)
            ok = (not any(bb.contains(px, py) for px, py in pts) and not any(bb.overlaps(o) for o in placed)
                  and box.contains(bb.x0, bb.y0) and box.contains(bb.x1, bb.y1))
            if ok:
                placed.append(bb)
                break
            t.remove()
    return placed

def badge(ax, x, y, n, state, fs=8.0):

def clean(ax):
    """Static diagrams drawn with Play.draw: drop hash ticks and yard numbers, clip text to the window."""
    for ln in list(ax.lines):
        xs, ys = ln.get_data()
        if len(xs) == 2 and ln.get_linewidth() == 0.5 and np.hypot(xs[1] - xs[0], ys[1] - ys[0]) < 1.0:
            ln.remove()
    for t in list(ax.texts):
        if t.get_text().isdigit() and t.get_alpha() == 0.55:
            t.remove()
        else:
            t.set_clip_on(True)


def moment(ax, p, t, lateral, window, rings=None, plan=None, ghosts=(), trail_keys=None, r=0.9):
    """Everyone where he is at time t, faint trails of where he has been, the ball, and (optionally)
    the called routes as faint grey dashes, so a single panel shows both the plan and the present."""
    fld = frame_field(ax, p, lateral, window)
    ppy = fld.pts_per_yard()
    lo, hi = sorted(lateral)
    inside = lambda d, w: window[0] + 0.6 <= d <= window[1] - 0.6 and lo + 0.6 <= w <= hi - 0.6  # noqa: E731
    for pts in (plan or {}).values():
        xy = [XY(fld, p, *q) for q in pts]
        xs, ys = zip(*xy)
        ax.plot(xs, ys, color=style.MUTED, lw=0.9, ls=(0, (2, 2)), alpha=0.75, zorder=2)
        ax.add_patch(FancyArrowPatch(xy[-2], xy[-1], arrowstyle="-|>", mutation_scale=6, color=style.MUTED, lw=0,
                                     alpha=0.75, zorder=2))
    if t > 0:
        ts = np.linspace(0, t, 32)
        for k, pl in p.players.items():
            if (pl.pos in NONROUTE) if trail_keys is None else (k not in trail_keys):
                continue
            pts = np.array([p.position(k, tt) for tt in ts])
            if np.ptp(pts[:, 0]) + np.ptp(pts[:, 1]) < 0.3:
                continue
            xs, ys = fld.xy(*p.to_abs(pts[:, 0], pts[:, 1]))
            if pl.side == "off":
                ax.plot(xs, ys, color=style.OFFENSE, lw=1.1, alpha=0.5, zorder=3)
            else:
                ax.plot(xs, ys, color=style.DEFENSE, lw=1.1, ls=(0, (3, 2)), alpha=0.8, zorder=3)
    for d, w, text in ghosts:
        x, y = XY(fld, p, d, w)
        ax.add_patch(Circle((x, y), 0.72, fc="none", ec=style.OFFENSE, lw=1.0, ls=(0, (2, 1.5)), zorder=6))
        if text:
            ax.text(x, y - 1.45, text, ha="center", va="top", fontsize=max(5.0, ppy * 0.7), color=style.INK_2,
                    zorder=6)
    for k, pl in p.players.items():
        d, w = p.position(k, t)
        if inside(d, w):
            dot(ax, *XY(fld, p, d, w), pl.side, pl.label, ppy, r=r, ring=(rings or {}).get(k))
    thrown = [(to, te) for kind, to, te in p.ball_events if kind == "pass" and t > te]
    if thrown:
        bd, bw = p.ball_position(t)
        sd, sw = p.ball_position(thrown[0][1])
        rd, rw = p.position(thrown[0][0], t)
        caught = np.hypot(bd - rd, bw - rw) < 0.4
        ax.add_patch(FancyArrowPatch(XY(fld, p, sd, sw), XY(fld, p, bd, bw), arrowstyle="-|>", mutation_scale=8,
                                     color=style.BALL, lw=1.3, ls=(0, (2, 1.5)), zorder=9, shrinkA=0,
                                     shrinkB=9 if caught else 2))
        football(ax, *XY(fld, p, bd + (0.3 if caught else 0.0), bw - (1.25 if caught else 0.0)))
    elif t > 0.15 and "QB" in p.players:
        qd, qw = p.position("QB", t)
        football(ax, *XY(fld, p, qd + 0.3, qw - 1.2))
    return fld


def eyes(fld, p, t, target, color=GREEN):
    """The quarterback's eyes: a dashed line from him to the player he is reading."""
    (qd, qw), (rd, rw) = p.position("QB", t), p.position(target, t)
    x0, y0 = XY(fld, p, qd, qw)
    x1, y1 = XY(fld, p, rd, rw)
    fld.ax.add_patch(FancyArrowPatch((x0, y0), (x1, y1), arrowstyle="-", color=color, lw=1.4, ls=(0, (4, 2)),
                                     shrinkA=7, shrinkB=9, zorder=7))


def strip(draw, times, notes, ncols=2, width_in=6.5, aspect=1.0, note_lines=2):
    """Print stand-in for an animation: numbered panels with a caption under each."""
    nrows = int(np.ceil(len(times) / ncols))
    pw = width_in / ncols
    fig, axes = plt.subplots(nrows, ncols, figsize=(width_in, nrows * (pw * aspect + 0.2 + 0.16 * note_lines)),
                             squeeze=False)
    for ax in axes.flat[len(times):]:
        ax.axis("off")
    for i, (ax, t) in enumerate(zip(axes.flat, times)):
        draw(ax, t)
        ax.set_title(f"{i + 1}.  " + ("snap" if abs(t) < 1e-6 else f"{t:+.1f} s"), fontsize=8.2,
                     color=style.INK_2, loc="left", pad=3)
        ax.text(0.5, -0.01, notes[i], transform=ax.transAxes, ha="center", va="top", fontsize=7.1,
                color=style.INK, linespacing=1.2)
    fig.tight_layout(h_pad=1.2, w_pad=0.8)
    return fig


def video(draw, times, notes, t_end, width_in=5.6, aspect=1.0):
    """Web version of the strip: every tenth of a second, with the strip's captions as running titles."""
    import shutil
    from IPython.display import HTML, display
    fig, ax = plt.subplots(figsize=(width_in, width_in * aspect + 0.6))

    def update(t):
        ax.clear()
        draw(ax, t)
        cur = [n.replace("\n", " ") for tt, n in zip(times, notes) if tt <= t + 1e-6]
        head = "snap" if abs(t) < 0.05 else f"{t:+.1f} s"
        ax.set_title(textwrap.fill(head + ("   " + cur[-1] if cur else ""), 92), fontsize=7.6, color=style.INK,
                     loc="left")
        return []

    frames = list(np.round(np.arange(0.0, t_end + 0.05, 0.1), 2))
    anim = animation.FuncAnimation(fig, update, frames=frames, interval=100)
    html = (anim.to_html5_video(embed_limit=40).replace("<video ", '<video style="max-width:100%;height:auto" ')
            if shutil.which("ffmpeg") else anim.to_jshtml(default_mode="once"))
    plt.close(fig)
    display(HTML(html))


def play_out(draw, times, notes, t_end, ncols=2, aspect=1.0, note_lines=2, width_in=6.5):
    """Video on the website; a numbered, captioned frame strip in the PDF."""
    if PDF:
        strip(draw, times, notes, ncols=ncols, aspect=aspect, note_lines=note_lines, width_in=width_in)
        plt.show()
    else:
        video(draw, times, notes, t_end, aspect=aspect)


BOX_W = 3.9   # the tackle box: Rule 3-25 puts its edges at the OUTSIDE shoulders of the tackles, not their centers


def tackle_box(ax, fld, p, back):
    """The shaded tackle box (pocket area), from the line back to depth `back`."""
    xy = [XY(fld, p, *q) for q in [(-0.7, -BOX_W), (-0.7, BOX_W), (back, BOX_W), (back, -BOX_W)]]
    ax.add_patch(Polygon(xy, closed=True, fc=PURPLE, alpha=0.09, ec=PURPLE, lw=0.9, ls=(0, (3, 2)), zorder=1.4))


def pass_set(p, depths=(-2.6, -1.7, -1.4, -1.7, -2.6), t=0.8):
    """The five linemen set (tackles deeper and wider than the interior) and hold."""
    for k, dd in zip(OL_KEYS, depths):
        w0 = p.players[k].w
        timed(p, k, [(t, dd, w0 * 1.15)], "block")
    return p

R1_3_LONG = long3.loc[long3.ending == "Throw to the first read", "success"].mean()

The quarterback’s clock

Every dropback is a race against a clock the defense controls, and the quarterback runs a fixed sequence of jobs against it. Coaches teach the sequence by tying each job to a moment in the drop, so that the quarterback’s feet tell him what his eyes should be doing.

Much of the work is done before the clock starts. Before the snap he has identified the Mike and set the protection (Pass Protection), read the defense’s alignment for which half of the field he expects to work (The Pre-Snap Phase), and checked whether the look calls for a change of play or a shot down the field (a subject of Concepts Versus Coverages). After the snap, here is the order, for an ordinary dropback pass from the shotgun.

  1. At the snap: the protection check. Before he looks at a single receiver, the quarterback answers one question: did more rushers come than the protection can block? If yes, the ball goes now to the hot receiver or a receiver who sight-adjusted his route, before any progression starts. The protection call has already told him which rusher is his: the one the line leaves unblocked by design if he comes, whom the quarterback must beat with a quick throw. So this check takes a glance, not a study.
  2. The drop: read 1. If the rush is covered, his eyes go to the first read as his feet go back, and the decision comes as the last step of the drop lands: open, and the ball comes out as his back foot plants; covered, and he moves on.
  3. Each hitch step: the next read. If read 1 is covered, he takes a short step forward, the hitch step of Passing Game Fundamentals (a step, not to be confused with the hitch route), which resets his feet so he can throw again, and his eyes move to read 2. Another hitch step, read 3. Each hitch step takes about half a second.
  4. The last read: the checkdown. The final look is to a back or tight end a few yards past the line, the safe throw that keeps the play from becoming a loss.
  5. After that: off schedule. If the ball still isn’t out, the play has left its design. The quarterback now has to manage the pocket (step up, slide, or escape), and decide among a scramble, an improvised throw, a throwaway, or taking the sack.

The bottom half of Figure 26.1 is the part to remember. The first-read throw leaves the quarterback’s hand at a median of 2.3 seconds; a throw to a later read, about 0.6 seconds after that; and the throwaways and scramble-drill throws at the far right of the chart, after three seconds, when the rush has usually arrived. The checkdown is the odd one: it is the last look on paper, yet its median is no later than a second read’s. Two things pull it early. On many quick and half-field plays the back is read 2 or 3, not the last of four or five, so his turn comes fast; and some checkdowns are thrown before the deeper reads had a chance, the “bad checkdown” the checkdown section warns about. About 27% of checkdowns left the quarterback’s hand inside two and a half seconds. Every step down the list costs time, and, as you’ll see, value.

ImportantWhy it exists

Tying the eyes to the feet solves a problem that sounds trivial and isn’t: in the chaos of a rush, how does a quarterback know when to give up on a receiver? Counting seconds doesn’t work with three hundred pounds of defensive tackle in his face. Feet do. “Last step, read one; hitch step, read two; hitch step, checkdown” turns a judgment into a rhythm he can practice ten thousand times, and it keeps him on the same clock as his receivers, whose routes are timed to the same drop (Passing Game Fundamentals).

What it gives up: freedom. A quarterback who throws on rhythm throws to where the play says, not to where a receiver happens to be open. When the defense takes away the design, a rhythm passer can be stuck; that is where the second half of this chapter, the off-script game, comes in.

def upto(p, key, way, t_stop, kind="drop"):
    """Script a defender along `way` [(t, d, w), ...] but only until t_stop, the moment he reacts to the ball."""
    cut, prev = [], (0.0, *p._cur(key))
    for t, dd, ww in way:
        if t <= t_stop:
            cut.append((t, dd, ww))
            prev = (t, dd, ww)
        else:
            f = (t_stop - prev[0]) / (t - prev[0])
            cut.append((t_stop, prev[1] + f * (dd - prev[1]), prev[2] + f * (ww - prev[2])))
            break
    timed(p, key, cut, kind)

Three things are happening in Figure 26.2 that you can’t see in a static play diagram.

Every read has a key defender. Passing Game Fundamentals separated two kinds of read: a progression, which looks at receivers one at a time, and a defender read, which watches one defender. Dagger is a progression, but even on a progression each look is answered by one defender, and good quarterbacks look at him. At read 1 the question is not really “is Y open?” but “what is the safety on Y’s side doing?” In Quarters that safety is taught to carry the inside receiver if he goes deep, so a seam route is open only if he doesn’t, and here he does. At read 2 the question is the Mike linebacker: has he dropped deep enough to take away the dig, or is the window behind him open? Here he sits at ten yards with his eyes on the quarterback, and the safety is gone with Y, so the space at sixteen yards between them is empty. Which defender is the key on which play depends on the coverage, and that is a subject of its own: Concepts Versus Coverages teaches it, once Part 6 has taught the coverages themselves.

The order is a ranking of risk and reward, and a timetable. Read 1 is the biggest gain the play can produce (a seam is a touchdown if the safety guesses wrong); read 2 a solid one; read 3 a small, safe one. A progression runs from the most valuable throw the defense might allow to the least dangerous one, so that the quarterback takes the big play when it is there and the safe one when nothing else is. It is also a schedule: each route is built to come open just as his feet arrive at that read. Z’s dig breaks at about 2.4 seconds because that is when the hitch step lands. Many coaches sum it up as deep to short, but the timetable is the real rule, and on quick concepts the order starts short: stick and slant-flat key one underneath defender and throw whichever of two short routes he leaves (The Quick Game). Dagger is a full-field read in the language of Passing Game Fundamentals; on a half-field read the order is shorter, and what the quarterback saw before the snap can decide which half he starts on.

The ball comes out before the receiver is open. In panel 3 the dig hasn’t broken yet; Z is still running straight up the field. The quarterback throws anyway, because he has seen what he needed: the Mike is in front of the window, not in it, and the safety is committed to the seam. That kind of throw has a name.

NoteMadden vs. real life

In Madden every receiver carries a button icon, the camera sits high behind the quarterback, and you can see the whole field at once and press the button of whoever looks open. A real quarterback sees through a forest of six-and-a-half-foot linemen with helmets bouncing in his face, from eye level, and can usefully look at one area at a time. That is why a progression exists at all: he can’t scan everything, so the play tells him what to look at, in what order, and when to give up on each look. When an analyst says a quarterback “got through his progression,” he means the quarterback looked at the right things, in the right order, on time, which is much harder than pressing the right button.

Anticipation: throwing to a window that isn’t there yet

An anticipation throw is a pass released before the receiver is open, usually before he makes his break, aimed at the spot where the window between defenders will be when the ball gets there. It is the bigger, riskier cousin of the timing route of Passing Game Fundamentals: a timing route is thrown to a fixed spot on a fixed step, while an anticipation throw is a judgment, made on any route, about where defenders will be in a second.

Why throw to a receiver who isn’t open? Because the ball takes time to get there. A firm 20-yard throw is in the air for about a second, and in a second a linebacker can close five yards. A window that is open when the quarterback sees it is often closed by the time the ball arrives. What makes it a judgment rather than a timing route is what he reads in the half second before he lets go. On dagger the timing is in the play; the decision is the Mike. If the Mike is sinking as Z stems upfield, eyes turned to the receivers, he is going to be standing in the window, and the throw dies. If he sits with his eyes on the quarterback, he will be in front of it. The quarterback has to make that call while Z is still running straight upfield.

In the simulation the early throw is caught with the nearest defender 2.4 yards away; the late one with a defender 0.9 yards away, close enough to knock it down or catch it himself. The difference is half a second, one hitch step.

A simulation only shows what it was told, and this one was told that defenders break on the ball when it is thrown, not when they see the receiver break. So treat it as an illustration of the mechanism, not proof. The proof that windows matter is in tracking data. Next Gen Stats measures, for every target, how far the receiver was from the nearest defender when the ball arrived. Across 885 receiver-seasons with 50 or more targets from 2018 to 2025, each extra yard of average separation went with about 3.7 more percentage points of catch rate, at the same average depth of target.1 An anticipation throw is a way of buying that yard with the clock instead of the receiver’s speed.

The trade is just as plain. An anticipation throw is a bet on the future, made on information about the present: the quarterback is predicting that the receiver will break where and when he is supposed to, and that the defender he is reading will keep doing what he is doing. If the receiver slips, rounds his cut, or is held up, the ball arrives at an empty spot, often into a defender’s hands. And if the defender was disguising his intentions, sitting shallow to bait the throw, the window was never going to open. Anticipation is therefore the skill that most separates veteran quarterbacks from rookies: it takes years of reps with the same receivers to know exactly where they’ll be, and years of film to know what a linebacker’s first two steps mean.

Coaches often say the young quarterback “needs to see it before he throws it.” What they mean is that he is throwing a beat late, waiting for the receiver to look open, which, as Figure 26.3 shows, is too late against good defenses.

The pocket: climb, slide, escape

The progression assumes a quarterback standing calmly at his launch point. He rarely is. A pocket is a moving thing (the tackles giving ground, the interior bending), and the quarterback has to move inside it without losing his eyes or his feet, so that he can still throw on time from wherever he ends up. Coaches name three moves.

Climb. Climbing the pocket means stepping forward into the pocket, between the interior linemen, as the edge rushers are pushed past. It is the single most important pocket skill, and it follows directly from the geometry of Pass Protection: the tackles give ground on purpose and ride the ends on a wide arc behind the launch point, while the guards and center hold firm in front. A quarterback who stays at his launch point, or worse, drifts backward, walks into the spot where the ends’ arcs finish. One who steps up two yards lets the ends run past behind him, and throws from the cleanest spot on the field. Quarterback coaches drill the idea endlessly: pressure from the outside is answered by moving up, not back.

Slide. When pressure comes from one side (an end who beat the tackle inside, a blitzer through a gap) and the middle is still firm, the quarterback slides a yard or two the other way, keeping his shoulders square to the line of scrimmage and his eyes downfield. A slide buys a half second without ending the play: he is still in the pocket and can still throw to anyone in the progression.

Escape. When the pocket is gone, usually because the interior collapsed and the quarterback has nowhere to step up into, he leaves. The usual escape lane is outside one of the tackles, on the side where an end rushed too far upfield and was pushed past; it opens exactly because the tackle did his job. Escaping changes the play into something else, which is the next section’s subject.

All three moves are made with both hands on the ball. Quarterback coaches drill it as hard as the footwork: a quarterback who carries the ball loose in one hand while he climbs or slides is the one who gets it knocked out from behind by the end he never saw, the strip-sack.

The skill under all three is doing them without looking at the rush. A quarterback who drops his eyes to watch the defensive end has stopped reading the coverage, and a quarterback who has stopped reading the coverage has stopped running the play. Great pocket passers feel the rush (from the color at the edge of their vision, from the sound, from the pressure on a lineman’s back) and move their feet while their eyes stay downfield. Coaches call this pocket presence, and it is one of the few things that clearly separates good quarterbacks from bad ones, as Figure 26.5 shows in motion.

CautionCommon misconception

“A mobile quarterback should get out of the pocket whenever it gets crowded.” Quarterback coaches teach the opposite: don’t leave a pocket that still works. Outside it he has fewer throws (he can see only half the field and has to throw on the run), the rushers who were blocked are now chasing him unblocked, and the sideline is an extra defender. The best scramblers in the league are good at not leaving until the pocket is gone. Then they are devastating, as the next section shows.

Be careful with the obvious statistic here. In FTN’s charting from 2022 to 2025, quarterbacks were sacked on 8.3% of dropbacks once they were out of the pocket, against 6.2% in it,2 which looks like proof that leaving gets you sacked. It isn’t, because quarterbacks mostly leave because the pocket has already failed: from 2023 to 2025, 57% of out-of-pocket dropbacks were charted as pressured, against 24% of the rest. Compare like with like, pressured dropbacks only, and the ones that got outside ended in a sack less often (15% against 26%). Once the pocket has collapsed, escaping is a rescue. What the charting can’t tell you is what leaving a pocket that was still good costs, because nobody charts whether it was still good. That part of the rule rests on coaching judgment, not on this number.

Off-script football and the scramble drill

When the quarterback escapes the pocket, or holds the ball past the play’s last read, the play has gone off script. An off-script play (also called an off-schedule play) is one that has left its design: the called routes have run out, the protection has broken, and what happens next depends on improvisation.

Improvisation sounds like the opposite of coaching. It isn’t. The quarterback improvises, but his receivers follow rules, practiced every week, that tell them what to do when he starts running. Those rules are the scramble drill. Every team has its own version and its own vocabulary, but the core is nearly universal.3

  • Short goes deep, deep comes back. Receivers spread out vertically so the quarterback has a choice of depths, and so that two receivers never end up in the same spot dragging their defenders with them. The deep receiver comes back because a quarterback throwing on the run cannot make a 40-yard throw accurately; the short receiver goes deep because the defenders near him have all turned to chase the quarterback, and the space behind them is empty. “Short” and “deep” are measured against the linebackers: in front of them is short, behind them is deep.
  • Go where he goes. A quarterback running to his right can throw easily to his right and downfield, and only with difficulty back across his body to the left. (Throwing across your body on the run is one of the most dangerous throws in football, because the ball floats back toward the middle of the field where the defenders are.) So receivers work toward his side.
  • Backside receivers come across, deep. A receiver on the far side can’t stand still and wait; he runs back across the field behind the coverage, where defenders who have turned to chase the quarterback have left the deep middle.
  • The back leaks out; the linemen stay home. A back who stayed in to block releases toward the scramble side, a yard or two past the line, as the last, safest throw. The linemen keep blocking whoever they have and do not drift downfield: if the quarterback then throws, a lineman more than a yard beyond the line is ineligible downfield, a foul that wipes out the completion (RPOs).4
  • Get into his line of sight and keep moving. A receiver the quarterback can’t see is not an option. Receivers find grass on a different level from their teammates and stay in his vision, and if he crosses the line of scrimmage to run, they stop running routes and start blocking. Nobody calls out “scramble”: the cue is the clock. A receiver who reaches the end of his route without the ball turns to find the quarterback, and if he is no longer where the drop put him, the drill is on.

When two rules seem to apply, settle it by side first. A receiver on the far side from the scramble comes across: deep if he was deep, and a crosser keeps crossing at his depth, toward the quarterback. On the scramble side, short goes deep and deep comes back. Teams word this differently, but that order keeps the receivers on different levels, which is the whole point.

Where does the quarterback look? The rules hand him receivers at four or five depths; he can’t look at all of them. The throws he can make while moving come first: the receiver working back toward him along his sideline, and the crosser running with him. Both are short, firm throws toward the side he is running to. The deep receivers are the big play, and he takes that shot when he has room to stop, set his feet and throw; a quarterback running flat out usually can’t throw 30 yards accurately. Escaping to his throwing-hand side makes all of this easier. A right-handed quarterback rolling left has to turn his hips and square his shoulders back to the target before he can throw at all, so for him the come-back routes on that side matter even more.

Throw or run? Once he is out of the pocket he has a third choice, and it is the best-valued ending in the data, as Figure 26.10 will show. The rule of thumb is simple. As long as he is behind the line, his eyes stay downfield and every receiver is still an option. When the receivers are covered and there is open grass in front of him, ideally to the line to gain, he runs. Once he crosses the line of scrimmage the passing play is over, because a forward pass thrown from beyond the line is illegal.5 Running too early gives up the throws; running too late, after the defenders have rallied, gives up the grass.

Why does any of this work? Because the defense was built for the play the offense called, and that play is over. Against man coverage, where each defender follows one receiver, defenders are running with their backs to the quarterback; when a receiver stops and reverses, his defender has to stop, turn and find him, and he is a step behind. Against zone coverage, where each defender guards an area, defenders have dropped to spots that were chosen for the original routes, and they have to leave those spots to chase receivers who are now somewhere else, opening the areas they were guarding.

The defense has rules for this too, and the scramble drill works best against a defense that breaks them. Up front the rushers keep their rush lanes and the ends keep contain, not rushing past the quarterback’s depth, which is exactly the mistake that opens the escape lane in Figure 26.4. Some defenses assign a spy, a defender who mirrors the quarterback and tackles him if he runs. And in zone, once the quarterback leaves the pocket, coverage players are taught to stop guarding grass and lock onto the nearest receiver (coaches call it “plastering”). A defense that does all of this leaves the quarterback little; much of the off-script value in the next chart comes against defenses that don’t.

ImportantWhy it exists

What the scramble drill solves. A pass play is designed to finish in under three seconds. When it doesn’t, eleven offensive players are suddenly without instructions, and without rules each receiver would do something different: some keep running their routes into the sideline, some stop, two drift into the same spot. The scramble drill turns the broken play into a second, pre-planned play, so the quarterback’s improvisation lands on receivers who are where he expects them.

What it gives up. Nothing in the design, but plenty in the result: a play that goes off script is, on average, a worse play. It is a rescue, not a strategy, and the offense would usually rather the first play had worked. The exceptions are a handful of quarterbacks for whom the rescue is a weapon, and you can find them in the data.

Off-script value: who gains by leaving the pocket?

FTN charts whether the quarterback was outside the pocket on each dropback, whether by design (a rollout) or by escape; the flag does not say which. From 2022 to 2025 that happened on 18% of NFL dropbacks. On average those plays were worth +0.011 EPA (expected points added) per play, against +0.049 in the pocket. That average hides a split. About 48% of out-of-pocket dropbacks ended in a quarterback run or a scramble-drill throw, and those were worth a lot; another 22% ended in a throwaway, a sack or an interception, and those cost even more. Off script is a high-variance place: for the league as a whole it nets out slightly worse than the pocket, but for a few quarterbacks it is clearly better.

A short history: from Tarkenton to Mahomes

The scramble entered the NFL’s vocabulary with Fran Tarkenton, a preacher’s son from Athens, Georgia, who quarterbacked the expansion Minnesota Vikings from their first game in 1961. Behind a thin, new offensive line, he ran around behind it, often for long stretches, until someone came open. Coaches of his era, his own first head coach Norm Van Brocklin among them, thought scrambling undisciplined (the feud with Van Brocklin ended with Tarkenton demanding a trade in 1967); Tarkenton’s answer, preserved on his Hall of Fame page, was that “I scramble because I’m good at it, because I can twist and dodge those big pass rushers better than most guys, and we get a lot of touchdowns that way.” He retired in 1978 holding the career records for completions, passing yards and touchdown passes.6

For decades after that the pocket passer was the ideal: the quarterbacks who won the most, from Joe Montana to Peyton Manning and Tom Brady, did their damage on schedule, and the offenses built around them, timed to the drop, treated a scramble as a play that had failed. But the off-script line never died. John Elway passed for 3,000 yards and ran for 200 in seven straight seasons and built a career on late comebacks; Steve Young retired with 43 rushing touchdowns, then the second-most by any quarterback.7 Teams with quarterbacks like them practiced what to do when the play broke.

What changed in the last decade is scale. College football filled with spread offenses and athletic quarterbacks who had been improvising since high school, and NFL offenses stopped treating the scramble drill as an emergency procedure and started repping it weekly and designing around it. The quarterbacks who came out of that world (Mahomes, Allen, Lamar Jackson, Jalen Hurts) made off-script play part of the plan rather than an emergency.

Getting rid of it: the throwaway

The last of the quarterback’s options when nothing is open is the least glamorous and among the most valuable. A throwaway is a pass he deliberately throws where nobody can catch it, usually out of bounds or well over a covered receiver’s head, to avoid a sack or an interception. He gives up the down but keeps the yards. NFL quarterbacks threw the ball away on 3.5% of dropbacks from 2022 to 2025.

Why is that a skill and not a surrender? Look at what the alternatives cost. In the same seasons, the average throwaway was worth -0.78 EPA (an incompletion, a lost down), the average sack -1.77 (a lost down and six or seven yards), and the average interception -4.42. A quarterback who turns two sacks a game into throwaways gains about 2 points a game for his team, without completing a pass.

There is a catch, and it is a rule. Intentional grounding is a foul for throwing the ball away to avoid a sack when the pass isn’t in the direction and vicinity of an eligible receiver; it costs a loss of down and ten yards (or loss of down at the spot of the pass, if that is worse), and a safety if the quarterback throws it from his own end zone. The rule changes once the quarterback leaves the tackle box, the area between where the two offensive tackles lined up: outside it, any pass that lands at or beyond the line of scrimmage (including out of bounds past it) is legal, receiver or no receiver.8 Officiating, Replay, and the Rules of the Ball covers the full rule; here is what it means for the quarterback.

The rule rewards exactly the skills this chapter has been describing. A quarterback who climbs and slides inside the pocket can always find a covered receiver to throw near; a quarterback who escapes the pocket earns the right to throw it into the stands. The quarterback who gets in trouble is the one who stays in the box with nobody near him, panics, and flings it at the turf, or who, outside the box, tries to throw it away on the run and doesn’t get it back to the line.

The most famous grounding call of recent decades shows how expensive the panic is. On the New England Patriots’ first offensive play of Super Bowl XLVI, against the New York Giants in February 2012, Tom Brady dropped back from his own 6-yard line with Justin Tuck bearing down on him, and threw deep down the middle to nobody. The flag came out; Brady had thrown it from his own end zone, so the grounding penalty was a safety, and the Giants, who had punted on their own opening drive, led 2-0. They won by four.9

The checkdown: risk management, not cowardice

The last read in most progressions is the checkdown: a short, safe throw, usually to a running back or tight end a few yards past the line, underneath all the coverage. It is the quarterback giving up on the big play to take a small, nearly certain one. Fans boo it. Coaches mostly don’t, and the data says why.

That is what risk management means at quarterback. Each read in the progression has an expected value and a risk of disaster, and the checkdown is the read with the least of both. A quarterback who takes it at the right moment gives up a little upside to cut off the tail of sacks and interceptions, and over a season that tail decides games.

On third-and-long, a checkdown usually means a punt. Even so it beats the sack: when the ball is going to be punted anyway, the yards the back gains after the catch still flip a little field position, while a sack can push a field goal out of range. On third-and-short the same throw is a good bet. Good quarterbacks know the down and distance on every snap and adjust: checkdowns early, when four yards on first down keeps the offense on schedule, patience on third-and-long, when only a throw past the sticks will do.

CautionCommon misconception

“Checking it down is what scared quarterbacks do.” Every great quarterback checks the ball down, often, and good offenses design their checkdowns to gain yards: a back who catches it in space with one linebacker to beat. The difference between a good checkdown and a bad one is when: a checkdown after the deeper reads were honestly covered is a good decision; a checkdown on the first beat of a play, before the deep reads had time to open, is a quarterback who has stopped trusting his protection. Watch where his eyes were before he threw it.

Measuring a quarterback: time to throw, aDOT, CPOE and EPA

You can now describe a quarterback’s process. The data can describe his style and his results, and four numbers do most of the work.

  • Time to throw (seconds from the snap to the release) says how long he holds the ball: quick and decisive, or patient and hunting.
  • aDOT (average depth of target) says how far downfield he aims: underneath and safe, or deep and aggressive.
  • CPOE says how accurate he is, after allowing for how hard his throws were.
  • EPA per dropback says what it all adds up to: the points his dropbacks added, sacks, scrambles and turnovers included.

The first two describe style, and neither is good or bad on its own; the last two describe results. All four are public: the Next Gen Stats passing leaderboard (https://nextgenstats.nfl.com/stats/passing) has time to throw, air yards and its own CPOE for every starter, and rbsdm.com (https://rbsdm.com/stats/), built on the same nflverse data as this chapter, has EPA per dropback and CPOE, so you can profile this Sunday’s quarterback before kickoff. The one that needs explaining is CPOE.

CPOE: completion percentage, corrected for difficulty

Completion percentage is the oldest quarterback number and one of the most misleading, because it rewards easy throws. A quarterback who checks the ball down all day completes 75% of his passes; one who throws deep outs and posts completes 60% and may be the better passer. CPOE, or completion percentage over expected, fixes this by comparing each throw with a model’s estimate of how often a throw like it is completed. The nflverse model behind the numbers here knows each throw’s depth, which part of the field it went to, the down and distance, and whether the quarterback was hit as he threw;10 the Next Gen Stats version also knows, from tracking data, how far the receiver was from the nearest defender.11 A deep out to the sideline might have an expected completion rate of 45%; a checkdown to the flat, 85%. Complete the deep out and you gain 55 percentage points over expected; drop the checkdown and you lose 85. Average those differences over a season and you have CPOE.

CPOE is the best single public measure of accuracy, and it is steadier from season to season than completion percentage, but read it with three cautions. It credits the quarterback for everything that is not in the model: a receiver who makes a great catch, or a scheme that gets receivers wide open, raises it. It says nothing about where the ball went, only whether it was caught: a quarterback who throws safely and accurately to covered checkdowns can post a fine CPOE and a poor offense. And check what each version does with throwaways. They count as incompletions in plain completion percentage, which punishes a smart habit, but the nflverse model leaves most of them out, because only 18% of charted throwaways get a completion probability.

CautionCommon misconception

“Completion percentage measures accuracy.” It measures accuracy and the difficulty of the throws a quarterback chooses, mixed together, and difficulty dominates. A conservative passer and an aggressive one can be equally accurate and be ten points apart in completion percentage. If you want accuracy, look at CPOE; if you want to know how aggressive he is, look at aDOT; if you want to know whether it worked, look at EPA.

Whose sack is it?

Sacks are charged to the quarterback’s line in the box score, but a good share of them belong to him. FTN’s charters, who watch every sack on video, marked 40% of the sacks from 2022 to 2025 as the quarterback’s fault: held the ball too long, drifted into the rush, missed the hot throw. A cleaner number separates the line’s share from his: pressure-to-sack rate, the share of the dropbacks on which he was pressured that ended in a sack. The line decides how often he is pressured; what happens next is mostly him. From 2023 to 2025 pressured quarterbacks were sacked 22% of the time, but among those with 200 or more pressured dropbacks the rate ran from 13% (Nix) to 30% (Levis).12 That spread is pocket presence, measured. The Pass Rush takes up why pressure is the steadier number and the sack the noisier one.

The four corners of Figure 26.12 are four ways to play the position.

  • The distributor (quick and short) wins with decisions: the ball comes out on rhythm to the first open receiver, and the offense lives on YAC. He rarely takes sacks, and his ceiling depends on how many yards his receivers make after the catch.
  • The rhythm shooter (quick and deep) is the hardest style to play, because deep routes take time to develop; it takes anticipation and a scheme that gets receivers vertical fast.
  • The aggressive creator (holds it, throws deep) extends plays to give deep routes time, with a pocket he manages and an off-script game he trusts. His sack rate is higher and his big-play rate too.
  • The patient or stuck passer (holds it, throws short) is the corner to worry about: the ball comes out late and short, which usually means the first reads were not open and the quarterback ended up checking it down after a long look. Sometimes that is a scheme or a line problem rather than a quarterback problem, but it is rarely a good sign.

In 2025 the league MVP, Matthew Stafford, was a rhythm shooter: top-left corner, an aDOT of 9.3 yards at 2.80 seconds, and +0.24 EPA per dropback. Drake Maye, the runner-up, threw just as deep but held the ball longer, top-right, with the league’s best CPOE and its best EPA per dropback (+0.33). Brock Purdy sits at the far right, just below the average depth line, holding the ball longer than almost anyone, and still posted +0.22: the patient passer, not the stuck one.

In 2025 the top half of the map had no red at all; every quarterback below zero threw short. That fits the argument of Dropback Concepts that value lives downfield. But the bottom half also holds Purdy, Josh Allen and Jared Goff, three of the year’s better seasons. Style tells you how a quarterback plays; the color tells you whether it worked.

Why the quarterback dominates value

One last number explains why everything in this chapter matters so much, and why the quarterback is paid like no one else on the field.

First, he touches the ball on almost every play, and on most of them he decides where it goes: in 2025, 60% of NFL offensive plays were dropbacks.

Second, passing plays swing further than running plays. Play by play, the EPA of a 2025 dropback had a standard deviation of 1.57 points, against 1.00 for a designed run: dropbacks hold the touchdowns and the interceptions. And the player who decides which way they swing differs more from team to team. Among 2025 starters with 300 or more dropbacks, average EPA per dropback had a standard deviation of 0.110; among ball carriers with 150 or more carries, average EPA per carry had one of 0.075. Multiply by a season’s workload (536 dropbacks for the median qualifying quarterback, 232 carries for the median back) and a quarterback one standard deviation above average is worth about 59 points over a season, against about 17 for a back.13

Points become wins at a steady rate: since 2012, each extra win has taken about 35 points of season scoring margin.14 So that quarterback is worth something like 1.7 wins a year over an average starter, and the back about 0.5. Over a replacement, the backup a team could sign off the street, the quarterback’s margin is larger still, because the gap between an average starter and a backup is wider at quarterback than anywhere else (the Purdy and Mac Jones numbers in the film room are one example).

All of these numbers share credit with the blockers, the receivers and the play-caller, and the split is unknowable from the box score. But the gap is too big to be an accident. The quarterback is the player with the most chances to add or lose points, on the plays where the most points are at stake. That is the starting point of Building a Roster, where the salary cap turns this gap into the most important economic fact in the sport.

He did it so economically that it looks like nothing happened, which is the point: a quarterback who climbs well never seems to be under pressure at all. He also got rid of the ball. Even at 44 and 45 his average time to throw was among the quickest in the league: 2.50 seconds in 2021 (rank 2 of 30 qualifying passers) and 2.45 in 2022 (rank 1 of 30).17

NoteFilm room: San Francisco 49ers 2023 — Brock Purdy and the system-quarterback debate

Brock Purdy was the last player chosen in the 2022 draft, number 262, the pick nicknamed “Mr. Irrelevant.”22 In 2023, his first full season as the starter, he led the 33 quarterbacks with 250 or more dropbacks in EPA per dropback (+0.32, rank 1) and in the nflverse model’s CPOE (rank 1). The argument about him since has been whether those numbers are his or Kyle Shanahan’s: a scheme famous for motion, play-action and receivers who run after the catch, with Christian McCaffrey, Deebo Samuel, Brandon Aiyuk and George Kittle to throw to. Is he a “system quarterback”?

The data cuts both ways, which is why the argument never ends. Against the “YAC merchant” version of the charge: in 2023, 48% of his receiving yards came after the catch, about the league rate (48%), and his aDOT (8.2) was above the league average (7.8). He was throwing downfield and completing it. For the charge: the two CPOE models disagreed. The nflverse model, which doesn’t know how open the receiver was, ranked him first; Next Gen Stats’ model, which does, ranked him 5th. That is still very good, but the gap between the two rankings is the point: part of what the first model credited to Purdy’s accuracy was receivers being open, which the scheme creates. And in 2025, when a turf-toe injury cost Purdy eight games,23 his backup Mac Jones ran the same offense at +0.15 EPA per dropback, against Purdy’s +0.22: worse, but far better than Jones’s -0.06 across 2022–2024 with New England and Jacksonville.

What to notice: the question “is it the quarterback or the system?” is the right one to ask, and CPOE models that know about separation help answer it. But a quarterback who throws on time to the right read is part of what makes a system work. Watch Purdy’s anticipation on intermediate throws and judge for yourself.

Watch for it on Sunday

TipWatch for it

Watch the eyes. On replays, especially the end-zone camera that looks straight at the quarterback, count how many reads he got through before he threw.

  • Find his first look. Where do his eyes (and his helmet’s facemask) point as his drop ends? That is usually read 1. Did the ball go there? More than half the time it does.
  • Count the head turns and hitch steps. Each snap of the head to a new area, usually with a hitch step, is a new read. One look and a throw: a rhythm throw. Two or three: a progression. A long stare at one side: he was waiting for something to open, he was confused, or he was holding a defender with his eyes on purpose, a look-off (Concepts Versus Coverages). His eyes can lie; the ball can’t.
  • Note the time. Count “one-Mississippi” from the snap. Ball out by about two and a half: quick game or first read. Out after three: off schedule, or a deep shot with max protection.
  • Watch his feet when the edge closes. Step up (climb)? Slide? Or drift back into the rush? The good ones move forward.
  • When he leaves the pocket, watch the receivers. Did the deep ones come back and the short ones go deep? Is everyone moving toward his side? That is the scramble drill. Who got open?
  • Judge the ending. A checkdown on third-and-12 is a punt with a few extra yards; on first-and-10 it is a fine play. A throwaway when nothing is there is a small win. A sack after a long hold in a clean pocket is on the quarterback; a sack that arrives before he could have thrown is on the protection.

For your prediction log: for one quarter, write down for each dropback how many reads the quarterback got through (1, 2, 3, or “off script”), and whether the ball came out on time. By the end you will know which kind of quarterback you are watching.

Where should the quarterback go: back, up, left, right, or out of the pocket?

TipAnswer

Up, and a little to his left: climb into the space behind the left guard and the center. Read the rush picture piece by piece. The ends have been pushed past him, so drifting back walks him into them, and escaping outside means running around a rusher who is already at his depth. The right side of the interior is caving: the tackle has the right guard two yards deep, so stepping straight up, or up and right, takes him into that collision. The left side of the interior is holding firm, and that is where the cup still has room.

So he climbs at an angle, a step or two up and half a step left, which is a climb and a slide at once. He doesn’t have to leave the pocket, his eyes stay downfield, and he is still inside the tackle box, so if nothing is open he can still throw it near a covered receiver or check it down. Escaping would only be right if the left side collapsed too.

TipAnswer

Throw it away, out of bounds past the line of scrimmage, and kick the field goal. It is legal because he is outside the tackle box. Walk through his options. Z is covered with the cornerback on top, so a throw to him is a contested catch at best and an interception at worst, and an interception here throws away a likely three points. Using the scramble-drill rules, where should each of X, H, Y and Z go, and where will the quarterback look first?

TipAnswer

X comes back down the left sideline; H takes off deep; Y keeps crossing to the left at his depth; Z works back across the field, deep. The quarterback looks first at the throws he can make on the move, X coming back and Y crossing toward him, and takes the deep shot to H or Z only if he can set his feet.

Settle each receiver by side first, then depth.

  • X is on the scramble side and deep, so he comes back: he stops his go route and works back toward the line of scrimmage along the left sideline, settling at about 8 to 10 yards, into the space his cornerback (running deep with him) leaves behind.
  • H is on the scramble side and short, so he goes deep: he turns up the field from his hitch route, behind the defenders who have turned to chase the quarterback.
  • Y is a crosser, already moving toward the scramble side, so he keeps crossing toward the quarterback, at about 12 yards: a level between X coming back and H going deep.
  • Z is on the backside. Standing on a curl 30 yards from the quarterback is useless, so he runs back across the field, deep, into the middle the defense has emptied, and into the quarterback’s line of sight.

That gives four receivers at four depths: H deep, Z deep and crossing, Y at about 12, X at 8 to 10, all moving toward the quarterback’s side. Now the look order. He is right-handed and rolling left, so before he can throw at all he has to turn his hips and square his shoulders back toward the target. The short, firm throws toward the sideline he is running at, X coming back and Y crossing, are the ones he can make like that. H and Z are the big play, worth it only if the rush gives him time to stop and set. If all four are covered he has two good options left: run, if there is grass to the line to gain, or throw it out of bounds past the line; he is outside the tackle box. What he must not do is throw back across his body to the right.

Takeaways

  • The quarterback’s eyes run on his feet’s clock. Protection check at the snap, read 1 on the last step of the drop, a hitch step for each later read, then the checkdown. Each later read costs about half a second, and value.
  • Each read is answered by a defender, and the ball leaves before the receiver is open. The quarterback looks at the key defender for each read. An anticipation throw, released before the break, beats the defenders’ reaction; a “see it, throw it” ball arrives a hitch step late into a closing window.
  • Pocket movement is a response to the rush picture. Edges pushed past: climb. Pressure from one side: slide. Interior collapsed: escape, usually outside the tackle on the side the end was run past. Eyes stay downfield throughout.
  • Off-script football is coached. The scramble drill (short goes deep, deep comes back, go where he goes, backside comes across deep, the back leaks out) turns a broken play into a second play. Off script is high variance: for most quarterbacks it nets out below the pocket; for a few (Allen, Mahomes) it adds value.
  • Bad endings have a price list. A checkdown is worth about nothing, a throwaway a down, a sack a down and yards, an interception the ball. Throwaways must avoid intentional grounding: near a receiver inside the tackle box, past the line of scrimmage outside it.
  • Four numbers profile a passer. Time to throw and aDOT describe style; CPOE (completions over expected) and EPA per dropback describe results. Because he decides most plays, and passing varies more than running, the quarterback swings more points than anyone else on the field.
