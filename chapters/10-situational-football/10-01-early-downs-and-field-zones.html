# ---- field zones, in yards from the opponent's goal line (nflverse `yardline_100`) ----------------
# backed up: own 1-10 | coming out: own 11-25 | open field: own 26 to opp 41 |
# four-down territory: opp 40-21 | red zone: opp 20 to goal
ZONES = [("backed up", 90, 100), ("coming out", 75, 90), ("open field", 40, 75),
         ("four-down\nterritory", 20, 40), ("red zone", 0, 20)]
ZONE_COL = {"backed up": style.DEFENSE, "coming out": style.LINE_TO_GAIN, "open field": style.OFFENSE,
            "four-down\nterritory": style.THIRD, "red zone": style.DEFENSE_DARK}


def zone_of(y100):
    for name, lo, hi in ZONES:
        if lo < y100 <= hi:
            return name.replace("\n", " ")
    return None


def shade_zones(ax, alpha=0.07, labels=True, y=None, fs=6.6):
    """Faint zone bands on a chart whose x axis is BDB x (own goal line 10, opponent's goal line 110)."""
    for name, lo, hi in ZONES:
        x0, x1 = 110 - hi, 110 - lo
        ax.axvspan(x0, x1, color=ZONE_COL[name], alpha=alpha, lw=0, zorder=0)
        if labels:
            ax.text((x0 + x1) / 2, y, name.upper().replace("\n", " ") if hi - lo > 18 else name.upper(),
                    ha="center", va="top", fontsize=fs, color=style.INK_2, fontweight="bold", linespacing=1.0,
                    zorder=1)


def yardline_axis(ax, fs=7):

def _P(fld, p, d, w):
    return fld.xy(*p.to_abs(d, w))


def say(fld, p, d, w, text, color=style.INK_2, fontsize=6.8, ha="center", weight="normal"):
    """A short on-field label at (d, w), on a light patch so lines don't run through it."""
    x, y = _P(fld, p, d, w)
    fld.ax.text(float(x), float(y), text, fontsize=fontsize, color=color, ha=ha, va="center", zorder=11,
                fontweight=weight, linespacing=1.1,
                bbox=dict(boxstyle="round,pad=0.18", fc=style.SURFACE, ec="none", alpha=0.92))


def point_to(fld, p, text_at, target, text, color=style.INK_2, fontsize=6.6):
    """A label with a thin leader line to the player or spot it names."""
    (xt, yt), (xp, yp) = _P(fld, p, *text_at), _P(fld, p, *target)
    fld.ax.annotate(text, (float(xp), float(yp)), xytext=(float(xt), float(yt)), fontsize=fontsize,

def clean(ax):
    """Zoomed diagrams: drop the tiny hash ticks and painted yard numbers, clip text to the window."""
    for ln in list(ax.lines):
        xs, ys = ln.get_data()
        if len(xs) == 2 and ln.get_linewidth() == 0.5 and np.hypot(xs[1] - xs[0], ys[1] - ys[0]) < 1.0:
            ln.remove()
    for t in list(ax.texts):
        if t.get_text().isdigit() and t.get_alpha() == 0.55:
            t.remove()
        else:
            t.set_clip_on(True)


def draw(p, ax=None, width_in=5.8, **kw):

# ---- frame strips (print) and videos (web) for simulated plays --------------------------------
MARK_R = 0.95   # frame-strip marker radius in yards


def tracking(play):
    df = play.to_tracking()
    df["_id"] = df["displayName"].astype(str)
    return df


def panel(ax, play, df, t, lateral, window, rings=()):
    """One moment of a simulated play: markers, trails of on-screen players, rings."""
    tmap = df.groupby("frameId")["time"].first()
    fid = int((tmap - t).abs().idxmin())
    ylo, yhi = sorted((play.ball_y - lateral[0], play.ball_y - lateral[1]))
    fld = Field(ax=ax, orient="vertical", los=play.los, window=window, numbers=False, hashes=False,
                lateral=(ylo, yhi), to_go=play.to_go)
    cur = df[df["frameId"] == fid]
    shown = cur[cur["x"].between(play.los + window[0] + 0.4, play.los + window[1] - 0.4)
                & cur["y"].between(ylo + 0.4, yhi - 0.4)]
    ids = set(shown["_id"])
    trails = df[(df["frameId"] <= fid) & df["_id"].isin(ids) & (df["side"] != "ball")]
    for (_, side), g in trails.groupby(["_id", "side"]):
        xs, ys = fld.xy(g["x"].values, g["y"].values)
        ax.plot(xs, ys, color=style.OFFENSE if side == "off" else style.DEFENSE, lw=0.9, alpha=0.4, zorder=3)
    ppy = fld.pts_per_yard()
    for _, rr in shown.iterrows():
        x, y = (float(v) for v in fld.xy(rr["x"], rr["y"]))
        if rr["side"] == "ball":
            # a carried ball sits beside its carrier so it never covers his letter
            near = cur[cur["side"] != "ball"]
            held = (np.hypot(near["x"] - rr["x"], near["y"] - rr["y"]) < 1.2).any()
            bx, by = (x + MARK_R * 1.25, y + MARK_R * 0.55) if held else (x, y)
            ax.add_patch(Ellipse((bx, by), 0.7, 0.42, fc=style.BALL, ec="white", lw=0.5, zorder=9))
            continue
        lab = str(rr.get("label", "") or "")
        if rr["side"] == "off":
            ax.add_patch(Circle((x, y), MARK_R, fc=style.OFFENSE, ec="white", lw=0.6, zorder=8))
            tc = "white"
        else:
            ax.add_patch(RegularPolygon((x, y), 4, radius=MARK_R * 1.3, orientation=np.pi / 4,
                                        fc=style.DEFENSE_TINT, ec=style.DEFENSE, lw=1.0, zorder=8))
            tc = style.INK
        fs = max(4.0, min(8.0, ppy * (1.1 if len(lab) <= 1 else 0.85)))
        ax.text(x, y, lab, ha="center", va="center", fontsize=fs, color=tc, fontweight="bold", zorder=10)
    for key, col in rings:
        rr = cur[cur["displayName"] == key].iloc[0]
        x, y = fld.xy(rr["x"], rr["y"])
        ax.add_patch(Circle((float(x), float(y)), MARK_R * 1.55, fc="none", ec=col, lw=1.7, zorder=7))
    # the throw: a dashed brown line from where it was released to where the ball is now
    for kind, _, te in play.ball_events:
        if kind == "pass" and t > te:
            (d0, w0) = play.ball_position(te)
            (d1, w1) = play.ball_position(t)
            xs, ys = fld.xy(*play.to_abs(np.array([d0, d1]), np.array([w0, w1])))
            ax.plot(xs, ys, color=style.BALL, lw=1.3, ls=(0, (2.5, 1.8)), zorder=6)
    return fld


def strip(play, times, notes, lateral, window, ncols=2, rings=(), width_in=6.4):
    """Numbered, captioned frame strip: the PDF stand-in for the video."""
    df = tracking(play)
    nrows = int(np.ceil(len(times) / ncols))
    pw = width_in / ncols
    span_w, span_d = lateral[1] - lateral[0], window[1] - window[0]
    fig, axes = plt.subplots(nrows, ncols, squeeze=False,
                             figsize=(width_in, nrows * (pw * span_d / span_w + 0.55)))
    for ax in axes.flat[len(times):]:
        ax.axis("off")
    for i, (ax, t) in enumerate(zip(axes.flat, times)):
        panel(ax, play, df, t, lateral, window, rings)
        ax.set_title(f"{i + 1}.  " + ("snap" if abs(t) < 1e-6 else f"{t:+.1f} s"), fontsize=8.5,
                     color=style.INK_2, loc="left")
        ax.text(0.5, -0.03, notes[i], transform=ax.transAxes, ha="center", va="top", fontsize=7.5,
                color=style.INK, linespacing=1.15)
    fig.tight_layout(h_pad=1.6, w_pad=0.6)
    return fig


def video(play, times, notes, lateral, window, rings=(), width_in=5.6):
    """Web version of the strip: every frame, with the strip's notes as running captions."""
    from matplotlib import animation
    df = tracking(play)
    span_w, span_d = lateral[1] - lateral[0], window[1] - window[0]
    fig, ax = plt.subplots(figsize=(width_in, width_in * span_d / span_w + 0.4))
    tmap = df.groupby("frameId")["time"].first()

    def update(fid):
        ax.clear()
        t = float(tmap[fid])
        panel(ax, play, df, t, lateral, window, rings)
        cur = [n.replace("\n", " ") for tt, n in zip(times, notes) if tt <= t + 1e-6]
        ax.set_title(("snap" if abs(t) < 0.05 else f"{t:+.1f} s") + (("   " + cur[-1]) if cur else ""),
                     fontsize=8.0, color=style.INK, loc="left")
        return []

    anim = animation.FuncAnimation(fig, update, frames=sorted(df["frameId"].unique()), interval=100)
    html = (anim.to_html5_video(embed_limit=40).replace("<video ", '<video style="max-width:100%;height:auto" ')
            if shutil.which("ffmpeg") else anim.to_jshtml(default_mode="once"))
    plt.close(fig)
    display(HTML(html))


def animation_or_strip(play, times, notes, lateral, window, rings=(), ncols=2):
    """Video on the website; a numbered, captioned frame strip in the PDF."""
    if pdf_build():
        strip(play, times, notes, lateral, window, ncols=ncols, rings=rings)
        plt.show()
    else:
        video(play, times, notes, lateral, window, rings=rings)

S["be_1_10"] = int(be[(be > 0) & (be.index >= 0)].index.min())

S["w30"] = games[games.runs > 30].won.mean()

First quarter, no score, first-and-10 at the offense’s own 25. The back runs for two yards, and the analyst approves: they’re “establishing the run.” Next series, first-and-10 again, a quick throw gains nine, and now the analyst perks up: second-and-1 is “a free play,” the perfect time for a shot down the field. (That’s booth slang. On the field a free play is stricter: a snap on which the defense has jumped offside, so the offense can’t lose.) The offense hands the ball off again. A few series later the same offense faces third-and-3 at the other team’s 38, on a down where offenses threw 86% of the time when the ball sat in their own half, and this time it runs. Nothing about the players has changed. The situation has: which down it is, how far there is to go, and where the ball sits on a field that coaches carve into zones with names. This chapter teaches you to read those three things the way a play-caller does, and to check what the broadcast booth tells you against what teams actually do.

The situation comes before the play

A play-caller does not walk into a game with a list of “good plays” and pick his favorite. He walks in with a call sheet, the laminated card you see covering his mouth on the sideline, and that card is organized by situation. One box for first-and-10, one for second-and-long, one for second-and-short, a few for third down cut by distance, then boxes for the places on the field where the rules of the game change the math: backed up against your own goal line, coming out, the red zone, the goal line. A separate panel handles the clock: the two-minute drill and the four-minute offense, where the score and the time left are the situation.

One thing sits on top of the boxes: the opening script, a list of about 15 plays the play-caller picks before the game and calls in order on ordinary first and second downs. Bill Walsh’s 49ers made it famous, and The Game Plan shows how one is built. So many of a team’s first-quarter early downs come off a list written before the game, not off the situation boxes. Keep that in mind when this chapter measures what teams did early in games: some of it was planned, not reactive.

Strip away the score and the clock and three numbers define a situation: the down, the distance to the line to gain, and the yard line. This chapter is about all three on the first two downs of a series, and about the yard line everywhere.

Start with the downs. First and second down are the early downs: the snaps where the offense still has at least one more try in hand before the down that decides whether it punts. That one extra try is what makes them different. On third down the offense must gain the yards, and the defense knows it, so third down gets its own chapter (Third Down: The Money Down). On an early down nothing is forced. A play-caller can run, throw short, throw deep, fake one and do the other, and the defense has to be ready for all of it. Early downs are also most of the game: about three snaps in four are first or second downs.

NoteMadden vs. real life

In a video game you see the same playbook menu on every snap, and the “situation” is something you notice if you feel like it. A real play-caller’s options are pre-filtered by situation before the game starts. On his own 2-yard line, half of his passing game is crossed off the sheet; on second-and-1 a box of deep shots appears that he would never call on second-and-11. When you hear a coordinator say “we were in a good situation for that call,” he means the call was on the right part of the sheet.

Two facts sit side by side in that chart, and the tension between them is the whole early-down debate.

The first fact: first-and-10 is a coin flip, and it always has been. In every season since 1999, offenses in neutral situations have thrown on between 44% and 48% of their first-and-10 snaps (46% in 2025). Over the same years the league remade itself: the shotgun went from a passing-down formation to the default, the spread arrived from college, the rules protected quarterbacks and receivers more every few years. Second down moved with it, from about 54% passes in the first five seasons of the data to about 59% in the last five. First down didn’t.1

The second fact: the pass has been the better bet on first-and-10 every single year. A dropback out-earned a designed run by between 0.14 and 0.26 points of EPA per play, and the run has averaged a loss of expected points in all 27 seasons. A tenth of a point per play, you’ll remember from 01-02, is roughly the gap between a good offense and an average one. So the obvious question: if passing on first down is that much better, why hasn’t anyone’s first-down pass rate moved?

There are four honest answers, and none of them is “coaches are fools.”

  1. The averages aren’t a fair fight. First-down runs and passes are not run against the same defenses. Offenses call passes when the defense looks ready for the run, and many first-down passes are play-action, which borrows its value from the run’s existence. Take the run away and the pass faces a defense that knows it’s coming. This is the selection effect you met with box counts in A First Look at Defense, and Expected Points, EPA, and Success Rate will deal with it head on.
  2. Balance is the point of first down. First-and-10 is the one snap where the defense can’t lean. A team that throws 80% of the time on first down has handed the defense a tendency, and the defense will play pass on first down until the edge is gone. In the language of 08-02, the run is what keeps the pass from being a guess the defense wins.
  3. Coaches price risk, not just averages. A run almost never loses ten yards or the ball; a pass sometimes does both. EPA counts sacks and interceptions in its average, but a head coach who answers for every turnover on Monday weighs a disaster more heavily than a bonus. The economist David Romer found the same lean in NFL fourth-down decisions twenty years ago.2
  4. Runs buy things EPA doesn’t price. A run keeps the clock moving when you’re ahead, keeps a pass rush from teeing off on every snap, and spares your quarterback a hit.

Weigh those and you get the position most analysts actually hold. It isn’t “never run.” It’s that the average first-down run has been a worse bet than the average first-down pass for a generation, that most teams could throw a little more on early downs before the pass’s edge wore off, and that the teams who do it well tend to have the better offenses. That “how much more than expected” idea has a name, pass rate over expected (PROE), and a chapter of its own (Tendencies: PROE, Personnel, Motion, and Coverage).

ImportantWhy it exists: balance on first down

First-and-10 is balanced because balance is what first down is for. It is the one snap where a defense has nothing to go on but the formation and the film, so a play-caller who keeps it near a coin flip forces the defense to defend everything, and makes every later down easier to call. The price is the gap in the chart: each first-down run called “to stay balanced” gives up, on average, a slice of expected points. The good offenses keep the balance and shrink the price, with runs that look exactly like their passes and passes that look exactly like their runs.

On schedule and behind schedule

Coaches don’t talk about first down in EPA. They talk about the schedule. An offense is on schedule when its early downs leave it a manageable next down: about 4 yards or more on first-and-10, then a little more than half of what’s left on second down, so that third down is short or never comes. It’s behind schedule when a short gain, a loss, a sack, an incompletion or a penalty leaves second-and-long or third-and-long. The yardage version of success rate from 01-02, 40% of the needed yards on first down and 60% on second, is the schedule turned into a number: 4 of the 10 on first down, then 60% of whatever is left (4 of the 6 after a 4-yard gain), which leaves a third-and-2.

Read the bars as the price of first down. A first-down play that loses yards (a sack, a run stuffed in the backfield) leaves a series that converts only 49% of the time; an incompletion or a no-gain run, 57%. After that there’s no cliff: every yard is worth three or four percentage points of conversion, and 4 yards gets the series to 71%. Coaches draw the line at about 4 because second-and-6 or shorter still leaves room for a run and a pass, and puts a third-and-short within one ordinary play. The EPA bookkeeping from 01-02 lands in nearly the same place: the smallest first-and-10 gain whose average EPA is positive is 5 yards. The coaches’ rule of thumb and the analysts’ model describe the same slope and draw the line within a yard of each other.

The bars also show what coaches fear most on first down, and it isn’t the incompletion. It’s the negative play. A sack or a tackle for loss (8% of these first downs) does more damage than an incompletion, because it spends the down and lengthens the distance. A holding penalty is worse still: first-and-20. That is why so much of the modern early-down passing game is built to be safe as well as productive (the quick game, screens, play-action with the quarterback rolling away from pressure) and why a quarterback who takes a sack on first down hears about it in the film room.

CautionCommon misconception

“Second-and-long is no big deal: you still have two downs.” Look at the bars again. A first-down play that gains nothing leaves a series that converts 57% of the time; a 4-yard gain leaves one that converts 71%. Either way the offense has two downs left, but two downs to gain ten yards is a much harder problem than two downs to gain six, and the defense knows it: on second-and-long it can start playing the pass.

“Establish the run”: what the claim says, and what survives

Every broadcast has an announcer who says a team needs to “establish the run”: run the ball early and often, and good things will follow. It is one of the oldest pieces of football wisdom, and it is really four different claims bundled together. The data treats them very differently.

Claim 1: running wears the defense down, so the runs get better as the game goes on. If that were true, early-down runs in close games should gain more in the fourth quarter than in the first. They barely do. From 2016 to 2025 they averaged −0.09 EPA in the first quarter and −0.07 in the fourth, a hair better and nowhere near closing the gap to passes, which averaged +0.10 and +0.09.3 Big, physical offenses do tire defenses on some days, and you’ll see it on individual drives. It just doesn’t show up as a rule.

Claim 2: you have to run to make play-action work. This one has been tested hardest, and it fails. Play-Action, Bootlegs, and Screens walks through the evidence: play-action passes beat ordinary dropbacks by about the same margin whether a team’s run game is good, bad or barely used. Linebackers react to the look of the run (the line’s first step, the back’s path) because reacting is their job, not because a running back has been gaining yards. What survives is the weaker point: the fake has to look exactly like your real run, so you need a real run to copy.

Claim 4: the run keeps the defense honest. This one survives, and it’s the reason the debate never ends. A defense that doesn’t have to respect the run can drop seven or eight players into coverage on every early down, rush four, and make the pass much harder. The run’s value isn’t only its own EPA; part of it shows up in the pass’s EPA. That is constraint theory again, and it explains why the best offenses of the analytics era did not simply stop running.   • Second-and-medium (4 to 6) is the true toss-up: offenses split almost exactly half and half, even though the pass has out-earned the run by a wide margin here too. An offense here is on schedule, and every call on the sheet is live.
  • Second-and-long (7 to 10) is the most common second down, the usual result of an incompletion or a short run. The offense is behind schedule, and its job is to get half of it back: a 5-yard gain turns third-and-long into third-and-short.   • Second-and-very-long (11 or more) follows a sack, a penalty or a loss. The goal is a makeable third down, not the whole distance, which is where the draw and the screen earn their place.

Why second-and-short is a shot down

Every coach knows the logic, and you heard it from the analyst in the opening paragraph. On second-and-1 the offense has two downs to gain one yard. It can afford to spend one of them on a shot play, a pass aimed deep down the field, because if the shot falls incomplete it’s third-and-1, the easiest third down there is. Three things make the shot unusually good here, and the first one can be measured.

1. Failure is cheap. Losing a down costs less on second-and-short than anywhere else. From second-and-1, a series converted 94% of the time; from third-and-1, 81%. A wasted second down cost 13 percentage points of conversion. From second-and-5 or longer, a wasted down cost about 24 percentage points on average: from second-and-10 to third-and-10 is 57% to 33%. The same incompletion is about half as expensive on second-and-1.9

2. The offense’s own look pulls the defense into run mode. In the NFL a defense’s personnel follows the offense’s personnel, not the down. Teams often answer second-and-short with heavy personnel (two tight ends, or a fullback) and the quarterback under center, the same look they run from. Heavy personnel draws the defense’s base personnel; base personnel on a run-heavy down puts an eighth defender in the box, usually a safety walked down; and eight in the box leaves one deep safety to cover the whole middle of the field. Linebackers on short yardage are coached to attack the line the moment they see a run action, because a yard is all the offense needs. (Against three receivers, most defenses stay in nickel and walk a safety down instead. That one-high look invites the same shot.)

3. The fake is believable. Because the look is the run look, a run fake sells. On second-and-short, 42% of passes were play-action, against 24% on second-and-long and 13% on second-and-very-long.

And when teams take the shot, it pays. On second-and-1-to-3 from 2021 to 2025, passes that traveled 15 or more yards in the air averaged +0.29 EPA per attempt, against +0.03 for runs and +0.10 for shorter throws. Yet only about 5% of second-and-short snaps were a deep throw (231 in five seasons of neutral situations).10 Two cautions belong next to those numbers. A deep-throw average only counts passes that were thrown, not the sacks taken while waiting for a deep route to open, so the true value of calling the shot is lower. And the shot is a constraint play: it works because the defense is leaning toward the run. If every offense threw deep on every second-and-1, defenses would stop leaning, and the number would shrink. The case the data makes is narrower and still strong: most offenses could take this shot more often than they do before it stopped paying.

CautionCommon misconception

“An incomplete shot on second-and-1 is a wasted down.” It’s a spent down, and it was the cheapest down the offense will have all game. The offense still has third-and-1, converted more than two times in three, with fourth-and-1 behind it in much of the field. Compare that with an incompletion on second-and-10, which leaves third-and-10 and a series that converts about a third of the time.

Why most teams still run on second-and-short

If the shot is that cheap and that good, why did offenses still run on 70% of second-and-shorts? Four reasons, and each one is real.

  • The run usually just wins. A designed run on second-and-1-to-3 gained the first down on the spot 66% of the time, and 79% on second-and-1. That’s a fresh set of downs with no third down at all, and every coach values never having to face one.
  • The shot is expensive to call. It needs a long run fake or a deep drop, eight men in protection and three seconds or more, which means only two receivers and a real chance of the drive-killing negative play from the on-schedule section. A sack turns second-and-1 into third-and-8, which is no longer cheap at all.
  • The risk weighs more than the reward. Romer’s asymmetry again: an interception on a shot is replayed all week, a stuffed run is forgotten by the next series.
  • The defense doesn’t always cooperate. Some defenses answer second-and-short with two deep safeties precisely to take the shot away, and then the right call is the run, into a lighter box.

Second-and-short isn’t the only shot box on a call sheet, either. Most staffs also keep shots for sudden change, the first snap after a takeaway, while the defense that just ran onto the field is still settling (08-03’s call sheet has a box for it); for the first first down after crossing midfield; and for coming out, which you’ll meet below.

The second down after an incompletion

One more second-down habit is worth knowing, because it’s a tendency you can see from your couch. Second-and-10 comes about two ways: after a first-down incompletion, or after a first-down run that gained nothing. Same down, same distance, so it should be the same call. It isn’t. From 2021 to 2025, offenses threw on 83% of second-and-10s that followed a no-gain run, and on only 56% of those that followed an incompletion. The pattern holds a yard or three shorter, too: on second-and-7-to-9, teams threw 73% of the time after a first-down run and 56% after a first-down pass.11

Play-callers alternate. After a pass that didn’t work, there’s a pull toward a run, to avoid two incompletions in a row and a third-and-10, and to “get back on track” with a sure few yards. After a run that didn’t work, the pull goes the other way. It’s exactly the habit that 08-02’s misconception box warned about: alternating is a pattern too. A truly unpredictable caller’s next call wouldn’t depend on his last one, because if it did, the defense could guess the next play from the last. Economists who studied NFL play-calling found the habit in the data, and found that it costs something: the habit made plays less effective.12 That’s why defensive coordinators chart it, and now you can too.

Here is each zone in plain words. The names are the common ones; exact boundaries vary by staff, and the same staff will move them with the weather, the kicker and the score.13

  • Backed up: inside your own 10. A sack in the end zone, or a holding penalty there, is a safety: two points and the ball for the other team. Even the punt is cramped. A punter normally stands 14 or 15 yards deep; with the ball at the 2, the end line is only 12 yards back, so he sets up a step in front of it, the snap arrives sooner, and a blocked punt is a safety or a touchdown (Kicking and Punting). Many staffs split this zone again: a tight box for the minus 1 to about the minus 4 or 5, where even under-center play-action is rare and the sneak and quick-hitting runs dominate, and the rest of the zone out to the minus 10. That’s why 08-03’s call sheet draws “backed up” at the 5 and this chapter draws it at the 10: both are in use.
  • Coming out: roughly your own 10 to your own 25. The emergency is over, but the offense still wants its first first down, which moves the ball away from its goal line and turns a short punt into a long one (“flipping the field”).
  • Open field: roughly your own 25 to the opponent’s 40. Field position neither forces the offense’s hand nor tempts it. Since 2025, a kickoff that lands in the end zone comes out to the 35, and about two drives in three now start at the offense’s own 25 or better.14
  • Plus territory: the opponent’s half. Coaches write yard lines there with a plus sign (“plus 35”) and in their own half with a minus sign (“minus 25”), so you’ll hear a coordinator say “we got the ball in plus territory” after a turnover, often followed by that sudden-change shot. The plus side isn’t one zone: it holds the last ten yards of the open field and the next two zones.
  • Four-down territory, which coaches also call the fringe: roughly the opponent’s 40 to its 20. A punt from here buys little field (a little too long and it’s a touchback at the 20), and a field goal is long or uncertain, so the offense plans on using fourth down if it needs it.
  • The red zone, from 01-02: inside the opponent’s 20, where the field compresses and drives cash in or settle for three. It has zones of its own (inside the 10, the goal line), which belong to The Red Zone, Goal Line, and Short Yardage.

Backed up: get out, and don’t make it worse

Inside its own 10, an offense changes character. On first-and-10 there, teams passed 33% of the time, against 49% in the open field, and their quarterbacks were in the shotgun on 36% of snaps, against 46%. The passes they did call were built not to be sacked: the sack rate on first-down dropbacks fell from 4.8% in the open field to 2.7% backed up, which means quick throws, play-action with extra blockers, and quarterbacks told to throw the ball away rather than take the loss.15 Even the throwaway has a limit down there. A quarterback still inside the tackle box who throws the ball to no one to dodge a sack is guilty of intentional grounding, and from his own end zone that’s a safety too. Here’s the puzzle: by EPA, the run is a worse bet backed up than anywhere else on the field (you’ll see it in the EPA chart at the end of the chapter), and the pass still comes out ahead on average, sacks and safeties included. Why run anyway? Partly the averages again: the backed-up passes that teams do call are the safe, quick, well-protected kind, chosen precisely because they rarely end in disaster, and a defense that expected more of them would make them harder. Partly the objective is different. Backed up, the first goal isn’t a big gain, it’s getting out: a few yards so the punter can stand at his normal depth, a first down so the next series starts with room. And partly it’s the asymmetry Romer described: a safety or an interception at your own 5 loses a game in a way a three-yard run never does. Backed-up football is the most conservative football you’ll see all day, and for most teams it’s also their least productive: on first-and-10 inside the 10, plays succeeded only 35% of the time, against 43% in the open field.

Coming out: the first first down

From its own 10 to about its own 25, an offense’s priority is the first first down, and you can put a number on why. From 2021 to 2025, after a punt from the offense’s own 10 to 15, the opponent’s next drive started, on average, at its own 43; after a punt from the own 25 to 30, at its own 28.17 One first down before the punt is worth about 15 yards of the opponent’s field position: that’s what “flipping the field” buys. You’ll see play-callers here lean on their most reliable first-down calls, the ones they trust not to go backward, but you’ll also see the quarterback come back out in the shotgun and a normal pass rate return: first-and-10 in this zone was a pass 47% of the time, against 33% backed up and 49% in the open field. Some staffs also keep a box of play-action shots for this zone, because defenses that just watched a backed-up offense run three times in a row are expecting a fourth.

The open field: the whole menu

From your own 25 out to about the opponent’s 40, the yard line stops mattering much. First-and-10 pass rates are flat across the whole stretch, and so are shotgun rates. This is where the base plays live, where sequencing does its work and where a play-caller’s balance on first down earns its keep. If you want to learn an offense’s personality, watch it here: the open field is where coaches call what they like, not what the field forces on them.

Plus territory and four-down territory: one more down to spend

Cross midfield and something changes that you can’t see on the field: the offense gains a down. Not officially, but in practice, because once a punt buys little and the kick is long, most offenses now plan to go for it on fourth-and-short. In 2025 teams went for it on 77% of fourth-and-1-to-3s between midfield and the opponent’s 21, against 33% in their own half between their 21 and midfield (outside the last two minutes of each half).18 Is going that often right? Public fourth-down models, such as Ben Baldwin’s open-source nfl4th, generally favor going for it on fourth-and-short across most of this part of the field, so the rise mostly reflects teams catching up with the math; The Clock, the Two-Minute Drill, and Fourth Down prices the decision case by case.19

The extra down changes third down. Look at third-and-2-to-4 first. (Third-and-1 is a running down everywhere, the quarterback sneak alone sees to that: offenses threw on only 24% of them, wherever the ball was. So the yard line’s effect shows a yard or three further out.) In its own half, an offense facing third-and-2-to-4 threw 86% of the time; past midfield, 73%. It is still a passing down, but the run comes back into play: with a fourth down in hand, third-and-3 becomes the first of two tries, and a run that gains two yards is no longer a failure, because it sets up fourth-and-1. Notice in the bottom panel of the chart where the switch flips: right at midfield, a little earlier than the “plus 40” that the textbook definition of four-down territory uses. On short yardage, today’s NFL treats the whole plus side as four-down territory.

The fringe of field-goal range, and the red zone

Somewhere around the opponent’s 35, a drive enters field-goal range, and the cost of a negative play changes again. A field goal is the yard line plus about 18 (the kick is spotted 8 yards behind the line, and the posts stand 10 yards deep, on the end line), so a sack at the plus 33 doesn’t just cost a down and seven yards; it can turn a 51-yard field goal into a 58-yarder or a punt. That’s why you’ll see quarterbacks near the edge of range throw the ball away rather than hold it, and play-callers avoid slow-developing dropbacks on third-and-long there: three points in hand are worth more than a small chance at a first down.

That line is moving outward. NFL kickers tried 266 field goals of 50 yards or more in 2025, against 150 in 2016, and made 69% of them, up from 57%.21 With a strong-legged kicker, “range” can start near the plus 40, and four-down territory shrinks with it, which is one reason the boundaries on the map are drawn loosely.

Inside the 20, the field shrinks. The back of the end zone works like an extra defender, there’s no room to throw over the top, and both pass rates and shotgun rates fall again as the goal line nears. How offenses and defenses play that compressed field is the next chapter but one, The Red Zone, Goal Line, and Short Yardage.

The defense reads the same map

This section has been told from the offense’s side, but the defensive coordinator’s call sheet carves up the same field. Backed up, the defense is the aggressive side: it expects a run or a quick throw, crowds the line, and is happy to pressure, because a sack or a holding flag might be a safety and even a short punt hands it good field position. In the open field it plays its base calls and its tendencies. In four-down territory a stop on third down isn’t a stop yet, so the defense is really defending two short-yardage snaps, and its third-down call can’t sell out against the run or the pass. Near the edge of field-goal range a sack is worth more than usual, because it can push the offense out of range. And on second-and-short, as the shot section showed, it mostly answers the offense’s personnel. The defense’s half of these situations gets its full treatment in the rest of Part 10.

A chart like this answers a different question: “what did each choice earn?” Read it in three steps. First, find the gap between the lines: on early downs the pass has earned more almost everywhere, by the most when the offense is backed up, where teams run the most. Second, find where the lines meet: close to the opponent’s goal line, where the defense has no deep space to protect and a run is as good as a pass. Third, remember what the chart can’t see. Every point mixes situations (the backed-up passes are the safe ones; the third-and-short runs include quarterback sneaks, the best short-yardage play in football), and every point is an average of what teams chose to do, not of what they would have earned doing more of it. The gap is a strong hint about where offenses are too cautious. It isn’t a play-call.

TipWatch for it on Sunday: the situation tag

This is the 01-02 forecaster, upgraded with the field. For one half of one game, before every snap:

  • Say the tag out loud: the down-and-distance bucket plus the zone. “Second and short, open field.” “First and ten, backed up.” “Third and three, four-down territory.”
  • Predict run or pass from the tag, then glance at the formation and change your guess if it tells you something. Mark both.
  • Flag the four situations this chapter gave you an edge in: second-and-10 after an incompletion (less pass than usual, though still slightly more likely a pass); second-and-short with one deep safety and eight defenders near the ball (shot alert: count the deep safeties before the snap); backed up (count the shotgun snaps, and expect a quick throw or a run); third-and-2-to-4 past midfield (still usually a pass, but the run comes back into play, and expect a fourth-down try if it fails).

Run or pass? And name two calls this offense is very unlikely to make.

TipAnswer

Most likely a run, from exactly this kind of look; very unlikely: a shotgun dropback, or any play that asks the quarterback to hold the ball.

On first-and-10 inside their own 10, offenses in neutral situations passed only 33% of the time (2016–2025), against 49% in the open field. Two tight ends with the quarterback under center make a run more likely still. But even this run isn’t safe by default: the quarterback takes the snap in the field of play, and the handoff comes right around the goal line, so a defensive tackle who knifes into the backfield can make the tackle at the mesh for a safety. That’s why, backed up this tight, staffs cheat the back up a yard or two (he’s only two and a half yards deep in the end zone here) and call downhill runs that hit fast: a dive, an iso, duo, the quarterback sneak.

The calls to cross off are the ones that keep the ball behind the goal line for a long time: the quarterback in the shotgun (he’d start three yards deep in his own end zone), a deep dropback, a pass that needs four seconds to develop, and any run with a deep mesh or a long sideways path (a stretch, a toss, a counter with pulling linemen), which might be caught going sideways in the end zone. A sack or a holding penalty in the end zone is two points and the ball.

The surprise you’ll occasionally see: a play-action shot from this exact look, with both tight ends and the back staying in to block, because eight defenders near the line and one deep safety is exactly the picture a shot is built to beat. It is rare, and when it hits it can go 90-plus yards. The defense knows the odds, which is why it walked the strong safety down: it’s betting on the run, and accepting the risk of the shot.

Is a pass more or less likely here than on a typical second-and-10? What does the previous play tell you, and what does the defense tell you?

TipAnswer

Less likely than usual: still a pass more often than not, but only barely (56%), against 83% when the same second-and-10 follows a no-gain run.

Second-and-10 is usually a passing down, but which second-and-10 matters. From 2021 to 2025, offenses threw on 83% of second-and-10s that followed a no-gain run and on only 56% of those that followed an incompletion (4,179 snaps). After a failed pass, play-callers lean toward a run to avoid two straight incompletions and a third-and-10.

The defense adds a second reason. Two deep safeties and a nickel back leave six defenders in the box against six blockers (five linemen and the tight end), a light box that is all but inviting a run. Many offenses build this answer into the play itself, with a run-pass option or a quarterback check at the line: against six in the box, hand it off. The defense, for its part, knows that four or five yards here are not a disaster for it; what it can’t allow is the explosive pass that a single-high shell would risk.

If a pass does come, expect something quick that gets half the distance back: a 5-yard completion turns the down into third-and-5.

Run or pass? Would your answer change if this were third-and-3 at the offense’s own 38?

TipAnswer

Still more likely a pass, about three times in four, but a run is far more likely than it would be in the offense’s own half, and if it fails, expect the offense to stay on the field for fourth down.

On third-and-2-to-4 from 2016 to 2025, offenses threw 86% of the time in their own half and 73% of the time past midfield. The yard line is the reason. At the opponent’s 38 a punt buys little and a field goal would be 56 yards (38 plus 18), so most offenses plan to use fourth down: in 2025 teams went for it on 77% of fourth-and-1-to-3s between midfield and the opponent’s 21. That makes this third down the first of two tries, and a run that gains two yards isn’t a failure; it sets up fourth-and-1. The pistol helps the run here too: the back gets the ball going downhill, and the quarterback can keep it.

It also explains why the conversion rate on these third downs barely changes between the two halves of the field (54% in the offense’s own half, 56% past midfield): the run isn’t a worse call, just a different one. At its own 38, with a punt coming if it fails, the offense would probably throw, about six times in seven, and the defense would know it.

Takeaways

  • The situation comes before the play. A call sheet is organized by down and distance, field zone, and score and clock. On the early downs (first and second) nothing is forced, which is why they are where a play-caller’s philosophy shows.
  • First-and-10 is a coin flip, on purpose. Neutral first-down pass rates have sat near 46% for a quarter century, even though dropbacks have out-earned runs there every season. Balance keeps the defense guessing; the price is the gap. Good offenses shrink the price; they don’t abandon balance.
  • “Establish the run” is four claims. Running doesn’t measurably wear defenses down, play-action doesn’t need a successful run game, and teams run a lot because they’re winning, not the reverse. What survives: a run game the defense must respect makes the passing game better, and an efficient run game (2019 Ravens) is worth calling often.
  • Stay on schedule. About 4 yards on first down is the line between a series that usually converts and one that often doesn’t. The play that kills drives is the negative play, not the incompletion.
  • Second down comes in four buckets, and second-and-short is the cheapest down in football to spend on a shot: a failed shot costs about half what a wasted second-and-long does. Teams still mostly run there, because the run usually moves the chains and the shot carries sack and turnover risk; when they do take the shot, it usually pays. And the second-and-10 after an incompletion is less of a passing down than its down and distance say.
  • The field has zones. Backed up, offenses get conservative and leave the shotgun; coming out, they chase the first first down that flips the field; in the open field the whole menu is available; past midfield, fourth down becomes part of the plan, and on third-and-short the run comes back into play.
