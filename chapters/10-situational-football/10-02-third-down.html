def clean(ax):
    """Drop the painted yard-line numbers (they sit where receivers and safeties stand)."""
    for t in list(ax.texts):
        if t.get_alpha() == 0.55:
            t.remove()


def ruler(fld, play, fs=6.3, upto=40):
    """Depth labels (yards past the line of scrimmage) down the left edge of the field window, so the
    faint 5-yard lines read as a scale. Drawn on the Field's ax: gridiron has no ruler."""
    ax = fld.ax
    x_left = FIELD_W - fld.y1 + 0.35
    first = True
    for d in range(5, upto + 1, 5):
        y = play.los + d
        if play.to_go is not None and abs(d - play.to_go) < 1.0:
            continue          # the orange line to gain would run through the label
        if fld.x0 + 0.8 <= y <= fld.x1 - 0.8:
            ax.text(x_left, y, f"{d}" + (" yd" if first else ""), fontsize=fs, color=style.INK_2,
                    ha="left", va="center", zorder=12,
                    bbox=dict(boxstyle="round,pad=0.12", fc=style.SURFACE, ec="none", alpha=0.85))
            first = False


def timed(p, key, way, kind="path"):
    """Script a player: `way` = [(t, d, w), ...] absolute spots reached at time t (s after the snap).
    Standing still between two times is allowed. (gridiron's actions are speed-based; this turns
    'be here at this moment' into the speed and delay each leg needs.)"""
    cur = np.array(p._cur(key), dtype=float)
    tc, wait = 0.0, 0.0
    for t, d, w in way:
        nxt = np.array([d, w], dtype=float)
        dist = float(np.hypot(*(nxt - cur)))
        if dist < 1e-3:
            wait += t - tc
        else:
            p._add(kind, key, [tuple(cur), tuple(nxt)], speed=dist / (t - tc), delay=wait)
            wait = 0.0
            cur = nxt
        tc = t
    return p


def protect(p, keys=("LT", "LG", "C", "RG", "RT"), back=1.2):
    """Five linemen set back in pass protection: enough to say 'pass', no more."""
    for k in keys:
        p.block(k, pts=[(-back, 0.0)])

def snapshot(play, t, ax=None, lateral=(-21, 21), window=(-9.0, 17.0), width_in=6.5, notes=(), trail_from=None,
             ruler_fs=5.2):
    """Everyone where he is at time t, with faint trails of where he has been (since `trail_from`, if given),
    and the depth ruler down the left edge."""
    pos = {k: play.position(k, t) for k in play.players}
    mv = [pl.moved(d=float(pos[k][0]), w=float(pos[k][1])) for k, pl in play.players.items()]
    q = Play([m for m in mv if m.side == "off"], [m for m in mv if m.side == "def"],
             los=play.los, ball_y=play.ball_y, to_go=play.to_go)
    q.highlights = dict(play.highlights)
    fld = q.draw(ax=ax, lateral=lateral, window=window, width_in=width_in, title=False)
    ax, ppy = fld.ax, fld.pts_per_yard()
    clean(ax)
    if ruler_fs:
        ruler(fld, play, fs=ruler_fs, upto=int(window[1]))
    PP = lambda d, w: fld.xy(*play.to_abs(d, w))  # noqa: E731
    start = play._compile()[1] if trail_from is None else trail_from
    for k, pl in play.players.items():
        if pl.side == "off" and pl.pos in ("LT", "LG", "C", "RG", "RT"):
            continue
        if t <= start:
            continue
        pts = np.array([play.position(k, tt) for tt in np.linspace(start, t, 30)])
        if np.ptp(pts[:, 0]) + np.ptp(pts[:, 1]) < 0.3:
            continue
        xs, ys = PP(pts[:, 0], pts[:, 1])
        if pl.side == "off":
            ax.plot(xs, ys, color=style.OFFENSE, lw=0.9, alpha=0.45, zorder=3)
        else:
            ax.plot(xs, ys, color=style.DEFENSE, lw=1.1, ls=(0, (3, 2)), zorder=3)
    handed = any(kind == "handoff" and t >= te for kind, _, te in play.ball_events)
    if t <= 0 or handed:            # in the quarterback's hands the ball would hide his label
        bx, by = PP(*play.ball_position(t))
        draw_marker(ax, float(bx) - (0.9 if handed else 0.0), float(by), "ball", "", ppy)
    for note in notes:              # (text, (d, w)) or (text, (d, w), (d, w) of the player it names)
        text, (d, w) = note[0], note[1]
        x, y = PP(d, w)
        box = dict(boxstyle="round,pad=0.2", fc=style.SURFACE, ec=style.BASELINE, lw=0.5)
        if len(note) > 2:
            xp, yp = PP(*note[2])
            ax.annotate(text, (float(xp), float(yp)), xytext=(float(x), float(y)), fontsize=6.0, color=style.INK,
                        ha="center", va="center", zorder=13, bbox=box,
                        arrowprops=dict(arrowstyle="-", color=style.INK_2, lw=0.7, shrinkA=1, shrinkB=5))
        else:
            ax.text(float(x), float(y), text, fontsize=6.0, color=style.INK, ha="center", va="center", zorder=13,
                    bbox=box)
    return fld


def strip(play, times, labels, notes_at=None, ncols=2, width_in=6.5, lateral=(-21, 21), window=(-9.0, 17.0),
          trails_from=None):
    """The print version of an animation: panels at chosen moments of one simulated play.
    `trails_from` = {panel index: time} starts that panel's trails later, so only the late movement shows."""
    n = len(times)
    nrows = int(np.ceil(n / ncols))
    pw = width_in / ncols
    h = nrows * (pw * (window[1] - window[0]) / abs(lateral[1] - lateral[0]) + 0.34)
    fig, axes = plt.subplots(nrows, ncols, figsize=(width_in, h), squeeze=False)
    for ax in axes.flat[n:]:
        ax.axis("off")
    for i, (ax, t, lab) in enumerate(zip(axes.flat, times, labels)):
        snapshot(play, t, ax=ax, lateral=lateral, window=window, notes=(notes_at or {}).get(i, ()),
                 trail_from=(trails_from or {}).get(i))
        ax.set_title(lab, fontsize=8.2, color=style.INK, loc="left")
    fig.tight_layout(h_pad=0.6, w_pad=0.4)
    return fig

Why third down is the money down

Start with what third down is. On first and second down a bad play is a setback; on third down it is usually the end of the drive. Convert, and the offense gets three more tries and keeps the ball. Fail, and outside field-goal range it punts, giving the ball away 40 yards downfield. That cliff is why the down gets its nickname.

Put a number on the cliff. In the 2021–2025 regular seasons, a converted third down was worth +1.8 expected points to the offense on average, and a failed one -1.3. The difference, about 3.0 points, rides on one snap.1 No other ordinary play swings that much so often: a first-and-10 that gains four yards or loses one leaves the drive alive either way. Third downs are only about 21% of all runs and passes, but they decide whether most drives become points or punts.

And the league converts them at a remarkably steady rate: between 39% and 42% in every season from 2016 to 2025, about two in five.2 Rule changes, the passing boom and the two-high defenses of the last decade have barely moved it. That stability is a clue. Third down is a contest both sides prepare for more than any other, and the offense’s gains and the defense’s answers have stayed roughly matched.

What makes third down special isn’t just the stakes; it’s the information. On first-and-10 the defense has to guard against everything, because the offense can do anything. On third-and-8 both sides know what the offense needs: eight yards, almost certainly through the air. Knowing that, the defense can specialise. It can take off run defenders and put on pass defenders, take off a big run-stopping tackle and put on a quick pass rusher, and set every underneath defender’s depth by the line to gain. The offense specialises too: its best receiver moves to where he can get a free release, its routes are cut to a depth measured from the sticks, and it calls the plays it has practised all week for exactly this distance. Third down is the down on which the situation tells both teams the most, so it is the down on which they each carry a separate plan.

NoteMadden vs. real life

In a video game you pick a play from a menu on every down, and the menu looks the same on third-and-8 as on first-and-10. In the NFL, third down has its own section of the call sheet, its own practice day (in a typical week the third-down plan is installed on Thursday, after the early-down plan; The Game Plan), its own personnel and often its own play-caller’s mindset. The plays on that page are sorted by distance: third-and-1, third-and-2 to 3, third-and-4 to 6, third-and-7 to 10, third-and-long. A coordinator calling a third-down play is choosing from a short list built for exactly this distance against exactly this opponent’s third-down habits, not from the whole playbook.

Short, 1–3 yards: converted 60%. A run can get it, so the offense’s whole menu is open and the defense has to defend both. Medium, 4–6 yards: converted 44%. The most contested distance. The offense will almost certainly throw, but it doesn’t need to throw far: a 6-yard route run well converts. The defense plays tight man coverage, mixes in pressure, and leans on mugs and disguise to make the quarterback guess. It is the bucket of option routes, rubs and mesh on one side and of Cover 1 and A-gap mugs on the other.

Long, 7–10 yards: converted 31%. Almost always a pass, and one that has to travel past the defenders waiting at the sticks. Now the offense needs a route that takes time to run, and that time is what the defense attacks: this is where the blitz peaks, because a hot throw to beat it usually lands short of the line.

Very long, 11 or more: converted 15%. The defense can concede anything short and still win, so it drops more players into coverage, plays deep, and rallies to tackle. Some offenses take a shot; many take the safe completion, the screen or the draw, and play for a better punt or a field-goal try.

And the long kinds are common: 46% of third downs needed 7 yards or more. That is why coaches care so much about first and second down: a four-yard gain on first down doesn’t look like much, but it is the difference between a third down at the top of that list and one at the bottom (Early Downs and Field Zones).

The sticks route: a yard past the line

If the target is a known distance, aim past it. That is the whole idea behind a sticks route: a route whose depth is set by the line to gain, usually so the receiver breaks or settles a yard or two beyond it. On first-and-10 a curl might be run at 12 yards because that’s where it fits the defense. On third-and-6 the same receiver runs it at 7 or 8, because what matters now is that the catch is a first down at the catch, with no need for yards after it. Coaches teach it bluntly: get past the sticks before you break, and don’t break short of them.

Why a yard or two past, and not exactly on the line? Two reasons, both about the catch point.

  1. Receivers drift back toward the ball. A receiver who breaks at 7 and turns back to the quarterback is caught at 6 and a half, short of a third-and-7. A route designed at 8 leaves room for that.
  2. Defenders sit on the sticks. On third down the underneath defenders’ landmark is the line to gain (more below). A ball caught right on the line is caught in front of a defender who is already there and only has to tackle forward.

But a sticks route can’t go too far past the line, because every extra yard of depth costs completion probability: it takes the receiver longer to get there, and the quarterback has to hold the ball against the rush. The design problem of third-and-medium is to get a receiver open as close past the sticks as possible, as fast as possible. Against the man coverage defenses favour at those distances, that means beating a defender who knows where you’re going.

The option route is the heart of it (Passing Game Fundamentals taught the rule). A man defender has to line up on one side of his receiver, and that side, his leverage, is the side he is protecting. Here the nickel sits inside H, taking away the quickest throws, the slant and the in-cut toward the middle. So H pushes to 8 yards, past the line, and breaks the other way, toward the sideline, where the nickel can’t get to him in time and the robber can’t help.

That inside alignment is a choice, not the default. With a robber in the middle, many defenses play the slot defender outside and funnel the receiver inside to the help; this one is taking away the quick slant instead. Against that outside alignment the rule sends H inside, straight toward the robber, and the quarterback has to see it and look elsewhere. And the rule has a third branch that coaches always teach: if the defender is soft, head-up, or dropping into a zone, sit it down in the open space past the sticks, face the quarterback and catch the ball. That is why the quarterback reads two things, not one: the nickel’s leverage and where the robber is standing. The receiver doesn’t guess; he reads, and the quarterback reads with him, which is why option routes go to the receivers a quarterback trusts most.

The rest of the concept exists to protect that one decision. The two go routes outside are clear-outs: they take the cornerbacks, who are in man coverage and must run with them, out of the area where H will break. Y’s seam up the middle gives the single deep safety something to worry about, so he can’t drift toward the sideline to help on H. And the back is the checkdown. If the defense blitzes, he blocks; if it doesn’t, he is the safe throw, short of the sticks, for when the option route is covered.

The option route is one of a family of third-down man-beaters, all of which do the same job by different means: give the receiver an advantage over a defender who has no help.

  • Rubs and bunches. Line receivers up close together, a bunch, and send them across each other’s paths. Each man defender has to fight through traffic to follow his receiver, while zone defenders would just swap receivers. (Dropback Concepts draws the concepts; where a rub becomes illegal offensive pass interference is in Contact Rules and Penalties as Strategy.)
  • Mesh. Two shallow crossers who brush past each other in the middle (mesh), with the same traffic problem for their defenders; the crossers usually settle a few yards past a third-and-medium line to gain.
  • Slants and quick outs. The fastest throws on the field, aimed at the receiver who has the leverage he wants before the snap. They beat a blitz too, because the ball is out before the rush arrives.
  • Matchups. The offense moves its best receiver to wherever the defense’s worst cover man will have to take him, usually the slot, against a linebacker or a safety. On third down, who covers whom often matters more than the concept, which is why defenses answer the best receiver with a bracket: two defenders on him, one under and one over, or one inside and one outside. “Take away their guy and make somebody else beat us” is a third-down plan in itself.
  • Empty. Five receivers spread across the field and nobody in the backfield (empty). It thins the box, forces the defense to show man or zone by how it lines up, and usually leaves a linebacker on a receiver. Offenses lined up in empty on 19% of third-down dropbacks in 2020–2022, against 12% on first and second down.4 The price is protection: with no back to block, a sixth rusher is the quarterback’s problem, and he has to throw hot.

Why does all this aim at man coverage? Because that is what defenses play on third-and-short and medium. In 2023–2025, charted third-down dropbacks at 1 to 3 yards to go were played in man coverage 58% of the time, and at 4 to 6 yards 58%; at 11 or more it fell to 26%.5 A zone defense on third-and-4 has defenders spread across the field waiting for the ball, and any route that finds the space between two of them, five yards deep, converts. Man coverage takes that space away by putting a defender on every receiver. The offense’s reply is to make that defender’s life as hard as possible for about a second and a half.

Man coverage has one more price on third down: flags. A man defender is attached to his receiver, and when he is beaten his last resort is a grab. On charted third-down dropbacks in 2023–2025, defenders were flagged for pass interference, defensive holding or illegal contact on 4.8% of snaps in man coverage, against 1.6% in zone, about three times as often. All three give the offense an automatic first down whatever the distance. Across 2021–2025, about one third-down snap in 25 (4.0%) ended in a first down by defensive penalty, pass interference and holding the most common.6 The conversion rates in this chapter leave those out, so they slightly understate how often drives survive third down, and every one of those flags is a quiet cost of playing man.

ImportantWhy it exists

The sticks route exists because on third down the distance is fixed in advance, so the offense can aim at one yard and give the receiver the right to choose his break. What it gives up: a route set to the sticks hands the defense a target depth too, and the two sides have been trading moves over that yard ever since. Defenders learn to sit at the sticks and drive on the break; receivers answer by selling the sit and turning upfield (a “stick-nod”), quarterbacks by throwing to the back shoulder, away from the defender’s drive. Defenses answer with man coverage from depth and safeties over the top, so that the receiver who turns upfield runs into help; offenses answer that with rubs. That steady trading of moves is why the league’s third-down rate barely moves.

Throws aimed one or two yards past it converted 63%, the best of any group. Notice the dip for throws aimed right at the line (52%, with completions falling off too): that is very likely, at least in part, the sticks-sitting defender at work, and it is the reason the sticks route is set a yard or two beyond.

So the broadcasters are mostly right. But “past the sticks” is a rule about averages, and there are four situations in which a throw short of the line is the right one.

  1. Third-and-short. Even a catch behind the line can convert when the line is only a yard or two away. Completed throws short of the sticks converted 80% of the time on third-and-1 and 69% on third-and-2, against 36% on third-and-7 to 10 and 11% on 11 or more. On third-and-1 a pass to the flat, caught a yard behind the line with nobody near the receiver, is a good play; it’s a run with a better start.
  2. The receiver who will run after the catch. Screens and quick throws to the offense’s best open-field runner are short by design: the bet is on the yards after the catch, and against a defense playing deep it can be a good bet.
  3. Four-down territory. When an offense near midfield has already decided it will go for it on fourth down if it needs to, third-and-6 is really “third-and-6 with two tries.” A five-yard completion that sets up fourth-and-1 is then a fine outcome (Early Downs and Field Zones has the field zones; The Clock, the Two-Minute Drill, and Fourth Down the decision).
  4. Field-goal range and field position. On third-and-12 at the opponent’s 35, a 7-yard completion doesn’t convert, but it turns a 52-yard field-goal try into a 45-yard one. Deep in his own end, an offense facing third-and-15 may simply want a safe gain and a punt from further out.

How often do offenses throw past the line? It depends heavily on the distance: 78% of aimed passes on third-and-1 to 3 were thrown at or past it, 68% at 4 to 6, 52% at 7 to 10 and only 29% at 11 or more. The longer the distance, the more the defense can make the deep throw hard and the short throw easy, and the more often the quarterback takes the short one.

CautionCommon misconception

“A completion short of the sticks on third down is a quarterback mistake.” Sometimes. But on third-and-11 or more the defense’s whole design is to make the short completion the only open one, with everybody else deep and waiting at the line to gain. A quarterback who forces the ball into that coverage risks an interception to chase a first down that converts less than one time in five even when everything goes right. Taking the checkdown, getting five yards, and punting from further out is often the better play. The mistake is the other one: the third-and-5 checkdown that gets four, when a receiver was open at six.

Third-and-long: screens, draws and the punt

On third-and-12 the odds are poor whatever the offense calls: dropbacks converted 16% of third-and-11-or-more snaps in 2021–2025. Here a lot of offenses stop trying to win the down outright and start trying to win the possession: to avoid the sack, the interception and the holding penalty, and to make the punt that follows a little shorter. Two plays do most of that work, and both turn the defense’s third-and-long plan against it.

The screen lets the pass rush through and throws behind it. On third-and-long the defense’s rushers are going all-out for the quarterback and its coverage players are deep and facing the quarterback, so a back or receiver who catches the ball behind the line with linemen in front of him has room. The draw shows a pass, then hands the ball off up the middle, into the space the linebackers just left by dropping into coverage. Its mechanics, and Paul Brown’s Browns who made it famous, are in Gap Schemes. Here is why it lives on third down.

Watch what the defense’s third-and-long plan does to itself. It has taken off a linebacker for a sixth defensive back, so there is one fewer big body near the line. Its remaining linebacker drops to 10 yards at the snap, because that is where the passing windows are. Its rushers come upfield as fast as they can, because they are trying to reach a quarterback who is about to throw. The draw uses all three: the linemen let the rushers run themselves past the play, and the back runs into the space the dropping linebacker left. Against a defense playing the pass, it’s one of the few runs with numbers on its side.

And it still usually fails to convert. Designed runs on third-and-11 or more converted only 5% of the time in 2021–2025, because the defensive backs who dropped to the sticks are waiting there to tackle. So why call it? Partly because a 9-yard run on third-and-12 is a better outcome than a sack, an interception or an incompletion, especially near midfield, where it improves the punt or sets up a long field-goal try. Partly to protect the next third down: a defense that has been gashed by a draw rushes a half-step more carefully the next time, and its linebacker drops a half-step later. That is a constraint play, a call that exists to keep the defense honest against another (Constraint Theory).

The same logic explains the third-and-long screen, which also rarely converts but pays in the same currency. Both are cheap ways to lose the down safely, with a small chance of winning it big against a defense that guessed wrong.

The defense’s third-down package

When the defense knows the offense will throw, it changes the people on the field. That is a third-down package: the personnel and calls a team keeps for third down. On defense it means more defensive backs, faster pass rushers, and the pressures and disguises it has practised all week. On offense it means its best quick concepts, its protections for the opponent’s pressures, and the receivers and back it trusts most on the money down.

Pass rushers. The same logic applies up front. A 330-pound run-stopping tackle who is excellent on first-and-10 is wasted on third-and-9, so many defenses replace him with a lighter, quicker rusher, or put four edge rushers on the field together, with two of them lined up inside over the guards. Steve Spagnuolo’s 2007 Giants had a package of four pass rushers that Sports Illustrated later called his “Four Aces”; by 2011 the Giants’ version, four defensive ends on third-and-long, was known as their “NASCAR” package, for its speed.9 Most NFL defenses now have some version of it, and many rosters carry a rusher or two who play mostly on passing downs. And the most common third-down “pressure” is not a blitz at all: it is four rushers running a stunt, an end and a tackle crossing paths so that one of them comes free through a gap the protection thought was covered (The Pass Rush). Sticks defense: keep it in front

The simplest third-down defense is the one that makes the offense’s own target its landmark. If the offense needs 12 yards, the defense can give up 11. Coaches call it playing the sticks: zone defenders drop to (or just past) the line to gain, keep every receiver in front of them, and tackle the catch short of it.

Why does this work? Because the offense’s routes have to be finished to be useful. A curl at 14 yards takes about two seconds to run; a defender who starts at 12 and only has to take one step and tackle has almost no ground to cover. And the defense doesn’t need to stop the catch, only the extra yard. This is very likely part of the reason for the dip in the past-the-sticks chart at throws aimed right at the line: the defender standing on the line to gain catches the receiver there.

What does it give up? Everything short. A sticks defense on third-and-12 hands the offense an open 6-yard route on almost every snap, which is fine until the offense is in four-down territory and happy to take it, or has a receiver who can make a man miss. It also gives the quarterback time, because the underneath defenders are reading him rather than rushing him. And it asks for good tackling: a defense that misses the first tackle at the sticks has turned a stop into a first down. The cure for the short completion is a fast defense that rallies to it; for the time, mixing in pressure; for the tackling, there is no scheme.

Man under, two deep: 2-Man

Zone at the sticks has a cousin built on man coverage. In 2-Man (also called Cover 5), two safeties split the deep field in halves and the five defenders underneath play man, usually in trail technique: a step behind the receiver and on his inside hip, daring him to run past, because a safety is waiting over the top. It is the classic third-and-medium-to-long man call against option-route offenses. The underneath defender can sit on the short break, the sticks route, with no fear of the deep ball, and the receiver who turns upfield runs into a safety. Defenses played it on 8% to 9% of charted third-and-4-to-10 dropbacks in 2023–2025, against 5% on early downs.11

The price is the rush and the middle. Two deep and five in man leaves four rushers and nobody to spy or rob, so a quarterback who escapes the pocket has open grass in front of him: the man defenders are running with their receivers, backs turned. That is why 2-Man is a favourite against pocket passers and a risk against quarterbacks who run (the spy, below, is the answer to that). The pre-snap tell is two high safeties with the cornerbacks and the nickel tight and inside their receivers; the trap is that the same picture can turn into a two-deep zone at the snap.

Pressure on third down: simulated pressure and the blitz

The second family is pressure. Zone Blitzes, Fire Zones, and Simulated Pressure built the tools; here is why third down is where they live.

On third down the quarterback has to hold the ball until a route past the sticks is open, so every fraction of a second the defense takes away is worth more than on any other down. Pressure that forces the ball out early forces it out short, usually to a hot receiver in front of the line to gain, and on third-and-8 a hot throw that gains five is a defensive win.

That is why the blitz peaks at third-and-7 to 10. In 2023–2025 defenses sent five or more rushers on 38% of third-and-7 to 10 dropbacks, against 25% on first and second down. At that distance the hot throw is short of the sticks, so the defense can live with it; the routes that convert take time to develop, so pressure can kill them. At 11 or more the blitz rate falls to 17%: the defense doesn’t need to force the ball out, because a short throw already loses. At 4 to 6, blitzing is more dangerous, because the hot throw can convert, so blitz rates there are lower, and most defenses that do blitz play man coverage behind it, not zone (the third-down chart in Zone Blitzes, Fire Zones, and Simulated Pressure counts it).

Simulated pressure fits third down even better. It shows a blitz, gets the protection to commit and the quarterback to plan a hot throw, then rushes four from unexpected places and drops seven, often with a lineman sitting in exactly the short window the hot throw was aimed at. On third-and-long the quarterback who throws hot into that window gains nothing, and the one who waits for a route past the sticks is facing a rusher his protection didn’t count. To see why “didn’t count” matters, look at the other side of the ball.

The offense’s answer: count, slide, keep the back in

Remember the quarterback in the opening, pointing at a safety and shouting. He was counting. Before every third-down snap the offense names the defenders it thinks can rush, picks one as the reference point (the Mike identification), and every blocker counts his man from him, so all of them block the same picture (Pass Protection). On third down that call does most of the work:

  • Slide toward the threat. The line slides toward the side the extra rusher is most likely to come from, blocking gaps rather than men, so a stunt or a late blitzer runs into a blocker.
  • Six-man protection. The back stays in to take the first extra rusher (the half-slide is the everyday version), which answers a five-man pressure with six blockers. The cost is one fewer receiver, on the down when the offense most needs a man open.
  • Hot and sight adjust. When the defense brings more than the protection can block, the quarterback throws hot to the receiver whose defender just blitzed, and a receiver who sees the same blitz may cut his route short (sight adjust). That is the defense’s bet: the hot throw is fast, and on third-and-8 it is usually short.

So when a simulated pressure walks a safety down and then drops him, it isn’t just “showing blitz.” It is attacking the count: the protection slid toward the safety, the back was told to check him, and the fourth rusher comes from the other side, where the count left a gap. Simulated pressures are designed to beat the protection’s counting rules (“who’s the fifth?”), and that is why third down is where they live.

Cover 0: all in

The most aggressive third-down call is Cover 0: man coverage on every eligible receiver, nobody deep, and everyone else rushing. Man Coverage and Blitzing drew it; this section is about when.

Across the league it is above all a short-yardage call. On third-and-1 to 3, Cover 0 was played on 9% of charted dropbacks in 2023–2025; on third-and-11 or more, 2%. Why short? Two reasons.

  1. A completion costs less. With nobody deep, any completion that beats its man can go a long way. But on third-and-2 the defense loses the down on almost any completion anyway, so the bet risks less than on third-and-10, where a short completion is still a stop and the defense would be giving up its safe outcome to gamble.
  2. Every defender is close to the line, which helps against the run. On third-and-short the offense might run; Cover 0 puts eight or nine defenders within a few yards of the ball.

Some coordinators use it at longer distances too, for a third reason: it forces the decision now. A quarterback facing six rushers and no safety has well under two seconds. He must throw hot, against tight man coverage, before the route has developed. On third-and-2 that hot throw can convert, but on third-and-6 or 7 the routes fast enough to beat the rush mostly end short of the line. That is the Cover 0 of the film rooms below. Spagnuolo’s Chiefs and Flores’s Dolphins called it on short third downs like everyone else, but kept calling it on long ones, where most of the league nearly stops: on third-and-7 or longer, the 2021 Dolphins played Cover 0 on 24% of charted dropbacks (first in the NFL; league median 3%), and the 2025 Chiefs on 10% (second; median 2%).13 They were betting that the hot throw would land short. The league habit (short yardage, for the run and because the completion converts anyway) and theirs (medium and long, because the completion lands short) are two answers to the same question: where is the quick completion least damaging?

What it gives up is everything behind it. A Cover 0 blitz that doesn’t get home is one broken tackle away from a touchdown, and offenses that see it coming (both safeties walking down, corners pressed, linebackers mugged in the A gaps) answer with slants, rubs, and protection that keeps an extra blocker in.

Rush three, drop eight

At the other end of the dial is a call that sounds like giving up: rush three, drop eight. Three rushers, eight defenders in coverage, usually on third-and-long.

The logic is the sticks defense taken to its limit. On third-and-12 the offense needs a completion past 12 yards, and eight defenders can put a body in every window at that depth and still keep three deep. The quarterback is offered the checkdown and little else.

The price is time. Three rushers against five blockers will rarely get home quickly: the offense has two spare linemen to double-team the best rusher, and the quarterback can hold the ball for four seconds or more. Given that long, receivers can run a second route, and a quarterback who can move will find the gaps in the rush and scramble through them. That is why rush-three calls often come with the next section’s idea attached, a defender assigned to the quarterback, and why defenses mix them with pressure: a quarterback who sees three rushers on one third-and-long can’t assume he’ll get time on the next.

How rare is it, and does it work? Across 2023–2025 defenses rushed three or fewer on 8% of third-and-11-or-more dropbacks and 4% at 7 to 10. On third-and-7 or longer, offenses converted 21% against three or fewer rushers, 24% against four and 31% against five or more.16 Sack rates run the way you’d expect: 7.7% with three rushers, 12.4% with five or more.

Read those conversion rates carefully, because they seem to say the blitz is the worst call on third-and-long. Most of that is the distance hiding inside the average. Three-man rushes come mostly on the very longest downs, which are the hardest to convert whatever the defense does, and blitzes mostly at 7 to 10, which are easier. Hold the distance still and the gap nearly closes: on third-and-7 to 10 alone, offenses converted 31% against four rushers and 34% against five or more. So the blitz doesn’t stop third-and-long more often than a four-man rush, on average. It does change how the down ends: more sacks, more hurried throws, and the occasional receiver left alone. That near-tie is roughly what you’d expect if coordinators are mixing their calls well. If one call were clearly better, everyone would call it more, offenses would prepare for it, and its edge would shrink. The blitz earns its place by being one option in a mix the quarterback can’t predict, not by winning every time it is called.

Some defenses lean on it. Brian Flores’s 2023 Vikings, the league’s heaviest blitzers that season, also rushed three or fewer on 30% of their third-and-7-or-longer dropbacks, the most in the NFL: a defense that might send seven or three, from the same look. In 2025 the leaders were Jesse Minter’s Chargers (19%; Minter is the Ravens’ head coach in 2026) and Kelvin Sheppard’s Lions (18%).

ImportantWhy it exists: rush three, drop eight

What it solves: on third-and-long the offense needs a completion past the sticks, so the defense puts its numbers where the completion has to happen, and accepts a free short throw. What it gives up: the pass rush. With only three rushing, the quarterback gets time, and time eventually beats any coverage. It works best against quarterbacks who don’t run and offenses without a receiver who can turn a checkdown into twelve yards, and it works only as one call in a mix.

The spy

The last family answers a problem that third down’s own calls create. Drop eight into coverage, or even seven, and the coverage covers everyone except the man with the ball. Play man coverage, and it is worse: every man defender is running with his receiver, back turned to the quarterback, so nobody sees him take off. (Zone defenders at least face the quarterback and can come up when he runs.) A quarterback who can run looks downfield, sees every receiver covered, sees the rush lanes open, and goes. Scrambles were only 6% of third-down dropbacks in 2021–2025, but on third-and-7 to 10 they converted 48% of the time, against 32% for dropbacks as a whole.17 Against the quarterbacks who scramble most, that is a weapon a defense has to plan for.

The defense’s answer is the spy (or QB spy): a defender assigned not to a receiver or a rush lane but to the quarterback himself. He stands at linebacker depth, usually 5 to 7 yards off the ball and in front of the quarterback, keeps his eyes on him, and mirrors him. If the quarterback throws, the spy has done nothing, which is the price. If he runs, the spy meets him.

Who spies? Someone fast enough to match a running quarterback and big enough to tackle him: usually a linebacker, sometimes a safety or nickel back against the fastest quarterbacks, occasionally a defensive lineman who “rushes” by stepping up and sitting in the middle. A linebacker assigned to the back can also be a green dog (some staffs say hug or add-on), who joins the rush only if the back stays in to block (Linebackers and the Second Level). The spy comes with a rush plan too: against a running quarterback the rushers play a mush rush (some say cage rush), staying square in their rush lanes and pushing the blockers back into the quarterback instead of racing around them (The Pass Rush). The ends keep contain, the pocket shrinks around him but doesn’t open, and the spy is the last line if he finds a crease anyway.

The trade is the familiar one: the spy is a defender who isn’t covering anyone. In Cover 1 he usually replaces the robber, so the middle of the field loses its best in-cut defender; in zone he is one fewer underneath player. A spy can also be used. Offenses run a receiver right at him to make him choose, or send the back on a route he has to respect, and a linebacker spying a quarterback who is faster than he is can be outrun to the sideline. Like every third-down call, it works best as one of several from the same look.

Is third down a skill?

Third down feels like the purest test of a team, and it is a big one. But it is also a small sample: a team faces roughly 200 to 230 third downs a season, and every one is a coin weighted by distance. Two consequences are worth knowing.

First, early downs decide a lot of third down. A team that is on schedule faces third-and-3, not third-and-9, and converts more often without being any better at third down. How do we know? With a correlation, a number between 0 and 1 (for this kind of comparison) that measures how well one season’s figure lines up with the next: 0 means last year tells you nothing about this year, 1 means it tells you everything, and anything around 0.2 or 0.3 is a weak link, real but swamped by luck. Across 2016–2025, a team’s success rate on first and second down predicted its third-down conversion rate the next season (correlation 0.33) nearly as well as its own third-down rate did (0.39).20 Neither is strong, and the early-down number, which comes from about four times as many plays, does almost as well.

Second, defensive third-down numbers are noisy. A defense’s third-down conversion rate allowed carried over from one season to the next with a correlation of only 0.20: barely more than nothing. A defense that “gets off the field” one year often doesn’t the next, which is a reason to judge third-down defenses on what they do (pressure, coverage, tackling) more than on the conversion rate alone. Offenses are more stable, but the gap between the best and worst is still modest: in 2025 the best offense converted 51% of its third downs, the worst 32%.

None of that makes third down unimportant; it makes it consequential and noisy at once, which is exactly what you’d expect of a single high-stakes snap. Expected Points, EPA, and Success Rate has the tools to measure it.

That looks like a contradiction: throws aimed right at the line are the chart’s weak spot. Welker beat the dip for the two reasons the dip exists. It comes from balls thrown into a defender sitting at the sticks; the option route against zone finds the gap between two of them and sits down in it, facing the quarterback. Why did it work? Because the option route attacks the defense’s third-down plan at its core. Against man coverage the slot defender has to choose a side, and the receiver breaks the other way; against zone the receiver sits down in the gap at the sticks, facing the quarterback. Brady could throw before the break was finished, because he and his receiver read the same defender. What to notice on old Patriots third downs: the slot receiver’s first few steps are the same every time; the break comes at the sticks; the ball is in the air before he turns.

TipWatch for it

Past the sticks? On every third down for one half, make a prediction before the snap and check it after.

  • Say the bucket out loud: short (1–3), medium (4–6), long (7–10) or very long (11+). Then make the base-rate call: short, run or quick pass; medium, quick man-beater; long, a route past the sticks against pressure; very long, a safe completion or a draw.
  • Read the offense. Did the best receiver move to the slot? Is the back still in the backfield (likely to stay in and block a blitzer) or did the offense go empty (five out, and the quarterback has to beat any extra rusher himself)?
  • Read the package. How many defensive backs came on? Did a big tackle leave for a pass rusher? How many defenders are within five yards of the line, and how many deep?   • Find a spy. Against a running quarterback, look for a linebacker or safety sitting in front of him who isn’t lined up on any receiver.

If it wants the first down, the soft spot of a two-deep shell is the middle of the field between and under the safeties, 14 to 18 yards deep: a dig or a seam, thrown on time. If it wants the possession, the draw and the screen are built for this picture: a light box, one linebacker who will drop, rushers chasing upfield. From its own 20, a 9-yard draw that sets up a punt from the 29 is a reasonable outcome, and an interception at the 35 is the disaster to avoid.

What is the Mike’s job, what coverage is behind him, and where is the offense’s best throw?

TipAnswer

The Mike is probably a spy, in Cover 1. Every receiver has a defender pressed or tight on him (the corners on X and Z, the nickel on H, the safety on Y, the Will on the back), one safety is deep, and the Mike is lined up on nobody, square to the quarterback at spy depth. Against a quarterback who runs on third-and-5, the defense has given up its robber to keep him in the pocket.

Where to throw: not right in front of the spy. His eyes are on the quarterback, so he sees every throw that crosses his face at 5 to 8 yards, and spies rob those throws all the time. The good throws are the ones he can’t get to:

  • Behind him and in front of the free safety: a dig or an over route at 10 to 14 yards. In Cover 1 with no robber, that is the space the defense emptied, and the spy can’t turn and run back to it while watching the quarterback.
  • Away from leverage, outside: an out or option-out by a slot receiver whose defender is aligned inside him, toward the sideline the spy can’t reach.
  • Make the spy choose: send the back on a route right at him, or a receiver across his face, so that covering the route means leaving the quarterback.

And the quarterback still has his legs: a spy who is slower than him has to be right about the angle.

Takeaways

  • Third down decides drives, about three points of expected value on one snap, and the league converts about two in five, every year. The distance is the whole story: short third downs are converted most of the time, very long ones rarely.
  • The distance is information, so both sides specialise. Defenses add defensive backs and pass rushers as the distance grows; offenses set their routes by the line to gain, put their best receiver where he can win, count the protection, and keep a third-down back.
  • Aim a yard or two past the sticks. Third-down throws aimed there convert most often; throws short of the line rarely do, except on third-and-short, in four-down territory, or when the yards after the catch are the plan. Option routes, rubs and mesh exist to win that yard against man coverage, which costs the defense flags as well as catches.
  • The defense picks a point on a dial: Cover 0 and tight man on short yardage, pressure (real and simulated) and 2-Man on third-and-7 to 10, sticks defense and rush three/drop eight on very long, and a spy when the quarterback can run. Each call gives something up, none wins on average once the distance is held still, and the best defenses show the same look for all of them.
  • Third-and-long is often about the possession, not the down. Screens and draws rarely convert, but they avoid disasters, shorten punts, and keep the rush honest for the next one.
  • Third down is consequential and noisy. Early-down success predicts it nearly as well as past third-down success does, and defensive third-down rates barely carry over year to year.
