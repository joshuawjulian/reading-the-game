def line5(split=1.6, names=False, d=LINE_D):
    """Five offensive linemen; only the center is lettered unless names=True."""
    out = []
    for k, m in (("LT", -2), ("LG", -1), ("C", 0), ("RG", 1), ("RT", 2)):
        out.append(O(k, k, d, m * split, k if names else ("C" if k == "C" else "")))
    return out


def spread(rb_w=-1.5, x_w=-15.0, h_w=-9.0, y_w=9.0, z_w=15.0, y_pos="TE"):
    """11 personnel, shotgun 2x2: X and Z on the line (seven on the line with the five linemen),
    H and Y a step off the ball in the slots."""
    return line5() + [O("QB", "QB", GUN, 0.0, "Q"), O("RB", "RB", GUN, rb_w, "R"),

def route_to(p, key, *pts, delay=0.0, speed=None):
    """A route through absolute (d, w) points."""
    start = np.array(p._cur(key))
    return p.route(key, [tuple(np.array(q) - start) for q in pts], frame="field", delay=delay, speed=speed)


def move_to(p, key, *pts, kind="path", delay=0.0, speed=None, target=None):
    """Any other movement through absolute (d, w) points: path, run, block or pull."""
    start = np.array(p._cur(key))
    rel = [tuple(np.array(q) - start) for q in pts]
    if kind == "block":
        return p.block(key, pts=rel, delay=delay, speed=speed)
    if kind == "pull":
        return p.pull(key, rel, target=target, delay=delay, speed=speed)
    if kind == "rush":
        return p.rush(key, pts=rel, delay=delay, speed=speed)
    return {"path": p.path, "run": p.run}[kind](key, rel, delay=delay, speed=speed)

    """Drop the painted yard numbers unless asked (they sit where receivers stand); clip all text."""

def set_tracks(p, tracks, t_end=None):
    """Hand-made keyframes {key: [(t, d, w), ...]} for some players, after all routes are added.
    gridiron's man() ends on top of the receiver and cannot be slowed by traffic, so defenders
    who get rubbed, and every body in a pile, are keyframed by hand."""
    kf, t0, t1 = p._compile()
    kf = dict(kf)
    for key, rows in tracks.items():
        q = p.players[key]
        rows = [(min(t0, -1.0), q.d, q.w), (0.0, q.d, q.w)] + [tuple(r_) for r_ in rows]
        kf[key] = np.array(rows, dtype=float)
    p._compiled = (kf, t0, max(t1, t_end or 0.0, *[rows[-1][0] for rows in tracks.values()]))


def ease(points, step=0.05):
    """Smooth keyframes through (t, d, w) waypoints (ease in and out on each leg)."""
    rows = []
    for (ta, da, wa), (tb, db, wb) in zip(points[:-1], points[1:]):
        for tt in np.arange(ta, tb, step):
            f = (tt - ta) / (tb - ta)
            f = f * f * (3 - 2 * f)
            rows.append((round(float(tt), 3), da + (db - da) * f, wa + (wb - wa) * f))
    rows.append(tuple(points[-1]))
    return rows


def frame(ax, p, t, lateral, window, rings=(), ltg=True, numbers=False, mr=0.75):
    """Everyone at time t, with faint trails of where they have been and the ball's flight so far."""
    ylo, yhi = sorted((p.ball_y - lateral[0], p.ball_y - lateral[1]))
    fld = Field(ax=ax, orient="vertical", los=p.los, window=window, numbers=numbers, lateral=(ylo, yhi),
                to_go=p.to_go if ltg else None)
    ppy = fld.pts_per_yard()
    wl, wr = sorted(lateral)
    ring = dict(rings)
    for k, q in p.players.items():
        d, w = p.position(k, t)
        if not (window[0] + 0.4 < d < window[1] - 0.4 and wl + 0.4 < w < wr - 0.4):
            continue
        if t > 0:
            pts = np.array([p.position(k, tt) for tt in np.linspace(0.0, t, 40)])
            if np.ptp(pts[:, 0]) + np.ptp(pts[:, 1]) > 0.3:
                xs, ys = P(fld, p, pts[:, 0], pts[:, 1])
                ax.plot(xs, ys, color=style.OFFENSE if q.side == "off" else style.DEFENSE, lw=1.0,
                        alpha=0.5, zorder=3)
        x, y = P(fld, p, d, w)
        draw_marker(ax, float(x), float(y), q.side, q.label, ppy, ring=ring.get(k), r=mr)
    for kind, to, te in p.ball_events:
        if kind == "pass" and t > te:
            ts = np.linspace(te, t, 20)
            b = np.array([p.ball_position(tt) for tt in ts])
            xs, ys = P(fld, p, b[:, 0], b[:, 1])
            ax.plot(xs, ys, color=style.BALL, lw=1.4, ls=(0, (2, 1.5)), zorder=9)
            ax.add_patch(Ellipse((float(xs[-1]) - 0.9, float(ys[-1])), 1.0, 0.62, fc=style.BALL, ec="white",
                                 lw=0.5, zorder=11))
    clean(ax, numbers)
    return fld


def strip_grid(p, times, notes, lateral, window, rings=(), ncols=2, width_in=6.5, deco=None, fs=6.9, ltg=True,
               mr=0.75):
    """Print stand-in for an animation: numbered panels, each with a sentence underneath."""
    nrows = int(np.ceil(len(times) / ncols))
    pw = width_in / ncols
    ph = pw * (window[1] - window[0]) / abs(lateral[1] - lateral[0])
    fig, axes = plt.subplots(nrows, ncols, squeeze=False, figsize=(width_in, nrows * (ph + 0.8)))
    for ax in axes.flat[len(times):]:
        ax.axis("off")
    for i, (ax, t) in enumerate(zip(axes.flat, times)):
        fld = frame(ax, p, t, lateral, window, rings, ltg=ltg, mr=mr)
        if deco:
            deco(t, fld, p)
        lab = "before the snap" if t < -1e-6 else ("snap" if abs(t) < 1e-6 else f"+{t:.1f} s")
        panel_title(ax, f"{i + 1}.  {lab}")
        under_note(ax, notes[i], fs=fs)
    fig.tight_layout(h_pad=3.2, w_pad=0.8)
    return fig


def video(p, lateral, window, t_end, rings=(), width_in=5.6, t_start=-0.4, notes=None, times=None, ltg=True,
          mr=0.75, deco=None):
    """Web version of a strip: the same frame drawn every tenth of a second, with the strip's
    notes as a running caption."""
    from IPython.display import HTML, display
    span_w, span_d = abs(lateral[1] - lateral[0]), window[1] - window[0]
    fig, ax = plt.subplots(figsize=(width_in, width_in * span_d / span_w + 0.5))
    ts = np.round(np.arange(t_start, t_end + 1e-9, 0.1), 2)

    def update(i):
        ax.clear()
        t = float(ts[i])
        fld = frame(ax, p, max(0.0, t), lateral, window, rings, ltg=ltg, mr=mr)
        if deco:
            deco(max(0.0, t), fld, p)
        txt = "before the snap" if t < -1e-6 else ("snap" if abs(t) < 1e-6 else f"+{t:.1f} s")
        if notes and times:
            cur = [n for tt, n in zip(times, notes) if tt <= t + 1e-6]
            if cur:
                txt += "   " + cur[-1].replace("\n", " ")
        ax.set_title(txt, fontsize=8, loc="left", color=style.INK)
        return []

    fig.tight_layout()
    anim = animation.FuncAnimation(fig, update, frames=len(ts), interval=100, blit=False)
    html = (anim.to_html5_video(embed_limit=40).replace("<video ", '<video style="max-width:100%;height:auto" ')
            if shutil.which("ffmpeg") else anim.to_jshtml(default_mode="once"))
    plt.close(fig)
    display(HTML(html))


def pct(x, nd=0):
    return f"{100 * x:.{nd}f}%"

SEAS = [2022, 2023, 2024, 2025]

PB = pbp(SEAS, columns=_cols)
PB = PB[PB["season_type"] == "REG"].copy()
_ftn = nfl().load_ftn_charting(seasons=SEAS).to_pandas()[["nflverse_game_id", "nflverse_play_id", "is_qb_sneak"]]
PB = PB.merge(_ftn, left_on=["game_id", "play_id"], right_on=["nflverse_game_id", "nflverse_play_id"], how="left")
# FTN has no tush-push flag: is_qb_sneak covers the plain sneak and the pushed version alike
PB["sneak"] = PB["is_qb_sneak"].eq(True) & PB["play_type"].eq("run")
PB["conv"] = PB["first_down"].eq(1) | PB["touchdown"].eq(1)
# scrimmage plays only: no kicks, no kneel-downs, and no tries (the extra point is snapped from the 15 and the
# two-point try from the 2, so leaving them in would invent "red-zone trips" for every touchdown)
PLAYS = PB[PB["play_type"].isin(["pass", "run"]) & PB["qb_kneel"].ne(1) & PB["down"].notna()
           & PB["two_point_attempt"].ne(1)].copy()

# --- short yardage: sneaks vs everything else, 1 yard to go
ONE = PLAYS[PLAYS["ydstogo"] == 1]
SNK1 = ONE[ONE["sneak"]]
RUN1 = ONE[(ONE["play_type"] == "run") & ~ONE["sneak"] & ONE["qb_scramble"].ne(1)]
PAS1 = ONE[ONE["qb_dropback"] == 1]

The field runs out

Start with geometry, because everything in this chapter follows from it. The red zone is the last 20 yards before the goal line. Behind the goal line is the end zone, 10 yards deep, and behind that the end line, the back edge of the playable field. A pass caught beyond the end line is out of bounds. So from the opponent’s 20, the offense has at most 30 yards to work with; from the 10, 20 yards; from the 2, where two-point tries are snapped, 12.

Why does that matter so much? Because of what defenses do between the 20s. A normal defense spends its safeties, and sometimes its corners, on the deepest part of the field, 15, 25, 40 yards away, because a receiver who gets behind everyone there scores. The defense has to keep its depth. Near the goal line that problem disappears: the end line protects the deepest part of the field for free. It is the twelfth defender that never misses a tackle. Every defender who had been guarding grass 30 yards downfield is now available to do something else: crowd the receivers, cover a man, or rush. That is the compressed field: less grass to defend, the same eleven defenders to defend it.

Defenses use that freedom in three big ways, and you can see all three on any red-zone snap.

Deep zones shrink, then disappear. A zone defense asks defenders to guard areas. The picture above shows what happens if the defense stays in zone: the deep areas get squeezed against the end line until they are barely deeper than the underneath ones, the hole between the two layers closes, and the spaces between defenders, the “windows” a quarterback throws into, get small. But a zone with nothing behind it is wasting its deep defenders, which is why, inside the 10, many defenses stop pretending: with no deep middle to protect, they bring the safeties down. In FTN’s charting of 2023–2025 dropbacks, two-deep shells (Cover 2, Cover 4 and their relatives, which keep two safeties deep) were called on 43% of plays outside the red zone but only 18% inside the 5.1

Man coverage and press take over. Man coverage’s great weakness is the receiver who beats his man deep, with no help behind. Near the goal line, “deep” doesn’t exist: a receiver who runs past his man runs out of the end zone. So defenses play much more man coverage, and much more of it press, with the defender a yard from the receiver at the snap, jamming him to wreck the timing. In Man Coverage you saw that charters label roughly four in five dropbacks inside the 5 as man, and that Cover 0 (man coverage with nobody deep at all) becomes the single most common call there.

The box fills up. Every defender not needed in deep coverage can line up near the ball. That squeezes the run game: the extra defenders take away the cutback lanes and the creases a running back finds between the 20s.

The freed defenders have two more jobs, both from Man Coverage. A safety who no longer needs to stay deep can rob the middle: in a Cover 1 robber or lurk call he sits a few yards deep in the middle of the end zone, reading the quarterback, waiting for the slant or the crossing route. And a defense with a defender to spare can double the best target: a bracket, two defenders on one star tight end or receiver, is expensive at midfield and affordable when nobody is needed deep.

The offense suffers in quieter ways too. Its deep routes are gone, so the threat that stretches the defense vertically, the one that keeps safeties honest between the 20s, is gone with them. Every route is shorter, so every throw arrives with defenders closer to the catch point. Yards after the catch shrink, because there are fewer yards left to gain. And the playbook narrows: concepts that need 15 or 20 yards of depth to develop (the deep crosser, the post, most of Dropback Concepts) simply don’t fit.

One rule quietly tilts the last few yards toward the run. A touchdown is scored the instant any part of the ball, held by a runner, breaks the plane of the goal line, the invisible wall rising from its front edge. A quarterback stopped dead on a sneak scores if he got the ball over the line before he was stopped. A receiver has a harder test: to make a catch in the end zone he must control the ball and get both feet (or a knee) down in bounds, and every red-zone throw near the sideline or the end line is a fight with that boundary as well as with the defender.2

ImportantWhy it exists

The compressed field isn’t a scheme; it’s a consequence of the rulebook’s geometry. But it explains a whole family of schemes. Defenses get more aggressive near the goal line because the end line gives them deep help for free: man coverage, press and Cover 0 are cheap there in a way they never are at midfield. Offenses answer with plays that win quickly and close to the line (slants, rubs, fades thrown before the safety can help, quarterback runs) or that create a numbers advantage the defense can’t cover. The trade: everything is faster and more crowded, so mistakes cost more. A tipped ball near the goal line is an interception risk in a crowd; a one-yard loss on third-and-goal can be the difference between seven points and three.

What each yard is worth

The numbers make the squeeze concrete. Across the 2022–2025 regular seasons, 6,881 drives ran at least one play inside the opponent’s 20. Of those, 57% ended in a touchdown and 29% in a field goal; the rest ended in turnovers, failed fourth downs, missed kicks or the end of a half.3 That 4-point gap between a touchdown (with its extra point) and a field goal is the whole economics of the red zone, and it is why coaches judge their red-zone offense by the share of trips that end in touchdowns, not by yards.

CautionCommon misconception

“At the goal line, smart teams pass; running into a crowd is old-fashioned.” From the 1, passing and handing off come out about even, and the quarterback sneak beats both. In 2022–2025, dropbacks snapped from the 1 scored on 52% of plays and ordinary runs on 55%, while quarterback sneaks scored on 67%.5 (These are rates per play; the 90% in the chart above is per drive, which gets up to four tries. The sneak section returns to why one yard at the goal line is harder than one yard anywhere else.) A pass needs a receiver to beat tight man coverage in a tiny space; a run needs one yard. Neither play is always right, and the best goal-line offenses keep both alive so the defense can’t sell out against either.

Red-zone pass concepts

Because the field is short and the coverage is tight and usually man, red-zone pass plays share a style. They are quick (the ball is out in a second or two), they attack a specific matchup or leverage, and many of them are built to create separation that man coverage can’t prevent, rather than to find holes in zones. Here are the six you will see most.

The fade

The fade is the simplest red-zone pass and the most argued-about. The receiver releases outside his defender and runs for the back corner of the end zone; the quarterback throws the ball high and outside, over the receiver’s outside shoulder, toward the pylon, to a spot where only the receiver can catch it. It needs room, so the receiver usually lines up with five to eight yards between himself and the sideline: room for the ball to land outside him and still in bounds, with space for both feet. Line up much wider and there is no room for the ball; much narrower, and the safety in the middle has a shorter run to help. And it needs a matchup: a tall receiver with strong hands, ideally against a smaller corner.

Why does it fit the red zone? The fade beats a pressed corner with no deep help outside, and the compressed field produces plenty of those. If the corner lines up inside the receiver (inside leverage, to take away the slant), the receiver releases outside, and the corner is left chasing from behind and inside, unable to see the ball. The safety is usually in the middle of the end zone, too far away to arrive before a ball that is in the air for barely a second.

The back-shoulder throw

Against a corner who has learned to run with the fade, stride for stride and over the top, the back-shoulder throw is the answer. The quarterback throws short and behind the receiver, to his back (outside) shoulder. The receiver, reading the throw, stops, turns back to the ball and catches it, while the defender, running full speed with his back to the quarterback, is carried a step or two past. A defender taught to take away the deep ball can’t take this away without opening the deep ball again.

The fade is famous for being thrown too often. It looks easy (one receiver, one defender, a jump ball) and it fails a lot: the ball has to be put in a small space near the sideline and the end line, and a corner who stays on the receiver’s hip can knock it away at the last moment. In 2022–2025, passes thrown into the end zone toward the outside from inside the 10, a rough stand-in for fades and back-shoulder throws that the public data can’t separate, were completed 40% of the time, against 68% for throws that stayed short of the goal line. But a completion in the end zone is a touchdown, and a short completion usually isn’t: per throw, the end-zone throws scored 40% of the time and the short ones 26%.6 So the fade is not a bad play. It fails more often than it works, but when it works it ends the drive in the best possible way; a short completion often just moves the ball a few yards closer and leaves another down to win. The danger is the offense that has nothing else and throws it on every third-and-goal, where a defense that knows it is coming can sit on it.

Contact helps the fade too. Defensive pass interference in the end zone gives the offense a first down at the 1, the most valuable spot on the field (nine drives in ten score from there, as the first-and-goal chart showed), so a corner who grabs a fading receiver is gambling too. The other way round, penalties near the goal line lose their teeth: half the distance means a 5-yard foul on the 4 moves the ball only 2 yards.

NoteFilm room: Green Bay Packers, the Rodgers–Nelson years (2010s)

The back-shoulder throw belongs to many quarterbacks, but its most famous partnership was Aaron Rodgers and Jordy Nelson in Mike McCarthy’s Green Bay offense. The Packers’ Hall of Fame profile of Nelson notes that the throw began as a red-zone play and that the two of them took it all over the field; Nelson, 6-foot-3, described it as a reaction rather than a set call: no signal, no pre-snap adjustment, just the ball arriving behind him when the corner ran too well.7 What to notice: the receiver’s head. On a fade he keeps running and looks up late; on a back-shoulder throw he snaps his head around early, sinks his hips, and the corner keeps running. If you see a corner win a fade early in a game, watch for the back-shoulder throw against him next time.

The slant, and leverage

If the corner shades outside to take away the fade, the receiver has the inside, and the quick slant (two steps up, then a sharp break inside) is the answer. It is caught a yard or two into the end zone, in the space between the corner and whoever is playing the middle, before either can close. Often a second receiver runs to the flat underneath it, the slant-flat pairing from Quick Game, so the defender between them can’t take both. Put the two together and you have the red zone’s first lesson about leverage: the corner can take away the fade or the slant, not both, and the receiver’s release attacks the side he gives up.

Defenses know the odds, and most red-zone corners play inside leverage on purpose. They use the sideline and the end line as help and concede the fade, the lowest-percentage throw on the field, to take away the slant, a high-percentage one thrown at their backs. The leverage game is a bet on both sides: the corner bets the quarterback can’t make the hard throw, and the offense bets on its receiver against that corner.

The slant’s risk lives in the middle of the field. The throw goes inside, where linebackers and safeties stand, so a defender who reads it can jump in front. That’s what happened on the most famous red-zone slant ever thrown, coming up in the film room on rubs.

The sprint-out

The fourth concept moves the quarterback. On a sprint-out he takes the snap and runs toward one sideline, with a group of routes flooding that side at three depths: high (a corner route to the back pylon), middle (an out at the goal line) and low (a flat route in front of the goal line). The sprint-out does three things near the goal line. It cuts the field in half, so the quarterback reads only one side. It moves the launch point away from the middle of the pocket, where interior rushers come free quickest. And it makes the quarterback a runner, so the defender in the flat faces a choice: cover the receiver in front of him, or come up to stop the quarterback. Either answer is wrong.

Rubs: the man-beater’s home field

The rub is the red zone’s signature pass. A rub route is a route designed to cross close to a teammate’s, so a defender chasing one receiver has to go around the other. Near the goal line it is at its most powerful, for a simple reason: the compressed field produces the man coverage rubs are built to beat. A zone defender doesn’t care if two receivers cross in front of him; a man defender who has to follow one particular receiver does.

Offenses set up rubs with alignment. Two receivers lined up one behind the other (a stack) or three in a tight triangle (a bunch) put their defenders close together, where any crossing route creates traffic. Watch the animation (the frame strip in print): the front man in a stack runs a slant inside, the back man runs out toward the pylon, and the nickel back covering the back man has to get through the slant to follow him. The stack can be run either way round: the front man can run the “pick” and the back man the slant, as Seattle did in the Super Bowl coming up below. What matters is that one receiver’s route crosses the path of the other’s defender.

def catch_time(p, which=0):
    """When the ball arrives, the way gridiron times a pass (ball speed and the receiver's path)."""
    from gridiron.play import BALL_SPEED, _interp
    kf, _, _ = p._compile()
    kind, to, te = [e for e in p.ball_events if e[0] == "pass"][which]
    start, t_arr = p.ball_position(te), te + 0.5
    for _ in range(4):
        t_arr = te + float(np.linalg.norm(_interp(kf[to], t_arr) - start)) / BALL_SPEED
    return round(t_arr, 1)

When does a rub become a pick? The receiver in front may run his route; he may not block. The NFL calls offensive pass interference on a receiver who deliberately initiates contact with a defender more than one yard past the line of scrimmage, before the pass is touched, to get a teammate open. So offenses teach the rub man to run through the defender’s path rather than at the defender, and to keep running, and they design the crossing to happen as close to the line as they can. Officials judge intent at full speed, and a quarterback can’t know in advance whether the flag will come, which is part of why rubs are both effective and maddening.

How defenses answer. The cleanest answer is not to be in pure man. Against a stack or bunch, defenders can play a banjo: instead of each locking onto one receiver before the snap, they decide at the release, the outside defender taking whoever breaks outside and the inside defender whoever breaks inside, so nobody has to fight through the traffic. Another answer is to stop the rub at its source: a defender who jams the front receiver at the line knocks the traffic out of the way before it can form. That second answer decided a Super Bowl.

The shovel pass

The shovel pass is a run play that happens to be a pass. The quarterback takes the snap and moves sideways along the line, as if rolling out or keeping the ball; a teammate (a tight end, a back, sometimes a slot receiver) comes across just in front of him, behind the line of scrimmage, and the quarterback flips the ball to him underhand. The receiver then runs behind a pulling lineman into the gap the defense has just vacated.

Why does it work so well near the goal line? Because goal-line defenses are built to attack. Defensive ends charge upfield at a rolling quarterback; linebackers fly toward the first thing they see. The shovel pass turns that aggression into empty space: the end who chases the quarterback has taken himself out of the gap the ball is about to come through.

The flip is a forward pass, and that has two consequences. The good one: a dropped shovel is merely an incomplete pass, not a fumble. The constraint: because it counts as a pass, the offensive linemen may not run upfield until it is thrown. The NFL’s ineligible-downfield rule applies to every forward pass, even one caught behind the line: no lineman may be more than a yard past the line before the throw unless he got there while blocking.9 So the flip comes fast, while the linemen are still fighting at the line, and only then does the pulling guard lead the way up the field. College spread offenses made the shovel option famous; in the NFL it is a change-up, used a few times a season, usually near the goal line or on a short third down. Andy Reid’s Kansas City offenses have used it for years: the go-ahead touchdown in a 27–20 win over Philadelphia in September 2017 was a 15-yard shovel to tight end Travis Kelce, from a look that for its first half-second resembled a quarterback keep.10

Two-point plays: designing a play from the 2

After a touchdown, the scoring team gets one more snap, a try. It can kick an extra point for one point or run a play from the 2-yard line for two: the two-point conversion. The NFL adopted it in 1994, borrowing it from college football (which snaps its try from the 3) and from the American Football League, the NFL’s 1960s rival, which had used it throughout its ten seasons. In 2015 the league moved the extra-point kick back to the 15-yard line, turning a near-automatic kick into a 33-yard attempt, and began letting the defense score two points by returning a failed try to the other end zone.11 (The rules are unchanged for 2026: still snapped from the 2.) Whether to go for two, and when the scoreboard demands it, belongs to The Clock, the Two-Minute Drill, and Fourth Down. This section is about what to call.

From 2015 to 2025, NFL teams converted 47% of their 1,297 regular-season two-point tries (counting only tries that produced a run or a pass, not the ones wiped out by penalties). Runs converted 55% of the time, passes 44%, yet teams passed on about 73% of tries.12 (Is going for two worth it at all? Two points times 47% is about 0.94 points a try, almost exactly what the 94% extra-point kick is worth. That near-tie is why the scoreboard, not the average, decides, and it is 10-04’s subject.) Why not run every time?

Part of the answer is selection. The teams that run on two-point tries are mostly the ones built to: a running quarterback, a heavy line, a reliable short-yardage play. Passes include the desperate tries late in games, when everyone knows the offense needs two and the defense has called its best pass defense. The bigger part is game theory. A two-point try is a single snap the defense knows is coming, from a spot it has studied all week. If an offense ran every time, defenses would load the box with every man they have, and the run rate would fall. The value of a two-point play depends on what the defense expects, and the best calls are ones that work against more than one defense.

The defense’s side of the try looks like any goal-to-go snap. It matches the offense’s personnel rather than bringing in a goal-line package: on the 2023–2025 tries charted with personnel, the defense had five or more defensive backs on the field 71% of the time and a goal-line group of three or fewer only 9%, because most offenses went for two with three wide receivers. Its coverage is the red zone’s, only more so: 72% of charted two-point dropbacks faced man coverage.13

That gives a simple design rule for a two-point play: two good options that put one defender in conflict, ideally a run threat and a pass, or a man-beater and a zone-beater on the same snap. Here are four common shapes. (The public data can’t say which shape converts best: play-by-play records the run or the pass, not the concept, so the case for each is in the football, not in a success rate.)

Each shape answers a defensive call. Against the man coverage that dominates the goal line, the bunch rub (A) is the classic: three receivers starting close together, two crossing just behind the third, and three defenders who must either fight through traffic or switch assignments perfectly. Against zone, the snag triangle (B), from Quick Game (the snag itself is a receiver who runs five yards and sits down facing the quarterback), puts three receivers in one area guarded by two defenders: whichever receiver the defenders leave, the quarterback throws to. The sprint-out option (C) doesn’t care much which coverage it gets, because it attacks one defender with two threats. And the quarterback run (D), the extra-hat idea from Option Football, gives the offense one more blocker than a handoff can, which matters most when two yards is all it needs.

Notice what’s missing: deep routes, long developing reads, anything that needs a second and a half to come open. From the 2 the end line is 12 yards away, and a pass rush arrives in about two and a half seconds. Two-point plays are designed to be decided by the quarterback’s first look.

Goal-line personnel: the biggest people on the field

Inside the 3 or so, both teams change their players. The offense brings in its heaviest personnel: two or three tight ends, a fullback, sometimes a sixth offensive lineman (a jumbo package) who has to report as eligible: tell the referee, who announces it, because a lineman’s number normally makes him an ineligible receiver. And sometimes the extra body is a defensive lineman playing fullback. The most famous of those was William “The Refrigerator” Perry, the Chicago Bears’ rookie defensive tackle, who scored a 1-yard touchdown in Super Bowl XX in January 1986.15 The idea is simple arithmetic: every receiver taken off the field is replaced by a blocker, so the offense can match the defense body for body near the ball.

The defense answers in kind with a goal-line defense: extra linemen and linebackers in place of defensive backs, everyone lined up close to the ball, every gap along the line filled before the snap. The two classic shapes are the 6-2 (six down linemen, two linebackers, three defensive backs) and the 5-3 (five down linemen, three linebackers, three defensive backs). Both put eight defenders within about three yards of the ball.

Why two shapes? Against an offense with a tight end at each end of the line there are eight gaps to fill (A, B, C and D on each side, from The Language of the Run Game). The 6-2 puts six of those eight defenders on the line, in the gaps, low, and lets the two linebackers fill the last two: it is the stronger shape right at the ball, built to win the first yard by force. The 5-3 takes one lineman off the line and adds a third linebacker. It gives up a little of that force, but it gains a defender standing up, who can see the play, fill whichever gap the run goes to, or pick up a tight end or fullback who slips out for a pass. So the 6-2 suits the last yard against an offense that just pounds the ball, and the 5-3 suits a defense that fears play-action from the same look.

Football coaches have a three-word rule for this part of the field: low man wins. When two big men collide at the line, the one whose pads are lower gets under the other, lifts him and drives him back. On an ordinary down that matters; at the goal line it is almost everything, because the play will be decided within a yard of the ball. So goal-line defensive linemen often line up in four-point stances, both hands on the ground, and fire under the blockers, aiming to knock the offensive line back into the backfield and create a pile the runner can’t cross. Linebackers fill whatever gap is left, often by going over the top of the pile.

ImportantWhy it exists

A goal-line defense exists because near the goal line the run is a real threat on every snap and the pass has little room to hurt you. Trading cornerbacks and safeties for linemen and linebackers trades coverage, which the compressed field has made cheap, for gap control and size, which the last yard has made precious. What it gives up: coverage. Goal-line defenses are built to stop a run up the middle, and their worst nightmare is the play that looks exactly like that run and isn’t: the play-action pass to a tight end or a fullback slipping out, or a quarterback rollout to the edge where fewer defenders are standing. That is why, when heavy personnel does throw near the goal line, it almost always fakes the run first: inside the 3 in 2023–2025, offenses with one wide receiver or none passed on only 28% of snaps, but 66% of those dropbacks were play-action, against 31% from lighter groupings.16

The quarterback sneak

The shortest play in football is the QB sneak. The quarterback takes the snap under center and immediately drives forward behind his center, usually into one of the gaps beside him, to gain a yard or less. No handoff, no fake, no read.

It is the most reliable play in the book for one yard, and the reasons are worth spelling out:

  • The ball starts closest to the line. On a handoff, the ball goes backward to a running back five to seven yards deep, who then has to gain all of that back before he gains anything. On a sneak, the ball is a foot behind the line when the quarterback has it.
  • It is the fastest play there is. The ball goes from the center’s hands to the quarterback’s and forward in a fraction of a second, before linebackers can read the play and fill.
  • The quarterback is the extra blocker’s dividend. On every other run, the quarterback hands off and stops being a factor; on the sneak he is the runner, so every other player can block, the same extra-hat arithmetic as QB power.
  • The defense can’t see it coming any differently. A quarterback under center on third-and-1 might sneak, hand off, or fake and throw. Defenses have to defend all three.

And its blocking is simple. Coaches call it wedge blocking: every lineman drives low and forward toward the center’s hip, so the front moves as one block with no seams for a defender to slip through, often with the center and a guard double-teaming whoever lines up over the ball. The quarterback’s coaching point is just as simple: follow the center’s backside into the first crease. And the cadence, the quarterback’s call that tells his line when the ball will be snapped, is a weapon either way. A quick count snaps the ball on the first sound, before the defense has settled into its stances; a hard count, a long, loud call with a sudden bark meant to draw the defense offside, aims for the free yard without snapping it at all.

The tush push

The tush push is a quarterback sneak with help from behind. Two or three teammates (usually a running back and a tight end, sometimes a third player) line up directly behind the quarterback, and at the snap they drive into him and push. The offensive line fires low and forward; the quarterback, crouched behind his center, surges into the line; the pushers add their weight from behind, and the whole mass moves forward, usually just far enough. Philadelphia fans called it the Brotherly Shove.

Why does a shove from behind help that much? Physics, mostly. A sneak is a pushing contest at the line: the offense’s linemen and quarterback against the defense’s linemen and linebackers, both sides trying to move a pile a yard. Two pushers behind the quarterback add a few hundred pounds of driving force at exactly the point of the contest, and they add it immediately, at the snap, before the defense’s second level can arrive. The defense can’t add the same force, because its extra bodies have to come from the other side of a pile they can’t get into. And the quarterback, pushed from behind, can stay low and keep his feet driving instead of being stopped upright by the first linebacker to hit him.

Notice what the push gives up. A plain sneak hides inside a formation that could also hand off or throw; the push announces itself, because nobody lines up two players against the quarterback’s hips for anything else. The push trades disguise for force, and over one yard, force wins far more often than not.

How it became legal: assisting the runner

For most of the NFL’s history, the tush push was illegal. The rule on assisting the runner said teammates could not push or pull a ball carrier forward, a 10-yard foul. In the mid-2000s (accounts differ by a year) the NFL removed “push” from the rule and kept the ban on pulling. The reason was practical: in a pile, officials could never tell who was pushing whom (a runner, a teammate in contact with the runner, the pile itself), so the rule was close to unenforceable. As Mike Pereira, another former officiating chief, put it, pushing “became really too difficult to officiate,” so the league simply made it legal.19 Pulling is still a foul: a teammate who grabs the runner’s arm or jersey and hauls him forward costs his team 10 yards. Watch for it in a tush-push pile. The college rulebook followed in 2013.20

The timing carries a footnote of its own. On October 15, 2005, USC’s Matt Leinart scored the winning touchdown at Notre Dame with running back Reggie Bush shoving him across the goal line from behind, the “Bush Push.” College rules still banned pushing then, no flag was thrown, and the play became a legend: the illegal, unforgettable ancestor of a play that is now legal in the NFL.21

For more than 15 years after the change, NFL teams pushed runners only in passing, as piles formed. Then the Philadelphia Eagles turned it into a designed play.

The ban debate

The push’s success made it controversial. Critics made three arguments: safety (a pile of the heaviest players on the field driving forward with heads down), aesthetics (a near-automatic yard is dull football), and fairness (one team’s mastery of a play is a reason to rethink it). Defenders of the play answered that injuries on it were rare, that other teams could copy it, and that you don’t ban a play for being executed well.

In 2025 Green Bay proposed a ban. Its first version, barring teammates from pushing the quarterback immediately at the snap, was tabled at the spring league meeting; a broader revision, barring any pushing, pulling, lifting or assisting of the runner except by blocking opponents, came to a vote on May 21, 2025, and failed 22–10, two votes short of the 24 needed (rule changes need three-quarters of the 32 owners). The league tightened its officiating guidance instead. In practice that meant watching for linemen who move before the snap (a false start, easy to miss when everyone is crouched shoulder to shoulder) and for players lined up in or across the neutral zone, the ball-length strip between the two lines; the officiating department’s own training video in September 2025 said an Eagles guard should have been flagged for moving early on one push.25 For 2026, no team proposed a ban; the competition committee’s co-chairman said in February 2026 that he didn’t expect one, and the league said it would only “lightly” revisit how the play is officiated. The tush push is legal for the 2026 season.26

CautionCommon misconception

“The tush push is unstoppable.” It isn’t, and the next section’s numbers show that. Defenses stop it by winning the snap: crowding both A gaps beside the center, timing the count so their linemen are moving first, and sending linebackers over the top to meet the quarterback before the pile moves. (Leaping over the pile is legal on a running play; many fans assume it isn’t.27) Defensive linemen also try to jump the snap count, accepting the occasional offside penalty for the chance of getting into the backfield early. What the push does is shift the odds of a one-yard contest, and on one yard that shift is worth a great deal.

Does it work? The sneak in the data

FTN’s charting service marks every play it thinks was a quarterback sneak (is_qb_sneak), and the flag covers the plain sneak and the push alike, because FTN does not chart them separately. So the numbers below are for sneaks of every kind.28

On 1 yard to go, 2022–2025 regular seasons, sneaks gained the first down or the touchdown 81% of the time (1,089 plays). Other runs on 1 yard to go converted 70%, and passes 58%. A gap of 11 percentage points on a play that decides whether a drive lives is an enormous difference. With 2 yards to go, sneaks were far rarer and far less successful (54% of 37), which tells you what the play is for: the last yard, not the last two. A sneak is a pile moving forward, and a pile of a dozen big men moves a yard, maybe a yard and a half, before it stalls (watch the frame strip above). Ask it for two and it usually stops short.

You have now met three numbers for “one yard”: sneaks from the 1 scored on 67% of plays, sneaks on any 1-yard-to-go snap converted 81%, and drives with first-and-goal at the 1 scored 90% of the time. They measure different things. The drive number is highest because a drive gets up to four tries at the yard. And the yard before the goal line is harder than the yard before a first down: there the defense’s back line is the goal line itself, so it can put every defender within a few yards of the ball, and it brings in its goal-line personnel to do it. In the middle of the field, a defense crowding the line on third-and-1 still has to worry about a pass over its head.

Two lessons hide in that chart. First, volume, not rate, is what separates the teams. Most teams’ conversion rates overlap once you account for sample size: 40 sneaks is not enough to tell an 80% team from an 85% team. What differs enormously is how often teams call the play. The Eagles stand alone at one end; at the other, Kansas City was charted with 1 in four seasons, because Andy Reid’s offense almost never asks Patrick Mahomes to sneak.

If the sneak converts that often, why don’t the other teams run it more? None of the reasons is foolish. The sneak puts the quarterback, usually the most valuable and most expensive player on the roster, at the bottom of a pile of the biggest men on the field. Many modern offenses live in the shotgun, so their quarterback rarely takes a snap under center, and the sneak is a play they would have to practise specially. And the play needs a line that can win the first half-second together, which not every team has. A coach weighing a few extra yards a season against those costs can reasonably say no.

Second, a high rate is partly selection. Teams sneak when the distance is truly short, often inches rather than a full yard (the play-by-play records both as “1 to go”), and most teams sneak only when they like the look. That explains a puzzle you may have noticed. The push adds hundreds of pounds of force, yet Philadelphia’s sneak rate in the chart sits close to the league line, and Hurts’s mark in the film room is no better than Brady’s or Allen’s on their own. The Eagles called the play on about half of all their 1-yard downs: on full yards as well as inches, against defenses that knew it was coming and lined up to stop it. Other teams sneaked only when the spot and the look were good. Comparing the two rates compares a play run every time with a play run only when it was easy, which is why the Eagles’ rate is more impressive than it looks. How much the push really adds once you account for that is a harder question, taken up in the Eagles case study.

Putting it together: a red-zone possession

Here is how the pieces fit on one drive. First-and-10 at the opponent’s 18: the defense still has some depth to protect, so you’ll see two-high safeties, zone coverage and the offense’s normal playbook. A completion to the 7 makes it first-and-goal, and the picture changes: the safeties come down, the corners press, and the coverage is mostly man. The offense answers with quick throws: a slant or a fade on the outside if it likes a matchup, a rub from a bunch if it sees man, a sprint-out if it wants an easy read. Inside the 3, both teams change personnel: jumbo against goal-line defense. Now it’s a run, a quarterback sneak or push, or the play-action pass that the heavy personnel sets up. And if the drive ends in a touchdown, there’s one more decision: kick, or one more play from the 2.

TipWatch for it on Sunday: fade or rub?

When the ball crosses the 20

  • Count the safeties. Two deep at the 18 is normal; one or none inside the 10 means the defense has traded depth for bodies near the ball.
  • Look at the corners’ depth. A corner a yard off the receiver is pressing: expect man coverage.

Inside the 10: read the receivers’ splits

  • A receiver alone, wide, with 5 to 8 yards to the sideline? That spacing is for a fade or back-shoulder throw. Check the corner’s leverage: lined up inside him, think fade; outside him, think slant.
  • Receivers bunched or stacked, tight to the formation? That’s rub territory. Predict a crossing pattern against man, and watch whether the defenders switch (banjo) or chase through the traffic.
  • The quarterback in the gun with receivers flooded to one side? Look for a sprint-out to that side.

Inside the 3

  • Heavy personnel on both sides? Watch the quarterback: under center with players stacked directly behind him is a tush push; a fullback and tailback deep behind him is a handoff or play-action.
  • On the push, watch the snap and the first half-second, not the pile: whichever line fires first and lower usually wins.

After a touchdown

  • If the offense stays on the field, guess the concept before the snap from its formation: bunch (rub), spread with a back beside the quarterback (QB run or sprint-out), or heavy (run or play-action).

Takeaways

  • Near the goal line the end line becomes the deepest defender. The compressed field squeezes zones, frees safeties to come down, and makes man coverage, press and Cover 0 cheap. The offense loses its deep threat, and every throw arrives into a crowd.
  • Red-zone pass concepts win quickly and close: the fade and back-shoulder throw against press with no outside help, the slant when the corner takes away the fade, rubs from stacks and bunches against man, the shovel pass against an attacking front, and the sprint-out that puts a flat defender in conflict.
  • A good two-point play gives the offense two answers that put one defender in conflict, so it works against more than one defensive call. Runs have converted more often than passes since 2015, partly because of who chooses to run, and a try is worth about what the kick is, so the scoreboard decides.
  • Goal-line defenses (6-2, 5-3) trade defensive backs for linemen and linebackers and fill every gap; low man wins. Their weakness is the play that looks like the run and isn’t, which is why heavy personnel that throws near the goal line almost always fakes a run first.
  • The QB sneak is the best one-yard play in football: the ball starts closest to the line, the play is instant, and the quarterback-as-runner adds a blocker. The tush push adds pushers behind him; it has been legal since the NFL removed the ban on pushing the runner in the mid-2000s (2005 or 2006; pulling is still a foul), survived a 22–10 ban vote in 2025, and is legal in 2026.
  • In the data, a little under six red-zone trips in ten end in touchdowns; a first-and-goal from the 1 scores about nine times in ten (a drive gets four tries); and sneaks convert about four in five on 1 yard to go, but only about half the time on 2. Teams differ far more in how often they sneak than in how well, and the teams that sneak most face the hardest looks.
