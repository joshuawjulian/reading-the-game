def T(t, side=1):

def line(labels="none", puller=None):
    """The five offensive linemen ("none": only the center lettered, as in the other run chapters).
    `puller` letters one more lineman "G" (the guard the figure is about)."""
    out = []
    for p in offense("singleback"):
        if p.key in OL_KEYS:
            lab = p.key if labels == "full" else ("C" if p.key == "C" else ("G" if p.key == puller else ""))
            out.append(p.moved(label=lab))
    return out


def te(side=1, key=None, d=-0.7):
    key = key or ("Y" if side > 0 else "U")
    return O(key, "TE", d, side * 4.8, key)


def i_form(wrs=True, puller=None):
    """I-formation, 21 personnel, tight end (Y) on the right; X on the line left, Z off the ball right."""
    out = line(puller=puller) + [O("QB", "QB", -1.9, 0, "Q"), O("FB", "FB", -4.9, 0, "F"), O("RB", "RB", -7.4, 0, "R"), te(+1)]
    if wrs:
        out += [O("X", "WR", -0.7, -15.0, "X"), O("Z", "WR", -1.5, 15.0, "Z")]
    return out


def single_back_11(gun=False):
    """11 personnel 2x2: Y in line right, Z wide right, H in the left slot, X wide left (X on the line)."""
    qb = O("QB", "QB", -5.0 if gun else -1.9, 0, "Q")
    rb = O("RB", "RB", -5.0, -1.6, "R") if gun else O("RB", "RB", -7.0, 0, "R")
    return line() + [qb, rb, te(+1), O("Z", "WR", -1.5, 15.0, "Z"), O("H", "WR", -1.5, -9.0, "H"),
                     O("X", "WR", -0.7, -15.0, "X")]


def over_front(side=1):
    """The 4-3 Over of Gap and Power Runs: 5, 1, 3, 5 on the line from the offense's left, the Sam walked up
    on the line outside the tight end (a wide 9; 05-02 draws the Over's Sam off the ball), the Mike over the
    strong A gap, the Will over the weak B gap."""
    s = side
    return [D("WE", "DL", 1.0, T("5", -s), "5"), D("NT", "DL", 1.0, T("1", -s), "1"),
            D("DT", "DL", 1.0, T("3", s), "3"), D("SE", "DL", 1.0, T("5", s), "5"),
            D("SAM", "SAM", 1.0, s * 7.0, "S"), D("MIKE", "MIKE", 4.6, s * 0.9, "M"),
            D("WILL", "WILL", 4.6, -s * 2.5, "W")]


def even4(sam_w=None):
    """A four-man even front (5, 1, 3, 5 from the left; 3-technique to the right) with two or three linebackers.
    The right end is a true 5 (outside shoulder of the tackle, inside the tight end)."""
    out = [D("WE", "DL", 1.0, T("5", -1), "5"), D("NT", "DL", 1.0, T("1", -1), "1"),
           D("DT", "DL", 1.0, T("3", 1), "3"), D("SE", "DL", 1.0, T("5", 1), "5"),
           D("MIKE", "MIKE", 4.6, 1.2, "M"), D("WILL", "WILL", 4.6, -2.4, "W")]
    if sam_w is not None:
        out.append(D("SAM", "SAM", 4.2, sam_w, "S"))
    return out


def go_to(p, key, end, kind="block", delay=0.0, speed=None, via=()):
    """Move `key` through absolute (d, w) points `via` to `end`, as a block, pull, path or run."""
    start = np.array(p._cur(key))
    pts = [tuple(np.array(q) - start) for q in (*via, end)]
    if kind == "block":
        return p.block(key, pts=pts, delay=delay, speed=speed)
    if kind == "pull":
        return p.pull(key, pts, delay=delay, speed=speed)
    return {"path": p.path, "run": p.run}[kind](key, pts, delay=delay, speed=speed)

def dashed(fld, p, pts, color=style.THIRD, lw=1.5, dash=(0, (3, 2.2)), end="arrow"):

def clean(ax):
    """Box-zoomed diagrams: drop hash ticks and yard numbers, clip anything outside the window."""
    for ln in list(ax.lines):
        xs, ys = ln.get_data()
        if len(xs) == 2 and ln.get_linewidth() == 0.5 and np.hypot(xs[1] - xs[0], ys[1] - ys[0]) < 1.0:
            ln.remove()
    for t in list(ax.texts):
        if t.get_text().isdigit() and t.get_alpha() == 0.55:   # yard numbers, not technique labels
            t.remove()
        else:
            t.set_clip_on(True)

MARK_R = 0.62


def tracking(play):
    df = play.to_tracking()
    df["_id"] = df["displayName"].astype(str)
    return df


def panel(ax, play, df, t, lateral, window, rings=(), ring_r=0.9):
    """One moment of a simulated play on `ax`: markers, trails of on-screen players, rings."""
    tmap = df.groupby("frameId")["time"].first()
    fid = int((tmap - t).abs().idxmin())
    ylo, yhi = sorted((play.ball_y - lateral[0], play.ball_y - lateral[1]))
    fld = Field(ax=ax, orient="vertical", los=play.los, window=window, numbers=False, hashes=False,
                lateral=(ylo, yhi), to_go=play.to_go)
    cur = df[df["frameId"] == fid]
    snap = df[df["frameId"] == int((tmap - 0.0).abs().idxmin())]
    first = snap[snap["x"].between(play.los + window[0] + 0.8, play.los + window[1] - 0.8)
                 & snap["y"].between(ylo + 0.8, yhi - 0.8)]       # players whose marker fits at the snap
    ids = set(first["_id"]) | {"football"}
    shown = cur[cur["_id"].isin(ids) | (cur["side"] == "ball")]
    trails = df[(df["frameId"] <= fid) & df["_id"].isin(ids) & (df["side"] != "ball")]
    for (_, side), g in trails.groupby(["_id", "side"]):
        xs, ys = fld.xy(g["x"].values, g["y"].values)
        ax.plot(xs, ys, color=style.OFFENSE if side == "off" else style.DEFENSE, lw=0.9, alpha=0.4, zorder=3)
    ppy = fld.pts_per_yard()
    for _, r in shown.iterrows():
        x, y = (float(v) for v in fld.xy(r["x"], r["y"]))
        if r["side"] == "ball":
            ax.add_patch(Ellipse((x, y + 0.45), 0.7, 0.42, fc=style.BALL, ec="white", lw=0.5, zorder=9))
            continue
        lab = str(r.get("label", "") or "")
        if r["side"] == "off":
            ax.add_patch(Circle((x, y), MARK_R, fc=style.OFFENSE, ec="white", lw=0.6, zorder=8))
            tc = "white"
        else:
            ax.add_patch(RegularPolygon((x, y), 4, radius=MARK_R * 1.3, orientation=np.pi / 4,
                                        fc=style.DEFENSE_TINT, ec=style.DEFENSE, lw=1.0, zorder=8))
            tc = style.INK
        fs = max(4.0, min(8.0, ppy * (0.85 if len(lab) <= 1 else 0.62)))
        ax.text(x, y, lab, ha="center", va="center", fontsize=fs, color=tc, fontweight="bold", zorder=10)
    for key, col in rings:
        r = cur[cur["displayName"] == key].iloc[0]
        x, y = fld.xy(r["x"], r["y"])
        ax.add_patch(Circle((float(x), float(y)), ring_r, fc="none", ec=col, lw=1.6, zorder=7))
    for patch in ax.patches:
        patch.set_clip_on(True)
    return fld


def strip(moments, lateral, window, notes=None, rings=(), ncols=2, width_in=5.2, after=None):
    """Numbered frame strip. `moments` is a list of (play, t, tag): several plays may share one strip,
    which is how two versions of the same play (same start, two endings) are shown in print."""
    cache = {}
    nrows = int(np.ceil(len(moments) / ncols))
    pw = width_in / ncols
    span_w, span_d = lateral[1] - lateral[0], window[1] - window[0]
    fig, axes = plt.subplots(nrows, ncols, squeeze=False,
                             figsize=(width_in, nrows * (pw * span_d / span_w + (0.62 if notes else 0.25))))
    for ax in axes.flat[len(moments):]:
        ax.axis("off")
    for i, (ax, (play, t, tag)) in enumerate(zip(axes.flat, moments)):
        if id(play) not in cache:
            cache[id(play)] = tracking(play)
        fld = panel(ax, play, cache[id(play)], t, lateral, window, rings)
        if after:
            after(fld, play, t, i)
        lab = "snap" if abs(t) < 1e-6 else f"{t:+.1f} s"
        ax.set_title(f"{i + 1}.  {lab}" + (f"   {tag}" if tag else ""), fontsize=8.3, color=style.INK_2, loc="left")
        if notes:
            ax.text(0.5, -0.03, notes[i], transform=ax.transAxes, ha="center", va="top",
                    fontsize=7.1, color=style.INK)
    fig.tight_layout(h_pad=1.3 if notes else 0.6, w_pad=0.6)
    return fig


def video(segments, lateral, window, rings=(), width_in=4.6, after=None):
    """Web version: each (play, tag, [(t, note), ...]) segment played in turn, notes as captions."""
    import shutil
    from matplotlib import animation
    from IPython.display import HTML, display
    span_w, span_d = lateral[1] - lateral[0], window[1] - window[0]
    fig, ax = plt.subplots(figsize=(width_in, width_in * span_d / span_w + 0.45))
    frames = []
    for play, tag, notes in segments:
        df = tracking(play)
        tmap = df.groupby("frameId")["time"].first()
        frames += [(play, df, float(tmap[f]), tag, notes) for f in sorted(df["frameId"].unique())]

    def update(k):
        play, df, t, tag, notes = frames[k]
        ax.clear()
        fld = panel(ax, play, df, t, lateral, window, rings)
        if after:
            after(fld, play, t, -1)
        txt = ("snap" if abs(t) < 0.05 else f"{t:+.1f} s") + (f"  {tag}" if tag else "")
        cur = [n for tt, n in notes if tt <= t + 1e-6]
        ax.set_title(txt + (("\n" + cur[-1].replace("\n", " ")) if cur else ""), fontsize=8.5, color=style.INK,
                     loc="left")
        return []

    anim = animation.FuncAnimation(fig, update, frames=range(len(frames)), interval=100)
    html = (anim.to_html5_video(embed_limit=40).replace("<video ", '<video style="max-width:100%;height:auto" ')
            if shutil.which("ffmpeg") else anim.to_jshtml(default_mode="once"))
    plt.close(fig)
    display(HTML(html))


def animation_or_strip(moments, notes, lateral, window, rings=(), after=None, ncols=2, width_in=5.2):
    """Video on the website; a numbered, captioned frame strip in the PDF."""
    if pdf_build():
        strip(moments, lateral, window, notes=notes, rings=rings, ncols=ncols, width_in=width_in, after=after)
        plt.show()
        return
    segs = []
    for play, t, tag in moments:
        if not segs or segs[-1][0] is not play:
            segs.append((play, tag, []))
    for (play, t, tag), n in zip(moments, notes):
        for s in segs:
            if s[0] is play:
                s[2].append((t, n))
    video(segs, lateral, window, rings=rings, after=after)

First-and-10, early in the third quarter. The offense runs power to the right: a guard pulls from the left side, the fullback races toward the outermost defender on the line on the right, and for a split second there is a hole outside the tight end that looks big enough to drive a truck through. Then the end man does something that looks like a mistake. Instead of standing up to the fullback, that end man throws himself underneath him, toward the inside, and the two of them go down in a heap right where the hole was supposed to be. The pulling guard has nowhere to go. The running back bounces outside, finds green grass for a stride, and is met by a safety who was twelve yards deep when the ball was snapped. Gain of two. The analyst says, “Great fit by the defense.” Nobody on the defense made a spectacular play. Nobody beat a block. So what, exactly, was great? This chapter is the answer: a run defense is not eleven men chasing the ball, but eleven assignments that add up, and the end who “lost” that collision was the most important part of the sum.

You’ll be able to:

  • Account for every gap against a formation and name the defender who owns each one, including the gap a safety brings with him when he comes down.
  • Find the force defender on each side before the snap, and say from the coverage shell whether he is likely to be an end, a linebacker, a safety or a cornerback.
  • Tell spill from box (squeeze) when a pulling guard or fullback arrives, and say where the ball should go next and who must be waiting there.
  • Do the box math, including the quarterback who runs and the safety who rotates down at the snap, and say who the free defender is.
  • Explain how a defense fits zone, power and the zone read, why the gap exchange exists, and what each answer costs.
  • Read a chart of rushing EPA by box count, and explain why a defense that keeps two safeties deep and invites the run is making a reasonable bet.
NoteYou’ll need

  • Option Football: the read key (the defender the quarterback reads instead of blocking), the zone read, and the scrape exchange, which this chapter rebuilds from the defense’s side.
  • Also useful: Gap and Power Runs (power, the kick-out and the puller); A First Look at Defense (the box, box count, light and loaded boxes, one-high and two-high shells); and Defensive Line Play (how a lineman fights a down block, setting the edge and contain).

How to read the diagrams. Offense is blue circles, defense orange squares, and left and right always mean the offense’s left and right. On offense, C is the center (the other four linemen are unlettered), Q the quarterback, R the running back, F the fullback, Y the tight end, and X, Z and H receivers. Defensive linemen are labelled by their technique number (5, 3, 1), S, M and W are the Sam, Mike and Will linebackers, $ is the nickel back, CB a cornerback, and FS and SS the free and strong safeties. Blue lines ending in a short bar are blocks; orange arrows are where defenders go; the thick dark-blue arrow is the ball carrier. In the frame strips, faint lines are where each player has been. Rings keep one meaning all chapter: purple is the force (or the plus-one), green the alley player or another free fitter, orange the end man (the spill or box player) or another defender whose technique the figure is about, and gray the offensive player to watch, usually the ball carrier. Where the Sam (S) stands on the line beside a 5-technique (5), he wears the orange ring, so the two never get confused.

What a run fit is

Start with the offense, because run defense only makes sense as an answer to it. Every run play in Part 3 is built around one of two ideas. Either it creates a crease the defense did not expect (zone’s moving wall and cutback), or it creates an extra blocker at one spot (power’s pulling guard, the fullback on an iso, the quarterback’s read on the option). Both ideas attack the same weakness: a defender who is not where his teammates think he is.

So the defense’s first answer is not a technique or a tackle. It is a plan. A run fit is the defense’s plan for which defender takes which gap or job against a run, and also the act of carrying it out: each defender getting to his gap after the snap, on the correct side of whoever is blocking him. Coaches use the word both ways. “Our fits were bad” means people went to the wrong places; “he fit it perfectly” means one player arrived in his gap at the right moment, on the right shoulder.

The principle underneath every fit has a name of its own: gap integrity. Every gap the offense can run through has exactly one defender responsible for it, and each defender stays in his gap until the ball shows him where it is going. “Exactly one” cuts both ways. A gap with nobody in it is an obvious problem: that is where the long run goes. A gap with two defenders in it is the same problem in disguise, because the defense only has so many players. If two defenders are in the B gap, some other gap is empty, and a good running back will find it.

You met the counting in Defensive Line Play: against a formation with one tight end there are seven gaps along the line (two A gaps either side of the center, two B gaps, two C gaps, and a D gap outside the tight end), and a one-gap defense puts one defender in charge of each. That count was the static version. Run fits are what happens when the count starts moving, and it moves for three reasons:

  1. Blockers carry gaps with them. A gap is not a patch of grass; it is the space between two blockers. When the tight end blocks down on the man inside him, the C gap slides inward with him, and a new space opens outside.
  2. Backs add gaps. A fullback who runs to the line, or a guard who pulls, is one more blocker at one spot, and the space beside him is one more gap. Coaches say every back in the backfield can add a gap. Power’s whole point is to create a gap at the point of attack that the defense did not have a man for before the snap.
  3. The defense moves too. Linemen slant, linebackers blitz, safeties rotate. Each of those moves takes a defender out of one gap and puts him in another, so somebody else has to inherit the gap he left.

A fit is the defense’s way of keeping the count right while all of that happens. Before the snap, each defender knows his gap. After it, each knows how his gap changes when he sees a particular block: if the tackle in front of me blocks down, my gap is outside; if the guard pulls, the linebacker behind me fast-flows and I inherit his. The names you will learn in this chapter (force, spill, box, alley) are the vocabulary of those changes.

ImportantWhy it exists

A defense has eleven players and the offense knows where the ball is going; the defense does not. Run fits exist so that the defense does not need to know. If every gap has an owner and every owner knows how his gap moves when the blocking changes, the ball can go anywhere and somebody is waiting. The price is discipline. A fit only works if all eleven players do their job instead of chasing the ball, which means trusting the man next to you. The most talented player on the field can wreck a defense by “making a play” outside his gap and leaving it empty.

NoteMadden vs. real life

In Madden, the move every user learns is to grab the safety, wait for the handoff and sprint at the ball carrier. It works because the computer plays the other ten defenders’ gaps perfectly. In real football, a player who leaves his gap to make a play is the most common reason for a long run, and coaches grade it as a missed assignment even when he makes the tackle. When you see a 60-yard run on a broadcast, rewind to the snap and look for the defender who ran to the wrong side of a block. It is rarely the man who missed the tackle at the end.

Force: who turns the ball back inside

Every run has an inside and an outside, and the outside is where the defense has fewest bodies. So before anyone talks about gaps, the defense names one player on each side of the ball whose job is the outside edge. He is the force defender, or “the force”: the defender responsible for keeping the ball carrier from getting outside him on a run to his side and for turning (forcing) the ball back inside, toward the teammates who are chasing it. He does not have to make the tackle. He has to make the runner change direction, back into the crowd.

You have met two cousins of this idea. In Defensive Line Play, to set the edge was a technique: the outermost defender on the line takes on a blocker with his outside arm free so the run can’t get outside him. Contain was a rule about the quarterback escaping the pocket or running a bootleg. Force is a role. It belongs to whichever defender the call assigns, and the call changes from play to play. Sometimes the force is the defensive end, setting the edge exactly as 05-01 drew it. Often it is an outside linebacker, a nickel back or a safety a few yards outside the end. In some coverages it is a cornerback.

Why would a cornerback be the force? Because the coverage decides who is free. A defender who has a deep zone to protect cannot also come up to the line to turn a run inside; a defender with a short zone, or with no zone at all, can. That is why the first question to ask about a run defense is not “who’s in the box?” but “what are the safeties and corners doing?” Two common answers, which you will see drawn in full in the coverage chapters:

  • Safety force (Cover 3, “sky”). Cover 3 splits the deep field into three zones, the deep thirds: one for each corner and the middle one for the free safety. The other safety is spare, so he comes down to the strong side, outside the tight end, and becomes the force. The corners are deep, so they cannot be. This is the eighth man in the box. Coaches call it a sky rotation: s for the safety who comes down.
  • Corner force (Cover 2, “cloud”). Cover 2 keeps both safeties deep, each guarding a deep half of the field. The corners have no deep zone, so they sit at about five yards, a step outside the widest receiver, and come up to force the run. This short corner is the hard corner of 06-04, and coaches call his side a cloud side: c for the corner who stays short.

The words describe who comes down, not the coverage number, so don’t equate cloud with Cover 2. A Cover 3 defense can also “cloud” one side (the corner stays short and a safety runs to the deep third behind him), and in a buzz rotation the Cover 3 safety comes down inside, as the alley player, and the outside linebacker or nickel back already standing out there is the force. Cover 3 draws all three rotations.

Who stands just inside the force is the next question, and it is the subject of the next section, but you can see both answers at once in Figure 32.1.

The force has a hard job, and the technique he uses is mostly patience. He comes up outside-in: he approaches the ball from outside its path, keeping his inside shoulder toward the runner and his outside arm free, never letting the runner’s helmet get to his outside. Coaches often name two speeds of it. Hard force is fast and tight: the squatting Cover 2 corner attacks the line and meets the ball close to it. Soft force is slower and deeper: a safety coming from a deep zone keeps his outside leverage and gives a little ground, accepting a few yards to be sure nothing gets outside him. Either way the failures are the same. Come up too fast and too far inside, and a back with any speed simply runs around him. Come up too slowly, and the back turns upfield inside him for seven yards before anyone arrives. A good force player is the defense’s answer to two of the offense’s most dangerous plays: the outside run that wins the corner, and the bounce, which is a run that was designed to go inside and ends up outside. The bounce is the force’s specialty, and it is where the next idea comes in.

CautionCommon misconception

“The defensive end is always responsible for the outside.” Sometimes he is, and then he sets the edge. But in many calls the end is told to do the opposite: take on blocks inside and let somebody else have the outside. (Keep two names apart here. The defensive end is a position, the lineman at the end of the front. The end man is whoever is outermost on the line on that side; with an outside linebacker walked up beside the end, the linebacker is the end man, not the end.) Whether the end is the force or not is decided by the call and the coverage, and you can often tell before the snap. If a safety or a corner is standing five to seven yards deep just outside the tight end, the end probably isn’t the force.

Spill or box: two ways to meet a kick-out

Now go back to the play this chapter opened with. Power’s kick-out block, from Gap and Power Runs, asks a question of the end man on the line of scrimmage, the outermost defender on the line on that side. Here he is the Sam linebacker, walked up on the line outside the tight end (a wide 9; Even Fronts drew the 4-3 Over’s Sam off the ball, and walking him up is a common variation), not the 5-technique defensive end inside him. The question: a fullback is coming at you from the inside, and a pulling guard and the running back are right behind him. What do you do?

You have met both answers already, from the offense’s side in 03-03 (kick or log) and from the lineman’s side in Defensive Line Play (the wrong arm). Here is the full version, because the choice between them is a team decision, not one player’s.

Spill. The defender attacks the blocker’s inside half with a wrong arm, his outside shoulder and arm thrown across and under the blocker toward the hole. He is not trying to win. He is trying to make a pile in the hole: his body, the fullback’s body, and soon the pulling guard, who arrives to find his lane full. The running back, who was following that guard, has nowhere to go inside, so the ball spills outside, sideways along the line.

Box technique, also called squeeze. The defender keeps his outside arm free, meets the blocker with his inside shoulder, and drives him back inside, so the hole gets smaller and smaller. He is “boxing” the ball in: it cannot get outside him, so it has to stay inside, where the linebackers are filling. (Don’t confuse box technique with the box, the area near the ball where defenders are counted. The words are unrelated; football just reuses them.)1 Coaches also say squeeze for the thing an end does a moment earlier, closing down the line behind a tackle’s or tight end’s down block (05-01). The two usually come in sequence: squeeze the down block, then box or spill the kick-out.

The two techniques send the ball to opposite places, and that is the whole point of choosing between them. A defender who spills is promising that someone is waiting outside. A defender who boxes is promising that someone is waiting inside. In Figure 32.2 the same power play meets both.

Why choose one or the other?

Both techniques work. Both fail. The difference is how they fail, and a defense picks the failure it can live with.

Spill sends the ball to speed. A run that bounces sideways is running at the slowest angle in football: it gains no yards while it moves, and every defender on the field can chase it. Defenses with fast linebackers and safeties love to spill, because they would rather tackle a back in space with three pursuers than in a crowded hole with one. Spill also attacks the heart of a gap scheme. Power and counter depend on the puller turning up into a clean hole; take the hole away and the play’s best blocker becomes a spectator. The risk is the outside. A spilled ball is going exactly where the defense has fewest bodies, and if the force is late, blocked or out of position, the back is around the corner with nobody between him and the end zone.

Box keeps the ball in the crowd. Boxing keeps everything inside, where the big bodies are, and it is simpler for the defenders outside: the force can stay outside and wait, because nothing is supposed to get to him. The risk is inside. A boxing end who gets driven off the ball (the kick-out does its job) leaves a hole wider than the one the play designed, and if the linebacker filling it is a step late or gets caught by the puller, the back runs through the middle of the defense untouched.

The offense has an answer to each. Spill is exactly the picture Gap and Power Runs called log: when the end man gets inside the kick-out, the puller stops trying to kick him out, takes his outside shoulder instead and seals him in with a log block, and the back bounces outside the log. So a spill call is a dare. The defense is inviting the offense to bounce the ball, because it has a man waiting outside, and the whole bet rides on the force. If he is there, the logged bounce runs into him and gets turned back toward the pursuit. If he is late or blocked, the log has opened the edge. Against box, the offense’s answer is simply the play as drawn: the kick-out block is the right block, and the back cuts up inside it, so the defense is betting on the linebacker who fills that hole.

A common way to organize all this, though staffs differ, is a pattern coaches sometimes call spill inside, force outside: the players closest to the ball, linemen and inside linebackers, spill anything coming at them, and the outermost defender on each side keeps everything inside him, so that every spilled ball runs into a man who stops it from getting further out. In the top row of Figure 32.2, that is exactly the shape: the Sam spills, and the strong safety, the force, keeps the ball inside him. The ball gets squeezed between them. Many spill defenses spill at two levels: the end man spills the fullback, and the play-side linebacker spills the puller too, attacking his inside half instead of running around the pile; then the play-side end, fighting back across the tight end’s down block, owns the gap inside the pile. Other defenses ask their end man to box and their linebackers to spill the puller, and some change the call by formation or by opponent. What matters for you watching on Sunday is the promise each technique makes. When the end man wrong-arms a blocker, look outside for the man who has to be there.

NoteMadden vs. real life

In Madden you pick the defense and every player simply knows his job. In real football, spill or box, and who is the force, travel with the call. The defensive coordinator’s call reaches one player through the radio in his helmet, the green dot of What a Play Really Is, usually a linebacker (see Linebackers), who repeats it to the other ten. Many defenses build the run fit into the call itself, so that a coverage word also says which safety comes down and whether the edge spills or boxes. Then, at the line, the Mike and the safeties adjust it to the formation they see, shouting a strength call (“Strong right!”, from Even Fronts), a “crack!” warning when a receiver cuts his split, or a check that changes the spill player, until the snap. The fit that eleven players carry out is finished at the line, in the last seconds before the snap, by the players who can see the offense.

TipWatch for it

The tell is the end man’s first contact. If he dives under the kick-out block toward the ball, it is a spill, and the run should bounce: watch for the force. If he stays square, outside arm free, and takes the block on his inside shoulder, it is a box, and the run should stay inside: watch the linebackers fill. When a broadcast shows a power run from the end-zone camera, the pile in the hole after a spill is impossible to miss.

The alley player

Spill a ball outside and it does not go straight to the force. It goes first into the space between the box and the force, a lane a few yards wide just outside the last defender on the line. Coaches call that lane the alley, and the defender responsible for it is the alley player: the man who fills the gap between the force player and the box, meeting a run that has bounced outside the box but not outside the force.

Look back at Figure 32.1. In the Cover 3 picture the strong safety is the force and the free safety, coming down from the deep middle, is the alley player; in some calls it is the Mike linebacker, running around the outside edge of the pile. In the Cover 2 picture the cornerback is the force and the safety on that side is the alley player, coming down from his deep half. In quarters (Quarters and Match), a fast-growing two-high coverage,2 the outside linebacker or nickel back is usually the force and the safety is the alley player. Names and exact assignments differ by staff.3 The geometry does not. Every sound fit has, on each side of the ball, an outside man who forces and a man inside him who fills the alley.

The two come at the ball from opposite sides, and that is the point. The force comes outside-in. The alley player comes inside-out: he approaches from inside the ball’s path, a step behind it, keeping the runner on his outside shoulder, the same leverage a linebacker keeps when he flows (Linebackers). A runner caught between them can’t cut back inside the alley player and can’t get outside the force, so the only way out is into the narrowing lane between two tacklers.

Why does the alley player usually come from the secondary? Because he has to come from outside the blocks. The linebackers start in the box, where blockers can reach them: a pulling guard leads up on the Mike, a tight end climbs to the Will. A safety eight or twelve yards deep starts where no lineman can get to him before he reads the play.

How does he know it’s a run, and in time? He reads a key, just as a linebacker does, usually the nearest tight end or offensive tackle. Coaches call it high hat, low hat. A blocker who fires out low and forward, or a guard who pulls, is a low hat: run, so come down. A blocker who pops up tall and sets back to protect is a high hat: pass, so play the deep zone.4 The read takes a fraction of a second, and it is exactly what the offense attacks. Play-action is built to show low hats and a handoff fake, pull the alley safety forward a few steps, and send a receiver into the deep zone he just left. That is what the alley player costs: a safety filling the alley is not covering the zone he started in.

This is a typical way to fit the play, not one team’s playbook: defenses vary the spill and box calls, who flows where, and which safety comes down.

Three things are worth noticing.

Everybody has a job, including the players far from the ball. The left cornerback is fifteen yards from the play and still has an assignment, and if he leaves it to chase the run, a play-action pass to the X receiver is a touchdown. Fits are eleven-man plans, which is why defensive coaches say a run defense is only as good as its worst fit.

The defense counts, and so does the offense. Power brought seven blockers to the party (five linemen, the tight end and the fullback). The defense brought eight defenders into and around the box (four linemen, three linebackers and the strong safety), plus a free safety who can arrive late. Count who is left unblocked, though, and you find two: the backside end and the strong safety. That is not a contradiction. The offense chooses which defender to leave free, and power, as Gap and Power Runs showed, leaves the one farthest from the hole: the backside end, who has to chase down the line behind a wall of bodies, while the left tackle’s hinge only closes the gap the pulling guard left. So the free defender who matters is the one at the point of attack, and that is where the defense spends its eighth man: as the force, outside the play, exactly where a spilled ball ends up. The plus-one, below, gives that man a name. The fit chart and the box count are the same idea seen from two sides.

Gaps move, and the fits move with them. Before the snap, the Mike was over the right A gap. After the snap, the right A and B gaps have become one crowded space: the 3-technique, the right guard and right tackle doubling him, and the center blocking back next to them. The 3 holds it alone, and the Mike’s job has moved outside the pile, four gaps over. He did not freelance. He read his key, saw power, and went where the plan said power’s gap would appear, and the Will, who shuffles behind the double team, inherits the gap the Mike left if the ball cuts back into it. Every linebacker’s read in Linebackers and the Second Level is a question about where his gap has gone.

Fits against zone: when the gaps keep moving

Power tells the defense where it is going. A pulling guard is a signpost, and a defense that reads it can fit the gap the play creates. Zone does the opposite. As Zone Running showed, the whole line steps the same way and the running back picks the hole after the snap, so there is no single gap to fit. The gaps slide sideways with the blockers, and the back is reading the defenders to see which one is out of place.

Against zone, then, a fit is less a spot than a relationship: stay on the correct side of your blocker, and stay a gap behind the ball. Figure 32.4 draws it against outside zone.

  • The play-side linebacker runs, but behind the ball. Fast flow, but never ahead of the running back’s helmet. A linebacker who over-runs the play (gets outside the ball) has given the back a lane behind him, and zone is designed to find exactly that lane.
  • The backside owns the cutback. This is where zone hurts defenses. Outside zone’s biggest runs are cutbacks, so the backside linebacker slow-flows (shuffles, waiting) and owns the cutback lane, and the backside safety comes down behind him, the last defender in it. The backside end squeezes down the line, chasing from behind, but with one eye on the quarterback: outside zone’s best friend is the bootleg the other way.

Zone also explains the most common failure of gap integrity on a broadcast. When a zone run gets eight yards on a cutback, rewind and look at the backside linebacker. Usually he flowed with everyone else, two defenders ended up in the same play-side gap, and the backside gap he owned was empty. The back didn’t make the hole. The defense gave it to him.

The plus-one: where the extra defender comes from

So far this chapter has assumed something it hasn’t said out loud: the defense has enough players to own every gap and still have someone left over. Where does that someone come from?

The Language of the Run Game introduced hat math: count the offense’s blockers, count the box, and compare. A First Look at Defense turned it into a scale (light, normal, loaded). The defensive side of that arithmetic has a name. The plus-one is having one more run defender than the offense has blockers, so one defender is unblocked and free to make the tackle. With six blockers, it means seven run defenders; with seven, eight.

Why one more? Because if the numbers are even, every defender can be blocked. A hat on every hat means the runner reaches the second level untouched unless some defender beats his blocker, and the defense would be betting its whole run defense on winning one-on-one fights. With a plus-one, the defense doesn’t need to win a single block. It needs every defender to hold his gap, and the extra man makes the tackle. Look at the fit chart again: the strong safety is the plus-one, and his job, force, is the one nobody on the offense was assigned to block.

The plus-one almost always comes from the secondary, because that is where the spare players are. The defense starts with eleven; the corners are usually busy with the wide receivers, and one deep safety stays back to prevent the long pass. Everyone else can be a run defender. So the cleanest way to add a defender to the box is to bring a safety down, and the cleverest way is to bring him down at the snap, after the quarterback has looked at the defense.

The offense’s answers, and the defense’s answers to those

The plus-one is not a fixed fact. It is the opening position in a negotiation, and each side has a short list of moves.

The offense can add a blocker. A second tight end or a fullback makes seven blockers, so the defense needs eight to stay plus-one. That is why heavy personnel is a run-game tool, and why it came back: in 2025 NFL offenses used fewer than three wide receivers on 41.7% of plays, the first season at or above 38% since Next Gen Stats began tracking in 2016.5

The offense can spread the defense out. Every receiver split wide pulls a defender with him. Four wide receivers leave the defense with fewer players to put in the box at all.

The offense can read the free man instead of blocking him. This is the option idea from Option Football, and it is the plus-one’s natural enemy. If the quarterback reads the unblocked defender and keeps the ball whenever that defender chases the back, the offense has “blocked” him without spending a blocker, and the count is even again. So the rule of box math with a running quarterback is simple: count the quarterback as a blocker. Six blockers plus a quarterback who reads is seven. Designed quarterback runs get there from the other direction. On QB power or QB counter the quarterback carries the ball and the running back, no longer the runner, becomes one more lead blocker: the same plus-one for the offense.

The offense can throw. Every safety in the box is a safety not covering a deep zone. The run-pass option (RPOs) and play-action both turn the plus-one into a target.

And the defense’s answers? It can bring yet another safety, and play with no deep safety at all, a gamble you will meet in Blitz and Man Pressure. It can change the technique instead of the numbers: put five men on the line, with a nose who two-gaps the center, so that the offensive line can’t reach the linebackers and they stay free to fit. That is the tite front of Odd and Hybrid Fronts. Or it can accept the light box and fit the run from depth, which is the next section, and the biggest strategic argument in the NFL right now.

Fitting the option: the gap exchange

The zone read is plus-one arithmetic with a quarterback in it. Option Football showed the offense’s view: block everyone but the backside end, read him, and hand the ball to whichever runner he is not taking. The defense’s fit chart has to say who takes the running back and who takes the quarterback, and the simplest version says the end takes the quarterback (he stays home) and the linebacker behind him takes the inside gap. The offense knows that chart, so the end’s behaviour tells the quarterback what to do.

The defense’s answer is to change the chart after the snap. A gap exchange is a planned swap of gap responsibilities between two defenders after the snap: one takes the other’s gap and vice versa. The best-known version is the scrape exchange from 03-04: the end crashes inside at the running back, which is the “keep” read, and the linebacker behind him scrapes around the end’s outside to take the quarterback. You have seen it twice already, from the quarterback’s side in Option Football and from the Will’s in Linebackers. What is new here is the ownership. The quarterback did everything right. He read the end, the end took the back, and he kept the ball. The defense simply moved the man who owned the quarterback, and it did it after the quarterback’s eyes had stopped looking for him. Notice the gap the Will left behind: it is not empty, because the crashing end is now standing in it. That is the whole trick of an exchange: every gap still has exactly one owner, only the names have changed.

The exchange still has costs, and offenses aim at them:

  • Arc. A tight end or H-back runs around the read end to block the scraping linebacker (the arc block of 03-02). Now the scraper is blocked, the end is inside, and nobody has the quarterback.
  • Bend. An end who crashes too flat and too deep goes past the back, who bends behind him into the gap the Will just left. The crash has to arrive under control, on the back’s near hip.
  • Read the scraper. The offense can block the end and read the linebacker instead, or flip the roles with the inverted veer or bash of 03-04, so that the exchange points the wrong way.
  • Depth. The scraper has to get from about four and a half yards deep to the quarterback’s path before the quarterback gets there. A linebacker lined up deeper or wider than usual, to give himself time, is a tell the quarterback can see before the snap.

A gap exchange is a bet that the offense will read the first picture.

Gap exchanges are not only for the option. A defensive line slant (05-01) is a gap exchange for a whole front: every lineman moves over one gap and the linebackers move the other way to fill what they left. An end who “spikes” (slants hard to the gap inside him at the snap) while the outside linebacker takes his C gap is one; so is a linebacker who blitzes through the A gap while the nose loops into the B gap. The idea is always the same: keep gap integrity (every gap still has exactly one owner) while changing which defender owns it, so the blocking scheme and the quarterback’s read, both built on the pre-snap picture, are wrong. That is also the logic of the simulated pressures of Part 7, applied to the pass game.

Against a team built on the option, defenses also rewrite the chart in option language. Instead of gaps, a coach assigns dive (the running back on the inside run), quarterback and, if there is one, pitch, and he makes sure each has an owner on every play, with the assignments rotating between players from snap to snap so the quarterback never sees the same picture twice. That is the logic you will see from the defensive side of The Ravens’ Option Run Game.

Two-high and the light box: inviting the run on purpose

Everything so far says the defense wants a plus-one in the box. For most of football history, that was the end of the discussion. Then, starting in the late 2010s, more and more NFL defenses began doing something that looked backward: keeping both safeties deep, playing six (sometimes five) defenders in the box against six blockers, and daring the offense to run.

In Figure 32.7 the box is even: six defenders, six blockers, a hat on every hat. Nobody in the box is free. The plus-one still exists, but it is twelve yards away: the two safeties, who read run and come downhill as alley players. They are the defense’s free men, and they arrive late, somewhere around four or five yards downfield.

Why four or five? Rough arithmetic, not a measurement, makes the number believable. Give the safety half a second to read his key and be sure it is a run. Then he runs downhill, speeding up, at an average of about six yards a second: by two seconds after the snap he has covered about nine yards on a diagonal toward the ball, which brings him from twelve yards deep to about five. The running back takes the handoff about seven yards behind the line and crosses it about 1.2 seconds after the snap, running about six yards a second, so by two seconds he is about five yards past the line. Their paths cross four or five yards downfield. A safety who reads it a tenth of a second faster, or starts at ten yards instead of twelve, saves a yard; one who hesitates gives one away. That is the price the defense has agreed to pay on every run, in exchange for something it values more.

Who is the force in a light box? Apply the rule from the force section to each side. On the nickel’s side the force is the $, the overhang defender, who is outside the box precisely so he can be both the edge against the run and a cover man on the slot. On the tight end’s side there is no one outside the end before the snap, so either the end sets the edge himself or, as in the figure, a safety comes down outside-in and the end plays inside him. Either way, the force on that side is arriving late, which is one more reason runs aimed at the tight end’s edge do well against two-high.

Why a defense would do this

Explosive passes cost more than runs. In EPA terms (the primer is in Downs, Distance and Field Position), the average pass is worth more than the average run, and the plays that lose games for a defense are explosive passes. Two deep safeties are the best protection against them. A defense that keeps two high is choosing to give up four-yard runs rather than forty-yard passes.

Two-high shells can disguise. A defense that shows two safeties can still rotate into one-high at the snap (the plus-one of Figure 32.5) or play any of the two-high coverages behind it. The light box keeps the offense guessing about who will be in it a second after the snap.

It has been spreading. Next Gen Stats measured split-safety (two-high) coverage on 32.9% of plays in 2018, 40.4% in 2024 and 42.0% in 2025, the highest since its tracking began.6 The box counts moved with it: in the nflverse data, designed runs met six or fewer defenders in the box about 38% of the time in 2018–2022, and 58% and 55% of the time in 2024 and 2025.7

The coach most associated with the approach is Vic Fangio, an NFL assistant for more than thirty years before Denver made him a first-time head coach at 60 (2019–2021). His two-high defenses in Chicago and Denver spread across the league, carried first by coaches who had worked for him, such as Brandon Staley.8 His approach is simple to state: keep two safeties deep to take away the explosive pass, and ask the front to hold the run without help. His linemen are asked to win their own fights and, at times, to “steal” back gaps the box does not have bodies for, as a Philadelphia beat writer put it in 2024.9 How the shift to two-high started (this book calls it the two-high counterrevolution), and the answer to it, are history for The Modern Era and a case study, Fangio’s Two-High Revolution and Macdonald’s Answer.

What it costs

The light box is a trade, and the bill comes in four parts.

  • Yards. Runs against six-man boxes average more than runs against seven or eight: on early downs in the data below, about a quarter of a yard more per carry than against seven and more than half a yard more than against eight or more. A defense that lives in light boxes will look bad in rushing yards allowed, and it has accepted that.
  • The front must be good enough to hold. A six-man box has no margin. If one lineman is moved off his gap, there is no spare linebacker behind him; the next defender is a safety at twelve yards. Light-box defenses need linemen who can take on double teams and two-gap when they have to, which is why they pay for big interior players and why the tite front of Odd and Hybrid Fronts became their favourite shape.
  • The safeties must tackle. A safety fitting from depth meets the runner at full speed, in space, often one-on-one. A missed tackle there is not a four-yard run. It is a forty-yard run, because there is nobody behind him.
  • Heavy personnel attacks it. Against a light box, the offense’s simplest move is to add blockers, and the rise of two- and three-tight-end sets in 2024 and 2025 is partly that. Each extra blocker forces the defense to choose again: match it with a heavier package (base defense was on the field for 29.7% of snaps in 2025, up from a low of 20.5% in 2023) or accept being outnumbered in the box.10
CautionCommon misconception

“A light box means the defense is soft against the run.” The best run defense of 2024 by EPA was one of the lightest boxes in the league: Fangio’s Eagles (see the film room below). A light box is not the absence of a run fit. It is a fit that uses the safeties instead of an eighth man at the line, and that asks more of the defensive line.

Read the left panel first. The more defenders in the box, the worse runs do: on these downs, designed runs averaged about −0.03 EPA against five or fewer in the box, −0.05 against six, −0.09 against seven and −0.15 against eight or more, and their success rates fell from about 41% to 34% across the same range.11 Both data sources agree on the slope. Now the right panel, which is the part that explains the two-high era. The same extra defender helps the passing game. Dropbacks against five or fewer in the box averaged about +0.06 EPA; against seven, about +0.12; against eight or more, about +0.16 in the 2018–2022 data. The box is a budget: a defender spent near the line is a defender not spent in coverage, and the offense’s passing game cashes the difference. (The 2023–2025 point for eight or more dropbacks rests on only about 800 plays, which is why its interval is so wide.)

And look at the two panels side by side. Against every box count, the average dropback beats the average designed run. Even against the lightest boxes, runs averaged less than zero EPA; even against the heaviest, dropbacks averaged more than zero. The slopes are about the same size, too: an extra defender in the box takes roughly as much from the run as it gives to the pass. So, on averages alone, the light box is close to a wash. Why call it a reasonable bet, then? Three numbers from the same data tip it.

  • Where the big plays come from. Against six in the box, 8% of dropbacks gained 20 yards or more; against seven, 11%. The share of designed runs gaining 10 or more barely moved: 12% against six, 12% against seven.12 Taking a safety out of the deep field buys the offense explosive passes; putting him back buys the run game almost nothing big. A defense that fears the long ball more than the four-yard run, as Fangio’s does, is reading this line.
  • The run it gives up is still a losing play. At −0.05 EPA, a designed run against a six-man box still leaves the offense slightly worse off, on average, than before the snap. Conceding it costs less than it looks on a rushing-yards table.
  • Don’t expect the offense to be “steered.” You might think a light box lures offenses into running more. In these situations the raw numbers say the opposite: offenses ran on 43% of plays against six in the box and 55% against seven (same footnote). That is the defense’s choice showing through, not the offense’s: defenses lighten the box against spread formations that usually throw and load it against heavy ones that usually run. The bet is not that the offense will take the bait. It is that whatever the offense does, the defense has spent its players where the bigger danger is.

If dropbacks beat runs at every box count, why does anyone run? Because the averages only hold at the mix offenses actually use. Runs keep defenses from dropping everyone into coverage; an offense that stopped running would soon see those same dropbacks against eight defenders in coverage, and the gap would close. And this chart leaves out the situations where running is clearly right: short yardage, the goal line, protecting a lead late. The light box is a bet on the margin, not a claim that the run is dead.

Three cautions keep this honest.

  • The box count is measured at the snap. A safety who rotates down after the snap, like the one in Figure 32.5, is counted out of the box. Some “six-man boxes” in the data were seven-man fits a second later, which makes light boxes look a little better against the run than the fits really were.
  • Boxes are chosen, not assigned at random. Defenses load the box when they expect a run and empty it when they expect a pass, and offenses check into runs against light boxes. Holding the down and the game situation fixed, as this chart does, removes the worst of that, but not all of it. A First Look at Defense showed how flat these comparisons look when you don’t.
  • The data changes source between 2022 and 2023. Next Gen Stats (the NFL’s player-tracking data) supplied the box count through 2022, and FTN, a data company whose staff chart every play by hand, from 2023. They draw the box differently, so read the two lines as two measurements, not one trend.

The “Watch for it” rule (the outermost defender within about seven yards of the line on each side, ignoring a corner who is deep or pressed on his man, but counting a corner who squats at about five yards when both safeties are deep) is easy to turn into code. Show the Python (find the probable force player on each side at the snap) 
ROLE = {"WE": "backside end (5)", "SE": "play-side end (5)", "SAM": "Sam (S)", "SS": "strong safety",
        "FS": "free safety", "LCB": "left corner", "RCB": "right corner", "NB": "nickel ($)",
        "MIKE": "Mike", "WILL": "Will", "NT": "nose (1)", "DT": "tackle (3)"}


def force_candidates(trk, max_depth=7.0, corner_depth=(3.0, 6.0), deep=10.0):
    """Outermost defender on each side within `max_depth` yards of the line at the snap: the probable force.
    A corner counts only if he squats `corner_depth` yards off the ball AND at least two defenders are deeper
    than `deep` (the Cover 2 'cloud' corner); a deep or pressed corner is ignored. A pre-snap guess:
    rotation after the snap can change it."""
    snap = trk[trk["frameType"] == "SNAP"]
    ball = snap[snap["side"] == "ball"].iloc[0]
    los, mid = ball["x"], ball["y"]
    dfn_all = snap[snap["side"] == "def"].copy()
    dfn_all["depth"] = dfn_all["x"] - los
    two_deep = (dfn_all["depth"] > deep).sum() >= 2
    rows = []
    for side_name, sgn in (("offense's right", -1), ("offense's left", 1)):   # offense faces +x: right = -y
        dfn = dfn_all[(dfn_all["y"] - mid) * sgn > 0].copy()
        dfn["width"] = (dfn["y"] - mid) * sgn
        corner = dfn["position"] == "CB"
        squat = corner & dfn["depth"].between(*corner_depth) & two_deep
        dfn = dfn[(dfn["depth"] <= max_depth) & (~corner | squat)]
        top = dfn.sort_values("width").iloc[-1]
        rows.append({"side": side_name, "probable force": ROLE.get(top["displayName"], top["displayName"]),
                     "yards outside the ball": round(top["width"], 1), "depth": round(top["depth"], 1)})
    return pd.DataFrame(rows)

Film room

NoteFilm room: Seattle Seahawks 2012–2015 (the eighth man)

The plus-one’s most famous modern home was Pete Carroll’s Seattle defense, built on single-high Cover 3. Earl Thomas played the deep middle, alone, and Kam Chancellor, a strong safety built like a linebacker, played near the line as the eighth man in the box. Seattle led the NFL in scoring defense four years running, 2012 through 2015.13 Watch any of their games from those seasons for one thing: where Chancellor is at the snap. He is the force on runs to his side, the plus-one in the box count, and the reason offenses that wanted to run on Seattle needed an extra blocker. Then watch what happens to Thomas when the offense fakes the run: he is the only deep defender, which is the cost of the eighth man.

NoteFilm room: Philadelphia Eagles 2024 (the light box that held)

Vic Fangio’s first Eagles defense, in 2024, is the light-box argument at full strength, and it started badly. Through the first two weeks the Philadelphia Inquirer reported six or fewer defenders in the box on 73.5% of snaps, the third-highest rate in the league, and the run defense looked shaky after the Falcons ran through it in Week 2.14 Fangio kept the shape. Over the full season, by this book’s count, opponents’ designed runs met six or fewer in the box about two thirds of the time (a different measure from the Inquirer’s early snap count, and again the third-highest rate), and the Eagles allowed the fewest EPA per designed run in the league.15 Every defense Fangio has run since 2019 (he spent 2022 as a consultant) has been among the five that met runs with six or fewer in the box most often.16 The bet rested on the interior of the line: tackles Jalen Carter and Jordan Davis, whom Fangio called “a better run defender than he is a pass rusher,” and Milton Williams.17 What to watch: the two safeties at 12–14 yards at the snap, the six-man box, and how often an interior lineman ends up with two blockers on him and doesn’t move. The defense is betting on the front, and in 2024 the bet paid off on the way to a Super Bowl LIX win over Kansas City, 40–22.

NoteFilm room: Baltimore Ravens 2019 (the box that wasn’t loaded enough)

The 2019 Ravens rushed for 3,296 yards, breaking an NFL team record that had stood since 1978, and Lamar Jackson ran for 1,206 of them on his way to a unanimous MVP.18 Box math explains why defenses could not simply load up. On a designed run with Jackson reading a defender, every box needed one more defender than it appeared to, because the quarterback counts as a blocker, and on a designed quarterback run (Jackson had 120 of them), the running back could turn into one more lead blocker. Defenses met the Ravens’ designed runs with seven or more in the box about 65% of the time, slightly more than they did the rest of the league, and still lost: the Ravens averaged about +0.10 EPA per designed run against seven-man boxes and about +0.04 against eight or more, where the league averaged −0.10 and −0.13. Jackson’s own designed runs averaged about 7.1 yards.19 What looks like a normal seven-man box on the broadcast was, against Baltimore, a light one. Watch any 2019 Ravens game for the defender the quarterback is reading, and count him out of the box.

NoteFilm room: Philadelphia Eagles 2024 offense (running at two-high)

The same season, on the other side of the ball, the Eagles showed what a good run game does to a light box, with a quarterback, Jalen Hurts, whom defenses also had to account for as a runner. Saquon Barkley ran for 2,005 yards, the ninth 2,000-yard season in NFL history.20 Against Philadelphia, defenses played six or fewer in the box on about 61% of designed runs (the league rate was about 59%), and the Eagles averaged about 5.4 yards per designed run against those boxes, against the league’s 4.7. About one in seven of those runs gained ten yards or more.21 That is the light box’s bill, paid in full: when the front cannot hold six blockers and the safeties arrive at five yards instead of three, the runs that get past them go a long way. What to watch: the safeties’ depth at the snap, the double teams on the interior, and the moment Barkley reaches the second level one-on-one with a safety coming downhill.

Watch for it on Sunday

TipWatch for it

Before the snap

  • Count the box, then count the blockers. Five linemen, plus every tight end and fullback near the line. If the quarterback is a runner who reads a defender, add him. More defenders than blockers: the defense has its plus-one in the box. Even: the plus-one is a safety, and he will arrive late.
  • Find the force on each side. Look for the outermost defender within about seven yards of the line. Ignore a corner who is deep or pressed on his receiver, but count a corner squatting at about five yards when both safeties are deep. A safety walked down outside the tight end: safety force (likely Cover 3, sky). Corners squatting at five yards with both safeties deep: corner force (likely Cover 2, cloud). A nickel or linebacker over the slot: he is the force on that side. Nobody out there but the end: the end is the force. With two tight ends or an unbalanced line, do it side by side: each side has its own end man and its own force.
  • Check the safeties’ depth. At 12–14 yards, they are alley players from depth and the box is what you see. At 8–10 yards and creeping, one of them is probably coming down at the snap.

After the snap

  • Did a safety rotate down? That is the plus-one arriving.
  • Watch the end man meet the kick-out or the puller. Dives under it: spill, so look outside for the force. Stays square, outside arm free: box, so look inside for the linebacker filling.
  • Did the run bounce? If it did, who met it outside, the force or nobody? If nobody, rewind and find the empty gap.
  • On a cutback, find the backside linebacker. If he flowed with everyone else, that is the gap the runner used.

One more defender belongs in the count: the nickel back ($), the overhang defender outside the box on the left. With the back beside the quarterback on the left, the zone read’s keep goes left, straight at him, so on the keep side he can play the quarterback and the math there is really seven against seven. That is why offenses pair the read with a quick throw to the slot receiver (a bubble or a glance, the RPO idea): if the $ comes down to take the quarterback, the throw goes behind him, and the slot “blocks” him without touching him.

Expect a zone read, or a run with the quarterback as a threat, likely with that throw attached, and expect it to gain four or five yards before a safety arrives, unless the defense has a plan. Its plans: rotate a safety down at the snap (watch for one creeping toward 8–9 yards before the ball is snapped), or exchange gaps on the read man (the end crashes, a linebacker scrapes to the quarterback, Figure 32.6). If the safeties stay deep and the end sits still, this is the picture the 2019 Ravens punished whenever they got it, and it is a good one for the offense.

TipAnswer

A crack block on the force, so the cornerback must replace him. The defense’s fit against power depends on the strong safety: he is the force, the man outside the spill. A receiver in a tight split is in perfect position to block down on him at the snap, a crack block (Perimeter Runs and Misdirection), from the side, where the safety is not looking. If the defense spills the kick-out and the force has been cracked, the bounce has nobody outside it.

So the defender who must be ready is the right cornerback. His rule is crack-replace: when the receiver in front of him blocks in, he stops thinking about his deep third and comes up to become the force himself. Many defenses tie this to the split before the snap: a reduced split triggers a “crack!” call, so the corner and the safety both know the crack-replace is on, and some staffs also switch the end man from spill to box so the ball stays inside. A tight split is also the classic tell for crack-toss, where the ball goes straight to the edge the crack has cleared. Watch the corner’s first step after the snap. If it is forward and outside, the defense saw the crack coming. If it is backward, the run is going to the sideline.

Back to the replay

Return to the opening play. The offense ran power to the right; the end man threw himself under the fullback; the back bounced outside and was tackled by a safety who started twelve yards deep. Now you can name every piece of it. The end man was the spill player, and the pile he made was the point: the pulling guard had no lane, and the ball went sideways. The safety who made the tackle was either the force (if he had come down to the edge) or the alley player (if he filled the lane inside the force from depth), and he had been free all along, the defense’s plus-one, never assigned to any blocker. The analyst said “great fit” because every defender between the ball and the sideline did his job and nobody did anyone else’s. Nobody beat a block. The arithmetic did the work.

Takeaways

  • A run fit is an eleven-man plan. Every gap has exactly one owner (gap integrity), and every owner knows how his gap moves when the blocking changes. Two defenders in one gap means an empty gap somewhere else.
  • Force turns the ball back inside. The force defender is a role, not a position: the end, a linebacker, a safety or a corner, depending on the call. The coverage usually decides who is free to be him (safety force in Cover 3, corner force in Cover 2).
  • Spill and box send the ball to opposite places. Spill wrong-arms the kick-out and bounces the ball outside to speed (and dares the offense to log it); box keeps the outside arm free and squeezes the ball inside to the linebackers. Each is a promise about who is waiting, and the alley player fills the lane between the box and the force, coming inside-out as the force comes outside-in.
  • The plus-one is the extra defender, and it usually comes from a safety. Offenses erase it with an extra blocker, a quarterback read or a pass; defenses get it back with rotation, gap exchanges or two-gapping fronts. Count the quarterback when he runs.
  • The light box is a bet, not a mistake. Two-high defenses give up a few yards per carry, four or five before the safeties arrive, to protect against the explosive pass. In 2018–2025 data, each extra box defender cost runs about as much on average as it gave passes, but it bought the passing game far more big plays than it took from the run game, and the run the light box concedes still averages below zero EPA.
