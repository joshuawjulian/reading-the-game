OL_KEYS = ("LT", "LG", "C", "RG", "RT")


def pdf_build():
    return os.environ.get("RTG_FORMAT", "").lower() in ("pdf", "typst", "print")


def O(key, pos, d, w, label):
    return Player(key, pos, "off", d, w, label=label)


def D(key, pos, d, w, label):
    return Player(key, pos, "def", d, w, label=label)


def line(labels="full"):
    """The five offensive linemen, labelled LT..RT ("full") or with one letter, T G C G T ("short")."""
    short = {"LT": "T", "LG": "G", "C": "C", "RG": "G", "RT": "T"}
    return [p.moved(label=(p.key if labels == "full" else short[p.key])) for p in offense("singleback")
            if p.key in OL_KEYS]


def te(side=1, key=None, label=None):
    """An in-line tight end: Y on the right, U on the left."""
    key = key or ("Y" if side > 0 else "U")
    return O(key, "TE", -0.7, side * 4.8, label or key)


def T(t, side=1):
    """Width of technique t on the offense's right (side=+1) or left (side=-1)."""
    return technique(t, side)


def go_to(p, key, end, kind="path", delay=0.0, speed=None, via=()):
    """Move `key` through absolute (d, w) points `via` to `end`, as a path, run, route, block or pull."""
    start = np.array(p._cur(key))
    pts = [tuple(np.array(q) - start) for q in (*via, end)]
    if kind == "block":
        return p.block(key, pts=pts, delay=delay, speed=speed)
    if kind == "pull":
        return p.pull(key, pts, delay=delay, speed=speed)
    if kind == "route":
        return p.route(key, pts, frame="field", delay=delay, speed=speed)
    return {"path": p.path, "run": p.run}[kind](key, pts, delay=delay, speed=speed)


def path_len(pts):
    """Length in yards of a path through (d, w) points."""
    return float(sum(np.hypot(b[0] - a[0], b[1] - a[1]) for a, b in zip(pts[:-1], pts[1:])))


def _P(fld, p, d, w):
    return fld.xy(*p.to_abs(d, w))


def say(fld, p, d, w, text, color=style.INK_2, fontsize=7.2, ha="center", weight="normal"):
    """A short on-field label at (d, w)."""

def point_to(fld, p, text_at, target, text, color=style.INK_2, fontsize=7, shrink=7):

def spot(fld, p, d, w, color=style.DEFENSE_DARK, ms=10):
    """Mark a point (a tackle, a meeting point) with an X."""
    x, y = _P(fld, p, d, w)
    fld.ax.plot([x], [y], marker="X", ms=ms, color=color, mec="white", mew=0.8, zorder=12)


def ring(fld, p, key, color=GREEN, r=1.25, lw=1.8, ls="-"):
    """A ring around a player at his alignment."""
    q = p.players[key]
    x, y = _P(fld, p, q.d, q.w)
    fld.ax.add_patch(Circle((float(x), float(y)), r, fc="none", ec=color, lw=lw, ls=ls, zorder=7.5))


def lane(fld, p, w0, w1, d0, d1, color=style.OFFENSE, alpha=0.10, ls=(0, (3, 2))):
    """Shade a rectangle from width w0 to w1 and depth d0 to d1."""
    (x0, y0), (x1, y1) = _P(fld, p, d0, w0), _P(fld, p, d1, w1)
    fld.ax.add_patch(Rectangle((min(x0, x1), min(y0, y1)), abs(x1 - x0), abs(y1 - y0), fc=color, alpha=alpha,
                               ec=color, lw=0.9, ls=ls, zorder=0.8))


def dashed(fld, p, pts, color=style.THIRD, lw=1.5, dash=(0, (3, 2.2)), end="arrow"):
    """A dashed 'what if' or 'intent' path through absolute (d, w) points."""

    xs, ys = np.asarray(xs, float), np.asarray(ys, float)
    if end == "arrow":
        hx, hy = xs[-2] + 0.75 * (xs[-1] - xs[-2]), ys[-2] + 0.75 * (ys[-1] - ys[-2])
        xs = np.concatenate([xs[:-1], [hx, xs[-1]]]); ys = np.concatenate([ys[:-1], [hy, ys[-1]]])
        fld.ax.plot(xs[:-1], ys[:-1], color=color, lw=lw, ls=dash, zorder=6)
        fld.ax.add_patch(FancyArrowPatch((xs[-2], ys[-2]), (xs[-1], ys[-1]), arrowstyle="-|>",
                                         mutation_scale=10, color=color, lw=lw, zorder=6))
    else:
        fld.ax.plot(xs, ys, color=color, lw=lw, ls=dash, zorder=6)


def solid(fld, p, pts, color=style.DEFENSE, lw=1.6, z=5, head=True):
    """A solid path with an arrowhead, through absolute (d, w) points."""

                                         (xs[-1], ys[-1]), arrowstyle="-|>", mutation_scale=10, color=color,
                                         lw=lw, zorder=z))


def eyes(fld, p, a, b, color=style.THIRD):

def bracket(fld, p, d, w0, w1, color=style.DEFENSE_DARK, lw=1.4, tick=0.35):
    """A bar from w0 to w1 at depth d, with end ticks."""

def clean(ax):
    """Box-zoomed diagrams: drop hash ticks and yard numbers, clip anything outside the window."""
    for ln in list(ax.lines):
        xs, ys = ln.get_data()
        if len(xs) == 2 and ln.get_linewidth() == 0.5 and np.hypot(xs[1] - xs[0], ys[1] - ys[0]) < 1.0:
            ln.remove()
    for t in list(ax.texts):
        if t.get_text().isdigit() and t.get_alpha() == 0.55:
            t.remove()
        else:
            t.set_clip_on(True)

# --- fronts used throughout (tight end, and so the strength, on the offense's RIGHT) -------------------
def over_front(s=1, sam=True, mike_w=0.8, will_w=-2.0):
    """4-3 Over to side s (as in 05-02): weak 5, 1 shaded weak, 3 strong, strong 5; Sam off the ball
    outside the TE, Mike over the strong A gap, Will over the weak (uncovered) guard."""
    out = [D("WE", "DE", 1.0, T("5", -s), "5"), D("NT", "DT", 1.0, T("1", -s), "1"),
           D("DT", "DT", 1.0, T("3", s), "3"), D("SE", "DE", 1.0, T("5", s), "5"),
           D("MIKE", "LB", 4.5, s * mike_w, "M"), D("WILL", "LB", 4.5, s * will_w, "W")]
    if sam:
        out.append(D("SAM", "LB", 4.2, s * 6.2, "S"))
    return out


def nickel_front(mike=(4.6, 1.6), will=(4.6, -1.6), wide_ends=True):
    """A 4-2-5 nickel front against a one-tight-end offense (TE right): 5 (wide) on the left, 3 on the left
    guard, 1 shaded right of the center, 5 on the right tackle; two linebackers."""
    le = -5.0 if wide_ends else T("5", -1)
    return [D("WE", "DE", 1.0, le, "5"), D("DT", "DT", 1.0, T("3", -1), "3"),
            D("NT", "DT", 1.0, T("1", 1), "1"), D("SE", "DE", 1.0, T("5", 1), "5"),
            D("MIKE", "LB", mike[0], mike[1], "M"), D("WILL", "LB", will[0], will[1], "W")]

MARK_R = 0.62


def tracking(play):
    df = play.to_tracking()
    df["_id"] = df["displayName"].astype(str)
    return df


def panel(ax, play, df, t, lateral, window, rings=(), enter=(), ring_r=0.85):
    """One moment of a simulated play on `ax`: markers, trails of on-screen players, rings."""
    tmap = df.groupby("frameId")["time"].first()
    fid = int((tmap - t).abs().idxmin())
    ylo, yhi = sorted((play.ball_y - lateral[0], play.ball_y - lateral[1]))
    fld = Field(ax=ax, orient="vertical", los=play.los, window=window, numbers=False, hashes=False,
                lateral=(ylo, yhi), to_go=play.to_go)
    cur = df[df["frameId"] == fid]
    snap = df[df["frameId"] == int((tmap - 0.0).abs().idxmin())]
    first = snap[snap["x"].between(play.los + window[0] + 0.8, play.los + window[1] - 0.8)
                 & snap["y"].between(ylo + 0.8, yhi - 0.8)]
    ids = set(first["_id"]) | {"football"}
    inside = cur[cur["x"].between(play.los + window[0], play.los + window[1]) & cur["y"].between(ylo, yhi)]
    ids |= set(inside["_id"]) & set(enter)
    shown = cur[cur["_id"].isin(ids) | (cur["side"] == "ball")]
    trails = df[(df["frameId"] <= fid) & df["_id"].isin(ids) & (df["side"] != "ball")]
    for (_, side), g in trails.groupby(["_id", "side"]):
        xs, ys = fld.xy(g["x"].values, g["y"].values)
        ax.plot(xs, ys, color=style.OFFENSE if side == "off" else style.DEFENSE, lw=0.9, alpha=0.4, zorder=3)
    ppy = fld.pts_per_yard()
    for _, r_ in shown.iterrows():
        x, y = (float(v) for v in fld.xy(r_["x"], r_["y"]))
        if r_["side"] == "ball":
            ax.add_patch(Ellipse((x, y + 0.45), 0.7, 0.42, fc=style.BALL, ec="white", lw=0.5, zorder=9))
            continue
        lab = str(r_.get("label", "") or "")
        if r_["side"] == "off":
            ax.add_patch(Circle((x, y), MARK_R, fc=style.OFFENSE, ec="white", lw=0.6, zorder=8))
            tc = "white"
        else:
            ax.add_patch(RegularPolygon((x, y), 4, radius=MARK_R * 1.3, orientation=np.pi / 4,
                                        fc=style.DEFENSE_TINT, ec=style.DEFENSE, lw=1.0, zorder=8))
            tc = style.INK
        fs = max(4.0, min(8.0, ppy * (0.85 if len(lab) <= 1 else 0.62)))
        ax.text(x, y, lab, ha="center", va="center", fontsize=fs, color=tc, fontweight="bold", zorder=10)
    for key, col in rings:
        r_ = cur[cur["displayName"] == key].iloc[0]
        x, y = fld.xy(r_["x"], r_["y"])
        ax.add_patch(Circle((float(x), float(y)), ring_r, fc="none", ec=col, lw=1.6, zorder=7))
    for patch in ax.patches:
        patch.set_clip_on(True)
    return fld


def strip(moments, lateral, window, notes=None, rings=(), ncols=2, width_in=5.2, after=None, enter=()):
    """Numbered frame strip. `moments` is a list of (play, t, tag)."""
    cache = {}
    nrows = int(np.ceil(len(moments) / ncols))
    pw = width_in / ncols
    span_w, span_d = lateral[1] - lateral[0], window[1] - window[0]
    fig, axes = plt.subplots(nrows, ncols, squeeze=False,
                             figsize=(width_in, nrows * (pw * span_d / span_w + (0.62 if notes else 0.25))))
    for ax in axes.flat[len(moments):]:
        ax.axis("off")
    for i, (ax, (play, t, tag)) in enumerate(zip(axes.flat, moments)):
        if id(play) not in cache:
            cache[id(play)] = tracking(play)
        fld = panel(ax, play, cache[id(play)], t, lateral, window, rings, enter)
        if after:
            after(fld, play, t, i)
        lab = "snap" if abs(t) < 1e-6 else f"{t:+.1f} s"
        ax.set_title(f"{i + 1}.  {lab}" + (f"   {tag}" if tag else ""), fontsize=8.0, color=style.INK_2, loc="left")
        if notes:
            ax.text(0.5, -0.03, notes[i], transform=ax.transAxes, ha="center", va="top",
                    fontsize=6.9, color=style.INK)
    fig.tight_layout(h_pad=1.3 if notes else 0.6, w_pad=0.6)
    return fig


def video(segments, lateral, window, rings=(), width_in=4.6, after=None, enter=()):
    """Web version: each (play, tag, [(t, note), ...]) segment played in turn, notes as captions."""
    import shutil
    from matplotlib import animation
    from IPython.display import HTML, display
    span_w, span_d = lateral[1] - lateral[0], window[1] - window[0]
    fig, ax = plt.subplots(figsize=(width_in, width_in * span_d / span_w + 0.45))
    frames = []
    for play, tag, notes in segments:
        df = tracking(play)
        tmap = df.groupby("frameId")["time"].first()
        frames += [(play, df, float(tmap[f]), tag, notes) for f in sorted(df["frameId"].unique())]

    def update(k):
        play, df, t, tag, notes = frames[k]
        ax.clear()
        fld = panel(ax, play, df, t, lateral, window, rings, enter)
        if after:
            after(fld, play, t, -1)
        txt = ("snap" if abs(t) < 0.05 else f"{t:+.1f} s") + (f"  {tag}" if tag else "")
        cur = [n for tt, n in notes if tt <= t + 1e-6]
        ax.set_title(txt + (("\n" + cur[-1].replace("\n", " ")) if cur else ""), fontsize=8.5, color=style.INK,
                     loc="left")
        return []

    anim = animation.FuncAnimation(fig, update, frames=range(len(frames)), interval=100)
    html = (anim.to_html5_video(embed_limit=40).replace("<video ", '<video style="max-width:100%;height:auto" ')
            if shutil.which("ffmpeg") else anim.to_jshtml(default_mode="once"))
    plt.close(fig)
    display(HTML(html))


def animation_or_strip(moments, notes, lateral, window, rings=(), after=None, enter=(), ncols=2, width_in=5.2):
    """Video on the website; a numbered, captioned frame strip in the PDF."""
    if pdf_build():
        strip(moments, lateral, window, notes=notes, rings=rings, ncols=ncols, width_in=width_in,
              after=after, enter=enter)
        plt.show()
        return
    segs = []
    for play, t, tag in moments:
        if not segs or segs[-1][0] is not play:
            segs.append((play, tag, []))
    for (play, t, tag), n in zip(moments, notes):
        for s in segs:
            if s[0] is play:
                s[2].append((t, n))
    video(segs, lateral, window, rings=rings, after=after, enter=enter)

import re
from gridiron.data import nfl, pbp

SEASONS_P = list(range(2016, 2026))


def n_db(s):
    """Defensive backs on the field, from nflverse participation's defense_personnel string.
    2016-2022 rows read like '4 DL, 2 LB, 5 DB'; 2023+ rows list positions ('3 CB, 1 FS, 1 SS, ...')."""
    if not isinstance(s, str):
        return np.nan
    m = re.search(r"(\d+) DB", s)
    if m and "CB" not in s:
        return int(m.group(1))
    return sum(int(k) for k, pos in re.findall(r"(\d+) ([A-Z]+)", s) if pos in ("CB", "FS", "SS", "S", "DB", "SAF"))


rows = []
for season in SEASONS_P:
    part = nfl().load_participation([season]).to_pandas()[["nflverse_game_id", "play_id", "defense_personnel"]]
    plays = pbp(season, columns=["game_id", "play_id", "season_type", "play_type", "down"])
    plays = plays[(plays.season_type == "REG") & plays.play_type.isin(["pass", "run"]) & plays.down.notna()]
    m = plays.merge(part, left_on=["game_id", "play_id"], right_on=["nflverse_game_id", "play_id"])

The hardest job on the field

A linebacker lines up behind the defensive line, about four or five yards off the ball, at what coaches call the second level of the defense: the linemen are the first level, the safeties the third. You met the three classic linebackers in The Twenty-Two: the Mike in the middle, the Sam on the strong (tight end’s) side and the Will on the weak side. In Even Fronts you saw where they stand in a 4-3, and that each front leaves one guard uncovered and puts a linebacker over him.

This chapter is about off-ball linebackers: the ones who stand off the line of scrimmage, behind the linemen. The name “outside linebacker” can mean two different players, and it is worth sorting out now. In a 4-3, the Sam and the Will are outside linebackers in the old sense: off the ball, on either side of the Mike. In a 3-4, “outside linebacker” usually means a stand-up edge (Odd and Hybrid Fronts), an edge rusher who lines up on the line and rushes the passer; that is a different job, and this chapter mostly leaves it alone. When it says inside linebackers, it means the off-ball linebackers in the box: the Mike and the Will in a 4-3, the two linebackers of a nickel defense, the two inside linebackers of a 3-4 (whom staffs name differently: Mike and Will is common).

Why call it the hardest job? Put yourself in his cleats for the first second of a play.

  • He has two jobs that point in opposite directions. Against a run he has a gap to fill, and it is in front of him: he has to go forward, fast, because the running back reaches the line about a second after the snap. Against a pass he has a zone to drop to or a man to cover, and that is behind him or beside him. Every snap, he has to choose before he can be sure.
  • He is the one who gets blocked by someone twice his size. Defensive linemen are big men fighting big men. A linebacker weighs about 225 to 245 pounds, and the man coming to block him, a guard who has climbed off a double team, weighs 315. He cannot win that collision by force. He wins by getting to the spot first, or by meeting the block on his own terms and getting off it.
  • He tackles in space. A defensive lineman makes his tackles in a crowd, with the runner barely moving. A linebacker meets the ball carrier after he has come through the line, at full speed, with room to cut both ways.
  • He covers people who are faster than he is. Running backs and tight ends run routes against him on every pass play, and offenses spend whole weeks scheming to get their best receiver matched up with a linebacker.
  • He often runs the defense. In most NFL defenses the Mike linebacker hears the call in his helmet radio, relays it to the other ten, and adjusts it to the formation, all in the 15 seconds before the snap.

The position grew out of a lineman standing up. In the early 1950s most NFL defenses played a five-man line with a middle guard over the center. The Chicago Bears’ middle guard, Bill George, kept hitting the center and then watching passes sail over his head, so during a 1954 game against the Philadelphia Eagles he started dropping back off the line at the snap, and two plays later he intercepted a pass, the first of his career. He kept doing it. A five-man line with its middle man standing up is a four-man line with a linebacker in the middle. Nobody can be sure which middle guard first stood up, but George is the usual answer, and he is often credited, by accident, with the 4-3 as well.1 Two years later Tom Landry, coaching the New York Giants’ defense, built a 4-3 around a rookie from West Virginia, Sam Huff, and in 1960 CBS wired Huff with a microphone for a half-hour special, The Violent World of Sam Huff. The middle linebacker became the most famous defensive player in football, and has stayed near the top ever since: Dick Butkus, Ray Nitschke, Mike Singletary, Ray Lewis, Brian Urlacher, Luke Kuechly.2

This chapter takes the linebacker’s job apart in the order he does it: where he stands, what he looks at, which way he goes, how he gets there, how he tackles, and what he does when it’s a pass. Then it explains why the position looks so different today from the one Sam Huff played.

Where a linebacker stands

Before the snap a linebacker picks a spot, and the spot is a decision about two things: how far he is from the blockers who will come for him, and how far he is from the receivers he may have to cover. Three alignments cover most of what you will see.

Stacked. In a stack alignment the linebacker lines up directly behind a defensive lineman, four or five yards deep. The lineman is his shield. A blocker who wants the linebacker has to go around the lineman to get to him, and by then the linebacker has seen the play. The stack also hides the defense’s intentions: a lineman and the linebacker behind him own the two gaps on either side of that lineman between them, and the offense cannot tell before the snap which man will take which. On one snap the lineman slants left and the linebacker fills right; on the next, the reverse. The left panel of Figure 31.1 is the extreme version, a 3-3 stack, with all three linebackers hidden behind all three linemen; you met it as the 3-3-5 front in Odd and Hybrid Fronts. Four-man fronts usually stack at most one linebacker (in many Under fronts, the Will behind the 3-technique) and play the others open or over a shaded lineman.

The cost of the stack is that his own lineman is in his way. To get to either gap he has to go around his own man, so he is a step slower to it than a linebacker standing over open grass, and if the lineman in front of him gets driven backward, the linebacker is caught in the pile (coaches say “in the wash”).

Open. The opposite is to stand over an uncovered offensive lineman, a guard nobody is lined up on. That is the middle panel, and it is the bubble you met in Even Fronts: every even front leaves one guard uncovered and puts a linebacker over him. The linebacker gets a clear view of the whole backfield, and nobody is in his way when he goes. The cost is that nobody is in the guard’s way either. He can climb straight up to the linebacker at the snap, so the open linebacker has to be quick enough to beat him to a spot two yards away.

NoteGo deeper

Coaches number linebacker spots the way they number linemen’s techniques, often with a zero added: a linebacker in a 30 is lined up on the guard’s outside shoulder, the spot behind a 3-technique, and one in a 10 is on the center’s shoulder, over the A gap. You won’t hear it on a broadcast, but it is how a linebacker coach talks, and it tells you that linebacker alignments are as precise as the line’s.

Depth. Off-ball linebackers usually stand about four to six yards deep, a little deeper against a team that likes play-action or against a running back aligned deep behind the quarterback, a little shallower on short yardage. The trade is simple. Every yard closer to the line is a step quicker to the hole and a step closer to the blockers; every yard deeper is more time to read and less time to get there.

A First Look at Defense introduced the defender who stands there, the overhang defender, usually a nickel back. When a linebacker goes there instead, coaches call it a walk-out: he walks out of the box before the snap to stand over or near the slot receiver.

Figure 31.2 shows why a walk-out happens. The offense has spread four receivers across the field, two on each side. A base 4-3 has only four defensive backs: two cornerbacks on the wide receivers and two safeties deep. Somebody has to stand near the two slot receivers. The defense has three answers: walk the outside linebackers out, as here; rotate a safety down over one slot and keep one linebacker in the box (a single-high look); or take a linebacker off and bring in a nickel back, which the defense can only do before the snap, by substitution or a pre-planned check.

Why walk out? Because leaving him inside is worse. A linebacker left in the box against a slot receiver is eight or nine yards from a man he may have to cover, which is an easy completion every time the quarterback looks at him. From the apex, he can still come downhill at the edge of the box against a run, outside-in, keeping the ball inside him, and he can get under or in front of the slot receiver’s route against a pass. (Whether he squeezes the run or bounces it outside is the call’s decision, built in Run Fits and Box Math.)

What does it cost? Count the box. Five defenders against five offensive linemen and a running back: the box is now light, and every run defender can be blocked. And each walked-out linebacker is now the defense’s conflict defender: if he comes inside for the run, the slot receiver behind him is open; if he widens for the pass, the edge of the box is open. The offense can read him after the snap and throw or run accordingly, which is exactly what the RPOs of Part 4 are built to do.

That is the problem that pushed linebackers off the field. A defense that expects the slot receiver to run routes would rather have a defensive back at the apex, a nickel back who can cover him, and it takes a linebacker off to make room. The last section of this chapter follows that trade to its end. For now, notice the walk-out as a tell: when a linebacker walks out with a receiver, the defense has told you it is short of defensive backs for this formation, and the offense will usually try to find out whether he can cover.

Keys: what a linebacker looks at

Here is the problem with standing five yards off the ball. Things happen faster than a linebacker can see all of them. At the snap, eleven offensive players move at once; the running back has the ball, or doesn’t, half a second later; the linebacker has to be moving in the right direction by then. He cannot watch eleven people. So he watches one or two.

A key is the offensive player a defender watches at the snap to learn what the play is and where it is going. (It is the defense’s version of an idea; don’t confuse it with an option play’s read key, which is the defender a quarterback watches.) For an inside linebacker, the two classic keys are the guards and the near running back, and coaches talk about reading “through” the guard to the back: eyes on the guard’s first step, the back in the edge of the picture.

Which guard? The simplest rule is the guard nearest him, the one he is lined up on or over. Many teams have their inside linebackers read both guards, the near one first, because the far guard’s pull is the single most valuable thing a linebacker can see (the Mike in Figure 31.3, below, reads his near guard and sees the far one pull). You’ll hear the versions by name: a “guard read” or “high hat, low hat” read; a “triangle” of the near guard, the far guard (or the near tackle) and the near back (not the quarterback’s triangle read of Part 4); and a “flow” or “backfield” read, which watches the backs and trusts the linemen in front to keep the guards off.

What the guard tells him

Offensive linemen can’t hide what kind of play it is, because the physics of blocking and the rules tell them how to move (a lineman may not go downfield on a pass, a rule the play-action section below puts to work). A guard’s first step is the most honest signal on the field.

Fires forward, pads low (“low hat”) Run, and the guard is blocking the man in front of him or doubling a lineman Downhill: attack his gap
Comes straight up at the linebacker Run, and the linebacker is the guard’s target; the ball is coming near him Beat the guard to the spot, or take him on and keep his gap

The “high hat, low hat” read is the same one How to Watch a Broadcast gave you for spotting run and pass from the couch. The linebacker uses it a quarter-second after the snap, from five yards away.

The last row is where reading gets hard. A draw is built to look like a pass: the linemen set, the quarterback drops, and then he hands the ball to a back who waited. A linebacker who reads “high hat, pass” and sprints to his zone is running away from it. The tells are small: the back stays square in front of the quarterback instead of stepping up to block or releasing; the guards set but don’t punch, inviting the rushers upfield; the quarterback’s drop is a little too quick. So linebackers are taught to open to their drop with their eyes still on the back for a beat. A screen shows the same high hat first, then the linemen let their rushers through and release, running sideways after a pass set, while the back slips out behind them. Linemen running flat after setting to pass is the screen’s signature.

The pull is the most valuable signal of all, because the gap-scheme runs of Gap and Power Runs need a pulling lineman to get an extra blocker to the hole, and that lineman has to run there. Wherever he is going, the ball is going too. That is the read in the chapter’s opening scene, and Figure 31.3 plays it out, jab step and all.

What the back tells him

The near back is the second key. His first two steps say three things: direction (left, right, or straight ahead), speed (a sprint sideways or a patient downhill track) and depth (does he get deeper, for a toss or a pitch, or come straight at the line). If he steps up to block, or swings out of the backfield to the flat, it is a pass. If there is a fullback, he is the most honest back of all: on an iso or a lead play, he runs straight at the linebacker he is going to block, which tells that linebacker where the ball is going.

Why not just read the back? Because the back can lie and the linemen mostly can’t. Offenses build whole plays on the backfield lie. The counter starts with the back stepping away from the play while the guard and tackle pull toward it; a linebacker who believes the back takes a false step and is out of position before he knows it. Play-action shows him a back who seems to have the ball. That is why most coaches teach the guards as the primary key and the back as the confirmation: if they say the same thing, go; if they disagree, believe the linemen.

Keys also depend on the defense. Linebackers in a two-gap 3-4, who sit behind big linemen and are protected from blockers, often read the guards almost exclusively. Linebackers in an attacking one-gap 4-3 (the Tampa 2 teams of the 2000s are the classic case) often read the backfield flow and trust their linemen to hold the guards. And a Sam on the line’s strong side, or a walked-out apex player, keys the tight end or tackle in front of him and the near back, because the play coming his way starts out there: a tight end who blocks down means the ball is coming outside him, one who releases means pass, and from the apex the eyes go on through to the slot receiver.

ImportantWhy it exists

Keys exist because a linebacker who waits to be sure is always late. The back reaches the line in about a second; a linebacker four or five yards away needs most of that second to get there. So he decides from the first step of one or two blockers, who have to show their intent early. The price is that the offense knows what he is looking at. Counters, traps, play-action and screens are all, in part, plays built to show a linebacker’s key a lie.

CautionCommon misconception

“Great linebackers just follow the ball.” Watch the replay of a linebacker making a tackle for no gain, and then watch his first step: almost always it came before the ball went anywhere, and it went where a lineman was going. The ball is the last thing a good linebacker looks at, not the first. Linebackers who “see the ball” are usually the ones a counter or a play-action pass catches out.

Keys tell a linebacker what the play is. Flow tells him how to move. In coaches’ language, flow is the direction and speed of the backfield action at the snap: where the backs (and, on zone, the whole line) are going, and how fast. Staffs use different words, but nearly all of them sort it into two kinds.

Fast flow is the ball going sideways in a hurry: outside zone, a toss, a sweep, the back sprinting flat for the edge. The linebacker’s job is to run with it, and to run with it correctly. Correctly means inside-out: a yard or so behind his own linemen and a step behind the ball, so that the ball is always a little outside him. (You will see the phrase again; it is the linebacker’s master rule.) If the back keeps going, the linebacker can keep up and fill as soon as he turns upfield. If the back stops and cuts back, which is what outside zone is designed to make him do (see Zone Running), the linebacker is still inside him and can stop. The backside linebacker, the Will in Figure 31.4, shuffles over the top: behind the Mike and a little deeper, ending up behind the spot where a cutback would come through. He is the cutback player: if both linebackers race to the sideline, the back has the whole field behind them.

Slow flow is the ball coming straight at the line, or coming slowly: inside zone, an iso, a draw, a delayed handoff, the counter’s first step. Now the linebacker presses: he steps forward into his gap, square to the line, shoulders parallel to it, and makes the back declare first. The worst thing he can do against slow flow is to run sideways, because every inside run is built around a back who reads the linebackers and cuts into the gap one of them just left.

There is a third case, and coaches have a phrase for it: split flow, the backs going in opposite directions (one back left, the other right, or a back one way and the quarterback the other). Split flow is the offense saying misdirection. A linebacker who sees it plays his gap and does not chase either one until the ball shows.

Fitting the run: fill, plug, scrape and run-through

Once a linebacker knows it’s a run and where it’s going, he has to get into the right gap. The defense’s plan for who takes which gap is the run fit, and its full arithmetic is the subject of Run Fits and Box Math. What a linebacker does to get to his gap has four common names, and you will hear all four on broadcasts and in coaches’ interviews.

  • Fill. To fill is to attack downhill into the gap the run is aimed at and meet whatever comes through it, a lead blocker or the back, as close to the line of scrimmage as possible. It is the general verb: “the Mike filled the hole.” The closer to the line he meets the play, the less room the back has to choose a different hole, so coaches measure fills in yards: a fill at the line of scrimmage is a good play even if the linebacker doesn’t make the tackle, because it forces the back to stop or bounce.
  • Plug. To plug is to step straight forward into a gap directly in front of you and close it with your body. It is the inside linebacker’s partner to the defensive line’s slant (05-01): when the 3-technique slants into the A gap, the B gap he left behind needs a body, and the linebacker stacked behind him plugs it. Plugs are short and violent, and they usually end in a collision with a lineman.
  • Scrape. To scrape is to run sideways, behind and close to your own defensive linemen (“over the top” of them), to a gap farther outside. It is how a linebacker gets to an outside gap without getting caught in the traffic at the line, and it is the move you saw in fast flow. Scraping linebackers stay square to the line, shuffling or crossing over, so that they can turn upfield and fill the instant the back commits. The defense’s answer to the zone read, below, is built on it.
  • Run-through. A run-through is the aggressive version of a fill: the linebacker attacks through his gap at the snap (or on his first key), before the blocker assigned to him can climb, and meets the play in the backfield. “Run-through linebacker” is also a compliment, for the kind of player who does it naturally. It wrecks zone schemes, whose climbing linemen need a fraction of a second to come off their double teams, and it makes tackles for loss. It is also the fit most vulnerable to misdirection: a linebacker who runs through a gap on a counter step or a play-action fake is out of the play, or out of his coverage.

The call and the front decide a lot (a defense whose linemen slant and stunt asks its linebackers to plug and scrape behind them; one that wants plays in the backfield against a zone team asks them to run through), but from the couch you can usually match the fit to what the run looks like:

What the run shows the linebacker The fit it usually calls for
A fullback or lead blocker coming straight at him (iso, lead) Fill: meet the lead blocker in the hole, at the line
His own lineman slanting away, leaving the gap in front of him empty Plug that gap
The ball going outside, or the end in front of him squeezing inside (and the zone read, below) Scrape over the top to the outside gap
Zone blocking, with linemen who need a moment to climb to him, and a call that says go Run-through before they arrive

The first step, and the block he can’t avoid. None of these starts with a lunge. A linebacker’s first step is a short read step (some coaches say “pop step”): a few inches, downhill, that keeps him moving without committing him. His second step is the one that tells you what he saw. The cardinal sin is the false step, a first step the wrong way, which is exactly what counters and play-action are built to draw. And sometimes a climbing lineman gets to him anyway. Then he does not run around the block, which would open the gap behind him (unless the call tells him to). He meets the blocker with his hands, a punch to the chest, keeps his gap-side arm free, and gets off the block toward his gap. Which shoulder he takes the blocker on with, keeping the ball inside him (box, or squeeze) or bouncing it outside (spill), is a whole-defense decision, and Run Fits and Box Math builds it.

NoteMadden vs. real life

In Madden, the linebacker you control can run anywhere and the computer fills the gap you left. On a real field, every linebacker’s gap is somebody else’s assumption. If the Mike scrapes outside when the call told him to plug, the gap he was supposed to fill is empty, and the defensive lineman next to it, who slanted away because the Mike would be there, is now on the wrong side of the ball. That is why coaches care more about a linebacker being in the right gap than about him making the tackle, and why “he was out of his gap” is one of the harshest things a linebacker coach says on film.

The scrape against the zone read

The scrape has its most famous use against the zone read. On a zone read the offense leaves the backside end unblocked and the quarterback reads him: if he stays wide, hand off; if he crashes down on the back, keep the ball and run where the end was. The defense’s answer, the scrape exchange from Option Football, has the end crash on the back on purpose and the linebacker behind him scrape around him to the spot where the quarterback will keep. The quarterback reads “keep” correctly and runs into the linebacker.

The scraper has three things to get right. Timing: he has to leave at the snap, before the read, because a quarterback keeps the ball and is past the line in about a second and a half. Path: tight to the crashing end, so that the quarterback can’t run between them and the uncovered left tackle, climbing at the snap, can’t get to him first; coaches tell scrapers to “rub the end’s hip.” Leverage: he has to arrive inside the quarterback, on his inside hip, with a teammate (the nickel back or a safety in Figure 31.6) outside, so that the quarterback’s only way out is toward that teammate. An overrun, on the right, is the scraper’s classic mistake: he gets to the spot so eagerly that he passes it, and an athletic quarterback cuts back underneath him into the hole the end left. Offenses fight back by giving the scraper something else to worry about, like a tight end or H-back who arcs around the end to block him, which is why the scrape exchange is a call the defense mixes in, not something it can do every snap.

Getting there: pursuit angles

Most runs don’t end in the hole. The back gets through, or bounces outside, and the play becomes a footrace in which several defenders who started in different places converge on one runner. How they converge is the difference between a five-yard gain and a fifty-yard one.

A pursuit angle is the path a defender takes to a ball carrier who is already moving. The instinct, for anyone who has never been coached, is to run at the runner. That is almost always wrong.

def intercept(R0, vR, L0, sL):
    """Earliest time t at which a defender from L0 at speed sL can reach the runner R0 + vR t."""
    d = np.asarray(R0, float) - np.asarray(L0, float)
    a = vR @ vR - sL ** 2
    b = 2 * d @ vR
    c = d @ d
    if abs(a) < 1e-9:
        return -c / b
    disc = b * b - 4 * a * c
    ts = [(-b - np.sqrt(disc)) / (2 * a), (-b + np.sqrt(disc)) / (2 * a)]
    return min(t for t in ts if t > 0)

Figure 31.7 makes the point with real geometry. A linebacker who runs at the back (the grey path) is always aiming at a spot the back has already left. In this example the linebacker is the faster man, and he is still two yards behind after a second and a half; he gets within arm’s reach only after 3.6 seconds, 23 yards downfield. Against a back who is as fast as he is, or faster, which in the open field is most backs, he never catches him at all. A linebacker who aims at where the back will be (orange) cuts the distance he has to run and meets the back in about a second and a quarter. That is a pursuit angle. The catch is the dashed path: aim too far ahead, and the linebacker gets outside the back, who simply plants his foot and cuts back behind him. Every defender who has been run past has done one of those two things: chased, or overrun.

The rule coaches give for it is inside-out again: take the angle that keeps the ball carrier a step outside you, never inside. If you are inside him, he can only go one way, outside, and outside is where the next defender is.

That next defender is the leverage player: the man on the outside whose job is to keep the ball inside, turning the runner back toward the pursuit. In Figure 31.7 it is the cornerback; on other plays it is the end who set the edge, an outside linebacker, a safety who came down. (The defense names him in advance, as its force player, in Run Fits and Box Math.) Put the two together and you have the shape of good team pursuit: one defender outside the ball, squeezing it in, and everyone else inside-out on it, each a little deeper and farther inside than the man in front, so that whichever way the runner cuts, someone is already there. Coaches drill it constantly, and it is one of the clearest things you can see on the end-zone replay: on a good defense, the arrows all converge.

Tackling, and why tackles get missed at the second level

Then the linebacker has to bring the man down. Tackling has changed more in the last decade than in the fifty years before it, for two reasons: safety, and the realization that the old way missed too many tackles.

The old way, the one a lot of fans still picture, was a head-on collision: eyes up, head across the runner’s body (“head across the bow”), drive through him. It produces highlight hits. It also uses the head as the first point of contact, and it fails whenever the runner moves at the last moment, because a tackler who commits his whole body to one spot can’t adjust.

The modern approach is leverage tackling, and its most influential teachers were Pete Carroll’s Seattle Seahawks. In 2014 they made a 20-minute instructional video of their shoulder tackling, borrowed from rugby, and that August the league sent it, with a letter from Carroll, to coaches and parents at every level of the game; a second video in 2015 named its basic tackle the hawk tackle.3 The ideas fit together:

  • Leverage first. The tackler approaches with the runner on one side of him, usually inside-out, so the runner’s only escape is toward the sideline and the next defender. The pursuit angle in Figure 31.7 is the start of the tackle, not something separate from it.
  • The near shoulder to the near thigh. The target is the runner’s thighs (“eyes through the thighs,” in Seattle’s coaching points), not his chest; the shoulder makes contact and the head goes to the side, behind the runner on a tackle from an angle, out of the line of contact. The legs are what carry the ball, and they can’t dodge as easily as a torso can.
  • Wrap and roll. Arms around the thighs, keep the feet moving, and roll the runner to the ground, taking him toward the help rather than trying to drive him backward.
  • Breakdown in space. Within a couple of yards of the runner, the tackler shortens his steps and lowers his hips, so he can move either way with the runner’s last cut.

The rules have pushed in the same direction. In March 2024 the league banned the hip-drop tackle, in which a tackler grabs a runner from behind and swings his hips and weight down onto the runner’s legs, with a 15-yard penalty; the details are in Contact Rules and Penalties.4 For the 2026 season the league and the players’ union added a three-strike system: a player called for three of five fouls (hip-drop, horse-collar, launching, hits on a defenseless player and blindside blocks) in a season is suspended for a game.5 Most of those fouls are committed by defenders tackling in pursuit, which is the linebacker’s daily life.

The chart is also a warning about the statistic fans quote most. A linebacker’s tackle total counts every tackle the same, but a tackle five yards downfield is a run that beat the defense first, and the best fill of the game, the one that meets the fullback at the line and makes the back bounce into a teammate, often goes in the box score under someone else’s name. Coaches grade linebackers on whether they were in their gap on time; the tackle is a by-product.

Missed tackles are not in the play-by-play, but Pro Football Reference charts them for every defender, and nflverse carries the numbers. Per hundred tackle attempts from 2021 to 2025 (runs and passes together), defensive linemen missed 7.2, inside linebackers 7.5, outside linebackers 8.4, safeties 8.5 and cornerbacks 9.9.7 The farther from the line a defender usually tackles, the more often he misses. A lineman tackles a back who is still in a crowd, barely moving, with nowhere to go. Past the line the runner is moving at full speed, with room to cut both ways, and the tackler is often coming off a block, from an angle, alone. The runner chooses his cut after seeing the tackler commit; the tackler has to commit first.

Two things stand out. Inside linebackers make their tackles in far more space than linemen do and miss barely more often: that is what leverage tackling, and all the drilling behind it, buys. (Take away one direction with your angle, and you only have to win the other.) And the misses that hurt are the ones past the line, because the cost of a miss grows with the space behind it: a lineman who misses leaves the back a crowd, a linebacker who misses leaves him a safety ten yards away.

Linebackers in coverage

Half the snaps in a game are passes, and on most of them the linebackers are in coverage. Their jobs fall into three families.

Zone: the hook/curl defender

In zone coverage, the linebacker drops to an area. The most common linebacker zone is the one between the offensive tackle and the numbers, about 8 to 12 yards deep, which combines the hook zone over the tackles and the curl zone outside it (both defined in Coverage Fundamentals). A linebacker who drops there is a hook / curl defender.

The hook/curl defender’s job is a sequence. First, wall: get in the path of the slot receiver’s inside route, so the slant or the dig has to go around him. If the slot runs straight up the seam instead, carry him: run with him, inside and underneath, until a deep defender can take over, about 10 to 12 yards downfield. Four receivers running deep against Cover 3’s three deep thirds is the classic way to beat it, and the hook/curl linebacker’s seam carry is what stops it. Then, sink: as the outside receivers push deep, back up under the curls and comebacks. All the while, eyes on the quarterback, because a zone defender breaks on the throw, not on the receiver. Linebackers in modern NFL defenses rarely drop to a fixed spot and wait; most play some version of pattern matching, zone that turns into man coverage on whichever receiver enters his area. The Cover 3 and Tampa 2 chapters show the most demanding linebacker drops: in Tampa 2, the Mike runs straight down the middle of the field to take away the seam, 15 or 20 yards deep.

The model hook defender of the last decade was Seattle’s Bobby Wagner, the Mike in the 2013 Seahawks’ Cover 3, behind the Under front of Even Fronts and in front of a secondary nicknamed the “Legion of Boom.” Watch his early-down drops in Super Bowl XLVIII, a 43–8 win over Denver in which he made ten tackles: how fast he gets depth, how he walls the crossing routes, and then how his run fills come square and downhill. He went on to six first-team All-Pro selections, more than two thousand career tackles, and the 2025 Walter Payton Man of the Year award, playing for Washington.8

Offenses build plays specifically to get the back in a route against a linebacker in space, like the angle route (a fake to the outside and a break back inside), and to put a fast tight end on a slow linebacker.

Play-action, revisited

Play-Action, Boots, and Screens showed play-action from the offense’s side: a fake run that pulls the linebackers forward, and a throw into the space behind them. Now that you know what a linebacker looks at, you can see why it works, and what the good ones do about it.

The fake is aimed at his keys, his run-fit triggers. The linemen fire out low, like a run. The back runs his run path. The quarterback reaches the ball toward him. Every key says run. But one key can’t say it completely, because of a rule: on a pass, an offensive lineman may not be more than a yard downfield when the ball is thrown. So on play-action, the linemen who would climb to the linebackers on the real run can’t. They block at the line and stay there. A linebacker who keys the back sees a run. A linebacker who keys the uncovered guard, the man who should be coming for him on a run, sees a guard who fires out and then doesn’t come. That is the tell.

Two things in Figure 31.11 are worth remembering. First, the linebacker who keys the back is not a bad linebacker; he is doing exactly what he is trained to do against the run, and on most snaps it is the right call. That is why, as 04-05 showed, play-action works about as well for teams that run the ball badly as for teams that run it well: the linebacker is reacting to action, not to statistics. Second, the cure is in the keys, and it costs something. A linebacker who waits for the guard to climb is a fraction of a second slower on every real run. Linebackers as pressure players

The last job is going forward on a pass play. Linebackers have traditionally been the most common extra rushers, the blitzers, because they start close to the line and in the middle, where a rusher reaches the quarterback fastest (modern defenses send their nickel backs and safeties too). Their pressure roles, all built in Part 7, come in a few recognizable shapes:

  • The A-gap blitz. Both inside linebackers walk up into the gaps on either side of the center before the snap, the double A-gap mug, threatening the quickest path to the quarterback. The offense has to account for both; the defense may send both, one, or neither and drop them into coverage at the snap (Blitz and Man Pressure).
  • The green dog (or hug, or add-on). A linebacker assigned to cover the running back who, when the back stays in to block, rushes instead: there is nobody left for him to cover.
  • The simulated pressure. A linebacker rushes while a defensive lineman drops into coverage, so the offense sees five threats but gets four rushers from unexpected places: simulated pressure, in Zone Blitz and Simulated Pressure.
  • The spy. Against a quarterback who runs, a linebacker is assigned to mirror him, staying at linebacker depth and following him wherever he goes: a spy, covered in Third Down. Against the fastest quarterbacks, teams often spy with a safety or the nickel back instead, for the same speed reason the Sam lost his job.

Every one of these trades coverage for pressure, or pressure for coverage, and the linebacker is the piece the defense moves to make the trade.

The part-time Sam and the safety-sized linebacker

Put the jobs together, the run fits, the zone drops, the man coverage on backs and tight ends, the blitzes, and you can see the problem the modern NFL created for linebackers. Offenses spread out. The more snaps an offense plays with three wide receivers, the more snaps one of those receivers is in the slot, and the more often the defense has to choose between a linebacker who walks out over him (Figure 31.2) and a defensive back who replaces that linebacker.

For most of the last two decades, defenses chose the defensive back. By Pro Football Focus’s count, base defense (four defensive backs, three linebackers) was on the field for 54% of plays in 2008 and 45% in 2011, and by 2014 defenses had five or more defensive backs out for more than 60% of plays.9 The linebacker they removed was usually the Sam, the strong-side linebacker built to take on tight ends and fullbacks, because the 11-personnel offenses that spread the field had one tight end and no fullback for him to take on. The nickel package, five defensive backs and two linebackers, became the real base defense of the NFL, as A First Look at Defense showed.

In 2016 base defense was on the field for 31% of snaps; by 2023 it was down to 22%. Then it came back: 31% in 2025, close to the 29.7% Next Gen Stats measured, because offenses returned to two-tight-end and fullback sets and defenses answered with bigger personnel.10 The Sam didn’t disappear; he became a part-time player.

The other half of the story is the linebackers who stayed. If two linebackers are going to be on the field for most snaps, and those two have to cover running backs and tight ends in space as well as fill gaps, teams want them faster, and faster usually means lighter. The right half of Figure 31.12 shows the trend at the NFL Combine. Off-ball linebackers in the 2000–2009 draft classes averaged 242 pounds and ran the 40-yard dash in 4.75 seconds; the classes since 2019 have averaged 233 pounds and 4.60 seconds.11 Nine or ten pounds and a tenth and a half of a second doesn’t sound like much. At the second level it is the difference between running with a running back and watching him.

The clearest sign of the shift is where some linebackers come from. In 2015 the Arizona Cardinals moved Deone Bucannon, a safety they had drafted in the first round a year earlier, to inside linebacker, at about 211 pounds. Their depth chart listed him as “$ILB,” and the coaches called him the moneybacker: a safety’s speed and coverage in a linebacker’s spot, playing the run by beating blockers to the hole rather than taking them on.12 Other teams followed with safety-sized linebackers and linebacker-sized safeties, until in some defenses the line between the two positions is mostly a matter of which word is on the depth chart. The big nickel of A First Look at Defense, with a third safety in the box, is the same idea from the other direction.

What the safety-sized linebacker gives up is the thing the Sam was built for: taking on a fullback or a pulling guard in the hole and holding his ground. A 215-pound linebacker who has to stand still and absorb a 315-pound guard loses. So the modern linebacker is coached to not stand still: to beat the blocker to the spot, or run through the gap before he arrives, the fits from earlier in this chapter. The best of today’s linebackers combine both. San Francisco’s Fred Warner, a third-round pick in 2018 and first-team All-Pro in 2020, 2022, 2023 and 2024, is about 6-foot-3 and 230 pounds: the coverage linebacker teams hoped the safety-sized linebacker would be, in a full-sized body. Watch him carry a tight end up the seam in his zone drops, the hook/curl seam carry of Figure 31.9, and then fill against the run.13 And when offenses go heavy, with two tight ends and a fullback, as many did again in 2025, defenses bring back a third linebacker, or a bigger one. The Sam isn’t extinct. He is a specialist who comes in when the offense does.

CautionCommon misconception

“Linebackers are the defense’s run-stoppers; the defensive backs cover.” In the modern NFL the two off-ball linebackers are on the field for most passing snaps and are in coverage on most of them, against running backs, tight ends and slot receivers. A linebacker who cannot cover cannot stay on the field, however well he tackles. That, more than anything, is what has changed about the position.

What does he actually do with it?

  1. He relays the call. The defensive coordinator calls the front, the coverage and any pressure into one helmet. The green dot passes it to the other ten, in a huddle or with hand signals, before the radio goes dead.
  2. He sets the strength. Every rule in the call depends on which side is “strong,” and the strength depends on what the offense shows: the tight end, the trips side, the running back, or the wide side of the field. The green dot decides and shouts it, and the front sets off his word, the way Even Fronts showed. The hard part is that many defenses set the front to the tight end (or the back) and the coverage to the receivers. In Figure 31.13 they agree; against trips with the tight end on the other side they don’t, and he has to call both: “Front left, coverage right.”
  3. He adjusts to what the offense does. When the offense shifts or sends a man in motion, the strength may change, and with it the front and the coverage rules. With the radio already off, the green dot is the one who re-sets everyone, in a second or two, with a word. (If the offense substitutes after the radio has cut out, a new call comes from the sideline by hand signal, and he passes it on.)
  4. He makes the alerts. Linebackers study opponents’ tendencies all week. The green dot is the one who sees a formation and yells “Watch the screen!” or “Boot! Boot!” before the snap, from a list built in film study.
  5. He plays a game with the protection. The offense’s center identifies a defender as the “Mike” for its blocking count, the Mike identification of Pass Protection. The defense knows it. Linebackers walk up, back out and swap spots before the snap to make that identification wrong.

It is a lot to ask of one player, and it is why teams prize linebackers who are, in the clichéd but accurate phrase, coaches on the field. It is also why the green dot has started to move. A linebacker who comes off the field in some packages takes the radio with him, and the backup green dot has to take over. More and more teams have given the radio to a safety or another defensive back who never leaves the field: six teams did in 2021, according to one count.15 The Mike’s job doesn’t vanish when he loses the dot; he still sets the front and the run fits in the box, while the safety with the radio runs the coverage.

TipWatch for it

Back to the huddle

The chapter opened with a middle linebacker who went the “wrong” way, away from the back’s jab step, and stopped the run for no gain, while the analyst circled the left guard. Now you can see what he saw. The guard nearest him fired out (a run), and the left guard opened his hips and ran flat behind the line (a pull, so a gap-scheme run going where he went). The back’s jab step was window dressing. The Mike went with the puller, had the shorter trip, and met him at the line, where the hole was supposed to be. The back never had a chance to choose.

Takeaways

  • Where a linebacker stands is a trade. Stacked behind a lineman, he is shielded and hard to read but a step slower to the gaps; open over an uncovered guard, he sees everything and is easy to block; walked out to an apex over a slot receiver, he can cover but he has left the box light and become the offense’s conflict defender.
  • Linebackers read keys, not the ball. The guards’ first steps (fire out, down block, pull, zone step, pass set) and the near back’s direction tell him run or pass and where, early enough to get there. The offense knows it, which is why counters, traps and play-action are built to make his keys lie.
  • Flow tells him how to move; the fit tells him where. Fast flow: run with it, inside-out, with the backside linebacker a step behind for the cutback. Slow flow: press the gap and stay square. The fit is a fill, a plug, a scrape or a run-through, and the scrape is also the defense’s answer to the zone read.
  • The second level is where runs are decided. Pursuit angles aim where the runner will be, a step inside him, with a leverage player outside; leverage tackling takes away one direction and wraps the legs. Inside linebackers make a fifth to a quarter of the tackles on runs up to 9 yards; misses rise with the space around the tackle, and a run that beats the linebackers is a defensive back’s tackle.
  • On passes, linebackers are hook/curl defenders, man defenders on backs and tight ends, robbers in the hole and blitzers. Play-action targets their keys; the uncovered guard who doesn’t climb is the tell.
  • The position has shrunk. Spread offenses made the Sam a part-time player, replaced by a nickel back, and made the remaining linebackers lighter and faster, down to safety-sized “moneybackers.” The green-dot linebacker, who sets the strength and adjusts the defense before every snap, is the job that hasn’t changed.
