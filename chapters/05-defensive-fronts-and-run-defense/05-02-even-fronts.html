OL_KEYS = ("LT", "LG", "C", "RG", "RT")


def pdf_build():
    return os.environ.get("RTG_FORMAT", "").lower() in ("pdf", "typst", "print")


def O(key, pos, d, w, label):
    return Player(key, pos, "off", d, w, label=label)


def D(key, pos, d, w, label):
    return Player(key, pos, "def", d, w, label=label)


def line(labels="full"):
    """The five offensive linemen, labelled LT..RT ("full") or with one letter, T G C G T ("short")."""
    short = {"LT": "T", "LG": "G", "C": "C", "RG": "G", "RT": "T"}
    return [p.moved(label=(p.key if labels == "full" else short[p.key])) for p in offense("singleback")
            if p.key in OL_KEYS]


def te(side=1, key=None, label=None):
    """An in-line tight end: Y on the right, U on the left."""
    key = key or ("Y" if side > 0 else "U")
    return O(key, "TE", -0.7, side * 4.8, label or key)


def T(t, side=1):
    """Width of technique t on the offense's right (side=+1) or left (side=-1)."""
    return technique(t, side)


def go_to(p, key, end, kind="block", delay=0.0, speed=None, via=()):
    """Move `key` through absolute (d, w) points `via` to `end`, as a block, path or run."""
    start = np.array(p._cur(key))
    pts = [tuple(np.array(q) - start) for q in (*via, end)]
    if kind == "block":
        return p.block(key, pts=pts, delay=delay, speed=speed)
    if kind == "pull":
        return p.pull(key, pts, delay=delay, speed=speed)
    return {"path": p.path, "run": p.run}[kind](key, pts, delay=delay, speed=speed)


def _P(fld, p, d, w):
    return fld.xy(*p.to_abs(d, w))


def say(fld, p, d, w, text, color=style.INK_2, fontsize=7.5, ha="center", weight="normal"):
    """A short on-field label at (d, w)."""
    x, y = _P(fld, p, d, w)
    fld.ax.text(x, y, text, fontsize=fontsize, color=color, ha=ha, va="center", zorder=9, fontweight=weight,
                bbox=dict(boxstyle="round,pad=0.18", fc=style.SURFACE, ec="none", alpha=0.9))


def point_to(fld, p, text_at, target, text, color=style.INK_2, fontsize=7, shrink=7):
    """A label with a thin leader line to the player or spot it names."""
    (xt, yt), (xp, yp) = _P(fld, p, *text_at), _P(fld, p, *target)
    fld.ax.annotate(text, (float(xp), float(yp)), xytext=(float(xt), float(yt)), fontsize=fontsize,
                    color=color, ha="center", va="center", zorder=9,
                    arrowprops=dict(arrowstyle="-", color=color, lw=0.7, shrinkA=2, shrinkB=shrink),
                    bbox=dict(boxstyle="round,pad=0.18", fc=style.SURFACE, ec="none", alpha=0.9))


def gap_tag(fld, p, d, w, text, color=GREEN, fontsize=8.5):
    """A gap letter in a small rounded box."""
    x, y = _P(fld, p, d, w)
    fld.ax.text(x, y, text, fontsize=fontsize, color=color, ha="center", va="center", zorder=9,
                fontweight="bold", bbox=dict(boxstyle="round,pad=0.2", fc=style.SURFACE, ec=color, lw=1.0))


def spot(fld, p, d, w, color=style.DEFENSE_DARK, ms=10):
    """Mark a point (a tackle, a meeting point) with an X."""
    x, y = _P(fld, p, d, w)
    fld.ax.plot([x], [y], marker="X", ms=ms, color=color, mec="white", mew=0.8, zorder=12)


def ring(fld, p, key, color=GREEN, r=1.25, lw=1.8, ls="-"):
    """A ring around a player at his alignment (drawn under the marker's label)."""
    q = p.players[key]
    x, y = _P(fld, p, q.d, q.w)
    fld.ax.add_patch(Circle((float(x), float(y)), r, fc="none", ec=color, lw=lw, ls=ls, zorder=7.5))


def lane(fld, p, w0, w1, d0, d1, color=style.OFFENSE, alpha=0.10, ls=(0, (3, 2))):
    """Shade a rectangular lane (a run lane, a gap) from width w0 to w1 and depth d0 to d1."""
    (x0, y0), (x1, y1) = _P(fld, p, d0, w0), _P(fld, p, d1, w1)
    fld.ax.add_patch(Rectangle((min(x0, x1), min(y0, y1)), abs(x1 - x0), abs(y1 - y0), fc=color, alpha=alpha,
                               ec=color, lw=0.9, ls=ls, zorder=0.8))


def dashed(fld, p, pts, color=style.THIRD, lw=1.5, dash=(0, (3, 2.2)), end="arrow"):
    """A dashed 'what if' or 'intent' path through absolute (d, w) points."""
    xs, ys = _P(fld, p, np.array([q[0] for q in pts]), np.array([q[1] for q in pts]))
    xs, ys = np.asarray(xs, float), np.asarray(ys, float)
    if end == "arrow":   # dashed all the way to a short solid head on the last 25% of the last leg
        hx, hy = xs[-2] + 0.75 * (xs[-1] - xs[-2]), ys[-2] + 0.75 * (ys[-1] - ys[-2])
        xs = np.concatenate([xs[:-1], [hx, xs[-1]]]); ys = np.concatenate([ys[:-1], [hy, ys[-1]]])
        fld.ax.plot(xs[:-1], ys[:-1], color=color, lw=lw, ls=dash, zorder=6)
        fld.ax.add_patch(FancyArrowPatch((xs[-2], ys[-2]), (xs[-1], ys[-1]), arrowstyle="-|>",
                                         mutation_scale=10, color=color, lw=lw, zorder=6))
    else:
        fld.ax.plot(xs, ys, color=color, lw=lw, ls=dash, zorder=6)


def ghost(fld, p, d, w, label, side="def"):
    """A faint, dotted outline where a player USED to be (before a shift)."""
    x, y = _P(fld, p, d, w)
    x, y = float(x), float(y)
    if side == "def":
        fld.ax.add_patch(RegularPolygon((x, y), 4, radius=0.75 * 1.25, orientation=np.pi / 4, fc="none",
                                        ec=style.DEFENSE, lw=1.0, ls=(0, (1.5, 1.2)), zorder=7))
        fld.ax.text(x, y, label, ha="center", va="center", fontsize=6.5, color=style.MUTED, zorder=7)
    else:
        fld.ax.add_patch(Circle((x, y), 0.75, fc="none", ec=style.OFFENSE, lw=1.0, ls=(0, (1.5, 1.2)), zorder=7))


def bracket(fld, p, d, w0, w1, color=style.DEFENSE_DARK, lw=1.6):
    """A bar spanning the gap(s) a player owns, with end ticks."""
    (xa, ya), (xb, yb) = _P(fld, p, d, w0), _P(fld, p, d, w1)
    fld.ax.plot([xa, xb], [ya, yb], color=color, lw=lw, solid_capstyle="butt", zorder=5)
    for w in (w0, w1):
        (x0, y0), (x1, y1) = _P(fld, p, d - 0.35, w), _P(fld, p, d + 0.35, w)
        fld.ax.plot([x0, x1], [y0, y1], color=color, lw=lw, zorder=5)


def clean(ax):
    """Box-zoomed diagrams: drop hash ticks and yard numbers, clip anything outside the window."""
    for ln in list(ax.lines):
        xs, ys = ln.get_data()
        if len(xs) == 2 and ln.get_linewidth() == 0.5 and np.hypot(xs[1] - xs[0], ys[1] - ys[0]) < 1.0:
            ln.remove()
    for t in list(ax.texts):
        if t.get_text().isdigit() and t.get_alpha() == 0.55:   # yard numbers, not technique labels
            t.remove()
        else:
            t.set_clip_on(True)

def over_front(s=1, sam=True):
    """4-3 Over to side s: weak 5, 1 shaded weak, 3 strong, strong 5; Sam off the ball outside the TE,
    Mike over the strong A gap, Will over the weak (uncovered) guard."""
    out = [D("WE", "DE", 1.0, T("5", -s), "5"), D("NT", "DT", 1.0, T("1", -s), "1"),
           D("DT", "DT", 1.0, T("3", s), "3"), D("SE", "DE", 1.0, T("5", s), "5"),
           D("MIKE", "LB", 4.5, s * 0.8, "M"), D("WILL", "LB", 4.5, -s * 2.0, "W")]
    if sam:
        out.append(D("SAM", "LB", 4.2, s * 6.2, "S"))
    return out


def under_front(s=1, sam=True):
    """4-3 Under to side s: Sam on the line outside the TE (a 9), strong 5, nose shaded strong (1),
    3-technique weak, weak end in a loose 5 ("5w", a yard outside a normal 5); Mike over the strong (uncovered) guard, Will over the weak A."""
    out = [D("WE", "DE", 1.0, T("5", -s) - s * 1.0, "5w"), D("DT", "DT", 1.0, T("3", -s), "3"),
           D("NT", "DT", 1.0, T("1", s), "1"), D("SE", "DE", 1.0, T("5", s), "5"),
           D("MIKE", "LB", 4.5, s * 1.9, "M"), D("WILL", "LB", 4.5, -s * 1.1, "W")]
    if sam:
        out.append(D("SAM", "LB", 1.0, T("9", s), "S"))
    return out

MARK_R = 0.62


def tracking(play):
    df = play.to_tracking()
    df["_id"] = df["displayName"].astype(str)
    return df


def panel(ax, play, df, t, lateral, window, rings=(), enter=(), ring_r=0.85):
    """One moment of a simulated play on `ax`: markers, trails of on-screen players, rings."""
    tmap = df.groupby("frameId")["time"].first()
    fid = int((tmap - t).abs().idxmin())
    ylo, yhi = sorted((play.ball_y - lateral[0], play.ball_y - lateral[1]))
    fld = Field(ax=ax, orient="vertical", los=play.los, window=window, numbers=False, hashes=False,
                lateral=(ylo, yhi), to_go=play.to_go)
    cur = df[df["frameId"] == fid]
    snap = df[df["frameId"] == int((tmap - 0.0).abs().idxmin())]
    first = snap[snap["x"].between(play.los + window[0] + 0.8, play.los + window[1] - 0.8)
                 & snap["y"].between(ylo + 0.8, yhi - 0.8)]
    ids = set(first["_id"]) | {"football"}
    inside = cur[cur["x"].between(play.los + window[0], play.los + window[1]) & cur["y"].between(ylo, yhi)]
    ids |= set(inside["_id"]) & set(enter)
    shown = cur[cur["_id"].isin(ids) | (cur["side"] == "ball")]
    trails = df[(df["frameId"] <= fid) & df["_id"].isin(ids) & (df["side"] != "ball")]
    for (_, side), g in trails.groupby(["_id", "side"]):
        xs, ys = fld.xy(g["x"].values, g["y"].values)
        ax.plot(xs, ys, color=style.OFFENSE if side == "off" else style.DEFENSE, lw=0.9, alpha=0.4, zorder=3)
    ppy = fld.pts_per_yard()
    for _, r_ in shown.iterrows():
        x, y = (float(v) for v in fld.xy(r_["x"], r_["y"]))
        if r_["side"] == "ball":
            ax.add_patch(Ellipse((x, y + 0.45), 0.7, 0.42, fc=style.BALL, ec="white", lw=0.5, zorder=9))
            continue
        lab = str(r_.get("label", "") or "")
        if r_["side"] == "off":
            ax.add_patch(Circle((x, y), MARK_R, fc=style.OFFENSE, ec="white", lw=0.6, zorder=8))
            tc = "white"
        else:
            ax.add_patch(RegularPolygon((x, y), 4, radius=MARK_R * 1.3, orientation=np.pi / 4,
                                        fc=style.DEFENSE_TINT, ec=style.DEFENSE, lw=1.0, zorder=8))
            tc = style.INK
        fs = max(4.0, min(8.0, ppy * (0.85 if len(lab) <= 1 else 0.62)))
        ax.text(x, y, lab, ha="center", va="center", fontsize=fs, color=tc, fontweight="bold", zorder=10)
    for key, col in rings:
        r_ = cur[cur["displayName"] == key].iloc[0]
        x, y = fld.xy(r_["x"], r_["y"])
        ax.add_patch(Circle((float(x), float(y)), ring_r, fc="none", ec=col, lw=1.6, zorder=7))
    for patch in ax.patches:
        patch.set_clip_on(True)
    return fld


def strip(moments, lateral, window, notes=None, rings=(), ncols=2, width_in=5.2, after=None, enter=()):
    """Numbered frame strip. `moments` is a list of (play, t, tag)."""
    cache = {}
    nrows = int(np.ceil(len(moments) / ncols))
    pw = width_in / ncols
    span_w, span_d = lateral[1] - lateral[0], window[1] - window[0]
    fig, axes = plt.subplots(nrows, ncols, squeeze=False,
                             figsize=(width_in, nrows * (pw * span_d / span_w + (0.62 if notes else 0.25))))
    for ax in axes.flat[len(moments):]:
        ax.axis("off")
    for i, (ax, (play, t, tag)) in enumerate(zip(axes.flat, moments)):
        if id(play) not in cache:
            cache[id(play)] = tracking(play)
        fld = panel(ax, play, cache[id(play)], t, lateral, window, rings, enter)
        if after:
            after(fld, play, t, i)
        lab = "snap" if abs(t) < 1e-6 else f"{t:+.2f} s".replace("0 s", " s")
        ax.set_title(f"{i + 1}.  {lab}" + (f"   {tag}" if tag else ""), fontsize=8.3, color=style.INK_2, loc="left")
        if notes:
            ax.text(0.5, -0.03, notes[i], transform=ax.transAxes, ha="center", va="top",
                    fontsize=7.1, color=style.INK)
    fig.tight_layout(h_pad=1.3 if notes else 0.6, w_pad=0.6)
    return fig


def video(segments, lateral, window, rings=(), width_in=4.6, after=None, enter=()):
    """Web version: each (play, tag, [(t, note), ...]) segment played in turn, notes as captions."""
    import shutil
    from matplotlib import animation
    from IPython.display import HTML, display
    span_w, span_d = lateral[1] - lateral[0], window[1] - window[0]
    fig, ax = plt.subplots(figsize=(width_in, width_in * span_d / span_w + 0.45))
    frames = []
    for play, tag, notes in segments:
        df = tracking(play)
        tmap = df.groupby("frameId")["time"].first()
        frames += [(play, df, float(tmap[f]), tag, notes) for f in sorted(df["frameId"].unique())]

    def update(k):
        play, df, t, tag, notes = frames[k]
        ax.clear()
        fld = panel(ax, play, df, t, lateral, window, rings, enter)
        if after:
            after(fld, play, t, -1)
        txt = ("snap" if abs(t) < 0.05 else f"{t:+.1f} s") + (f"  {tag}" if tag else "")
        cur = [n for tt, n in notes if tt <= t + 1e-6]
        ax.set_title(txt + (("\n" + cur[-1].replace("\n", " ")) if cur else ""), fontsize=8.5, color=style.INK,
                     loc="left")
        return []

    anim = animation.FuncAnimation(fig, update, frames=range(len(frames)), interval=100)
    html = (anim.to_html5_video(embed_limit=40).replace("<video ", '<video style="max-width:100%;height:auto" ')
            if shutil.which("ffmpeg") else anim.to_jshtml(default_mode="once"))
    plt.close(fig)
    display(HTML(html))


def animation_or_strip(moments, notes, lateral, window, rings=(), after=None, enter=(), ncols=2, width_in=5.2):
    """Video on the website; a numbered, captioned frame strip in the PDF."""
    if pdf_build():
        strip(moments, lateral, window, notes=notes, rings=rings, ncols=ncols, width_in=width_in,
              after=after, enter=enter)
        plt.show()
        return
    segs = []
    for play, t, tag in moments:
        if not segs or segs[-1][0] is not play:
            segs.append((play, tag, []))
    for (play, t, tag), n in zip(moments, notes):
        for s in segs:
            if s[0] is play:
                s[2].append((t, n))
    video(segs, lateral, window, rings=rings, after=after, enter=enter)

First-and-10. The offense breaks the huddle with its tight end on the right. Before the quarterback reaches the line, the defensive tackle who had drifted up over the right guard while the offense huddled jogs across to the left guard, the other tackle slides the opposite way, and a linebacker who had been off the ball walks up and plants himself on the line just outside the tight end, hand on his knees. The middle linebacker is pointing and shouting one word. Two plays later the tight end lines up on the left, and this time the tackles don’t trade places at all, and the linebacker outside the tight end stays five yards back. Nobody on the broadcast mentions it. The defense has just played two different fronts, chosen by one word and the position of one offensive player, and each was a bet about where the run would go. This chapter teaches you to read that bet.

You’ll be able to:

  • Draw a 4-3 Over and a 4-3 Under from the offense’s strength, and say which guard each leaves uncovered, who owns each gap, and which runs each front is built to stop.
  • Explain the wide-9 front’s trade, more pass rush for bigger running lanes inside the ends, and who on the defense has to pay for it.
  • Diagram the 46 and its bear front, and explain why covering the center and both guards breaks so many blocking schemes, and why offenses made it disappear.
  • Show how an even front changes when the offense swaps a fullback for a third wide receiver (21 to 11 personnel), and what stays the same.
  • Name the front from the broadcast angle before the snap, using two landmarks: the tight end and the 3-technique.
NoteYou’ll need

  • From earlier parts: formation strength (the side the defense calls “strong,” usually the tight end’s), covered and uncovered linemen and the A-to-D gaps from The Language of the Run Game, and the 4-3 and nickel packages from A First Look at Defense.

How to read the diagrams. Offense is blue circles, defense orange squares, and plays go up the page. Left and right always mean the offense’s left and right. LT, LG, C, RG and RT are the offensive linemen (T, G and C in the smallest pictures), Y and U are tight ends, Q the quarterback, R the running back and F a fullback. Defensive linemen are labelled with their technique number: 5w is a loose 5, a yard or so wider than a normal 5 on the tackle’s outside shoulder, and w9 a wide-9. M, W and S are the Mike, Will and Sam linebackers; SS and FS are the safeties and $ a nickel back. Green letters are gaps, never players. Unless a picture says otherwise, the tight end, and so the offense’s strength, is on the right.

An even front sets its down linemen in pairs, two on each side of the ball, with nobody head-up on the center. In practice that means a four-man line, the family of fronts that grew up around the 4-3. An odd front puts a man on the center (the nose of a 3-4, or the middle man of a five-man line) and is the subject of Odd and Hybrid Fronts.

Why does the center matter so much? Because of what you learned about covered and uncovered linemen in 03-01. An offensive lineman with nobody on him, head-up or on a shoulder, is the offense’s spare part: he can double-team a neighbor or climb to a linebacker. In an odd front the nose covers the center from the first instant, and the spare parts are the guards. An even front has only two defensive tackles for the three interior offensive linemen, the center and the two guards, so one of those three is always uncovered. Which one is the even front’s first decision. In the balanced front of Figure 30.1 it is the center. In the two fronts this chapter is built around, a tackle sits on one of the center’s shoulders, which covers him, and the free man is a guard instead. Every even front is a decision about which lineman to leave free, and where to put the linebacker who has to deal with him.

The perfectly balanced version in Figure 30.1 (a 3-technique on each guard, an end on each tackle) exists, and you will see it on passing downs, when it gives each of the two best interior rushers a guard to himself. On early downs, most even fronts are unbalanced on purpose. One tackle slides in to a 1-technique, shaded on a shoulder of the center, and the other stays a 3-technique, outside a guard. Now the line leans: the 3-technique’s side has its linemen spread wide, the 1-technique’s side has them pinched in toward the ball, and the guard between the 1 and the end on that side has nobody on him. Which way the line leans, toward the offense’s strength or away from it, decides which guard that is, and it is the difference between the two fronts this chapter is built around.

CautionCommon misconception

“Even and odd just mean four linemen or three.” That’s the fan’s shortcut and it usually works. But the coaches’ test is the center and the pairs, and the two counts can disagree: the bear front later in this chapter has four men with their hands down and a nose head-up on the center, and a 3-4 team can put a fourth player’s hand on the ground without changing who covers whom. When the counts argue, believe the alignment. (The bear is in this chapter anyway, for history rather than taxonomy: it grew out of a 4-3 team’s playbook. By the alignment test it is an odd front.)

Setting the front: the strength call

A front leans toward or away from something, so before it can line up, the defense needs a word for which side is which. That word is the strength, the side the defense calls strong, from Formation Rules and Vocabulary. For fronts, the usual rule is simple: the strength is the tight end’s side. The tight end is a sixth blocker attached to the line, so his side has one more gap (the D gap outside him) and one more man to block with. Runs go there more often, and the front wants to know about it.

When there is no tight end, or one on each side, defenses fall back on other rules. Common ones are the field (the wide side of the field when the ball is on a hash), the running back (the side he is offset to in the shotgun), or the side with more receivers. Which rule a team uses, and in what order, is in its playbook, and the call before the snap says which one applies. Two tight ends on the same side make that side strong twice over. A tight end who is detached, split a few yards off the line or in the slot, usually stops counting: he is no longer a blocker attached to the tackle, so many defenses treat him as a receiver and set the front by their next rule.

That last case is why many defenses keep two strengths. The run strength, usually the tight end’s side, sets the front: where the linemen and linebackers go. The pass strength, the side with more receivers, sets the coverage. Against a tight end on the right and three receivers on the left, the front can lean one way and the coverage the other. In this chapter “strength” means the run strength unless it says otherwise.

The call is relative, and that is the whole point. The defense does not call “3-technique on the left guard.” It calls something like “Over” and lets the formation decide which guard: Over means the line leans toward the strength, Under means it leans away from it.1 A Mike linebacker shouting “Strong right! Over!” is translating one word into eleven alignments for this particular snap.

Start with the left half of Figure 29.2, the 4-3 Over. On the strong side (the tight end’s), the defense puts its two most active linemen: the 3-technique on the strong guard’s outside shoulder and the strong end on the tackle, here in a 5. (Some teams set him wider, on the tight end: a 7 on his inside shoulder or a 6 head-up on him, so the tight end can’t release inside cleanly or block down on the 3-technique.) On the weak side it puts the 1-technique, shaded on the center’s weak shoulder, and the weak end in a 5. The linebackers fill in around them. The Mike stacks over the strong A gap, standing directly behind the line a few yards deep (the stack alignment of linebacker play); the Will stands over the weak guard; the Sam is off the ball, a few yards outside the tight end.

Now count the gaps, the way 05-01 taught. Seven gaps, seven men, one each: the four linemen take the gaps they are shaded into (weak C, weak A, strong B, strong C), and the three linebackers take the three left over (weak B, strong A, and the D gap outside the tight end). This is one-gap football. Every man knows his gap before the snap and can play it at full speed. That includes the 1-technique: he owns the weak A gap, not the center, and when the center and the free guard double-team him, his job is the one 05-01 drew: sink his hips, give no ground and keep both blockers on him, so neither can leave cleanly for a linebacker. (Some Under teams ask their nose to two-gap instead; the four-man fronts in this chapter are drawn the one-gap way.)

Why lean toward the tight end? Because that’s where the offense has the most to work with. The strong side has three blockers on the line (guard, tackle, tight end) and four gaps, and the classic power runs, the off-tackle plays of Gap and Power Runs, go there. The Over meets them with linemen. Its strong-side B and C gaps belong to big men with their hands in the ground, not to linebackers who have to read and then come downhill, so the runs the offense likes best run into the defense’s weight. And the Sam, off the ball, is free to do two jobs: fill the D gap on a run, or turn and cover the tight end on a pass. Play-action to the tight end is one of the oldest plays in football; an off-ball Sam can see it.

What does it give up? Look at the green ring. The weak guard is uncovered, and an uncovered guard on the side away from the tight end is a lineman with a clean path to the Will linebacker. That empty patch of the line, with no defensive lineman over a blocker, is what coaches call a bubble (no relation to the bubble screen of the passing game: same word, different thing), and offenses aim runs at bubbles: the guard can help double the 1-technique and then climb to the Will, or go straight to him. Over teams live with it by asking a lot of the Will, the run-and-hit linebacker who has to beat a 300-pound guard to his gap. There is a second, subtler cost: the 3-technique is on the side with the most offensive blockers, so it’s the side where he is easiest to double-team.

The 4-3 Under

Flip the lean and you get the right half of Figure 29.2, the 4-3 Under. The 3-technique moves to the weak guard. The nose slides to a 1-technique on the center’s strong shoulder. The strong end stays in a 5 on the tackle, and the weak end, with no tight end to worry about, widens into a loose 5 (labelled 5w), a yard or so outside the tackle’s shoulder instead of on it. That is a small trade: it makes the weak C gap a little wider for the run, but the end sees the whole backfield from there, has a better angle to rush the passer, and is in position to chase down the runs, bootlegs and reverses that come back his way. Then the linebackers shift the other way: the Sam walks up onto the line, outside the tight end, and the Mike moves over the strong guard.

Count the gaps again. On the strong side, the Sam owns the D gap and the end the C. The Mike, stacked over the uncovered strong guard, owns the B gap, and the nose owns the strong A. On the weak side, the Will owns the A, the 3-technique the B and the weak end the C. Seven for seven, just distributed differently.

Why lean away? Three reasons, and each one is a bet.

  1. A hard strong-side edge. With the Sam on the line and a big end in the 5, the tight end’s side has two defenders on the line of scrimmage outside the tackle. The off-tackle run has to block both, and the edge they set turns everything back inside. Seattle’s defensive coordinator in 2011, Gus Bradley, put the whole philosophy in two sentences: “We want to set the edge. Turn everything back inside to our big guys.”2
  2. A one-on-one 3-technique. On the weak side there is no tight end, so the offense has fewer ways to double-team the 3-technique without leaving someone else free. The Under puts its best interior pass rusher where he is likeliest to get a guard alone. Monte Kiffin’s Tampa Bay defense had a name for him, the under tackle, after the side he played, and built its pass rush around Warren Sapp in that spot.3
  3. The bubble in front of the right linebacker. Every four-man front leaves one guard free, so some linebacker has to take him on. The Over asks the Will, usually the smallest and fastest of the three. The Under puts the free guard in front of the Mike, usually the biggest and best at taking on a blocker, and in exchange the Will plays behind a weak side where every lineman is covered, free to run to the ball. The Under doesn’t hide the bubble; it chooses who faces it.

Put the Sam on the line, lean the line away from him, and the Under starts to look like something else. Squint at the right half of Figure 29.2: five men on the line, a nose on the center’s shoulder, two big ends, a stand-up player on the edge, two inside linebackers. That is close to a 3-4 with one outside linebacker’s hand on the ground, which is why Pete Carroll has described his Seattle fronts as a 4-3 with 3-4 personnel.4 The labels in this part of football blur at the edges; the alignment is what’s real.

What does it give up? The bubble is now on the strong side, over the right guard, in front of the Mike. Runs at that bubble, inside zone and duo (03-03’s gap run with two double teams and no puller) toward the tight end, put the uncovered guard on the Mike with a running start, and if the Mike loses, the hole is on the side with the most blockers. And the Sam on the line is a Sam who can’t easily drop into coverage: if he rushes or sets the edge, someone else has to take the tight end on play-action, usually a safety.

ImportantWhy it exists

Why carry both? An offense’s blocking scheme is built on a guess about where the defensive linemen will be. A defense that can lean either way from the same eleven players, without substituting anyone, makes that guess wrong some of the time, and a run designed against an Over that meets an Under is aimed at a 3-technique instead of a bubble. That is why most 4-3 teams carry both, and why the call often comes late.

NoteGo deeper

The dialects. Not every staff means the same thing by “Under.” For some, it is enough that the line leans away from the tight end: Monte Kiffin’s Tampa Bay defenses, and Lovie Smith’s Bears after them, set the under tackle on the weak side and the nose on the strong side of the center but usually kept the Sam off the ball, and some coaches call that an Under too. Others reserve “Under” for the version in Figure 29.2, with the Sam on the line. Some teams set their ends by side (left and right) rather than by strength, so the ends never move and only the tackles and linebackers shift. The strong end in an Over may play a 5, a 7 or a 6, or even a 9 outside the tight end: 05-01’s gap diagram drew the Over with a 9 end owning the D gap and the Sam inside him owning C. That is the same front with the end and the Sam trading gaps, not a contradiction. Many staffs also call direction with paired code words that start with R and L (Rip and Liz, Lion and Larry), which is what you are hearing on a mic’d-up linebacker. When a coach draws “his” Over or Under on a whiteboard, read the alignments, not the word.

Fronts come with coverages. The four men up front don’t tell you what is behind them, but some pairings became famous. Kiffin’s and Smith’s four-man fronts played Cover 2, two deep safeties with five underneath, and its Tampa 2 variation, in which the Mike runs down the deep middle. Carroll’s Under played Cover 3, one deep safety and a strong safety down near the box. Part 6 builds both; for now, notice that the same four linemen can sit in front of very different defenses.

Figure 29.4 is why tight-end shifts and motions became so common (Motion and Shifts has the offense’s side). One player jogs a few yards; if the defense answers by flipping its Over (top right), five defenders have to re-set, and they have about a second and a half to do it. Several things can go wrong. A tackle who is still moving at the snap is a tackle who hasn’t got his stance, and a one-gap lineman without a stance has lost his head start. Two linebackers who don’t hear the new call fill the same gap. And the offense learns something either way: if the 3-technique crosses the center every time the tight end moves, the offense now knows his rule and can put him wherever it wants him, for instance on the side away from the play it is about to run.

Defenses have answers, and each one is a trade. The simplest leaves the line alone. An Over to the right and an Under to the left put the four linemen in exactly the same spots, so when the tight end crosses, the call becomes “Under”, the Sam follows the tight end and walks up outside him, and nothing crosses the center (Figure 29.4, bottom). Defensive line coaches like this answer because nobody has to get into a new stance at the last second. Its price is that the defense no longer chooses its front: the shift chose it, and the offense knows that every shift to the left turns this Over into an Under. An Under team has the mirror problem: its answer to a shift is to check to Over, and now it is the Sam who has to come off the line and cross. Other defenses set the line by the field or by the running back instead, so a tight-end shift doesn’t move anything. And some use a check, a one-word call that changes the plan at the line, to tell everyone to ignore the shift for this snap and play the front as it stands, leaning the “wrong” way. Every one of those answers gives up some of the precision the front was built on, which is the point of shifting in the first place.

The offense’s line has to read the new picture too. Before every snap the center (or the quarterback) names the linebacker the blocking rules will count from, the Mike identification of Pass Protection, and a front that changes late can leave five blockers counting from a picture that no longer exists. Shifts cut both ways.

NoteMadden vs. real life

In Madden you pick “4-3 Over” from a menu and the line appears, already leaning the right way, every time. On a real field the front is a rule, not a picture. The call says “Over” and the defenders work out where that means for this formation, then work it out again when the tight end moves, often while the offense is still shifting. A defense whose front is a step late re-setting is a defense the offense has already beaten before the ball is snapped.

def corridor(f, p, pts, width=1.0, color=style.DEFENSE, alpha=0.15):
    """A shaded band of `width` yards along the polyline pts (absolute (d, w))."""
    pts = np.array(pts, float)
    left, right = [], []
    for i in range(len(pts)):
        a, b = pts[max(i - 1, 0)], pts[min(i + 1, len(pts) - 1)]
        v = b - a
        n = np.array([-v[1], v[0]]) / np.linalg.norm(v) * width / 2
        left.append(pts[i] + n); right.append(pts[i] - n)
    poly = np.vstack([left, right[::-1]])

Why play it? Pass rush, first and last. From a 5, an end has a blocker on him at the snap and has to beat that man before he goes anywhere. From a wide-9 nobody is on him: the tackle has to kick slide a long way out to meet him, and the end is at full speed by the time they touch. His path is longer in yards but often quicker in time, and it arrives on an angle that turns the tackle’s shoulders. With two of them, the offense has to decide on every pass which side gets the tight end’s or the back’s help, and the other side is one-on-one in space. (Jim Washburn’s linemen, 05-01’s wide-9 story, were told simply to attack.)

What does it cost? Look at the blue shading in Figure 29.5. With the ends that far out, the space between each end and the nearest tackle, the off-tackle lane, is five or six yards from one defender to the other, two gaps wide on each side, and on a run the ends are going upfield, not closing it. Somebody else has to: on the tight end’s side the Sam, on the other side the Will, and since one linebacker can’t own two gaps, usually a safety coming down behind them. The weak side is the new problem a second wide-9 creates. In a normal Over the weak end in a 5 owns the C gap and the Will only has to worry about the bubble; with that end gone wide, the Will’s side has a bubble and an empty C gap. The wide-9 moves run defense from the defensive line to the linebackers. A wide-9 team with linebackers who are late, or a 3-technique who gets moved off the ball and widens the lane from the inside, gives up the long runs that gave the front its reputation. Ends who rush upfield on every snap also invite the two plays built to punish exactly that: the draw, which shows pass and hands off behind them, and the running-back screen, which lets them run past and throws over their heads. And because a wide-9 end is the easiest defender on the field to leave unblocked, he is a natural target for the zone read and RPOs, which leave him alone on purpose and make him choose.

That is why few teams line up in a wide-9 on every snap. Most move their ends in and out by down, distance and formation: wide on passing downs and against spread sets, tighter against heavy personnel. On obvious passing downs a wide-9 team will even slide an end inside, over a guard, to rush against a smaller or slower blocker; Brandon Graham’s strip-sack of Tom Brady late in Super Bowl LII came on a series when Philadelphia had moved him from end to tackle.5

So which is it, a pass-rush front that can’t stop the run, or something else? CautionCommon misconception

“The wide-9 can’t stop the run.” It gives the run a bigger target, and it makes the players behind the ends more important. The 46 and the bear front

The even fronts so far lean, or stretch, a four-man line. The last one in this chapter crushes it together, and for a few seasons in the 1980s it was the most feared defense in football.

Buddy Ryan’s defense

Buddy Ryan was a defensive line coach who had won a Super Bowl with the 1968 Jets and coached Minnesota’s line after that before the Chicago Bears made him their defensive coordinator in 1978. A Korean War veteran from rural Oklahoma, he was blunt, combative and loyal to his players, and he feuded openly with his own head coach. His problem in Chicago was the problem every defense of that era had: offenses ran the ball out of two-back formations, the I-formation and the split backs, with a fullback leading the way and guards pulling, and they ran it well. His answer, worked out over several seasons and in full bloom by 1981, was to put more players near the ball than the offense could block.9

He named it after a player. Doug Plank was a hard-hitting safety whom Ryan moved down from his deep spot to play, in effect, as a linebacker. Ryan “circled my number three times,” Plank recalled, “and said, ‘we’re going to call this the 46 defense.’”10 So the 46 defense is not a count of anything: it is Plank’s jersey number. (Counted the usual way, it has four men with their hands down, three linebackers and a safety playing a fourth, and three defensive backs behind them.) Plank retired after the 1982 season, three years before the defense became famous.

Read Figure 29.6 from the middle out. The heart of it is the line: a nose head-up on the center and a 3-technique on each guard. That interior, three men covering the center and both guards, is called a bear front, after the team that made it famous.11 Outside it, the weak end lines up a yard or two outside the weak tackle. Then Ryan did something no 4-3 did: he moved both outside linebackers to the tight end’s side, one on the tight end’s inside shoulder to jam his release and one outside him. The Mike stands over the strong tackle, and the strong safety comes down to linebacker depth over the weak tackle.12 The names stay with the players, not the spots: the Will is still the Will after Ryan moves him to the tight end’s side, and the strong safety is still the strong safety standing over the weak tackle. Count them: eight defenders in the box, against seven blockers. Behind them, the two cornerbacks press the wide receivers (line up within a yard of them and jam them at the snap) in man coverage, and a single safety plays deep. And on many snaps five, six or seven of those eight came after the quarterback; whichever linebackers and safety did not rush took the tight end and the backs man-to-man, and which ones rushed changed from snap to snap.

By the alignment test from the start of this chapter, the bear is an odd front: it has a nose head-up on the center. It is here for history, not taxonomy. It was built by a 4-3 team out of 4-3 personnel, and it is the logical end of the even front’s question, which interior lineman do we leave free?, answered with “none.”

Why covering the guards breaks the blocking

Every run scheme in Part 3 is easiest with at least one interior lineman who isn’t covered. Zone needs him to start a double team and climb. Power, counter and the trap need him to pull. The iso needs the guards to drive two defensive linemen apart while a fullback hits the linebacker. The bear front takes all of that away at once.

Work through what the bear does to each kind of play.

  • No easy double teams. With the center and both guards covered, nobody inside can help a neighbor without leaving his own man free. Every interior block is one-on-one, against three linemen chosen precisely because they win one-on-one. The double team that moves a defensive tackle off the ball, the foundation of inside zone and power alike, never forms.
  • No climbers. The linebackers behind a four-man front survive by beating a guard who climbs to them. In the bear, the guards can’t climb: a guard who leaves his 3-technique has let a penetrator into the backfield. So the linebackers, and the safety playing linebacker, run to the ball with no one in their way. An offense with a fullback can send him at one of them, but the Mike and the strong safety are both free to run, so the fullback can take only one of them, and the two outside linebackers on the tight end’s side still need the tackle and the tight end.
  • Hard pulls. Pull a guard on power or a trap, and the 3-technique he was covering has a free run into the backfield before the play has started. The plays that beat aggressive one-gap fronts by letting defenders through and then hitting them from the side need a lineman who can leave. Against the bear the only way to free a guard is for the tackle to block down on the 3-technique, a long, hard block from the side, which in turn frees whoever the tackle was supposed to block.
  • Hard pass protection. On a pass, the three interior rushers each have a blocker of their own, the center can’t slide to help, and the outside linebackers stacked on the tight end’s side create more rushers on that side than the offense has blockers. Ryan’s defense then decided, every snap, which of the eight to send.

The 1985 Bears

The result was one of the great defensive seasons in NFL history. The 1985 Bears went 15–1 and allowed 198 points, the fewest in the league; they also led the league in yards allowed and takeaways. In the playoffs they shut out the Giants 21–0 and the Rams 24–0, and then beat the New England Patriots 46–10 in Super Bowl XX on January 26, 1986.13 New England ran the ball 11 times in that game for 7 yards, still the Super Bowl record for fewest rushing yards, and the Bears sacked New England’s quarterbacks seven times, tied for the most in any Super Bowl; the defensive end Richard Dent was the game’s MVP.14 After the game, Ryan’s players carried him off the field on their shoulders, alongside the head coach, Mike Ditka, with whom he had feuded openly.

Why it disappeared

The 46’s one loss in 1985 told the rest of the league how to beat it. On December 2, on Monday Night Football, the Miami Dolphins beat the Bears 38–24. Miami’s quarterback, Dan Marino, had one of the quickest releases in the game, and the Dolphins’ receivers won the one-on-one matchups the 46’s man coverage gave them. An eight-man front with three defensive backs behind it is built for a two-back, two-receiver offense. Against three receivers and a quarterback who throws before the rush arrives, its strength becomes its problem.

Offenses spent the next decade doing to the 46 what Figure 29.8 shows. Three-receiver sets pulled a linebacker or the safety out of the box or left a slow man covering a fast one. Quick, timed throws got the ball out before the extra rushers arrived. Runs went outside the bear instead of into it: tosses and stretch plays, and runs to the weak side, away from the linebackers stacked on the tight end. And as the run-and-shoot and the West Coast offense spread through the league (The 1978 Liberation tells that story), the two-back offense the 46 was designed to strangle became the exception. The full 46 is now a curiosity, but its pieces never left. Ryan coached it again in Philadelphia, as head coach from 1986 to 1990, and in Houston and Arizona; his sons Rex and Rob carried versions of it into the 2000s.

The bear front itself is the part that survives. Near the goal line, against two tight ends or a fullback, defenses still cover the center and both guards to take away the interior double teams. It has also come back on ordinary early downs as an answer to zone-running offenses, whose double teams and climbing linemen are exactly what the bear takes away. By PFF’s charting its use dipped from 2016 to 2018 and then came back: in the three seasons before 2021 it was the call on roughly 12% of early-down snaps between the 20-yard lines, usually with a one-high coverage behind it.15 Fangio’s Two-High Revolution and Macdonald’s Answer picks up that story. And the idea behind it, decide which interior linemen to leave uncovered, is alive in the tite front that NFL defenses use against inside runs today: the tite covers the center and the tackles’ inside shoulders, the bear covers the center and the guards, and both are answers to the same question.

From two backs to three receivers: the even front in nickel

The fronts in this chapter grew up against offenses with two backs and one tight end, 21 personnel. That offense has nearly disappeared: in 2025, 21 personnel was about 7% of NFL snaps, while 11 personnel, one back, one tight end and three wide receivers, was about 59%.16 The four-man even front didn’t disappear with it. Figure 29.9 shows the pattern most even-front teams follow.

The line is set by the run strength, and it stays. The tight end hasn’t moved, so the front’s rule hasn’t changed: the Under still leans away from him, the 3-technique still on the weak guard. The four linemen are the same players, and the three inside them are in the same spots; only the open-side end may loosen. That’s the great practical virtue of a four-man even front. It is the same from base to nickel, so the defensive line doesn’t need a second set of rules.

The third linebacker is the part that changes. Against a third wide receiver, the Sam has three choices, and two of them are bad. He can stay in the box, where he can’t cover the slot receiver. He can walk out over the slot, an apex alignment between the box and the receiver, where a linebacker is now covering a receiver in space. Or he can leave the field for a nickel back, a cornerback who can cover and is big enough to tackle. Most NFL defenses choose the third, and the nickel package that results, four linemen, two linebackers and five defensive backs, is the 4-2-5 you will meet as a front family of its own in Odd and Hybrid Fronts.

Someone still has to do the Sam’s old job. On the right of Figure 29.9 the strong safety steps in over the tight end to take the D gap, and the coverage behind him stays one-deep, with a single safety in the middle of the field. The box goes from eight defenders to seven, and that is not a concession: the offense took out a blocker too. Against 21 personnel the box was eight against seven blockers (five linemen, the tight end and the fullback); against 11 it is seven against six. Either way the defense has one more man than the offense can block (the quarterback, who hands off, isn’t counted). That is roughly how Pete Carroll’s Seattle defenses of the 2010s played it, with a big strong safety, Kam Chancellor, as the down safety: close to the line against the run, in an underneath zone against the pass.17 The other option is to keep two safeties deep and let the box be six against six blockers, the “light box” that A First Look at Defense measured, and accept that the offense can run. That choice, and its arithmetic, is the subject of Run Fits and Box Math.

The open-side end often gets wider. Against 21 personnel the weak end has a fullback who can lead at him and a tight run game to squeeze, so he stays close, in his loose 5. Against 11 personnel there is no fullback, a slot receiver and a nickel back are now outside him, and the odds say pass. So he slides out to a wide-9’s angle, where he rushes better and still owns the C gap and the edge on a run. Watch an even front against 11 personnel on third down and you will often see one end in a 5 on the tight end’s side and the other as wide as a wide-9 on the open side.

The trend is running the other way again. In 2025 NFL offenses used fewer than three wide receivers on 41.7% of plays, the highest share since Next Gen Stats began tracking in 2016, and defenses answered by playing base personnel on 29.7% of snaps, up from a low of 20.5% in 2023.18 The 4-3 Over and Under, with a real Sam on the field, are coming back with the tight ends and fullbacks. How often teams line up in each front is harder to say: the public play-by-play and participation data count defenders in the box and pass rushers but don’t name the front, so for that you are back to the broadcast, and to the next section.

Reading the front from the broadcast

The diagrams in this book look down from above. The broadcast doesn’t. The main camera sits high on one sideline, near the line of scrimmage, so you see the line from the side, and the one thing a front is made of, which shoulder each lineman is on, runs straight away from you into the screen.

def project(pts, cam, target):
    """Pinhole camera: world points (downfield d, offense's LEFT, height) -> screen (x, y) and depth."""
    C, Tg = np.asarray(cam, float), np.asarray(target, float)
    fwd = (Tg - C) / np.linalg.norm(Tg - C)
    right = np.cross(fwd, [0.0, 0.0, 1.0]); right /= np.linalg.norm(right)
    up = np.cross(right, fwd)
    P = np.atleast_2d(np.asarray(pts, float)) - C
    z = P @ fwd
    return (P @ right) / z, (P @ up) / z, z

From the high sideline shot, you can’t measure a technique, and the bubble is hard to see: the defenders stand one behind another, and the space across from the uncovered guard is only a little wider than the space next to it. What you can find, every snap, are three landmarks.

  1. The tight end. He is the strength, so find him first: the man at the end of the line, a step off the tackle, often with a hand on his knee. If he is on the near side of the screen, the strong side is near; if he is on the far side, it is far.
  2. The 3-technique. Find the center, then the two defensive tackles. One, the 1, is almost straight across from the center. The other, the 3, stands clearly off to one side, out over a guard. If the 3 is on the tight end’s side, it’s an Over; if he is on the other side, it’s an Under. That is the same fact as the bubble, seen from the other direction: the empty guard is always on the side away from the 3.
  3. The edge outside the tight end. A linebacker standing on the line beside the tight end, part of the defensive line, is the Under’s Sam. Nobody there, and a linebacker a few yards off the ball, points to an Over (or to the Tampa-style Under of the dialect box).

Then add the details. Ends lined up so wide they are nearly out of the shot are a wide-9. Three linemen packed across the middle with no tackle off to one side, and extra men stacked on the tight end’s side, is a bear. And the sideline read is a first guess, not a verdict: when the replay comes from the end-zone camera, check it. From behind the offense every shoulder is visible, and after a few weeks of checking, the sideline guess gets fast and mostly right.

  • Toward the tight end, the lane between the 3-technique and the right end spans two gaps, the C and the D, and the wide-9’s whole job on the snap is to go upfield, not to close it. Against this front the right tackle and the tight end are both uncovered, so the blocking is clean: the right guard takes the 3, the right tackle climbs to the Mike, the tight end blocks the Sam, and the end is left alone to run himself out of the play. The Sam is the only defender who owns that lane, so he has to fill it before the back gets there, ideally meeting the tight end at the line so the back has to cut back inside. If he’s late, the next defender is a safety ten yards downfield.
  • Away from the tight end, the lane between the 1 and the left end is even wider on paper, and it sits on the bubble: the left guard is uncovered. The center and left guard double the 1 and one climbs, the left tackle climbs to the Will, and again the end is left alone. Here the Will decides, with the same two-gap problem and a 300-pound tackle coming at him.

Many coaches would still go strong first, because the tight end gives them an extra blocker on that side and a running back can bounce outside him if the Sam over-fills. Either way, the end can help by squeezing when he reads run, closing down the line behind the play rather than rushing past it, and the defense’s real answer is a safety who arrives in time. That is the trade from Figure 29.5, live: the ends’ rush is a bet that the linebackers and safeties will be on time.

The best modern answer of all uses the end’s alignment against him. A shotgun offense can run inside zone, leave the backside wide-9 unblocked and read him: if he chases the back, the quarterback keeps the ball and runs where the end used to be. That is the zone read of Option Football, and its pass-game cousin, the RPO, punishes the same man the same way.

Back to the huddle

The chapter opened with a defensive tackle jogging across the center when the tight end lined up on the right, a linebacker walking up onto the line outside the tight end, and, two plays later, the same defense staying put. Now you can read it. The tackle over the right guard was only standing there while the offense huddled: until the tight end shows his side, the defense doesn’t know where its front goes. On the first play the call was a 4-3 Under: once the tight end set up on the right, the 3-technique went to the weak guard, away from him, and the Sam walked up to make the strong-side edge two men deep at the line. On the second, with the tight end on the left, the call was an Over: the tackle was already on the guard on the tight end’s side, so nobody had to cross, and the Sam stayed off the ball. One word, and the formation, set eleven alignments. And each was a bet about which guard to leave free and where the run would be met.

Takeaways

  • An even front sets its linemen in pairs, with nobody head-up on the center. Two tackles can’t cover three interior linemen, so the center or a guard is always free; every even front is a decision about which one, and which linebacker answers for him.
  • Over and Under are the same players, leaning opposite ways from a strength call. The Over puts the 3-technique and its linemen toward the tight end and leaves the weak guard uncovered for the Will; the Under sends the 3-technique to the weak side to rush one-on-one, puts the Sam on the line to harden the strong-side edge, and leaves the strong guard uncovered under the Mike. Runs aim at the uncovered guard: the bubble.
  • Strength is relative, so offenses move it. A tight-end shift can make half a front re-set in a second and a half, or the defense can check Over to Under and move only the linebackers, because an Over one way is the same line as an Under the other way. Every answer gives something up.
  • The wide-9 buys pass rush with run lanes. Unblocked at the snap, the ends reach full speed before the tackle can touch them, and the two-gap lanes inside them become the linebackers’ and safeties’ problem. With a stout interior and fast linebackers (the 2017 Eagles) it can stop the run; without them it gets run through.
  • The bear front covers the center and both guards, and the 46 built a defense on it. With no uncovered interior lineman, double teams, climbing guards and pullers all become hard. Buddy Ryan’s 1985 Bears smothered two-back offenses with it; three receivers and quick throws made it a situational front, now used near the goal line and against zone-running teams.
  • Against 11 personnel the line stays and the Sam goes. The four linemen keep their rule; the third linebacker becomes a nickel back, and someone, often a safety, takes his job on the tight end.
