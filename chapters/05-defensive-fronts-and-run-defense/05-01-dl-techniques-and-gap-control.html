OL_KEYS = ("LT", "LG", "C", "RG", "RT")


def pdf_build():
    return os.environ.get("RTG_FORMAT", "").lower() in ("pdf", "typst", "print")


def O(key, pos, d, w, label):
    return Player(key, pos, "off", d, w, label=label)


def D(key, pos, d, w, label):
    return Player(key, pos, "def", d, w, label=label)


def line(labels="full"):
    """The five offensive linemen, labelled LT..RT ("full") or with one letter, T G C G T ("short")."""
    short = {"LT": "T", "LG": "G", "C": "C", "RG": "G", "RT": "T"}
    return [p.moved(label=(p.key if labels == "full" else short[p.key])) for p in offense("singleback")
            if p.key in OL_KEYS]


def te(side=1, key=None, label=None):
    """An in-line tight end: Y on the right, U on the left."""
    key = key or ("Y" if side > 0 else "U")
    return O(key, "TE", -0.7, side * 4.8, label or key)


def T(t, side=1):
    """Width of technique t on the offense's right (side=+1) or left (side=-1)."""
    return technique(t, side)


def go_to(p, key, end, kind="block", delay=0.0, speed=None, via=()):
    """Move `key` through absolute (d, w) points `via` to `end`, as a block, path or run."""
    start = np.array(p._cur(key))
    pts = [tuple(np.array(q) - start) for q in (*via, end)]
    if kind == "block":
        return p.block(key, pts=pts, delay=delay, speed=speed)
    if kind == "pull":
        return p.pull(key, pts, delay=delay, speed=speed)
    return {"path": p.path, "run": p.run}[kind](key, pts, delay=delay, speed=speed)


def _P(fld, p, d, w):
    return fld.xy(*p.to_abs(d, w))


def say(fld, p, d, w, text, color=style.INK_2, fontsize=7.5, ha="center", weight="normal"):
    """A short on-field label at (d, w)."""
    x, y = _P(fld, p, d, w)
    fld.ax.text(x, y, text, fontsize=fontsize, color=color, ha=ha, va="center", zorder=9, fontweight=weight,
                bbox=dict(boxstyle="round,pad=0.18", fc=style.SURFACE, ec="none", alpha=0.9))


def point_to(fld, p, text_at, target, text, color=style.INK_2, fontsize=7, shrink=7):
    """A label with a thin leader line to the player or spot it names."""
    (xt, yt), (xp, yp) = _P(fld, p, *text_at), _P(fld, p, *target)
    fld.ax.annotate(text, (float(xp), float(yp)), xytext=(float(xt), float(yt)), fontsize=fontsize,

def gap_tag(fld, p, d, w, text, color=GREEN, fontsize=8.5):

def spot(fld, p, d, w, color=style.DEFENSE_DARK, ms=10):
    """Mark a point (a tackle, a meeting point) with an X."""
    x, y = _P(fld, p, d, w)
    fld.ax.plot([x], [y], marker="X", ms=ms, color=color, mec="white", mew=0.8, zorder=12)


def lane(fld, p, w0, w1, d0, d1, color=style.OFFENSE, alpha=0.10, ls=(0, (3, 2))):
    """Shade a rectangular lane (a run lane, a gap) from width w0 to w1 and depth d0 to d1."""
    (x0, y0), (x1, y1) = _P(fld, p, d0, w0), _P(fld, p, d1, w1)
    fld.ax.add_patch(Rectangle((min(x0, x1), min(y0, y1)), abs(x1 - x0), abs(y1 - y0), fc=color, alpha=alpha,
                               ec=color, lw=0.9, ls=ls, zorder=0.8))


def dashed(fld, p, pts, color=style.THIRD, lw=1.5, dash=(0, (3, 2.2)), end="arrow"):
    """A dashed 'what if' or 'intent' path through absolute (d, w) points."""
    xs, ys = _P(fld, p, np.array([q[0] for q in pts]), np.array([q[1] for q in pts]))
    if end == "arrow":   # dashed all the way to a short solid head on the last 25% of the last leg
        xs, ys = np.asarray(xs, float), np.asarray(ys, float)
        hx, hy = xs[-2] + 0.75 * (xs[-1] - xs[-2]), ys[-2] + 0.75 * (ys[-1] - ys[-2])
        xs = np.concatenate([xs[:-1], [hx, xs[-1]]]); ys = np.concatenate([ys[:-1], [hy, ys[-1]]])
    fld.ax.plot(xs[:-1] if end == "arrow" else xs, ys[:-1] if end == "arrow" else ys, color=color, lw=lw,
                ls=dash, zorder=6)
    if end == "arrow":
        fld.ax.add_patch(FancyArrowPatch((xs[-2], ys[-2]), (xs[-1], ys[-1]), arrowstyle="-|>",
                                         mutation_scale=10, color=color, lw=lw, zorder=6))


def bracket(fld, p, d, w0, w1, color=style.DEFENSE_DARK, lw=2.2, stem_to=None):
    """A bar spanning the gaps a two-gapper owns, with end ticks and an optional stem to the player."""
    (xa, ya), (xb, yb) = _P(fld, p, d, w0), _P(fld, p, d, w1)
    fld.ax.plot([xa, xb], [ya, yb], color=color, lw=lw, solid_capstyle="butt", zorder=5)
    for w in (w0, w1):
        (x0, y0), (x1, y1) = _P(fld, p, d - 0.35, w), _P(fld, p, d + 0.35, w)
        fld.ax.plot([x0, x1], [y0, y1], color=color, lw=lw, zorder=5)
    if stem_to is not None:
        (x0, y0), (x1, y1) = _P(fld, p, d, (w0 + w1) / 2), _P(fld, p, *stem_to)
        fld.ax.plot([x0, x1], [y0, y1], color=color, lw=1.0, ls=(0, (2, 1.5)), zorder=4.5)


def clean(ax):
    """Box-zoomed diagrams: drop hash ticks and yard numbers, clip anything outside the window."""
    for ln in list(ax.lines):
        xs, ys = ln.get_data()
        if len(xs) == 2 and ln.get_linewidth() == 0.5 and np.hypot(xs[1] - xs[0], ys[1] - ys[0]) < 1.0:
            ln.remove()
    for t in list(ax.texts):
        if t.get_text().isdigit() and t.get_alpha() == 0.55:   # yard numbers, not technique labels
            t.remove()
        else:
            t.set_clip_on(True)


def draw(p, width_in=5.8, ax=None, **kw):
    """Play.draw, sized for the page, cleaned for a box-sized window."""
    fld = p.draw(width_in=width_in, ax=ax, **kw)
    clean(fld.ax)
    return fld


def front(spec, lbs=(), others=()):
    """Defensive linemen from (key, technique, side) triples, labelled with their technique number,
    plus (key, d, w, label) linebackers and any other defenders."""
    out = [D(k, "DL", 1.0, T(t, s), t) for k, t, s in spec]
    out += [D(k, "LB", d, w, lab) for k, d, w, lab in lbs]
    out += [D(k, "S", d, w, lab) for k, d, w, lab in others]
    return out

MARK_R = 0.62


def tracking(play):
    df = play.to_tracking()
    df["_id"] = df["displayName"].astype(str)
    return df


def panel(ax, play, df, t, lateral, window, rings=(), enter=(), ring_r=0.85):
    """One moment of a simulated play on `ax`: markers, trails of on-screen players, rings."""
    tmap = df.groupby("frameId")["time"].first()
    fid = int((tmap - t).abs().idxmin())
    ylo, yhi = sorted((play.ball_y - lateral[0], play.ball_y - lateral[1]))
    fld = Field(ax=ax, orient="vertical", los=play.los, window=window, numbers=False, hashes=False,
                lateral=(ylo, yhi), to_go=play.to_go)
    cur = df[df["frameId"] == fid]
    snap = df[df["frameId"] == int((tmap - 0.0).abs().idxmin())]
    first = snap[snap["x"].between(play.los + window[0] + 0.8, play.los + window[1] - 0.8)
                 & snap["y"].between(ylo + 0.8, yhi - 0.8)]       # players whose marker fits whole at the snap ...
    ids = set(first["_id"]) | {"football"}
    inside = cur[cur["x"].between(play.los + window[0], play.los + window[1]) & cur["y"].between(ylo, yhi)]
    ids |= set(inside["_id"]) & set(enter)                         # players allowed to walk into the picture
    shown = cur[cur["_id"].isin(ids) | (cur["side"] == "ball")]    # ... stay in every frame (clipped at the edge)
    trails = df[(df["frameId"] <= fid) & df["_id"].isin(ids) & (df["side"] != "ball")]
    for (_, side), g in trails.groupby(["_id", "side"]):
        xs, ys = fld.xy(g["x"].values, g["y"].values)
        ax.plot(xs, ys, color=style.OFFENSE if side == "off" else style.DEFENSE, lw=0.9, alpha=0.4, zorder=3)
    ppy = fld.pts_per_yard()
    for _, r in shown.iterrows():
        x, y = (float(v) for v in fld.xy(r["x"], r["y"]))
        if r["side"] == "ball":
            ax.add_patch(Ellipse((x, y + 0.45), 0.7, 0.42, fc=style.BALL, ec="white", lw=0.5, zorder=9))
            continue
        lab = str(r.get("label", "") or "")
        if r["side"] == "off":
            ax.add_patch(Circle((x, y), MARK_R, fc=style.OFFENSE, ec="white", lw=0.6, zorder=8))
            tc = "white"
        else:
            ax.add_patch(RegularPolygon((x, y), 4, radius=MARK_R * 1.3, orientation=np.pi / 4,
                                        fc=style.DEFENSE_TINT, ec=style.DEFENSE, lw=1.0, zorder=8))
            tc = style.INK
        fs = max(4.0, min(8.0, ppy * (0.85 if len(lab) <= 1 else 0.62)))
        ax.text(x, y, lab, ha="center", va="center", fontsize=fs, color=tc, fontweight="bold", zorder=10)
    for key, col in rings:
        r = cur[cur["displayName"] == key].iloc[0]
        x, y = fld.xy(r["x"], r["y"])
        ax.add_patch(Circle((float(x), float(y)), ring_r, fc="none", ec=col, lw=1.6, zorder=7))
    for patch in ax.patches:
        patch.set_clip_on(True)
    return fld


def strip(moments, lateral, window, notes=None, rings=(), ncols=2, width_in=5.2, after=None, enter=()):
    """Numbered frame strip. `moments` is a list of (play, t, tag): several plays may share one strip,
    which is how a branching play (same start, two endings) is shown in print."""
    cache = {}
    nrows = int(np.ceil(len(moments) / ncols))
    pw = width_in / ncols
    span_w, span_d = lateral[1] - lateral[0], window[1] - window[0]
    fig, axes = plt.subplots(nrows, ncols, squeeze=False,
                             figsize=(width_in, nrows * (pw * span_d / span_w + (0.62 if notes else 0.25))))
    for ax in axes.flat[len(moments):]:
        ax.axis("off")
    for i, (ax, (play, t, tag)) in enumerate(zip(axes.flat, moments)):
        if id(play) not in cache:
            cache[id(play)] = tracking(play)
        fld = panel(ax, play, cache[id(play)], t, lateral, window, rings, enter)
        if after:
            after(fld, play, t, i)
        lab = "snap" if abs(t) < 1e-6 else f"{t:+.2f} s".replace("0 s", " s")
        ax.set_title(f"{i + 1}.  {lab}" + (f"   {tag}" if tag else ""), fontsize=8.3, color=style.INK_2, loc="left")
        if notes:
            ax.text(0.5, -0.03, notes[i], transform=ax.transAxes, ha="center", va="top",
                    fontsize=7.1, color=style.INK)
    fig.tight_layout(h_pad=1.3 if notes else 0.6, w_pad=0.6)
    return fig


def video(segments, lateral, window, rings=(), width_in=4.6, after=None, enter=()):
    """Web version: each (play, tag, [(t, note), ...]) segment played in turn, notes as captions."""
    import shutil
    from matplotlib import animation
    from IPython.display import HTML, display
    span_w, span_d = lateral[1] - lateral[0], window[1] - window[0]
    fig, ax = plt.subplots(figsize=(width_in, width_in * span_d / span_w + 0.45))
    frames = []
    for play, tag, notes in segments:
        df = tracking(play)
        tmap = df.groupby("frameId")["time"].first()
        frames += [(play, df, float(tmap[f]), tag, notes) for f in sorted(df["frameId"].unique())]

    def update(k):
        play, df, t, tag, notes = frames[k]
        ax.clear()
        fld = panel(ax, play, df, t, lateral, window, rings, enter)
        if after:
            after(fld, play, t, -1)
        txt = ("snap" if abs(t) < 0.05 else f"{t:+.1f} s") + (f"  {tag}" if tag else "")
        cur = [n for tt, n in notes if tt <= t + 1e-6]
        ax.set_title(txt + (("\n" + cur[-1].replace("\n", " ")) if cur else ""), fontsize=8.5, color=style.INK,
                     loc="left")
        return []

    anim = animation.FuncAnimation(fig, update, frames=range(len(frames)), interval=100)
    html = (anim.to_html5_video(embed_limit=40).replace("<video ", '<video style="max-width:100%;height:auto" ')
            if shutil.which("ffmpeg") else anim.to_jshtml(default_mode="once"))
    plt.close(fig)
    display(HTML(html))


def animation_or_strip(moments, notes, lateral, window, rings=(), after=None, enter=(), ncols=2, width_in=5.2):
    """Video on the website; a numbered, captioned frame strip in the PDF."""
    if pdf_build():
        strip(moments, lateral, window, notes=notes, rings=rings, ncols=ncols, width_in=width_in,
              after=after, enter=enter)
        plt.show()
        return
    segs = []
    for play, t, tag in moments:
        if not segs or segs[-1][0] is not play:
            segs.append((play, tag, []))
    for (play, t, tag), n in zip(moments, notes):
        for s in segs:
            if s[0] is play:
                s[2].append((t, n))
    if len(segs) > 1:   # later branches share the opening frames: show the first branch's opening notes too
        opening = [(t, n) for t, n in segs[0][2] if t < 0.5]
        for s in segs[1:]:
            s[2][:0] = [x for x in opening if x[0] < min(tt for tt, _ in s[2])]
    video(segs, lateral, window, rings=rings, after=after, enter=enter)

Second quarter, first-and-10. The running back takes the handoff, starts toward the right side of the line, and is hit before he gets to it, by a defensive tackle who is already two yards into the backfield. The replay shows the tackle lined up just outside the right guard and gone past him before the guard has finished his first step. “He lives in the backfield,” says the analyst. Two drives later, the same offense runs right up the middle. This time the defender in the center of the line, the biggest man on the field, doesn’t go anywhere at all. The center hits him, a guard comes over to help, and he simply stays where he is, two blockers leaning on him, until the running back arrives and he throws one of them aside and makes the tackle. “He ate up two blockers,” the analyst says. Two defensive linemen, two opposite ways to be great: one wins by going, the other by staying. This chapter is about why both exist, what each costs, and how to tell which one you are watching.

The defensive lineman’s three jobs

Watch a defensive lineman on a running play and it looks like a shoving match he either wins or loses. It is more specific than that. On every snap he has three jobs, in order:

  1. Defeat the block. A 300-pound man is coming at him, from the front, the side, or two at once. Before he can do anything else, he has to stop that man from moving him where the offense wants him to go.
  2. Control his gap. Every defense gives each run defender a gap (or two) to be responsible for, so that wherever the ball carrier goes, someone is already there. A lineman who beats his blocker but ends up in the wrong place has won the fight and lost the play.
  3. Find the ball and get to it. Only after the first two does he get to chase the ball carrier, and by then, on most runs, the ball carrier is already close.

The tension in defensive line play is between the first two jobs. A lineman who is on a blocker, hands locked on him, can feel where the block is taking him and can hold his ground, but he is not going anywhere fast. A lineman who is in a gap, shooting through the space beside the blocker, can get into the backfield in a heartbeat, but if he guesses wrong about where the ball is going, he has run himself out of the play. Every defense chooses, snap by snap and player by player, which of those two to ask for. That choice has names, one-gap and two-gap, and they are the spine of this chapter. First, though, look at where the linemen stand, because the alignment is already half of the choice.

The nose: 0 and 1

The man over the center is the nose, the nose tackle of The Twenty-Two. He lines up in one of two ways. Head-up on the center he is a 0-technique, and nothing about his alignment tells the offense which side of the center he will play. On one of the center’s shoulders he is a 1-technique, a shade nose, sitting in that side’s A gap.

Either way, he is the player offenses most like to double-team. Look at where he stands: the center in front of him and a guard on either side, all within a step. Two of those three can reach him with no effort at all, and the combo block you met in 03-01, center and guard together, is aimed at men in exactly his spot. So the nose’s first quality is not speed but mass and pad level: getting lower than the blocker, so that the blocker’s push goes up into the nose’s chest while the nose’s push goes straight through the blocker, and being heavy enough that two men leaning on him can’t move him. That is why noses are the heaviest players on the field. Vince Wilfork, the model for a generation of them, measured 6-foot-1 and 323 pounds at his Scouting Combine; modern noses such as the Eagles’ Jordan Davis (341 pounds) and Dexter Lawrence (342), a Giant until an April 2026 trade sent him to Cincinnati, are just as heavy and much taller.1

What does the defense get from a man who absorbs two blockers? Arithmetic. In hat math the offense has a fixed number of blockers. Every blocker spent on the nose is one who isn’t climbing to a linebacker, and an unblocked linebacker is a tackle waiting to happen. A great nose’s best plays often show up in a linebacker’s tackle total, not his own.

The 3-technique: the penetrator

You met the 3-technique in 03-01 as the alignment that became a position: a defensive tackle on the guard’s outside shoulder, in the B gap. What is new from the defense’s side is why the defense puts its best interior athlete there. The spot gives him the best odds of a single blocker. The center is two men away and can’t easily help. The tackle can, and against the run the guard-tackle double team on a 3 is one of the most common blocks in football; but every snap the tackle spends on him is a snap he isn’t climbing to a linebacker or blocking the end, so the offense has to choose which problem to leave alone. On passing downs the choice usually goes the 3’s way: the center slides to help the other guard, the tackle has the edge rusher, and the 3 gets the guard alone. So the defense puts its quickest, most explosive interior player there, built to get off the ball and shoot the gap before the guard can get his body in the way: Warren Sapp in the 1990s, Aaron Donald in the 2010s, both about 6-foot-1 or 6-foot-2 and listed at roughly 280 to 300 pounds, both much lighter than a nose and much quicker.2 Teams pay for that: in the contracts file in October 2026, the ten richest deals for interior defensive linemen averaged about $31 million a year, more than any defensive position except edge rushers.3

Which side does he line up on? That is the defense’s choice, and it is usually made by a rule tied to the offense’s formation: many 4-3 teams put the 3-technique toward the tight end (the Over front), others away from him (the Under), and some toward the wide side of the field or the running back. Spot the rule and you can predict where the 3 will be before the offense breaks the huddle. The fronts and their rules are in Even Fronts.

The cost is the mirror image. A 3-technique who is moved off his spot by a double team, or who shoots upfield and misses, leaves the B gap empty. That’s the bargain one-gap football makes, and we’ll come back to it.

The tackle’s techniques: 4i, 4 and 5, the long men

Three alignments sit on the offensive tackle: the 4i on his inside shoulder, the 4 head-up on him, and the 5-technique on his outside shoulder. These are the classic spots of the defensive ends in a three-man front, and the body they want is long: tall players with long arms, around 290 to 310 pounds, who can lock out an offensive tackle at arm’s length and still see past him to the ball. J.J. Watt, 6-foot-5 and 290 at his combine, played the 5 for the Houston Texans and turned it into a pass-rushing position: he led the league in sacks twice.4

Each spot hints at a job, though the call has the final say. The 4, head-up, is the classic two-gap end of the old 3-4: he controls the tackle and owns the B and C gaps on either side of him. The 5 usually owns the C gap he sits in. The 4i usually owns the B gap he sits in, as a one-gapper, but some defenses ask him to control the tackle and play the C gap too, so watch his first step.

Why the inside shoulder? 03-01 gave the offense’s view: same gap as a 3, different blocker occupied. From the defense’s side the point is the blocking angles. On a zone run toward him, the guard has to reach a man who is already outside him; on a run away, the tackle has to cut off a man who is already inside him. And because the 4i can play off either blocker, he muddles which of them should take him and which should climb. It is the end’s favorite alignment in the tite fronts that NFL defenses use to stop inside runs (five men on the line: three linemen in a 4i, a 0 and a 4i, plus two edge players standing outside them), which you will meet in Odd and Hybrid Fronts. The 5, outside the tackle in the C gap, is the end’s spot in four-man fronts as well, where, with no tight end on his side, he is often the last man on the line and has the edge.

The 9, the wide-9 and the edge

The 9 sits on the tight end’s outside shoulder. Whoever lines up there is usually the defense’s edge on that side: in a 4-3 a defensive end, in a 3-4 an outside linebacker standing up rather than in a three-point stance (T.J. Watt in Pittsburgh is the type). He is lighter again, 250 to 270 pounds, because his two jobs both happen in space: rush the passer around the tackle, and on runs, keep the ball from getting outside him. That second job is called setting the edge, and it has its own section below.

Go wider still and you reach the most extreme alignment on the line, the wide-9: a defensive end lined up well outside the tight end, usually one to two or more yards beyond the tight end’s outside shoulder (the exact width changes with down and distance), or, with no tight end, just as far beyond where one would stand. He is usually in a sprinter’s stance: a three-point stance with his feet staggered and his weight well forward on his hand, like a sprinter in the blocks, often tilted in toward the quarterback. The difference from a 9 is not a new number but distance and posture: a 9 is close enough to touch the tight end and is ready to take him on; a wide-9 is out of reach and ready only to run. The wide-9 exists for one reason: it gives a fast pass rusher the best possible angle around the tackle. The tight end can’t block him without help (he is too far outside), and the tackle has to cover a lot of ground to meet him. The price is just as plain. With the end that far out, the space between him and the next defender inside is enormous, and somebody, usually a linebacker, has to fill it on runs.

The numbers this map leaves out

03-01’s chart had a few more numbers. They exist, but you will see them less often, because each is a compromise between two of the jobs above. The 2i (the guard’s inside shoulder) and the head-up 2 put a man on a guard rather than the center: a 2i squeezes the A gap from outside, and some odd and under fronts use one, but a lineman there is easy for the center and guard to double, without the 3’s one-on-one or the nose’s mass to make it worth it. The 6 and 7 are head-up and on the inside shoulder of a tight end: spots for a strong-side end whose first job is to stop the tight end from blocking down on the linemen inside him. You will meet them in the front families of the next two chapters.

ImportantWhy it exists

Every alignment is a trade between the run and the pass, and between size and speed. Inside, where blockers are close together and double teams are easy, the defense wants mass that can’t be moved: the nose. One step outside, where a defender can get a single blocker, it wants quickness: the 3-technique. On the tackle, it wants length: the 4i, 4 and 5. On the edge, it wants speed, and the wider it aligns its edge rusher the more pass rush it gets and the more run defense it gives up. When you read a front, read it as a list of bets. Which spots did the defense fill with which kind of player, and what is it hoping you’ll do?

One-gap: attack a gap

Now the big choice. In a one-gap defense, each defensive lineman is responsible for exactly one gap. He lines up in it or shaded toward it (that is what an odd number or an “i” usually signals), and at the snap he attacks it. His goal is penetration: getting across the line of scrimmage and into the offense’s backfield before the play can develop, through his gap. If a blocker gets in his way, he defeats the blocker while staying in his gap; he does not need to read where the ball is going first, because the other gaps belong to somebody else.

How does he attack it? Watch three things, the same three a defensive line coach teaches first. His stance: shaded toward his gap, with the hand and foot on the gap side slightly back, so his first step goes straight into the gap instead of at the blocker. His key: he does not wait for the quarterback’s cadence (the defense never knows the snap count), so he watches the ball, or the blocker’s hand beside it, and moves the instant it moves. And his hat and hands: helmet into the gap he owns, hands on the blocker’s near shoulder or chest, pads lower than the blocker’s, so that even if the blocker gets to him he is driving through the gap rather than standing in front of the man.

That last point, staying in his gap, is what makes one-gap football work: the gaps add up. Seven gaps across a line with one tight end, seven defenders in the box, one defender for each. The linemen take the gaps they are aligned in; the linebackers behind them take the rest. Everybody has one job and can play it fast. (Change the formation and the count changes with it: no tight end means six gaps, two tight ends mean eight, and the defense adds or removes a box defender to keep one man per gap. The full accounting is Run Fits and Box Math.)

Why it exists. One-gap play turns the defense’s linemen loose. A lineman who doesn’t have to read before he moves gets a half-step head start on every snap, and on the line half a step is everything: the blocker who is reaching or pulling or setting for a pass is moving sideways or backward while the defender is moving forward. The results are the plays fans remember, tackles for loss, sacks, a running back hit as he takes the handoff, and they wreck an offense’s plans in a way a two-yard gain never does. One-gap play also fits the players the modern game produces. A lineman who wins by quickness, the Donald type, is wasted standing still with his hands on a blocker; turn him loose in a gap and he is the most disruptive player on the field.

What it costs. Three things.

  • A wrong guess is a big hole. A one-gapper who penetrates past the play, or gets moved out of his gap by a double team, leaves his gap empty, and nobody else was assigned to it. A run that gets through a vacated gap is usually not a short gain, because the next defender is a linebacker who was busy with his own gap.
  • Offenses build plays to punish it. An aggressive penetrator is exactly what the trap in Gap and Power Runs is designed for: let him into the backfield unblocked, then hit him from the side with a pulling lineman, and run the ball through the space he left. Draws and screens work on the same idea: invite the rush, then go where it was.
  • It needs everyone else to be right. One-gap defense is only as good as its accounting. If one linebacker fills the wrong gap, there are two defenders in one gap and none in another. That is why one-gap teams spend so much practice time on run fits, the subject of Run Fits and Box Math.

Where it came from. For much of football’s history, defensive linemen were taught to read and react. The great attacking interior linemen changed that by being too quick to keep at home. The best known story is Pittsburgh’s in 1974: late that season the Steelers began lining Joe Greene up at the center, his stance tilted at about 45 degrees toward the A gap, so he could knife through before anyone could block him. Accounts differ on whose idea it was: the Hall of Fame credits Greene, other histories credit his line coach, George Perles. The Steelers called it the stunt 4-3 (a period name: today “stunt” usually means the looping pass-rush games of The Pass Rush, which this was not). It began as a practice-field experiment, got its first heavy use late that regular season, and carried the Steelers through the 1974 playoffs to their first Super Bowl win.7 Two decades later the Tampa Bay Buccaneers of Tony Dungy and Monte Kiffin built a whole defense around a one-gap 3-technique, Warren Sapp, who was the NFL’s Defensive Player of the Year in 1999.8

Two-gap: control the man, then pick a side

The other answer is older, heavier and quieter. In a two-gap defense, a lineman lines up head-up on a blocker (a 0, a 2, a 4: the even numbers) and is responsible for the gaps on both sides of him. At the snap he does not attack a gap. He attacks the man: strikes him with both hands inside the blocker’s chest, straightens his arms to keep the blocker off his body (coaches call it locking out), and stands him up at the line of scrimmage. Then, with the blocker held at arm’s length, he looks into the backfield, sees which side the ball is going, and throws the blocker the other way to make the tackle.

Coaches have a rule for how a two-gapper knows which way to go: read the blocker’s head. A blocker who wants to wall a defender off from one side has to get his helmet to that side of him. Say the center’s helmet slides to the right of the nose (the offense’s right, as always in this book): the center is trying to keep him from going right, which means the ball is going right. So the two-gapper “plays across the face”: he keeps his arms locked, works to the side the blocker’s head went (right), throws the blocker the other way (left), and is standing in the right A gap when the ball arrives. The block tells him where the ball is going before he ever sees the ball.

Head-up is the natural spot for this, because the defender starts with a gap on each side and the blocker in the middle. A shaded player can be asked to two-gap too, and some 4-3 teams ask exactly that of a 1 or a 4i. He plays it the same way, by striking and controlling the man he is shaded on, but his near gap is already his and his far gap is the hard one: to reach it he has to work back across the blocker’s face, so his alignment is a lean, not a commitment. That is why “odd number or an i” is only a hint about the job.

Why it exists. Look back at the right half of Figure 28.3. If one lineman can be responsible for two gaps, three linemen can hold six of them, and the defense has players left over. In the classic two-gap 3-4, the two inside linebackers have no gap of their own to rush to at the snap. They read the play and fill wherever the ball goes, and the two-gappers’ job is to make that possible. Three linemen can’t occupy five blockers just by standing there: the guards are uncovered, and if nothing stops them they climb straight to the linebackers. What stops them is the nose. A nose who can’t be moved by the center alone forces a guard to stay and help him, as in Figure 28.5, so at most one blocker gets up to the linebackers, and the 4s keep the tackles from climbing at all. That is the two-gapper’s real product: not tackles, but clean linebackers. And because he doesn’t commit before the snap, the offense can’t build its blocking around where he will go.

Two-gapping matters even more in the modern NFL than its reputation suggests. Defenses now keep two deep safeties on many snaps to stop the pass, which leaves only six men in the box against an offense that may have six or seven blockers (A First Look at Defense has the numbers). With six defenders and seven gaps, the accounting of one-gap football does not add up. The usual fix is to ask a big nose to play both A gaps, so six men can cover seven gaps. That is one of the ideas behind the light-box fronts of Vic Fangio and his followers, taken up in Odd and Hybrid Fronts and the case study Fangio’s Two-High Revolution and Macdonald’s Answer.

What it costs.

  • Rare bodies. Holding a 310-pound center at arm’s length while a guard hits you from the side takes size and strength that very few people have: two-gap noses are typically well over 320 pounds, and the pool of such players who can also move is small.9
  • No head start. Reading before moving means the two-gapper is always reacting. He rarely makes a tackle behind the line and rarely gets a sack. On a pass his job is mostly to push the center straight back into the quarterback’s lap, collapsing the middle of the pocket so the quarterback can’t step up, which matters but doesn’t show up as a sack. That is why classic two-gap noses often leave the field on third down for a lighter, quicker pass-rush package.
  • His value hides. The stat sheet credits the man who makes the tackle, not the man who made the tackle possible. A great two-gapper can have a quiet box score in a game he dominated.
CautionCommon misconception

“A 3-4 means two-gap, and a 4-3 means one-gap.” It is a decent first guess, nothing more. The classic 3-4s of the 1970s through the 2000s, Bill Parcells’s and Bill Belichick’s among them, did two-gap with big ends and a big nose. But Wade Phillips, son of the Bum Phillips who devised the technique numbers, spent decades running an attacking, one-gap 3-4: when he took over the Houston Texans’ defense in 2011, the team’s own preview called it “his one-gap scheme,” and the line coach explained that it was “not like a two-gap type defense where you need that 350-pound nose man to be able to just clog up and take on two blocks.”10 And plenty of 4-3 teams ask a nose to two-gap. The number of linemen tells you how many there are; their first step tells you how they play.

NoteMadden vs. real life

In Madden, a defensive lineman is mostly someone you steer at the quarterback, and the run defense takes care of itself. On a real early down, a lineman’s job is mostly the gap: hold it, or attack it, exactly as the call says. A nose tackle who abandons both A gaps to chase a sack can give up a 20-yard run even if he beats his man, which is why defensive line coaches grade every snap on assignment before they grade it on result.

Fighting the blocks

Whichever style he plays, a defensive lineman has to beat whatever block the offense throws at him, and the offense has a whole vocabulary of them. You learned them from the blocker’s side in 03-01. Here are the defender’s answers to the three that matter most, the down block, the reach and the double team, and the one principle underneath all of them: fight pressure with pressure. Whatever direction a block is pushing you, push back into it. A blocker pushes you where the offense wants you, toward the hole’s edge or out of your gap, so leaning into the pressure is how you stay where the defense wants you.

Against a down block

A down block hits a defender from the side: the blocker outside him comes inside, at an angle, to drive him toward the center and seal the inside of the hole. Two defenders are involved, and they have opposite jobs.

  • The man being down-blocked (the 3-technique in panel 1) feels pressure from his outside. If he gives way, he is washed down the line and the hole doubles in size. So he fights pressure: drops his outside shoulder into the blocker and pushes back outward, holding his gap and never slipping behind the block. A few defenses let a dominant player fight across the blocker’s face and into the hole himself; if he doesn’t win that fight, he has given up the inside.
  • The man left alone (the 5 in panel 1) has just seen the tackle in front of him leave. A blocker who vanishes is a warning, not a gift. On a gap-scheme run like power, the tackle blocks down because someone else is coming for the end: a pulling guard or a fullback, to kick him out. So he squeezes: steps down the line, hip to hip with the down-blocker, shrinking the hole, with his eyes inside for the puller. When the puller arrives, the defense’s call decides what he does next. He can keep his outside arm free and force the ball back inside (box or squeeze), or attack the puller’s inside shoulder and pile up the hole, pushing the ball outside to a teammate waiting there (spill, the “wrong-arm” you met in 03-03). Which one, and who is waiting, is the heart of Run Fits and Box Math.

There is a cousin of this rule that every defensive line coach teaches: if nobody blocks you, something is coming. A lineman who is let through untouched should close inside at once and get small: turn his shoulders and drop his pads, so a blocker coming from the side has only a narrow shoulder to hit and can’t knock him out of the gap. The likeliest explanation is a trap block from the side, the gap play built to catch penetrators. But coaches say “trap or read,” because there is a second explanation. On an option play the offense leaves an end unblocked on purpose and lets the quarterback read him (Option Football): if the end closes inside, the quarterback keeps the ball and runs where he was. How does a lineman know which it is? Partly from what he sees (a quarterback holding the ball in the back’s belly with his eyes on him is reading him; a guard pulling toward him is trapping him), and mostly from the call. The call tells him in advance whether he takes the back or the quarterback, and against option teams defenses often swap those jobs between the end and a linebacker, the scrape exchange of 03-04, precisely so the read is wrong.

Against a reach

A reach block is the blocker trying to get his head and body outside a defender who is already a step to the play side of him, and seal him inside. The defender’s whole job is not to let that happen, and he has the head start. A one-gapper fast enough to be through his gap before the blocker arrives beats the reach before it starts, as in Figure 28.4. Once the blocker has got to him, every defender, one-gap or two-gap, does the same thing: he steps with the blocker, keeps his outside arm free (so there is always an arm between the blocker and the sideline), and presses the blocker back into the backfield. Panel 2 shows the effect: the runner, who wanted to turn upfield just outside the guard, has to keep running sideways. Every yard he goes sideways is a yard of pursuit for the rest of the defense.

The mistake to avoid is the one that looks most athletic: running around the block. A defender who loops around the blocker’s back to get outside him has left his gap. On outside zone, that is exactly the hole the runner is looking for when he cuts back. The rule coaches use is to beat the block through the blocker, not around him.

Against a double team

The double team, or combo block, is the hardest thing a defensive lineman faces and the most important to survive. Two blockers want two things from him: to move him off the ball, and then, once the linebacker behind him commits, for one of them to come off and block that linebacker. The defender’s goal is to deny both.

  • Don’t get moved. He widens his base, sinks his hips, and fights the pressure of the blocker coming from the side. Coaches call that blocker the drive man (he does the moving) and the one in front of the defender the post man (he stops him from going forward). The defender drops the hip nearest the drive man and leans into him, so the pair can’t turn his shoulders and walk him out of his gap. Giving up a yard backward is how a double team creates the hole.
  • Keep both blockers. The longer he keeps both men busy, the longer the linebacker behind him goes unblocked. The double team’s second blocker is supposed to climb to the second level; a defender who stays heavy on him keeps him from leaving.
  • If he is losing, make a pile. A lineman being driven back is taught to drop his hips, even to go to a knee in the gap, so that the bodies pile up where the hole was supposed to be. It looks like losing. It is a win if the running back has nowhere to go.

A few great linemen split double teams, slipping between the two blockers into the backfield, and it is a spectacular play when it works. It is also a gamble. If he doesn’t get through, he has turned his shoulders and left his gap, and the double team walks him out of it.

Against blocks going away

When the play goes away from a defender, his blocker uses a cut-off block (or the whole backside line scoops), stepping toward the play to get between him and the ball. Here he does chase. He flattens down the line, staying at about the depth of the ball, so that if the runner cuts back he is waiting. Backside blockers on zone runs often cut him instead, diving at his thighs to put him on the ground; he plays the cut with his hands, pushing the blocker’s shoulders down and away, and keeps his feet free. But the defender farthest from the play has a different rule, and it comes next.

Setting the edge and keeping contain

The last man on the line on each side has a special job. Inside him, his teammates can count on someone being in the gap next to them. Outside him there is nobody, just the sideline. If a run gets outside him, it gets outside everybody.

To set the edge is to be that last man and win: take on the blocker coming at you (a tight end, a tackle, a pulling guard, a fullback) at or near the line of scrimmage, with your outside arm free, and don’t let him push you back or get his head outside you. Then the run has only one place to go, back inside, toward the linebackers and safeties chasing it. A defense talks about a hard edge (the defender holds his ground at the line and turns the ball in quickly) and a soft edge (he gets pushed back or upfield, and the runner gets around him). Setting a hard edge is one of the least glamorous and most valued skills an edge player has, and it is a big part of why teams still want 270-pound defensive ends when a 245-pound linebacker might rush the passer just as well. On a given play, the player who turns the ball back inside is also called the force defender; who that is, and how it changes with the coverage, is in Run Fits and Box Math.

The end who did both jobs as well as anyone was Reggie White, the “Minister of Defense,” who played for the Eagles and the Packers and retired with 198 sacks, then the NFL record (Bruce Smith later passed him). In Super Bowl XXXI, after the 1996 season, he sacked Drew Bledsoe three times, a Super Bowl record that has since been tied but never broken, as Green Bay beat New England 35–21.11 If you find his games, watch the runs at him rather than the sacks: the blocker in front of him going backward, White’s outside arm free, and the ball carrier turning back inside. That is a hard edge, from a man who was also one of the great pass rushers of his era: proof that an edge player doesn’t always have to choose between the two jobs.

Contain is the edge player’s other rule, and its spirit is the same: nothing gets outside me, but with an emphasis on patience. On the backside of a run, the end who has contain does not chase the play flat down the line. He trails it at about the depth of the ball, because the plays most likely to come back his way are the ones designed to punish a chasing end: the bootleg, the reverse, the quarterback keeping the ball on a read (Perimeter and Misdirection has those plays from the offense’s side). Not every call keeps the end outside, though: against teams that run the read option, the end is often told to crash on the running back while a linebacker loops outside him to take the quarterback, the scrape exchange from the trap-or-read discussion above, a swap built to confuse the read. Contain still belongs to somebody; the call says who. On a pass rush, contain means the ends rush outside the tackles but not past the quarterback’s depth. An end who loops too far upfield opens a lane between himself and the quarterback, and a mobile quarterback steps up and runs through it. You will hear analysts call this rush-lane discipline; how pass rushers balance it against getting home is in The Pass Rush.

Both jobs ask a fast, aggressive player to be patient on a down when patience isn’t rewarded with a statistic. That is exactly why offenses attack them. A defensive end who gets a little too eager for the sack or the tackle for loss is the end the bootleg is built for.

A slant is a called movement in which the linemen step hard across the face of the blocker in front of them at the snap, into the next gap over, usually all in the same direction, gaining only a little ground: the slanter’s job is to arrive in the next gap, not to go deep. The linebackers move the opposite way to fill the gaps the line just left, so the whole front’s gap assignments shift by one. An angle is a shorter movement, typically by one or two linemen: a step at about 45 degrees into the adjacent gap that moves the defender over while he keeps going upfield. Think of the slant as relocating the line, and the angle as a penetrator changing lanes. (The words are not standardized. Some staffs call every movement a slant, some have names for a slant toward the tight end and a slant away from him, and inward and outward moves by pairs of linemen are often called a pinch and an open. The footwork varies by staff too: the clinic source cited here starts the slant with a roughly 45-degree step that gains a little ground, and nearly every coach warns that a lineman who moves purely sideways can be caught by the blocker and pushed along the line. What separates the two movements is where the lineman is headed, across the line or through it.)12

Why move? Because the offense’s blocking is decided before the snap. Every offensive lineman has looked at the front and decided whether he is covered or uncovered, who doubles whom and who climbs to which linebacker. Movement changes the answers after it is too late to change the call. A slant can turn a covered lineman uncovered and an uncovered one covered in a single step. A double team aimed at a 3-technique hits air when the 3 angles into the A gap. A pulling guard runs into a defender who wasn’t in his path a moment ago. Movement is also how a defense of quick, lighter linemen deals with bigger blockers: they can’t win a head-on shove, so they don’t take one. As one college staff put it in a coaching clinic, “we use a lot of slants up front (gap control) to try and keep the offensive linemen occupied, which then will allow our linebackers the freedom to roam and get to the ball.”13

What it costs. Movement is a bet on where the play is going, and whether the bet pays depends on the blocking scheme as much as the direction.

  • Against gap schemes, slanting toward the run is the jackpot. Power and counter are built on pre-snap assignments and pullers aimed at where the defenders were; a front that slants into the play’s path arrives at the hole before the blockers do, and the puller finds a defender where his gap was supposed to be. Slant away from a gap play and every defender has stepped out of the gap the ball is coming through, with the replacing linebackers a step farther from it.
  • Against zone, slanting toward the run gains less than it looks. Zone linemen block whoever arrives in their area, so a slanting defender just runs into the next blocker’s arms (Zone Running explains why), and the play side gets crowded without getting stopped. The real danger is behind the slant: every lineman has left a gap, and the zone back’s cutback is aimed at exactly those gaps. Against zone, a slant lives or dies on the linebackers replacing the gaps the line left.
  • Either way, the linebackers have to move with the line every time. If one of them forgets the slant was called, two players fill one gap and the other is empty.

When two linemen go further and loop around each other on a pass rush, the movement becomes a stunt, which belongs to The Pass Rush.

How do the linemen know, and can you see it coming? The call comes from the sideline to the defense’s green-dot player, usually a linebacker, who relays it in the huddle or, against a fast offense, with a word or a hand signal at the line. It includes the front (where each lineman lines up), whether each plays one gap or two, and any movement, and it can be changed at the line when the offense shifts. From the stands or the couch you can sometimes see it leak. Look for the tells you learned for offensive linemen in How to Watch a Broadcast, applied to the other side of the ball: a lineman whose stance is cheated (weight leaning, back foot staggered toward the gap he is about to take, his body tilted the way Joe Greene’s was), a lineman who widens or tightens his alignment just before the snap, or a linebacker creeping toward the gap he will have to replace. Good defenses know this and try to give the same look on every snap, so treat any tell as odds, not a promise.

Putting a front together

Step back and the parts fit together. A defensive coordinator calling a front is choosing, for each lineman, three things you can now name:

  1. Where he lines up (his technique), which tells the offense who is covered and shapes its blocking.
  2. How he plays it: one-gap (attack the gap he is in) or two-gap (control the man and read).
  3. Whether he moves: straight ahead, or a slant or an angle into another gap at the snap.

Different answers to those questions are what separate the fronts in the rest of this part: the even fronts with their 3-techniques and wide ends (Even Fronts), the odd and tite fronts with their 0s and 4is (Odd and Hybrid Fronts), and the linebackers whose jobs change with every one of these choices (Linebackers and the Second Level). A defense can also mix them: two-gap the nose, one-gap the 3-technique, angle the end. The modern NFL front is less a system than a menu.

At the snap: the first step.

  • Does the lineman’s first step go into a gap, upfield (one-gap), or into the blocker, with his hands, standing him up (two-gap)?

  • Which way did the run go relative to the 3-technique: at him, or away? Did he get into the backfield, get double-teamed, or get trapped?
  • Did the run get outside the edge? If it did, ask who had the edge on that side and whether he got hooked. On a bootleg or scramble, ask who had contain.

The chapter opened with two great defensive linemen on the same afternoon. Now you can name them.

That is penetration, and the analyst’s “he lives in the backfield” is its signature.
  • The biggest man on the field, over the center, a guard comes over to help, and he stays where he is: a two-gap nose. The center and guard double-teamed him, he gave no ground, and when the back arrived he shed one blocker to the side the ball went and made the tackle. “He ate up two blockers” is the other half of the story: on a play he doesn’t finish himself, that guard never reaches the linebacker, who does.

One wins by going and one by staying, and both are doing the same arithmetic: making the offense’s blockers add up to less than the defense’s gaps.

Takeaways

  • Alignment is a job description. The nose (0 or 1) anchors against double teams and wants mass; the 3-technique is the likeliest to get a guard one-on-one and wants quickness; the 4i, 4 and 5 want length to control a tackle; the 9 and the wide-9 want speed, and the wider the end, the more pass rush and the bigger the run lane inside him.
  • One-gap attacks. Each lineman owns one gap and attacks it, aiming for penetration. It wins with quickness, creates sacks and tackles for loss, and turns the linemen loose, but a wrong guess leaves an empty gap, and traps, draws and screens are built to punish it.
  • Two-gap controls. A head-up lineman strikes the blocker, locks out, reads (the blocker’s head tells him where the ball is going) and sheds to the ball’s side. One man covering two gaps, and forcing a second blocker to stay on him, keeps the linebackers clean, at the cost of rare bodies, little pass rush, and a stat sheet that hides his value.
  • Fight pressure with pressure. Against a down block, the man blocked fights back outward and the man left alone squeezes and looks for the puller (or, if nobody blocks him at all, asks “trap or read?”); against a reach, step with the blocker and keep the outside arm free; against a double team, lean into the drive man, give no ground and keep both blockers.
  • The edge is everybody’s boundary. Setting the edge turns runs back inside; contain keeps the quarterback and the misdirection plays from getting outside. Both ask an aggressive player for patience.
  • Movement changes the picture after the snap. Slants shift the whole line a gap; angles move one man while he keeps attacking. Both beat pre-snap blocking assignments, especially gap schemes, and both are bets on direction that the linebackers have to back up, above all against zone’s cutback.

def depth_table(plays, times=(0.3, 0.6, 1.0)):
    rows = []
    for name, play, key in plays:
        trk = play.to_tracking()
        los = trk.loc[(trk["displayName"] == "football") & (trk["time"] == 0.0), "x"].iloc[0]
        me = trk[trk["displayName"] == key].set_index("time")

The same calculation on real frames is a few lines with gridiron.bdb. The 2025 Big Data Bowl release (2022 season, weeks 1–9) has every player’s location ten times a second, and its players file has his position label, so you can find each interior defensive lineman on each run and measure how deep he is at one second. Aggregated by player, it separates the penetrators from the anchors, with two caveats. Position labels are rosters, not jobs: filtering on “DT” and “NT” misses 3-4 ends who play a 4i or a 3 and edge players who slide inside on passing downs. And depth alone can’t tell a penetrator from a defender the offense let through: an unblocked man on a trap, an option read or an influence block shows up deep in the backfield too. import pandas as pd
from gridiron import bdb

plays = pd.read_csv("data/bdb/plays.csv")
players = pd.read_csv("data/bdb/players.csv")[["nflId", "position", "displayName"]]
runs = plays[plays["isDropback"] == False]                       # not a dropback: designed runs (plus kneels)
trk = bdb.load_tracking("data/bdb/tracking_week_*.csv")
if "position" not in trk:                                        # 2025 tracking has no position column:
    trk = trk.merge(players[["nflId", "position"]], on="nflId", how="left")   # take it from players.csv

out = []
for g, pl in runs[["gameId", "playId"]].itertuples(index=False):
    one = bdb.prepare(trk, plays, game_id=g, play_id=pl, label="position")
    los = one.loc[(one["side"] == "ball") & (one["time"] == 0.0), "x"].iloc[0]
    at1 = one[(one["time"].round(1) == 1.0) & one["position"].isin(["DT", "NT"])]
    out += [{"nflId": r.nflId, "depth": r.x - los} for r in at1.itertuples()]

depth = (pd.DataFrame(out).merge(players, on="nflId")
         .groupby("displayName")["depth"].agg(["mean", "count"]).query("count >= 50").sort_values("mean"))
