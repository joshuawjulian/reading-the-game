OL_KEYS = ("LT", "LG", "C", "RG", "RT")


def pdf_build():
    return os.environ.get("RTG_FORMAT", "").lower() in ("pdf", "typst", "print")


def O(key, pos, d, w, label):
    return Player(key, pos, "off", d, w, label=label)


def D(key, pos, d, w, label):
    return Player(key, pos, "def", d, w, label=label)


def line(labels="full"):
    """The five offensive linemen, labelled LT..RT ("full") or with one letter, T G C G T ("short")."""
    short = {"LT": "T", "LG": "G", "C": "C", "RG": "G", "RT": "T"}
    return [p.moved(label=(p.key if labels == "full" else short[p.key])) for p in offense("singleback")
            if p.key in OL_KEYS]


def te(side=1, key=None, label=None):
    """An in-line tight end: Y on the right, U on the left."""
    key = key or ("Y" if side > 0 else "U")
    return O(key, "TE", -0.7, side * 4.8, label or key)


def T(t, side=1):
    """Width of technique t on the offense's right (side=+1) or left (side=-1)."""
    return technique(t, side)


def gun11(y_attached=True):
    """Shotgun, 11 personnel: Y attached on the right (or flexed to a slot), H in the left slot, X and Z wide."""
    sk = [O("QB", "QB", -5.0, 0.0, "Q"), O("RB", "RB", -5.0, -1.6, "R"),
          O("X", "WR", -0.7, -15.0, "X"), O("H", "WR", -1.5, -9.0, "H"), O("Z", "WR", -1.5, 15.0, "Z")]
    sk.append(O("Y", "TE", -0.7, 4.8, "Y") if y_attached else O("Y", "TE", -1.5, 9.0, "Y"))
    return line("full") + sk


def go_to(p, key, end, kind="block", delay=0.0, speed=None, via=()):
    """Move `key` through absolute (d, w) points `via` to `end`, as a block, path, run or motion."""
    start = np.array(p._cur(key))
    pts = [tuple(np.array(q) - start) for q in (*via, end)]
    if kind == "block":
        return p.block(key, pts=pts, delay=delay, speed=speed)
    if kind == "pull":
        return p.pull(key, pts, delay=delay, speed=speed)
    if kind == "motion":
        return p.motion(key, pts=pts, speed=speed)
    return {"path": p.path, "run": p.run}[kind](key, pts, delay=delay, speed=speed)


def _P(fld, p, d, w):
    return fld.xy(*p.to_abs(d, w))


def say(fld, p, d, w, text, color=style.INK_2, fontsize=7.5, ha="center", weight="normal"):
    """A short on-field label at (d, w)."""
    x, y = _P(fld, p, d, w)
    fld.ax.text(x, y, text, fontsize=fontsize, color=color, ha=ha, va="center", zorder=9, fontweight=weight,
                bbox=dict(boxstyle="round,pad=0.18", fc=style.SURFACE, ec="none", alpha=0.9))


def point_to(fld, p, text_at, target, text, color=style.INK_2, fontsize=7, shrink=7):
    """A label with a thin leader line to the player or spot it names."""
    (xt, yt), (xp, yp) = _P(fld, p, *text_at), _P(fld, p, *target)
    fld.ax.annotate(text, (float(xp), float(yp)), xytext=(float(xt), float(yt)), fontsize=fontsize,
                    color=color, ha="center", va="center", zorder=9,
                    arrowprops=dict(arrowstyle="-", color=color, lw=0.7, shrinkA=2, shrinkB=shrink),
                    bbox=dict(boxstyle="round,pad=0.18", fc=style.SURFACE, ec="none", alpha=0.9))


def gap_tag(fld, p, d, w, text, color=GREEN, fontsize=8.5):
    """A gap letter in a small rounded box."""
    x, y = _P(fld, p, d, w)
    fld.ax.text(x, y, text, fontsize=fontsize, color=color, ha="center", va="center", zorder=9,

def spot(fld, p, d, w, color=style.DEFENSE_DARK, ms=10):
    """Mark a point (a tackle, a meeting point) with an X."""
    x, y = _P(fld, p, d, w)
    fld.ax.plot([x], [y], marker="X", ms=ms, color=color, mec="white", mew=0.8, zorder=12)


def lane(fld, p, w0, w1, d0, d1, color=style.OFFENSE, alpha=0.10, ls=(0, (3, 2))):
    """Shade a rectangular lane (a run lane, a gap) from width w0 to w1 and depth d0 to d1."""
    (x0, y0), (x1, y1) = _P(fld, p, d0, w0), _P(fld, p, d1, w1)
    fld.ax.add_patch(Rectangle((min(x0, x1), min(y0, y1)), abs(x1 - x0), abs(y1 - y0), fc=color, alpha=alpha,
                               ec=color, lw=0.9, ls=ls, zorder=0.8))


def dashed(fld, p, pts, color=style.THIRD, lw=1.5, dash=(0, (3, 2.2)), end="arrow"):
    """A dashed 'what if' or 'intent' path through absolute (d, w) points."""
    xs, ys = _P(fld, p, np.array([q[0] for q in pts]), np.array([q[1] for q in pts]))
    xs, ys = np.asarray(xs, float), np.asarray(ys, float)
    if end == "arrow":
        hx, hy = xs[-2] + 0.75 * (xs[-1] - xs[-2]), ys[-2] + 0.75 * (ys[-1] - ys[-2])
        xs = np.concatenate([xs[:-1], [hx, xs[-1]]]); ys = np.concatenate([ys[:-1], [hy, ys[-1]]])
        fld.ax.plot(xs[:-1], ys[:-1], color=color, lw=lw, ls=dash, zorder=6)
        fld.ax.add_patch(FancyArrowPatch((xs[-2], ys[-2]), (xs[-1], ys[-1]), arrowstyle="-|>",
                                         mutation_scale=10, color=color, lw=lw, zorder=6))
    else:
        fld.ax.plot(xs, ys, color=color, lw=lw, ls=dash, zorder=6)


def bracket(fld, p, d, w0, w1, color=style.DEFENSE_DARK, lw=2.0):
    """A bar spanning the gaps a defender owns, with end ticks."""
    (xa, ya), (xb, yb) = _P(fld, p, d, w0), _P(fld, p, d, w1)
    fld.ax.plot([xa, xb], [ya, yb], color=color, lw=lw, solid_capstyle="butt", zorder=5)
    for w in (w0, w1):
        (x0, y0), (x1, y1) = _P(fld, p, d - 0.32, w), _P(fld, p, d + 0.32, w)
        fld.ax.plot([x0, x1], [y0, y1], color=color, lw=lw, zorder=5)


def hand(fld, p, d, w, half=0.66):
    """A hand on the ground: a dark knob on the front edge of a defender's square marker."""
    x, y = _P(fld, p, d - half, w)
    fld.ax.add_patch(Circle((float(x), float(y)), 0.26, fc=style.DEFENSE_DARK, ec="white", lw=0.5, zorder=11))


def hands(fld, p, keys, t=None, half=0.66):
    for k in keys:
        d, w = (p.players[k].d, p.players[k].w) if t is None else p.position(k, t)
        hand(fld, p, d, w, half)


def clean(ax):
    """Box-zoomed diagrams: drop hash ticks and yard numbers, clip anything outside the window."""
    for ln in list(ax.lines):
        xs, ys = ln.get_data()
        if len(xs) == 2 and ln.get_linewidth() == 0.5 and np.hypot(xs[1] - xs[0], ys[1] - ys[0]) < 1.0:
            ln.remove()
    for t in list(ax.texts):
        if t.get_text().isdigit() and t.get_alpha() == 0.55:   # yard numbers, not technique labels
            t.remove()
        else:
            t.set_clip_on(True)


def draw(p, width_in=5.8, ax=None, keep_numbers=False, **kw):
    """Play.draw, sized for the page, cleaned for a box-sized window."""
    fld = p.draw(width_in=width_in, ax=ax, **kw)
    if not keep_numbers:
        clean(fld.ax)
    return fld


def front(spec, lbs=(), others=()):
    """Defensive linemen from (key, technique, side) triples, labelled with their technique number,
    plus (key, d, w, label) linebackers / edges and any other defenders."""
    out = [D(k, "DL", 1.0, T(t, s), t) for k, t, s in spec]
    out += [D(k, "LB", d, w, lab) for k, d, w, lab in lbs]
    out += [D(k, "S", d, w, lab) for k, d, w, lab in others]
    return out


def secondary(nickel_w=None, high=11.5, safety_w=7.5):
    """Two corners on X and Z, two deep safeties and (optionally) a nickel back at apex width."""
    out = [D("LCB", "CB", 7.0, -15.0, "C"), D("RCB", "CB", 7.0, 15.0, "C"),
           D("FS", "FS", high, -safety_w, "FS"), D("SS", "SS", high, safety_w, "SS")]
    if nickel_w is not None:
        out.append(D("NB", "NB", 4.6, nickel_w, "$"))
    return out


def box_outline(fld, p, w0, w1, d0=-0.2, d1=6.6, text=None, text_at=None):

MARK_R = 0.62


def tracking(play):
    df = play.to_tracking()
    df["_id"] = df["displayName"].astype(str)
    return df


def panel(ax, play, df, t, lateral, window, rings=(), enter=(), ring_r=0.85, relabel=None):
    """One moment of a simulated play on `ax`: markers, trails of on-screen players, rings.
    `relabel(key, t)` may return a new label for a player at time t (a lineman who changes technique)."""
    tmap = df.groupby("frameId")["time"].first()
    fid = int((tmap - t).abs().idxmin())
    ylo, yhi = sorted((play.ball_y - lateral[0], play.ball_y - lateral[1]))
    fld = Field(ax=ax, orient="vertical", los=play.los, window=window, numbers=False, hashes=False,
                lateral=(ylo, yhi), to_go=play.to_go)
    cur = df[df["frameId"] == fid]
    snap = df[df["frameId"] == int((tmap - 0.0).abs().idxmin())]
    first = snap[snap["x"].between(play.los + window[0] + 0.8, play.los + window[1] - 0.8)
                 & snap["y"].between(ylo + 0.8, yhi - 0.8)]
    ids = set(first["_id"]) | {"football"}
    inside = cur[cur["x"].between(play.los + window[0], play.los + window[1]) & cur["y"].between(ylo, yhi)]
    ids |= set(inside["_id"]) & set(enter)
    shown = cur[cur["_id"].isin(ids) | (cur["side"] == "ball")]
    trails = df[(df["frameId"] <= fid) & df["_id"].isin(ids) & (df["side"] != "ball")]
    for (_, side), g in trails.groupby(["_id", "side"]):
        xs, ys = fld.xy(g["x"].values, g["y"].values)
        ax.plot(xs, ys, color=style.OFFENSE if side == "off" else style.DEFENSE, lw=0.9, alpha=0.4, zorder=3)
    ppy = fld.pts_per_yard()
    for _, r in shown.iterrows():
        x, y = (float(v) for v in fld.xy(r["x"], r["y"]))
        if r["side"] == "ball":
            ax.add_patch(Ellipse((x, y + 0.45), 0.7, 0.42, fc=style.BALL, ec="white", lw=0.5, zorder=9))
            continue
        lab = str(r.get("label", "") or "")
        if relabel is not None and r["side"] == "def":
            lab = relabel(str(r["displayName"]), t) or lab
        if r["side"] == "off":
            ax.add_patch(Circle((x, y), MARK_R, fc=style.OFFENSE, ec="white", lw=0.6, zorder=8))
            tc = "white"
        else:
            ax.add_patch(RegularPolygon((x, y), 4, radius=MARK_R * 1.3, orientation=np.pi / 4,
                                        fc=style.DEFENSE_TINT, ec=style.DEFENSE, lw=1.0, zorder=8))
            tc = style.INK
        fs = max(4.0, min(8.0, ppy * (0.85 if len(lab) <= 1 else 0.62)))
        ax.text(x, y, lab, ha="center", va="center", fontsize=fs, color=tc, fontweight="bold", zorder=10)
    for key, col in rings:
        r = cur[cur["displayName"] == key].iloc[0]
        x, y = fld.xy(r["x"], r["y"])
        ax.add_patch(Circle((float(x), float(y)), ring_r, fc="none", ec=col, lw=1.6, zorder=7))
    for patch in ax.patches:
        patch.set_clip_on(True)
    return fld


def strip(moments, lateral, window, notes=None, rings=(), ncols=2, width_in=5.2, after=None, enter=(),
          relabel=None):
    """Numbered frame strip. `moments` is a list of (play, t, tag)."""
    cache = {}
    nrows = int(np.ceil(len(moments) / ncols))
    pw = width_in / ncols
    span_w, span_d = lateral[1] - lateral[0], window[1] - window[0]
    fig, axes = plt.subplots(nrows, ncols, squeeze=False,
                             figsize=(width_in, nrows * (pw * span_d / span_w + (0.62 if notes else 0.25))))
    for ax in axes.flat[len(moments):]:
        ax.axis("off")
    for i, (ax, (play, t, tag)) in enumerate(zip(axes.flat, moments)):
        if id(play) not in cache:
            cache[id(play)] = tracking(play)
        fld = panel(ax, play, cache[id(play)], t, lateral, window, rings, enter, relabel=relabel)
        if after:
            after(fld, play, t, i)
        lab = "snap" if abs(t) < 1e-6 else f"{t:+.1f} s"
        ax.set_title(f"{i + 1}.  {lab}" + (f"   {tag}" if tag else ""), fontsize=8.3, color=style.INK_2, loc="left")
        if notes:
            ax.text(0.5, -0.03, notes[i], transform=ax.transAxes, ha="center", va="top",
                    fontsize=7.1, color=style.INK)
    fig.tight_layout(h_pad=1.3 if notes else 0.6, w_pad=0.6)
    return fig


def video(segments, lateral, window, rings=(), width_in=4.6, after=None, enter=(), relabel=None):
    """Web version: each (play, tag, [(t, note), ...]) segment played in turn, notes as captions."""
    import shutil
    from matplotlib import animation
    from IPython.display import HTML, display
    span_w, span_d = lateral[1] - lateral[0], window[1] - window[0]
    fig, ax = plt.subplots(figsize=(width_in, width_in * span_d / span_w + 0.45))
    frames = []
    for play, tag, notes in segments:
        df = tracking(play)
        tmap = df.groupby("frameId")["time"].first()
        frames += [(play, df, float(tmap[f]), tag, notes) for f in sorted(df["frameId"].unique())]

    def update(k):
        play, df, t, tag, notes = frames[k]
        ax.clear()
        fld = panel(ax, play, df, t, lateral, window, rings, enter, relabel=relabel)
        if after:
            after(fld, play, t, -1)
        txt = ("snap" if abs(t) < 0.05 else f"{t:+.1f} s") + (f"  {tag}" if tag else "")
        cur = [n for tt, n in notes if tt <= t + 1e-6]
        ax.set_title(txt + (("\n" + cur[-1].replace("\n", " ")) if cur else ""), fontsize=8.5, color=style.INK,
                     loc="left")
        return []

    anim = animation.FuncAnimation(fig, update, frames=range(len(frames)), interval=100)
    html = (anim.to_html5_video(embed_limit=40).replace("<video ", '<video style="max-width:100%;height:auto" ')
            if shutil.which("ffmpeg") else anim.to_jshtml(default_mode="once"))
    plt.close(fig)
    display(HTML(html))


def animation_or_strip(moments, notes, lateral, window, rings=(), after=None, enter=(), ncols=2, width_in=5.2,
                       relabel=None):
    """Video on the website; a numbered, captioned frame strip in the PDF."""
    if pdf_build():
        strip(moments, lateral, window, notes=notes, rings=rings, ncols=ncols, width_in=width_in,
              after=after, enter=enter, relabel=relabel)
        plt.show()
        return
    segs = []
    for play, t, tag in moments:
        if not segs or segs[-1][0] is not play:
            segs.append((play, tag, []))
    for (play, t, tag), n in zip(moments, notes):
        for s in segs:
            if s[0] is play:
                s[2].append((t, n))
    video(segs, lateral, window, rings=rings, after=after, enter=enter, relabel=relabel)

The graphic in the corner of the screen says the defense is in a 3-4. You count. Four players at the line of scrimmage, two of them standing up, two with a hand on the ground. Next snap, after a substitution you barely notice (a huge man jogging on, a linebacker jogging off), there are five across the line, three of them enormous and wedged inside the offensive tackles, with a single linebacker behind them. The offense hands the ball to its running back on an inside run, the kind that worked all afternoon last week, and he runs into a wall two yards deep. No blocker ever touched the linebacker. “That’s their tite front,” says the analyst. “Three big bodies inside, and nobody can get to the second level.” So which is it: a 3-4, a four-man line, a five-man line? And how did a defense with only six men near the ball stop a run aimed at six blockers? This chapter is about the other family of fronts, the odd ones, and about why, in the modern game, the names on the graphic tell you less than the picture on the field.

You’ll be able to:

  • Look at a front before the snap, find whether anyone is head-up on the center, and say whether it is odd or even.
  • Diagram the 3-4 Okie, and tell from the linemen’s alignment and first step whether it is being played two-gap (control and read) or one-gap (attack).
  • Recognize a tite (or mint) front, explain why it walls off inside runs against spread offenses, and name what it gives up.
  • Identify the nickel fronts (4-2-5, 3-3-5, 2-4-5) by counting hands on the ground and players standing on the edges, and explain why a roster label and the picture so often disagree.
  • Explain how a defense picks a front to stop inside zone and a different one to stop outside zone.
  • Decode front dialect (Okie, Eagle, G, reduced) and know when one word means different things in different buildings.
NoteYou’ll need
  • Even Fronts: an even front has nobody head-up on the center (the 4-3 Over puts its 3-technique toward the tight end, the Under away from him), and every even front has a rule, usually tied to the tight end, that tells you where the linemen will line up.
  • Defensive Line Play: technique numbers from the defense’s side (the 0 nose, the 3, the 4i, 4 and 5 on the tackle, the 9 and the wide-9), one-gap versus two-gap, setting the edge and contain.
  • A First Look at Defense: base defenses (the 4-3 and the 3-4), nickel, the box and the light box, and two-high safety shells.
  • Zone Running: inside and outside zone, double teams that climb to linebackers, and the cutback.

How to read the diagrams. As in the rest of Part 5: offense is blue circles, defense orange squares, plays go up the page, and left and right are always the offense’s left and right. Defensive linemen are labelled with their technique number (a 0 is a nose head-up on the center, a 4i is on a tackle’s inside shoulder). E is an edge player standing up at the end of the line; M and W are inside linebackers; $ is the nickel back; C a cornerback; FS and SS the safeties. Offensive linemen are LT, LG, C, RG, RT, shortened to T G C G T in the smaller panels; R is the running back and F a fullback. If nobody is head-up on the center, the front is even: the defensive tackles line up on or around the guards, the center is uncovered or only shaded, and you are looking at one of the four-man fronts of Even Fronts. If a big man stands right over the ball, nose to nose with the center, the front is odd: it is balanced around a 0-technique nose tackle. The classic odd fronts have three down linemen (the 3-4, which puts five on the line once you count its two standing outside linebackers) or five (the old 5-2), rather than four.

Why does one alignment decide the family? Because of what it does to the offensive line’s arithmetic. In an even front, nobody is square in front of the center: at most a tackle shaded on one of his shoulders, a man he can work on together with a guard, and often nobody at all, in which case he is free to help or to climb to a linebacker. The defense has eased the center’s job in exchange for putting its tackles where they can attack the guards. In an odd front, the center has a man in his face on every snap, and a nose head-up on him is responsible for both A gaps, one on each side of the center. The center can’t block him alone at a good angle in either direction, so one of the two guards usually has to help. That is the founding idea of every odd front: put a heavy man on the center, make the offense spend two blockers on him, and let somebody else run free. You met that nose in Defensive Line Play. This chapter is about the fronts built around him.

A warning before we start, which the rest of the chapter will keep proving: “odd” and “even” describe where linemen stand, not who is on the roster. A team that calls itself a 3-4 team plays an even front on plenty of snaps, and a 4-3 team can put a nose on the center. Keep your eyes on the center.

The 3-4 Okie

The 3-4 Okie is the classic 3-4, the one drawn in every coaching book: a nose head-up on the center, a defensive end on each offensive tackle, a stand-up edge at each end of the line, and two inside linebackers behind the guards. The name is short for Oklahoma, because the alignment is Bud Wilkinson’s Oklahoma 5-2 of the late 1940s and 1950s with its two outside men standing up (more on that history below).

Read Figure 30.2 as a list of jobs, the way you learned to in Defensive Line Play.

  • The nose takes on the center, and very often a guard as well. He is usually the heaviest man on the defense, because his whole value is in not moving.
  • The ends sit on the offensive tackles. In the classic version they are head-up there, in a 4-technique, and they control the tackle and play the gap on either side of him. Others line up on the tackle’s outside shoulder, a 5. They are long and strong, the 290-to-320-pound bodies built to lock out a tackle at arm’s length.
  • The outside linebackers are standing up at each end of the line. Against the run they set the edge; against the pass, one or both rush; and on some snaps one drops into coverage. They are the lightest players in the front, edge rushers in all but stance.
  • The inside linebackers stand behind the guards, and here is the design’s point. Count the offensive linemen who don’t have a defender on or near them before the snap: the two guards. In zone-blocking terms they are uncovered (Zone Running), and an uncovered guard wants to climb straight to a linebacker. The Okie’s whole design is to make that impossible: the nose is too much for the center alone, so one guard has to stay and help; the ends hold the tackles so they can’t climb either. If the front does its job, at most one blocker reaches the second level, and two linebackers are left to make tackles.

One alignment, two ways to play it

A 3-4 can be a two-gap front, in which the three linemen control blockers and read, or a one-gap front, in which every defender attacks a gap at the snap. The two-gap 3-4 is the old-school version, and its appeal is the math in the left half of Figure 30.3. Three linemen can hold six gaps if each of them can stand up a blocker and play either side of him. That leaves the two inside linebackers gap-free: they don’t have to guess at the snap; they watch the play develop and run to the ball, and the linemen in front of them keep the blockers off them. (Against one back, that is. Against a fullback, like the one in Figure 30.2, the offense has an extra blocker who can lead through a hole, and one inside linebacker has to meet him there while the other fills off the nose; Linebackers has that fight.) Bill Parcells’s New York Giants of the 1980s played it, with Bill Belichick as his defensive coordinator, and Belichick’s New England Patriots of the 2000s played it with linemen like Vince Wilfork, Richard Seymour and Ty Warren, all built like the job: big, long and patient.1

The cost, as Defensive Line Play explained, is that two-gappers are always reacting. They rarely make plays behind the line or sack the quarterback, they need bodies the draft produces only a few of each year, and on passing downs the defense usually has to take the nose off for someone quicker.

The one-gap 3-4 keeps the personnel and changes the job. Line the ends up shaded into gaps, shade the nose, and tell every defender to attack his gap at the snap, and the 3-4 becomes an attacking front. The right half of Figure 30.3 shows the most common way to do it, and it holds a secret: lined up this way, a “3-4” is almost exactly the 4-3 Under of Even Fronts, a nose shaded to the tight end, a 3-technique away from him, a 5 and a 9 to the tight end’s side, with the Sam linebacker called an outside linebacker and standing up. Run this chapter’s one-glance test on it: nobody is head-up on the center any more, so by alignment this “3-4” is an even front.

That is why “3-4” by itself is a weak clue. The number of linemen tells you how many there are; their alignment and first step tell you how they play. A two-gapper’s first step goes into the blocker, hands first, and he stays at the line. A one-gapper’s goes into a gap, and he is across the line before the blocker has finished his first step. Watch the ends for two snaps and you will know which 3-4 you are watching.

ImportantWhy it exists

The 3-4 trades linemen for linebackers, and two-gap play is what makes that trade affordable. Take a defensive lineman off the field and put a linebacker on, and you gain a player who can rush, drop into coverage or fill a run lane, and who keeps the offense guessing which of those he’ll do. You lose one big body at the line. The two-gap 3-4 covers the loss with technique: three linemen hold six gaps, and the linebackers run free. The one-gap 3-4 covers it with movement and speed: every man attacks, and the extra linebacker is just one more fast player who can come from anywhere. Either way, the defense is betting that a fourth linebacker is worth more than a fourth lineman. In an era of spread offenses, three-receiver sets and quarterbacks who throw on first down, that bet has mostly paid.

Where the 3-4 came from (and why nobody agrees)

Ask who invented the 3-4 and you will get at least four answers, each true about something. The honest version is a family tree, not a birth certificate.

The ancestors: five men on the line. In 1947 Earle “Greasy” Neale, the Philadelphia Eagles’ head coach, built his defense around a five-man line with two linebackers behind it. It was called the Eagle defense, and the Eagles won NFL championships with it in 1948 and 1949.2 Bud Wilkinson took the five-man line to the University of Oklahoma, where it became the Oklahoma 5-2, the defense behind his Sooners’ 47-game winning streak in the 1950s. (That is where Okie comes from.) Now look at a 5-2 the way a coach of the 1970s did. The two men at the ends of the line were defensive ends, but they were usually smaller and quicker than the three linemen inside them, and their jobs were to set the edge and, on some plays, to drop. Stand them up, call them linebackers, and the same seven players become three linemen and four linebackers. Nobody had to invent the 3-4; it was hiding inside the 5-2.3

The NFL arrivals, early 1970s: several claimants.

  • Bill Arnsparger, Miami, 1972. The Dolphins’ defensive coordinator under Don Shula had a linebacker, Bob Matheson, who could play with his hand on the ground or standing up. Arnsparger moved him around the front, sometimes as a fourth linebacker, sometimes as a rushing end, and the package was named for his jersey: the 53 defense. The Dolphins went 17–0. Was it a 3-4? On some snaps, yes, and it is often named as the 3-4’s NFL ancestor; the story belongs to The 1978 Liberation.4
  • Chuck Fairbanks, New England, 1974. Fairbanks had coached the Oklahoma 5-2 as the Sooners’ head coach, and when he took over the Patriots he brought it with him, standing up its ends. His 1974 Patriots are usually named as the first NFL team to use the 3-4 as their regular base defense.5
  • Bum Phillips, Houston, 1974–75. Phillips, the cowboy-hatted Texan who had devised the technique numbers you have been using since The Language of the Run Game, was the Oilers’ defensive coordinator under Sid Gillman, and had talked Gillman into a 3-4, when they traded for nose tackle Curley Culp in October 1974; he kept the 3-4 when he became head coach in January 1975. Culp, a former NCAA heavyweight wrestling champion at Arizona State, was strong enough to need two blockers, and Phillips gave him the credit: “Curley made [the 3-4 defense] work. He made me look smart.”6

The boom and the bust. Within a few years the 3-4 was everywhere. John Madden’s Raiders switched to it in 1976, after injuries thinned their defensive line, and won the Super Bowl; Arnsparger’s Dolphins went to it full time in 1977; and by the late 1970s and early 1980s it was the league’s most common base defense.7 Then offenses changed again, the one-back passing games of the 1980s and 1990s made four-man fronts with fast defensive ends fashionable, and by the mid-1990s only a handful of teams, most notably Buffalo and Pittsburgh, still lined up in a 3-4. The revival came in the 2000s, led by the Steelers and Belichick’s Patriots, and took a new shape that the rest of this chapter is about: the stand-up edge rusher, the one-gap 3-4, and in the 2020s the five-man tite front. The longer arc, including the 3-4’s role in the rise of the hybrid edge (the stand-up edge below, treated as a position of its own: part end, part linebacker), is in Dynasties and Counterpunches and The Single Wing to the Scoring Drought.

So who invented it? Wilkinson’s 5-2 supplied the alignment, Arnsparger the idea of a hybrid end-linebacker, Fairbanks and Phillips the commitment to it as a base defense, and Madden the Super Bowl that sold it. That is the usual shape of football history: a problem (offenses running more and throwing more on early downs), several smart people solving it in the same few years, and credit that depends on whom you ask.

The stand-up edge

The 3-4’s most important legacy isn’t the nose. It is the player at the end of the line who isn’t sure whether he is a lineman or a linebacker, and doesn’t need to be.

A stand-up edge lines up at the end of the line of scrimmage in a two-point stance, standing, instead of with a hand in the dirt. In a 3-4 he is called an outside linebacker, and the roster lists him that way; on the field, most snaps, he is an edge rusher doing a defensive end’s job. Lawrence Taylor in the 1980s, Kevin Greene and Greg Lloyd in the 1990s, DeMarcus Ware and Von Miller in the 2000s and 2010s, T.J. Watt today: the most feared pass rushers of the 3-4 era stood up.

Why stand? Three reasons, each a trade.

  1. Vision and options. Standing, he sees the whole backfield, and he can move in any direction at the snap: rush, take on a blocker, or turn and drop into coverage. A lineman in a three-point stance, weight forward on his hand, is committed to going forward. The offense can’t rule out a drop from a standing edge, so it can’t be sure who is rushing. That uncertainty is the seed of the simulated pressures of Part 7: four-man rushes in which a lineman drops into coverage and a player the offense didn’t count rushes instead.
  2. Width and a view of the tackle. Standing, he can align wider and watch how the tackle sets before he commits to an inside or outside move. It isn’t a free win: plenty of rushers still prefer a hand down for a faster get-off on obvious passing downs.
  3. Disguise. A front with two standing edges can become a four-man line, a five-man line or a three-man rush without anyone substituting. That is the trick behind the opening of this chapter, and it gets its own animation below.

What it costs. Mostly leverage against the run. A defender in a three-point stance comes out low, and the low man wins at the line; a standing edge has to sink as he takes on a tackle or a tight end, or he gets pushed upfield and the edge goes soft. That is why the stand-up edges who last are the ones who can set the edge as well as rush (in Defensive Line Play’s terms, who keep their outside arm free against a 300-pound tackle), and why many of them put a hand down in short yardage and at the goal line, where leverage is everything. The most common hand-down moment of all is on passing downs in nickel, when the 3-4 outside linebacker becomes, in everything but the roster, a 4-2-5’s defensive end (more on that below). The body type is in between, too: usually 245 to 265 pounds, lighter than a defensive end, heavier than an inside linebacker. Teams that play four-man fronts draft the same players and call them defensive ends. The player is the same; the label follows the scheme.

CautionCommon misconception

“A 3-4 outside linebacker is a linebacker, so he covers people.” Some do, some of the time. Most of the great ones spend the large majority of their passing-down snaps rushing the quarterback. When you see a 3-4 outside linebacker drop into coverage, it is usually a deliberate surprise, designed to put a defender where the quarterback expected an empty space, not his everyday job.

The Phillips branch: the attacking 3-4

The 3-4’s revival in the 2000s split along the line from Figure 30.3. Belichick’s Patriots and the Steelers kept versions that leaned on two-gap play up front. The other branch came from one family, the Phillipses.

Wade Phillips, Bum’s son and his linebackers coach in Houston from 1976 to 1980, spent four decades as a coordinator and head coach running his father’s 3-4, and running it the attacking way, not the two-gap way the Parcells and Belichick branch made famous.8 Phillips’s linemen were shaded into gaps and attacked them; his outside linebackers rushed; the ends were penetrators, not anchors. When he took over the Houston Texans’ defense in 2011, the team’s own preview called it “his one-gap scheme,” and the line coach said it was “not like a two-gap type defense where you need that 350-pound nose man.”9 The proof arrived the following year, when the Texans’ 5-technique end, J.J. Watt, led the league with 20.5 sacks and was Defensive Player of the Year: the end of a 3-4, the position that used to exist to let linebackers make tackles, had become the defense’s best pass rusher.10

Four years later Phillips took the same ideas to Denver, where the 2015 Broncos allowed the fewest yards in the league and won Super Bowl 50 with two stand-up edges, Von Miller and DeMarcus Ware, chasing the quarterback.11 Their film room is below.

A two-gap 3-4 and a one-gap 3-4 use the same eleven bodies and ask opposite things of them. The modern descendants of both branches meet in one front, which is next: it has a two-gap nose like Belichick’s and one-gap linemen beside him like Phillips’s.

The tite front: walling off the middle

The tite front is an odd front with five men on the line: a nose on the center, two defensive linemen in 4i-techniques on the inside shoulders of the offensive tackles, and an edge player outside each tackle. Fangio and his coaching tree call it tite; Kirby Smart’s Georgia calls the same picture the mint front; and, as the dialect section below shows, some playbooks call it Eagle.

It is a reduced front on both sides. In a reduced front, the defensive linemen slide a gap inward from their usual spots and an edge player takes the outside gap they left. The 4-3 Under of Even Fronts reduces one side (its backside end moves down toward the guard); the tite takes the Okie and reduces both ends, from the tackles’ outside shoulders (5s) to their inside shoulders (4is). That one move inside is the whole front.

Why move the ends inside? Put a nose on the center and a 4i in each B gap, and the whole interior of the line, from one tackle’s inside shoulder to the other’s, has a big body in it. Now run the covered-or-uncovered test from Zone Running for inside zone to the offense’s right. The center is covered (the nose is on him), the right guard is covered (the 4i is on his play side) and so is the left tackle. The left guard and the right tackle are uncovered, and in an ordinary front those are the two men who would climb to linebackers. Here they can’t, because each stands next to a defender who can’t be blocked one-on-one: the left guard has to help the center with the nose, and the right tackle has to help the right guard with the 4i. The offense can still double-team, but every double team is spent on a lineman, and every blocker who stays on a lineman is a blocker who isn’t on a linebacker.

If that sounds like the bear front of Even Fronts, it should: both fronts decide that no interior blocker will be free. The bear does it by covering the center and both guards (a 0 and two 3s), and the 46 that made it famous put eight men near the ball to fill everything else. The tite sits on the center and the tackles’ inside shoulders, which does the same job against the inside runs with five on the line and two safeties still deep. The price of that choice, as we’ll see, is the space outside the tackles, where the bear had bodies and the tite has only its edges.

Walk through Figure 30.4 with the zone rules from Zone Running in mind. Inside zone wins with double teams: two linemen drive a defensive lineman backward, and when the linebacker behind commits, one of them comes off and blocks him. That requires two things: a lineman the pair can move, and a linebacker within reach. The tite takes away both.

  • The doubles can’t move their men. The nose plays the center and whichever guard joins the double: what analysts of these light-box fronts call “gap-and-a-half” play, in which a big two-gap nose controls one A gap and leans into the other.12 The 4i sits in the B gap, exactly where inside zone’s aiming point is, so the right guard and the right tackle have to block him together. Neither can come off to climb: the 4i is already in the hole, so the moment one of them leaves, the man who stays is blocking him from the wrong side and the hole closes.
  • Nobody climbs. With every blocker spent on a lineman, the one inside linebacker is never touched. That doesn’t mean he has no job. He owns a gap like everyone else (the next section counts them), but he can fill it without having to beat a block first, and the wall in front of him has already told the running back where not to go.
  • The cutback is plugged. Inside zone’s best friend is the bend back to the weak side. In the tite, the backside A and B gaps have the nose and the backside 4i in them, and the backside edge is squeezing down the line behind them. He squeezes square, though, not flat down the line: against a shotgun team the quarterback can pull the ball and run where the edge just left, so the edge closes the gap without giving up the quarterback.

What’s left for the runner is outside the tackle, and here it matters how he gets there. A bounced inside run is late and alone: the back has pressed the B gap, stopped, and turned, and while he did, the Mike, the play-side edge and a safety coming down have had a full second to get there. That is the trade the tite offers on purpose. It turns the inside runs that win against two-high defenses into bounced runs that its linebacker and safeties can chase down. A run designed to go outside is a different matter, because it arrives with its blockers already out there. That is the tite’s weak spot, and we’ll come to it.

Why the tite exists now: six men, seven gaps

The tite has existed in college for a long time, but it took over the NFL conversation in the late 2010s for a reason you have met before: two deep safeties. Defenses facing modern passing offenses want to keep both safeties deep, as long as possible (A First Look at Defense has the numbers: split-safety looks reached a record 42 percent of dropbacks in 2025).13 Two safeties deep and five defensive backs on the field leaves six men in the box. Against an offense with a tight end attached, that is six defenders for six blockers and seven gaps. By the one-man-per-gap accounting of one-gap football, the numbers don’t work.

Figure 30.5 shows how the tite makes the numbers work. One nose who holds both A gaps is the extra defender the box doesn’t have: five men on the line take six of the seven gaps, the Mike takes the last one without a blocker in his way, and the safeties can stay home on the pass and fit the run late, from depth, with the alley outside the edges left to a safety or the nickel. The 4is are one-gap players (each has his B gap), but they play it from an inside shade, by locking onto the tackle rather than darting past him, which is why they need the length and strength of two-gappers.14 The assignments vary by staff and by how many linebackers are on the field. A college tite with two inside linebackers often lets the nose play just one A gap, the one away from the center’s first step, with a linebacker filling the other;15 the NFL’s nickel tite, with one linebacker, leans on the nose to hold both.

The Fangio tree, Vic Fangio himself and the coaches who worked under him, Brandon Staley among them, built their light-box run defense on exactly that bargain. Fangio’s 2018 Chicago Bears, his last season as a coordinator before he became Denver’s head coach, played their 3-4, often in the tite, with big interior linemen (Akiem Hicks and the nose Eddie Goldman) and two stand-up edges (Leonard Floyd and Khalil Mack, acquired in a trade the week before the season), and allowed the fewest points in the league (283) while leading it in takeaways (36).16 The light-box bargain is clearest in Staley’s 2020 Rams, who by one count played a light box on about 78 percent of snaps, the second-highest rate in the league, and allowed the fewest points in the NFL (296).17 Fangio himself is still at it: he is the Eagles’ defensive coordinator in 2026, his third season there.18

What the tite gives up

Every front is a trade. The tite’s costs are specific, and offenses know them.

  • The pass rush. Three of its five linemen are big interior players built to hold blocks, not beat them; the 4is especially are aligned to take the run away, not to rush. To get four rushers, the tite often has to rush both edges and two interior men, or send an off-ball linebacker. Cody Alexander, one of the most widely read writers on defensive structure, names this as the front’s main drawback: “its inability to get a rush without sending an off-ball defender.”19
  • The perimeter. With three big bodies inside the tackles, the defense outside them is thin: one edge on each side, a linebacker flowing from the middle, and safeties coming from depth. Runs that are designed to start outside, outside zone, tosses, the jet sweeps of Perimeter and Misdirection, arrive there with blockers in front of them and go straight at that thinness. The next section shows how.
  • Angles for gap runs. Linemen lined up inside their blockers hand a gap scheme easy down blocks: on power to the right, the right tackle blocks down on the 4i and the right guard down on the nose, both from the side. The tite survives power with technique, not alignment: the 4i squeezes the down block and closes the B gap anyway, the edge meets the pulling guard and spills the ball back inside, and the lone linebacker fits off the spill. If any of the three loses, there is nobody behind them but safeties. (Run Fits and Box Math draws a full fit against power.)
  • The quick game and the RPO. A light box with a single linebacker leaves big spaces underneath. The RPO was built for exactly this defender: a lone linebacker who must fill the run will leave the space behind him open for a throw.
  • Rare players, again. Like the two-gap 3-4, the tite needs a nose who can hold two blockers and two 4is with the length to play from an inside shade. The front is only as good as those three men.

Two practical notes before we leave it. First, the tite is balanced, so unlike the Over and Under it has no strength call for its interior: the 0 and the two 4is stay put wherever the tight end lines up, and only the edges and the Mike adjust (the edge on the tight end’s side widens to his outside shoulder, and the Mike’s gap goes with him). Second, when the offense detaches its tight end and goes 2x2 or empty, there are only five blockers and six gaps, and the five-man line covers all six by itself, so the Mike becomes a truly free player. That sounds ideal, but those formations are usually passing formations, and a five-man line that rushes four is exactly the problem in the first bullet above, which is one reason defenses so often meet them with a four-man front instead.

Inside zone or outside zone? How a front chooses

The tite is the clearest example of a principle that runs through all of defensive front design, and it is the best way to understand why a defense changes fronts from week to week: every alignment makes some blocks easier and others harder, and a defense picks the alignment that makes the offense’s favourite blocks hard.

Recall the two zone plays from Zone Running:

  • Inside zone is won by double teams that move a defensive lineman back and then climb to a linebacker. Its aiming point is the play-side B gap.
  • Outside zone is won by reach blocks: each blocker steps sideways and tries to get his head outside the defender in front of him. Its aiming point is the tight end’s outside leg, and its biggest gains come from the edge being reached (the bounce) or the defense over-running the play (the cutback).

Now ask what kind of alignment makes each one hard.

The rule of thumb that falls out of Figure 30.6 is simple to state:

  • To beat the double team, line up on top of or just inside the blockers. A nose on the center and 4is inside the tackles leave the offense’s two uncovered linemen stuck on doubles; the pairs can’t move their men and can’t leave them. That is the tite’s answer to inside zone and duo, the double-team runs. (Gap runs like power, as the last section showed, are a fight it wins with technique, not with alignment.)
  • To beat the reach, line up outside the blockers, and leave no gap empty on the edge. A reach block is hardest when the defender is already a step to the play side of the blocker: a 3-technique outside the guard, a 5 outside the tackle, a 9 or a wide edge outside the tight end. Against outside zone the tite’s problem isn’t that its linemen are easy to reach; the 4i is actually a hard reach for the guard. It is that the C gap is empty, so the play-side tackle is free to help on the edge or climb to the lone linebacker, and that its biggest men have to run sideways. That is why the classic answer to an outside-zone team is a wider front, with shades outside the blockers, often a four-man even front with a 3-technique who can penetrate before the guard’s reach lands, as in Defensive Line Play’s one-gap animation.

The two answers pull in opposite directions, which is why no front stops everything. A tite that pulls its ends inside to choke inside zone has emptied the gaps outside the tackles that outside zone attacks; a wide four-man front that beats the reach leaves the interior gaps to linebackers who can be climbed to. Defenses solve the dilemma in three ways, and you can watch for all of them:

  1. Change the front by opponent. A defense facing an inside-zone and duo team plays a lot of tite; the same defense facing a wide-zone team widens its edges and plays more four-man fronts. Fangio’s own defenses varied in exactly this way: Alexander notes that in Denver Fangio favoured a four-down front more often than his former assistant Staley, who preferred the five-man front.20
  2. Change the front by down, distance and formation. Tite on first-and-10 against two tight ends, four-man front on second-and-long against three receivers.
  3. Change the front after the offense lines up. Shift from one to the other at the last moment, so the offense’s blocking calls are made against the wrong picture. That is the last section of this chapter.

Who, in either front, makes the tackle on the run that gets through? That is the job of the run fits, the full eleven-man accounting of who owns which gap and who forces the ball back inside, in Run Fits and Box Math.

Nickel fronts: 4-2-5, 3-3-5, 2-4-5

Everything so far has been about base defenses. But the modern NFL plays nickel, five defensive backs, on most snaps: about 60 percent of all runs and passes in 2025, with base (four defensive backs) on about 30 percent and dime on most of the rest.21 So the front you will see most often on Sunday is a nickel front, and nickel fronts have their own three-number names. All three end in 5: five defensive backs. The first two numbers split the other six players between the line and the linebackers.

Why can’t a 3-4 team just stay in its Okie against three receivers? Because the third receiver has to be covered by somebody. In the Okie that somebody is an outside linebacker, who walks out from the end of the line to stand over the slot. That leaves a four-man front in the box with one soft, empty edge where he used to be, and a 250-pound rusher in coverage on a receiver. Taking a big man off for a nickel back, and rebuilding the front from what is left, is the better trade. The three nickel fronts are the three ways to rebuild it.

The 2-4-5 is what a 3-4 team usually plays when the offense brings out a third receiver: the nose tackle comes off for a nickel back, the two ends slide inside to play the tackle spots, and the two outside linebackers stay on the edges, standing. You met it as a footnote in A First Look at Defense; look at it now in Figure 30.8, next to a 4-2-5, and the two are the same front. Four players on the line, two linebackers behind them, five defensive backs. If the 2-4-5’s outside linebackers put their hands down, you couldn’t tell them apart at all.

So why the different name? Because the name is built from the roster, not the picture. The 2-4-5’s edges are listed as linebackers because, in the team’s base defense, they are the 3-4’s outside linebackers. Bring the same two players onto a 4-3 team and they would be listed as defensive ends, and the identical alignment would be called a 4-2-5.

3-3-5: three meanings

The 3-3-5 is three down linemen, three linebackers and five defensive backs, and it is the one name in this section that means different things depending on who says it.

  • When an NFL coach calls a 3-3-5 front, he means a five-man line in nickel: three linemen inside, a stand-up edge on each end, and one linebacker behind them. The tite of Figure 30.4 is exactly this, the 3-4 team’s nickel with its nose back on the field. It is the light-box run front of the Fangio tree, and the Staley branch of that tree calls it Penny, a nickel with a nose.22
  • When the roster count says 3-3-5, it usually means something else: a four-man line with one edge listed as a defensive end and the other as a linebacker. That is what most of the 3-3-5 snaps in the data below are.
  • In college, “3-3-5” usually means the 3-3-5 stack, a different defense with three linemen and three linebackers stacked directly behind them, usually credited to Joe Lee Dunn at Memphis in 1991 and spread in the 1990s as a way to put more speed on the field against spread offenses.23 It belongs to The College Laboratory.

Which one you are looking at, the hands will tell you: three down inside the tackles with an edge on each side is the coach’s 3-3-5; four men on the line is the roster’s.

That is why this book, and every coach you will read, asks you to count hands, not names. Before the snap, count the players with a hand on the ground and the players standing at the ends of the line, and note where the man nearest the center is. Four down is a four-man line, whatever the graphic says. Two down with two standing edges is a four-man line too, built from 3-4 personnel. Three down inside the tackles with an edge on each side is a tite, five on the line. The personnel label tells you which players are on the field, which matters for what they can do (a 3-4 outside linebacker can drop into coverage more comfortably than a 4-3 end); the alignment tells you what they are about to do.

Multiple defenses: one personnel group, many fronts

If the label is personnel and the front is alignment, nothing stops a defense from playing several fronts with the same eleven players. Almost every NFL defense now does. A multiple defense (also called a hybrid defense) lines up in more than one front family, odd and even, four-man and five-man, often from the same personnel and sometimes switching between them before the snap. “Now every team is ‘multiple,’” one analyst wrote more than a decade ago, and it has only become more true.24

Belichick’s Patriots were the model of the idea: a defense that might be a two-gap 3-4 one week and a four-man front the next, built to take away whatever the opponent did best. The modern versions go further, changing fronts within a series, and the most effective way to do it is a shift: show the offense one front, then move into another as the quarterback is calling the protection.

Why go to that trouble? Because, as Zone Running explained, an offensive line decides its blocking before the snap: who is covered, who doubles, who climbs, and, on a pass, which way the center slides and who picks up which rusher. A front that shifts at the last moment makes those decisions against the wrong picture. In Figure 30.10 the line set its rules against a four-man even front with two linebackers; at the snap it faced a five-man line with one, and on the pass, the player it had counted as a rusher (the walked-up Mike) was in coverage. Notice what the shift does not do: it never puts a 250-pound edge rusher on a tackle’s inside shoulder. Every interior spot of the tite is still held by a defensive lineman; the only man who changes stance is the end who stands up on the outside. Georgia’s defense under Kirby Smart is known for exactly this move, from a 4-2-5 into its mint front without a substitution: one tackle to the nose, the other tackle and an end to 4is, the other end standing up and walking out.25

The cost of being multiple is complexity. Every front has its own gap assignments and its own fits, and every player has to know all of them; a shift that is a beat late leaves a gap empty at the snap. That is why multiple defenses are built around smart, versatile players (edges who can rush and drop, linemen who can play a 3 and a 4i) and why they live and die on the communication of the green-dot linebacker who relays the call.

The Eagle front has meant at least three things. In the 1940s it was Greasy Neale’s five-man line, the Philadelphia Eagles’ defense and the 3-4’s ancestor. In the late 1980s Fritz Shurmur, the Los Angeles Rams’ defensive coordinator, built a linebacker-heavy package he called the Eagle, and later wrote a coaching book on it, The Eagle Five Linebacker Defense (1993). Its usual shape had only two down linemen, with five linebackers; in the Rams’ 21–7 playoff win over Philadelphia after the 1989 season, Shurmur went further, playing zone on every snap and using five linebackers and six defensive backs, with no defensive linemen at all, on long-yardage downs.26 And in modern 3-4 playbooks, “Eagle” is a reduction, but staffs disagree about which one. In some it is the Okie with both ends pinched inside to 4is: the middle panel of Figure 30.11, the same alignment other staffs call tite or mint. In others it is a one-side change-up: an outside linebacker walks up onto the tackle and the end beside him reduces inside to a 3, which is the Under-like 3-4 of Figure 30.3.27 Several meanings, one word, and only context tells you which.

The G front is a tag rather than a whole front: it moves the nose from the center (or from his shade on the center, in a four-man front) onto a guard’s inside shoulder, a 2i. “G” is for guard. It is a small move with a specific purpose. Inside zone’s first double team is usually the center and a guard on the nose, both stepping forward into him so they can drive him back and then let one man climb to the linebacker. With the nose on the guard instead, the center has to step sideways to reach him. A sideways step carries no push, so the double forms a beat late and moves nobody, and by the time either blocker can come off it, the linebacker behind the nose has already filled. You’ll most often see the G tag on four-man Over and Under fronts, where it moves a 1-technique to a 2i. The right panel of Figure 30.11 borrows the same move for the Okie to show what it does; whether a three-man front uses the letter G for it varies by staff.28 Run the one-glance test on it, too: with the nose on a guard, nobody is head-up on the center, so a G front is, by alignment, even, whatever the base defense is called. That is the point of the tag. It changes who is covered.

Mint is the same kind of word: Georgia’s name, under Kirby Smart, for the front Fangio’s staffs call tite. The point is not to memorize every staff’s dictionary. It is to read the alignment first and attach the name second. If an analyst says “Eagle” and you see 4is inside the tackles, you know what he means even if his playbook and yours disagree.

NoteMadden vs. real life

In Madden, the defensive playbook is a menu of named fronts: pick “3-4 Odd” or “Nickel 3-3-5 Wide” and the players run to their spots. Real defenses don’t call a front from a menu so much as build it from rules. The call names a front, a tag or two (a G, a slant, who is the edge), and a coverage, and each player’s alignment then depends on where the offense puts its tight end and its back, which side is the wide side of the field, and whether the offense shifts. Two teams can call the same word and line up differently, and the same team can call two different words and end up in the same alignment. The names are shorthand for a set of instructions, and every staff writes its own.

TipWatch for it

Before the snap: count the hands, find the center.

  • Is anybody head-up on the center? A nose over the ball means an odd front; nobody there means an even one.
  • Count the hands on the ground. Four is a four-man line, whatever the graphic says. Two down plus two standing edges is also a four-man line, usually a 3-4 team’s 2-4-5 nickel. Three inside the tackles plus a standing edge on each side is a five-man tite (or mint) front.
  • Who is standing on the edges? A standing edge is a rusher who might drop; note his number and watch him on passes.
  • Where are the ends: outside the tackles (5s, built to beat the reach and set the edge), head-up (4s, two-gappers) or inside them (4is, a tite, built to choke inside runs)? From the high sideline camera, look at the two helmets: is the defender’s helmet on the guard’s side of the tackle’s helmet, on top of it, or on the far side? If you can’t tell, the end-zone replay angle (see Even Fronts) shows it at once.

At the snap: the first step and the count.

  • Did the front change in the last seconds? A shift (an edge sliding inside, a linebacker walking up) means the defense is trying to beat the offensive line’s blocking calls.
  • Do the ends step into the tackles’ chests (two-gap) or into the gaps (one-gap)?
  • On a pass, how many rushed, and who? Five on the line rushing four means somebody standing at the line dropped.

After the play: where did the run go?

  • Against a tite, did the inside run get through, or did it bounce outside? If it bounced, who made the tackle: an edge, the linebacker, or a safety?
  • Did the offense’s play fit the front? Inside zone into a tite is a fight on the defense’s terms; outside zone and perimeter runs test its edges.

For your prediction log (How to Watch a Broadcast): on first and second down, write down the number of hands on the ground and whether the front was odd or even, then whether the run went inside or outside. After a quarter you’ll know which front the defense uses to stop what, and whether the offense noticed.

If the reaches fail, the back can’t get outside and has to cut up early, into two linebackers.

In Front A, the trouble is the gap nobody is in. The play-side C gap, between the right tackle and the tight end, is empty, so the right tackle is uncovered: he can help the tight end, an excellent blocker, on the edge, or climb straight to the only linebacker. The 4i and the nose, big men built to hold ground, have to chase the play sideways. Everything rests on one edge against two blockers, with one linebacker and a safety coming from depth to fill the alley. Expect the ball to bounce outside if the edge is reached, or to cut back if the Mike over-runs it.

What Front B gives up: the interior. Against the same team’s inside zone, the 1 and the 3 can be doubled and the doubles can climb to two linebackers. That is why defenses facing an outside-zone team mix fronts by down and formation instead of living in one, and why the tite’s teams widen their edges (or switch to four-man fronts) against wide-zone offenses.

TipAnswer

Snap A is two-gap. The alignment couldn’t tell you, but the first step can: each lineman has stepped straight into the blocker in front of him and put his hands on his chest, about to lock out and read. That means three linemen are holding six gaps, the edge on the tight end has the D gap, and both inside linebackers are free: they have no gap of their own at the snap and will read and fill wherever the ball goes. If the linemen win, nobody blocks either of them.

Snap B is one-gap, from the same head-up alignment. Every lineman has slanted into a gap beside his blocker, all toward the tight end: the left end into the B gap, the nose into the A gap toward the tight end, the right end into the C gap. Every lineman has one gap, so the linebackers have one each too: the Mike the B gap toward the tight end, the Will the other A gap, exactly the assignments of the right half of Figure 30.3. Nobody is free, but everybody is fast. One-gap teams usually line up shaded, as in that figure, but plenty of them line up head-up and slant, precisely so that the alignment gives nothing away. That is why the first step, not the alignment, is the tell.

Back to the broadcast

The chapter opened with a graphic that said 3-4 and a picture that kept changing. Now you can read both.

  • The graphic says 3-4, and you count four on the line, two standing: that is the 3-4 team’s 2-4-5 nickel, its nose on the bench and its two outside linebackers on the edges. The roster says 2-4-5; the football is a four-man line.
  • Next snap, five across the line, three of them enormous and wedged inside the tackles, one linebacker behind them: a tite, built by sending the nose back on for a linebacker, the coach’s 3-3-5. A multiple defense like Georgia’s can reach the same picture without substituting at all, by shifting a four-man line into it at the last moment.
  • The inside run into a wall, the linebacker never touched: the tite’s whole purpose. A nose and two 4is occupy all five offensive linemen, the doubles can’t climb, and the one linebacker fills his gap unblocked and makes the tackle. That is how six defenders stop a run aimed at six blockers: one of them, the nose, is doing the work of two.

Takeaways

  • Find the center. A defender head-up on the center makes the front odd; nobody there makes it even. The odd fronts are built around a heavy nose who takes two blockers so somebody else can run free.
  • The 3-4 Okie is one alignment played two ways. Two-gap (head-up, control and read) keeps the inside linebackers clean at the cost of rare bodies and little pass rush; one-gap (shaded, attack) turns the 3-4 into an aggressive front, often an Under with a standing Sam. The first step tells you which.
  • The stand-up edge is the 3-4’s great legacy. Standing gives him vision, width and the option to drop, which makes the defense’s rush uncertain; the price is leverage against the run.
  • The tite walls off the middle. A 0 and two 4is leave the offense’s uncovered linemen stuck on doubles that can neither move them nor climb, so a light box with two safeties deep can still stop inside runs; the one linebacker owns the last gap, unblocked. It gives up pass rush, the perimeter (an empty C gap against outside zone) and the space underneath.
  • Fronts choose what to stop. Line up on or inside blockers to beat double teams (inside runs); line up outside them, with every outside gap filled, to beat reaches (outside zone). No front does both, so defenses change fronts by opponent, by situation and by shifting at the snap.
  • Nickel labels are roster labels. As roster counts, 4-2-5, 2-4-5 and 3-3-5 describe nearly the same box count and pass rush; what differs is whether the edges are listed as ends or linebackers. (When a coach calls a 3-3-5, he means the tite.) Count hands, not names, and read alignment before dialect.

Measuring the wall. The tite’s promise is that nobody climbs: every offensive lineman is still engaged with a defensive lineman a second after the snap, so the linebacker behind them is untouched. Tracking data can test that. A simple measure is the distance from the linebacker to the nearest offensive blocker: two players whose centres are about a yard apart are in contact, so anything under about 1.5 yards means a blocker has reached him. Here it is at three moments for the Mike in the simulated tite play above, and, for contrast, for the Mike in the same inside zone against a four-man front that leaves the play-side guard uncovered (a 1 on the center and a 3 on the left guard), so that the guard can climb. def nearest_blocker(play, key, times=(0.5, 1.0, 1.5)):
    trk = play.to_tracking()
    me = trk[trk.displayName == key].set_index("time")
    ol = trk[trk.displayName.isin(OL_KEYS + ("Y",))]
    out = {}
    for t in times:
        m = me.loc[round(t, 1)]
        o = ol[ol.time.round(1) == round(t, 1)]
        out[f"+{t:.1f} s"] = f"{np.hypot(o.x - m.x, o.y - m.y).min():.1f} yd"
    return out
