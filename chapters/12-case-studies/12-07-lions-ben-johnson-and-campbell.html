def O(key, pos, d, w, label=None):
    return Player(key, pos, "off", d, w, label=label)


def D(key, pos, d, w, label=None):
    return Player(key, pos, "def", d, w, label=label)


def line5(d=-0.7):
    """The five offensive linemen (only the center labelled)."""
    return [O(k, k, d, w, label="C" if k == "C" else "") for k, w in
            (("LT", -3.2), ("LG", -1.6), ("C", 0.0), ("RG", 1.6), ("RT", 3.2))]


def pre(p, key, pts):
    """Pre-snap movement through absolute (d, w) points (gridiron's motion takes offsets from the start)."""
    d0, w0 = p._cur(key)
    p.motion(key, pts=[(d - d0, w - w0) for d, w in pts])


def go(p, key, pts, kind="path", **kw):
    """Post-snap movement through absolute (d, w) points: kind = path, run, route or block."""
    d0, w0 = p._cur(key)
    rel = [(d - d0, w - w0) for d, w in pts]
    if kind == "route":
        return p.route(key, pts, absolute=True, **kw)
    if kind == "block":
        return p.block(key, pts=rel, **kw)
    return {"path": p.path, "run": p.run}[kind](key, rel, **kw)


def retime(p, key, frames):
    """Replace `key`'s pre-snap keyframes with explicit (t, d, w) rows (t <= 0).
    gridiron ends every pre-snap motion at the snap and starts it whenever its length requires,
    so a shift, a one-second freeze and then a motion can't be expressed directly. This sets
    the schedule by hand (implemented here, not in gridiron). Call it after every action is added."""
    kf, t0, t1 = p._compile()
    post = kf[key][kf[key][:, 0] > 1e-9]
    new = np.array(frames, dtype=float)
    if new[-1, 0] < 0:
        new = np.vstack([new, [0.0, *new[-1, 1:]]])
    kf[key] = np.vstack([new, post])
    p._compiled = (kf, min(float(f[0, 0]) for f in kf.values()), t1)


# ---------------------------------------------------------------- drawing helpers
def no_numbers(ax):
    """Remove the painted yard numbers (in zoomed windows they sit where players stand)."""
    for t in list(ax.texts):
        if t.get_alpha() == 0.55:
            t.remove()


def P_of(fld, p):
    """(d, w) in the play's frame -> plot coordinates on this field."""
    return lambda d, w: tuple(float(v) for v in fld.xy(*p.to_abs(d, w)))

def point_to(ax, P, at, to, text, color=style.INK, fs=6.6):
    """A label at `at` with a thin leader line to `to` (both (d, w))."""
    ax.annotate(text, xy=P(*to), xytext=P(*at), fontsize=fs, color=color, ha="center", va="center", zorder=12,

def ghost_marker(ax, x, y, label, ppy, side="def"):
    """A hollow, dashed marker: where a player stood before he moved."""
    col = style.DEFENSE if side == "def" else style.OFFENSE
    if side == "def":
        pat = RegularPolygon((x, y), 4, radius=MARKER_R * 1.25, orientation=np.pi / 4, facecolor="none",
                             edgecolor=col, lw=1.0, ls=(0, (2, 1.2)), zorder=6)
    else:
        pat = Circle((x, y), MARKER_R, facecolor="none", edgecolor=col, lw=1.0, ls=(0, (2, 1.2)), zorder=6)
    ax.add_patch(pat)
    if label:
        ax.text(x, y, label, ha="center", va="center", color=col, fontweight="bold", zorder=6,
                fontsize=max(4.0, min(8.0, ppy * 0.72)))

def snapshot(play, t, ax=None, lateral=(-17, 17), window=(-9, 15), ghosts=(), ghost_t=None, trail_from=None):
    """Everyone where he is at time t, a faint trail behind each mover, and a hollow 'ghost'
    where each listed player stood at `ghost_t`."""
    pos = {k: play.position(k, t) for k in play.players}
    moved = [pl.moved(d=float(pos[k][0]), w=float(pos[k][1])) for k, pl in play.players.items()]
    q = Play([m for m in moved if m.side == "off"], [m for m in moved if m.side == "def"],
             los=play.los, ball_y=play.ball_y, to_go=play.to_go)
    q.highlights = dict(play.highlights)
    fld = q.draw(ax=ax, lateral=lateral, window=window, title=False)
    ax, ppy = fld.ax, fld.pts_per_yard()
    no_numbers(ax)
    P = P_of(fld, play)
    start = play._compile()[1] if trail_from is None else trail_from
    for k, pl in play.players.items():
        if pl.pos in ("LT", "LG", "C", "RG", "RT", "DE", "DT", "NT"):
            continue
        pts = np.array([play.position(k, tt) for tt in np.linspace(start, t, 50)])
        if np.ptp(pts[:, 0]) + np.ptp(pts[:, 1]) < 0.4:
            continue
        xs, ys = fld.xy(*play.to_abs(pts[:, 0], pts[:, 1]))
        col = style.OFFENSE if pl.side == "off" else style.DEFENSE
        ax.plot(xs, ys, color=col, lw=1.1, alpha=0.5, ls="-" if pl.side == "off" else (0, (3, 2)), zorder=3)
    if ghost_t is not None:
        for k in ghosts:
            d0, w0 = play.position(k, ghost_t)
            d1, w1 = play.position(k, t)
            if abs(d1 - d0) + abs(w1 - w0) > 1.0:
                ghost_marker(ax, *P(d0, w0), play.players[k].label, ppy, play.players[k].side)
    bd, bw = play.ball_position(t)
    carried = any(np.hypot(*(play.position(k, t) - np.array([bd, bw]))) < 0.3
                  for k, pl in play.players.items() if pl.side == "off" and k != "C")
    bx, by = P(bd + (0.95 if carried and t > 0 else 0.0), bw)
    ax.add_patch(Ellipse((bx, by), 0.75, 1.1, facecolor=style.BALL, edgecolor="white", lw=0.6, zorder=9.5))
    for art in list(ax.texts) + list(ax.patches) + list(ax.lines):   # nothing drawn outside the window
        art.set_clip_on(True)
    return fld


def strip(play, times, labels, ncols=2, width_in=6.5, legend=True, annot=None, **kw):
    """The print version of an animation: one panel per chosen moment, each with a caption.
    `annot` maps a panel index to a function f(ax, P) that adds print-only labels to that panel."""
    lateral, window = kw.get("lateral", (-17, 17)), kw.get("window", (-9, 15))
    n = len(times)
    nrows = int(np.ceil(n / ncols))
    pw = width_in / ncols
    h = nrows * (pw * (window[1] - window[0]) / abs(lateral[1] - lateral[0]) + 0.42) + (0.35 if legend else 0.1)
    fig, axes = plt.subplots(nrows, ncols, figsize=(width_in, h), squeeze=False)
    for ax in axes.flat[n:]:
        ax.axis("off")
    for i, (ax, t, lab) in enumerate(zip(axes.flat, times, labels)):
        fld = snapshot(play, t, ax=ax, **kw)
        if annot and i in annot:
            annot[i](ax, P_of(fld, play))
        ax.set_title(lab, fontsize=7.8, color=style.INK, loc="left", pad=3, linespacing=1.15)
    fig.tight_layout(h_pad=0.6, w_pad=0.5, rect=(0, 0.05 if legend else 0, 1, 1))
    if legend:
        hs = [Line2D([], [], color=style.OFFENSE, lw=1.2, alpha=0.7, label="Offense: path so far"),
              Line2D([], [], color=style.DEFENSE, lw=1.2, ls=(0, (3, 2)), label="Defense: path so far"),
              Patch(facecolor="none", edgecolor=style.DEFENSE, lw=1.0, ls=(0, (2, 1.2)),
                    label="Dashed outline: where he started")]
        fig.legend(handles=hs, loc="lower center", ncol=3, fontsize=7.0, frameon=False, bbox_to_anchor=(0.5, 0.0))
    return fig


def show_play(play, times, labels, web_caption, annot=None, **kw):
    """Web: an animation with a caption that changes with the phase of the play.
    Print: a labelled frame strip (panel titles = `labels`)."""
    if PDF:
        strip(play, times, labels, annot=annot, **kw)
        plt.show()
        return
    df = play.to_tracking()
    tmap = df.groupby("frameId")["time"].first()
    anim = animate(df, to_go=play.to_go)
    ax = anim._fig.axes[0]
    cap = ax.text(0.01, 0.02, "", transform=ax.transAxes, ha="left", va="bottom", fontsize=9, color=style.INK,
                  zorder=21, bbox=dict(boxstyle="round,pad=0.3", fc=style.SURFACE, ec=style.THIRD, lw=1.0))
    base = anim._func

    def upd(fid, *a):
        out = base(fid, *a)
        cap.set_text(web_caption(float(tmap.get(fid, 0.0))))
        return out
    anim._func = upd
    if shutil.which("ffmpeg"):
        html = anim.to_html5_video(embed_limit=40).replace("<video ", '<video style="max-width:100%;height:auto" ')
    else:
        html = anim.to_jshtml(default_mode="once")
    plt.close(anim._fig)
    display(HTML(html))


def phases(cuts):
    """[(t_upper, caption), ...] -> function of t (seconds from the snap) for show_play."""
    def f(t):
        for up, txt in cuts:
            if t < up:
                return txt
        return cuts[-1][1]
    return f

def pct(x, nd=0):
    return f"{100 * x:.{nd}f}%"


def ordinal(n):
    n = int(n)
    return f"{n}{'th' if 10 <= n % 100 <= 20 else {1: 'st', 2: 'nd', 3: 'rd'}.get(n % 10, 'th')}"


def rk(tab, season, team, asc=False):
    return int(tab.loc[season].rank(ascending=asc, method="min")[team])

for team in ("DET", "CHI"):
    for s in range(2019, 2026):
        g = rs[(rs.season == s) & ((rs.home_team == team) | (rs.away_team == team))]

# A rough yardstick for "points gained by going instead of kicking": for every go in the first
# three quarters, subtract the league's average EPA on punts and field goals from fourth downs in
# the same yards-to-go and field-position bucket (2021-2025).
q3 = f4[(f4.qtr <= 3) & f4.season.between(2021, 2025)].copy()
q3["yb"] = pd.cut(q3.ydstogo, [0, 1, 2, 3, 5, 10, 99])

GOS = q3[q3.go].join(KICK, on=["yb", "fb"])

The problem: a franchise with nothing to protect

Start with how deep the hole was. Before January 2024, the Lions had won one playoff game since their last championship in 1957: a victory over Dallas on January 5, 1992.2 In 2020 they went 5–11, fired head coach Matt Patricia in November, and finished the season with an interim coach. In January 2021 they hired a new general manager, Brad Holmes, from the Rams’ scouting department, and a new head coach, Dan Campbell.

Campbell was not a hot coordinator, the kind of in-demand assistant every team with an opening wants to interview. He had been a tight end for 11 NFL seasons, the Dolphins’ interim head coach for 12 games in 2015, and for the five seasons before Detroit, Sean Payton’s tight ends coach and assistant head coach in New Orleans. At his introductory press conference, on January 21, 2021, he promised a team that would take on the identity of the city, absorb a punch and, on the way back up, “bite a kneecap off.”3 It became a meme, and not a kind one. One national commentator said the new coach had “3-13 written all over him.”

Then came the trade. Nine days later the Lions agreed to send Matthew Stafford, their quarterback for twelve seasons, to the Rams for Jared Goff, a 2021 third-round pick and the Rams’ first-round picks in 2022 and 2023.4 Goff had just been pushed out in Los Angeles. To most observers he was a contract the Rams wanted to get rid of; the picks were the prize.

That is the problem this case study is about. Detroit had no young stars, a quarterback the league had written off, and a coach most people thought was a motivational speaker. What do you do with a team like that?

The answer, part 1: build the trenches and the core through the draft

Holmes and Campbell’s plan was old-fashioned in one way and modern in another. The old-fashioned part was the order of building: linemen first, on both sides of the ball, mostly through the draft. Detroit’s first three picks in 2021 were a tackle and two defensive tackles (Sewell, Levi Onwuzurike and Alim McNeill), and its first pick in 2022 was an edge rusher, Aidan Hutchinson.5 The modern part was volume: extra picks from the Stafford trade, and a willingness to trade around the board. Table 65.1 isn’t the whole draft record; it’s the core that the offense in this chapter was built around, which is why it tilts toward receivers and backs once the lines were started.

The Rams’ two first-round picks didn’t simply turn into two players: Detroit used them in trades. In 2022 the Rams’ pick (No. 32) went to Minnesota with two more picks to move up to No. 12 for Jameson Williams; in 2023 the Rams’ pick (No. 6) went to Arizona in a trade down that brought back No. 12 (Gibbs) and No. 34 (LaPorta). Two more names matter for the offense: right tackle Sewell, the seventh overall pick in 2021, and St. Brown, a fourth-rounder who became one of the league’s best receivers. You’ll see both in the plays below. A scheme can only ask its players to do what they can do, and Johnson’s scheme asked a lot of a tackle and a slot receiver.

The answer, part 2: when you are the underdog, choose variance

The second half of the plan was a decision-making style, and it showed up before the team was any good. In 2021 the Lions went 3–13–1. They also went for it on fourth down 41 times, the most in the league (the average team went for it 25 times), and ran three successful fake punts.6 Why would a bad team be the most aggressive team in football?

ImportantWhy it exists

Underdogs should increase variance. A basketball team that is outmatched shoots more three-pointers: it will miss more, but the only way it wins is a hot night, so it buys more chances at one. Football works the same way. If you expect to lose a game playing it safe, a safe strategy mostly guarantees the expected result: a respectable loss. A riskier strategy, more fourth-down attempts, onside and fake kicks, trick plays, makes both the blowouts and the upsets more likely. When you start out behind, the upsets are where your wins live (a statistician would say the good tail of a wider distribution). The price is real: bad outcomes get worse and more visible, and a coach who chooses variance will be second-guessed on every failure. A rebuilding team with nothing to protect pays that price most cheaply.

There was a second, less mathematical reason: identity. Campbell talked about building a team that played like it had nothing to lose, and an aggressive fourth-down habit is a weekly, public signal to the players that the coach trusts them. That is harder to measure, but it is part of why the habit survived after the Lions became favourites, which is where the math gets more interesting (later in this chapter).

Ben Johnson: the designer

Ben Johnson is an unlikely NFL play-caller in the best sense. He was a walk-on quarterback at North Carolina who graduated with degrees in mathematics and computer science and worked briefly as a software developer before going into coaching: graduate assistant and then tight ends coach at Boston College, seven seasons on the Miami Dolphins’ offensive staff (quarterbacks, tight ends and receivers), and then Detroit in 2019 as an offensive quality-control coach, the job of breaking down film and data for the coaches above him.7 Campbell’s arrival in 2021 kept him and made him passing-game coordinator. When the offense sputtered under coordinator Anthony Lynn, Campbell took over the play-calling at midseason, leaning on Johnson for the plan; after the season Lynn left and Johnson became the offensive coordinator and play-caller.8

The results came fast. Detroit’s offense ranked 23rd in EPA per play in 2021, 6th in 2022, 6th in 2023 and 3rd in 2024, when the Lions went 15–2 and led the league with 564 points. Goff, the castoff, threw for 4,629 yards and 37 touchdowns that year. Johnson was named the AP’s Assistant Coach of the Year for 2024.9

How much of that was Goff and how much was Johnson? The cleanest look is the same quarterback under different play-callers, measured in EPA per dropback (every pass, sack and scramble), against the league average in the same seasons:

  • Rams, 2019–2020 (Sean McVay calling): +0.06, league +0.06. An average starter over those two seasons.
  • Lions, 2021 (Lynn, then Campbell): -0.06.
  • Lions, 2022–2024 (Johnson): +0.18, league +0.02, and +0.28 in 2024 alone.
  • Lions, 2025 (Morton, then Campbell): +0.17, league +0.04.

Goff kept most of the gain after Johnson left, so it wasn’t all scheme: he and his receivers had become a very good passing team. But his best season came in Johnson’s last one, and it fell back by more than a third without him. Read it as a quarterback the system suited, in a system that got the most out of him.

What was his offense? Not one invention. Its base was familiar: a zone and gap run game behind Sewell and a strong interior line, play-action off it, and quick, timed throws to St. Brown and LaPorta. What made it Johnson’s was a set of habits layered on top that made every snap look like several others. Here are the four that show up most clearly on film and in the data, with what each buys and costs.

Habit 1: the quarterback goes back under center

In the 2010s the shotgun took over the NFL. Johnson’s Lions went the other way: in 2022 their quarterback was under center on 47% of their runs and passes, the highest share in the league, and in 2024 on 54%, again the highest.10

For comparison, the league average was 32% in 2022, fell to 27% in 2023 and was 29% in 2024. That 2023 dip is real, not a charting quirk: the play-by-play’s own shotgun flag rose from 68% to 72% of plays the same year. 2023 was the shotgun’s high-water mark, and the league-wide turn back toward under center came after it, in 2024 and 2025. Detroit was already there.

Why go back? Because under center, a run and a pass begin with the same motion. The quarterback turns his back to the defense, the running back comes downhill toward him, and for the first second the linebackers can’t see the ball. From the shotgun, the run fake is a sideways mesh the defense reads more easily. Under center is a disguise before it is anything else, and the payoff is play-action. In 2024 the Lions used a run fake on 35% of their dropbacks, the league’s highest share (league 23%).

Did it pay? Careful here, because play-action flatters everyone. It is called on run-friendly downs, against defenses set to stop the run, so it beats ordinary dropbacks across the whole league. The fair test compares Detroit’s edge with the league’s edge on the same downs. On first and second down in 2022–2024, a run fake was worth +0.07 EPA per dropback to the rest of the league, and +0.19 to Detroit: nearly three times the league’s edge.

The under-center split needs the same care. Detroit’s plays from under center averaged +0.11 EPA, its shotgun plays +0.10: no difference. That comparison can’t show the disguise working, because the two groups are different plays in different situations (most under-center snaps are runs, most shotgun snaps passes). What the numbers do show is narrower. Going under center more than anyone cost Detroit nothing, and when it threw from under center it threw unusually well: +0.31 EPA per dropback, against +0.10 for the rest of the league on the same kind of snap.

The costs are real. On a run fake the quarterback turns his back to the defense and loses sight of the coverage for about a second, so he comes out of the fake re-reading it. That is why play-action passes, Johnson’s included, are usually built as simple high-low reads to defined landmarks (the deep crosser, the post, the flat) rather than full-field progressions. And on a true dropback without a fake, the quarterback needs a beat longer to set up than from the shotgun, which is why under-center teams lean on play-action rather than pure drop-back passing.

Habit 2: bodies, and the lineman who might be a receiver

Johnson loved heavy personnel (extra tight ends) and jumbo sets with a sixth offensive lineman, and he made that sixth lineman a weapon by having him report eligible. Formation Rules counted the numbers: Detroit reported a lineman on 170 runs and passes from 2022 to 2024, third most in the league, and passed on about 35% of them against about 24% for the league.

A reported lineman sends the defense two messages at once. “Big formation” says run, so linebackers and a safety creep toward the line. “Eligible” says this 300-pounder might go out for a pass, so somebody has to account for him. The left play is the one fans remember, because the catch is funny. The right one is the one that does damage, and it is far more common. That trade is the whole design: the reported lineman is mostly a threat, and the offense cashes the threat by throwing somewhere else.

What does jumbo cost? Every lineman you add is a receiver you take off. With six linemen on the field, one of the five eligible players is a 320-pound tackle who catches a few passes a season, so the real receiving threats drop to three or four, and the defense can answer with its own big bodies (an extra linebacker instead of a nickel back) without fear of being spread out. The boot in the left panel shows a second cost: once #70 releases, nobody blocks the end on that side, so the quarterback has to throw on time or pull up. And there is a rule tax. Play-action from a run look asks the linemen to fire off the ball as if it were a run, and a lineman who is more than a yard past the line when a forward pass is thrown beyond it is an ineligible man downfield: five yards, and the play doesn’t count. The more the pass looks like a run, the closer the blockers come to that line. It is the foul that wiped out the Sewell lateral described below.

CautionCommon misconception

“Reporting a tackle eligible is a trick play.” Mostly it isn’t. Across the league, a reported lineman usually means a run, and Detroit ran on about two of every three such snaps. The report is a cheap way to add a blocker who is also a possible receiver, and the passes it sets up usually go to the regular receivers. The genuinely tricky part, a throw to the lineman himself, is the rare constraint play that keeps the threat honest.

The lesson for a viewer is to listen for the referee’s announcement, and then watch the safeties rather than the lineman.

Habit 3: shift, set, motion: make the defense declare twice

Before a defense can stop a play, it has to agree on what it’s looking at: which side is the strength, which linebacker has which gap, who covers whom. Every shift asks it to agree again. Johnson’s offense layered two kinds of movement: first a shift that changed the formation (a tight end trading sides, a receiver tightening his split), then, after the required one-second freeze, a man in motion at the snap. Count what the defense had to do in under seven seconds: identify a two-tight-end formation and set its strength; watch two players move at once, re-identify the strength and slide three players; recognise a jet motion and rotate its safeties; and then, at the snap, decide whether the jet was real. Each of those is a call that must be heard by eleven people in a loud stadium. Each is a chance for one of them to be a step late or in the wrong gap. A defense that answers every movement correctly has still been made to show its answers, which tells the quarterback who is likely to be where.

ImportantWhy it exists

Shifts and motion buy information and mistakes. The defense’s adjustment reveals its coverage (a defender who travels with motion suggests man; a safety who spins suggests a rotation), though less reliably than it used to, since by the mid-2020s many zone defenses also “bump” a linebacker or safety across with the motion man. That is part of why Johnson layered a shift first: two adjustments are harder to fake than one. And every re-declaration is a chance for a miscommunication. Layering a shift and a motion means the defense has to answer twice on one snap. The cost: the offense must be perfectly drilled too. After a shift every player has to be still for a full second before anyone goes in motion, or it is a five-yard illegal shift, and the more players move, the more ways there are to get it wrong. It also burns play clock, which is why these sequences cluster on plays where the offense broke the huddle early.

Johnson’s Lions were not the league’s heaviest motion team by volume. FTN (a company that hand-charts every NFL play, in data the course loads through nflverse) logged motion on 59% of Detroit’s plays in 2024, 6th in the league. The Rams led, at 77%, but the two teams moved for different reasons. The Rams lined up with three wide receivers on 85% of their 2024 plays, so their motion mostly moved receivers around one familiar personnel group; Detroit used three receivers on only 55% (league 63%), with tight ends, a sixth lineman or two backs the rest of the time. The distinctive thing was the combination: motion plus shifts plus under center plus heavy personnel, so the same eleven players could arrive at many different formations, and the defense, having substituted for one personnel group, could rarely substitute its way out of the formation it ended up facing.

Habit 4: trick plays as constraint plays

Every team has a few trick plays. Johnson’s Lions ran them more than anyone and treated them as part of the system rather than a surprise package. FTN’s charters flagged 61 Detroit snaps as trick plays from 2022 to 2025, the most in the league (the next team had 56), and those snaps averaged +0.76 EPA, against +0.11 for everyone’s trick plays.13 The samples are small, a few dozen plays, so read the EPA as “worked often” rather than as a precise number.

The logic is constraint theory. A trick play isn’t random. It is a constraint play built on a specific habit the defense has learned from your film: a cornerback who always rallies to a short catch, a defense that ignores the running back as a passer, a safety who flies at every run fake. The offense’s own self-scout tells it which of its plays the defense has studied hardest, and the trick play starts as exactly that play. The 2024 season shows the pattern: the same catch-and-pitch idea four times, and three different non-quarterbacks throwing.

FTN flagged 17 Detroit snaps as trick plays in the 2024 regular season. Table 65.2 isn’t all of them: 8 were runs by wide receivers (end-arounds and reverses), and the table keeps the two kinds this chapter is about, the laterals and the throws by players who aren’t the quarterback.

Two rules make those plays possible. First, anyone can throw a forward pass (it has to be thrown from behind the line, and only one per play), so a receiver or a running back can be the passer. Second, anyone can catch a backward pass, a lateral or a pitch, including a lineman who hasn’t reported, because the eligibility rules apply only to forward passes.14 (Both are covered in Officiating, Replay, and the Rules of the Ball.) That is how a right tackle like Sewell can take a toss: the toss goes backward, as a backward pass. The running-back throws have a favourite home near the goal line, where the ball can be snapped straight to the back (a direct snap, the idea behind the wildcat) with the quarterback split out as a receiver. The defense then has to account for a back who can run behind an extra blocker, a quarterback who can catch, and tight ends who are eligible but block on almost every goal-line snap, and it tends to forget that the back can throw.

NoteMadden vs. real life

In a video game a trick play is a menu option you pick when you’re feeling lucky. In the NFL it is a scheduled part of the week. A trick play is usually installed against one opponent’s habit, practised all week, and called when the situation matches the film: the right down, the right hash, the right defensive call. Coaches think of it like any other tendency breaker: it exists because the offense has been predictable on purpose, and it works only if the play it imitates has been run often enough that the defense trusts its read.

Three readings of Figure 65.5.

The habit came first, the good team second. Detroit’s neutral go rate was 9% in 2020, below the league. In Campbell’s first season it was 25%, 3rd in the league, for a team that won three games: the underdog logic from the start of this chapter. (Detroit led the league in raw attempts that year; by this rate, which strips out desperate situations, it was a whisker behind the leader’s 26%.) Then it stayed near a third of neutral fourth downs, first in the league in 2022, 2023 and 2024 (33% in 2024, against a league average of 14%), after Detroit had become a favourite. That is the more interesting decision, because the variance argument no longer applies. A favourite going for it is betting on the expected-points case alone, and on its own offense.

It paid, by a rough yardstick, though mostly in one season. Detroit converted 56% of its attempts in 2021–2025 (the league converted 54%). Converting is the wrong test, though, as 10-04 argued: what matters is the attempt compared with the kick it replaced. So compare each of Detroit’s 114 first-three-quarter attempts with what the league’s punts and field goals produced from the same spots. The attempts come out about 67 points ahead over five seasons, 4th in the league over the period. That total is lumpy: about 31 came in 2024 alone, and the other four seasons averaged about 9.

The total is a net figure, and the bad tail is inside it. The conversions gained about 216 points against kicking; the 47 failures gave back about 148. Each failure hands the opponent a short field, and the ones that come late in close games are the ones fans remember. The yardstick is also crude. Teams go for it when they like their chances, so the attempts are a selected sample, and that same selection is why the league average go in Figure 65.5 looks worth +0.45 points over kicking: the average go looks good partly because teams only go when it looks good. A proper answer needs a win-probability model of each decision, which is built in Win Probability and Decision Analysis. But the direction for Detroit is not in doubt.

The more you go, the less each go is worth. Detroit’s value per attempt (+0.59 points) was only a little above the league’s (+0.45), and well below the Eagles (+1.11) and the Chiefs (+1.45). The reason shows up inside Detroit’s own attempts. The first fourth downs a coach adds are the obvious ones, fourth-and-1 or 2 near midfield; the next ones are closer calls, and 26% of Detroit’s attempts came with four or more yards to go, against 19% for the league. Detroit’s attempts with one to three yards to go were worth +0.77 points each against kicking; its attempts with four or more were worth +0.10. That is diminishing returns: each extra, longer attempt is still slightly positive on average, but adds far less than the short ones, and it fails more often. (The right panel of Figure 65.5 can’t show this on its own: across teams, attempts and value per attempt are nearly uncorrelated, 0.06, because teams differ in far more than aggressiveness.)

CautionCommon misconception

“Campbell is a gambler.” A gambler takes bets with negative expected value for the thrill. Campbell’s fourth-down record is a long series of bets with positive expected value, some of which lost, a few of them in playoff games. The fair criticism is narrower: in a handful of late, high-leverage spots (the 2023 NFC Championship Game, covered in 10-04, is the famous one), the models themselves called the decision close, and a coach with a strong kicker might reasonably have kicked. Judge the habit by the whole record, and the close calls by the odds before the snap.

The habit extended past the field. Detroit’s front office took its aggressive streak to the league’s rule-making meetings: a Lions proposal produced the 2024 rule that gives a coach a third challenge when either of his first two succeeds (it had required both),15 and in 2025 the Lions proposed letting wild-card teams with better records be seeded above division winners, a proposal withdrawn before a vote.16 A team that thinks in expected value tends to notice where the rules leave value on the table.

How opponents responded

Defenses had answers, and they were the standard ones for this kind of offense. None of them is specific to Detroit’s opponents in any one game; they are what defensive coaches teach against under-center play-action, jumbo and heavy motion:

  • Keep the backside end home. On a run fake away from him, the end on the boot side doesn’t chase; he stays to meet the bootlegging quarterback. That is the reason the leak in Figure 65.2 has to come out on time.
  • Give the reported lineman an owner. A linebacker or the walked-down safety takes #70 if he releases, so the leak is covered and the bait still has to beat the defenders underneath.
  • Play single-high and accept the risk. Two deep safeties stop the play-action shots but leave a light box for the run game, so many defenses chose the other trade: a safety down, eight in the box, and a deep shot given up now and then. Detroit saw two-high coverage on 36% of its 2024 dropbacks against a league average of 39% (23rd of 32 teams, where 1st means the most two-high), so slightly more single-high than most. The gap is small, so read it as a lean, not a strategy.
  • Simplify against motion. Instead of rotating with every shift and motion, a defense can “lock” its coverage (nobody travels, nobody rotates) and play the call it started with, accepting a worse match-up in exchange for no busts.

None of these took the offense away: it ranked 6th, 6th and 3rd in EPA per play from 2022 to 2024. The worst day came in January 2025, in the divisional round against Washington. Detroit, the 15–2 top seed, lost 45–31 after turning the ball over five times: four interceptions, one of them returned for a touchdown, and a lost fumble. Two of the trick plays in the table above came in that game. It wasn’t a scheme answer so much as the bad tail of a high-variance team arriving on the worst day, against a Detroit defense that had been gutted by injuries since October: Aidan Hutchinson broke his leg in the 47–9 win at Dallas, and by December the injured defenders also included Alim McNeill, Alex Anzalone and Carlton Davis.17

The league’s response was the more durable one, and it came in two forms. The first was imitation, though the Lions were one influence among several (by 2025 the Rams took more snaps from under center than anyone): league-wide, the share of plays from under center went from about 27% in 2023 to about 34% in 2025, heavy personnel climbed to a decade high, and the neutral fourth-down go rate kept rising.18 The second was the coaching market. Within three days in January 2025, the Lions lost both coordinators to head-coaching jobs: Johnson to the Chicago Bears, a division rival, and defensive coordinator Aaron Glenn to the New York Jets.19 That is the usual price of success for a staff, and it set up what economists call a natural experiment: nobody designed it, but the same ideas now had one coach removed and two teams to compare. The rest of this chapter runs that comparison.

After Johnson: Detroit in 2025 and 2026

Campbell hired John Morton, a veteran assistant, as offensive coordinator and play-caller, and Kelvin Sheppard replaced Glenn as defensive coordinator. After a loss to Minnesota in Week 9, Campbell took the play sheet himself, starting with a 44–22 win over Washington.20 The Lions finished 9–8 and missed the playoffs. The offense was still good, 7th in EPA per play, with the same quarterback, receivers and running backs, but no longer great, and it looked more like everyone else’s (though still well above the league’s habits): under center 48% of the time (from 54%), play-action on 25% of dropbacks (from 35%). What a Play Really Is compares the offense under Morton and under Campbell in detail. After the season Morton was fired, and the Lions hired Drew Petzing, previously the Cardinals’ coordinator, as their third offensive coordinator in three years; Sheppard stayed on defense.21

The data in this chapter stop at the end of 2025, and the 2026 season is only weeks old as this is written, so it is too early to say what Petzing’s offense is. The test is easy to run yourself as the season goes on: if Detroit’s under-center and play-action rates in Figure 65.6 climb back toward their 2024 levels, the habits belonged to the building; if they stay near the league average, they left with Johnson.

What transferred to Chicago

The Bears Johnson inherited were a classic Detroit-2021 problem: 5–12 in 2024, an offense 26th in EPA per play, and a rookie quarterback, Caleb Williams, sacked 68 times. In 2025 they went 11–6, won the NFC North, beat Green Bay 31–27 in the wild-card round (the franchise’s first playoff win since the 2010 season) and lost to the Rams 20–17 in overtime in the divisional round.22 The offense jumped to 8th in EPA per play; Williams was sacked 24 times, and his EPA per dropback rose from -0.09 to +0.07.

The picture is unusually clean for football data, at least for the three most visible habits. In one offseason, Chicago went from 17th to 5th in under-center rate (28% to 48%), from 29th to 2nd in play-action share (18% to 31%), and from 28th to 1st in trick-play rate. Detroit’s numbers moved the other way. Those habits were the play-caller’s, and they travelled with him.

The other two habits moved less. Chicago’s motion rate rose from 46% to 58% of plays (12th in the league), about twice the league’s rise, but not to the top. Reported linemen rose from 1.1 to 4.0 per 100 plays, in a year when the whole league jumped from 2.4 to 4.5, so Chicago ended up slightly below the league average. A habit that depends on personnel (a sixth lineman Johnson trusts as a blocker and as a decoy) takes longer to install than one that depends on the play sheet. Shifts can’t be measured from FTN’s public charting at all.

The fourth-down habit didn’t travel the same way. Detroit’s neutral go rate fell from 33% to 25% but stayed 5th in the league under Campbell; Chicago’s rose from 17% to 21%, in a season when the whole league rose. Aggression on fourth down had been the head coach’s.

What the fingerprint can’t tell you is how much of Chicago’s improvement came from the scheme. The Bears also rebuilt the middle of their offensive line before 2025, trading for guards Joe Thuney and Jonah Jackson (a Lions starter in Johnson’s first two seasons as coordinator) and signing center Drew Dalman,23 and Williams was in his second season, when most young quarterbacks improve. The sack total falling from 68 to 24 belongs to all three. In short: the play-caller’s signature habits transferred at once, the personnel-dependent ones partly; the results transferred substantially, with help.

What survives

The Johnson-Campbell Lions lasted four seasons as a unit, from 2021 to 2024, and they leave a few ideas that have outlasted the staff.

  • Under center is a disguise, not a throwback. The Lions, with the Rams, showed that a modern passing offense can take a large share of its snaps from under center at no cost, and throw unusually well off the run fake, because the fake and the run look the same. The league followed.
  • Personnel is a message you can lie with. A reported lineman, a third tight end, a jumbo set: each tells the defense something, and an offense can choose whether to deliver on it.
  • Make the defense answer twice. Shifts layered with motion turn a single snap into two or three communication tests. Most defenses pass most of them; the offense only needs the occasional failure.
  • Trick plays are part of the system. Built on the defense’s habits and the offense’s own self-scout, they are constraint plays with a sense of humour, and their cost is visible on the bad days.
  • Fourth-down aggression became normal, and its edge got smaller. When one team goes for it a third of the time, it’s a strategy; when the league catches up, the remaining edge is in choosing which fourth downs, which is a harder skill to see from the stands.
  • A rebuild can choose variance. A bad team that plays to the expected-value math and accepts the ugly failures can turn into a good team that keeps doing it, which is what Detroit did.

Watch for it on Sunday

TipWatch for it: the trick detector

Pick a Lions or Bears game (or any team that runs a lot from under center) and flag every snap with something unusual in it.

  • Before the snap, listen. “Number 70 is reporting as eligible”: note the down and distance. Then watch the safeties, not the lineman. If they creep down, expect the run fake and a throw behind them.
  • Count the movements. One mark for a shift, one for a motion. Snaps with two or more marks are the ones where the defense has to communicate twice; look for a defender still pointing or moving at the snap.
  • Find the odd alignment. A quarterback split wide, a running back alone in the backfield taking a direct snap, a receiver stacked right behind another one a yard off the line, a tackle at the end of the line. Write down what the defense did about it.
  • After the snap: who caught the first ball? On a short catch near the sideline, keep your eyes on the player behind the catcher. If someone is trailing at full speed, the hook-and-lateral is on.
  • On fourth down, guess first. Before the offense decides, say go, kick or punt using the rules of thumb from 10-04. Then compare with what the coach did, and note whether he was more or less aggressive than you.
  • At the end of the game, count your flags and how many of them gained yards. You have just charted a team’s disguise rate, the raw material for a tendency report.

TipAnswer

Detroit goes for it; most teams don’t; and the play will be a shotgun pass. This is the “go zone” from 10-04: the opponent’s 30 to 40, too far for a comfortable field goal (a 53-yarder at two-thirds or less) and too close for a punt to gain much, so the models usually say go even on fourth-and-4 or 5. Teams mostly ignore that. On fourth-and-3 to 5 between the opponent’s 30 and 40 in the first three quarters of 2022–2024, the league went for it 40% of the time and Detroit 73% (a small sample, 11 snaps). This is exactly where Campbell’s habit differed from everyone else’s: not on fourth-and-1, where most teams now go, but on the medium-yardage calls the models like and the sidelines still fear.

The play is the surprise. You might expect the under-center, play-action identity, and on fourth-and-1 or 2 you’d be partly right: Detroit was under center on 61% of those attempts (league 51%) and used play-action on 15% (league 9%). But on fourth-and-3 to 5, Detroit was in the shotgun on all 31 of its attempts in 2022–2024, and 94% were dropbacks, just like the league. When the down and distance tell the defense “pass”, there is nothing left to disguise, and even this offense lines up to throw. The menu isn’t one play.

TipAnswer

The default is a run by the back who takes the snap, behind the wing tight end as an extra blocker; the trick is a pass by that back; and the men not to forget are the tight ends. This is the direct-snap look from the trick-play section. The back can run without a handoff, which gives the offense an extra blocker in the count, and the quarterback, standing in the slot as a receiver, takes a defender with him. The defense’s eyes go to the back and to the quarterback (everyone knows the quarterback can catch out there). What it tends to forget is that the back can throw, and that the tight ends, the players who block on almost every goal-line snap, are eligible: Y is the end of the line and U is a back on the wing. In Week 5 of 2025 at Cincinnati, Detroit took a direct snap to David Montgomery on second-and-3 from the 3, and Montgomery threw a touchdown pass to tight end Brock Wright.24 If the defense covers the quarterback and the back’s run, a tight end is the most likely target.

Takeaways

  • Detroit’s 2021 rebuild had two halves: build the lines and the core through the draft (Sewell, Hutchinson, St. Brown, Gibbs, LaPorta), and choose variance as an underdog. The fourth-down aggression came before the good team did.
  • Ben Johnson’s offense (2022–2024) was a familiar run and play-action base with four habits on top: the quarterback under center (the league’s highest share in 2022 and 2024), heavy and jumbo personnel with a reported lineman, shifts layered with motion, and trick plays built as constraint plays. Each makes one snap look like several; each costs drilling time, play clock and the occasional disaster.
  • Campbell’s fourth downs led the league in neutral go rate in 2022–2024 and, by a rough yardstick, gained about 67 points over kicking in five seasons, most of it in 2024. The extra, longer attempts were worth far less than the short ones: aggression has diminishing returns.
  • Defenses used the standard counters (a boot-side end who stays home, an owner for the reported lineman, more single-high, simpler motion rules), and the offense still ranked near the top every year. What finally broke it up was the coaching market: other teams hired its coordinators. Johnson went to the Bears and Glenn to the Jets in January 2025. Detroit then went through Morton, Campbell calling plays and, in 2026, Petzing: three coordinators in three years.
  • What transferred: in one season Chicago’s under-center, play-action and trick-play rates jumped to near the top of the league and Detroit’s fell back; motion and reported linemen moved less. The play-caller’s signature habits moved with him, the fourth-down habit stayed with the head coach, and the Bears’ results moved with help from a rebuilt line and a second-year quarterback.
