import os
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

from gridiron import Play, Player, Field, offense, technique, style

OL_NAMES = {"LT": "LT", "LG": "LG", "C": "C", "RG": "RG", "RT": "RT"}
GREEN = "#0f7a55"           # darker step of the aqua highlight, for text on the light field
PURPLE = style.SERIES[6]    # a ring colour that is neither offense blue nor defense orange


def named_line(players):
    """Label the five offensive linemen so the reader can follow each one."""
    return [p.moved(label=OL_NAMES[p.key]) if p.key in OL_NAMES else p for p in players]


def O(key, pos, d, w, label):
    return Player(key, pos, "off", d, w, label=label)


def D(key, pos, d, w, label):
    return Player(key, pos, "def", d, w, label=label)


def line5():
    return [q for q in named_line(offense("gun_2x2")) if q.key in OL_NAMES]


def pistol(extra=(), te=("Y", "U")):
    """The pistol box: five linemen, the quarterback 4 yards deep, the back 7 yards deep directly behind him,
    and inline tight ends (Y on the right, U on the left). Receivers are left out unless passed in `extra`."""
    tes = []
    if "Y" in te:
        tes.append(O("Y", "TE", -0.7, 4.8, "Y"))
    if "U" in te:
        tes.append(O("U", "TE", -0.7, -4.8, "U"))
    return line5() + [O("QB", "QB", -4.0, 0.0, "Q"), O("RB", "RB", -7.0, 0.0, "R"), *tes, *extra]


def front62(fs=False):
    """Eight in the box against two tight ends: a 9-technique outside each tight end, a 5 on each tackle,
    a 1 and a 3 inside, and two linebackers (a 6-2 look). Optional free safety in the deep middle."""
    dfn = [D("BE", "OLB", 1.0, -6.3, "9"), D("B5", "DE", 1.0, -4.0, "5"), D("DT1", "DT", 1.0, -0.7, "1"),
           D("DT3", "DT", 1.0, 2.4, "3"), D("P5", "DE", 1.0, 4.0, "5"), D("PE", "OLB", 1.0, 6.3, "9"),
           D("MIKE", "MIKE", 4.5, 1.8, "M"), D("WILL", "WILL", 4.5, -2.0, "W")]
    if fs:
        dfn.append(D("FS", "FS", 12.5, 0.0, "FS"))
    return dfn


def check_legal(players):
    """Seven offensive players on the line, and only the two at its ends may be eligible receivers."""
    off = [q for q in players if q.side == "off"]
    on = sorted([q for q in off if q.d > -1.0], key=lambda q: q.w)
    assert len(on) == 7, [q.key for q in on]
    for q in on[1:-1]:
        assert q.key in OL_NAMES, f"{q.key} is covered up but not a lineman"


def go_to(p, key, end, kind="block", delay=0.0, speed=None, via=()):
    """Move `key` through absolute (d, w) points `via` to `end`, as a block, path, run or pull."""
    start = np.array(p._cur(key))
    pts = [tuple(np.array(q) - start) for q in (*via, end)]
    if kind == "block":
        return p.block(key, pts=pts, delay=delay, speed=speed)
    return {"path": p.path, "run": p.run, "pull": p.pull}[kind](key, pts, delay=delay, speed=speed)


def _P(fld, p, d, w):
    return fld.xy(*p.to_abs(d, w))


def branch(fld, p, pts, color=style.THIRD, label=None, label_at=None, lw=2.0, fontsize=7.5,
           dash=(0, (3, 2.2)), zorder=6):
    """A dashed 'option' path through absolute (d, w) points, with an arrowhead."""
    xs, ys = _P(fld, p, np.array([q[0] for q in pts]), np.array([q[1] for q in pts]))
    fld.ax.plot(xs[:-1], ys[:-1], color=color, lw=lw, ls=dash, zorder=zorder, solid_capstyle="round")
    fld.ax.add_patch(FancyArrowPatch((xs[-2], ys[-2]), (xs[-1], ys[-1]), arrowstyle="-|>",
                                     mutation_scale=11, color=color, lw=lw, zorder=zorder))
    if label:
        x, y = _P(fld, p, *(label_at or pts[-1]))
        fld.ax.text(x, y, label, fontsize=fontsize, color=color, fontweight="bold", ha="center",
                    va="center", zorder=9,
                    bbox=dict(boxstyle="round,pad=0.2", fc=style.SURFACE, ec=color, lw=0.8))


def read_line(fld, p, a, b, color=style.THIRD):
    """The quarterback's eyes: a dashed green line between two (d, w) spots."""
    (x0, y0), (x1, y1) = _P(fld, p, *a), _P(fld, p, *b)
    fld.ax.plot([x0, x1], [y0, y1], color=color, lw=1.3, ls=(0, (4, 2)), zorder=4)


def say(fld, p, d, w, text, color=style.INK_2, fontsize=7.5, ha="center", weight="normal"):
    """A short on-field label at (d, w)."""
    x, y = _P(fld, p, d, w)
    fld.ax.text(x, y, text, fontsize=fontsize, color=color, ha=ha, va="center", zorder=9, fontweight=weight,
                bbox=dict(boxstyle="round,pad=0.18", fc=style.SURFACE, ec="none", alpha=0.9))


def point_to(fld, p, text_at, target, text, color=style.INK_2, fontsize=7, shrink=7, weight="normal"):
    """A label with a thin leader line to the player or spot it names."""
    (xt, yt), (xp, yp) = _P(fld, p, *text_at), _P(fld, p, *target)
    fld.ax.annotate(text, (float(xp), float(yp)), xytext=(float(xt), float(yt)), fontsize=fontsize,
                    color=color, ha="center", va="center", zorder=9, fontweight=weight,
                    arrowprops=dict(arrowstyle="-", color=color, lw=0.7, shrinkA=2, shrinkB=shrink),
                    bbox=dict(boxstyle="round,pad=0.18", fc=style.SURFACE, ec="none", alpha=0.9))


def tackle_x(fld, p, d, w, color=style.DEFENSE_DARK):
    x, y = _P(fld, p, d, w)
    fld.ax.plot([x], [y], marker="X", ms=10, color=color, mec="white", mew=0.8, zorder=12)


def box_shade(fld, p, d0, d1, w0, w1, color=GREEN):
    """Shade the box (alignment coordinates) to count defenders against blockers."""
    (x0, y0), (x1, y1) = _P(fld, p, d0, w0), _P(fld, p, d1, w1)
    fld.ax.add_patch(Rectangle((min(x0, x1), min(y0, y1)), abs(x1 - x0), abs(y1 - y0), fc=color, alpha=0.07,
                               ec=color, lw=0.8, ls=(0, (3, 2)), zorder=1))


def clean(ax):
    """Box-zoomed diagrams: drop hash ticks and yard numbers, clip anything outside the window."""
    for ln in list(ax.lines):
        xs, ys = ln.get_data()
        if len(xs) == 2 and ln.get_linewidth() == 0.5 and np.hypot(xs[1] - xs[0], ys[1] - ys[0]) < 1.0:
            ln.remove()
    for t in list(ax.texts):
        if t.get_text().isdigit() and t.get_alpha() == 0.55:
            t.remove()
        else:
            t.set_clip_on(True)


def draw(p, width_in=5.8, ax=None, **kw):
    """Play.draw, sized for the page, cleaned for a box-sized window."""
    fld = p.draw(width_in=width_in, ax=ax, **kw)
    clean(fld.ax)
    return fld


def pdf_build():
    return os.environ.get("RTG_FORMAT", "").lower() in ("pdf", "typst", "print")


# ---------------------------------------------------------------- frame strips and videos (as in 03-04)
MARK_R = 0.6   # frame-strip marker radius in yards


def tracking(play):
    df = play.to_tracking()
    df["_id"] = df["displayName"].astype(str)
    return df


def carrier(play, t):
    """Who holds the ball at time t (QB until the first handoff or pitch arrives)."""
    holder = "QB"
    for kind, to, te in sorted(play.ball_events, key=lambda e: e[2]):
        if t >= te + (0.25 if kind == "pitch" else 0.0):
            holder = to
    return holder


def panel(ax, play, df, t, lateral, window, rings=()):
    """One moment of a simulated play: markers, trails, the ball, rings (and a green ring on the ball carrier)."""
    tmap = df.groupby("frameId")["time"].first()
    fid = int((tmap - t).abs().idxmin())
    ylo, yhi = sorted((play.ball_y - lateral[0], play.ball_y - lateral[1]))
    fld = Field(ax=ax, orient="vertical", los=play.los, window=window, numbers=False, hashes=False,
                lateral=(ylo, yhi), to_go=play.to_go)
    cur = df[df["frameId"] == fid]
    shown = cur[cur["x"].between(play.los + window[0] + 0.4, play.los + window[1] - 0.4)
                & cur["y"].between(ylo + 0.4, yhi - 0.4)]
    ids = set(shown["_id"])
    trails = df[(df["frameId"] <= fid) & df["_id"].isin(ids) & (df["side"] != "ball")
                & df["x"].between(play.los + window[0], play.los + window[1])]
    for (_, side), g in trails.groupby(["_id", "side"]):
        xs, ys = fld.xy(g["x"].values, g["y"].values)
        ax.plot(xs, ys, color=style.OFFENSE if side == "off" else style.DEFENSE, lw=0.9, alpha=0.4, zorder=3)
    ppy = fld.pts_per_yard()
    for _, r in shown.iterrows():
        x, y = (float(v) for v in fld.xy(r["x"], r["y"]))
        if r["side"] == "ball":
            ax.add_patch(Ellipse((x, y + 0.5), 0.75, 0.45, fc=style.BALL, ec="white", lw=0.5, zorder=11))
            continue
        lab = str(r.get("label", "") or "")
        if r["side"] == "off":
            ax.add_patch(Circle((x, y), MARK_R, fc=style.OFFENSE, ec="white", lw=0.6, zorder=8))
            tc = "white"
        else:
            ax.add_patch(RegularPolygon((x, y), 4, radius=MARK_R * 1.3, orientation=np.pi / 4,
                                        fc=style.DEFENSE_TINT, ec=style.DEFENSE, lw=1.0, zorder=8))
            tc = style.INK
        fs = max(4.0, min(8.0, ppy * (0.8 if len(lab) <= 1 else 0.6)))
        ax.text(x, y, lab, ha="center", va="center", fontsize=fs, color=tc, fontweight="bold", zorder=10)
    for key, col in list(rings) + [(carrier(play, t), style.THIRD)]:
        rr = cur[cur["displayName"] == key]
        if len(rr):
            x, y = fld.xy(rr["x"].iloc[0], rr["y"].iloc[0])
            ax.add_patch(Circle((float(x), float(y)), 1.05, fc="none", ec=col, lw=1.7, zorder=7))
    return fld


def strip(play, times, lateral, window, notes=None, rings=(), width_in=6.5, ncols=2):
    """Frame strip of one simulated play, wrapped over rows, with a caption line under each panel."""
    df = tracking(play)
    nrows = int(np.ceil(len(times) / ncols))
    pw = width_in / ncols
    span_w, span_d = lateral[1] - lateral[0], window[1] - window[0]
    fig, axes = plt.subplots(nrows, ncols, squeeze=False,
                             figsize=(width_in, nrows * (pw * span_d / span_w + (0.6 if notes else 0.25))))
    for ax in axes.flat[len(times):]:
        ax.axis("off")
    for k, (ax, t) in enumerate(zip(axes.flat, times)):
        panel(ax, play, df, t, lateral, window, rings)
        ax.set_title(f"{k + 1}.  " + ("snap" if abs(t) < 1e-6 else f"{t:+.1f} s"), fontsize=8,
                     color=style.INK_2, loc="left")
        if notes:
            ax.text(0.5, -0.03, notes[k], transform=ax.transAxes, ha="center", va="top", fontsize=7.2,
                    color=style.INK)
    fig.tight_layout(h_pad=1.4 if notes else 0.6, w_pad=0.5)
    return fig


def video(play, times, lateral, window, notes=None, rings=(), width_in=5.4):
    """Web version: the same panel drawn every tenth of a second, with the current step's note."""
    import shutil
    from matplotlib import animation
    from IPython.display import HTML, display
    df = tracking(play)
    grid = np.round(np.arange(float(df["time"].min()), float(df["time"].max()) + 1e-9, 0.1), 2)
    span_w, span_d = lateral[1] - lateral[0], window[1] - window[0]
    fig, ax = plt.subplots(figsize=(width_in, width_in * span_d / span_w + 0.6))

    def update(t):
        ax.clear()
        panel(ax, play, df, t, lateral, window, rings)
        txt = "snap" if abs(t) < 0.05 else f"{t:+.1f} s"
        if notes:
            cur = [m for tt, m in zip(times, notes) if tt <= t + 1e-6]
            txt += ("\n" + cur[-1].replace("\n", " ")) if cur else ""
        ax.set_title(txt, fontsize=7.5, color=style.INK, loc="left")
        return []

    anim = animation.FuncAnimation(fig, update, frames=list(grid), interval=100)
    html = (anim.to_html5_video(embed_limit=40).replace("<video ", '<video style="max-width:100%;height:auto" ')
            if shutil.which("ffmpeg") else anim.to_jshtml(default_mode="once"))
    plt.close(fig)
    display(HTML(html))


def play_animation(play, times, lateral, window, notes=None, rings=(), **kw):
    """Video on the website; a numbered, captioned frame strip in the PDF."""
    if pdf_build():
        strip(play, times, lateral, window, notes=notes, rings=rings, **kw)
        plt.show()
    else:
        video(play, times, lateral, window, notes=notes, rings=rings)

Lamar Jackson stands four yards behind the center, with Mark Ingram directly behind him, three yards deeper. The defense has seen the film. It walks a safety down and crowds eight men within a few yards of the ball, the kind of picture a broadcaster calls “selling out to stop the run.” The ball is snapped, Jackson puts it in Ingram’s stomach, and an unblocked defensive end on the edge freezes for half a heartbeat, unsure whether to take the back or the quarterback. The problem: a quarterback who didn’t fit (2018)

Lamar Jackson arrived in Baltimore with the last pick of the first round of the 2018 draft, 32nd overall, after a Heisman Trophy at Louisville in 2016 and a college career in which he ran for more than 4,000 yards.1 He also arrived with a question hanging over him that a lot of NFL people asked out loud: could a quarterback who ran like that be a real NFL passer, or would he have to move to another position? The loudest version came from the Hall of Fame executive Bill Polian, who said that February that Jackson should switch to wide receiver (and admitted in 2019 that he had been wrong).2 The Ravens had a veteran starter, Joe Flacco, a Super Bowl MVP five years earlier, and an offense built for a pocket passer.

Then, in November 2018, Flacco hurt his hip, and on November 18 against Cincinnati Jackson made his first NFL start.3 What followed was improvised. The coaching staff, with Marty Mornhinweg as offensive coordinator, didn’t have a Jackson offense, so it bolted his college game onto Flacco’s: lots of quarterback runs, read plays, extra tight ends, and as few dropbacks as it could get away with. The data shows how sudden the swerve was. With Flacco starting, weeks 1 through 10, the Ravens called a designed run on 34% of their plays; with Jackson starting, weeks 11 through 17, it was 59%. The runs also got better. Measured in EPA per play (expected points added: how much a play raised or lowered the number of points the offense could expect to score on the drive, the yardstick 01-02 introduced), they went from -0.02, slightly worse than an average snap, to +0.08.4 Jackson ran 147 times that season, a record for a quarterback at the time, and Baltimore won six of his seven starts to take the AFC North.

The improvisation had a ceiling, and the wild-card game on January 6, 2019 found it. The Los Angeles Chargers had played Baltimore two weeks earlier, and for the rematch their defensive coordinator, Gus Bradley, did something almost nobody did: he put seven defensive backs on the field for 58 of the Ravens’ 59 snaps, with two safeties playing linebacker. The Chargers had used seven defensive backs for a total of 50 snaps all season. Their head coach, Anthony Lynn, said later that when Bradley proposed it his first reaction was to ask whether the staff was insane. It wasn’t. The Ravens gained 90 rushing yards and 229 in total, the fewest the Chargers had ever allowed in a playoff game, Jackson was sacked seven times, and Los Angeles won 23–17 after leading 23–3 in the fourth quarter.5 (Figure 63.7, later in the chapter, draws the idea.)

The lesson the Ravens took from that afternoon was not “stop running the quarterback.” It was that a run game added on to an offense is a run game a defense can study, match and outnumber in a week. If Jackson’s legs were going to be the foundation, the whole offense had to be built on them, including the passing game, so that the defense’s best answer to the run would open something else. That is a design problem, and the man the Ravens handed it to had been working on exactly that problem for most of a decade.

The scheme answer: Greg Roman’s 2019 rebuild

In January 2019 the Ravens promoted Greg Roman, their assistant head coach and tight ends coach, to offensive coordinator in Mornhinweg’s place.6 Roman had run the 49ers’ offense from 2011 to 2014, when Colin Kaepernick’s pistol read game took San Francisco to a Super Bowl, and then Buffalo’s with another running quarterback, Tyrod Taylor: the Bills led the NFL in rushing yards in 2015, and kept leading it in 2016 with the run game he had installed, although he was fired two games into that season.7 He was, in other words, the league’s most experienced designer of run games with a quarterback in them.

His first public description of the plan was blunt. The staff would not try to fit Jackson into something that “other people had once done”; it would “rebuild the thing from the ground up” and build an offense “that really accommodates” him.8 What came out of that rebuild had four pieces, and each one answers a specific question the Chargers had asked.

1. Heavy personnel: more blockers, and slower defenders. The Chargers’ answer had been speed: little defenders who could run with Jackson. Roman’s first move was to make little defenders expensive with heavy personnel. The 2019 Ravens put fewer than three wide receivers on the field on 53% of their snaps, against 37% for the league, using tight ends Mark Andrews, Hayden Hurst and Nick Boyle and the fullback (and part-time defensive lineman) Patrick Ricard (02-01 has the personnel fingerprint). Every tight end or fullback is one more blocker on a run, which forces the defense to answer with bigger bodies. Against Baltimore in 2019, defenses were in a base defense (four defensive backs, seven bigger front players) on 40% of snaps; against the rest of the league they were in base on 27%.9 A linebacker is a better run defender than a safety, but he is a worse chaser of a sprinter-fast quarterback in space, and a much worse cover man against a tight end like Andrews.

2. The pistol: no tell, and both ends in play. The pistol (02-02) puts the quarterback about four yards from the ball with the back directly behind him. The Ravens made it their home. In 2019 they lined up in it on 54% of their plays, 558 snaps; the rest of the league combined used it on 3% of plays, and the next-heaviest user had 68 snaps all season.10 75% of Baltimore’s designed runs started there. Why the pistol rather than the shotgun? Three reasons, each one a fix for something the shotgun gives away:

  • No directional tell. In the shotgun the back stands beside the quarterback, and which side he is on usually hints at where the run is going. From the pistol the back can go either way with the same first step, so the defense can’t lean before the snap.
  • A downhill back. A back seven yards deep and straight behind the ball runs at the line with momentum, like a back in the I-formation, instead of starting sideways from the quarterback’s hip. That makes gap runs (power, counter, duo) work from the same look as reads.
  • Both ends are read keys. With the quarterback at four yards and the back behind him, the mesh can go to either side, so either defensive end can be the read key on any snap. A defense that wants to tell its backside end to “always crash” has to tell both ends, and an end can’t know before the snap which one he is.

3. A menu of reads. On top of the pistol, Roman ran the whole option family that 03-04 teaches, and added a few of his own: the zone read and bash, the inverted veer (power read), the counter read (two linemen pull and the backside end is read), counter bash (the back goes away from the pullers, the quarterback follows them), and the occasional speed option.11 The point of having so many is not variety for its own sake. Each one leaves an end alone but sends the runners to different places, so a defense that solves one read with a stunt or an exchange (03-04) is wrong against the next.

4. Designed quarterback runs: the back becomes a blocker. And sometimes there was no read at all. Jackson carried on purpose, on QB power, QB counter, QB draws and sweeps, and on those plays the running back or the fullback became one more lead blocker. In 2019 Jackson had 113 designed runs, about 7.5 a game, averaging 7.4 yards and +0.32 EPA per carry.12

None of the four pieces works without the people who block for it, and the 2019 Ravens had them. Left tackle Ronnie Stanley and right guard Marshal Yanda (his eighth Pro Bowl) were Pro Bowl starters that season, right tackle Orlando Brown Jr. was an alternate, and Patrick Ricard went to the Pro Bowl as a fullback for his lead blocking.13 A scheme that asks the line to block everybody but one defender needs linemen who win those blocks.

Put those together and you have what this book calls a pistol option offense: an offense that makes the pistol its base alignment and builds its run game on the quarterback as a runner, with option reads and designed quarterback runs behind extra tight ends and fullbacks, so that the defense must account for one more ball carrier on every snap. Chris Ault built the college version at Nevada, where the pistol was born in 2005 and Kaepernick learned to run it (03-04 and The College Laboratory tell that story). The 2019–2022 Ravens are the NFL’s defining example.

ImportantWhy it exists

A pistol option offense exists because a quarterback who is a real running threat is worth one blocker on every run, and the pistol lets an offense collect that blocker from either side, behind a downhill back, without showing which way it is going. The heavy personnel makes the defense match with bigger, slower players, and then the passing game punishes the bigger, slower players. The trade is the quarterback’s body (he is a ball carrier, or a target, on most snaps), a passing game built from fewer receivers, and a style that works best when the offense is ahead or level, because the run is its engine.

The rebuild worked at once. The 2019 Ravens went 14–2, led the league in scoring with 531 points, set the NFL record for team rushing yards with 3,296, and Jackson ran for 1,206 of them, a quarterback record, while also leading the league with 36 touchdown passes on his way to a unanimous MVP award.14 By the measure this chapter uses, Baltimore’s designed runs averaged +0.11 EPA per play that season, 1st in the league, while the league as a whole averaged -0.07. And the passing game the run was meant to open up did open up: on dropbacks (passes, sacks and scrambles) the 2019 Ravens averaged +0.32 EPA per play, also 1st in the league, against a league average of +0.03.15 The rest of this chapter explains why.

Why the quarterback changes every run fit

Run Fits gave the rule in one line: count the quarterback as a blocker when he reads. It is worth slowing that down, because the Ravens’ whole era is an argument about what the rule really costs a defense, and the cost is larger than one man.

Start with a defense’s ordinary plan against a run. Every gap has exactly one owner, and the defense tries to have one more run defender than the offense has blockers: the plus-one. Against a quarterback who hands off and watches, the plus-one is easy to find. The offense blocks as many defenders as it has blockers; the one left over is usually the end on the far side of the play, and he can do the most useful thing a free defender can do, chase the play down the line from behind and tackle the back if he cuts back.

Now put Jackson at quarterback (Figure 63.1, right). Nothing about the blocking changes, but the free man isn’t free any more. If the backside end chases the back, Jackson keeps the ball and runs into the space the end just left. So the end has to stay home and play the quarterback, and the moment he does, the defense has lost its plus-one: seven blockers and a read against eight defenders is a hat on every hat. To get the plus-one back, the defense has to bring a ninth defender down, and in Figure 63.1 the only one left is the free safety, twelve yards deep and the last man in the deep middle.

That is the arithmetic. The Ravens’ insight was that the arithmetic doesn’t only apply on read plays. It applies on every run, for three reasons that compound.

The threat works on plays that aren’t reads. Most of the Ravens’ runs were simple handoffs. But a backside end can’t see the play call. On every pistol handoff he sees the same mesh he sees on a read, and if he chases the back on the one snap in five that was a read, Jackson is gone. So he learns to stay home every time, and on the four plain handoffs the offense gets for free what it earned on the read: an end who isn’t chasing the cutback. Coaches call this “freezing” the backside, and it is why a quarterback’s run threat improves his running backs’ numbers even when he never touches the ball.

Every defender’s fit gets one more line. A run fit (05-05) is a list of who owns which gap. Against a running quarterback, the defense needs one more owner on that list, quarterback, on every play, and the person who owns him can’t own anything else. If it’s the end, he can’t squeeze; if it’s a linebacker, his gap needs a new owner; if it’s a safety, the deep middle does. There is no free version.

Pursuit stops being free. Even after the ball is handed off, a defense normally gets a bonus: everybody on the far side of the play turns and runs to the ball. That backside pursuit cleans up mistakes. Against Jackson, backside defenders who turn and run before they’re sure the quarterback doesn’t have it get punished by the keep, so they turn a step later, and a step later is the difference between a four-yard run and a twelve-yard one.

CautionCommon misconception

“Loading the box stops a running quarterback.” Loading the box is how a defense gets its plus-one against a handoff. Against a quarterback who reads, an eight-man box is a seven-man box, because one of the eight is assigned to the quarterback. In fact, the 2019 Ravens saw loaded boxes about as often as everyone else did (eight or more defenders on 22% of their designed runs, against 21% for the league), and still led the league in rushing efficiency. What defenses did differently against Baltimore was get bigger, not more numerous: base personnel on 40% of snaps against 26% against everyone else, a gap that held through Roman’s four seasons. The boxes got heavier later: in 2021 and 2022 Baltimore saw eight or more on 30% and 33% of its designed runs, against about 23% for the league, and its runs still ranked 13th and 2nd.16 The extra defender has to come from somewhere, and when he comes from the deep secondary, the passing game is aimed at the hole he leaves.

Walk through what Figure 63.2 asks of the defense.

Before the snap, the defense can’t lean. In the shotgun, a back offset to the left usually means a run to the right; from the pistol, the back is straight behind the quarterback and gives nothing away. The defense has to stay balanced: a 9 on each side, a 5 on each side, linebackers over the middle.

At the snap, the line tells a story of power. The tight end and tackle block down, a guard pulls across. A linebacker who reads the guards has been taught to follow a pulling guard, because on power the puller leads the ball carrier through the hole. Here he does lead a ball carrier through the hole, but which one?

The end man has a bad choice. The play-side 9 is the man the power play would normally kick out, or log (seal inside from the outside so the runner can bounce around him). On this play nobody touches him. If he does what he’s taught against power, squeeze the hole from the outside so the pulling guard has nowhere to go, he has stepped inside the back, who is running outside him at full speed. Out there the play-side wide receiver blocks the cornerback, and the back’s problem is the alley player, usually a safety coming down from depth. If he widens to take the back, he has opened the hole for the quarterback, who follows the guard through it. Either way the defense’s best edge player is blocked by a decision.

Why build it from the pistol with tight ends instead of the shotgun with receivers, as most college teams did? The tight ends do two jobs a receiver can’t. The play-side tight end’s down block on the 5 creates the hole’s inside wall from a tighter angle, so the quarterback’s keep lane is a real gap, not a crease. And the backside tight end blocks a man a receiver couldn’t: the backside 9, the defender who on a shotgun power read is left to chase the quarterback from behind. With him walled off, the eight-man box is fully accounted for, seven blocks and a read, which is the arithmetic of Figure 63.1 made into a play.

What does it cost? Two things. The back is running sideways on the give, so a wrong read (a give when the 9 was actually widening) loses yards. And the keep sends Jackson between the tackles, where the linebackers and the safety arrive, on every snap that the 9 widens. That second cost is real enough that the Ravens ran the play in measured doses, which brings us to the formation that made the doses unpredictable.

Read Figure 63.3 as the defense’s coordinator, on the sideline, deciding what to put on the field when three tight ends run on.

If he plays it as a run formation, with base personnel and eight in the box (defenses chose base on about 77% of Baltimore’s 57 such snaps in 2019, a small sample, so read every 13-personnel number here as a tendency), he is still a man short against the run (right panel): eight blockers plus Jackson’s read against eight. To get even he needs the free safety in the box, and then nobody is deep. Meanwhile his linebackers are covering tight ends. Mark Andrews, who led the 2019 Ravens in catches, was a 6-foot-5 receiver in a tight end’s job description; a linebacker in man coverage on him up the seam (left panel) is the mismatch the formation was built to create.

If he plays it as a pass formation, with smaller players to cover the tight ends, the run arithmetic gets worse: smaller defenders to block, and seven of them in the box against eight blockers and a read.

And here is the surprise. The league ran the ball on about 58% of its 13-personnel snaps in 2019, as you’d expect from a goal-line-looking grouping. Baltimore threw on 67% of them.17 Three tight ends didn’t mean “run” for the Ravens any more than three wide receivers meant “pass”: that season they ran on 49% of their 11-personnel snaps (three receivers), against 29% for the league. The personnel stopped being a tell. That is the deepest point of Roman’s design: once the quarterback is a runner, every grouping can run, so every grouping can also throw, and a defense that reads personnel to guess run or pass, as 02-01 teaches you to, guesses wrong more often against Baltimore than against anyone.

Three things make this play the Ravens’ and not just anyone’s.

The fake beats both kinds of linebacker. Linebackers are taught to read one of two things at the snap (their key), and the counter beats each one differently. A linebacker who keys the backfield reads the running back: if he steps left, the run is going left. Against Jackson he also has to wonder whether the quarterback is about to keep the ball and run left himself, so his false step left is bigger, and the cutback right, which is where the counter goes, is bigger too. A linebacker who keys the guards is not fooled about the direction: he sees the guard and the tight end pull right, and the pullers tell the truth. What beats him is the half-second of doubt before he trusts them. The mesh looks like every read play he has seen, and if he flies after the pullers and Jackson hands the ball off going left, he has run himself out of the play. So he hesitates for a beat, and a beat is all the tight end needs to arrive at him with a head of steam.

The pullers are a tight end and a guard, not two linemen. Many counter plays pull the backside guard and tackle. Pulling the tackle leaves a second hole in the backside of the line, where he used to stand, that someone has to step back and fill so a defender can’t run through it into the backfield. Pulling the backside tight end instead keeps the tackle home on the 5, and the tight end is usually the better athlete for the second puller’s job: turning up through the hole and finding a linebacker in space. The running back, after his fake, turns around and blocks the backside 9 (Figure 63.4, panel 4), the job he would have on any counter. Count the blocks: eight, against eight. That is the designed-run arithmetic of 03-04 with the Ravens’ extra tight end supplying the eighth.

It punishes the defense for following the rules. The linebackers who follow the first flow are doing what they’re coached to do. The 9 who squeezes the down blocks is doing what he’s coached to do. Both are wrong, because the play is built to make their rules point the wrong way. That is the pattern of the whole offense: not a trick, but a set of plays where each defender’s textbook reaction is the one the next play attacks.

The price is the one the “Why it exists” box named: Jackson is a ball carrier between the tackles, so the Ravens rationed these runs. How much they rationed them, and how that changed, is in the data.

How opponents responded

If the Ravens’ run game was the league’s best for seven years, what did defenses do about it? They tried four broad answers, and each one worked in some games, often in the games that mattered most.

The trade in Figure 63.7 is stark. The defense concedes the box count on paper (six defenders against seven blockers and a read), and gives up size at every position: a 210-pound safety at linebacker loses a lot of collisions with a pulling guard. In return, every defender can run. Against a quarterback whose best plays are outside, pursuit from the inside is the defense’s real plus-one, and safeties pursue faster than linebackers do. And a defense of seven defensive backs is already in the right personnel if the offense throws.

Why didn’t everybody copy it? Because it was built for a particular offense at a particular moment. The 2018 Ravens were short on receivers, and Jackson, making only his eighth NFL start, could not yet punish a defense for selling out to his legs. Roman’s 2019 rebuild was in large part a reply to this game. The heavy personnel made small defenders a liability, since six defensive backs can’t hold up against two tight ends and a fullback blocking down on them all afternoon. The passing game, aimed at the middle of the field with Andrews, punished defenses that took linebackers off it. The seven-DB idea never became a standard defense against Baltimore; as the base-personnel numbers earlier in the chapter show, most opponents went the other way and got bigger. Its premise, speed in the box can be worth more than size against this quarterback, survives in a milder form: the big nickel, a nickel package whose fifth defensive back is a third safety, big enough to fit a run and fast enough to chase one (02-05 has the packages).

Answer 2: get ahead and make them throw

The answer that worked best wasn’t schematic. It was the scoreboard. A pistol option offense is built on the run, and the run is most valuable when the game is close. Get ahead by two scores and the offense has to throw, and a team that has spent the week preparing for runs is suddenly defending a passing game built from fewer receivers.

Tennessee’s plan, as analysts described it at the time, also included more defenders at the second level who could flow to the ball, a version of answer 1, but the scoreboard did the most damage.25

Answer 3: pressure and zero

The third answer attacked the passing half of the bargain, and accepted a risk against the run to do it. In a Cover 0 pressure there is no deep safety; every defender either rushes or covers a man, and the quarterback has to beat the blitz with a quick throw before it reaches him. Miami did this to Jackson on a Thursday night in November 2021, blitzing on most of his dropbacks and playing zero coverage behind it, and the 2–7 Dolphins beat the 6–2 Ravens 22–10. 10-02 has the blitz numbers and the film.

Why is zero a gamble against this quarterback in particular? Because in zero every defender is either rushing or running with a receiver with his back to the ball, so nobody is left facing the quarterback. If Jackson escapes the rush, or simply runs on a play the defense read as a pass, there is no one to tackle him short of the end zone. Zero works against him only with two refinements. The rushers keep their lanes, each one responsible for one side of the quarterback so that he has no crease to escape through, and the defenders covering the backs play the hug rule (also called a green dog): if the back stays in to block, his man has nobody to cover and rushes too, a step late, often straight at the quarterback. That late rusher is a spy on the move. Miami’s plan had both, and the run game didn’t rescue Baltimore that night either: its designed runs lost value (-0.14 EPA per run on 19 of them), with Jackson held to 21 yards on 5 designed carries.26

Zero also takes away the time a play-action pass needs: a heavy formation’s fake pulls linebackers up, but a blitzing linebacker was coming up anyway. The Ravens’ longer-term answer was the one the next sections describe: a passing game with more receivers who could win quickly, built under a new coordinator.

Answer 4: spy him and disguise it

The fourth answer is the most direct: assign one defender to Jackson, a spy, and hide which one. The cost of a spy is the cost of every answer in this list: a defender assigned to the quarterback is not in coverage and not filling a gap. The results were mixed, which is what you’d expect from a tool rather than a cure. The Chiefs spied him with linebacker Leo Chenal on the 2024 opening night and won 27–20, but not because the spy stopped the run: Jackson ran 16 times for 122 yards and Baltimore’s designed runs averaged +0.26 EPA, in a game decided on its final snap. The Lions’ plan in Week 3 of 2025 worked better: Jackson ran 7 times for 35 yards, the designed runs averaged -0.40, and Detroit won 38–30 in Baltimore, with its coordinator crediting not the spy but a disguise that made every defender look the same before the snap.27 10-02 tells both games.

The everyday answers

The four answers above are the famous ones, built for single games. Underneath them, every defense that played Baltimore used the ordinary tools of option defense from 03-04 and 05-05, mixed snap to snap so the quarterback’s read kept changing:

  • The scrape exchange and the gap exchange. The read end crashes on the back, and a linebacker (or a safety) loops outside him to take the quarterback. Jackson reads “keep” and runs into the man who was waiting for him.
  • The mesh charge. The read end attacks the mesh itself, straight at the point where the ball is being handed off, so the quarterback has to decide in a fraction of a second, and a hurried decision is more often wrong.
  • Make him pay. Give him the keep, and hit him on every one. It concedes yards on the play to tax the quarterback over four quarters and a season, and it is the reason the Ravens rationed his carries.
  • Late safety rotation. Show two deep safeties before the snap, so the box looks light and the read looks easy, then rotate one down into the alley after it. It is the light-box idea of the Fangio-era defenses that 12-08 studies, and it was built in part for quarterbacks like this one.

What the answers have in common

Look at the four together and a pattern appears. None of them stopped the run game for a season; the chart in Figure 63.5 says so. What they did was make the Ravens play from behind, or make them throw against a defense built to stop the throw, in single games. That is the honest summary of the era’s defensive history, and it leads straight to the question every Ravens fan has argued about.

Three readings.

The offense was worse in the playoffs, by more than the opposition explains. Across the Jackson-era regular seasons the Ravens averaged +0.089 EPA per play, an elite offense. In their nine playoff games they averaged -0.008, roughly an average one. Some drop is normal: playoff defenses are better, and they have time to prepare. But the typical playoff offense of these years played about 0.06 EPA per play worse in January than in its own regular season, and Baltimore fell by 0.10, close to twice as much. Nor was it simply the quality of the opponents: in the regular season, against teams that went on to make the playoffs, the Ravens still averaged +0.083, barely below their overall +0.089.28 That part of the narrative is true.

The run game failed early in the era, and not at the end. Overall, the Ravens’ designed runs fell from +0.08 EPA per play in the regular season to -0.02 in the playoffs, but the diamonds in Figure 63.8 show where that came from. The runs lost value in each of the first four losses: the Chargers in January 2019 (the seven-defensive-back game), the Titans in January 2020, Buffalo in January 2021 and the Huntley game in Cincinnati in January 2023. They gained value in the last two losses and in all three wins. Take out just the January 2019 and January 2020 games and the playoff designed runs averaged +0.06. So the early answer to “was the run game solved?” is partly yes, in single games, by defenses with a week to prepare; the later answer is no.

The losses came from falling behind and giving the ball away. The right panel is the mechanism. In the regular season the Ravens trailed by more than a touchdown on only 10% of their snaps; in the playoffs, on 37%. They turned the ball over 1.1 times a game in the regular season and 1.8 in the playoffs, with three or more turnovers in 4 of the 6 losses. Did the staff abandon the run on its own, or did the score take it away? The split by score answers it. With the game within a touchdown either way, the Ravens called designed runs on 47% of their playoff snaps, against 45% in the regular season: in close games they ran as much as ever. The overall run share fell, from 46% to 41%, only because they spent so much more of their playoff time behind. That five-point fall looks small, but it is an average over nine games; in the two worst deficits (January 2020 and January 2024) the designed-run share was 23% and 18%. This is answer 2 from the last section, playing out across a decade: a run game built to win close games was asked, again and again, to win from behind.

Look at the losses one at a time and both stories appear. In January 2019 against the Chargers and January 2020 against the Titans, the run game was stuffed, and early deficits and turnovers finished the job. In January 2021 at Buffalo, a 3–3 game at halftime turned late in the third quarter: down 10–3, Jackson drove the Ravens to the Bills’ 9 and threw an interception that Taron Johnson returned 101 yards for a touchdown, and two plays later Jackson left with a concussion, and the Ravens lost 17–3 with Tyler Huntley at quarterback.29 In January 2024 against Kansas City, the Ravens trailed 17–7 at halftime and ran only 16 times all game, with their running backs carrying six of them; their designed runs averaged +0.17 EPA, better than their regular-season average, and they lost 17–10 after three turnovers.30 In January 2025 at Buffalo they ran for 176 yards and still lost 27–25, after an interception, a fumble on a sack and a Mark Andrews fumble, then a failed two-point conversion that would have tied the game.31 The later losses look less like a scheme being solved and more like an offense built on the run being taken out of it by the scoreboard.

How much of that is bad luck? Some, certainly: nine games, about 597 plays, is a small sample, and turnovers in particular are noisy from game to game. But a pattern repeated across six seasons, four lead running backs (Gus Edwards, Ingram, Dobbins and Henry) and three coordinators (Mornhinweg, Roman and Monken) deserves an explanation, and the most honest one is this: the pistol option offense was at its best protecting a lead or playing even, and the games it lost were the games where it had to do the thing it was built to do least. The 2023 change of coordinator was, in part, an attempt to fix exactly that.

NoteMadden vs. real life

In a video game, a great offense is great in every situation, because you can call any play from any score. Real offenses are built for a range of situations: personnel, practice time and a quarterback’s skills all get spent on the situations the staff expects most. An offense built to run is spending its practice week on runs, and when it falls behind by fourteen, it is asking players to win with the part of the playbook they practised least. That is a trade-off, not a flaw, and every offense makes one.

The Monken changes (2023)

After the 2022 season Greg Roman left. The Ravens’ offense under him had ranked first, seventh, 17th and 19th in points in his four years as coordinator, and the passing game had never quite caught up to the run.32 In February 2023 Baltimore hired Todd Monken, who had just won two straight national championships as Georgia’s offensive coordinator and had run NFL offenses in Tampa Bay and Cleveland.33

Monken’s mandate was to make Jackson a quarterback who could win from the pocket, without throwing away what made him dangerous. Figure 63.6 shows what he did:

  • From the pistol to the shotgun. Pistol snaps fell from 26% of plays in 2022 to 7% in 2023, and the shotgun rose to 78%.34 The shotgun gives the quarterback a better view of the coverage and a faster launch, at the cost of the pistol’s downhill back and its lack of a tell.
  • Fewer designed quarterback runs. Jackson’s designed runs fell from 5.5 a game to 4.1. He still ran, but more of it was scrambling, improvising when a pass play broke down, rather than running on purpose.
  • More receivers, and more passing on early downs. The Ravens drafted receiver Zay Flowers in the first round, used three-receiver sets more, and on neutral early downs ran the ball on 43% of plays, 27th in the league, after ranking first in that category in 2019.

The result was a career-high 3,678 passing yards for Jackson and his second MVP award, won with 49 of 50 first-place votes, and a 13–4 record and the AFC’s top seed.35 The run game didn’t collapse without the volume of quarterback runs. It ranked 3rd in EPA per designed run, and the Ravens still led the league in rushing yards. The likeliest reason is that the threat was doing some of the work the carries used to do: a backside end who had to respect Jackson every snap was still frozen, whether Jackson ran six times or four. Play-by-play can’t see a frozen end, but it can see the runs Jackson didn’t carry, and they held up. Baltimore’s designed runs by its running backs and receivers averaged +0.002 EPA in 2023, 5th in the league and well above the league’s -0.08; in Roman’s last season, with more Jackson carries, they had ranked 9th.36

NoteFilm room: AFC Championship, Kansas City at Baltimore, January 28, 2024

The Ravens beat Houston 34–10 in the divisional round, then hosted the Chiefs for a trip to the Super Bowl. They lost 17–10, and the game became the sharpest criticism of the Monken offense: Baltimore ran only 16 times, and its running backs carried just six of those.37

What to notice: the first half and the turnovers, not the play-calling alone. Kansas City led 17–7 at halftime, and the Ravens turned the ball over three times: a fumble on a sack in the second quarter, then, in the fourth, Zay Flowers’s fumble at the Kansas City 1-yard line and an interception in the end zone. Watch Steve Spagnuolo’s defense on Jackson’s dropbacks: four-man rushes, disguised coverages, and rushers who kept their lanes so that Jackson’s scrambles went nowhere. Then ask the question the chapter keeps asking: was the run abandoned, or was it taken away by the scoreboard? The answer is some of both, and it was the last time the Ravens’ run game rested mainly on Jackson.

The Derrick Henry seasons (2024–2025)

In March 2024 the Ravens signed Derrick Henry, the 30-year-old former Titans back who had run for 2,027 yards in 2020 and had beaten them almost single-handedly in that January 2020 playoff game, on a two-year contract.38 It was a signing that made the threat run both ways. For five years, defenses had to decide whom to fear on a read: the quarterback or a good running back. Now the back was one of the most punishing runners of his generation, a 247-pound back who had led the league in rushing twice.

2024 was the best offense of the era. Henry ran for 1,921 yards and 16 touchdowns, second in the league in yards behind Saquon Barkley’s 2,005; Jackson threw for 4,172 yards, 41 touchdowns and only four interceptions; and Baltimore led the NFL in EPA per play, in rushing yards, and, again, in EPA per designed run.39 The team went 12–5, beat Pittsburgh in the wild-card round as Henry ran for 186 yards, and lost at Buffalo, the game described above.

The run game moved under center. The data shows how differently the Henry offense was built. The Ravens took 28% of their 2024 snaps with Jackson under center, nearly three times their 2022 rate, and 45% of their designed runs started there.

Why under center, after five years of selling the pistol, and for the most downhill back in the league? The pistol and the I-formation both put the back about seven yards deep, so depth isn’t the difference. The differences are what happens at the mesh and what the defense can see.

  • A straight track to the hole. From the pistol the quarterback stands four yards deep, directly in the back’s way, so the back has to bend around his hip to take the handoff. From under center the quarterback opens out of the back’s path and hands off on the move, and the back runs his zone or duo track with his shoulders square to the line. That was the game Henry had built his career on: from 2019 to 2023 no player ran for more yards with his quarterback under center (5,795, by ESPN’s count). His former Tennessee teammate Taylor Lewan said publicly that Henry wanted to run downhill and disliked turning his shoulder on read plays, although Henry himself said he was comfortable with them.40
  • Play-action with the quarterback’s back to the defense. Under center, the quarterback turns away from the line to fake the handoff, so linebackers and safeties can’t see the ball, and the fake to a back as feared as Henry holds them longer. Henry’s Titans had averaged more yards per play-action pass than any other team since 2016, and the Ravens’ +0.31 EPA per dropback in 2024 ranked 1st in the league.
  • A quarterback who was now a passer first. Monken had already moved Jackson’s game toward the pocket. With Henry carrying the load, the offense no longer needed Jackson’s legs on every snap, and under center it no longer had to show them.

What it gave up was the read: from under center the quarterback can’t easily read an end and keep the ball. So the read game didn’t vanish; it became a change-up, and it kept its sting. Henry’s carries from the pistol, 92 of them across the two seasons, averaged 7.8 yards and +0.21 EPA, against 5.1 yards and +0.09 EPA from under center.41 Read that comparison with care. It is a small sample, and a change-up is called when the staff thinks the defense is set up for it, so its numbers flatter it; the same play called on every snap would not average that much. But it is the read offense in miniature: an end who has to stay home for Jackson is an end who isn’t chasing Henry.

What to notice: Jackson’s alignment before each run. Under center, watch the back’s track and the second level: the linebackers have to fit Henry’s downhill run, not a read. 2025 was the end. Jackson missed four games, three with a hamstring injury in October and one with a back injury in December, and was less of a runner when he played, saying afterwards that his health had limited his mobility:43 by this chapter’s count 28 designed runs in 13 games, about two a game. Henry still ran for more than 1,500 yards, the run game still ranked 3rd in EPA per designed run, and the Ravens ran on early downs more than all but one team. But a 1–5 start, a defense that gave up 398 points, and a 26–24 loss at Pittsburgh in Week 18 with the division on the line left them 8–9 and out of the playoffs.44 On January 6, 2026, the Ravens fired John Harbaugh after 18 seasons.

What survives: the era closed, and the 2026 epilogue

The coaches have scattered. Harbaugh is the Giants’ head coach in 2026, Monken is the Browns’ head coach, and Roman spent 2024 and 2025 as Jim Harbaugh’s offensive coordinator with the Chargers, who fired him in January 2026.45 The 2026 Ravens belong to Jesse Minter, a Ravens assistant from 2017 to 2020 and the Chargers’ defensive coordinator in 2024 and 2025, as head coach, with Declan Doyle, who came from Ben Johnson’s Chicago staff, as offensive coordinator and Anthony Weaver as defensive coordinator.46 Jackson is still the quarterback. What Doyle builds around him is a 2026 story, and this book’s coverage stops at the end of the 2025 season, so whether his Ravens line up in the pistol, the shotgun or under center is a question for your own Sunday viewing. The Watch-for-it list below is how to answer it.

What survives the era is bigger than one team.

  • The arithmetic is now conventional wisdom. “Count the quarterback as a blocker” was an option coach’s slogan in 2018. By 2025 it was how every NFL defense planned for Jalen Hurts, Josh Allen and Jayden Daniels, and how every analyst explained why those offenses ran well against light boxes.
  • Heavy personnel came back. In 2025 NFL offenses used fewer than three receivers on 41.7% of plays, the most since Next Gen Stats began tracking.47 The Ravens weren’t the only cause (the 49ers, Rams and Eagles all pushed the same way), but they showed that a modern offense could live in 12, 21 and 22 personnel and still throw.
  • The quarterback run became a managed resource. The Ravens’ own data tells the story of the last seven years in miniature: from seven or eight designed runs a game in 2019 to about two in 2025, with the efficiency of the run game barely moving. The lesson the league took is that a running quarterback’s value is mostly in the threat, which can be spent a few carries at a time, on the downs and in the games where it matters, and saved the rest of the time.
  • The pistol didn’t. The pistol as a base alignment was a Ravens thing, and it left with Roman. It survives as a run-heavy niche look (02-02’s data has a few teams living in it), and as a change-up that offenses built around the shotgun or under center use when they want a read, as the Henry-era Ravens did. That is a good example of the course’s larger point: the principle (an extra blocker, either side, no tell) outlived its favourite formation.
TipWatch for it

On Sunday: count the defenders against a running quarterback. Any team with a quarterback who runs, Baltimore or not, will do.

Before the snap

  • Count the blockers: five linemen, plus every tight end and fullback near the line.
  • Count the box: the box count is every defender within about 5–7 yards of the line and inside the tight ends.
  • Add the quarterback. If he’s in the pistol or shotgun with a back beside or behind him and is a real runner, add one to the offense’s side. Now compare: more defenders than blockers-plus-one, and the defense has its plus-one; even, and the defense’s extra man has to come from a safety.
  • Find where the extra man is. A safety walked down at eight yards, or a single deep safety in the middle? Then the play-action pass is aimed at him.
  • Where is the back? Directly behind the quarterback (pistol): no directional tell. Beside him: note the side. On zone runs and zone reads the run usually goes the other way; gap runs from the shotgun often go to the back’s side, so look for a puller before you commit.

At the snap

  • Watch the two ends. The one nobody touches is being read. Does he stay home (give coming) or chase (keep coming)?
  • Look for a puller. A guard pulling toward the unblocked end: power read. A guard and a tight end pulling the same way while the quarterback and the back step the other way: QB counter. The same two pullers with the back following them: counter read or counter bash, with the quarterback reading an end.

After the play

  • On the replay, check the backside end on a plain handoff. Did he chase the cutback, or stay home for a quarterback who didn’t have the ball? Count how often he stays, over a quarter. That is the hidden yardage.
  • Note the score. Is the offense still running when it’s behind by two scores? That’s the question that decided the Ravens’ playoff games.

Seven blockers plus Jackson’s read against eight: a hat on every hat. Five linemen and two tight ends make seven blockers. The box has eight: five linemen, the walked-down strong safety on the right edge, and two linebackers. On a handoff with a pocket quarterback the defense would be plus-one. With Jackson reading one end, it is even (Figure 63.1, Figure 63.2).

A defense that shows this picture can’t be right about both, which is why the Ravens could call either.

TipAnswer

Give. The 9 has squeezed inside, toward the hole the quarterback would run through: he has taken the keep away himself. So Jackson hands the ball to the back, who is already running outside the 9 at full speed (Figure 63.2).

Who has to tackle him? Probably nobody in the box. Every other box defender has a blocker assigned (the down blocks, the right guard climbing to the Will, the pulling guard leading for the Mike), and the 9 is inside the runner. The one box defender with a chance is the Mike, if he reads the play fast and beats the pulling guard over the top, which is why a give is never a sure thing. Otherwise the tackle has to come from outside or behind: the cornerback on that side (if the wide receiver hasn’t blocked him) or the free safety, coming from 12 yards deep across the field. That is the power read’s promise, a running back in space against a defensive back, and its risk: if the corner or safety arrives in time and the back was running sideways, the gain is small.

If the 9 had widened with the back instead, the answer flips: keep, and Jackson runs inside him behind the pulling guard.

TipAnswer

The box says run. It is a light box: six defenders against six blockers (five linemen and the tight end) and a quarterback who reads: the offense is effectively plus-one. Against light boxes, efficient running teams do well, and the Ravens were the most efficient running team of their era (Figure 63.5).

The scoreboard says “don’t panic.” Down 14 with more than a quarter and a half to play is two possessions, not a two-minute drill. A good run on second-and-8 is a good play in any quarter, and this two-high shell is inviting you to throw: if you throw against two deep safeties and five underneath (the nickel, both linebackers and both corners, with four rushing), you are throwing into the coverage the defense chose. (The win-probability tools of 13-03 make this precise.)

But is running really better for this team? That is the comparison that decides it, and it isn’t automatic: 10-01 showed that passing is usually the more efficient choice, and the 2019 Ravens’ dropbacks (+0.32 EPA) were worth more per play than their runs (+0.11). Two things tip this snap toward the run. The box: against six or fewer, Baltimore’s designed runs averaged +0.09 EPA across 2019–2025, against the league’s -0.02. And the score: in the Jackson-era regular seasons, when the Ravens trailed by more than a touchdown, their designed runs averaged +0.11 EPA and their dropbacks +0.06, because a defense protecting a lead plays pass and leaves the box light.48

The trap is the one in Figure 63.8: a deficit that pulls a run team off the run, and the turnovers that come with throwing from behind. The right call here is the run the defense is conceding (a read, or a designed run with the back blocking), and the right plan is to stay patient for another series. The two-score deficit becomes a three-score deficit far more often from an interception than from a three-yard run.

Takeaways

  • The 2019–2022 Ravens built a pistol option offense: the pistol as the base alignment, heavy personnel, a menu of option reads and designed quarterback runs, so that the defense had to account for one more ball carrier on every snap.
  • A running quarterback changes every run fit, not just the reads: the end he might read has to stay home on plain handoffs too, every defender’s fit gets a “quarterback” line, and backside pursuit stops being free. An eight-man box against him is a seven-man box.
  • The signature plays each make a defender’s textbook reaction wrong: the pistol inverted veer (squeeze the hole and the back runs outside you), 13 personnel (play it as a run formation and the seam opens; Baltimore threw from it two times in three), and the QB counter (follow the first flow and the play goes back the other way).
  • The data fingerprint: Baltimore’s designed runs were above the league average every season from 2018 to 2025, top three in six of them, under two coordinators and five lead backs. What changed was the recipe: pistol and quarterback runs under Roman, the shotgun and fewer designed runs under Monken (2023), under-center runs with Derrick Henry (2024–2025), chosen for his downhill track and for play-action, with the pistol read kept as a change-up.
  • Defenses answered with speed, pressure, spies, the everyday option answers and, most effectively, the scoreboard. In the playoffs (3–5 with Jackson) the offense fell off by about twice the usual playoff drop. Early in the era prepared defenses stuffed the run in single games; later, the losses came from deficits and turnovers. In close playoff games the Ravens ran as often as ever.
  • The era closed in January 2026 with Harbaugh’s firing. What survives is the arithmetic (count the quarterback as a blocker), heavy personnel, and the idea that a running quarterback’s value is mostly the threat, spent a few carries at a time.
