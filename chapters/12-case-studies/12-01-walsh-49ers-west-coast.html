def line():
    """The five offensive linemen with one-letter labels, T G C G T."""
    short = {"LT": "T", "LG": "G", "C": "C", "RG": "G", "RT": "T"}
    return [p.moved(label=short[p.key]) for p in offense("singleback") if p.key in OL_KEYS]


def T(t, side=1):

def split_backs(y_side=1, z_w=13.0, x_w=-13.0, h_side=-1):
    """Walsh's base look: split backs (the pro set), one back behind each tackle, a tight end, a split end (X)
    and a flanker (Z). The halfback (H) is on the offense's left by default; h_side=+1 puts him on the right."""
    return line() + [O("QB", "QB", -1.9, 0.0, "Q"), O("HB", "RB", -5.0, 3.2 * h_side, "H"),

                     O("Y", "TE", -0.7, 4.8 * y_side, "Y"), O("X", "WR", -0.7, x_w, "X"),
                     O("Z", "WR", -1.5, z_w, "Z")]


def era_43(shell="one", ss_side=1):
    """A 1980s 4-3 Over against a pro set (tight end on the offense's right), corners 7 off."""
    dfn = [D("WE", "DE", 1.0, T("5", -1), "E"), D("WT", "DT", 1.0, T("1", -1), "T"),

def go_to(p, key, end, kind="block", delay=0.0, speed=None, via=()):
    """Move `key` through absolute (d, w) points `via` to `end`, as a block, pull, path, run, rush or route."""
    start = np.array(p._cur(key))
    pts = [tuple(np.array(q) - start) for q in (*via, end)]
    if kind == "block":
        return p.block(key, pts=pts, delay=delay, speed=speed)
    if kind == "pull":
        return p.pull(key, pts, delay=delay, speed=speed)
    if kind == "rush":
        return p.rush(key, pts=pts, delay=delay, speed=speed)
    if kind == "route":
        return p.route(key, [tuple(q) for q in (*via, end)], absolute=True, delay=delay, speed=speed)
    return {"path": p.path, "run": p.run}[kind](key, pts, delay=delay, speed=speed)


def pass_pro(p, keys=OL_KEYS, depth=-1.2):

def clean(ax, numbers=True):
    """Drop hash ticks (and optionally yard numbers); clip text to the window."""
    for ln in list(ax.lines):
        xs, ys = ln.get_data()
        if len(xs) == 2 and ln.get_linewidth() == 0.5 and np.hypot(xs[1] - xs[0], ys[1] - ys[0]) < 1.0:
            ln.remove()
    for t in list(ax.texts):
        if numbers and t.get_text().isdigit() and t.get_alpha() == 0.55:
            t.remove()
        else:
            t.set_clip_on(True)

MARK_R = 0.62


def tracking(play):
    df = play.to_tracking()
    df["_id"] = df["displayName"].astype(str)
    return df


def catch_time(df, key, t_throw):
    """First tracking time after the throw when the ball reaches `key` (the simulated catch)."""
    b = df[df.side == "ball"].set_index("time")[["x", "y"]]
    q = df[df.displayName == key].set_index("time")[["x", "y"]]
    gap = ((b - q) ** 2).sum(axis=1) ** 0.5
    gap = gap[gap.index > t_throw + 0.05]
    return float(gap[gap < 0.6].index[0])


def panel(ax, play, df, t, lateral, window, rings=(), ring_r=0.95, mark_r=None, trails=None):
    """One moment of a simulated play on `ax`: markers, trails (only for the keys in `trails`, if given), rings."""
    tmap = df.groupby("frameId")["time"].first()
    fid = int((tmap - t).abs().idxmin())
    ylo, yhi = sorted((play.ball_y - lateral[0], play.ball_y - lateral[1]))
    fld = Field(ax=ax, orient="vertical", los=play.los, window=window, numbers=False, hashes=False,
                lateral=(ylo, yhi), to_go=play.to_go)
    cur = df[df["frameId"] == fid]
    shown = cur[cur["x"].between(play.los + window[0] + 0.5, play.los + window[1] - 0.3)
                & cur["y"].between(ylo + 0.5, yhi - 0.5) | (cur["side"] == "ball")]
    ids = set(shown["_id"])
    keep = ids if trails is None else ids & set(trails)
    tr = df[(df["frameId"] <= fid) & df["_id"].isin(keep) & (df["side"] != "ball")]
    for (_, side), g in tr.groupby(["_id", "side"]):
        xs, ys = fld.xy(g["x"].values, g["y"].values)
        ax.plot(xs, ys, color=style.OFFENSE if side == "off" else style.DEFENSE, lw=0.9, alpha=0.4, zorder=3)
    ppy = fld.pts_per_yard()
    mr = mark_r or MARK_R
    for _, r_ in shown.iterrows():
        x, y = (float(v) for v in fld.xy(r_["x"], r_["y"]))
        if r_["side"] == "ball":
            ax.add_patch(Ellipse((x, y + 0.45), 0.75, 0.45, fc=style.BALL, ec="white", lw=0.5, zorder=12))
            continue
        lab = str(r_.get("label", "") or "")
        if r_["side"] == "off":
            ax.add_patch(Circle((x, y), mr, fc=style.OFFENSE, ec="white", lw=0.6, zorder=8))
            tc = "white"
        else:
            ax.add_patch(RegularPolygon((x, y), 4, radius=mr * 1.3, orientation=np.pi / 4,
                                        fc=style.DEFENSE_TINT, ec=style.DEFENSE, lw=1.0, zorder=8))
            tc = style.INK
        fs = max(4.0, min(8.0, ppy * mr / MARK_R * (0.85 if len(lab) <= 1 else 0.6)))
        ax.text(x, y, lab, ha="center", va="center", fontsize=fs, color=tc, fontweight="bold", zorder=10)
    for key, col in rings:
        r_ = cur[cur["displayName"] == key].iloc[0]
        x, y = fld.xy(r_["x"], r_["y"])
        ax.add_patch(Circle((float(x), float(y)), ring_r, fc="none", ec=col, lw=1.6, zorder=7))
    for patch in ax.patches:
        patch.set_clip_on(True)
    return fld


def strip(moments, lateral, window, notes=None, rings=(), ncols=2, width_in=6.2, after=None, mark_r=None,
          trails=None, note_h=0.8):
    """Numbered frame strip. `moments` is a list of (play, t, tag)."""
    cache = {}
    nrows = int(np.ceil(len(moments) / ncols))
    pw = width_in / ncols
    span_w, span_d = lateral[1] - lateral[0], window[1] - window[0]
    fig, axes = plt.subplots(nrows, ncols, squeeze=False,
                             figsize=(width_in, nrows * (pw * span_d / span_w + (note_h if notes else 0.25))))
    for ax in axes.flat[len(moments):]:
        ax.axis("off")
    for i, (ax, (play, t, tag)) in enumerate(zip(axes.flat, moments)):
        if id(play) not in cache:
            cache[id(play)] = tracking(play)
        fld = panel(ax, play, cache[id(play)], t, lateral, window, rings, mark_r=mark_r, trails=trails)
        if after:
            after(fld, play, t, i)
        lab = "snap" if abs(t) < 1e-6 else f"{t:+.1f} s"
        ax.set_title(f"{i + 1}.  {lab}" + (f"   {tag}" if tag else ""), fontsize=8.3, color=style.INK_2, loc="left")
        if notes:
            ax.text(0.5, -0.03, notes[i], transform=ax.transAxes, ha="center", va="top",
                    fontsize=7.1 if ncols < 3 else 6.6, color=style.INK)
    fig.tight_layout(h_pad=1.3 if notes else 0.6, w_pad=0.6)
    return fig


def video(play, notes, lateral, window, rings=(), width_in=4.8, after=None, mark_r=None, trails=None):
    """Web version: the play frame by frame, with each moment's note shown as the caption."""
    import shutil
    from matplotlib import animation
    from IPython.display import HTML, display
    span_w, span_d = lateral[1] - lateral[0], window[1] - window[0]
    fig, ax = plt.subplots(figsize=(width_in, width_in * span_d / span_w + 0.5))
    df = tracking(play)
    tmap = df.groupby("frameId")["time"].first()
    frames = [float(tmap[f]) for f in sorted(df["frameId"].unique())]

    def update(k):
        t = frames[k]
        ax.clear()
        fld = panel(ax, play, df, t, lateral, window, rings, mark_r=mark_r, trails=trails)
        if after:
            after(fld, play, t, -1)
        cur = [n for tt, n in notes if tt <= t + 1e-6]
        ax.set_title(("snap" if abs(t) < 0.05 else f"{t:+.1f} s") + (("\n" + cur[-1].replace("\n", " ")) if cur else ""),
                     fontsize=8.2, color=style.INK, loc="left")
        return []

    anim = animation.FuncAnimation(fig, update, frames=range(len(frames)), interval=100)
    html = (anim.to_html5_video(embed_limit=40).replace("<video ", '<video style="max-width:100%;height:auto" ')
            if shutil.which("ffmpeg") else anim.to_jshtml(default_mode="once"))
    plt.close(fig)
    display(HTML(html))


def animation_or_strip(play, moments, lateral, window, rings=(), after=None, ncols=2, width_in=6.2, mark_r=None,
                       trails=None, note_h=0.8):
    """Video on the website; a numbered, captioned frame strip in the PDF. moments = [(t, tag, note)]."""
    if pdf_build():
        strip([(play, t, tag) for t, tag, _ in moments], lateral, window, notes=[n for *_, n in moments],
              rings=rings, ncols=ncols, width_in=width_in, after=after, mark_r=mark_r, trails=trails, note_h=note_h)
        plt.show()
        return
    video(play, [(t, n) for t, _, n in moments], lateral, window, rings=rings, after=after, mark_r=mark_r,
          trails=trails)

Third-and-3 at the Dallas 6, fifty-eight seconds left in the 1981 NFC Championship Game, the 49ers behind 27–21. Joe Montana takes the snap and sprints to his right with three Cowboys chasing him toward the sideline. His first receiver has slipped. Montana pump-fakes, backpedals, and throws the ball high toward the back of the end zone, so high that the Cowboys’ defensive end Ed “Too Tall” Jones, who had left his feet on the fake, can’t reach it. Dwight Clark jumps and gets his fingertips to it. It’s the most famous catch in the sport, and it gets called a miracle, a desperation heave, a throwaway that a receiver happened to catch. Clark always said it was none of those things: it was a play from the book, run on the practice field so many times that he knew where the ball would be.1 Which was it? The problem: 2–14, and no way to run the ball

In January 1979 the San Francisco 49ers hired a 47-year-old Stanford coach to fix a franchise that had just gone 2–14. Bill Walsh had waited a long time for the job. He had coached in pro football since 1966, with the Raiders (then an AFL team) and then for eight seasons with Paul Brown’s Cincinnati Bengals, where he ran the passing game. When Brown retired from coaching after the 1975 season he picked another assistant, Bill “Tiger” Johnson, as his successor; Walsh resigned, and later said Brown had worked against him when other teams asked about him for head-coaching jobs. He spent 1976 as the Chargers’ offensive coordinator, coaching a young Dan Fouts, and two seasons winning at Stanford before the 49ers called. He was also the general manager, which mattered: the man drawing the plays was the man choosing the players.2

What he inherited was a team with no good answer to the oldest question in football: how do you keep the ball? In the 1970s the answer was the run. A team that could gain four yards on first down and four more on second down controlled the clock, the field and the game, and passing was what you did when that failed. Walsh’s 49ers couldn’t do it. They didn’t have the linemen to push other teams around, and Walsh had been in exactly this spot before. Of the young expansion Bengals he had coached in the late 1960s, he said: “We couldn’t control the football with the run; teams were just too strong. So it had to be the forward pass, and obviously it had to be a high-percentage, short, controlled passing game.”3

That sentence is the whole chapter in miniature. Walsh didn’t decide to pass more instead of running. He decided that the pass could be his running game, doing the same job (four yards at a time, the clock moving, the ball kept) with players and plays he could actually win with. He had already proved it once, with Virgil Carter in Cincinnati (the story is in The Quick Game); San Francisco let him build a whole team around it, just after the 1978 rules had made it far easier.

You’ve met the outline of this story in 1978 and the Passing Revolution: the hire, Montana in the third round of the 1979 draft (with Dwight Clark in the tenth), 2–14 and then 6–10, and Super Bowl XVI two years later.4 This chapter is the version with the clock running: what the plays looked like snap by snap, why each piece was there, and what happened to the system after its inventor left.

The scheme answer: the pass as a run

Here is the idea in one term. A horizontal passing game is an offense built on short, timed throws toward the sidelines and across the field, caught a few yards past the line of scrimmage and turned into gains by the run after the catch. Its throws are so short that they’re often described as extended handoffs.5 The name contrasts it with the vertical passing of Don Coryell’s Chargers, which attacked down the field.

What does a run actually do for an offense? Three things. It gains a few yards with little risk of losing many. It puts the ball in a fast player’s hands while he’s moving forward, so he can make a defender miss. And it attacks the defense at a spot the offense chose. A short pass to the flat does all three, and Figure 59.1 shows the comparison that Walsh’s offense made every week: two ways to get the halfback to the right edge with the ball.

Look at what the pass removes. The toss sweep is a bet that four blockers each win their fight in the same second; the flat pass is a bet that one throw is accurate. Against a defense that’s bigger and stronger than your line, the second bet is the better one, and in 1979 the 49ers’ line was the weak point. The pass also changes who does the tackling. On a run the first defenders to arrive are linemen and linebackers, the biggest men on the field. On a quick throw to the edge the tackler is usually a defensive back, 30 or 40 pounds lighter, and often backing up when the ball arrives.

Then there is the horizontal stretch, which is what makes the throw hard to take away. You can see it in the right panel: in this Cover 3 the strong safety is the curl-flat defender, responsible for both the tight end’s curl and the flat, and two receivers are in his area at once. Whichever he takes, the quarterback throws the other. A zone defender who covers the flat can’t also cover a route breaking inside him, so the offense puts a receiver on each side of him and throws to whichever he leaves. Spread that across the field (the backs releasing to both flats, receivers crossing underneath) and every short defender has two receivers to worry about. The defense can’t add enough players underneath to cover them all without emptying the deep part of the field. That, Walsh argued, was when the long throw came open.

ImportantWhy it exists, and what it costs

The horizontal passing game exists because a team that can’t win at the line of scrimmage can still win in space. It trades the run’s physical contest for a contest of precision and speed: the ball gets to the edge before the defense does, and the tackle is made, or missed, by a smaller man in open field.

The bill comes in four parts. It needs an accurate quarterback, more than a strong-armed one: a throw a yard behind the receiver turns a five-yard gain into a tackle at the catch. It needs receivers who run after the catch, because the throw itself only travels a few yards. A short throw can still be intercepted, and a linebacker who reads it and jumps it has a clear run the other way. And it gets squeezed: once a defense stops worrying about the deep ball, its underneath defenders can crowd the short throws, which is why Walsh insisted the offense keep threatening deep. A West Coast team that only throws short isn’t running Walsh’s offense.

CautionCommon misconception

“The West Coast offense is dink-and-dunk.” Walsh hated the word finesse, which writers attached to his offense, and the record supports him.6 The short throws were the setup, not the point: they pulled linebackers and safeties forward, and the 49ers then threw over them. Jerry Rice caught 22 touchdown passes in twelve games in 1987, many of them long runs after slants and deep throws, which is not what dink-and-dunk looks like.7 What is true is that the offense was built so that a short gain was an acceptable result, which most 1970s passing games couldn’t say.

And the running game came back. “Couldn’t run” describes the 49ers Walsh inherited, not the team he built. In 1979 they ran on 44% of their plays, against 54% for the league as a whole, and threw more passes than any other team. By 1981 they ran on 52% (league: 51%), and in 1984 on 52% again (league: 49%), when Wendell Tyler set a team record with 1,262 rushing yards.8 The short passing game didn’t replace the run; it made the run possible again. Linebackers who had to respect two backs releasing into routes, and safeties pulled forward by short throws, left fewer defenders to stop a handoff. The run in turn gave the 49ers their play-action passes: a fake to a back from split backs, with the quarterback under center and his back to the defense, froze linebackers who had been burned by both. That under-center play-action is the bridge to Mike Shanahan’s offense, later in this chapter.

Design goal 1: timing

Walsh’s offense is a timing offense, the purest version of the idea you met in Passing Fundamentals: the quarterback’s drop and the receiver’s route are designed as one unit, so that the ball leaves his hand at the moment the receiver breaks, and arrives before the defender can react. That chapter’s chart of drops and route depths is the West Coast offense’s skeleton: a three-step drop for quick routes caught about 5 to 7 yards downfield (a slant breaks after three hard steps, about 4 yards, and is caught at 6 or 7), a five-step drop with a hitch for routes at 10 to 14, a seven-step drop for the deep comebacks and posts.

On a broadcast replay, a timed throw has a look you can learn. The ball leaves the quarterback’s hand before the receiver has finished his break, often before the receiver has turned his head, and the receiver catches it without slowing down. A throw that waits until the receiver is visibly open, and standing still, is a late throw, however good the result.

Two things made Walsh’s version different from a generic timing game.

The quarterback’s feet were his reading clock. Each step of the drop meant a receiver in the progression. On the last step of a quick drop he threw to the first receiver if he was open. If not, the hitch step that followed was his permission to go to the second, and so on down to the checkdown. This is why the offense could be taught: a quarterback didn’t have to decide when to give up on a receiver, because his feet decided for him.

The backs were receivers, not spare blockers. The original West Coast offense lined up in split backs: one back behind each tackle, a tight end, and two wide receivers.9 On a pass play both backs had rules that turned them into receivers. Each checked his blocking assignment first (a linebacker who might blitz) and released into a short route if nobody came, which is the check-release. A defense that covered all five possible receivers had to account for both backs, and a defense that blitzed found a back already in its way.

When the blitz brought more rushers than the backs and line could block, the answer was the same as in any dropback offense: a hot route, a receiver (often the tight end or the back on the blitz side) who knows to look for the ball immediately, and a sight adjustment by a wide receiver who converts his route into a quick slant when he sees the blitz. The quick throw goes where the blitzer came from. (The protection side of this is in Pass Protection.)

Compare it with the plate in Dropback Concepts, drawn from a modern spread, and you’ll notice two differences. The first is the read order. That chapter reads drive low to high, shallow first, because against man coverage the shallow is open early or not at all. Walsh’s version is read high to low, because of the drop. On a five-step drop with a hitch, the dig is timed to arrive just as the quarterback finishes his drop and hitches up, so it gets the first look; the shallow keeps running across the field and is still available on the next hitch; the backs come last. Dig on time, shallow late, back last. Both orders work, because the Mike still can’t take both routes; what changes is which one the timing favours.

The second difference is where the outlets come from. In the spread they’re receivers; in Walsh’s version they’re the two backs, who were also the offense’s protection a second earlier. That double duty is the West Coast offense in one picture: the same eleven players doing two jobs, so the defense has to cover more men than it expected.

Design goal 2: yards after the catch

Every one of these throws asks the receiver to keep running after he catches it. That’s not a happy accident; it’s what the routes were built for. The slant catches the receiver moving across the field at full speed. The shallow cross catches him running away from the defender who started across from him. The flat route catches the back moving toward the sideline with the cornerback backpedaling away from him, at least against Cover 3, where the corner is bailing to his deep third. (Against Cover 2 the corner squats in the flat, so the flat throw is the one being taken away, and the West Coast answer was to throw behind him instead: the curl, or a corner route into the hole between the corner and his safety.) In each case the receiver is already in motion when the ball arrives, and the nearest defenders are either behind him or moving the wrong way.

A run after the catch is worth more than it looks. On the slant in particular, the receiver catches the ball in the gap between the underneath zone and the deep zone, the only defenders who can tackle him are a step behind, and one missed tackle can turn seven yards into seventy. That’s how the 49ers’ short offense produced long touchdowns.

It also explains the 49ers’ roster. Walsh’s best players were the ones who could turn a short catch into a long gain:

  • Jerry Rice, a first-round pick in 1985 out of Mississippi Valley State, who finished his career with more catches, receiving yards and touchdowns than anyone in NFL history (1,549, 22,895 and 208 total touchdowns).10 His signature was the slant, caught in stride and taken the distance.
  • Roger Craig, a second-round pick in 1983, a running back who was also one of the best receivers on the team. In 1985 he became the first player to run for 1,000 yards and catch passes for 1,000 more in the same season, and he led the league with 92 receptions.11 A running back leading the NFL in catches is the West Coast offense’s fingerprint in a single statistic. Many of those catches came on the back’s option route: he runs straight at the linebacker covering him, reads which way the linebacker leans, and breaks the other way, in or out. The linebacker has to commit before the back does, and against a back as quick as Craig a wrong guess is a catch in open field.
  • Dwight Clark and Freddie Solomon, who did most of the catching in the early 1980s, and later John Taylor, Rice’s partner on the other side.

The yards after the catch also mean the offense isn’t at the mercy of how far the quarterback can throw. Montana, a third-round pick, wasn’t known for arm strength; nor, before him, was Virgil Carter. What both could do was put the ball in the right place at the right time, which leads to the third design goal. (YAC as a statistic, and the modern version that measures it against what a typical receiver would have gained, YAC over expected, belong to the Shanahan case study, which is where the 49ers’ current offense picks this thread back up.)

Design goal 3: ball placement

If the receiver has to keep running, where the ball arrives matters as much as whether it arrives. A pass thrown to the receiver’s back hip makes him stop, turn and reach behind him, and by the time he’s turned upfield again the defender is on him. The same pass thrown a foot in front of him, at chest height on his upfield shoulder, lets him catch it without breaking stride. Walsh coached quarterbacks to throw to a spot, placed for the run after the catch and away from the nearest defender, and receivers to finish their routes so the spot would be right. The difference between A and B is perhaps a yard and a quarter of a second, and it is the difference between the play Walsh designed and a two-yard gain. That’s why the West Coast offense has a reputation for chewing up quarterbacks who aren’t accurate: the plays are easy to draw and hard to execute, and everything depends on a throw that has to be right every time. It’s also why its coaches have so often preferred an accurate quarterback with an average arm to the reverse. The cost of that preference shows up against defenses that crowd the short area and dare the quarterback to beat them deep.

NoteMadden vs. real life

In a video game, a pass is complete or it isn’t; the receiver catches it and runs. In real life, a “completion” covers everything from a catch in stride with open grass ahead to a lunge at a ball thrown behind a receiver who is tackled where he lands. Box scores count them the same, and coaches don’t. When you watch replays, look past caught or not to where on the receiver’s body the ball arrived, and whether he had to slow down to catch it.

Scripting and practice: taking the guesswork out

Walsh is remembered for one planning habit above all: before every game he scripted his first fifteen or so offensive plays (sometimes as many as 25) and called them in order.12 How a modern staff builds and uses an opening script, and whether scripted plays actually gain more, is the subject of The Game Plan; here the question is why it mattered so much to this offense in particular.

A timing offense can’t improvise its way out of trouble. A run-first team that has a bad week of practice can still line up and run its best play harder. A West Coast team whose quarterback and receivers aren’t in sync throws incompletions and interceptions. So Walsh treated the week as rehearsal for decisions, not just for plays. The script moved the play-calling decisions from the noise of the first quarter to the calm of the week, and it let the players rehearse the exact plays they’d run first, in order, until they ran them without thinking. It also turned the opening drives into a set of questions (how does this defense line up against split backs? what does it do when the flanker goes in motion?) whose answers set up the rest of the game.

The practices were built the same way. A 1990 review of Walsh’s book on building the 49ers describes a plan in which he counted out the days of training camp and the season in advance, used four practice formats (from a hard two-and-a-half-hour session to a light half-hour the day before a game), and never scheduled two hard sessions in a row. Individual technique was drilled to the step, down to how many steps the quarterback takes and when he looks for each receiver. Plays were repeated until they were automatic, and answers to special situations (“third down and four at the goal line”) were prepared in advance rather than invented during a game. His stated goal was that players “take the field to learn something each session,” not just to have their toughness tested. He also coached the coaches, which is the beginning of the tree at the end of this chapter.13

There’s a direct line from that practice field to the play this chapter opened with. The 49ers didn’t draw up Sprint Right Option in the dirt with fifty-eight seconds left. They had run it, including its second option, all season.

ImportantWhy it exists

Scripting and rehearsal exist because a precise offense is only as good as its worst repetition. Walsh’s plays left no slack in the timing, so he took the slack out of everything else: the order of the calls, the week’s schedule, the footwork. The trade-off is flexibility. A script is a plan made before the defense has shown its hand, and a coach who follows it too faithfully can call a play the defense is already set to stop, which is why every script has exits for third down, the red zone and bad field position.

Signature play: The Catch (1981 NFC Championship)

January 10, 1982, Candlestick Park. The winner would go to Super Bowl XVI. The Dallas Cowboys had been to five Super Bowls in the previous eleven seasons; the 49ers hadn’t been to one. Dallas cornerback Everson Walls had intercepted Montana twice, and his fumble recovery late in the game set up the touchdown that put Dallas ahead 27–21.

San Francisco got the ball at its own 11 with 4:54 left and all three timeouts. What happened next turned this chapter’s idea inside out. All day the 49ers had used the pass where a defense expected a run; now, with Dallas expecting passes, they used the run. The Cowboys played pass defense, and Walsh ran the ball: Lenvil Elliott, a backup running back, carried on several plays, and Freddie Solomon gained 14 yards on a reverse. In between came short throws to Solomon, Clark and the fullback Earl Cooper. A team famous for passing drove into field-goal range mostly by doing what the defense didn’t expect. By the time they reached the Dallas 6 on third-and-3, with fifty-eight seconds left, the 49ers had used two timeouts and had one left.

Why third-and-3, not third-and-goal? Because the line to gain was short of the goal line. A new set of downs gives the offense ten yards to gain, so a first down earned outside the 10 sets the line to gain in front of the goal line; on this series it was at the Dallas 3. The 49ers needed three yards for a first down, or six for the game. (Goal-to-go starts only when the line to gain would be at or beyond the goal line; see Downs, Distance, and Field Position.)

The call was Sprint Right Option; the full call in the 49ers’ playbook is “Change Left Slot – Sprint Right Option.”14 Read it the way Offensive Systems taught you to read a West Coast call: the first half is the formation, the second is the play. “Left Slot” names a formation with two receivers on one side, one of them in the slot; “Sprint Right” is the quarterback’s launch point, with the line and backs protecting for a quarterback running to his right; “Option” names the pass pattern. We couldn’t find “Change” defined in a source we could check. Solomon actually lined up in the right slot, so it may have flipped the formation, but treat that as a guess. The 49ers lined up in split backs with Solomon in the slot on the right and Clark wide outside him. The design:

  1. Montana sprints right. A sprint-out moves the whole passing game toward one sideline. The quarterback gets away from the middle of the rush and runs toward his receivers, and the throws get shorter and simpler, because he only has to read half the field.
  2. Solomon is the first option, on a short out toward the right corner of the end zone. The 49ers had scored on this exact play, to Solomon, earlier in the game.
  3. Clark is the second option. He releases inside, across the path of Solomon’s defender, a rub that makes it harder to stay with Solomon, then runs across the end zone to the left, stops, and comes back to the right along the end line, where the quarterback, still moving right, will find him.

Montana is out of room, a few yards from the sideline with three defenders closing. A quarterback without a plan has two bad choices. He can take the sack, which loses yards and a down. Was it a throwaway that happened to be caught? People have argued about it ever since, and Montana’s own description of the throw has varied over the years. Clark’s answer never did: the route back along the end line was a practised backup, and he knew where the ball would come. Both things can be true. The throw was a quarterback under pressure putting the ball where nothing bad could happen; the route that put Clark under it was in the playbook and on the practice schedule. That is the practice-field argument of the last section in one snap.

ImportantWhy the sprint-out exists

The sprint-out (the quarterback running toward one sideline at the snap, with the routes flowing the same way) trades half the field for simplicity and time. The quarterback reads two or three receivers on one side instead of five across the field; he throws on the run, closer to his receivers; and he moves away from the rushers who started in the middle. Near the goal line, where the field is short and the end zone has edges a ball can be thrown over, those trades are worth more. The costs are real too: the whole back side of the field is gone, the throws are harder to make on the move, and a defense that knows the sprint is coming can chase it, as Dallas did. The modern cousins of the sprint-out are the bootleg and the moving-pocket throws you’ll meet in Play-Action, Boots, and Screens.

Now read down. The best bar on the left belongs to throws of six to ten yards, which succeeded 63% of the time, and the 11-to-15-yard throws aren’t far behind. That band is where Walsh’s five-step game lived: the digs, crossers and curls you saw in Figure 59.3, thrown on time to a receiver on the move. The deep throws are worth the most per play but succeed less than half the time and carry most of the interceptions. And the last bar is the honest reminder that “passing” includes sacks and scrambles, which bring the average for every dropback back down: a short throw isn’t better than passing, it’s better than running, which is the claim Walsh made.

That raises the analyst’s obvious question. If the deep bars are worth the most per play, was Walsh’s short-first offense wrong? Three things to weigh. First, the bars are averages over plays that weren’t assigned at random: a quarterback throws deep when the coverage allows it, often because the short throws have pulled defenders forward, so the deep bar takes credit for a look the short game helped create. This chart shows association, not a controlled test. Second, the deep throw’s average hides its spread: it fails more than half the time, and an offense that lives on it lives with incompletions, sacks while waiting for routes to develop, and interceptions. Third, Walsh’s real competitor was the run, and modern data agrees with him there emphatically. Where today’s analytics would push him is on the mix. Even with sacks and scrambles counted, an early-down dropback averaged +0.08 EPA against -0.07 for a designed run, so a modern Walsh would probably throw more often on first down, and more of those throws in the six-to-fifteen-yard band his five-step game was built for. Whether his short game was optimal or just optimal for a team with a weak line in 1979 is a fair argument; it was certainly better than what it replaced.

The league has absorbed the lesson so completely that it no longer looks like a system. In these three seasons 55% of early-down throws travelled five yards or less in the air, and 26% were thrown to a receiver at or behind the line. Every team in the league now runs some of Walsh’s offense, which is the subject of the coaching tree two sections down.

CautionCommon misconception

“Short passes are just safe passes.” The chart says something stronger and something weaker. Stronger: on early downs, short passes beat runs on consistency, which is what an offense needs to stay ahead of the chains. Weaker: a short pass is not automatically a good play. Its value comes from the defense leaving room underneath, and a defense that sits on short throws can make them fail. And on third-and-8 a five-yard catch is a punt, however efficient it looks on first down; there the deeper throw is the right one (Third Down).

How opponents responded

A system that wins three Super Bowls gets studied, and defenses found four answers. Each attacked one of the design goals.

Hit the receivers, and the quarterback. The timing depends on receivers reaching their spots on time, and inside five yards contact is still legal, so the first answer was to jam receivers at the line and knock them off schedule. The physical extreme was Bill Parcells’s New York Giants, who beat the 49ers in the playoffs after the 1985 season (17–3) and the 1986 season (49–3), the second game remembered for the hit by nose tackle Jim Burt that knocked Montana out of it.22 One story credits Parcells with the name “West Coast offense” itself, as a jab after the first of those wins; another traces it to a 1993 remark by quarterback Bernie Kosar, publicized in Sports Illustrated, in which the label meant the Coryell family of offenses from San Diego and was then attached to Walsh’s by mistake. Walsh disliked it either way.23

Rush with safe pressure. A quick-throw offense beats an ordinary blitz by getting the ball out before it arrives, so defenses invented pressure that didn’t leave a hole behind it. The Cincinnati Bengals, Walsh’s Super Bowl opponent twice, made Dick LeBeau their defensive coordinator in 1984 and became the home of the zone blitz, which drops a lineman into the throwing lane of a hot route while a linebacker rushes, and which the NFL’s own history describes as built to stop the West Coast timing game; the story is in Zone Blitzes.24

Keep everything in front, and tackle. If the offense lives on yards after the catch, take away the “after.” The Tampa 2 of the late 1990s and 2000s was built on fast linebackers and corners who sat short, let the short throw be completed, and swarmed the catcher. Its best-known coach, Tony Dungy, had been Dennis Green’s defensive coordinator in Minnesota, and Green was a Walsh assistant: the tree was learning to stop itself (Cover 2 and Tampa 2).

Learn the language. The West Coast call is long and explicit, and once a few dozen coaches had learned it, they all spoke it. The best-known payoff came in Super Bowl XXXVII, after the 2002 season, when Jon Gruden’s Tampa Bay Buccaneers beat the Oakland Raiders, the team he had coached the year before, 48–21. Rich Gannon threw five interceptions, and Raiders players said afterwards that Oakland had kept much of Gruden’s terminology and that Tampa’s defenders seemed to know what was coming. (Derrick Brooks remembers it differently: Gruden played Gannon himself in practice, and “we didn’t know what play they was running, but we knew their personalities.”)25

Today’s version of the answer is the two-high safety shell of the 2020s, which takes away the deep throw first and rallies to everything short. The West Coast descendants answered that in turn, with under-center play-action and the outside-zone run game, which is where the Shanahan case study begins.

The coaching tree

A coaching tree is the family of coaches descended from one head coach, traced through who worked on whose staff. A branch runs from a head coach or coordinator to an assistant who later ran a staff of his own; because most coaches work for several bosses, one coach can sit on several branches. Walsh’s is usually called the largest in football. Figure 69.7 draws the branches that run through his offensive assistants, and the two boxes underneath list the head coaches who came out of it in the 2025 and 2026 seasons.

Why did one staff produce so many head coaches? Partly luck and winning, which gets assistants hired. But Walsh also built a system that could be carried away. Everything was written down: the 1982 playbook ran past 300 pages (the call for The Catch is on page 325), and he later co-wrote a coaching manual, Finding the Winning Edge, on how to run a staff, a practice week and a passing game.26 The plays came with their reasons attached, as this chapter has shown: why this drop goes with that route, why the back checks before he releases. And everything was scripted and rehearsed, so his assistants spent their weeks planning and teaching the system, not just carrying it out; in the words of the 1990 review quoted earlier, he “coached the coaches.” An assistant who leaves with a written, explained, rehearsed system can install it somewhere else. One who leaves with a feel for his boss’s play calls can’t.

CautionCommon misconception

“Same tree, same offense.” A coaching tree records where coaches learned, not what they run. Reid’s 2025 Chiefs, Shanahan’s 49ers and Harbaugh’s Ravens share a common ancestor and almost nothing on the field: shotgun spread and RPOs in Kansas City, under-center wide zone in San Francisco, and a quarterback-run offense in Baltimore (and Harbaugh coached special teams on Reid’s staff, not offense).33 The data agrees. In 2025 the ten tree teams threw 56% of their early-down passes five yards or less in the air, the other 22 teams 55%, which is no real difference, and Liam Coen’s Jaguars, a McVay-branch team, ranked 30th of 32 in short early-down throws. What travels down a tree is a way of working (the explicit language, the scripting, the attention to detail in practice) more than a playbook. Trees are also generous: almost every coach worked for several bosses, and the drawer chooses the line.

There’s one more branch the figure can’t draw. In the mid-1980s (1987, by the league’s account) Walsh began bringing minority coaches into the 49ers’ training camp, to give them NFL experience and a reference from an NFL staff at a time when the profession was almost entirely white. The league later adopted the program for every team; it now carries his name as the Bill Walsh NFL Diversity Coaching Fellowship. Dennis Green and Ray Rhodes, two of the first Black head coaches of the modern era, got their NFL starts on Walsh’s staffs. Herm Edwards was the first graduate of the program to become an NFL head coach, and Marvin Lewis, Lovie Smith, Mike Tomlin and Raheem Morris followed.34 That might be the most important tree he planted.

What survives

Forty-odd years later, almost nothing about the 1981 49ers looks modern: the split backs, the quarterback under center on nearly every snap, the seven-step drop. And almost everything they did is still here.

  • The short pass as a run. Every modern offense has a set of throws that does the running game’s job: the quick game, the screen, and the RPO, which goes one step further and lets the quarterback choose between the handoff and the throw after the snap (RPOs). The fingerprint chart above is the league running Walsh’s offense.
  • Backs as receivers. Roger Craig’s 1,000/1,000 season of 1985 was matched in 1999 by Marshall Faulk and in 2019 by Christian McCaffrey, then with Carolina, who was the 2025 Comeback Player of the Year in Kyle Shanahan’s descendant of the same offense in San Francisco.35
  • Timing and placement. Quarterbacks are still taught to throw on rhythm, with their feet as the clock, and to put the ball where the receiver can keep running. The Quarterback Play chapter is built on those ideas.
  • The script and the rehearsal. Nearly every NFL play-caller scripts the opening plays, and every staff plans its practice week to the minute. Andy Reid’s index card of first-and-second-down calls, in The Game Plan, is Walsh’s script, one branch down.
  • The language. The long, explicit West Coast call lives on in the Shanahan, McVay and Reid families. Jon Gruden’s favourite, “Spider 2 Y Banana,” reads like Sprint Right Option: a formation-and-protection piece (“Spider 2”), then one receiver and his route (the tight end, Y, runs the route the playbook calls “banana”), with the rest, including the fullback in the flat, left to the base pattern. Many staffs now shorten calls with wristbands and one-word names (Offensive Systems).

What faded is mostly the look. The two-back pro set gave way to one back and three receivers, and then to the shotgun; the fullback nearly disappeared (Shanahan’s 49ers are an exception). But the look is coming partly back: in 2025, offenses lined up with fewer than three wide receivers on 41.7% of plays, the most since the NFL’s tracking data began in 2016, and the quarterback was under center on about a third of snaps, up from about 27% in 2023.36 The descendants of Walsh’s tree are leading that return, for reasons the Shanahan case study explains.

Watch for it on Sunday

TipWatch for it: a West Coast census

Pick a team from the tree (the 49ers, Rams, Packers, Chiefs, Vikings, Bengals, Jaguars or any of the others in the 2026 box) and chart its first-half first and second downs. For every pass, note two things:

  • Where was the ball caught, or where would it have been? Behind the line, one to five yards past it, or deeper? Count the first two together as short horizontal throws.
  • Was the receiver running when he caught it? If yes, how far did he go after the catch?

Then do the arithmetic. Across the league in 2025, 56% of early-down throws in competitive games travelled five yards or less in the air; teams ranged from 46% (CHI) to 64% (PIT). A team well above the league figure is leaning on the horizontal game; a team well below it is throwing downfield on early downs and running to set up its short game.

Don’t expect a tree team to land high just because it’s a tree team. In 2025 the tree’s ten teams averaged almost exactly what everyone else did (see the coaching-tree section): the short game is now the whole league’s, and what your count tells you is how this staff uses it this week.

Post-snap, on the replay: pick one short completion and watch the receiver’s hands. Did the ball arrive in front of him, at chest height, so he never slowed down (A in Figure 59.4), or behind him? And look for the backs: count how many times a running back releases into a route after checking for a blitzer. That’s the 1984 49ers, forty years on.

You’re Walsh, and the defense is built to break your timing: press the wide receivers so they can’t reach their spots on schedule, and hit everybody. Your receivers will be jammed at the line, and the pressure will come fast. You have two ideas: a crossing route by X, to run him away from his press corner, or a route for one of the backs. Which matchup is better, and what route do you give him?

TipAnswer

The backs, on option routes. Nobody can press a running back, because he starts five yards behind the line. And the men assigned to them are linebackers, who are bigger and slower than a back like Roger Craig. So give the back an option route: he runs straight at his linebacker, and breaks in or out, whichever way the linebacker doesn’t lean. (A flat or swing that makes the linebacker run to the sideline, or an angle route that starts toward the flat and snaps back inside, uses the same edge.)

The crossing route is the weaker idea here, and the defense drew it that way. Crossers are a good answer to man coverage in general, but X has to beat a press corner first, which costs time, and then run straight through the middle, where the Mike is sitting as a rat waiting for exactly that. The back’s route stays outside the rat’s area.

The trade-off is protection. If the backs run routes, they aren’t blocking, and a defense playing press man often blitzes. That’s what the check-release is for: each back looks for his blitzer first and releases only if nobody comes. Against a blitz the quarterback has to throw before the unblocked rusher arrives, which is why the quick throw to the back, or the hot route, is the first thing he looks at.

The same play as The Catch, on a different day, with one more thing gone wrong. The out to Z was jumped, and the free safety has read the play and parked himself at the back of the end zone, exactly where X is heading. It’s third-and-goal from the 6, 45 seconds left, one timeout, and you’re down by 6. Where should the ball go, and what’s the worst thing the quarterback can do?

TipAnswer

Nowhere in the end zone: throw it away, out of bounds. The second option worked in The Catch because only one defender was near Clark, and the throw could go high, where only the receiver could reach it. Here a second defender is already standing on the spot, facing the quarterback, so a high throw to X is a jump ball with two defenders under it, or an interception. Both designed answers are covered, so the right play is the one the design keeps in reserve: throw the ball over the back corner, or over the sideline, where no one can catch it.

That isn’t giving up. An incompletion stops the clock and brings up fourth-and-goal from the 6 with about 40 seconds and a timeout left: one more full play, with a new call. The worst outcomes are the two that take that play away. An interception ends the game. A sack is nearly as bad: it loses yards, makes fourth down longer, and keeps the clock running, so the offense has to burn its last timeout just to get one more snap. This is the other half of the lesson of The Catch. Montana’s throw was placed so that it could only be a touchdown or an incompletion; when the touchdown isn’t there, take the incompletion.

Takeaways

  • Walsh’s problem was a team that couldn’t yet run the ball, and his answer was to make the pass do the run’s job. The horizontal passing game uses short, timed throws to the edges and across the field, caught by receivers who are already running, so the ball beats the defense’s pursuit and the tackle is made in space by a smaller man. Once it worked, the run came back: by 1981 the 49ers ran as often as the league did.
  • Timing, yards after the catch and ball placement were design goals, not happy accidents. Drops matched route depths and the quarterback’s feet ran his progression; routes caught receivers moving away from defenders; and the ball was thrown to the spot that let them keep running. The bill was an offense that needed an accurate quarterback and perfect rehearsal.
  • Scripting and Walsh’s practice methods were the system’s insurance. A timing offense can’t improvise, so he planned the week, the calls and the footwork in advance, and rehearsed even the backup options. The Catch was Sprint Right Option’s second option, practised all season.
  • The data still backs him. On early downs from 2023 to 2025, a pass thrown one to five yards past the line succeeded more often than a designed run, and a screen did about the same job as a run. The deeper throws are worth more per play, which says a modern Walsh would throw more, not less.
  • Defenses answered with press and physical play, the zone blitz, rally-and-tackle zones like Tampa 2, and by learning the language, and today’s two-high shells are the latest round.
  • The coaching tree is usually called the largest in football: ten of the 32 head coaches in 2025 and twelve in 2026 came from its branches. It grew because Walsh’s system was written down, explained and rehearsed, so it could be carried away. What travels down it is a way of working more than a playbook: in 2025 the tree’s teams threw short no more often than anyone else.
