def O(key, pos, d, w, label=None):
    if label is None and pos in ("WR", "TE"):
        label = key            # receivers and tight ends wear their letters (X, Y, Z, H, U)
    return Player(key, pos, "off", d, w, label=label)


def D(key, pos, d, w, label=None):
    return Player(key, pos, "def", d, w, label=label)


def line5(d=-0.7):
    """The five offensive linemen (only the center labelled)."""
    return [O(k, k, d, w, label="C" if k == "C" else "") for k, w in
            (("LT", -3.2), ("LG", -1.6), ("C", 0.0), ("RG", 1.6), ("RT", 3.2))]


def eleven_attached(rb_w=1.6):
    """11 personnel from the shotgun, the tight end (Y) attached on the right: Z wide right off the
    ball, X wide left on the ball, H in the left slot off the ball. Seven on the line: five
    linemen, Y and X, so Y and X are the ends and everyone else is a back."""
    return line5() + [O("Y", "TE", -0.7, 4.8), O("Z", "WR", -1.5, 14.0), O("X", "WR", -0.7, -14.0),
                      O("H", "WR", -1.5, -8.5), O("QB", "QB", -5.0, 0.0, "Q"), O("RB", "RB", -5.0, rb_w, "R")]


def eleven_2x2():
    """11 personnel 2x2 from the shotgun: X and H left, the tight end flexed as the right slot (Y)
    and Z on the ball outside him. Seven on the line: five linemen, X and Z."""
    return line5() + [O("X", "WR", -0.7, -15.0), O("H", "WR", -1.5, -9.0), O("Y", "TE", -1.5, 8.5),
                      O("Z", "WR", -0.7, 15.0), O("QB", "QB", -5.0, 0.0, "Q"), O("RB", "RB", -5.0, 1.5, "R")]


def twelve_under_center():
    """12 personnel under center: two tight ends attached (U left, Y right), X and Z off the ball."""
    return line5() + [O("Y", "TE", -0.7, 4.8), O("U", "TE", -0.7, -4.8, "U"), O("X", "WR", -1.5, -14.0),
                      O("Z", "WR", -1.5, 14.0), O("QB", "QB", -1.9, 0.0, "Q"), O("RB", "RB", -7.0, 0.0, "R")]


def tite_quarters():
    """Fangio-style nickel tite against 11 personnel with the tight end on the right: a nose (0) and
    two 4i-techniques, a stand-up edge (E) outside each tackle (the right one outside the tight end),
    one linebacker (M), the nickel ($) inside the left slot, corners 7 off, safeties 12.5 deep."""
    return [D("LE", "OLB", 1.0, -5.3, "E"), D("L4", "DL", 1.0, T("4i", -1), "4i"), D("NT", "DL", 1.0, 0.0, "0"),
            D("R4", "DL", 1.0, T("4i", 1), "4i"), D("RE", "OLB", 1.0, T("9"), "E"),
            D("MIKE", "LB", 4.7, 0.8, "M"), D("NB", "NB", 5.0, -8.0, "$"),
            D("LCB", "CB", 7.0, -14.6, "CB"), D("RCB", "CB", 7.0, 14.6, "CB"),
            D("FS", "FS", 12.5, -7.0, "FS"), D("SS", "SS", 12.5, 7.0, "SS")]


def go_to(p, key, to, kind="path", **kw):
    """Post-snap movement to an absolute (d, w) spot."""
    d0, w0 = p._cur(key)
    rel = [(to[0] - d0, to[1] - w0)]
    if kind == "block":
        return p.block(key, pts=rel, **kw)
    if kind == "rush":
        return p.rush(key, pts=rel, **kw)
    return {"path": p.path, "run": p.run}[kind](key, rel, **kw)


# ---------------------------------------------------------------- drawing helpers
def no_numbers(ax):
    """Remove the painted yard numbers (in zoomed windows they sit where players stand)."""
    for t in list(ax.texts):
        if t.get_alpha() == 0.55:
            t.remove()


def P_of(fld, p):
    """(d, w) in the play's frame -> plot coordinates on this field."""
    return lambda d, w: tuple(float(v) for v in fld.xy(*p.to_abs(d, w)))


def say(ax, P, d, w, text, color=style.INK, fs=6.8, ha="center", ec=None, weight="normal"):
    x, y = P(d, w)
    ax.text(x, y, text, fontsize=fs, color=color, ha=ha, va="center", zorder=12, linespacing=1.12,
            fontweight=weight,
            bbox=dict(boxstyle="round,pad=0.22", fc=style.SURFACE, ec=ec or "none", lw=0.6, alpha=0.94))


def point_to(ax, P, at, to, text, fs=6.8, ha="center"):
    """A label at `at` with a thin leader arrow to `to` (both (d, w) yards)."""
    ax.annotate(text, xy=P(*to), xytext=P(*at), fontsize=fs, color=style.INK, ha=ha, va="center", zorder=12,
                linespacing=1.12, bbox=dict(boxstyle="round,pad=0.25", fc=style.SURFACE, ec=style.BASELINE,
                                            lw=0.6, alpha=0.96),
                arrowprops=dict(arrowstyle="-|>", color=style.INK_2, lw=0.8, shrinkA=2, shrinkB=6))


def ruler(fld, play, fs=6.0, top=30):
    """Depth labels (yards past the line of scrimmage) down the left edge of the window."""
    ax = fld.ax
    x_left = FIELD_W - fld.y1 + 0.35
    for d in range(5, top + 1, 5):
        y = play.los + d
        if fld.x0 + 0.8 <= y <= fld.x1 - 0.8:
            ax.text(x_left, y, f"{d}" + (" yd" if d == 5 else ""), fontsize=fs, color=style.INK_2,
                    ha="left", va="center", zorder=12,
                    bbox=dict(boxstyle="round,pad=0.12", fc=style.SURFACE, ec="none", alpha=0.85))


def box_outline(ax, P, d0, d1, w0, w1, text, text_at, color=style.INK_2):
    """A dashed rectangle around the box, with a caption."""
    (xa, ya), (xb, yb) = P(d0, w0), P(d1, w1)
    ax.add_patch(Rectangle((min(xa, xb), min(ya, yb)), abs(xb - xa), abs(yb - ya), fill=False,
                           ec=color, lw=1.0, ls=(0, (4, 3)), zorder=2))
    say(ax, P, text_at[0], text_at[1], text, fs=6.8, weight="bold")


def ghost_marker(ax, x, y, label, ppy, side="def"):
    """A hollow, dashed marker: where a player stood before he moved."""
    col = style.DEFENSE if side == "def" else style.OFFENSE
    if side == "def":
        pat = RegularPolygon((x, y), 4, radius=MARKER_R * 1.25, orientation=np.pi / 4, facecolor="none",
                             edgecolor=col, lw=1.0, ls=(0, (2, 1.2)), zorder=6)
    else:
        pat = Circle((x, y), MARKER_R, facecolor="none", edgecolor=col, lw=1.0, ls=(0, (2, 1.2)), zorder=6)
    ax.add_patch(pat)
    if label:
        ax.text(x, y, label, ha="center", va="center", color=col, fontweight="bold", zorder=6,
                fontsize=max(4.0, min(8.0, ppy * 0.72)))


# ---------------------------------------------------------------- animations and their print strips
def snapshot(play, t, ax=None, lateral=(-24, 24), window=(-8, 20), ghosts=(), ghost_t=None, trail_from=None):
    """Everyone where he is at time t, a faint trail behind each mover, and a hollow 'ghost'
    where each listed player stood at `ghost_t`."""
    pos = {k: play.position(k, t) for k in play.players}
    moved = [pl.moved(d=float(pos[k][0]), w=float(pos[k][1])) for k, pl in play.players.items()]
    q = Play([m for m in moved if m.side == "off"], [m for m in moved if m.side == "def"],
             los=play.los, ball_y=play.ball_y, to_go=play.to_go)
    q.highlights = dict(play.highlights)
    fld = q.draw(ax=ax, lateral=lateral, window=window, title=False)
    ax, ppy = fld.ax, fld.pts_per_yard()
    no_numbers(ax)
    P = P_of(fld, play)
    start = play._compile()[1] if trail_from is None else trail_from
    for k, pl in play.players.items():
        if pl.pos in ("LT", "LG", "C", "RG", "RT"):
            continue
        pts = np.array([play.position(k, tt) for tt in np.linspace(start, t, 50)])
        if np.ptp(pts[:, 0]) + np.ptp(pts[:, 1]) < 0.4:
            continue
        xs, ys = fld.xy(*play.to_abs(pts[:, 0], pts[:, 1]))
        col = style.OFFENSE if pl.side == "off" else style.DEFENSE
        ax.plot(xs, ys, color=col, lw=1.1, alpha=0.55, ls="-" if pl.side == "off" else (0, (3, 2)), zorder=3)
    if ghost_t is not None and t > ghost_t + 0.05:
        for k in ghosts:
            d0, w0 = play.position(k, ghost_t)
            d1, w1 = play.position(k, t)
            if abs(d1 - d0) + abs(w1 - w0) > 1.0:
                ghost_marker(ax, *P(d0, w0), play.players[k].label, ppy, play.players[k].side)
    bd, bw = play.ball_position(t)
    bx, by = P(bd, bw)
    for kind, _, t_rel in play.ball_events:
        if kind == "pass" and t > t_rel:
            sx, sy = P(*play.ball_position(t_rel))
            ax.add_patch(FancyArrowPatch((sx, sy), (bx, by), arrowstyle="-|>", mutation_scale=8, color=style.BALL,
                                         lw=1.3, ls=(0, (2, 1.5)), zorder=9))
    ax.add_patch(Ellipse((bx, by), 0.75, 1.1, facecolor=style.BALL, edgecolor="white", lw=0.6, zorder=9.5))
    for art in list(ax.texts) + list(ax.patches) + list(ax.lines):
        art.set_clip_on(True)
    return fld


def strip(play, times, labels, ncols=2, width_in=6.5, legend=True, annot=None, **kw):
    """The print version of an animation: one panel per chosen moment, each with a caption.
    `annot` maps a panel index to a function f(ax, P) that adds print-only labels to that panel."""
    lateral, window = kw.get("lateral", (-24, 24)), kw.get("window", (-8, 20))
    n = len(times)
    nrows = int(np.ceil(n / ncols))
    pw = width_in / ncols
    h = nrows * (pw * (window[1] - window[0]) / abs(lateral[1] - lateral[0]) + 0.42) + (0.35 if legend else 0.1)
    fig, axes = plt.subplots(nrows, ncols, figsize=(width_in, h), squeeze=False)
    for ax in axes.flat[n:]:
        ax.axis("off")
    for i, (ax, t, lab) in enumerate(zip(axes.flat, times, labels)):
        fld = snapshot(play, t, ax=ax, **kw)
        if annot and i in annot:
            annot[i](ax, P_of(fld, play))
        ax.set_title(lab, fontsize=7.6, color=style.INK, loc="left", pad=3, linespacing=1.15)
    fig.tight_layout(h_pad=0.6, w_pad=0.5, rect=(0, 0.05 if legend else 0, 1, 1))
    if legend:
        hs = [Line2D([], [], color=style.OFFENSE, lw=1.2, alpha=0.7, label="Offense: path so far"),
              Line2D([], [], color=style.DEFENSE, lw=1.2, ls=(0, (3, 2)), label="Defense: path so far"),
              Patch(facecolor="none", edgecolor=style.DEFENSE, lw=1.0, ls=(0, (2, 1.2)),
                    label="Dashed outline: where he started")]
        fig.legend(handles=hs, loc="lower center", ncol=3, fontsize=7.0, frameon=False, bbox_to_anchor=(0.5, 0.0))
    return fig


def show_play(play, times, labels, web_caption, annot=None, **kw):
    """Web: an animation with a caption that changes with the phase of the play.
    Print: a labelled frame strip (panel titles = `labels`)."""
    if PDF:
        strip(play, times, labels, annot=annot, **kw)
        plt.show()
        return
    df = play.to_tracking()
    tmap = df.groupby("frameId")["time"].first()
    anim = animate(df, to_go=play.to_go)
    ax = anim._fig.axes[0]
    cap = ax.text(0.01, 0.02, "", transform=ax.transAxes, ha="left", va="bottom", fontsize=9, color=style.INK,
                  zorder=21, bbox=dict(boxstyle="round,pad=0.3", fc=style.SURFACE, ec=style.THIRD, lw=1.0))
    base = anim._func

    def upd(fid, *a):
        out = base(fid, *a)
        cap.set_text(web_caption(float(tmap.get(fid, 0.0))))
        return out
    anim._func = upd
    if shutil.which("ffmpeg"):
        html = anim.to_html5_video(embed_limit=40).replace("<video ", '<video style="max-width:100%;height:auto" ')
    else:
        html = anim.to_jshtml(default_mode="once")
    plt.close(anim._fig)
    display(HTML(html))


def phases(cuts):
    """[(t_upper, caption), ...] -> function of t (seconds from the snap) for show_play."""
    def f(t):
        for up, txt in cuts:
            if t < up:
                return txt
        return cuts[-1][1]
    return f

This case study ties together three earlier chapters and leans on several more. One line each:

  • Simulated pressure (Zone Blitzes, Fire Zones, and Simulated Pressure): a simulated pressure rushes only four but makes the offense think more are coming, often by dropping a lineman (a dropper) and rushing a linebacker or defensive back in his place; a pressure look shows more possible rushers than will come, for example a double A-gap mug with linebackers standing in the gaps beside the center.

The problem: one deep safety was a promise the offense could cash

To see why two coaches built what they built, start with what NFL defenses looked like in 2017.

The dominant defense of the 2010s was Seattle’s Cover 3: one safety deep in the middle, the other walked down as an eighth man near the line. It was simple, fast and physical, and for a while it was the best defense in football. But it made a promise before every snap. With one safety deep, the defense had already committed its eighth defender to the run and told the quarterback where the deep help was: in the middle, and only there.

Two kinds of offense learned to cash that promise. Sean McVay’s and Kyle Shanahan’s play-action offenses pulled the eighth man forward with a run fake and threw into the space he left, and Patrick Mahomes punished the other half: with one deep safety, any receiver outside that safety’s reach was one-on-one deep. (The Modern Era tells both stories in full.)

The defensive problem had a precise shape. A one-high defense decides before the snap who will stop the run and where the deep help is, and modern offenses had learned to make that decision cost something on every play. The fix had to do three things at once:

  1. take away the explosive pass, so a quarterback like Mahomes had to be patient;
  2. stop the run without committing an eighth man before the snap, so the play-action fake would hit nothing;
  3. give the quarterback no reliable pre-snap information, so the offense’s beautifully designed answers couldn’t be called in advance.

Two safeties back solves the first. The other two are hard. Vic Fangio spent most of a long career working out how to do them.

Vic Fangio: four decades in the room

Victor John Fangio was born in 1958 in Dunmore, Pennsylvania, a small town next to Scranton. He started coaching at his old high school in 1979, worked as a graduate assistant at North Carolina, and got his first professional job in 1984 as a defensive assistant for the Philadelphia Stars of the USFL. In 1986 he followed the Stars’ head coach, Jim Mora, to the NFL’s New Orleans Saints as linebackers coach, where he coached one of the most famous linebacker groups ever assembled (Rickey Jackson, Sam Mills, Pat Swilling and Vaughan Johnson, the “Dome Patrol”). From there he spent more than twenty years as a coordinator nobody outside the profession knew: the expansion Carolina Panthers (1995–1998), the Colts, the expansion Texans, a stint as an assistant in Baltimore, a year at Stanford under Jim Harbaugh, the 49ers (2011–2014) and the Bears (2015–2018). He was 60 when Denver made him a head coach for the first time, in 2019.2

Those who have worked for him describe a coach of the old school in most things: long meetings, padded practices, clips of hard hits in the film session. When he took over the Eagles’ defense in 2024 he told the players, in effect, that it wasn’t their fault they didn’t know how to work, “But that’s going to change now that I’m here.” At his introductory press conference he put it more broadly: “People are not expecting as much out of players as we used to expect.”3 He is also, by every account, a relentless student of the opponent’s habits, the coach who notices that a tight end’s alignment gives away the run that is coming.

What makes his career unusual is that his defense got more modern as he got older. The ideas that made him famous were not new, and he didn’t claim to have invented them: two-deep coverages and quarters had been staples in college (Michigan State, TCU and Nick Saban’s teams; see Quarters and the Two-High Match Family), and odd fronts are as old as the 3-4. What he did was assemble them into one system that answered the specific problem of the late 2010s NFL, and teach it to a generation of assistants.

Decision 1: keep two safeties deep, and show it every time

Two safeties at 12 to 14 yards take away the explosive pass. A receiver who runs past a corner still has a safety over the top of him on his side of the field, not one safety trying to cover all of it. That is the reason anyone plays two-high. Fangio’s twist is the every time. His safeties line up at about the same depth and width whatever the call, so the shell that answers “where is the deep help?” gives the same answer every snap. Ed Donatell, who coached with Fangio for eleven seasons, described the goal as “delayed identification”: the quarterback shouldn’t know the coverage until his first or second step of the drop.4

What it costs: two safeties back means two fewer defenders near the line. Someone else has to stop the run. That is decision 2.

Decision 2: accept a light box, and build a front that can hold it

Against 11 personnel (one back, one tight end, three receivers) a two-high nickel defense usually has six defenders in the box against six blockers. The old answer to “six against six” was “bring a safety down.” Fangio’s answer was to make the six good enough. The front he favored on early downs, the tite, puts five men on the line (in Figure 66.1: E, 4i, 0, 4i, E) so that every interior gap has a lineman in it. On paper the offense still has a spare blocker: the guards are uncovered. In practice the nose on the center and the 4is in the B gaps keep both guards busy helping on them, so neither can climb cleanly to the lone linebacker (Odd and Hybrid Fronts draws this against inside zone). The nose must hold both A gaps on his own; the 4is must take on the guards and tackles without being moved. Then the safeties fit the run late, from depth, arriving at five or six yards instead of two.

Against three receivers, Fangio’s own everyday nickel was a close cousin. His Bears and Eagles usually took the nose off and kept two inside linebackers behind four linemen, a 2-4-5: by this book’s count both inside linebackers were on the field for 80% of Chicago’s nickel snaps in 2018 (Danny Trevathan and Roquan Smith) and 84% of Philadelphia’s in 2024 (Zack Baun and Nakobe Dean). The box count is the same six against six. The five-man nickel tite with one linebacker is that idea pushed further, and it is the version Staley and others in the tree made their everyday nickel.5 Both are drawn in Figure 66.3, later in the chapter.

Why it exists: a run that gains five yards is not a disaster; a pass that gains forty is. Fangio’s whole defense prices those two outcomes honestly. He would rather give up 4.5 yards a carry than a deep touchdown, because the long drive gives the offense ten more chances to make a mistake.

What it costs: rushing yards, sometimes a lot of them, and an enormous bill for interior linemen. A light-box defense with ordinary defensive tackles gets pushed around. Every Fangio defense that worked had at least one interior lineman who didn’t need help: Justin Smith in San Francisco, Akiem Hicks and Eddie Goldman in Chicago, Jalen Carter and Jordan Davis in Philadelphia.

Decision 3: rush four, from the same places

With six in the box and two safeties deep, there is nobody left over to blitz without breaking the coverage. So the default is a four-man rush, and it has to win on its own. The 2018 Bears traded for Khalil Mack a week before the season to make sure it would. Blitzing is not banned, but it is rare: by this book’s count from the charting data, Fangio’s 2018 Bears sent five or more on 21% of dropbacks (28th-most in the league), his 2024 Eagles on 19% (28th), and his 2025 Eagles on 19% (31st).6

In the tite, four rush and the fifth man on the line drops: usually the edge on the tight-end side (as in Figure 66.1), sometimes a 4i or the nose on a creeper call that sends a linebacker in his place. That is a small disguise, but a readable one. The same five players stand on the line in the same spots every snap, so after a few series an offense knows who the dropper can be and where the four will come from. Hold on to that: it is the gap Macdonald’s defense fills.

What it costs: if the four can’t get home, a patient quarterback can sit behind good protection and pick at the underneath zones. Fangio’s defense accepts that trade too, on the theory that ten-play drives against a good rush and good tacklers usually stall.

Decision 4: play match quarters and Cover 6, so the safeties can do both jobs

The coverage behind it is the quarters family from Quarters and the Two-High Match Family, and this is where the run problem is really solved. A quarters safety reads the inside receiver on his side (#2). If #2 runs a route, the safety covers him; if #2 blocks, the safety becomes a run defender, coming downhill into the alley outside the edge (he is the alley player). So the “missing” eighth man is not missing. He is standing 12 yards deep, waiting for the offense to tell him whether the play is a run. That is the decisive difference from the old eighth-man defense: the extra run defender decides after the snap, from what the play shows him, so the play-action fake that beat the walked-down safety has a harder time fooling him.

The match in match quarters is pattern matching: each defender starts in a zone, but once the receivers show their routes he takes the one who enters his area and covers him like man coverage. The zone tells the safety where to start; #2’s route tells him whom to take.

Fangio’s family has a split-field version too, under two names. Cover 6 is quarter-quarter-half with the Cover 2 half on the side away from the passing strength (the side with more wide receivers: in Figure 66.1 the left, where the nickel lines up), the usual arrangement. Cover 8 flips it and puts the half toward the passing strength.7 The flip moves the help to where the offense has more receivers. On the two-receiver side the corner squats in the flat (a cloud corner) to sit on the quick outs and hitches that side is built to throw, with a safety taking the whole deep half over him; on the tight-end side the safety keeps his quarters read on Y, where the run strength is. The analyst Cody Alexander calls this targeted coverage. The alternative is a defense that drops each safety into a fixed zone and rotates the same way whatever the offense shows. A targeted defense decides, call by call, which side gets the Cover 2 help and puts it there. (The numbers are dialect: many college staffs, Saban’s among them, use 6 and 7 for different calls; Coverage Fundamentals has the table.)

What it costs: the read safety obeys a rule, and a rule can be used against him. A tight end who blocks for a count and then releases behind the safety who came downhill (play-action), or an RPO that reads the safety directly, punishes the rule. And the run help arrives late, at four or five yards, not at the line.

Decision 5: rotate late, so the shell lies

The last decision is what turns a sound defense into a frustrating one. Broadcasters and film-room coaches call the two-high shell middle of the field open (MOFO) and a one-high shell middle of the field closed (MOFC). From the same open picture, one safety can rotate down at the snap and the other run to the deep middle, closing it: the call has become Cover 3 (or Cover 1). One version Fangio leans on is a weak rotation: the safety on the side away from the passing strength comes down, and the other one runs to the middle. Alexander has written it up as Fangio’s “Cover 9.”8 The quarterback who saw two safeties and threw to the open middle finds a safety sitting in it.

Why it exists: a two-high shell that is always two-high coverage is a puzzle the quarterback solves once. A shell that is two-high or one-high depending on the call is a puzzle he has to solve after the snap, every snap, while the four-man rush is coming. Rotation is what makes decision 1’s “every time” pay off.

What it costs: every rotation is a small play of its own with a way to go wrong. A safety who rotates down has a long way to travel, and a defense that rotates often sometimes busts (a bust is a blown assignment: two defenders take the same receiver, or nobody takes him). And the charting data stops calling it two-high. That point matters for the data later in this chapter, so hold on to it: Fangio’s 2024 Eagles built their calls from a two-high shell, but after the snap they played a two-high coverage on only 34% of dropbacks, 27th in the league (The Modern Era has the full film room). The coverage charts measure what was played, not what was shown.

ImportantWhy it exists: the system in one sentence

Make the offense earn it slowly. Take away the big pass with two deep safeties; stop the run with a front that wins its own fights and safeties who fit late; get pressure without sending more than four; and decide the coverage at the snap, so the offense can’t call its answer in advance. Every part is a trade: rushing yards for explosive plays, a blitz for a sound coverage, pre-snap clarity for post-snap uncertainty. It works only if the front holds and the four can rush.

CautionCommon misconception

“Fangio invented two-high defense.” He didn’t, and he has never claimed to. Quarters was a college staple in the 2000s and Saban’s pattern-match coverages go back to his years with Bill Belichick in Cleveland in the early 1990s (Quarters and the Two-High Match Family).9 Fangio’s contribution was the package: an odd front that let a light box hold, a four-man rush, a coverage family that let the safeties play the run, and a rotation habit that made the shell lie, all taught as one system to assistants who then spread it. Credit for a scheme in football is almost always shared.

Signature seasons: Chicago 2018 and Philadelphia 2024

NoteFilm room: Chicago Bears 2018 — the system arrives

Fangio’s fourth Bears defense was the one the league noticed. Chicago allowed 283 points, the fewest in the NFL; it led the league with 27 interceptions and 36 takeaways; and by this book’s count it allowed the fewest EPA per opponent dropback, while blitzing less than all but four teams. Its runs-against-light-boxes rate (62% of designed runs met six or fewer in the box) was the highest in the league. Fangio was the AP’s Assistant Coach of the Year, and Denver hired him as head coach in January 2019.10

What to notice: on 2018 Bears film, watch the front before the coverage. Against three receivers: Hicks and Goldman (or Bilal Nichols) inside, often in 4i-techniques so the guards can’t climb cleanly, Mack and Leonard Floyd standing up on the edges, and Danny Trevathan and Roquan Smith behind them: six in the box against three-receiver sets. Then watch the two safeties, Eddie Jackson and Adrian Amos, at the snap. On a run they come downhill late and arrive in the alley; on a pass they often stay deep and let the receivers come to them. That December the Bears held the Rams, the league’s best offense, to six points (the game is the film room in The Modern Era).

The Denver years (2019–2021) are the other half of the record, and they matter for the argument. As head coach, Fangio went 19–30 with no playoff appearances. His defenses were good (third-fewest points allowed in 2021) but not dominant, ranking 15th, 14th and 18th in EPA per dropback allowed. The weak link was the one the system leans on hardest: by this book’s count Denver’s pressure rate ranked 31st, 25th and 19th, so the four-man rush rarely got home. The lesson the league drew was not that the system failed. It was that a system built on a four-man rush and an interior front lives or dies on its players, and that a defensive head coach without a quarterback has a ceiling.11

NoteFilm room: Philadelphia Eagles 2024 and Super Bowl LIX — the system at full strength

After a year as an Eagles consultant (2022) and one as Miami’s coordinator (2023), Fangio took over a Philadelphia defense that had ranked 30th in scoring. In 2024 it allowed the fewest yards in the league and the second-fewest points, and, by this book’s count, ranked 1st in EPA per designed run allowed, while meeting 66% of opponents’ designed runs with six or fewer in the box (3rd-most).12

Then it played the best version of the problem the system was built for: Patrick Mahomes in the Super Bowl. In Super Bowl LIX (February 9, 2025; Eagles 40, Chiefs 22) the charting shows four or fewer rushers on 42 of Kansas City’s 42 dropbacks, and the coverage charted as quarters (Cover 4) on 25 of them. Mahomes was sacked 6 times. The Eagles didn’t need to blitz the best quarterback in football. They needed four rushers who could win and seven defenders who kept everything in front of them, and they had both.13

What to notice: in the Super Bowl LIX replay, count the rushers on each Chiefs dropback, then watch the safeties. Two deep, almost every snap; the deep ball never comes; Mahomes holds the ball looking for something downfield, and the four-man rush gets there.

Philadelphia’s 2025 defense was not as dominant (5th in points allowed), and the Eagles lost in the wild-card round to San Francisco. Fangio remains their defensive coordinator in 2026, at 68.14

The tree: how the system spread, and why results varied

A scheme spreads through people. A coaching tree is usually drawn by who worked for whom, but this one is better drawn by who took the idea: a two-high shell, a light box held by the front, four rushers, and coverage targeted to what the offense shows. Most of these coaches sat in Fangio’s meetings; one never did.

  • Brandon Staley coached Fangio’s outside linebackers in Chicago (2017–2018) and Denver (2019), then became the Rams’ coordinator for one season, 2020. With Aaron Donald inside and Jalen Ramsey moving around the secondary, the Rams allowed the fewest points in the league, 296, and, by this book’s count, the fewest EPA per dropback. The Rams under Staley added a lot of visible disguise to the system; that season is the film room of Disguise, Rotation, and Split-Field Coverage. Staley then became the Chargers’ head coach (2021–2023), and his defenses there ranked 29th, 21st and 24th in points allowed.
  • Ed Donatell, Fangio’s secondary coach for years, ran Denver’s defense under him (2019–2021) and then Minnesota’s (2022), which finished 28th in points allowed; he was let go after one season.
  • The Rams’ line of succession: Staley’s Rams staff included Ejiro Evero (safeties coach, later Denver’s coordinator in 2022 and Carolina’s since 2023) and Chris Shula (the Rams’ coordinator since 2024). Joe Barry, another Fangio-tree coach, took the system to Green Bay (2021–2023), where his defenses ranked between 10th and 17th in points allowed.
  • Sean Desai, a Bears assistant through Fangio’s years there, became Chicago’s coordinator (2021, 22nd in points allowed) and then Philadelphia’s (2023). His 2023 Eagles finished 30th; that is the defense Fangio inherited, and a year later, with largely the same ideas, it was second.
  • Jesse Minter is the branch that shows the idea traveling without the man. He never worked for Fangio; he worked beside Mike Macdonald on the Ravens’ defensive staff (2017–2020), succeeded him as Michigan’s coordinator (2022–2023), and ran the Chargers’ defense (2024–2025) with the same targeted split-field coverage. His 2024 Chargers allowed 301 points, the fewest in the league. He is the Ravens’ head coach in 2026.15

The record across the tree is uneven, and that is the most useful thing about it. The 2020 Rams (Donald, Ramsey), the 2024 Eagles (Carter, Davis, a deep secondary) and the 2024 Chargers led the league; Staley’s 2021 Chargers were 29th in points allowed and Evero’s 2024 Panthers 32nd, last. Carolina lost its best interior lineman, Derrick Brown, to a season-ending knee injury in Week 1 of 2024 and its veteran linebacker Shaq Thompson to a torn Achilles in Week 4, and by this book’s count ranked 32nd in pressure rate and 32nd in EPA per designed run.16 A light-box defense without the interior lineman who doesn’t need help is the failure case of decision 2. With Brown back for 2025, the same coordinator’s defense rose to 15th. The same ideas, run by coaches who understood them, produced the best and the worst defenses in football. The system decides how you will be beaten (slowly, on the ground, underneath), not whether. Its outcomes are set by the front four and the safeties.

There is also a reason the tree mattered more than any single coach. By 2021 offenses were facing some version of this defense almost every week. NFL Next Gen Stats counted two-high coverage on 32.9% of dropbacks in 2018 and 42.0% in 2025, the highest in its records, with every defense above 30%.17 When everyone plays the same defense, the defense that wins is the one that plays it with something the offense hasn’t seen. That is where Macdonald comes in.

Mike Macdonald: the student who added the missing piece

Mike Macdonald was born in Boston in 1987, grew up a Patriots fan, and finished high school outside Atlanta. He went to the University of Georgia to study finance, graduated summa cum laude, and stayed on as a graduate assistant and quality-control coach in the football program (2010–2013) while earning a master’s degree. He joined the Baltimore Ravens as an intern in 2014, and over seven seasons worked his way through defensive assistant, defensive backs coach and linebackers coach. In 2021 Jim Harbaugh hired him as Michigan’s defensive coordinator; Michigan’s scoring defense went from 95th in the country to 8th and the team reached the College Football Playoff. John Harbaugh brought him back to Baltimore as coordinator in 2022. In January 2024, at 36, he became the Seahawks’ head coach, the youngest in the league at the time, and he kept calling the defensive plays himself, with Aden Durde as his coordinator.18

He never worked on Fangio’s staff. But he came of age as a coach in the years when Fangio’s ideas were taking over the league, and his defenses start from the same place: two safeties back, a light box, a four-man rush most of the time. What he saw was the weakness that a universal defense creates. If everybody plays two-high, offenses learn to beat two-high. They spread it with three receivers and run into the six-man box; they put on extra tight ends and run into it with seven blockers; they throw quickly before the four-man rush arrives; they read the safeties after the snap. A Fangio defense is deliberately simple up front and complicated behind: few complex fronts, and, in the words of Eagles linebacker Zack Baun, “a lot of exotic stuff coverage-wise.”19 Offenses, once they had seen enough of it, knew exactly what the four-man rush and the six-man box would look like before the snap.

Macdonald’s answer was to make the front, the rushers and the personnel as hard to read as the coverage.

Multiplicity: one picture, many plays

Multiplicity is building a defense so that one pre-snap picture can become many different plays: several fronts, rushers and coverages that all start from the same look, and players who can play several roles, so the offense learns little before the snap. It is the opposite of a defense that wins by doing a few things the same way every snap.

Fangio’s defense is multiple mainly in one dimension, the coverage. Macdonald’s is multiple in four:

  • The coverage, decided at the snap, as in Fangio’s system. Seattle in 2025 played two-high coverage after the snap on 52% of dropbacks (3rd-most). Next Gen Stats counted 54.8% from Week 14 through the Super Bowl, the highest of any playoff team, and described Cover 6 as Macdonald’s beloved call.20
  • The front. Fangio’s front also puts five on the line and drops one, but it is the same five in the same spots, so the offense soon learns who the dropper can be (decision 3). Macdonald changes who is on the line: a linebacker walked up into an A gap, an edge standing up over a guard, a safety on the line outside the tight end. Any of the five may drop, and the protection can’t be set from the picture. In Macdonald’s own words, “We put five on the line and drop a guy.”21 Figure 66.3 draws the difference.
  • The rushers. Four rush, nearly always, but which four is not known. This is where his reputation for simulated pressure comes from, and it needs a correction, from the coach himself: “I’m going to disappoint you guys: We do not run a lot of simulated pressures,” he said on Seattle radio in November 2025. Seattle has them, and “it’s a great tool,” but by his definition a sim (simulated pressure) is a specific thing, four rushers with one of them a player the protection didn’t count, and it isn’t a major part of the plan. What is constant is the threat: a pressure look on snap after snap, with five or six who could come and four who do.22 (The data section below squares this with the 2023 Ravens’ reputation in Zone Blitzes.)
  • The personnel. Macdonald’s defenses are built on defensive backs who can play two or three positions: Kyle Hamilton in Baltimore; Devon Witherspoon, Julian Love and the rookie Nick Emmanwori in Seattle. When three safeties are on the field (a big nickel) and any of them can line up deep, in the slot or in the box, the personnel stops telling the offense what is coming. Before the Super Bowl, Macdonald said that with Emmanwori on the field, “when we go to dime, we’re really a nickel [4-3] team.”23 Decoded: Emmanwori is a safety who plays like a linebacker, so a six-DB package behaves like a five-DB one, and an offense that subs in receivers to face a light dime finds a nickel’s box instead.

The personnel is also the clearest record of how Macdonald built the Seattle defense. His first Seahawks defense, in 2024, was good, not great: 11th in points allowed and 12th in EPA per dropback, by this book’s count. It played the big nickel on only 6% of snaps. In 2025, with the rookie Emmanwori, that share was 56%, and the defense was 1st in points and 1st against the run. Put the four together and you get McDaniels’s complaint. Sixty-five of seventy snaps look the same, two safeties deep, a crowded line, three safeties somewhere, and from that one picture the defense can play quarters with a four-man rush from the usual spots, Cover 6 with a linebacker rushing and an edge dropping, Cover 3 with a safety rotating down, or, rarely, a real blitz. The quarterback can’t set the protection with confidence and can’t pick a side of the field with confidence. He has to find out after the snap, which is the slowest and most dangerous time to find anything out.

ImportantWhy it exists

Fangio’s system hides the coverage; Macdonald’s hides everything. Once two-high became the league’s default, its predictability became the weakness. Offenses knew a Fangio-style defense would rush four from known spots and leave six in the box, so they could set protections, plan quick throws and build run plays for that exact box. Multiplicity takes those pre-snap certainties away without giving up the two-high coverage that stops explosive plays. The cost is complexity: a defense this multiple needs smart, versatile players (expensive and rare), a lot of practice time, and a play-caller who can manage it all in real time. The call reaches the field through one radio helmet, the green dot, and the other ten have to hear it, adjust it to motion, and check out of it together. When that communication fails, the busts are spectacular: a receiver running free because two defenders each thought the other had him.

    off = eleven_2x2()
    dfn = [D("LE", "OLB", 1.0, -5.6, "E"), D("LT_", "DT", 1.0, T("3", -1), "T"), D("RT_", "DT", 1.0, T("2i", 1), "T"),

           D("NB", "NB", 5.0, -8.5, "$"), D("LCB", "CB", 8.5, -15.6, "CB"), D("RCB", "CB", 8.5, 15.6, "CB"),
           D("FS", "FS", 13.0, -7.5, "FS"), D("SS", "SS", 13.0, 7.5, "SS")]
    p = Play(off, dfn, los=40, to_go=7)
    # pre-snap: the Mike walks into the left A gap (mugged, about 2 yards off the ball), the Will creeps
    p.motion("MIKE", to=(2.0, -0.8), speed=1.6)
    p.motion("WILL", to=(3.5, 3.0), speed=0.8)
    # offense: dagger-style concept to the right (Y clears up the seam, Z digs under him), a go and a hitch left
    p.route("Z", [(4, 0), (12, 0), (12, -11)], absolute=False)
    p.route("Y", [(9, 0.0), (16.5, -1.0)]); p.route("X", [(19.5, 0)]); p.route("H", [(5, 0)])
    p.dropback(2.5)
    p.pass_to("Z", t=2.1)
    # protection: the line blocks the men on it; the back is responsible for the Will if he comes
    for k, to in (("LT", (-1.4, -4.6)), ("LG", (-1.3, -2.2)), ("C", (-1.2, -0.7)), ("RG", (-1.3, 1.5)),
                  ("RT", (-1.6, 4.4))):
        go_to(p, k, to, "block", speed=1.5)
    go_to(p, "RB", (-3.3, 2.5), "path", speed=2.0, delay=0.3)
    # the rush: both tackles, the left end and the Will
    go_to(p, "LE", (-1.0, -5.2), "rush", speed=2.0); go_to(p, "LT_", (-0.4, -2.4), "rush", speed=1.2)
    go_to(p, "RT_", (-0.4, 1.4), "rush", speed=1.2)
    go_to(p, "WILL", (-0.6, 2.6), "rush", speed=3.6, delay=0.1)
    go_to(p, "WILL", (-2.6, 2.4), "rush", speed=2.2)
    # the drops: the Mike backs out to the hook, the right edge walls Y and carries him up the seam
    p.drop("MIKE", (7.5, -1.5), speed=4.0, delay=0.15)
    p.drop("RE", (10.0, 8.6), speed=4.2, delay=0.1)

Count the lies. Before the snap the quarterback saw five on the line and a creeping Will: six possible rushers for six blockers. The line takes the five men on it, which leaves the Will to the back. But nobody knows before the snap whether the Will is coming, so the back has to stay in and wait to find out; that is the guess. If the Will drops, the back blocks nobody, and either way he isn’t running a route, so four receivers face seven defenders. The quarterback also saw two safeties deep with an open middle, so the dig behind the linebackers looked like the right throw. After the snap there were four rushers, two of the “rushers” were in coverage (the Mike in the short middle, exactly where a hot throw would go), and the open middle had a robber in it. Neither lie is new. Fire zones have dropped linemen since the 1990s (Zone Blitzes), and two-high rotation is Fangio’s. The modern step is doing both from the same picture, with the same personnel, and mixing them with plays on which nothing changes at all, so the quarterback can never assume a lie.

CautionCommon misconception

“Macdonald’s defense is a blitz defense.” It’s the opposite. Seattle sent five or more rushers on 22% of dropbacks in 2025 (25th-most of 32) and his 2023 Ravens on 21%. What looks like a blitz is usually a pressure look: more players near the line than will rush. The pressure comes from where the four come from and when the offense finds out, not from how many there are.

What the data can’t see: the disguise

The column every reader of this chapter most wants, how often a defense showed one shell and played another, isn’t in any free public file: the nflverse charting has no field for the safeties’ alignment before the snap. (Some charting services estimate the shell by eye, and Next Gen Stats has published a pre-snap disguise model, but neither is open data.) There is one way to measure it yourself: tracking data. The 2025 Big Data Bowl released every frame of every play from Weeks 1–9 of the 2022 season, including the frames before the snap, together with the charted coverage. With it you can call the shell from the safeties’ depth and width at the snap and compare it with the coverage label, play by play. Tracking Data and the Big Data Bowl builds this from the ground up; 06-06 shows the safety-rotation figure that comes out of it. The cell below is the core of that calculation (it isn’t run here, because the Big Data Bowl files aren’t distributed with this book).28

                  on=["gameId", "playId"])

Two cautions keep that calculation honest. Defenses that rotate before the snap (during the cadence) will be caught at the snap already in their real shell and count as honest, so the lie rate measures late rotation only. And the coverage labels are a charter’s judgment, which can differ between services: the Next Gen Stats article cited earlier says Julian Love’s fourth-quarter interception in Super Bowl LX did not come against split-safety coverage, while FTN’s charting labels the same play 2-Man, a two-high coverage. Treat any lie rate as an estimate with a few percentage points of noise either way.

You can still make a back-of-envelope guess for Seattle. If McDaniels is roughly right that about nine snaps in ten look the same (two-high before the snap), and Seattle played two-high after the snap on 52% of dropbacks, then it rotated out of its shell on at least 38% or so of dropbacks, before counting any lie about the rush. It is an estimate built on a coach’s quote, but it shows the scale.

There is one more number the data offers, as a caution. FTN’s charting counts how many rushers on each play were not regular linemen or edge rushers, and the rate of four-man rushes that included one is the closest public proxy for simulated pressure. By it, the 2025 Seahawks ranked 21st of 32 and the 2025 Eagles 2nd, the reverse of each coach’s reputation, and exactly what Macdonald said (“We do not run a lot of simulated pressures”).29

That looks like a contradiction of Zone Blitzes, whose 2023 Ravens film room calls Baltimore one of the league’s heaviest users of simulated pressure. It mostly isn’t, for three reasons. Different team and season: the same proxy puts the 2023 Ravens 13th, around the middle, not near the bottom like the 2025 Seahawks. Different counting: 07-03’s ranking came from PFF’s charting through Week 6 of 2023, as Cody Alexander reported it, while this proxy depends on how FTN’s charters tag each rusher, and a dropped tackle with a safety rushing in his place is exactly the snap two charters can code differently. Different things: much of what reporters called sim pressure was a pressure look, five or six possible rushers with four coming from the usual spots, which no public column records. The fair reading is that Macdonald’s defenses show the threat far more often than they run a true sim.

How offenses responded

The two-high era produced its own counter-revolution on offense, told as history in The Modern Era. Here is how each answer maps onto the five decisions, and how a multiple defense changes the math.

  • Run into the light box (attacks decision 2). Against six in the box, add a seventh blocker: a second tight end, a fullback, or the quarterback as a runner. In 2025, offenses used fewer than three wide receivers on 41.7% of plays, the highest since Next Gen Stats began tracking personnel, and snaps under center climbed back to about a third.30 A defense that answers by bringing a safety down is one-high again; one that stays two-high is outnumbered. Macdonald’s response was to make the extra defender a safety who looks like a deep player (the third safety of the big nickel), so the offense can’t tell from the personnel whether it has the numbers.
  • Use motion and tempo to force the picture early (attacks decisions 1 and 5). Motion at the snap makes safeties declare before they can disguise, and tempo denies them time to set the picture. New England went no-huddle on much of its fourth quarter in Super Bowl LX, partly from necessity, and moved the ball more against a defense with less time to disguise and every reason to keep it in front.
  • Throw before the rush and the rotation arrive (attacks decision 3). Quick throws and screens beat a four-man rush that needs two and a half seconds, and beat a robber who needs a second to get to the hole (the short middle, 10 to 15 yards deep). The cost is that quick throws are short throws; it is exactly what the two-high defense is willing to give up.
  • Read the safeties after the snap, and throw away from the rotation (attacks decision 5). A two-way concept pairs a route combination that beats two-high coverage on one side with one that beats one-high on the other, so the quarterback picks after the snap (Concepts Versus Coverages).
  • Make the read safety wrong (attacks decision 4). Play-action that shows the safety a blocking tight end, then releases the tight end behind him; RPOs that read the safety directly. This is the cost named in decision 4, cashed.
  • A quarterback who extends plays. Every one of these defenses is built on a timed four-man rush and zone coverage that keeps the ball in front. A quarterback who escapes the rush turns a covered play into a scramble, and zone defenders with their eyes on the quarterback can’t also chase him. In the Super Bowl, Maye’s 4 scrambles gained 38 yards, some of New England’s best plays of the night.

Defenses answered the answers. When offenses went heavy, defenses played more base and big nickel: base defense rose to 29.7% of snaps in 2025, up from 20.5% in 2023.31 When offenses moved at the snap, defenses rehearsed rotating late from motion. When offenses threw quickly, defenses taught their underneath players to break on the ball and tackle. The cycle hasn’t ended; it has just sped up.

CautionCommon misconception

“Two-high defenses are soft against the run.” By this book’s count the 2024 Eagles ranked 1st and the 2025 Seahawks 1st in EPA allowed per designed run, both while meeting most runs with six or fewer in the box. A light box is a choice about who stops the run (the front and late-arriving safeties), not a decision to stop it less. It does give up more yards per carry on average than a loaded box, and it fails badly when the interior linemen aren’t good enough.

What survives, and what offenses do next

By 2026 the two-high counterrevolution is no longer a counter anything. It is the default. What survives of Fangio’s system is almost all of it: the two-high shell as the starting picture, the light box held by the front, the four-man rush, quarters and Cover 6 as everyday coverages, and the rotation at the snap. What survives of Macdonald’s is the next layer, which the rest of the league is now copying: three-safety personnel, pressure looks without the blitz, and the conviction that the front should be as hard to read as the coverage.

What do offenses do next? These are predictions, not facts, but each follows from the logic of this chapter, and you can check them against the 2026 season as it unfolds.

  1. More blockers the offense controls. If a multiple defense hides its run defenders among three safeties, the offense’s cleanest counter is to count blockers it controls: 12 and 13 personnel, fullbacks, and quarterbacks who run. Expect the 2025 rise in heavy personnel to continue.
  2. Faster declarations. Motion at the snap, tempo, and fake snap counts all try to make the defense show its real picture before the ball moves. Expect more of all three, especially no-huddle on early downs, which takes away the time a multiple defense needs to set its disguise.
  3. Plays that don’t care. Concepts that work against both two-high and one-high (two-way reads, quick game paired with a deep shot) take the disguise out of play by making the coverage irrelevant.
  4. Attack the dropper and the rotator. Every pressure look ends with someone dropping who doesn’t usually cover, and every rotation ends with a safety arriving late to a spot. Offenses will target the defensive lineman in coverage and the area the rotating safety left.
  5. Quarterbacks who decide after the snap. In the end, multiplicity is a test of how fast a quarterback can process what he sees after the snap. Offenses will keep paying for the quarterbacks who can, and building simpler reads for those who can’t.

And defenses will answer again. The likeliest next step is the one Macdonald has already taken: more players who can do more things, so the same personnel can play every call. The cost of that strategy is in the roster, and it is covered in Building a Roster.

Watch for it on Sunday

TipWatch for it: the disguise audit

Pick one defense (Seattle or Philadelphia is ideal) and chart one half of passing downs on paper. You need three columns and about 25 rows.

  • Column 1, at the snap: the shell. Two-high or one-high? Count the safeties 10 or more yards deep as the ball is snapped (the center’s head goes down), not when the offense first sets: many defenses spin a second before the snap, and that is a pre-snap rotation, not a lie. Note anything unusual: a linebacker in an A gap, five on the line, a safety creeping.
  • Column 2, one second after the snap: what it became. Is the middle of the field closed (one deep safety) or open (two)? How many rushed (count the defenders who crossed the line)? Did anyone who looked like a rusher drop?
  • Column 3: the lie. Mark R if the shell changed (two-high to one-high or back), P if the rush changed (a pressure look that sent four, or a quiet look that sent five), RP for both, and S for same. A worked row: “2-high, M in the A gap, 5 on the line | 1-high, 4 rushed, M dropped | RP.”

At the end, compute the defense’s lie rate: the share of rows that aren’t S. Then split it. A Fangio defense should lie mostly with R (the coverage); a Macdonald defense with both R and P. For scale, one charted NFL defense built around disguise (Evero’s Carolina, by MatchQuarters’ count) rotated out of its two-high shell on about half the snaps that started there (Disguise, Rotation, and Split-Field Coverage). Use the high, wide replay for column 2; the live angle often cuts off the safeties.

The numbers say run: six blockers, six in the box, and the two deep safeties are 12 yards away. Which run do you call, and what is the defense counting on to stop it?

TipAnswer

A run that doesn’t go where the tite is strongest: wide, or with a seventh blocker. The tite is built to stop inside runs. The nose and the two 4is wall off the A and B gaps, and the Mike is free to fill whatever gap is left. An inside zone into that wall is the run the defense wants you to call. Two better choices:

  • Outside zone or a toss to the tight-end side, which asks the edge (E) to hold up against Y and puts the ball in the alley, where the strong safety has to come 12 yards to tackle. A tite pulls its linemen inside to choke the inside runs, which empties the gaps outside the tackles (05-03’s section “Inside zone or outside zone? How a front chooses” in Odd and Hybrid Fronts draws it). It may gain five; that is the bargain.
  • Add a seventh blocker without changing personnel: motion Z down into a wing or a crack block, or read the backside edge with the quarterback (a zone read), so one defender is left unblocked on purpose. The motion isn’t free, though: a match-quarters corner usually travels with Z or trades assignments with his safety, so the extra blocker brings an extra defender with him.

What the defense is counting on: that its interior linemen win their one-on-ones, and that the safeties tackle in space. And there is a trap. If Y blocks, the strong safety’s read says “run” and he comes downhill, so the same picture is a play-action opportunity: fake the run, release Y behind the safety.

How many will rush, and what should the quarterback plan to do?

TipAnswer

Most likely four, and the coverage is unknown until the snap. Seattle rushed exactly four on 76% of opponents’ dropbacks in 2025 and on 45 of 53 in the Super Bowl. Six could come, so the protection has to account for six (five linemen and the back), but expect one or two of the crowded players to drop, and expect the dropper to land in the short middle, right where a quick hot throw against a blitz would go.

The plan, then:

  • Don’t throw hot to the short middle the Mike appears to vacate. That is exactly where he drops.
  • Protect six, and give the back the Will. The line takes the five on it, and the back is responsible for the Will if he comes; if the Will drops, the back check-releases (waits a beat, then runs a short route).
  • Attack the dropper. If a lineman or edge drops (here, most likely the right edge), throw into his zone: a quick out or stick route by Y, or a seam the dropped edge has to carry. An edge rusher is rarely the quickest man in coverage.
  • Then read the safeties. If one rotates down (as in Figure 66.5), throw away from him, outside the numbers on the side he left. If both stay deep, the throws that beat two-high are open: a curl-flat combination outside, or a dig under the safeties if the Mike drops shallow.

This is what “finding out after the snap” means, and it is the moment the defense was built to win.

Who has the numbers in the run game, and what are the defense’s options before the ball is snapped?

TipAnswer

The offense does, before the snap: seven blockers (five linemen and two tight ends) against six in the box (four linemen and two linebackers; the nickel is outside the box over the left side). The defense has four answers, and each one is the trade this chapter has been about:

  • Let the reads do it. This is the main reason quarters exists. Each safety’s #2 is a tight end: U for the free safety, Y for the strong safety. If U and Y block, both safeties read run at once and fit (Y’s side into the alley, U’s side into the cutback), so the defense has eight run defenders after the snap without showing them before it. The price: the safeties arrive at four or five yards, not at the line, and the play-action trap from Drill 1 is now available on both sides.
  • Match the personnel. Sub to base (a third linebacker, or Fangio’s 3-4 tite) when 12 personnel comes on. Across the league, base defense rose to 29.7% of snaps in 2025 for exactly this reason (see “How offenses responded”). The cost is a slower defense on the field if the offense then spreads out.
  • Bring a safety down before the snap to make it seven against seven. Now the defense is one-high, and 12 personnel under center is the formation built for play-action shots against exactly that (the safety who came down is the one the fake pulls forward).
  • Rotate a safety down at the snap: stay two-high in the picture and add a seventh run defender only after the snap. That only works if the front can hold for the half-second the safety needs, which is why these defenses pay so much for interior linemen.

If the defense simply stays as it is and loses the numbers, the likeliest run goes to the right, away from the nickel: on the left, the nickel is a seventh defender close enough to fit; on the right, nobody is outside the edge but a safety 12 yards away. A Macdonald defense has one more answer you can’t see here: a third safety in the nickel’s spot, a safety who can play like a linebacker, so the box count the offense made before the snap may be wrong. That is what the big nickel is for.

Takeaways

  • The problem was the pre-snap commitment. One-high defenses had to decide the eighth run defender and the deep help before the snap, and McVay-era offenses and Mahomes punished that commitment every week.
  • Fangio’s answer is five decisions that pay for each other: two safeties back to stop explosive plays; a light box held by a tite front that wins its own fights; a four-man rush; quarters and Cover 6 so the safeties fit the run after the snap; and late rotation so the shell can lie. It works only when the interior linemen and the four rushers are good enough, which is why his tree produced both the best and the worst defenses in the league.
  • Macdonald’s answer is multiplicity: keep the two-high coverage, but make the front, the rushers and the personnel as hard to read as the coverage. Fangio puts the same five on the line; Macdonald changes who the five are. Same picture, many plays. It is a four-man-rush defense, not a blitz defense, and by his own account not even a heavy simulated-pressure defense; its pressure comes from pressure looks and uncertainty.
  • Super Bowl LX was the system on the biggest stage: four rushers on nearly every snap, mostly two-high coverage, six sacks, and no New England points for three quarters.
  • The public data shows the common ground (low blitz rates, pressure without the blitz, light boxes, strong run defense) but not the disguise. Measuring the lie takes tracking data or your own eyes.
  • Offenses answer by counting blockers they control, forcing early declarations and deciding after the snap. Heavy personnel, motion at the snap, tempo and quarterbacks who process fast are the counters, and defenses will answer them with even more versatile players.
