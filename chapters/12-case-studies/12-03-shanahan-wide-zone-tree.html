def named_line(players):
    """Label the five offensive linemen so the reader can follow each one."""
    return [p.moved(label=OL_NAMES[p.key]) if p.key in OL_NAMES else p for p in players]


def off21(strength="right"):
    """21 personnel from an offset I: the fullback (F) offset to the tight end's side, tight receiver
    splits. Seven on the line (five linemen, the Y and the X); Z is off the ball."""
    s = 1 if strength == "right" else -1
    out = []
    for q in named_line(offense("offset_i")):
        if q.key == "X":
            q = q.moved(w=-9.5)
        elif q.key == "Z":
            q = q.moved(w=9.5)
        elif q.key == "RB":
            q = q.moved(label="R")
        out.append(q if q.key in OL_NAMES else q.moved(w=s * q.w))   # linemen keep their sides
    return out


def D(key, pos, d, w, label):

def bear(cb_w=10.2, s=1):
    """A bear front with a single-high safety behind it, against a tight end on side s.

    return [D("BE", "DE", 1.0, -s * 4.9, "E"), D("B3", "DT", 1.0, -s * technique("3"), "3"),

            D("MIKE", "MIKE", 4.5, s * 2.4, "M"), D("WILL", "WILL", 4.5, -s * 2.6, "W"),

def go_to(p, key, end, kind="block", delay=0.0, speed=None, via=()):
    """Move `key` through absolute (d, w) points `via` to `end`, as a block, path, run or route."""
    start = np.array(p._cur(key))
    pts = [tuple(np.array(q) - start) for q in (*via, end)]
    if kind == "block":
        return p.block(key, pts=pts, delay=delay, speed=speed)
    if kind == "route":
        return p.route(key, [*via, end], absolute=True, delay=delay, speed=speed)
    return {"path": p.path, "run": p.run}[kind](key, pts, delay=delay, speed=speed)


def _P(fld, p, d, w):
    return fld.xy(*p.to_abs(d, w))


def branch(fld, p, pts, color=style.THIRD, lw=2.0, dash=(0, (3, 2.2)), end="arrow", z=6):
    """A dashed path through absolute (d, w) points, ending in an arrowhead or a block's T-bar."""
    xs, ys = _P(fld, p, np.array([q[0] for q in pts]), np.array([q[1] for q in pts]))
    if end == "arrow":
        fld.ax.plot(xs[:-1], ys[:-1], color=color, lw=lw, ls=dash, zorder=z, solid_capstyle="round")
        fld.ax.add_patch(FancyArrowPatch((xs[-2], ys[-2]), (xs[-1], ys[-1]), arrowstyle="-|>",
                                         mutation_scale=11, color=color, lw=lw, zorder=z))
    else:
        fld.ax.plot(xs, ys, color=color, lw=lw, ls=dash, zorder=z)
        v = np.array([xs[-1] - xs[-2], ys[-1] - ys[-2]])
        perp = np.array([-v[1], v[0]]) / np.linalg.norm(v) * 0.75
        fld.ax.plot([xs[-1] - perp[0], xs[-1] + perp[0]], [ys[-1] - perp[1], ys[-1] + perp[1]],
                    color=color, lw=lw + 0.6, zorder=z)

def clean(ax):
    """Box-zoomed diagrams: drop hash ticks and yard numbers, clip anything outside the window."""
    for ln in list(ax.lines):
        xs, ys = ln.get_data()
        if len(xs) == 2 and ln.get_linewidth() == 0.5 and np.hypot(xs[1] - xs[0], ys[1] - ys[0]) < 1.0:
            ln.remove()
    for t in list(ax.texts):
        if t.get_text().isdigit() and t.get_alpha() == 0.55:
            t.remove()
        else:
            t.set_clip_on(True)

MARK_R = 0.6   # frame-strip marker radius in yards, smaller than the static diagrams' so crowds stay legible


def tracking(play):
    df = play.to_tracking()
    df["_id"] = df["displayName"].astype(str)
    return df


def panel(ax, play, df, t, lateral, window, rings=()):
    """One moment of a simulated play on `ax`: markers, trails of on-screen players, rings."""
    tmap = df.groupby("frameId")["time"].first()
    fid = int((tmap - t).abs().idxmin())
    ylo, yhi = sorted((play.ball_y - lateral[0], play.ball_y - lateral[1]))
    fld = Field(ax=ax, orient="vertical", los=play.los, window=window, numbers=False, hashes=False,
                lateral=(ylo, yhi), to_go=play.to_go)
    cur = df[df["frameId"] == fid]
    shown = cur[cur["x"].between(play.los + window[0] + 0.4, play.los + window[1] - 0.4)
                & cur["y"].between(ylo + 0.4, yhi - 0.4)]
    ids = set(shown["_id"])
    trails = df[(df["frameId"] <= fid) & df["_id"].isin(ids) & (df["side"] != "ball")
                & df["x"].between(play.los + window[0], play.los + window[1])]
    for (_, side), g in trails.groupby(["_id", "side"]):
        xs, ys = fld.xy(g["x"].values, g["y"].values)
        ax.plot(xs, ys, color=style.OFFENSE if side == "off" else style.DEFENSE, lw=0.9, alpha=0.4, zorder=3)
    ppy = fld.pts_per_yard()
    for _, r in shown.iterrows():
        x, y = (float(v) for v in fld.xy(r["x"], r["y"]))
        if r["side"] == "ball":
            ax.add_patch(Ellipse((x, y + 0.45), 0.7, 0.42, fc=style.BALL, ec="white", lw=0.5, zorder=11))
            continue
        lab = str(r.get("label", "") or "")
        if r["side"] == "off":
            ax.add_patch(Circle((x, y), MARK_R, fc=style.OFFENSE, ec="white", lw=0.6, zorder=8))
            tc = "white"
        else:
            ax.add_patch(RegularPolygon((x, y), 4, radius=MARK_R * 1.3, orientation=np.pi / 4,
                                        fc=style.DEFENSE_TINT, ec=style.DEFENSE, lw=1.0, zorder=8))
            tc = style.INK
        fs = max(4.0, min(8.0, ppy * (0.8 if len(lab) <= 1 else 0.6)))
        ax.text(x, y, lab, ha="center", va="center", fontsize=fs, color=tc, fontweight="bold", zorder=10)
    for key, col in rings:
        r = cur[cur["displayName"] == key].iloc[0]
        x, y = fld.xy(r["x"], r["y"])
        ax.add_patch(Circle((float(x), float(y)), 1.05, fc="none", ec=col, lw=1.7, zorder=7))
    return fld


def strip(play, times, lateral, window, ncols=3, notes=None, rings=(), width_in=6.5, marks=None):
    """Numbered frame strip with a fixed window, so markers stay legible in print. `marks` maps a frame
    index to a function(fld, play, t) that adds that frame's on-field labels."""
    df = tracking(play)
    nrows = int(np.ceil(len(times) / ncols))
    pw = width_in / ncols
    span_w, span_d = lateral[1] - lateral[0], window[1] - window[0]
    fig, axes = plt.subplots(nrows, ncols, squeeze=False,
                             figsize=(width_in, nrows * (pw * span_d / span_w + (0.4 if notes else 0.2))))
    for ax in axes.flat[len(times):]:
        ax.axis("off")
    for i, (ax, t) in enumerate(zip(axes.flat, times)):
        fld = panel(ax, play, df, t, lateral, window, rings)
        if marks and i in marks:
            marks[i](fld, play, t)
        ax.set_title(f"{i + 1}.  " + ("snap" if abs(t) < 1e-6 else f"{t:+.1f} s"),
                     fontsize=8.5, color=style.INK_2, loc="left")
        if notes:
            ax.text(0.5, -0.03, notes[i], transform=ax.transAxes, ha="center", va="top",
                    fontsize=7.4, color=style.INK)
    fig.tight_layout(h_pad=1.4 if notes else 0.6, w_pad=0.6)
    return fig


def video(play, times, lateral, window, notes=None, rings=(), width_in=5.6):
    """Web version of the strip: the same panel at every frame, with the strip's notes as captions."""
    import shutil
    from matplotlib import animation
    from IPython.display import HTML, display
    df = tracking(play)
    span_w, span_d = lateral[1] - lateral[0], window[1] - window[0]
    fig, ax = plt.subplots(figsize=(width_in, width_in * span_d / span_w + 0.4))
    tmap = df.groupby("frameId")["time"].first()

    def update(fid):
        ax.clear()
        t = float(tmap[fid])
        panel(ax, play, df, t, lateral, window, rings)
        txt = "snap" if abs(t) < 0.05 else f"{t:+.1f} s"
        if notes:
            cur = [n for tt, n in zip(times, notes) if tt <= t + 1e-6]
            txt += ("   " + cur[-1]) if cur else ""
        ax.set_title(txt, fontsize=9, color=style.INK, loc="left")
        return []

    anim = animation.FuncAnimation(fig, update, frames=sorted(df["frameId"].unique()), interval=100)
    html = (anim.to_html5_video(embed_limit=40).replace("<video ", '<video style="max-width:100%;height:auto" ')
            if shutil.which("ffmpeg") else anim.to_jshtml(default_mode="once"))
    plt.close(fig)
    display(HTML(html))


def play_animation(play, times, notes=None, **kw):

Sunday, first-and-10 at midfield. San Francisco lines up with a fullback, a tight end and two receivers squeezed in tight to the formation, the quarterback under center. A receiver jogs in motion across the formation, the ball is snapped, and all five offensive linemen take the same hard step to their right. The running back takes the handoff on a path toward the tight end, the linebackers sprint sideways to meet him, and he cuts upfield for six yards. Two plays later: the same personnel, the same formation, the same motion, the same step by the line, the same back running the same path. Except this time the quarterback still has the ball. He is rolling the other way, alone, and the tight end who started on the right side of the formation is running free across the field on the left. He catches it in stride and runs for 20 more yards after the catch. The broadcast calls it “a busted coverage.” It wasn’t. Both plays were built to look identical for the first second, and that one second is the whole idea. This chapter is about the family of coaches who built an offense around it, and about why their offense seems to work no matter who plays quarterback.

Two problems, thirty years apart

Every successful offense is an answer to a question. The Shanahan family’s offense was built twice, by a father and then by his son, and each time it answered a different question.

Denver, 1995: the defense won’t stand still

When Mike Shanahan became the Denver Broncos’ head coach in 1995 he brought two things together. The first was the passing game of the San Francisco 49ers: he had just spent three seasons (1992–94) as George Seifert’s offensive coordinator, running the West Coast offense with Steve Young and winning Super Bowl XXIX.1 The second was a line coach, Alex Gibbs, who had a theory about running the ball.

The problem in front of them was the defenses of the mid-1990s. Dom Capers’s Pittsburgh zone blitzes sent linebackers into gaps and dropped linemen into coverage, and fronts slanted and stunted on almost every snap. A line that blocked “the man in front of me” kept finding that the man had left and somebody else had arrived. Gary Kubiak, Denver’s offensive coordinator under Shanahan, later remembered that in a man scheme “you’re getting picked off and you look bad.”2 Gibbs had been refining his answer since his first stint on Denver’s line in the 1980s, against the slanting, stunting fronts of that decade; the zone blitz made it more valuable, not new. His answer was outside zone, also called wide zone: every lineman steps the same way at the snap and blocks whoever enters his area, so a defense that moves is still blocked, and the running back reads the blocks and cuts once. Zone Running builds the play from the ground up; here what matters is the philosophy Gibbs attached to it. He wanted very few runs, repeated until the linemen could block them against any front, lighter and quicker linemen who could run sideways, and backs who would press the hole and make one decisive cut. Gibbs was famous among coaches for his blunt teaching and his near-religious faith in repetition; his clinic tapes are still sold to high school coaches.3 His backside linemen were taught to cut the defenders chasing the play (block them low, at the thighs, and put them on the ground), which is how Denver stopped backside pursuit, why the scheme had a “dirty” reputation among defensive linemen, and why the rules on low blocks matter so much to this family (03-02 explains the block; 09-03 the rules).

Shanahan supplied the other half: the bootleg, the pass that looks exactly like the run until the quarterback turns the other way. John Elway, in the last four seasons of his career, faked wide zone and rolled away from it again and again. Denver won Super Bowls after the 1997 and 1998 seasons; Terrell Davis, a sixth-round pick, ran for 2,008 yards in 1998 and was named MVP, and when Davis got hurt a string of backs who were not first-round picks kept producing 1,000-yard seasons in the same scheme.4 The lesson the league took was that the system, not the star, was doing a lot of the work. Hold on to that idea; it comes back, aimed at quarterbacks, later in this chapter.

The system didn’t disappear between Elway and 2017; it travelled with the coaches. Kubiak took it to Houston as head coach in 2006, with a young Kyle Shanahan on his staff; Mike Shanahan ran it in Washington from 2010 with Kyle as his coordinator; Kyle carried it to Cleveland and then Atlanta, where the 2016 Falcons were its last stop before San Francisco. The tree section below follows those staffs in detail.

San Francisco, 2017: no quarterback, and defenses built for the spread

Twenty-two years later Mike’s son Kyle Shanahan took over a 49ers team that had gone 2–14 in 2016.5 Kyle had grown up in the Denver building, played receiver at Texas, and coached under Gary Kubiak in Houston, under his father in Washington, and in Cleveland and Atlanta, where as offensive coordinator in 2016 his offense led the league in scoring and Matt Ryan was named MVP.6 He inherited a roster with no starting quarterback (his first starters were Brian Hoyer and the rookie C. J. Beathard) and few skill players a defense feared.

His problem was not the zone blitz. It was the defenses of the 2010s. To stop spread passing, teams had made the nickel (five defensive backs) their real base defense and filled the field with fast, light linebackers and safeties who could run with receivers. How do you beat a defense like that when you can’t count on your quarterback to win from the pocket on third-and-8?

Kyle’s answer kept his father’s run-and-boot core and added two things on top. First, make the defense declare its personnel and its coverage before the snap, by using personnel it doesn’t like (a fullback) and motion it has to react to. Second, design the passing game so that the hard part of a completion is done by the play, not the passer: throws to receivers already running, in space that the run fake has emptied, caught by players chosen because they can run after the catch. A defense that is fast and light is exactly the defense that has to run sideways to stop wide zone, and exactly the defense that has left grass behind its linebackers when the ball comes out on a bootleg.

ImportantWhy it exists

The Shanahan offense exists to make the defense’s speed and movement work for the offense. Zone blocking means slants and blitzes don’t leave anyone unblocked; the run’s sideways flow pulls fast defenders out of position; the bootleg and play-action passes look identical to the run for the first second and throw into the space the flow empties. The trade: it needs a particular roster (quick linemen, one-cut backs, a fullback, receivers who block and run after the catch), it asks for a lot of patience on early downs, and it is at its least natural when there is nothing to fake: third-and-long, two-minute drills, playing from far behind. Whether that last weakness is real is one of the things the data will test.

The answer: four ingredients that are one idea

Most offenses can be described as a list of plays. The Shanahan offense is better described as one picture shown over and over, with four ingredients mixed in different amounts. Learn the four and why each one is there, and you can recognise the family from the stands.

1. Wide zone, the play they will run all day

Wide zone is the base play, the play the whole offense is organised to protect. From Constraint Theory, a base play is the one you want to be good at no matter what the defense does, and the plays around it exist to stop the defense from cheating against it. Shanahan teams treat wide zone exactly that way. They will run it into a defense that is stopping it, on purpose, because every run teaches the defense something: the linebackers learn to flow hard with the line’s first step, the backside end learns to chase the back from behind, and the safety learns to come down fast. Every lesson the defense learns is a habit that a later play will punish.

That patience produces a kind of running back the scheme seems to manufacture. Denver’s string of late-round 1,000-yard backs had a modern sequel in San Francisco: Raheem Mostert, an undrafted free agent who ran for 220 yards in the 2019 NFC Championship Game, and Elijah Mitchell, a sixth-round rookie who ran for 963 yards in 11 games in 2021.7 (The reverse is also true: when the 49ers had an elite back, Christian McCaffrey, he was the league’s Offensive Player of the Year in 2023.) The scheme doesn’t make every back great, but it makes the job simple enough that an ordinary one can succeed: press the aiming point, see one crease, cut once.

2. The fullback, a personnel problem for the defense

In the 2010s, the fullback nearly vanished from the NFL; many teams carried none. Shanahan’s 49ers made him a centrepiece, signing Kyle Juszczyk in 2017 to what was then the richest contract ever given to a fullback.8 Why spend money on a position the rest of the league had abandoned?

Because a fullback puts the defense in a bind before the snap. Two backs and one tight end (21 personnel) is a running look, and the defense has two bad choices:

  • Stay in nickel (five defensive backs, two linebackers) and the offense has an extra blocker at the point of attack: the fullback leads into the hole on wide zone, and a light box gets run over.
  • Go to base defense (a third linebacker for the fifth defensive back) and the defense has put a slower player on the field against an offense whose fullback, tight end and back can all catch the ball. Now the play-action pass is a linebacker trying to cover George Kittle.

The fullback is also the offense’s most flexible piece once the ball is snapped. He can lead on wide zone, kick out the backside end on split zone (an inside-zone run on which he crosses behind the line to block that end), block the same end on a protected bootleg, or run the bootleg’s flat route, all from the same spot behind the quarterback. Coaches call that crossing path a slice: the same run across the formation is a block on split zone and a pass route on the boot, and the defense can’t tell which until the last step. Teams that copy the system without a fullback use a tight end or H-back for the same jobs.

3. Motion, which makes the defense answer first

Shanahan’s offenses move players before almost every snap: a receiver running across the formation (across or “zoom” motion), a receiver sprinting toward the quarterback at the snap (jet motion), a receiver circling behind the quarterback (orbit motion). Motion and Shifts explains each one. In this system motion does three jobs at once:

  • It asks the defense a question the offense can read. If a defender travels across the formation with the motion man, the coverage is probably man-to-man; if defenders bump over a spot or the safeties rotate, it is probably zone. That was a clean first rule in 2017. In the 2020s defenses learned to blur it (zone defenders bump, man defenders pass the motion man to a teammate instead of following him), which is one reason McDaniel’s Dolphins moved so much of their motion to the snap: a man still moving when the ball is snapped gives the defense no time to disguise its answer.
  • It changes the run’s arithmetic. A receiver in jet motion is just behind the quarterback at the snap and keeps running toward the far sideline, so the edge defender on that side has to respect a jet sweep even when the ball goes elsewhere; motion away from the play drags a defender out of the box.
  • It moves the strength after the defense has set. Defenses line up their front to the tight end’s side. Shift the tight end across (a trade) or bring a receiver across, and the 3-technique the defense put on the strong side is suddenly on the weak side, or a linebacker is a gap out of place, with seconds to fix it.
  • It hides the play. Because the same motion is used on runs, bootlegs and dropback passes, the motion itself tells the defense nothing: one look, many plays.

By the charting of FTN (a data company whose staff tags every NFL snap by hand: motion, play-action, where the quarterback lined up), the 49ers used motion (before or at the snap) on more plays than all but one or two teams in every season from 2022 to 2025; the numbers are in the fingerprint section below.

4. Passes designed for yards after the catch

The last ingredient is the most distinctive. Most offenses judge a pass play by where the ball is caught. Shanahan’s judges it by what happens after the ball is caught.

Think about where receivers are when the ball arrives on the plays you’ve seen so far. On a bootleg the tight end is crossing the field at full speed, behind linebackers who just ran the other way. On play-action the slot receiver is running a crossing route through the space the linebackers vacated when they stepped up for the run. On a jet sweep or a screen the ball carrier already has blockers in front of him. In every case the receiver catches the ball moving, facing upfield and with space ahead of him. That is a play designed to produce YAC, yards after the catch.

The roster follows the design. The 49ers’ most famous skill players of the Shanahan years were chosen less for how they get open deep than for what they do with the ball: George Kittle, a tight end who blocks like a lineman and in 2018 gained 870 of his 1,377 receiving yards after the catch; Deebo Samuel, a receiver built like a running back, who in 2021 had 1,405 receiving yards and also ran for eight touchdowns lining up in the backfield; and Christian McCaffrey, a back who catches like a receiver.10

To measure this, you need a way to separate “the play put him in space” from “he did something with it.” That is what the next statistic is for.

YAC over expected: separating the play from the player

YAC over expected, or YACOE, is the yards a receiver gains after the catch minus the yards a model expects him to gain from that catch, averaged per reception. Positive means more than the situation predicted.

The idea works like CPOE for quarterbacks. Raw YAC rewards receivers whose catches come in easy places: a screen caught behind the line with three blockers in front of it will produce more yards after the catch than a 20-yard out caught at the sideline, no matter who catches either. So a model first estimates the expected YAC of each catch from its circumstances, and YACOE is what is left:

YACOE on one catch = actual YAC − expected YAC

A worked example: a crossing route caught 6 yards downfield, in the middle of the field, on the run. The model, having seen thousands of similar catches, expects about 6 yards after the catch. The receiver breaks a tackle and gains 15. His YACOE on that catch is +9. A screen caught behind the line where the model expects 8 and the receiver gains 5 is −3. Over a season the noise averages out and the per-catch average tells you whether a team’s receivers consistently beat what their catches should produce.

Two models are in common use, and they disagree in an instructive way:

  • nflverse’s expected YAC (the xyac_mean_yardage column in the free play-by-play) is built from what the play-by-play knows: air yards, where on the field the ball was caught, down, distance and field position. It doesn’t know where the defenders were. So when a play design creates empty grass in front of the receiver, the nflverse model doesn’t see the grass, and the design shows up as YACOE.
  • Next Gen Stats (the NFL’s tracking-data service) builds its expected YAC from player tracking: the receiver’s speed and direction at the catch and the positions of the defenders around him. Space the scheme created is already in its expectation, so its YACOE leans more toward what the receiver did with that space.

A team that ranks high on both is probably getting yards from design and from players. Keep that in mind for the fingerprint section, because that is exactly what the 49ers did.

Why the four are one idea

Put the four together and each one makes the others work.

Wide zone makes linebackers flow fast and sideways; that flow is what empties the crossing lanes the YAC passes run through. The fullback forces the defense to choose between a light box (good for wide zone) and slower linebackers on the field (good for the passes). Motion tells the quarterback which coverage he is facing and moves defenders before the play starts. And all of it comes from the same few formations, so the defense cannot tell, in the first second, which ingredient is the one that matters on this snap. 08-02 draws the signature plate of this idea, “one formation, four plays” (wide zone, a naked bootleg, counter and a play-action shot, identical for the first half-second); it is worth a second look before you read on.

Signature play: wide zone against the bear

Every great offense forces defenses to invent answers. The oldest answer to wide zone is to take away the double teams it depends on. Wide zone works by pairing linemen: an uncovered lineman helps his covered neighbor reach a defender (the reach-and-overtake), then climbs to a linebacker. A defense that covers every interior lineman leaves no free lineman to help or to climb. That front has a name: the bear front, a nose on the center and a 3-technique on each guard, named after the 1985 Chicago Bears’ 46 defense that made it famous. Walk through it the way a defensive coordinator would.

What the bear takes away. Look at the middle of the line. On a normal four-man front, one guard is uncovered, and that guard is the engine of zone: he helps his neighbor, then climbs to a linebacker. Against the bear, the center and both guards each have a man in front of them, so nobody is free to help or to climb. No single block is impossible; the damage is that every interior lineman is on his own. The center has to reach a nose who can play either A gap. If the nose fights across the center’s reach, he is in the front-side lane before the back gets there. If he lets the center’s momentum carry past him, he falls into the backside A gap, which is exactly where the back’s cutback goes (the bend of bang-bend-bounce: the cut against the flow, behind defenders who over-ran the play). The left guard can’t help, because he is busy cutting off the backside 3 alone. Add an eighth defender in the box and the offense has more defenders than blockers.

What the offense does about it. Four things, and you can watch for all of them.

  • Bring the fullback. The fullback is the seventh blocker, and against an eight-man box he is the difference between one unblocked defender and two. Here he leads outside the tight end onto the safety, the man who would otherwise meet the back in the alley (the lane just outside the box).
  • Choose who goes unblocked. With eight defenders and seven blockers, someone is free. The offense picks the one who can hurt it least: the backside end, the man farthest from the ball. On a wide-zone run to the right he has to run the length of the line to make the tackle, from behind.
  • Change the run. Shanahan teams don’t only run wide zone into a bear. From the same look they run split zone, where the fullback crosses behind the line and kicks out that backside end, so every defender has a blocker; crack-toss, where the Z blocks down on the safety who rolled into the box and the ball gets outside the 3-techniques instead of asking linemen to reach them; and gap runs like duo and counter, which don’t depend on reaching anyone.
  • Make the free man pay. An unblocked defender who is taught to chase the run is the exact defender the bootleg is built to beat. Which brings us to the play that keeps the bear honest.
ImportantWhy it exists

The bear exists to take away the double teams and climbs that zone runs depend on, and it is a reasonable thing to show a wide-zone team on early downs. It is one of two opposite ways to defend this offense, and they trade the same things in reverse. The bear puts eight in the box with one deep safety: it wins the numbers against the run and gives up the bootleg and the play-action shot over the top. Two deep safeties with a light box (the 2020s answer you’ll meet under “How defenses answered”) takes away the shot and the easy boot and gives up runs: it bets that six or seven defenders can hold the line. Neither is a solution; each is a choice of which plays to give up.

Notice three things.

The first second is the run, with one deliberate exception. Freeze the strip at frame 1 and compare it with Figure 61.1: the line’s steps, the back’s path, the quarterback’s turn and the linebackers’ flow are the same. A linebacker who waits to find out which play it is has already lost on the run; one who flows fast has already lost on the boot. The exception is the fullback. Instead of leading to the right he crosses behind the line toward the backside end, which is exactly what he does on split zone, when he kicks that end out. The family pairs the boot with that look on purpose. To the end, a fullback coming at him means “a block is coming,” so he attacks it, and the fullback slips past him into the flat. This is also why the boot is a 21-personnel play rather than the two-tight-end version in 04-05: a fullback starts behind the line with a running start and a view of the end, so his “block” and his route look identical until the last step, while a tight end has to get off the line of scrimmage first.

The unblocked end is now the victim. On the run, leaving him free was cheap: he was chasing a back who was running away from him. On the boot, chasing is exactly wrong. His job against a bootleg is contain (stay outside the quarterback and force him back toward the pursuit), and he has given it up to chase a back who doesn’t have the ball. In frame 3 he is close to the quarterback, but he is inside and behind him, running the wrong way; a quarterback with a two-yard head start toward the sideline is gone. The more he is taught by the run, the better the boot gets.

The read is simple, even if the throw isn’t. The quarterback’s first look is that end. If the end has stayed home and is on him, the ball goes to the fullback in the flat at once, the answer 04-05 teaches. If the end chased, the quarterback has three receivers on one side at three depths (the flat, the crosser, the deep over) and reads them high to low. His legs are the fourth option: the man he watches for the keeper is the first defender who could come up to meet him on that side, usually the flat defender (here the corner, who has gone deep with the X). If that man drops with the routes, nobody is in front of the quarterback and he runs. Throwing on the move is harder than throwing from the pocket, which is why the throws are short and to receivers already running; what the boot buys him is a simpler decision and time, because every rusher is behind him. That is the “easy throws” half of Kyle Shanahan’s answer to the 2017 problem, and the reason this chapter’s next section is about quarterbacks.

What if the end doesn’t chase? That is the squeeze of Constraint Theory and Sequencing: the end shuffles down the line square, eyes on the quarterback, and the naked boot dies. 08-02 drew that duel; what this offense adds is what it does next. A squeezing end has stopped closing the back’s bend lane from behind, so the cutback opens on wide zone; split zone puts the fullback on him so the run is fully blocked; and if the offense still wants to boot, the fullback blocks him instead of releasing (the protected bootleg of 04-05). Every defensive answer leaves something else open; the play-caller’s job is to find out which answer the defense has chosen.

NoteMadden vs. real life

In Madden, “PA Boot” is a play you pick when you feel like it. In a Shanahan game plan, the bootleg is earned: it is aimed at a habit, a backside end and linebackers who flow hard with the run. The staff learns that habit from the opponent’s film all week (scripted openings often include a boot when the film says the end chases) and confirms it in the first quarter from the booth, and it often calls the boot on the next first down after a successful run from the same look. Called blind, against an end who isn’t chasing, the same play is a much worse one.

The quarterback question

Here is the claim that makes the Shanahan system famous, and controversial: it makes quarterbacks efficient. Not great, necessarily; efficient. A quarterback in this offense completes a lot of passes, throws for a lot of yards per attempt, and produces strong EPA per dropback, and he does it with a smaller share of difficult throws than quarterbacks in most other systems.

Why the system helps the passer

The reasons are all in the four ingredients.

  • The run fake moves the defenders out of the throwing windows. Linebackers who flow with wide zone are not in the middle of the field when the ball comes out on play-action; the crossing routes run behind them.
  • The bootleg halves the field and simplifies the read. A quarterback rolling right reads three receivers on one side, high to low, with no need to scan the whole field. The throw itself is harder (on the move, without setting his feet), so the routes are short and timed; the gain is that the rushers are behind him and the decision is simple.
  • Motion tells him the coverage before the snap, so the read after the snap is a confirmation, not a discovery.
  • Receivers catch it running. Throws are timed to receivers already moving, often across the field, so a catch tends to turn into yards. A five-yard throw that becomes a 20-yard gain counts as 20 passing yards for the quarterback.
  • Under center, the fake is better. When the quarterback turns his back to the defense to fake a handoff, the defense can’t see the ball, and the fake is more convincing than one from the shotgun (04-05 compares the two). The cost is that the quarterback has less time to read the defense on dropback passes, which the system accepts.

The debate: is that the quarterback or the system?

That design creates the argument every Shanahan-tree quarterback lives with. If the scheme manufactures open receivers and yards after the catch, how much credit does the passer deserve? It matters, because quarterback contracts are the biggest decisions a franchise makes.

The case for “system” is a list of names. Jimmy Garoppolo was efficient in San Francisco and was not after he left. Brock Purdy was the last pick of the 2022 draft, the 262nd player chosen, “Mr. Irrelevant,” and in 2023 he led the league in passer rating (the NFL’s official passing formula) and yards per attempt.11 Sam Darnold struggled with the Jets and Panthers, spent 2023 as Purdy’s backup, then had his two best seasons in offenses from the same family: Kevin O’Connell’s Minnesota Vikings in 2024 and Klint Kubiak’s Seattle Seahawks in 2025, where he won Super Bowl LX. Read Figure 61.3 with care, because it is the kind of chart that can prove whatever you want. Darnold’s jump from -0.08 to +0.10 EPA per dropback and Mac Jones’s from -0.04 to +0.13 are large, and they happened with new teammates, new offensive lines and, in Darnold’s case, a few more years of experience. Tagovailoa is the cleanest comparison (same team, same player, a new head coach in 2022), and his line hides a collapse: his 2025 season under McDaniel was his worst since his rookie year, and McDaniel was fired that winter. Garoppolo’s drop after leaving San Francisco came at age 32 and after injuries. And Goff, traded by McVay to Detroit for Matthew Stafford in 2021, averaged +0.13 per dropback in Detroit against +0.12 in Los Angeles, and had his best season (2024) after the Lions built their own run-and-play-action offense around him. Goff’s line doesn’t show a limit; it shows that the ideas travel to a cousin offense.

The Goff-for-Stafford trade is the sharpest test of the “system” argument, and the averages can’t settle it. With the Rams, Goff (2017–2020) averaged +0.12 EPA per dropback and Stafford (2021–2025) +0.12; on third down with 7 or more to go, +0.01 and +0.01. Yet McVay made the trade, won Super Bowl LVI in Stafford’s first season, and in 2025, with Stafford at 37, the Rams led the league in play-action and still had the MVP at quarterback: Stafford threw a league-leading 46 touchdown passes.12 Whatever McVay was paying for lives in the handful of plays an average hides, the ones that decide January games. That is a coach’s judgment, not a number, and it’s honest to call it one.

The critics have one more argument: the big games. Kyle Shanahan’s offenses have led in the second half of three Super Bowls and lost all three: as Atlanta’s coordinator in Super Bowl LI (up 28–3 in the third quarter, lost 34–28 in overtime), and as the 49ers’ head coach in Super Bowl LIV (up 20–10 in the fourth quarter, lost 31–20) and Super Bowl LVIII (lost 25–22 in overtime).13 Those losses get pinned on play-calling with a lead and on an offense that, critics say, can’t throw its way out of obvious passing downs. The second claim is testable, and the data doesn’t support it. From 2019 to 2025, on third down with 7 or more yards to go (the downs where everybody knows it is a pass), the 49ers’ dropbacks averaged +0.18 EPA, 1st of 32 teams, against a league average of -0.10. Look at how, though: their throws travelled less far in the air than most teams’ (22nd in average air yards on those snaps) and their completions gained more after the catch (5th in YAC per catch). Even on third-and-long, the 49ers’ answer is designed: a schemed throw to a receiver on the move, not a quarterback inventing something.

So the honest summary is in between. The system reliably raises the floor: quarterbacks who can read a defense, deliver the ball on time and throw on the move become efficient in it, and some of them have looked like stars. Its third-and-long numbers say the design reaches further than critics think. What no design supplies is the throw after the design breaks (the coverage was right, the protection failed, and the quarterback has to create something), and those plays are rare enough that season averages barely see them. Whether they decide the biggest games is the real argument, and it is one the data in this chapter can’t end.

CautionCommon misconception

“System quarterback” means “bad quarterback.” It doesn’t, and the phrase is used carelessly in both directions. Every quarterback plays in a system; the Shanahan system just makes its contribution unusually visible. Matt Ryan’s MVP season (2016), Aaron Rodgers’s MVP seasons under Matt LaFleur (2020 and 2021) and Stafford’s in 2025 all came in this family’s offenses, from quarterbacks nobody calls system players. The useful question isn’t “system or quarterback?” but “how much does this quarterback add beyond what the design gives him?”, which is what EPA, CPOE and YAC over expected try to separate.

How would you recognise this offense from data alone, without watching a snap? Four measurements do most of the work: how often the quarterback is under center, how often he fakes a handoff, how much the receivers gain after the catch, and (to check that all this produces anything) how efficient the offense is.

The tree in 2025 and 2026

A coaching tree is the family of coaches descended from one head coach, traced through who worked on whose staff. The Walsh case study draws the whole Walsh tree, of which this family is the largest modern branch: Mike Shanahan was Seifert’s coordinator in San Francisco before he went to Denver. This section zooms in on the part of it that calls plays today.

The family has two roots in the Denver building of the 1990s. Gary Kubiak, Mike Shanahan’s coordinator, carried the system to Houston as head coach (2006–13) and back to Denver, where he won Super Bowl 50 after the 2015 season; his sons Klint and Klay are coaches in it now. Kyle Shanahan took it through Houston, Washington, Cleveland and Atlanta to San Francisco. The decisive moment was the Washington staff of 2010–13, under Mike Shanahan with Kyle as offensive coordinator: Sean McVay coached the tight ends, Matt LaFleur the quarterbacks, and Mike McDaniel (from 2011) was an offensive assistant and then the receivers coach. All four became NFL head coaches.24 (That Washington offense also produced one of the scheme’s purest seasons: in 2012 Alfred Morris, a sixth-round rookie, ran for 1,613 yards in wide zone, beside rookie quarterback Robert Griffin III and the read-option plays the staff borrowed from Baylor, which The College Laboratory covers.)25

From there it branched. McVay’s own staffs in Los Angeles produced head coaches Zac Taylor (Bengals, 2019), Kevin O’Connell (Vikings, 2022) and Liam Coen (Jaguars, 2025). LaFleur took the system to Green Bay in 2019; Aaron Rodgers won two MVPs in it. Kyle’s 49ers staffs produced McDaniel (Dolphins, 2022), Mike LaFleur (Matt’s brother) and Brian Fleury, and they gave Klint Kubiak a season on the staff before he went on to run offenses in New Orleans and Seattle.

That is also why the family’s offenses differ. A coordinator who leaves takes the ideas, but he builds them around his own quarterback, his own roster and his own instincts, which is how a shotgun team in Cincinnati and an under-center team in Los Angeles can share a grandfather.

How defenses answered

A system this successful for this long gets studied by every defensive staff in the league. Their answers fall into four groups, and you will see all four on any Sunday.

1. Change the front. The bear in Figure 61.1 is the oldest answer: cover the interior linemen so nobody can climb. Its cousin is the tite front (a nose and two linemen on the tackles’ inside shoulders), which walls off the middle and asks the edges to win; and its opposite is a wide four-man front with a quick 3-technique who penetrates before the guard’s reach can land (05-01 animates it). Odd and Hybrid Fronts explains why no single front stops both zone runs, so defenses change fronts by opponent and by down. The offense’s counter is everything in this chapter: the fullback, the unblocked backside end and the boot, and the change-ups from the bear section (split zone, crack-toss, duo and counter) out of the same looks when the front is built for zone.

2. Keep two safeties deep and play the run with fewer men. The most important defensive trend of the 2020s, the two-high shell, is partly an answer to this family. Play-action and bootlegs work best against a defense that brings a safety down to stop the run, so defenses led by Vic Fangio and his descendants kept both safeties deep, accepted a light box, and trusted bigger fronts to hold up against the run. Two-high coverage rose from 32.9% of snaps in 2018 to 42.0% in 2025 by Next Gen Stats’ count, the highest since tracking began.28 The fight between that idea and this one is the subject of Fangio’s Two-High Revolution and Macdonald’s Answer, and of The Modern Era.

3. Take away the middle and make the quarterback hold the ball. The crossing routes that produce all that YAC run through the middle of the field at linebacker depth. Fill that area with defenders, rush with stunts that take a moment to develop, and the quarterback who wanted to throw to a receiver on the move is holding the ball. The most famous example is New England’s plan in Super Bowl LIII, which held McVay’s Rams to 3 points; The Belichick case study draws it.

4. Hunt the tells. An offense built on looking the same can be beaten if it doesn’t quite look the same. Defensive staffs grade the running back’s depth, the fullback’s alignment, the receivers’ splits and the direction of the motion, looking for any detail that separates the run from the boot. That is the self-scouting war of 08-02: the offense’s job is to keep its tendencies balanced inside each look, and the defense’s job is to find the one look that isn’t.

And the system answered back. When defenses went light and two-high, the family ran the ball more and got heavier (in 2025 offenses used fewer than three wide receivers on 41.7% of plays, the highest rate in the Next Gen Stats era), and when the bear came, the boots came with it.29 When defenses took away the crossers, McVay went and got a quarterback who could beat them from the pocket.

What survives

Ideas from this family are now just how offense is played in much of the NFL, including by teams with no connection to it:

  • Under center is back, led by this family (the numbers are in the fingerprint section).
  • Motion is universal. PFF counted a shift or motion on 63.9% of plays in 2025, up from 38.4% in 2016; every team but one used it on at least half its plays.30
  • Heavier personnel is back, for the same reason the 49ers kept a fullback: to make light defenses pay.
  • YAC as a design goal. Receivers are now scouted and drafted for what they do after the catch, and offenses design throws to catch them on the move.
  • The illusion of complexity (few plays, many looks; many plays, one look) is standard teaching on almost every offensive staff, inside the tree and out.

What may not survive is the specific balance. The 2025 Rams leaned harder on play-action from under center than anyone, while the 49ers’ play-action rate was ordinary, and the 2025 second generation in Cincinnati and Atlanta looks more like a spread offense than like Denver in 1998. A tree spreads by mutating. That is also how you will know it is still alive: the family resemblance will be in the questions the offense asks a defense, not in any one formation.

The quarterback fakes wide zone left, keeps the ball and rolls right behind him: the fullback slices into the right flat, the tight end crosses from left to right behind the linebackers who flowed with the fake, and if nobody comes up to stop him, the quarterback runs. First-and-10 is the ideal down for it: the defense expects run, and a short completion keeps the offense on schedule.

Why not run wide zone again? Against the bear it was losing yards; the defense has the numbers. The offense’s answer is not to block the bear better but to punish the habit the bear’s backside end has fallen into.

TipAnswer

A run. Count the box: six defenders against six blockers (five linemen and the tight end), with both safeties 12 yards deep. That is a light box, and the system’s standing answer to two-high safeties is to run the ball at them. The motion told the quarterback two more things: nobody ran across with the Z, so the coverage is very likely zone, and the safeties didn’t rotate down, so the box is staying light.

Which way is the play-caller’s choice, and both answers are textbook in this family:

  • Wide zone left, toward the motion (or the jet sweep itself, handed to the Z). The Z is just behind the quarterback at the snap and keeps running through the mesh toward the left sideline. He doesn’t block anyone; he is a threat, and the defender on the left edge has to respect a jet sweep for a beat, which is a beat he isn’t attacking the run.
  • Fake the jet, wide zone right, away from it. The fake drags the Will and the nickel back toward the motion and thins the right side, where the run is going.

If you said “run,” you read the picture; the direction is the bonus question.

TipAnswer

A play-action shot from under center, with max protection: two deep routes against the one safety. Put the two changes together. The naked boot is off the table, because the end who used to chase is squeezing and would be waiting for the quarterback. Wide zone is running into eight in the box, which is what the spin was for. But the spin has a price: one safety is left to cover the whole deep field, and at the snap he is still running to the middle. That is the picture the yankee concept from 04-05 is built for: fake wide zone, keep the fullback and the back in to protect, and send two receivers deep, a post from one side and a deep over from the other. The linebackers and the safety who came down step up for the fake; the lone deep safety can take only one of the two routes, and the quarterback throws the other.

Why not the protected boot? It’s a fine safe call, and a good one on second-and-long. But it attacks the short and middle areas the eight-man box is already crowding; on first-and-10 with one deep safety still moving, the shot is the bigger reward. This back-and-forth, base play, constraint, a new answer, a new constraint, is sequencing, and it is how a Shanahan play-caller spends a game.

Takeaways

  • Two problems, one family. Gibbs’s and Mike Shanahan’s 1990s Broncos built wide zone and the bootleg to block defenses that never stood still; Kyle Shanahan’s 2017 49ers added the fullback, motion and YAC passing to beat fast, light defenses without a star quarterback.
  • Four ingredients, one picture. Wide zone moves the defense sideways, the fullback forces a personnel choice, motion makes the defense answer first, and YAC passes throw into the space all of that empties. The same few looks carry all of it.
  • Leave a man free on purpose, then punish him. Against a bear front the offense can’t block everyone, so it leaves the backside end and beats him with the bootleg-keeper when he crashes; when he squeezes, the cutback, split zone and the protected boot are waiting. The defense’s answer to the base play sets up the constraint play.
  • The system raises the quarterback’s floor, and the ceiling is still argued. Several quarterbacks were much more efficient inside the family than outside it; even the 49ers’ third-and-long success is designed. Goff thrived in a cousin offense in Detroit, and EPA can’t see the difference McVay paid for in Stafford: the plays after the design breaks are a coach’s judgment, not a number.
  • The fingerprint: under center and in motion, 21 or heavy personnel (49ers), play-action at the top of the league (Rams), and receivers who gain far more after the catch than the catch predicts. From 2019 to 2025 the 49ers were first in YAC over expected by a wide margin and top ten in EPA per play in six of seven seasons.
  • Label every staff by season. In 2025 the family’s first generation called plays in San Francisco, Los Angeles, Green Bay, Miami and Seattle (Super Bowl LX champions), with McVay’s former assistants calling them in Minnesota, Jacksonville, Cincinnati, Atlanta and Houston. In 2026 Kubiak (Las Vegas) and Mike LaFleur (Arizona) are head coaches who call plays, McDaniel calls the Chargers’ plays, Fleury Seattle’s, Slowik coordinates Miami’s offense and Stefanski is Atlanta’s head coach.
