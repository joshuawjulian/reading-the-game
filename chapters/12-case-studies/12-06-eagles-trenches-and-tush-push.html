def move_to(p, key, *pts, kind="block", delay=0.0, speed=None):
    """Move `key` through absolute (d, w) points as a block, rush, run or path."""
    start = np.array(p._cur(key))
    rel = [tuple(np.array(q) - start) for q in pts]
    if kind == "block":
        return p.block(key, pts=rel, delay=delay, speed=speed)
    if kind == "rush":
        return p.rush(key, pts=rel, delay=delay, speed=speed)
    return {"path": p.path, "run": p.run}[kind](key, rel, delay=delay, speed=speed)

def set_tracks(p, tracks, t_first=-2.6):
    """Hand-made keyframes {key: [(t, d, w), ...]} that may start before the snap. Every listed player
    starts at his alignment at t_first; players not listed stand still. The Philly Special is a play of
    pitches and fakes that gridiron's routes can't time, so every moving body is keyframed by hand."""
    kf, _, t1 = p._compile()
    kf = dict(kf)
    for key, rows in tracks.items():
        q = p.players[key]
        kf[key] = np.array([(t_first, q.d, q.w)] + [tuple(r_) for r_ in rows], dtype=float)
    p._compiled = (kf, t_first, max(t1, *[rows[-1][0] for rows in tracks.values()]))


def ease(points, step=0.05):
    """Smooth keyframes through (t, d, w) waypoints (ease in and out on each leg)."""
    rows = []
    for (ta, da, wa), (tb, db, wb) in zip(points[:-1], points[1:]):
        for tt in np.arange(ta, tb, step):
            f = (tt - ta) / (tb - ta)
            f = f * f * (3 - 2 * f)
            rows.append((round(float(tt), 3), da + (db - da) * f, wa + (wb - wa) * f))
    rows.append(tuple(points[-1]))
    return rows


def frame(ax, p, t, lateral, window, rings=(), ltg=True, numbers=False, mr=0.75):
    """Everyone at time t (negative = before the snap), with faint trails of where they have been
    since the snap."""
    ylo, yhi = sorted((p.ball_y - lateral[0], p.ball_y - lateral[1]))
    fld = Field(ax=ax, orient="vertical", los=p.los, window=window, numbers=numbers, lateral=(ylo, yhi),
                to_go=p.to_go if ltg else None)
    ppy = fld.pts_per_yard()
    wl, wr = sorted(lateral)
    ring = dict(rings)
    for k, q in p.players.items():
        d, w = p.position(k, t)
        if not (window[0] + 0.4 < d < window[1] - 0.4 and wl + 0.4 < w < wr - 0.4):
            continue
        if t > 0:
            pts = np.array([p.position(k, tt) for tt in np.linspace(0.0, t, 40)])
            if np.ptp(pts[:, 0]) + np.ptp(pts[:, 1]) > 0.3:
                xs, ys = P(fld, p, pts[:, 0], pts[:, 1])
                ax.plot(xs, ys, color=style.OFFENSE if q.side == "off" else style.DEFENSE, lw=1.0,
                        alpha=0.5, zorder=3)
        x, y = P(fld, p, d, w)
        draw_marker(ax, float(x), float(y), q.side, q.label, ppy, ring=ring.get(k), r=mr)
    clean(ax, numbers)
    return fld


def strip_grid(p, times, notes, lateral, window, rings=(), ncols=2, width_in=6.5, deco=None, fs=6.9, ltg=True,
               mr=0.75):
    """Print stand-in for an animation: numbered panels, each with a sentence underneath."""
    nrows = int(np.ceil(len(times) / ncols))
    pw = width_in / ncols
    ph = pw * (window[1] - window[0]) / abs(lateral[1] - lateral[0])
    fig, axes = plt.subplots(nrows, ncols, squeeze=False, figsize=(width_in, nrows * (ph + 0.8)))
    for ax in axes.flat[len(times):]:
        ax.axis("off")
    for i, (ax, t) in enumerate(zip(axes.flat, times)):
        fld = frame(ax, p, t, lateral, window, rings, ltg=ltg, mr=mr)
        if deco:
            deco(t, fld, p)
        lab = "before the snap" if t < -1e-6 else ("snap" if abs(t) < 1e-6 else f"+{t:.1f} s")
        panel_title(ax, f"{i + 1}.  {lab}")
        under_note(ax, notes[i], fs=fs)
    fig.tight_layout(h_pad=3.2, w_pad=0.8)
    return fig


def video(p, lateral, window, t_end, rings=(), width_in=5.6, t_start=-0.4, notes=None, times=None, ltg=True,
          mr=0.75, deco=None):
    """Web version of a strip: the same frame drawn every tenth of a second (pre-snap frames
    included), with the strip's notes as a running caption."""
    from IPython.display import HTML, display
    span_w, span_d = abs(lateral[1] - lateral[0]), window[1] - window[0]
    fig, ax = plt.subplots(figsize=(width_in, width_in * span_d / span_w + 0.5))
    ts = np.round(np.arange(t_start, t_end + 1e-9, 0.1), 2)

    def update(i):
        ax.clear()
        t = float(ts[i])
        fld = frame(ax, p, t, lateral, window, rings, ltg=ltg, mr=mr)
        if deco:
            deco(t, fld, p)
        txt = "before the snap" if t < -1e-6 else ("snap" if abs(t) < 1e-6 else f"+{t:.1f} s")
        if notes and times:
            cur = [n for tt, n in zip(times, notes) if tt <= t + 1e-6]
            if cur:
                txt += "   " + cur[-1].replace("\n", " ")
        ax.set_title(txt, fontsize=8, loc="left", color=style.INK)
        return []

    fig.tight_layout()
    anim = animation.FuncAnimation(fig, update, frames=len(ts), interval=100, blit=False)
    html = (anim.to_html5_video(embed_limit=40).replace("<video ", '<video style="max-width:100%;height:auto" ')
            if shutil.which("ffmpeg") else anim.to_jshtml(default_mode="once"))
    plt.close(fig)
    display(HTML(html))

def pts_(x, sign=True):

Third-and-1 at midfield, and everybody in the stadium knows what is coming. The Philadelphia quarterback crouches under center. Two teammates line up so close behind him that they could rest their hands on his hips. The defense crowds the middle, linebackers rocking on their toes, ready to leap the pile. The broadcast cuts to a graphic about how often this works, the snap comes, the whole mass lurches forward, and the chains move. Seven years earlier the same franchise won its first Super Bowl with the most famous trick play of the century, a play built entirely on the defense not knowing what was coming. The problem: a franchise built to outlast its quarterbacks

Every NFL team says it builds around its quarterback. The Eagles of the last decade had a harder problem: their quarterback situation kept changing under them. In 2017 the starter, Carson Wentz, tore his ACL in December, and the Super Bowl was won by his backup. In 2020 the team drafted Jalen Hurts in the second round, a pick many questioned barely a year after Wentz had signed a long extension, and within a year he was the starter in an offense rebuilt to suit him. Around the quarterback, the play-callers changed too: Nick Sirianni is in his sixth season as head coach (2021–2026), and he has had five offensive coordinators in those six seasons: Shane Steichen for the first two, then a new one every year.

What did not change, for more than a decade, was the line of scrimmage. Look at who stayed:1

That table is the thesis of this chapter. The Eagles treated their two lines as the part of the team that lasts, and everything else (the quarterback, the play-caller, the scheme of the season) as the part that changes. It is not the fashionable approach. Modern team building chases the highest-impact positions: the quarterback first, then the players who rush the passer and those who catch or cover passes. Offensive linemen and run-stuffing tackles are often treated as the places to save money. Philadelphia’s bet was that an elite line makes every quarterback better, survives every coordinator, and turns ordinary plays (a sneak, an inside run, a four-man rush) into extraordinary ones.

The rest of the chapter tests that bet across three Super Bowl teams: 2017, 2022 and 2024.

2017: RPOs, a backup quarterback, and an aggressive coach

Doug Pederson came to Philadelphia in 2016 from Andy Reid’s staff in Kansas City, and he brought Reid’s offense with him, including its fondness for the RPO (run-pass option), a run play with a quick throw attached that the quarterback chooses between after the snap by watching one defender. By PFF’s count, the 2017 Eagles ranked second in the league in RPO usage, behind only Reid’s Chiefs.2 The RPOs chapter tells the mechanics and that season’s story in full; here is what matters for this case study.

The RPO as a quarterback simplifier. Wentz was playing at an MVP level when he tore his ACL against the Rams on December 10, 2017. His backup, Nick Foles, was a pocket passer with a fine arm and none of Wentz’s improvising. The coaches leaned harder on RPOs, because an RPO hands the quarterback his answer: if the defender he is reading plays the run, throw; if he drops into coverage, hand off. A quarterback who reads one man and throws on time does not need to be a star. By PFF’s charting, Foles completed 15 of his 16 RPO throws in his first five starts after the injury,3 and in the playoffs the Eagles scored 38 points against the league’s best defense in the NFC Championship and 41 in the Super Bowl.

An offensive line that made the run real. An RPO only works if the run part is a real threat. If the defense can stop the handoff with the players it already has in the box, it never has to choose. The 2017 line (Jason Peters at left tackle until he tore his ACL and MCL on October 23, then Halapoulivaati Vaitai;4 Stefen Wisniewski, Kelce, Brandon Brooks and Lane Johnson inside and at right tackle) was one of the best in football, and Jay Ajayi, LeGarrette Blount and Corey Clement ran behind it.

Aggression on fourth down. Pederson was also one of the first NFL coaches to act on what analysts had been saying for years: that teams punt and kick far too often on fourth-and-short. An analyst on his staff, Ryan Paganetti, sat on the headset during games feeding him the math on fourth downs.5 In 2017 the Eagles went for it on fourth down 26 times, the 2nd-most in the league, and converted 65%.6 That habit produced the two most important snaps of their Super Bowl: a fourth-and-1 conversion in their own territory late in the fourth quarter (the Film room box at the end of this section), and the most famous trick play of the century.

The Philly Special

Super Bowl LII, February 4, 2018, in Minneapolis. The Eagles led New England 15–12 with 38 seconds left in the first half and faced fourth-and-goal from the 1. Almost every coach of that era would have kicked the field goal. During a timeout, Foles walked over to Pederson and asked, in words a sideline microphone caught, “You want Philly Philly?” Pederson paused, then said, “Yeah, let’s do it.”7

The play they called is the Philly Special, a direct snap to the running back, a reverse to a tight end, and a pass from the tight end back to the quarterback. Take it a step at a time, because each step has a reason; the four panels of Figure 64.1, after the steps, show them from the shotgun alignment to the catch.

  1. Foles walks up, and stays eligible. Only a few players may catch a forward pass: the two players on the ends of the line, and the backs, the players lined up at least a yard behind it. A quarterback under center, with his hands under the center’s to take the snap, is the one back who is not eligible. A quarterback in the shotgun is just a back. Foles started in the shotgun, so he was eligible from the start, and the walk didn’t change that: he stood behind his right tackle, more than a yard off the line, and never put his hands under the center. The walk was for deception. The gesturing, the shouted words and the walk itself told the defense he was changing the protection, the most ordinary thing a quarterback does before a snap. A walk like that is a shift, so the rules then require every offensive player to stand still for a full second before the snap, and Foles did. At the snap he did nothing for a beat, which is the hardest part of selling it.
  2. The direct snap goes to the running back. Kelce snapped the ball straight to Clement, who had motioned in behind Foles before the walk. To eleven defenders, a running back with the ball on the 1-yard line means a run, and every one of them has a run responsibility. The linebackers and the defenders on Clement’s side chased him.
  3. The reverse brings the ball back. Clement pitched to Burton, a tight end coming the other way. A reverse is a misdirection play by itself; here it moved the ball away from the defenders who had just committed to Clement. The pitch has to go sideways or backward. A team gets only one forward pass on a play, and only from behind the line; a forward toss here would have used it up, and Burton could not have thrown.
  4. The throw. Burton had played quarterback in high school and taken snaps at quarterback at Florida. He rolled right and lobbed the ball to Foles, who had walked into the end zone after every defender near him had turned toward the run and nobody had stayed with him. (The right end in panel 3 ran past Foles: his job was the ball, and the ball was in the backfield.) Torrey Smith, crossing the end zone, was a second option Burton didn’t need.

        return (p.position("RB", 0.35) + BALL_OFF) * t / 0.35
    if t < T_PITCH:
        return p.position("RB", t) + BALL_OFF
    if t < T_CATCH_PITCH:
        a, b = p.position("RB", T_PITCH) + BALL_OFF, p.position("Y", T_CATCH_PITCH) + BALL_OFF

    if t < T_CATCH:

        return a + (b - a) * (t - T_THROW) / (T_CATCH - T_THROW)
    return p.position("QB", t) + BALL_OFF

Why does it work? Because the defense’s rules, not its effort, let it down. At the goal line every defender has a gap to fill or a receiver to cover, and nobody has “the quarterback,” because the quarterback is the one player a defense never has to cover. The play turns that rule into a hole. It also lands at the perfect moment: fourth-and-goal from the 1 is when a defense is most committed to stopping the run, and least able to think about anything else.

What it gives up is just as clear. A trick play is a one-time bet. If a single defender stays home on the quarterback, Burton is rolling right with the ball and nowhere to throw it, on fourth down. And it can only fool a defense that hasn’t seen it, so its value falls every time it’s run.

It wasn’t new, either. The Eagles found it on film of the Chicago Bears, who had run a version of it for a touchdown against Minnesota late in the 2016 season and called it the “Clemson Special,” after a college team that had used it years before. Wide receivers coach Mike Groh, who had coached in Chicago, helped install it, and the Eagles had considered running it in the NFC Championship Game before the score made it unnecessary.8 Even the alignment was argued about afterward: the former officiating chief Mike Pereira said Alshon Jeffery, alone on the right, was a step off the line. The offense must have at least seven players on the line of scrimmage, so with Jeffery off it there would have been only six: an illegal formation. Jeffery had checked with the official before the snap, and no flag was thrown.9

CautionCommon misconception

“The Philly Special was a gamble a coach dreamed up on the sideline.” It was a gamble, but not an improvised one. The play had been installed and practised for the playoffs, the coaches had chosen the situation for it in advance, and the quarterback asked for it by name. What was bold was the decision to go for it on fourth down at all, and that was the 2017 Eagles’ habit, not a one-off. Trick plays look like inspiration; they are usually homework.

2021–2025: Sirianni, Hurts, and a run game built in the trenches

Pederson was fired after the 2020 season, and Nick Sirianni, the Colts’ offensive coordinator, replaced him in January 2021. His first offense started like most modern offenses: in the shotgun, throwing the ball. After seven games the Eagles were 2–5, and two things changed at once. Sirianni handed play-calling to his offensive coordinator, Shane Steichen, and the offense turned around. In weeks 1–7 the Eagles ran on 38% of their snaps; from week 8 on, on 59%. They won seven of their last ten games, made the playoffs, and led the league in rushing at 160 yards a game.11 Be careful about crediting either change alone: the play-caller and the run share changed in the same week, and seven games against ten is far too little to tell them apart. What the next five seasons show is that the running stuck after the play-caller left.

Why would a modern offense get better by running more? Because of who was running it.

The quarterback as an extra blocker

On a normal handoff, the quarterback gives the ball away and stops mattering, so the offense blocks with ten players against eleven defenders. Jalen Hurts is strong, fast and hard to tackle, so a defense has to account for him on every snap, and that changes the arithmetic. On a zone read one defender is left unblocked, and Hurts reads him: hand off if that defender crashes inside, keep it if he stays outside. On a designed quarterback run (QB power or QB counter), the running back becomes a blocker. Either way the offense blocks with one more player than usual. Coaches call a blocker a “hat,” so this is the “extra hat” idea from the Option Football chapter.

The RPO came back too, now with a quarterback who could also run. In 2022, FTN’s charting has Philadelphia using RPOs on 14% of its plays, more than any other team and well above the league’s 5%.12 The richest versions read two defenders in sequence. First, at the handoff, Hurts watched the unblocked end: if the end crashed toward the back, Hurts pulled the ball. Then, with the ball still in his hands, he watched a second defender, a linebacker or the nickel back: if that man came up to stop the quarterback run, Hurts threw behind him; if he stayed back, Hurts ran. Each defender had to guess right, and one wrong guess was enough. The share fell back toward the league’s in later seasons (4% in 2024, 13th in the league, and 6% in 2025) as the offense leaned on Barkley’s handoffs instead.

2022: the line-first idea on both sides

The second Super Bowl team, in Steichen’s second season as play-caller, is the clearest picture of the line-first idea working on both sides of the ball. The 2022 Eagles went 14–3. The offense ranked 3rd in EPA per play (expected points added), with Hurts the MVP runner-up and the tush push at the start of its run. The defense recorded 70 sacks, the most in the league and tied for the third-most by any team in NFL history, and it was the first team ever with four players reaching double figures.13 They lost Super Bowl LVII to Kansas City 38–35, a game in which Hurts ran for 70 yards and three touchdowns and threw for 304 more.

What to notice: on their 2022 film, how often the four-man rush got home by itself. Haason Reddick, Josh Sweat, Javon Hargrave and Brandon Graham each had at least ten sacks. The line, and the man who coached it

None of that works without the five men in front. Jeff Stoutland came to Philadelphia in 2013 from Nick Saban’s Alabama, hired by Chip Kelly, and he stayed through Pederson and Sirianni. That run is rare in a league where position coaches usually leave with the head coach who hired them. He was the Eagles’ run game coordinator as well as their line coach, and his reputation rested on turning unlikely players into starters:

  • Jordan Mailata, an Australian rugby league player who had never played a down of American football, was a seventh-round pick in 2018. He made his first starts in 2020, was the full-time left tackle from 2021, and in 2024 signed a three-year, $66 million extension that made him one of the four highest-paid tackles in the league.
  • Jason Kelce, a sixth-round pick in 2011, became a six-time first-team All-Pro. He was undersized for a center, and size is usually the point of the position: a center’s hardest job is holding his ground against a nose tackle lined up directly over him, often the heaviest man on the defense. The Eagles asked Kelce to move instead, reaching defenders on zone runs, pulling, and racing out to block for screens, where quickness matters more than mass, and his timing made their sneak the most feared in football.
  • Mekhi Becton, a former first-round pick the Jets had given up on as a tackle, signed a one-year deal in 2024, moved inside to right guard at the Eagles’ request, and started 15 games for a Super Bowl champion.15

The two plays ask different things of the line, and the Eagles’ line could do both. Duo (taught in the Gap Schemes chapter) is a straight-ahead shoving match: two double teams move the two defensive tackles backward, and the back runs through whichever gap the Mike linebacker leaves open. It needs linemen strong enough to move a 300-pound man two yards against his will. (This book files duo with the gap runs because it has no zone footwork; many coaches, especially in the Shanahan family, teach it as inside zone’s cousin, “power without the puller.” It is the same play under either label.) Outside zone (from the Zone Running chapter) is a race: the whole line steps sideways together, the defense has to run with it, and somewhere a defender overruns his gap and opens a seam. The back’s answers are the zone chapter’s bang, bend, bounce; against a light box that is chasing the play sideways, the bend, cutting back behind the defenders who overran it, is often the big one. It needs linemen quick enough to get their heads across a defender’s body. Johnson, Mailata, Landon Dickerson and Kelce (and from 2024 his successor at center, Cam Jurgens) were rare in being good at both.

Notice who is not blocked on outside zone: the backside end. Nobody has time to reach him, so the play leaves him alone and lets the quarterback hold him. If he chases Barkley from behind, Hurts can pull the ball and run (or roll out to throw) where the end used to be. That is the extra-blocker idea from the previous section, working on a handoff.

Why did all this matter so much in 2024? Because of what defenses were doing. Across the league, defenses were playing more and more two-high coverage, keeping both safeties deep to stop explosive passes, which leaves only six defenders near the line: a light box (the Modern Era chapter tells that story). A light box is the defense making a bet: we’ll give you four or five yards on the ground rather than twenty in the air. The Eagles’ line, with Barkley behind it, made that a losing bet.

The honest summary is that the Eagles’ offense was built on things that stayed (the line and Hurts), not on the play-caller who changed.

Be careful with the “why” of 2025, too. Several things changed at once: a new play-caller, a Barkley coming off the heaviest workload of his career (he ran for 1,140 yards, down from 2,005), defenses that crowded the box more often (see “How opponents responded”), and, according to reporting by the Philadelphia Inquirer and ESPN, a push by Sirianni and Patullo to diversify the run game that cut into Stoutland’s role as its designer; he lost the title of run game coordinator during the season. The Eagles still went 11–6 and won the NFC East, then lost at home to San Francisco in the wild-card round, 23–19.19 On February 4, 2026, Stoutland announced that he was leaving, after 13 seasons; his replacement is Chris Kuper, the Vikings’ line coach from 2022 to 2025, who had worked under Fangio in Denver.20 A single season can’t separate those causes. What it can tell you is that the one stable thing in the Eagles’ offense was also the one that slipped.

The tush push as a talent edge

The chapter on Red Zone, Goal Line, Two-Point Plays, and Short Yardage explains how the tush push works and shows that league-wide, quarterback sneaks of every kind (FTN charts the plain sneak and the push under one label) convert about four times in five on 1 yard to go. It ends on a puzzle: the Eagles’ conversion rate on the play sits close to the league line, even though they ran it far more often than anyone, against defenses that knew it was coming. This section takes that puzzle up. If the push is the Eagles’ signature, how big was their edge, really?

The ingredients are not in doubt. Hurts has unusual leg strength for a quarterback, and he took the snap already crouched, ready to drive. Kelce, and the guards beside him, fired out low at the snap. The two pushers lined up against Hurts’s hips, where a shove moves the whole man rather than bending his back. Johnson, Mailata and Dickerson were among the best linemen at their positions, and Stoutland drilled the play’s timing until the line moved as one at the instant of the snap. The question is what those ingredients produced.

Read the blue dots first. Over the four seasons, the Eagles converted 82% of their 137 sneaks on 1 yard to go, and the rest of the league converted 81% of theirs: a difference of about 1 point.21 Season by season, the edge was +3 points in 2022, +14 points in 2023, −3 points in 2024 and −14 points in 2025. That is not what a near-unstoppable play looks like. Three things explain the gap between the reputation and the numbers, and they are worth learning, because they apply to almost every “Team X is great at Y” claim you will hear on a broadcast.

1. Small samples. Thirty or forty sneaks a season is not many. A play that truly succeeds 82% of the time will, in a season of 35 tries, land anywhere from about 71% to 93% just by chance. That is what the vertical bars show, and in three of the four seasons the bar crosses zero. A single good or bad season of sneaks tells you very little. Four seasons pooled tell you more, and they say: about league average.

2. Selection. Every other team chose its sneaks. They could run the play when the ball was spotted inches from the line to gain, against a defense that looked soft in the middle, and call something else when it wasn’t. The Eagles ran it on 51% of all their 1-yard-to-go snaps in these four seasons, against a league median of 14%, so they must have sneaked on many snaps other teams would have passed up, against defenses that had a week to plan for it. The data can’t confirm that their snaps were harder: the play-by-play records anything from an inch to a full yard as “1 yard to go.” It is an inference from how often they called it, and it means a league-average rate on the Eagles’ sneaks probably hides some edge. That is why the orange dots matter. They compare every snap on 1 yard to go, whatever the call, so no team gets to pick its spots: a team that sneaks only on the easy ones has to convert the hard ones some other way, and both count. On that measure the Eagles converted more often than the league in every season, but only 2023, at +14 points, is clearly more than noise.

3. Volume, and what it was worth. Across the league, sneaks on 1 yard to go converted 81%, other runs 70% and passes 58%. Don’t read that as “a sneak adds eleven points of success on any snap”: those other plays were chosen too, often on the longer “1 yards” a team didn’t trust its sneak on, so the gap overstates what calling a sneak every time would add. What it does show is why a team that can call its best short-yardage play on most short-yardage snaps, without the rate collapsing, gains something. Most teams can’t: their quarterback isn’t built for it, their line can’t win the snap every time, or their coach won’t put the starter at the bottom of a pile forty times a year.

How much did the Eagles gain? The orange dots answer it directly. Multiply each season’s edge on all 1-yard snaps by the number of those snaps, and you get the extra first downs (or touchdowns) the Eagles converted compared with a league-average offense facing the same snaps: about 2 in 2022, 9 in 2023, and fewer than one in 2024 and in 2025.22 In 2023 that came to about one extra first down every other game, often the one that keeps a drive alive. That was a real edge. In the other three seasons the gain was small enough to be luck. The talent edge, in other words, wasn’t a higher sneak rate. It was being able to put the sneak on half of all short-yardage snaps and still convert like the league, and only in 2023 did that add up to much more.

And then it faded. The 2023 season, Kelce’s last, was the peak. In 2024 the sneak rate fell back to the league’s; in 2025 the Eagles converted 65% of their sneaks, well below the rest of the league’s 80%, and ran fewer of them (26). Their other plays on 1 yard to go picked up the slack, so across all their 1-yard snaps they still matched the league (the orange dot, +1 point); what disappeared was the push’s own advantage. The candidate explanations are the ones in “How opponents responded”: the center who timed the snap had retired, defenses had studied the play for three years, and the league told its officials to watch for linemen moving early. The sample is too small to rank those causes. The direction is not in doubt.

2024: Barkley, Fangio, and a four-man rush

The 2023 Eagles started 10–1 and collapsed, losing six of their last seven games, including a lopsided wild-card loss in Tampa. The defense was the main culprit: by EPA per play it ranked 29th in the league.23 Brian Johnson’s offense had slipped too, from 3rd the year before to 8th, but it was still one of the better offenses in football; a defense ranked 29th is what sank the season. The offseason that followed was the most consequential in the franchise’s recent history:

  • Vic Fangio was hired as defensive coordinator. Then 65, he had first coached in the NFL in 1986 and spent most of the four decades since in the league, and he was the man behind the two-high, light-box defenses that had spread across the league from his late-2010s Bears and Broncos24 (his ideas are the subject of their own case study). He was still the Eagles’ coordinator in 2026.
  • Saquon Barkley signed as a free agent, the rare case of the Eagles paying a running back.
  • The draft brought two rookie defensive backs, cornerback Quinyon Mitchell in the first round and Cooper DeJean in the second, who both started on a Super Bowl defense as rookies.
  • Zack Baun signed a one-year deal. The front office saw him as a backup edge rusher and special-teams player; Fangio, from a handful of snaps on Baun’s New Orleans film, saw an inside linebacker, moved him there that spring, and Baun became a first-team All-Pro.25 That is coaching as roster building: the cheapest way to add a star is to find one playing the wrong position.
  • Kellen Moore became offensive coordinator, the third in three seasons.

The result was a 14–3 season and a defense that allowed the fewest yards in the league and ranked 2nd in EPA per play, a year after ranking 29th. Fangio’s defense blitzed less than most. A blitz is a pass rush with five or more rushers, one more than the usual four; by FTN’s charting, the 2024 Eagles sent one on 19% of dropbacks, 28th of 32 teams, against a league rate of 26%.26 Their four rushers were usually, though not always, the four down linemen. It trusted those four to get there and kept seven players in coverage. Then, in the Super Bowl, it did that on every snap.

Super Bowl LIX: four rushers, no blitzes

February 9, 2025, New Orleans: Eagles 40, Chiefs 22. The Eagles led 24–0 at halftime and 34–0 late in the third quarter, and the defense did it without blitzing. NFL Next Gen Stats counted 42 Patrick Mahomes dropbacks and zero blitzes, only the fourth time a defense had gone a whole game without blitzing since it began tracking in 2014, and still credited the Eagles with 16 pressures, a pressure on 38.1% of dropbacks, and six sacks, the most Mahomes had taken in a game in his career.27 FTN’s charting agrees on the count that matters: it never had more than four Eagles rushing on any scrimmage dropback.28 After the game the defensive end Josh Sweat, who had 2.5 sacks, put it plainly: “Vic didn’t want to blitz. We knew if we had to blitz, we’d be in trouble.” Figure 64.5, below, shows the shape of it.

Why would not blitzing be the right plan against the league’s best quarterback? Think about what a blitz gives a quarterback like Mahomes, and what it costs.

  • A blitz trades coverage for speed. Every extra rusher is one fewer defender in coverage, so somebody is uncovered or left one-on-one. A quarterback who reads the blitz before the snap knows where that somebody is, and Mahomes was as good as anyone at punishing it with a quick throw or a scramble into the space the rushers left.
  • Four plus seven takes away the quick answer. With seven in coverage, here a quarters shell (four deep, three underneath, from the Quarters and Match chapter), every route has a defender near it. FTN charted the Eagles in quarters (Cover 4) on 27 of the 45 dropbacks, and in some two-deep-safety coverage (quarters, Cover 2 or Cover 6) on 30, far more than the 34% two-high rate it charted for them in the regular season. The quarterback has to hold the ball, and every half-second he holds it is another half-second for the rush.
  • It only works if four can win. That is the condition, and it is mostly a roster condition. Jalen Carter, Milton Williams, Jordan Davis, Josh Sweat and Nolan Smith could beat blockers one-on-one, on the interior as well as the edges. And that night the matchup helped: after an injury to Kansas City’s left tackle in December, Joe Thuney, an All-Pro left guard, had moved out to left tackle and started there in the Super Bowl, which weakened the edge and the interior at once.29 Most defenses can’t get to the quarterback rushing four, so they have to blitz.

Roster building: paying for the lines

How did the Eagles keep elite lines for a decade? The tempting answer is “they spent their high draft picks on linemen,” and it is only partly true. From 2013 to 2025 the Eagles spent 7 of their 25 first- and second-round picks on offensive or defensive linemen (28%), 30th among the 32 franchises and below the league median of 38%.31 Their famous lines were built another way:

  1. Develop the unlikely. Mailata (seventh round), Kelce (sixth round), Isaac Seumalo (third round) and Becton (another team’s discarded first-rounder) were turned into starters by Stoutland’s coaching. That is cheap talent, and a team that can make it has money left for everything else. (Baun’s move to inside linebacker was the same trick on defense.)
  2. Pay to keep them. When those players became good, the Eagles paid them. Using contract data from Over the Cap, a website that tracks every NFL contract, the Eagles’ offensive line took up 18.2% of the salary cap in an average season from 2017 to 2025, against a league median of 15.2%, and ranked between 4th and 13th in the league every year.32
  3. Spend the big picks where they move the needle. The first-round picks of the 2021–2025 drafts went to a wide receiver (DeVonta Smith, 2021), two defensive tackles and an edge rusher (Davis, Carter, Nolan Smith), a cornerback (Mitchell) and a linebacker (Jihaad Campbell, 2025). Pass rushers who win one-on-one are the scarcest players in football after quarterbacks, and you usually can’t develop them from the seventh round.
  4. Use the quarterback’s cheap years. In 2022 Hurts was still on his second-round rookie contract, among the cheapest starting quarterbacks in the league. That gap between what a good young quarterback earns and what a veteran costs is the rookie-contract window, and the Eagles spent it: they traded for the receiver A.J. Brown and signed Reddick that spring. Hurts’s five-year, $255 million extension came in April 2023.33

Notice how each choice connects to a scheme choice. A team that develops linemen cheaply can afford a run-first offense, because the line is good enough to win without trickery. A run game that wins with five blockers frees the quarterback from carrying the offense. A pass rush that wins with four lets the defensive coordinator play coverage instead of guessing with blitzes. And a sneak that converts four times in five, and can be called on almost every short-yardage snap, makes fourth-and-1 a play to run, not a gamble. The roster is the scheme.

The trade-offs are real too. Money spent on linemen is money not spent on receivers or a second cornerback. Linemen get old together: the Eagles had to replace Kelce, Brooks and Peters within a few seasons of one another. And an offense whose identity is “our line is better than yours” has a hard time when the line is merely good, which is one way to read 2025.

How opponents responded

The league’s responses to the Eagles came in three forms.

On the field. Defenses attacked the push the only way it can be attacked: by winning the snap. They crowded both A gaps beside the center with their biggest linemen, timed the cadence so their tackles were moving as the ball was, and sent linebackers leaping over the pile to meet Hurts before it moved. Some defensive linemen began jumping the snap count on purpose, accepting an occasional offside penalty for the chance to get into the backfield first. Against the run game, defenses crowded the box: on first and second down, the Eagles faced seven or more defenders in the box on 34% of snaps in 2024 and 42% in 2025, when only 5 offenses saw heavy boxes more often.34 (The details of those tactics are in the Red Zone, Goal Line, Two-Point Plays, and Short Yardage chapter.)

The most vivid version came in the NFC Championship Game on January 26, 2025. With the Eagles in their push formation at the Washington 2, linebacker Frankie Luvu tried to time a leap over the pile and was flagged for jumping early, twice; a defensive lineman then jumped a hard count for a third penalty. The referee, Shawn Hochuli, announced that the officials could award a touchdown if Washington kept fouling to prevent one. The Eagles scored anyway and won 55–23.35 The leap itself is legal on a running play if it is timed after the snap. (Since 2017 it has been banned only on field-goal and extra-point tries, a safety rule for kicks that fans often assume covers every play.36) Luvu’s problem was the clock: to beat the push he had to leave before the ball moved.

In the rulebook. When you can’t stop a play, you can try to ban it. In 2025 Green Bay proposed a rule against pushing the runner; after being tabled once, a revised version failed 22–10 on May 21, two votes short of the 24 needed. The league answered by telling its officials to watch more closely for false starts and players lined up in the neutral zone on the play. In 2026 no team proposed a ban, and the push is legal this season.37

By copying. The sincerest response was imitation. Other teams, Buffalo most of all, ran sneaks and pushes in large numbers, and the light-box run game that Philadelphia and others built became one of the league’s standard answers to two-high defenses. And other teams hired the Eagles’ coordinators: Steichen to Indianapolis and Moore to New Orleans as head coaches, and Jonathan Gannon, the 2021–22 defensive coordinator, to Arizona as head coach in 2023. Success exported the people who had built it, which is the other cause of the churn.

What survives

As of the 2026 season, here is where the experiment stands.

  • The head coach and the defense survive. Sirianni is in his sixth season, and Fangio is still the defensive coordinator. The four-man rush philosophy depends on the front office continuing to find linemen who can win one-on-one.
  • The quarterback survives. Hurts, under contract for the long term, is the constant on offense, and so is his ability to make the defense account for him as a runner.
  • The architect does not. Stoutland, the line coach behind every Eagles line since 2013, is gone, and the offense has its fifth coordinator in Sirianni’s six seasons. The most interesting question for the 2026 Eagles is whether the trench identity was a system that outlives its coach, or a person.
  • The push is legal, but its edge has shrunk. This chapter’s data stop at the end of 2025, so here is your baseline: in 2025 the Eagles sneaked on 54% of their 1-yard snaps and converted 65% of those sneaks, against 80% for the rest of the league. Whether the 2026 Eagles, with a new line coach, still call it that often, and whether it climbs back to the league’s rate, is the cleanest test you can run from your couch this season.

How many will rush? Then count: how many receivers go out against how many defenders in coverage, and how many blockers against how many rushers? Where does the pressure have to come from?

TipAnswer

Four will rush, the four linemen; this defense blitzed less than almost any in the league. Now the counts. With the back staying in, Kansas City has six blockers against four rushers, so two blockers are spare and can double the most dangerous rushers. It has four receivers going out against seven defenders in coverage, so every receiver has a defender on him and three more are sitting in the throwing lanes. Nothing quick is open.

So the defense is betting on time. It concedes that the rush will rarely be free; it needs the four to beat their blockers anyway, and it needs the push to come up the middle, because a quarterback can drift away from edge rushers but not from a tackle in his face. That is why Carter and Williams mattered so much in Super Bowl LIX.

Compare the double A-gap mug in the drill at the end of the Zone Blitzes, Fire Zones, and Simulated Pressure chapter: a pressure look shows six to force the offense to keep the back in, which produces this same four-against-seven count. But if the mugged linebackers then drop, the center and guards are freed to double a rusher, so a mug pays off only when the rush runs a twist into a protection that has set the wrong way. Fangio got the count without showing anything. What would change it: Kansas City sending the back out too (five receivers against seven, five blockers against four), or a defense that doesn’t trust its four, which has to blitz and give up a defender in coverage.

Takeaways

  • The Eagles’ decade is one long bet on the lines of scrimmage: elite blockers and pass rushers as the stable core, and the quarterback, the coordinator and the scheme around them allowed to change.
  • 2017 won with a backup quarterback because RPOs gave him simple reads, the line made the run real, and Pederson went for it on fourth down when others kicked. The Philly Special worked by exploiting a defensive rule (nobody covers the quarterback) at the moment the defense was most committed to the run.
  • The Sirianni-era offense ran because Hurts made the defense account for an extra runner and the line could win both the shoving match of duo and the race of outside zone. That made Barkley’s 2,000-yard season possible against the league’s light boxes. The offense held up through three play-callers, which says it rested on what stayed (the line and Hurts) more than on who called the plays.
  • The tush push edge was real but smaller than its reputation. The Eagles’ sneaks converted at about the league’s rate while they called the play far more often than anyone; counting every 1-yard snap, the edge was worth about ten extra first downs in 2023 and only a couple or fewer in 2022, 2024 and 2025. By 2025 the push itself converted below the league, though the Eagles’ 1-yard plays as a whole still matched it.
  • Super Bowl LIX was won with a four-man rush and seven in coverage (no blitzes, by Next Gen Stats’ count), which is possible only for a team whose four linemen can win one-on-one. That scheme was a consequence of draft picks, helped that night by a Kansas City line playing a guard at left tackle.
  • Roster and scheme are one decision. Develop linemen cheaply, pay to keep the good ones, spend first-round picks on pass rushers, and you can run, sneak and rush four. Everything has a price, and the Eagles paid theirs at receiver, at cornerback depth and in the age curve of a veteran line.
