def go_to(p, key, *pts, kind="path", delay=0.0, speed=None, target=None):
    """Move `key` through absolute (d, w) points, as a route, run, path, block or rush."""
    start = np.array(p._cur(key))
    rel = [tuple(np.array(q) - start) for q in pts]
    if kind == "route":
        return p.route(key, rel, frame="field", delay=delay, speed=speed)
    if kind == "block":
        return p.block(key, pts=rel, delay=delay, speed=speed)
    if kind == "rush":
        return p.rush(key, pts=rel, delay=delay, speed=speed)
    return {"path": p.path, "run": p.run}[kind](key, rel, delay=delay, speed=speed)

def clean(ax, numbers=False):
    """Drop hash ticks and painted yard numbers (they sit where receivers stand); clip all text to the window."""
    for ln in list(ax.lines):
        xs, ys = ln.get_data()
        if len(xs) == 2 and ln.get_linewidth() == 0.5 and np.hypot(xs[1] - xs[0], ys[1] - ys[0]) < 1.0:
            ln.remove()
    for t in list(ax.texts):
        if not numbers and t.get_text().isdigit() and t.get_alpha() == 0.55:
            t.remove()
        else:
            t.set_clip_on(True)

def set_tracks(p, tracks, t_end=None):
    """Hand-made keyframes {key: [(t, d, w), ...]} for some players, after all routes are added.
    gridiron's man() ends on top of the receiver and cannot be slowed by a jam, so the jammed
    receiver, the jamming corner and the defender who jumps the route are keyframed by hand."""
    kf, t0, t1 = p._compile()
    kf = dict(kf)
    for key, rows in tracks.items():
        q = p.players[key]
        rows = [(min(t0, -1.0), q.d, q.w), (0.0, q.d, q.w)] + [tuple(r_) for r_ in rows]
        kf[key] = np.array(rows, dtype=float)
    p._compiled = (kf, t0, max(t1, t_end or 0.0, *[rows[-1][0] for rows in tracks.values()]))


def ease(points, step=0.05):
    """Smooth keyframes through (t, d, w) waypoints (ease in and out on each leg)."""
    rows = []
    for (ta, da, wa), (tb, db, wb) in zip(points[:-1], points[1:]):
        for tt in np.arange(ta, tb, step):
            f = (tt - ta) / (tb - ta)
            f = f * f * (3 - 2 * f)
            rows.append((round(float(tt), 3), da + (db - da) * f, wa + (wb - wa) * f))
    rows.append(tuple(points[-1]))
    return rows


def catch_time(p, which=0):
    """When the ball arrives, the way gridiron times a pass (ball speed and the receiver's path)."""
    from gridiron.play import BALL_SPEED, _interp
    kf, _, _ = p._compile()
    kind, to, te = [e for e in p.ball_events if e[0] == "pass"][which]
    start, t_arr = p.ball_position(te), te + 0.5
    for _ in range(4):
        t_arr = te + float(np.linalg.norm(_interp(kf[to], t_arr) - start)) / BALL_SPEED
    return round(t_arr, 2)


def frame(ax, p, t, lateral, window, rings=(), ltg=False, mr=0.7):
    """Everyone at time t, with faint trails of where they have been and the ball's flight so far."""
    ylo, yhi = sorted((p.ball_y - lateral[0], p.ball_y - lateral[1]))
    fld = Field(ax=ax, orient="vertical", los=p.los, window=window, numbers=False, hashes=False,
                lateral=(ylo, yhi), to_go=p.to_go if ltg else None)
    ppy = fld.pts_per_yard()
    wl, wr = sorted(lateral)
    ring_of = dict(rings)
    for k, q in p.players.items():
        d, w = p.position(k, t)
        if not (window[0] + 0.4 < d < window[1] - 0.4 and wl + 0.4 < w < wr - 0.4):
            continue
        if t > 0:
            pts = np.array([p.position(k, tt) for tt in np.linspace(0.0, t, 40)])
            if np.ptp(pts[:, 0]) + np.ptp(pts[:, 1]) > 0.3:
                xs, ys = P(fld, p, pts[:, 0], pts[:, 1])
                ax.plot(xs, ys, color=style.OFFENSE if q.side == "off" else style.DEFENSE, lw=1.0,
                        alpha=0.5, zorder=3)
        x, y = P(fld, p, d, w)
        draw_marker(ax, float(x), float(y), q.side, q.label, ppy, ring=ring_of.get(k), r=mr)
    for kind, to, te in p.ball_events:
        if kind == "pass" and t > te:
            ts = np.linspace(te, t, 20)
            b = np.array([p.ball_position(tt) for tt in ts])
            xs, ys = P(fld, p, b[:, 0], b[:, 1])
            ax.plot(xs, ys, color=style.BALL, lw=1.4, ls=(0, (2, 1.5)), zorder=9)
            ax.add_patch(Ellipse((float(xs[-1]) - 0.9, float(ys[-1])), 1.0, 0.62, fc=style.BALL, ec="white",
                                 lw=0.5, zorder=11))
    clean(ax)
    return fld


def strip_grid(p, times, notes, lateral, window, rings=(), ncols=2, width_in=6.5, deco=None, fs=6.9, mr=0.7):
    """Print stand-in for an animation: numbered panels, each with a sentence underneath."""
    nrows = int(np.ceil(len(times) / ncols))
    pw = width_in / ncols
    ph = pw * (window[1] - window[0]) / abs(lateral[1] - lateral[0])
    fig, axes = plt.subplots(nrows, ncols, squeeze=False, figsize=(width_in, nrows * (ph + 0.85)))
    for ax in axes.flat[len(times):]:
        ax.axis("off")
    for i, (ax, t) in enumerate(zip(axes.flat, times)):
        fld = frame(ax, p, t, lateral, window, rings, mr=mr)
        if deco:
            deco(t, fld, p)
        lab = "before the snap" if t < -1e-6 else ("snap" if abs(t) < 1e-6 else f"+{t:.1f} s")
        panel_title(ax, f"{i + 1}.  {lab}")
        under_note(ax, notes[i])
    fig.tight_layout(h_pad=3.0, w_pad=0.8)
    return fig


def video(p, lateral, window, t_end, rings=(), width_in=5.6, t_start=-0.3, notes=None, times=None, deco=None,
          mr=0.7):
    """Web version of a strip: the same frame drawn every tenth of a second, the strip's notes as captions."""
    from IPython.display import HTML, display
    span_w, span_d = abs(lateral[1] - lateral[0]), window[1] - window[0]
    fig, ax = plt.subplots(figsize=(width_in, width_in * span_d / span_w + 0.5))
    ts = np.round(np.arange(t_start, t_end + 1e-9, 0.1), 2)

    def update(i):
        ax.clear()
        t = float(ts[i])
        fld = frame(ax, p, max(0.0, t), lateral, window, rings, mr=mr)
        if deco:
            deco(max(0.0, t), fld, p)
        txt = "before the snap" if t < -1e-6 else ("snap" if abs(t) < 1e-6 else f"+{t:.1f} s")
        if notes and times:
            cur = [n for tt, n in zip(times, notes) if tt <= t + 1e-6]
            if cur:
                txt += "   " + cur[-1].replace("\n", " ")
        ax.set_title(txt, fontsize=8, loc="left", color=style.INK)
        return []

    fig.tight_layout()
    anim = animation.FuncAnimation(fig, update, frames=len(ts), interval=100, blit=False)
    html = (anim.to_html5_video(embed_limit=40).replace("<video ", '<video style="max-width:100%;height:auto" ')
            if shutil.which("ffmpeg") else anim.to_jshtml(default_mode="once"))
    plt.close(fig)
    display(HTML(html))

The problem: eleven defenders can’t stop everything

Start with arithmetic. A defense has eleven players. An offense can put five receivers into a pattern, run in six or seven different gaps, throw short or deep, to either side. No arrangement of eleven people covers all of that. Every coverage you met in Part 6 has holes, and What Beats What is a catalogue of where they are. So every defense, every snap, is choosing what to be weak against.

Most defenses make that choice once, in the off-season. They pick a system, say Pete Carroll’s single-high Cover 3 or Tony Dungy’s Tampa 2, and they get very good at it. The holes are known, but the players have repeated the system thousands of times, so they play it fast and without mistakes, and the coaching is about doing the same thing better. “We’re going to do what we do” is a real philosophy, and it has won Super Bowls: 11-04 tells how Seattle’s 2013 defense won one with a system so simple that its players could play it at full speed.

Bill Belichick made the choice every week instead. That is the idea at the heart of this chapter, the game-plan defense: a defense rebuilt for each opponent, rather than run the same way every week. The staff decides what this particular offense most needs (a player, a concept, a part of the field), designs alignments and coverages to take it away, and chooses in advance what it will concede instead. The front, the coverages, even the personnel groups can change from Sunday to Sunday. The player who learned Belichick’s defense in training camp did not know, in September, what he’d be doing in December. “Our philosophy is always, we’re going to give them something a little bit different,” is how Josh Boyer, then the Patriots’ cornerbacks coach, put it after Super Bowl LIII.2

Belichick didn’t invent the idea. Coaches have always prepared for opponents, and the phrase “make them beat you left-handed” is older than he is. What he did was make it the whole method, for longer and at a higher level than anyone else: the defense had no fixed identity to protect, only a process. You met Michael Lombardi’s summary of it in 08-03: Belichick aims at what the opponent’s individual players do best, not at the offense in the abstract.3 That refinement runs through every plan in this chapter. The target is usually a person.

ImportantWhy it exists: the game-plan defense

What it solves. A defense with one system has one set of holes, and by mid-season every opponent has a tendency report that lists them. A defense that changes each week has no stable holes to scout: the opponent prepares for the defense it saw on film and meets a different one. And because the plan is aimed at this offense’s best thing, it attacks the part of the opponent that matters most rather than playing the averages.

What it costs. Three. First, learning: a new plan every week means meetings, installs and practice time spent on things that will never be used again, and players who must be smart enough to absorb them. Belichick’s teams famously valued versatile, coachable players over pure athletes for exactly this reason. (What a week of that looks like in one concrete case, the practice rep behind the Super Bowl XLIX interception, is below.) Second, execution: a defense running something new is more likely to blow a coverage than one running its thousandth rep of the same call. Third, the bill itself: every defender aimed at the best thing is taken away from somewhere else, and the plan only works if the thing it concedes really is less dangerous.

Where the habit came from

Some of this is family history. Belichick’s father, Steve Belichick, was an assistant coach and scout at the Naval Academy for more than thirty years and wrote a textbook on the subject, Football Scouting Methods (1962). As a boy in Annapolis, Bill sat with him while he broke down film of Navy’s opponents, learning to see a game as a set of habits that could be listed and attacked.4 He started in the NFL in 1975 as a $25-a-week assistant in Baltimore, coached special teams and linebackers for the Giants under Ray Perkins and then Bill Parcells, and was the Giants’ defensive coordinator from 1985 to 1990, winning two Super Bowls. He was head coach of the Cleveland Browns from 1991 to 1995 and of the New England Patriots from 2000 to 2023, where he won six more.5

That last line hides the important point. The plans in this chapter come from three different eras of football: 1990, when the threat was a no-huddle passing attack; 2001, when it was a “Greatest Show on Turf”; 2018, when it was a young coach’s play-action machine. The defenses look nothing alike. The method is identical, and it fits in three questions:

  1. What does this offense most need? Not what it does most often: what it can’t live without.
  2. What can we give it instead, that it would rather not take? The concession has to be real, and cheap.
  3. What is the bill, and who pays it? Which defender is alone, which part of the field is soft, which play are we willing to watch work?

Keep those three questions in mind; we’ll answer them for each Super Bowl in turn.

Super Bowl XXV: let Thurman Thomas run

The problem. The 1990 Buffalo Bills were the most explosive offense in football. They went 13–3, led the league with 428 points, and beat the Raiders 51–3 in the AFC Championship Game.6 Their weapon was the K-Gun, a no-huddle attack (the full story is in The 1978 Liberation and In-Game Adjustments): Jim Kelly in the shotgun, three wide receivers (Andre Reed, James Lofton and a third), a tight end who could block and catch, and Thurman Thomas, a running back who was as dangerous catching passes as running. Because the Bills didn’t huddle, the defense couldn’t substitute between plays. Whatever eleven players a defense had on the field when a drive began, it was stuck with them, and Kelly picked the play after seeing them.

You met this plan briefly in 08-03’s film room. Here is how it worked on the field.

The answer. Belichick’s plan started from the no-huddle. If you can’t substitute, you have to pick one personnel group that can survive every down, and he picked a light one: for most of the game the Giants had only two down linemen, Erik Howard and Leonard Marshall. Bill Parcells described it afterward as “a 2-4-5, or a 2-3-6” (linemen, then linebackers, then defensive backs, the same way you read a 4-3); at times there were six defensive backs. The outside linebackers, Lawrence Taylor and Carl Banks among them, stood up at the edges, where they could rush or drop.7 Then came the line that made his players think he had lost his mind. A week before the game he told them that if Thurman Thomas ran for 100 yards, the Giants would win. Banks called the idea “a collective brain fart.”

Here is the logic, in the three questions:

  1. What does Buffalo most need? Kelly’s throws, especially to Reed and Lofton, and Thomas catching the ball in space. A no-huddle offense lives on completions: they move the chains, keep the tempo fast and keep a tired defense stuck on the field. Big completions are how it scores quickly.
  2. What can the Giants give instead? The run. A run is one fewer pass. And a run that ends in bounds keeps the clock moving (only an incompletion or a run out of bounds stops it). That suited a Giants team that planned to win the game with its own offense, a ball-control attack that would keep Kelly on the bench.
  3. What’s the bill? Thomas, running against a front with two linemen, would gain yards. Belichick bet that he couldn’t gain them fast enough, and he paid the bill on purpose.

And the passes the Giants did allow would be punished. Every linebacker and defensive back was responsible for hitting any Bill who caught the ball. “As soon as [Andre] Reed got the ball, he needed a helmet in his ribs,” Banks remembered.8

So why did the plan hold? Because of the other half of it. A great runner on 15 carries is a smaller problem than a great passer on 50 throws. CautionCommon misconception

“Take away their best thing” does not mean stop their best thing at any cost, or stop everything. It means: choose which of the offense’s weapons you can live with, and point everything else at the one you can’t. A game-plan defense that “gives up” 135 rushing yards and wins has not failed; it has paid the bill it chose. When you judge a defensive plan, ask what it was trying to concede before you count what it gave up.

Super Bowl XXXVI: two words, Marshall Faulk

The problem. Eleven years later Belichick was a head coach, and the opponent was even better. The 2001 St. Louis Rams, “the Greatest Show on Turf,” scored 503 points and were 14-point favorites (the betting markets expected them to win by two touchdowns). Kurt Warner threw to Isaac Bruce, Torry Holt and Az-Zahir Hakim, and the whole machine turned on Marshall Faulk, a running back who was also one of the best receivers in football, the league’s offensive player of the year three seasons running.10 The Patriots had already played them, in November, and lost 24–17. In that game Belichick and his defensive coordinator, Romeo Crennel, had put a linebacker on Faulk. “That didn’t work out for us,” Crennel said later, laughing.11

The answer. Four years after the game, as a guest on ESPN’s pregame show for Super Bowl XL, Belichick explained the plan in one sentence: “In Super Bowl XXXVI, our game plan was defined by two words: Marshall Faulk.”12 Every snap began with one question, where is he?

  • Faulk directly behind Warner: expect a run, and play run defense.
  • Faulk offset to one side, or split out: expect a pass. “He’s not behind the quarterback; we’re thinking jam the receivers.” Press the wide receivers at the line to throw off the timing of Warner’s throws, which in the Rams’ system went to spots before the receiver had turned. And that “triggered the defensive ends, [Willie] McGinest, trying to hit Faulk and take him out of the game” as he released.
  • When we blitz, blitz over Faulk, from his side, so that he has to stay in and block: “then our blitzers would get to the quarterback and we could force the ball out in a hurry.”

And behind the jams? Not the man coverage New England had played in November. “We felt they would be ready for man coverage, since we gave them a lot of man the first time,” Crennel said, “so as an alternative to the man, we wanted to go with the cover two, which allowed us to be physical with the receivers.” So the usual call was Cover 2: the corners jam the outside receivers and then sit in the flats underneath them, two safeties split the deep field, and the rest drop into short zones, with one defensive back, instead of a linebacker, matched on Faulk wherever he went. It wasn’t pure zone; Martz said afterward that New England played “a lot of two-deep type coverages, a lot of man-press kind of things.” Mixing the two was part of the point. “It was in the game plan for us to hit Faulk no matter where he was on the field,” McGinest said.13

The Patriots played most of the night with five, six or even seven defensive backs: by one count, nickel or dime personnel on 54 of the Rams’ 69 snaps. That was the concession, a light defense daring the Rams’ coach, Mike Martz, to run.14

In the first meeting they had blitzed and played man; Crennel’s staff guessed the Rams would be ready for man again, so they played mostly Cover 2 with physical corners instead, and kept just one man-to-man job: the defensive back on Faulk.

Super Bowl LIII: take away the picture

The problem. Seventeen years after the first Rams Super Bowl, the second one. The 2018 Rams went 13–3, scored 32.9 points a game and were second in the league in EPA per play. Their offense, Sean McVay’s version of the Shanahan system (the Shanahan case study has the family tree), was a magic trick built on sameness. The Rams used 11 personnel (one back, one tight end, three receivers) on 93% of their snaps in 2018, by far the most in the league, often from tight formations with the receivers close to the line. From that one look they ran outside zone with Todd Gurley, and play-action passes that began exactly like outside zone. A defense couldn’t tell run from pass by the personnel, the formation, or the first step. And Goff had McVay’s voice in his helmet radio until 15 seconds were left on the play clock, helping him read what the defense showed.

New England’s staff found one more thing in the film: the Rams’ pass game was full of man-beaters, crossing routes and rubs designed to beat man-to-man coverage. And man-to-man was what New England played. In 2018 the Patriots played man coverage on 61% of charted opponent dropbacks, the highest rate in the league (the median defense was at 33%).16

The answer. The plan, put together by Belichick and Brian Flores (who called the defense that season), did two things at once.17

  • Up front, put the run out of business. The Patriots walked two linebackers up to the line of scrimmage, effectively playing a six-man defensive line on early downs. Outside zone runs on combo blocks: two linemen double-team a defender, and one of them climbs off to the linebacker behind. That only works if some lineman is uncovered, with nobody lined up on him. Six on the line against five linemen and a tight end puts a defender on every blocker, so there are no double teams and no free lineman to climb to the second level. Each blocker has to reach his man alone, the edges are set hard on both sides, and the only defenders left to fill (the strong safety and the nickel) come downhill late, with nobody blocking them. Stop the run on first and second down, and play-action stops working, because nobody believes the fake on third-and-8. Gurley finished with 35 yards on 10 carries.
  • Behind it, take away the picture. After a season of man coverage, New England opened in zone. It mixed one-deep and two-deep shells, man and zone, so that what Goff saw before the snap was a poor guide to what he’d face after it. Midweek the coaches gave cornerback Jonathan Jones, usually the slot defender, a safety’s job: he lined up close to the line like a blitzer, Goff kept expecting him to come, and he kept dropping into the middle of the field. Up front, the Patriots varied which of the six rushed and which dropped, and ran stunts (rushers looping around each other) they hadn’t shown on film. Stephon Gilmore, the best cornerback, covered the Rams’ fastest receiver, Brandin Cooks.

Of Goff’s 41 charted dropbacks, New England rushed four on 14, five on 19, six on 3 and three on 5, all from the same crowded pre-snap picture, so Goff couldn’t count the rush until the ball was snapped. (NFL Next Gen Stats, which counts differently, had the Patriots blitzing on half their snaps and pressuring Goff 14 times.)18 The personnel bar fits the same story once you split it by down. On first and second down New England was mostly in base or nickel, and that is where the six-man line came from, with linebackers walking up; six or more defensive backs were on the field for only 15 of 46 early-down snaps. On third and fourth down it was six or seven defensive backs on 11 of 13.

Five rushers with zone behind them is a fire zone (three defenders underneath, three deep), a standard and sound call taught in Zone Blitz and Simulated Pressure. Its bill is the spacing: with three defenders underneath instead of four or five, the windows between them (the hook areas and the seams) are wider, and a quarterback who reads the coverage right has someone open, fast. The Patriots’ bet was that Goff, guessing, would not read it right in time. Once he very nearly did. Late in the third quarter, with New England ahead 3–0 and the Rams at the Patriots’ 29, Cooks was alone deep in the end zone, and cornerback Jason McCourty sprinted across the field to knock the ball away just before it arrived.19 That is the bill almost coming due. Goff went 19 of 38 for 229 yards with an interception and was sacked four times; the Rams converted 3 of 13 third downs, gained 260 yards and scored 3 points, their fewest under McVay (matched only once since, at Green Bay in 2023).20

The ending turned the plan’s own pattern around. With 4:17 left, the Rams trailing 10–3 and driving, New England, having played zone and disguised its rush all night, sent a six-man blitz with no deep safety at all: a Cover 0 call, all man, that left Gilmore alone on Cooks. After a night of learning to look for zone, Goff now faced the one thing he had stopped expecting, the man coverage New England had played all season. Under pressure, he threw it up, and Gilmore intercepted it at the Patriots’ 3. That is why the snap sums up the plan: the plan was never “play zone”, it was “don’t be where he expects you”. A Stephen Gostkowski field goal made it 13–3. Sony Michel’s 2-yard run had scored the game’s only touchdown.

Super Bowl XLIX: the last play

The most famous snap of the Belichick era happened on the other side of the method: not a plan’s grand design, but one detail of one week’s preparation, met by a decision under a running clock.

The situation. February 1, 2015, Glendale, Arizona. New England led Seattle 28–24. After a juggling 33-yard catch by Jermaine Kearse, Seattle had first-and-goal at the Patriots’ 5. Marshawn Lynch, the most punishing runner in the league, ran for four yards to the 1 with 1:06 left. Seattle had one timeout; New England had two. And then, for about 40 seconds, nobody stopped the clock.

Two coaches and a clock. Pete Carroll’s reasoning, which he explained afterward, went like this. Seattle had one timeout and needed one yard, and it wanted three chances at it. If it ran on second down and was stopped, it would have to burn its last timeout and then could run once more, maybe twice, with the clock nearly gone. A pass on second down, if incomplete, stops the clock for free, keeping the timeout for later. And New England had sent its goal-line defense onto the field against Seattle’s three wide receivers. A goal-line defense is built to stop one yard of running: eight big linemen and linebackers, and only three defensive backs. Against Seattle’s lighter run-blocking group (three receivers instead of extra tight ends), that is a bad match for a run and a good one for a throw: “It’s not the right matchup for us to run the football, so on second down, we throw the ball,” Carroll said, with the plan of running on third and fourth down.22 Belichick, for his part, let the clock run instead of calling a timeout, which would have saved time for Tom Brady if Seattle scored. He said later that he had considered it, but that the Patriots’ goal-line defense had come on at about the same time as Seattle’s three receivers, and “in the flow of the game, that we were OK with where we were.”23 On NFL Network’s later documentary, he recalled looking across at the Seattle sideline, sensing hesitation, and telling his staff to stay in goal-line. The ball was snapped with 26 seconds left.

The preparation. New England’s coaches had studied Seattle’s goal-line passing game during the week, and one look stood out: two receivers stacked, one behind the other, close to the formation, with the front man running into the defender of the back man (a rub route) so that the back man could slip underneath on a quick slant. The defensive rule for it was simple, and more aggressive than the textbook answer. The usual man-coverage answer to a stack is a banjo (Man Coverage): the two defenders wait for the release and switch, one taking whoever breaks in, the other whoever breaks out. New England’s rule didn’t wait. The defender on the front man jams him at the line, so the rub never forms. The defender on the back man doesn’t give ground and jumps the first inward break. The research behind it came from Ernie Adams, Belichick’s long-time football research director, who had studied Seattle’s goal-line formations. In practice that week, the scout team ran the play against a rookie cornerback, Malcolm Butler, who hadn’t even been sure of a place in the goal-line package, and beat him for a touchdown. Belichick told him: if you see that formation, you can’t give ground.24

NoteGo deeper: was the throw a blunder?

“Worst call in Super Bowl history” was the instant verdict, and it’s worth testing, because it is the kind of claim data can check. In the fourteen seasons before that Super Bowl (2001–2014, nflverse play-by-play, all games), teams snapped the ball from their opponent’s 1-yard line 4,851 times. Runs scored a touchdown 55% of the time and were fumbled away 1.8% of the time; passes scored 51% of the time and were intercepted 2.9% of the time. So yes, a pass from the 1 was a little worse than a run on both counts. But not by much, and the averages hide what Carroll was buying: a pass that falls incomplete stops the clock and keeps his last timeout for later, which made three tries at the end zone possible instead of two. Put the league rates into his two menus. Two runs score at least once about 80% of the time. A pass first, then (if it falls incomplete) two runs, scores about 88% of the time, even after counting the interceptions. That’s a rough model (every try treated as an average snap from the 1, against defenses that weren’t all goal-line packages), but it turns “defensible” into arithmetic: the third try is worth more than the pass costs. The decision was defensible; the outcome was the kind that happens about once in 34 throws from there, and it happened because the defense had prepared for that exact snap.

Belichick’s own decision deserves the same test. Calling a timeout after Lynch’s run would have stopped the clock at about a minute, so if Seattle scored, Brady would have had time to answer. Letting the clock run kept his goal-line personnel matched against Seattle’s three receivers and kept Seattle hurrying. Analysts still argue about both choices. That’s the honest lesson of situational football: good decisions and good outcomes are different things, and the rehearsed detail (a jam, a jumped slant) can matter more than either coach’s clock math.

The chameleon offense

So far this chapter has been about defense, because that’s where game-plan football started. But Belichick’s offenses were built on the same idea, pointed the other way: be whatever this opponent can least handle this week. Dynasties and Counterpunches tells the three big shape changes as history: the 2007 spread (Randy Moss outside, Wes Welker in the slot, 589 points, then a record), the 2010–2012 two-tight-end offense with Rob Gronkowski and Aaron Hernandez, and the language underneath that made both possible, the Erhardt-Perkins system, in which one word names a concept that can be run from any formation and any personnel, so the offense can change its clothes without learning new plays.

The 2007 version is worth a picture in words, because it shows how far one season’s shape could move. That year the Patriots added Randy Moss, a deep threat no single corner could carry, and Wes Welker, a quick slot receiver, and put Tom Brady in the shotgun with three or four receivers spread across the field. They snapped from the shotgun on 50% of their plays, the most in the league (the median team was at 26%), and Brady threw 50 touchdown passes, then a record.26 Spread the defense out, and the safeties have to choose between Moss deep and Welker underneath. Three years later the same coach and quarterback won with two tight ends and a much tighter shape. Here we look at the shape-shifting as a game plan, week to week.

Gronkowski and the 12-personnel dilemma

The two-tight-end offense of 2011 needed two rare players. After Hernandez was gone, from 2013, the Patriots showed the dilemma survives with one: Rob Gronkowski, 6-foot-6 and about 265 pounds, a tight end who could block a defensive end and outrun a linebacker. With him and a second, ordinary blocking tight end on the field (12 personnel: one back, two tight ends, two receivers), the defense has to choose its personnel before it sees the formation, and every choice leaves a hole.

Panel D’s first three rows are the whole dilemma, and it’s the same order-of-events trick as in 11-04: the defense commits its personnel before it sees the formation, so an offense with a player who is good at two jobs gets to pick which job he does after the defense has chosen. Panel C is the defense’s way out, and it’s pure Belichick logic turned against him: take away the best thing (Gronkowski) and pay the bill somewhere else. That’s why the Patriots kept a receiver like Julian Edelman, quick enough to win against a linebacker or a nickel back on an option route, next to Gronkowski for most of the decade. The bracket on one created the matchup for the other.

The last row of panel D is the way out that doesn’t cost a second defender: big nickel, a third safety in place of a linebacker or a slot corner, big enough to take on a tight end’s block and fast enough to run with him. It dissolves the dilemma rather than solving it, because whatever formation the offense picks, the same player is on the field. Its price is that a safety is a compromise at both jobs, a little lighter than a linebacker against the run and a little slower than a corner in coverage, and it only works if you have three safeties good enough to play. That’s the “versatile players” lesson again, from the defense’s side.

Or look at the end of the 2018 season. The Patriots had been a pass-first team, even though in the regular season they used 21 personnel (two backs, with fullback James Develin) on 29% of their snaps, second-most in the league. That isn’t a contradiction. Two backs on the field make a defense think run and load up for it, which is exactly when a play-action pass works best, and New England threw on 44% of its 21-personnel snaps, about the league rate for that grouping (44%). NoteMadden vs. real life

In Madden, a team’s offense is its playbook: pick “Spread” or “Pro Style” and that’s who you are. Real NFL offenses carry a far bigger playbook than any one game uses, and each week’s game plan selects a slice of it for this opponent. The Patriots took that further than anyone: their system was a language (Erhardt-Perkins concepts that work from any formation) rather than a list of plays, so the slice could be almost anything. Fifty-four passes one week, thirty-nine runs the next, from the same quarterback and the same play-caller, Josh McDaniels, is not a team with an identity crisis. It’s a team whose identity is the game plan.

Situational mastery

The third leg of Belichick’s football was the one fans noticed least until the last two minutes: treating every situation (the end of a half, a lead to protect, fourth-and-short, a strange rule) as something to be studied and rehearsed in advance, like an opponent. Each of the plays below is told in full in The Clock, the Two-Minute Drill, and Fourth Down or the chapter linked; here they’re one line each, because the point is the pattern, not the stories.

  • Super Bowl XXXVI, 1:21 left, no timeouts, tied, own 17: John Madden said kneel; Belichick and Brady threw short against the Rams’ prevent-style shell and won it on a 48-yard field goal as time expired.
  • November 2003 at Denver, down a point at their own 1: an intentional safety, two points given away so the next kick would be a free kick from the 20 instead of a punt from the end zone; the field position won the game.
  • Super Bowl XLVI, leading 17–15 with about a minute left: let Ahmad Bradshaw score, so Brady gets the ball back with time. (They lost anyway; the logic was right.)
  • The 2009 fourth-and-2 at Indianapolis (11-04): the analytics said go, the outcome didn’t cooperate.
  • The rulebook as a playbook: the eligibility-reporting formations against Baltimore in January 2015, which the league changed the rule for 2015 (02-02).
  • Super Bowl XLIX, above: letting the clock run to keep the goal-line personnel he wanted.

Is “situational mastery” real, or is it a story told about a team with a great quarterback? Some of it is measurable.29 In 2001–2019 regular seasons the Patriots outscored opponents by 0.84 points per game in the last two minutes of the first half alone, second in the league over those nineteen seasons (behind Pittsburgh). That’s the “double score” habit: points just before halftime, then, if you deferred the opening coin toss, the ball again just after it. And in one-score games (decided by eight points or fewer) they went 86–43, a 0.667 winning percentage, the best in the league over the same seasons. Close-game records carry a lot of luck, and some of New England’s came from having Tom Brady. But nineteen years of them is a long streak of luck.

Read it as three separate claims, each with its own caveat.

  • Opponents scored less against New England than they did against everyone else, by 3.1 points a game, the biggest gap in the league (Baltimore and Pittsburgh, the era’s great system defenses, are right behind). That’s the plainest measure of a good defense, adjusted for who it played; it doesn’t by itself prove game-plan defense. An offense that holds the ball for long drives also lowers the other team’s scoring.
  • The opponent’s best receiver got less of the ball. Over these seasons a team’s top receiver drew about 24% of its targets. One quirk first: against the typical defense his share rose a little in this measure (by 0.4 points), because games in which he wasn’t targeted at all drop out of the data, and play-by-play can’t tell a shutout from a game he missed. So the fair comparison is with the median defense, not with zero. Against New England his share fell 1.7 points below what the typical defense allowed, the largest gap of any defense and nearly twice the next team’s (TEN, 0.9). On a team throwing about 29 targeted passes a game that’s small, about 0.5 targets a game, but it is exactly the fingerprint Lombardi’s line predicts: a defense that aims at the opponent’s individual players should push the ball away from the best of them, and this one did, for nineteen years.
  • The end of the half belonged to the Patriots. The situational habit shows up as almost a point a game in a two-minute window, second only to Pittsburgh.

None of these is enormous, and none should be: they’re averages over hundreds of games against every kind of opponent. What’s striking is that one team sits at or near the end of all three lines.

Was it all Tom Brady? The defensive half of the fingerprint can be tested, because Brady left after 2019 and Belichick coached four more seasons. By the same opponent-adjusted points measure, New England’s defense ranked 5, 2, 6 and 11 in the league from 2020 to 2023 (1 is best), while the team reached the playoffs only once in those four seasons.30 The game-plan defense kept working without a great offense; what disappeared was the offense that turned a good defense into a dynasty.

How opponents responded

Every scheme in this course creates its own counter, and Belichick’s had several.

Make the plan irrelevant with four rushers. The first two teams to beat Brady and Belichick in a Super Bowl, the 2007 and 2011 Giants, did it the way 11-04 describes: a four-man rush good enough to reach Brady without blitzing, with linemen pushing up the middle where a pocket passer can’t escape. A game plan on offense can hide a weakness from a coordinator, but not from four defensive linemen who win their matchups on every play. The 2015 Denver defense did it again in the AFC Championship Game, hitting Brady 20 times, the most on any quarterback that season.31

Have too many best things. A plan that takes one thing away works best against an offense with one thing. The 2017 Eagles in Super Bowl LII spread the ball among receivers, tight ends and backs, and their backup quarterback played the game of his life; there was no single target to take away. The great quarterbacks who beat New England could beat a defense “left-handed”, throwing to whoever was left: Peyton Manning did it in three AFC Championship Games (after the 2006, 2013 and 2015 seasons).32

Take away their best thing. Defenses turned the method around: bracket Gronkowski, double Welker, press Edelman, and dare the second and third options to win. The Patriots’ answer, again and again, was to change which player was the best thing that week.

Hire his assistants. The most direct response was to hire the people who knew how the plans were built. Romeo Crennel, Nick Saban (the Browns’ defensive coordinator under Belichick), Eric Mangini, Charlie Weis, Josh McDaniels, Bill O’Brien, Matt Patricia, Brian Flores and Joe Judge all became head coaches in the NFL or in college, and so did his former linebacker Mike Vrabel. Their records were mixed, which is itself a lesson: the plans came from a process (the film work, the weekly installs, the players chosen to absorb them), and a coordinator who leaves takes the habits but not the building. The sharpest test was Jerod Mayo, a Belichick linebacker and then his linebackers coach, who succeeded him in New England in 2024, in the same building, and was let go after one 4–13 season.33

What survives

Almost all of it, and much of it has become ordinary.

  • Every defense game-plans now. Weekly, opponent-specific defense was Belichick’s signature in 2001. By 2026 it is simply how the league works: coordinators build a disguise package and a set of matchup rules for each opponent, and the rise of two-high, rotate-at-the-snap defenses (The Modern Era) is the LIII plan as a league-wide style: show one picture, play another. The clearest recent example sits on the other sideline from New England. In Super Bowl LX in February 2026, Mike Macdonald’s Seattle defense sacked Drake Maye six times and beat the Patriots 29–13. New England’s head coach that night was Mike Vrabel, a linebacker on Belichick’s first three Super Bowl teams, with Josh McDaniels calling the offense.34 (Macdonald’s defense gets its own case study in From Two-High to Macdonald.)
  • Versatile players over specialists. Defensive backs who can play corner, safety and slot; linebackers who rush and drop; tight ends who block and run routes. A game plan is only as flexible as the players who have to learn it, and Belichick’s roster-building made that the priority it is now.
  • Situational football as a subject. Teams now carry analysts for fourth-down, two-point and clock decisions (see Win Probability and Fourth Downs), and practice end-of-half and end-of-game situations weekly.
  • Concept language. The Erhardt-Perkins way of naming plays (concepts, not formations) survives in many offenses, including McDaniels’s in New England in 2026.

Belichick left New England after the 2023 season. He became head coach at the University of North Carolina for the 2025 season, which ended 4–8, and in the January 2026 vote, his first year of eligibility, he was not elected to the Pro Football Hall of Fame, a result that drew wide criticism.35 One of his game plans is there already: the handwritten Super Bowl XXV sheets.

TipWatch for it on Sunday: the take-away test and the situation test

Pick one game. Before kickoff, write down the answer to Belichick’s first question for each offense: what does it most need? A player (its star receiver, its running back), a concept (play-action off the run, the quick game), or a part of the field (the deep middle, the perimeter run)?

Early in the game, watch the defense on that team’s first few snaps:

  • Where is the star? Is a second defender near him (a safety over the top, a corner outside, a linebacker underneath)? That’s a bracket, a take-away.
  • How many defenders stand on the line before the snap, and how many actually rush? A defense with six on the line and four rushers is hiding something.
  • Does the coverage look the same before and after the snap? If the picture changes, the defense is disguising: it’s aiming at the quarterback’s pre-snap read.

At halftime and at the end, check your note. Did the star get his usual share of targets or carries? Who got the ball instead? Was the thing the defense conceded (the run? the short throw? the other receiver?) the thing that beat it, or the thing it could live with? A defense that “gave up” 120 rushing yards and won may have been paying a bill it chose on Wednesday.

The situation test, for the head coach rather than the defense:

  • The coin toss. Did the winner defer (take the ball to start the second half)? Write it down; it matters at 0:58.
  • The last two minutes of the first half. With the ball, one or two timeouts and a minute or more, does the team attack (short throws, out of bounds, timeouts after completions) or kneel? If it scores and deferred, watch for the double score right after halftime.
  • The clock against it. When the opponent is driving late, does the coach use timeouts to save time for his own offense, or let the clock run? Either can be right; ask what he is buying.
  • Fourth-and-short in plus territory. Go or kick? Compare with the team’s habit (the go-rate charts in 10-04).

TipAnswer

A run from the pass look: a draw or an inside run, or a screen to the back that uses the end’s crash against him. Every key a defense uses is a tendency the offense can scout, and once the offense knows the defense trusts it, the key becomes a lever (the tendency breaker of Constraint Theory). Look at what the key has bought the defense: six defensive backs and one linebacker (M), corners pressed on the outside, an end flying upfield. A screen to his side lets the crashing end run himself out of the play.

Two notes. On second-and-6 a 4-yard run is a fine result, so the offense isn’t risking much by testing the key. And the defense’s coaches know this too. A good game-plan defense keeps the key for most of the game and then shows the key while playing something else, a base front against the offset back, say, on the down it expects the tendency breaker. TipAnswer

Run right, at the nickel back, with the star tight end blocking him. This is the dilemma from Figure 60.6 with tempo added. First down was the question: what do you do when our tight end splits out? Many coaches would kneel and take the tie into halftime. Where are the soft spots in this shell, how many completions would it take to reach field-goal range, and would you attack or kneel?

TipAnswer

Attack, carefully: throw into the gap between the two layers of the soft shell, get out of bounds or use a timeout after each completion, and aim for field-goal range. That’s what Belichick’s teams usually did. The arithmetic is the “double score” from The Clock, the Two-Minute Drill, and Fourth Down. Fifty-eight seconds and two timeouts is enough for four to six plays. Read the shell: the underneath defenders sit about six yards deep and the safeties about eighteen, and the corners are eight yards off. That leaves a band from roughly 10 to 16 yards deep between the two layers, widest outside the hashes in front of the corners, where an out or a comeback can also stop the clock by going out of bounds. Each of those throws is worth 8 to 12 yards, and from your own 30 you need about 35 yards to reach the opponent’s 35, a long but makeable field goal: three or four completions. Then, because this offense deferred the coin toss, it gets the ball again to start the second half. Two possessions in a row, separated only by halftime, can turn a tie into a 10-point lead in a few minutes of game time.

The risk is a turnover deep in your own territory, which is why the throws are short, the sideline is used, and nothing goes up for grabs; if the first two plays go nowhere, the right call becomes the kneel. The Patriots made this sequence a habit. In 2001–2019 they outscored opponents by 0.84 points a game in the last two minutes of the first half, second in the league (Figure 63.6). It’s the same thinking as Super Bowl XXXVI’s last drive, from your own 17 with no timeouts: a situation rehearsed in practice is a situation you can attack.

Takeaways

  • A game-plan defense chooses its weakness every week. Eleven defenders can’t stop everything, so Belichick’s defenses asked what this offense most needed, took that away, and named in advance what they’d concede. The concession is part of the plan, not a failure of it.
  • The target is usually a person. Kelly’s passing (and Thomas as a receiver) in Super Bowl XXV, Marshall Faulk in XXXVI, Goff’s pre-snap read in LIII. In 2001–2019 data, the opponent’s top receiver lost more of his share of the targets against New England than against any other defense.
  • Each plan looked nothing like the last. Two linemen and six defensive backs; a defense keyed on one running back’s alignment; six on the line with zone behind after a season of man. What repeated was the method: three questions, asked fresh each week.
  • The offense was a chameleon for the same reason. A concept language that works from any formation let the Patriots throw 54 times one playoff week and run 39 times the next, and turn Gronkowski into a personnel dilemma the defense had to solve before seeing the formation.
  • Situations were rehearsed like opponents. End-of-half drives, intentional safeties, letting a team score, the eligibility rules, the clock at the end of Super Bowl XLIX: the Patriots treated each as something to prepare, and the close-game and end-of-half numbers show it.
  • Super Bowl XLIX was decided by preparation, not by the call. A defensible pass met a jam and a jumped slant that New England had rehearsed that week against that formation.
