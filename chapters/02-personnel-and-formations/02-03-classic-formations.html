style.apply()

RB_RING = style.THIRD          # aqua ring: a running back of any kind (tailback, fullback, halfback, wingback)
TE_RING = style.LINE_TO_GAIN   # gold ring: a tight end
THREAT = "#0f7a55"             # darker aqua for "this back could go here" arrows (safe for text)

UC, FB_D, TB_D = -1.9, -4.5, -7.0   # quarterback under center, fullback and tailback depths (yards)

DL_LABELS = {"SDE": "E", "WDE": "E", "SDT": "T", "WDT": "T"}


def line5(split=OL_SPLIT):
    """The five offensive linemen; only the center is lettered."""
    return [Player(k, k, "off", LINE_D, w, label=("C" if k == "C" else ""))
            for k, w in [("LT", -2 * split), ("LG", -split), ("C", 0.0), ("RG", split), ("RT", 2 * split)]]


def P(key, pos, d, w, label=None):
    """An offensive skill player at (d, w): d = yards from the ball (negative = backfield),
    w = yards to the offense's right."""
    return Player(key, pos, "off", d, w, label=label or key)


def D(key, pos, d, w, label=None):
    return Player(key, pos, "def", d, w, label=label)


def ring_by_body(p, skip=()):
    """Ring every back (aqua) and tight end (gold), as in the personnel chapter: count rings = personnel."""
    for k, pl in p.players.items():
        if k in skip or pl.side != "off":
            continue
        if pl.pos in ("RB", "FB"):
            p.highlight(k, color=RB_RING)
        elif pl.pos == "TE":
            p.highlight(k, color=TE_RING)
    return p


def hide_yard_numbers(ax):
    """Drop the faint painted field numbers (they sit where receivers stand in zoomed plates)."""
    for t in list(ax.texts):
        if t.get_text().isdigit() and t.get_alpha() is not None and abs(t.get_alpha() - 0.55) < 1e-6:
            t.remove()

# The eleven classic formations, strength to the offense's right. Each is legal: seven men on the
# line (five linemen plus the two ends), only the ends of the line and the backs eligible.
# Letters: R = the running back (tailback), F = fullback, H = halfback (a third back), A/B = the
# flexbone's own back letters, W = the Wing-T wingback; Y and U = tight ends (in the goal-line set the
# third tight end, at the wing, is H, the H-back); X and Z = wide receivers; H in the singleback = slot.
# name: (typical personnel, one-line description, skill players)

SPLITS = {"Goal line": GL_SPLIT}


def players_of(name, mirror=False):
    pl = line5(SPLITS.get(name, OL_SPLIT)) + [q.moved() for q in FORMS[name][2]]
    return [q.mirrored() for q in pl] if mirror else pl


def check_legal(players):
    """Seven on the line, and every eligible receiver on the line is at an end of it."""
    off = [q for q in players if q.side == "off"]
    on = sorted([q for q in off if q.d > -1.0], key=lambda q: q.w)
    assert len(on) == 7, [q.key for q in on]
    for q in on[1:-1]:
        assert q.pos not in ("WR", "TE", "RB", "FB"), f"covered receiver {q.key}"


for _name in FORMS:
    check_legal(players_of(_name))


def plate(ax, name, lateral=(-16.5, 16.5), window=(-9.0, 2.8), mirror=False, sub=True, title=None):
    """One atlas plate: the formation alone, backs ringed aqua and tight ends gold."""
    code, desc, _ = FORMS[name]
    p = ring_by_body(Play(players_of(name, mirror), None, los=35, to_go=None))
    fld = p.draw(ax=ax, lateral=lateral, window=window, title=False)
    hide_yard_numbers(ax)
    ax.text(0.0, 1.03, title or name, transform=ax.transAxes, ha="left", va="bottom", fontsize=10,
            fontweight="bold", color=style.INK)
    if sub:
        ax.text(1.0, 1.03, f"{code} personnel · {desc}", transform=ax.transAxes, ha="right",
                va="bottom", fontsize=7.2, color=style.INK_2)
    return p, fld


def ring_legend(fig, y=0.0, ncol=3, extra=()):
    h = [Line2D([], [], marker="o", ls="", ms=9, markerfacecolor=style.OFFENSE, markeredgecolor=RB_RING,
                markeredgewidth=2.2, label="aqua ring: a running back"),
         Line2D([], [], marker="o", ls="", ms=9, markerfacecolor=style.OFFENSE, markeredgecolor=TE_RING,
                markeredgewidth=2.2, label="gold ring: a tight end"),
         Line2D([], [], marker="o", ls="", ms=9, markerfacecolor=style.OFFENSE, markeredgecolor="white",
                label="no ring: receiver, QB (Q), linemen")]
    h += list(extra)
    fig.legend(handles=h, loc="lower center", bbox_to_anchor=(0.5, y), ncol=ncol, fontsize=7.3,
               frameon=False, handletextpad=0.3, columnspacing=1.1)


def plate_pair(a, b, mirror=(False, False), height=2.05):
    fig, axes = plt.subplots(1, 2, figsize=(6.6, height))
    out = [plate(ax, n, mirror=m) for ax, n, m in zip(axes, (a, b), mirror)]
    fig.subplots_adjust(left=0.01, right=0.99, top=0.86, bottom=0.16, wspace=0.05)
    ring_legend(fig)
    return fig, out


def XY(fld, p, d, w):
    return fld.xy(*p.to_abs(d, w))


def label(fld, p, text, at, target=None, ha="center", fs=7.3, color=style.INK, va="center"):
    """A short boxed label at (d, w); with `target`, a thin leader line to that spot."""
    x, y = XY(fld, p, *at)
    box = dict(boxstyle="round,pad=0.22", fc=style.SURFACE, ec=style.BASELINE, lw=0.6)
    if target is None:
        fld.ax.text(x, y, text, fontsize=fs, ha=ha, va=va, color=color, zorder=12, bbox=box)
    else:
        tx, ty = XY(fld, p, *target)
        fld.ax.annotate(text, xy=(tx, ty), xytext=(x, y), fontsize=fs, ha=ha, va=va, color=color,
                        zorder=12, bbox=box,
                        arrowprops=dict(arrowstyle="-", color=style.INK_2, lw=0.7, shrinkA=0, shrinkB=7))


def threat(fld, p, pts, color=THREAT, lw=1.7, text=None, text_at=None, fs=7.0, ls=(0, (3, 2))):
    """A dashed 'this back could go here' arrow through absolute (d, w) points."""
    xs, ys = XY(fld, p, np.array([q[0] for q in pts]), np.array([q[1] for q in pts]))
    fld.ax.plot(xs[:-1], ys[:-1], color=color, lw=lw, ls=ls, zorder=6, solid_capstyle="round")
    fld.ax.plot(xs[-2:], ys[-2:], color=color, lw=lw, ls=ls, zorder=6)
    fld.ax.add_patch(FancyArrowPatch((xs[-2] + 0.75 * (xs[-1] - xs[-2]), ys[-2] + 0.75 * (ys[-1] - ys[-2])),
                                     (xs[-1], ys[-1]), arrowstyle="-|>", mutation_scale=10, color=color,
                                     lw=lw, zorder=6))
    if text:
        x, y = XY(fld, p, *(text_at or pts[-1]))
        fld.ax.text(x, y, text, fontsize=fs, color=color, fontweight="bold", ha="center", va="center",
                    zorder=9, bbox=dict(boxstyle="round,pad=0.18", fc=style.SURFACE, ec="none", alpha=0.92))


def base43(off, shell="two_high", fix=None):
    """A base 4-3 (four linemen, three linebackers, four defensive backs) aligned to `off`."""
    dfn = defense("4-3_over", shell, off)
    fix = fix or {}
    dfn = [q.moved(**fix[q.key]) if q.key in fix else q for q in dfn]
    return [q.moved(label=DL_LABELS.get(q.key, q.label)) for q in dfn]


# --- frame strips (PDF) and video (web), cropped to the box so the markers stay legible --------------
# gridiron's frame_strip fits all 22 players and has no rings; these panels zoom on the box, ring the
# backs the reader should follow, and put a one-line note under each frame.
MARK_R = 0.62


def pdf_build():
    return os.environ.get("RTG_FORMAT", "").lower() in ("pdf", "typst", "print")


def tracking(play):
    df = play.to_tracking()
    df["_id"] = df["displayName"].astype(str)
    return df


def panel(ax, play, df, t, lateral, window, rings=()):
    """One moment of a simulated play: markers, trails of on-screen players, rings on chosen players."""
    tmap = df.groupby("frameId")["time"].first()
    fid = int((tmap - t).abs().idxmin())
    ylo, yhi = sorted((play.ball_y - lateral[0], play.ball_y - lateral[1]))
    fld = Field(ax=ax, orient="vertical", los=play.los, window=window, numbers=False, hashes=False,
                lateral=(ylo, yhi), to_go=play.to_go)
    cur = df[df["frameId"] == fid]
    shown = cur[cur["x"].between(play.los + window[0] + 0.8, play.los + window[1] - 0.8)
                & cur["y"].between(ylo + 0.8, yhi - 0.8)]
    ids = set(shown["_id"])
    trails = df[(df["frameId"] <= fid) & df["_id"].isin(ids) & (df["side"] != "ball")]
    for (_, side), g in trails.groupby(["_id", "side"]):
        xs, ys = fld.xy(g["x"].values, g["y"].values)
        ax.plot(xs, ys, color=style.OFFENSE if side == "off" else style.DEFENSE, lw=0.9, alpha=0.4, zorder=3)
    ppy = fld.pts_per_yard()
    for _, r in shown.iterrows():
        x, y = (float(v) for v in fld.xy(r["x"], r["y"]))
        if r["side"] == "ball":     # the ball, drawn small and just ahead of whoever holds it
            ax.add_patch(Ellipse((x + 0.75, y + 0.55), 0.6, 0.36, fc=style.BALL, ec="white", lw=0.5, zorder=11))
            continue
        lab = str(r.get("label", "") or "")
        if r["side"] == "off":
            ax.add_patch(Circle((x, y), MARK_R, fc=style.OFFENSE, ec="white", lw=0.6, zorder=8))
            tc = "white"
        else:
            ax.add_patch(RegularPolygon((x, y), 4, radius=MARK_R * 1.3, orientation=np.pi / 4,
                                        fc=style.DEFENSE_TINT, ec=style.DEFENSE, lw=1.0, zorder=8))
            tc = style.INK
        fs = max(4.0, min(8.0, ppy * (0.85 if len(lab) <= 1 else 0.62)))
        ax.text(x, y, lab, ha="center", va="center", fontsize=fs, color=tc, fontweight="bold", zorder=10)
    for key, col in rings:
        r = cur[cur["displayName"] == key].iloc[0]
        x, y = fld.xy(r["x"], r["y"])
        ax.add_patch(Circle((float(x), float(y)), 1.05, fc="none", ec=col, lw=1.7, zorder=7))
    return fld


def strip(play, times, lateral, window, notes=None, rings=(), ncols=2, width_in=6.5):
    """Numbered frame strip with a fixed, box-sized window and a note under each frame."""
    df = tracking(play)
    nrows = int(np.ceil(len(times) / ncols))
    pw = width_in / ncols
    span_w, span_d = lateral[1] - lateral[0], window[1] - window[0]
    fig, axes = plt.subplots(nrows, ncols, squeeze=False,
                             figsize=(width_in, nrows * (pw * span_d / span_w + (0.6 if notes else 0.25))))
    for ax in axes.flat[len(times):]:
        ax.axis("off")
    for i, (ax, t) in enumerate(zip(axes.flat, times)):
        panel(ax, play, df, t, lateral, window, rings)
        ax.set_title(f"{i + 1}.  " + ("snap" if abs(t) < 1e-6 else f"{t:+.1f} s"),
                     fontsize=8.5, color=style.INK_2, loc="left")
        if notes:
            ax.text(0.5, -0.03, notes[i], transform=ax.transAxes, ha="center", va="top",
                    fontsize=7.2, color=style.INK)
    fig.tight_layout(h_pad=1.4 if notes else 0.6, w_pad=0.6)
    return fig


def video(play, times, lateral, window, notes=None, rings=(), width_in=5.2):
    """Web version of the strip: the same panel at every frame, with the strip's notes as captions."""
    import shutil
    from matplotlib import animation
    from IPython.display import HTML, display
    df = tracking(play)
    span_w, span_d = lateral[1] - lateral[0], window[1] - window[0]
    fig, ax = plt.subplots(figsize=(width_in, width_in * span_d / span_w + 0.4))
    tmap = df.groupby("frameId")["time"].first()

    def update(fid):
        ax.clear()
        t = float(tmap[fid])
        panel(ax, play, df, t, lateral, window, rings)
        txt = ("snap" if abs(t) < 0.05 else f"{t:+.1f} s")
        if notes:
            cur = [n for tt, n in zip(times, notes) if tt <= t + 1e-6]
            txt += ("   " + cur[-1].replace("\n", " ")) if cur else ""
        ax.set_title(txt, fontsize=9, color=style.INK, loc="left")
        return []

    anim = animation.FuncAnimation(fig, update, frames=sorted(df["frameId"].unique()), interval=100)
    html = (anim.to_html5_video(embed_limit=40).replace("<video ", '<video style="max-width:100%;height:auto" ')
            if shutil.which("ffmpeg") else anim.to_jshtml(default_mode="once"))
    plt.close(fig)
    display(HTML(html))


def animation_or_strip(play, times, **kw):
    """Video on the website; a numbered, captioned frame strip in the PDF."""
    if pdf_build():
        strip(play, times, **kw)
        plt.show()
    else:
        kw.pop("ncols", None)
        video(play, times, **kw)

Picture a third-and-1 for the 2024 Baltimore Ravens, late in a close game. The Ravens break the huddle and something about the picture looks old. The quarterback walks right up behind the center and puts his hands under him. Behind him, in a straight line, stand a 300-pound fullback and, seven yards deep, a 247-pound running back, hands on knees. One tight end, two receivers. The analyst says “they’re in the I,” and the defense, which has spent most of the afternoon with five defensive backs on the field, is already shouting and pointing and walking a safety down toward the line. This is a formation your grandfather would recognize. So why is one of the best offenses in football still using it, and why does the defense react as if it already knows what’s coming? This chapter is about those old shapes: what each was built to do, how they grew out of one another, why most of them faded from the NFL, and why the ones that survived are making a comeback.

You’ll be able to…

  • Name the classic formations on sight: the I-formation, offset I, Power I, pro set (split backs), singleback, ace, T-formation, wishbone, flexbone, Wing-T and the goal-line formation.
  • Say what each formation was invented to do (give the runner a lead blocker, let him run downhill, stay balanced, or fool the defense with misdirection) and what it gives up.
  • Match each formation to the personnel grouping that usually comes with it.
  • Explain why two-back, under-center football declined in the NFL, and point to the three places it survives in 2025: 21 personnel, short yardage and play-action.
  • Use the backfield as a pre-snap clue: under center or shotgun, one back or two, stacked, split or offset, and count blockers against the defenders near the ball.

Reading a formation from the back to the front

The personnel told you who is on the field. The formation tells you where they stand. Formation Rules and the Language of Alignment gave you the grammar: seven players on the line of scrimmage, the ends of the line eligible to catch passes, the receiver letters (X, Y, Z, H, F), and the quarterback’s three depths, under center (taking the snap directly from the center’s hands), the pistol (about three to four yards back) and the shotgun (about five yards back). This chapter and the next are the vocabulary: the named formations you’ll actually hear on a broadcast.

The classic formations share one thing: the quarterback is under center. For most of football history that was simply how the game was played, and the formations differ mainly in what stands behind the quarterback. So read them from the back to the front:

  1. Is the quarterback under center? If he is, you’re in this chapter’s world.
  2. How many backs, and how are they arranged? One or two or three; stacked one behind another, side by side, or one shifted to the side.
  3. Where are the tight ends and receivers? A tight end on one side or both, a wing, receivers wide.

One rule settles most arguments about step 2. Name the formation by where players stand; count the personnel by who they are. Anyone a yard or more behind the line and inside the width of the tight ends is in the backfield, so a tight end standing directly behind the quarterback makes the formation an I, whoever he is. The personnel comes from the substitution, as in Personnel: from the couch you can’t tell a tight end from a fullback by where he stands, and often not by his build. When the two disagree, say both: a tight end at fullback is “an I with 12 personnel.” The same rule explains the exceptions you’ll meet below. The Wing-T’s wingback and the flexbone’s slotbacks stand near the line but are running backs, so they count as backs; an H-back is a tight end wherever he stands.

Every formation in this chapter can be described in those same terms, and the names often tell you the shape. The I is a letter. The T is a letter (the quarterback and three backs form the crossbar and stem). The wishbone looks like one. The pro set was what the pros used. The naming isn’t standardized, and every team has its own playbook words, but the broadcast names are stable enough to learn once.

The letters in this chapter’s diagrams. Playbook letters get reused, so here is the key, and it never changes inside a figure. On offense (circles): Q quarterback, C center, R the running back (the tailback), F fullback, H a halfback (a third back in the old formations), X and Z wide receivers, Y and U tight ends. Two letters do double duty, as they do in real playbooks: in the one-back pictures H is the “move” player from Formation Rules, a slot receiver or an H-back, and the option and Wing-T formations bring their own back letters (A, B, W), named where they appear. The ring settles it: aqua is a running back, gold is a tight end, no ring is a receiver. On defense (squares): E end, T tackle, N nose tackle, S, M and W the Sam, Mike and Will linebackers, C cornerback, FS and SS the safeties.

NoteMadden vs. real life

In Madden you pick a formation from a menu (“I-Form Pro,” “Singleback Ace,” “Strong Close”) and then pick a play from the list that belongs to that formation, as if each formation owned its own plays. Real offenses work the other way around. A team has a few dozen run and pass plays and runs most of them from many formations, because a play the defense can predict from the formation is a play the defense can stop. The formation is the first word of the play call (“I Right 26 Power,” as What a Play Really Is spelled out), not the play. The menu names themselves are borrowed from real playbooks.

The ancestor: the T-formation

Almost every formation in this chapter descends from one shape. In the T-formation the quarterback stands under center and three backs line up side by side about four or five yards behind him: a fullback in the middle and a halfback on each side. Seen from the press box, the quarterback and the line of backs make a T.

The backs’ names are older than the T itself, which is why they’re confusing: the fullback is the bigger, blocking-first back, and the halfbacks are the runners. Any backfield with three backs behind an under-center quarterback, four players in all, is called a full house backfield. The T is the original full house; you’ll meet two more in this chapter (the Power I and the wishbone), and you’ll still see one on the goal line most Sundays.

Why it exists: speed to the line

The T’s predecessor was the single wing, the dominant formation of the 1910s to the 1930s, in which the center snapped the ball several yards back, through the air, to a tailback who ran, passed or handed off. The single wing is a power formation, but it’s slow: a direct snap takes time, the runner starts from a standstill several yards deep, and the defense can see who has the ball.

The T fixed all three problems with one change, putting the quarterback’s hands under the center. A hand-to-hand snap is instant and almost never fumbled. The quarterback can turn and hand the ball to a back who is already running, so the ball carrier hits the line of scrimmage at full speed one or two seconds after the snap. And because the quarterback can fake a handoff to one back and give to another with his back to the defense, nobody across the line knows for sure who has the ball. With three backs, all a few strides from the line and facing forward, the T threatens the middle and both flanks on every snap, and the defense has to respect all of them.

The modern T arrived with a bang. Coaches had used a tight T for decades, but in 1940 Clark Shaughnessy’s version of it, with wider splits, quicker hitting plays and a back sent in motion before the snap, went undefeated at Stanford, and George Halas’s Chicago Bears, who had been developing the same ideas with Shaughnessy’s help, used it to beat Washington 73–0 in the NFL Championship Game on December 8, 1940.1 Within a decade almost every team in the NFL had switched; the Pittsburgh Steelers, the last NFL holdout, used the single wing for the last time in 1952 and switched to the T in 1953.2 The Single Wing to the Scoring Drought tells that history in full; what matters here is that the T is the trunk of the tree.

What did the T give up? Width. With both ends tight and three backs packed behind the quarterback, it had no receivers outside the formation, so the defense could pack its own players near the ball too. Most of what happened next is the story of coaches taking the T and pulling one of its players out wide.

You don’t need to memorize the dates. The useful thing is the logic of the branches, because each one is a different answer to the same trade: every player you keep in the backfield is a blocker or a runner you gain, and a receiver you lose.

Two names on the tree never get a section of their own, because here they matter only as links. The split-T was Don Faurot’s 1941 Missouri version of the T. Its “split” was the line: wide gaps between the linemen that stretched the defensive front, while the quarterback rode down the line and either kept the ball or pitched it to a trailing halfback, depending on what one unblocked defender did. It may have been football’s first option offense.3 The veer was Bill Yeoman’s triple-option attack at Houston from the mid-1960s, one of the offenses Emory Bellard was trying to improve when he drew up the wishbone.4 Option Football has both.

The pro set: one back goes wide

The first big branch came from the passing game. In 1949 Shaughnessy, by then coaching the Los Angeles Rams, took one of his T-formation halfbacks, Elroy “Crazylegs” Hirsch, and lined him up wide as a receiver: a flanker.5 (Others had split a back wide before, so Hirsch’s claim to be the first flanker is disputed, but he became the model for the position.) Two backs stayed behind the quarterback, side by side. Add a tight end on one side and a receiver split wide on the other, and you have the formation that ran the NFL for the next quarter century.

The pro set, also called split backs, puts the quarterback under center and the halfback and fullback side by side behind him, usually about five yards deep and roughly behind the guards or tackles. Up front there’s a tight end on one side, the flanker (Z) a yard off the line on the tight end’s side, and the split end (X) on the line on the other side. That’s 21 personnel: two backs, one tight end, two receivers.

Why it exists: balance

The pro set’s virtue is balance. With a back on each side of the quarterback, the offense can run either way with a lead blocker, throw to either back in the flat, or keep both in to block a blitz on either side. The defense gets almost no clue from the backfield, because the backfield looks the same whichever way the play goes. Meanwhile the flanker and split end give the passing game two real receivers outside, which the T never had. That combination, a credible run game both ways plus two wide receivers and a tight end, made the pro set the standard NFL formation from the late 1950s into the 1980s. Bill Walsh’s early San Francisco offenses, the original West Coast offense, was built on split backs and threw many of its short passes to backs releasing from them.6

What does it give up? The deep tailback. Split backs can hit a hole fast: each is only about five yards deep and close to the line, and split-back dives and traps are among the quickest runs in football. What they lose is the luxury of a back seven yards deep, directly behind the quarterback, who has time to watch the blocks form and can cut to either side. And there’s no lead blocker lined up in front of the runner: if the play wants one, he has to come from the other side and cross the ball carrier’s path, which costs a beat and has to be timed so the two don’t collide. For power running straight at the defense, the next branch was better.

NoteFilm room: Green Bay Packers, 1959–1967

Vince Lombardi’s Packers won five NFL championships in seven seasons (1961, 1962, 1965, 1966 and 1967, the last two followed by wins in the first two Super Bowls), and they did it mostly from the pro set.7 Fullback Jim Taylor and halfback Paul Hornung lined up side by side behind Bart Starr, and the signature play was the power sweep: both guards (Jerry Kramer and Fuzzy Thurston) pulled toward the sideline, the fullback blocked the defensive end, and the halfback took the handoff wide behind a wall of blockers. Lombardi’s name for the formation was “Red”: split backs, set wide apart. The runner didn’t aim for a fixed hole; he followed his blockers and “ran to daylight,” the title of Lombardi’s 1963 book.

What to notice: the split backs are what made the sweep hard to diagnose. The same backfield ran the sweep, the fullback’s runs inside and the plays that punished defenses for chasing the sweep, and the defense couldn’t tell which from the backfield. (The sweep itself is drawn block by block in Gap and Power Runs.)

The I-formation: stacking the backs

If you want the ball carrier running straight at the defense, put him directly behind the quarterback, and put the blocker in front of him. That is the I-formation: quarterback under center, the fullback directly behind him about four to five yards deep, and the tailback directly behind the fullback about seven yards deep. Three players in a straight line make the I. With one tight end and two receivers it’s 21 personnel, and “I-formation” and “21 personnel” were close to synonyms for a generation.

The I is usually credited to Tom Nugent at the Virginia Military Institute in 1950 (stacked backfields appear in older playbooks too, so “invented” is disputed). It became famous at USC in the 1960s under John McKay, who got it from his assistant Don Coryell; McKay’s I-formation tailbacks (Mike Garrett, then O.J. Simpson) won Heisman Trophies and gave USC the nickname “Tailback U.”8 By the 1970s and ’80s it was everywhere, college and pro.

The deep tailback is the whole point. Seven yards is a long way back, and it buys three things.

A running start. The tailback takes the handoff already moving at speed, straight toward the line: running downhill, as Formation Rules put it. A runner who meets a linebacker at full speed going forward wins more collisions than one who meets him moving sideways.

Time to see. From seven yards the tailback can watch the blocks form in front of him before he commits. If the hole he was promised closes, he can see the one that opened instead. Because he’s centered, that includes the other side of the formation: he can press the play side and then cut back against the grain (turn back across the formation into a lane the over-running defense left), which is why zone-running teams kept the I.

A lead blocker. The fullback is between the tailback and the line, so on a run he goes first, into the hole, and blocks the first defender who fills it, usually a linebacker. That’s a lead block, and the simplest I-formation run of all, the iso (short for “isolation”), is built on it. The diagram below draws it against a base 4-3 defense: four defensive linemen, three linebackers and four defensive backs, the defense’s traditional starting lineup.

Count before you look at the arrows, because this is the arithmetic of every running play. The offense’s possible blockers are everyone except the quarterback and the ball carrier: here the five linemen, the tight end and the fullback, seven. The defenders who matter are the ones close enough to the ball to tackle the runner in the first second, the ones in what A First Look at Defense calls the box (roughly, from just outside one tight end to just outside the other, and back about five to seven yards). The two cornerbacks are out covering the wide receivers and the two safeties are 10 or more yards deep, so they don’t count yet. That leaves the four linemen and three linebackers: seven. Seven blockers for seven defenders means every defender near the ball can be blocked, and the runner meets nobody unblocked until the safeties arrive. Coaches call that a hat on a hat. When the defense has one more player in the box than the offense has blockers, one defender is free, and the runner has to beat him. Keep that count in mind; the rest of the chapter keeps using it.

In the iso, four linemen and the tight end block the man in front of them, the left guard climbs to the Mike linebacker, and the fullback leads through the gap between the left guard and left tackle onto the Will, “isolating” him, one on one. The tailback reads that block and runs off it. The iso and its relatives are taught properly in Gap and Power Runs; here it’s the cleanest picture of why the I exists.

What the I gives up

Every strength of the I has a price.

  • It’s slow to the edge. A tailback seven yards deep has a long way to go to get around the end. Outside runs from the I take time, and a fast defense can beat them to the sideline.
  • The backs are slow to get into routes. The fullback is in the middle of the formation and the tailback is deep; neither can get into a pass route quickly. On a straight dropback the I has really only three quick receivers (the two wideouts and the tight end), and the defense knows it. The exception proves the rule: on play-action, the fullback who “should” be blocking becomes a fourth receiver precisely because nobody is watching him.
  • It’s obvious. Two backs stacked behind an under-center quarterback says “run” loudly. In neutral situations (first and second down, game still close) from 2022 to 2025, NFL offenses lined up under center with two backs threw on only about 27% of plays, roughly three runs in four. With one back under center they threw about 38% of the time; from the shotgun with one back or none, about 71%.9 That’s the price, and also, as you’ll see, the opportunity.

Offset I: cheating the fullback

A fullback directly behind the quarterback has to choose a side after the snap, and the defense can’t tell which side. That’s good for disguise but bad for geometry: if the play goes wide, he has to run sideways before he can block anyone. The offset I moves the fullback a yard or two to one side, behind a guard or tackle, so he starts closer to where he’s needed.

Coaches say the fullback is offset strong (toward the tight end) or weak (away from it). Offset strong, he’s in position to lead on runs to the tight end side, where the offense has an extra blocker. Offset weak, he’s in position to lead on runs away from the tight end, or to block a blitzing linebacker coming from the weak side on a pass. The cost is a clue. A fullback offset to one side tilts the offense toward that side, and defensive coaches study which way each opponent offsets on which plays. Smart offenses run plays both ways from both offsets so the tilt means nothing.

Playbooks and broadcasts have their own words for these. An I offset strong is often a Strong I and offset weak a Weak I (Madden uses both). And a classic formation’s full name is usually the backfield plus the receivers: “I Pro” is the I with the tight end on one side and a receiver on each side, as drawn here; “I Twins” puts both wide receivers on the side away from the tight end; “I Slot” puts the flanker in the slot. When an analyst says “I-formation twins,” he has told you the backfield and then the receivers, back to front, exactly the order this chapter reads them. The receiver sets get their own chapter: Spread Formations.

Power I: two lead blockers

The Power I adds a third back beside the fullback, so the tailback has two blockers leading him: usually 31 personnel (three backs, one tight end, one receiver), another full house. It’s a short-yardage and goal-line formation, built for the moment when the offense needs one yard and the defense knows it.

Do the count. The Power I has eight blockers (five linemen, the tight end and two lead backs), one more than a base 4-3 has in the box. So the defense answers the way defenses always answer an extra blocker: it brings a safety down, and it’s eight against eight again. The extra back’s job is to make sure that eighth defender gets blocked too, by a player who arrives with a running start.

Singleback and ace: trading the fullback for a tight end

Through the 1960s and ’70s, defenses got better at stopping two-back formations, and in 1978 the NFL changed its rules to help the passing game (pass blockers could extend their arms, and defenders could contact receivers only within five yards of the line;10 1978 and the Passing Revolution has the story). Offenses wanted more receivers on the field. The simplest place to find one was the backfield.

A singleback formation keeps the quarterback under center but uses only one running back, directly behind him, usually about seven yards deep. The second back is gone: his spot on the field goes to a third receiver or a second tight end. With three receivers it’s 11 personnel, the NFL’s most common grouping; with two tight ends it’s 12.

The two-tight-end version has its own name. The ace formation is a singleback formation with a tight end on each side of the line and two wide receivers (12 personnel). (Some teams say “ace” for any one-back look; the broadcast usage is the two-tight-end one.)

Why it exists: a tight end does a fullback’s job, and more

Taking the fullback off the field sounds like giving up the lead blocker. The insight of the one-back coaches was that a tight end can do most of a fullback’s job from a different spot, and a lot more besides. A tight end next to the tackle can block the same linebacker a fullback would have led on, or he can run a pass route down the middle, which a fullback almost never does. A second tight end on the other side makes the formation balanced: the same on both sides. A defense usually sets its strong side to the tight end (its biggest linebacker and a safety go there), but with a tight end on each side there isn’t one, and the offense can run either way with an extra blocker.

The lone back benefits too. Seven yards deep and directly behind the quarterback, he can start toward either side and see the whole line. That’s exactly what the zone running schemes of the 1990s onward wanted, a back who reads the blocks and picks his hole, and it’s why one-back, two-tight-end sets became a main home of zone running, alongside the I (Zone Running draws it).

What did it give up? The fullback’s head start. A tight end blocking from the line meets the linebacker sooner, but with less of a running start, and the tailback no longer has a blocker leading him into the hole. The ace’s balance has a cost of its own, too: with a tight end on each side and the receivers tight, it has only two wide receivers, so it can’t stretch the coverage the way three-receiver sets can. One-back offenses answered the lead-blocker problem with motion: move a tight end or an off-the-line “H-back” across the formation before the snap so that he ends up where the lead blocker is needed.

NoteFilm room: Washington, 1981–1992

Joe Gibbs, a Don Coryell assistant who became Washington’s head coach in 1981, built the most influential one-back offense in NFL history, and won three Super Bowls with it, after the 1982, 1987 and 1991 seasons, with three different starting quarterbacks (Joe Theismann, Doug Williams and Mark Rypien).11 The story Gibbs told was about one player: the Giants’ Lawrence Taylor, a linebacker so fast off the edge that a running back couldn’t block him. Gibbs’s answer was to take the second back out of the backfield and put a second tight end, often a smaller, quicker one who lined up off the line beside the tackle or went in motion (the H-back, a version of the H, or move player), where he could get his hands on Taylor before Taylor got going.

The H-back became the one-back offense’s Swiss army knife: he could block like a fullback, release like a tight end, or move across the formation before the snap. Behind him, Washington’s huge line (the “Hogs”) ran the plays that made the era famous, including the counter trey, in which two linemen pulled the opposite way from the backfield’s first step.

What to notice: the H-back is a fullback and a tight end in one body, and he forces the defense to decide which he is on every snap.

Watch how one H-back changes the picture before the snap. The offense starts with both tight ends on the right: Y on the line and the H-back (H) a step behind the right tackle. The defense sets its strength there: its 3-technique tackle, its Sam linebacker and its strong safety go to the right. Then the H-back moves.

The option branch: wishbone and flexbone

The other way to use the T’s three backs is to keep all three and let the quarterback choose who gets the ball after the snap. That’s option football: the offense leaves a defender unblocked on purpose, the quarterback reads what he does, and gives the ball to whichever back the defender didn’t take. Option football has its own chapter. Here we only need the two formations it made famous.

The wishbone

The wishbone was drawn up at the University of Texas in 1968 by offensive coordinator Emory Bellard, who moved the T’s fullback up close behind the quarterback and set the two halfbacks deeper and wider, one on each side. A Houston sportswriter, Mickey Herskowitz, said it looked like a “pulley bone” (an old Southern name for a chicken’s wishbone), and head coach Darrell Royal turned that into “wishbone.”12 Texas won 30 straight games with it, Oklahoma and Alabama adopted it, and through the 1970s the wishbone was the most feared offense in college football.

The geometry is built for the triple option, the play that gives the formation its purpose: three possible ball carriers, and the quarterback decides who gets the ball by reading two unblocked defenders after the snap. The fullback, close to the line, is the first threat (the “dive”), straight ahead. The quarterback is the second, running along the line. The third is the halfback from the other side, who starts away from the play and sprints across behind the quarterback as the pitch man; that’s why the halfbacks line up deeper than the fullback, so he has room to get across. The playside halfback leads around the edge and blocks the defender who would tackle the pitch man. So every back has a job on every play, dive, lead or pitch, toward either side; that is what a full house buys an option team. (Option Football draws the reads.)

The flexbone

The wishbone’s weakness was width. With all three backs bunched behind the quarterback and only one receiver, defenses could crowd the line and outnumber the option. The flexbone, which grew out of the wishbone in the late 1970s and 1980s, takes the two halfbacks out of the deep backfield and stands them just off the line, outside the tackles, as slotbacks. A fullback stays behind the quarterback, and there are two wide receivers on the line, usually with no tight end. Who invented it is disputed: Bellard himself (at Mississippi State in 1979), Fisher DeBerry’s Air Force teams and Paul Johnson at Georgia Southern all have a claim.13

Why there? A slotback just off the line is three things at once. He can block a defender on the edge from a short distance. He can go in motion before the snap and arrive at the pitch man’s spot just as the ball is snapped, so the defense doesn’t know which side the option is going. And he can run a pass route immediately, because he’s already past the line’s congestion. The two receivers outside force the defense to keep two defenders wide, which takes them out of the box, the area near the ball where the run is defended (roughly tight end to tight end, five to seven yards deep). The flexbone spreads the defense like a passing formation while keeping three runners near the ball.

What does it give up against the wishbone? Inside power. With the halfbacks moved up to the edges, the flexbone has no lead blocker lined up behind the quarterback for a run between the tackles, and no tight end; it beats the defense with angles and width, not with bodies at the point of attack.

NoteFilm room: Navy and Georgia Tech, Paul Johnson’s flexbone

No coach is more closely tied to the flexbone than Paul Johnson. He won two Division I-AA titles with it at Georgia Southern (1999 and 2000), took over a Navy team that had gone 0–10 in 2001 and had it winning eight games within two seasons, then ran the same offense against Atlantic Coast Conference defenses at Georgia Tech from 2008 to 2018.14 Why would an undersized team choose an old formation? Because it changes the problem. The service academies can’t recruit 310-pound linemen; they can recruit disciplined players who block quickly at defenders’ legs and carry out fakes on every play, and the flexbone makes the defense play assignment football for sixty minutes, something defenses that see it once a year rarely practice. (Even Navy has since moved much of its option game into the shotgun.15)

What to notice: when you see a flexbone team on a Saturday, watch the slotbacks before the snap. The one who goes in motion is usually telling you which way the option is headed, and the other one is often a blocker.

You won’t see either formation in the NFL. Option football needs a quarterback who takes a hit on most plays, and NFL teams won’t spend their most expensive player that way; NFL defenders are also fast enough to cover the width the option relies on. But the idea, a quarterback who reads a defender instead of blocking him, came back to the NFL from the shotgun, as the zone read: the quarterback reads one unblocked defender and either hands off or keeps the ball himself. The College Laboratory traces that lineage, from the wishbone and veer to the spread option.

The Wing-T: misdirection

The last classic branch crossed the T with its ancestor. Around 1950 Dave Nelson, at Maine and then Delaware, combined the quick-hitting T backfield with the single wing’s wingback, a back lined up just outside and a yard behind the tight end.16 The Wing-T puts the quarterback under center, a fullback directly behind him, a halfback beside the fullback on the side away from the tight end, and the wingback outside the tight end on the other side, with a split end on the halfback’s side. Counted as backs, the wingback and halfback make it 31 personnel; teams that use a receiver at wingback call it 21.

Why it exists: making the defense guess

The Wing-T’s purpose is misdirection: making the defense commit to one direction and then going the other. With three backs placed so that each can carry the ball in a different direction, and a quarterback who turns his back to the defense and fakes to each of them in turn, every play looks the same for its first second. The Wing-T is organized in series: groups of plays that start with the same backfield action and end differently. The famous one is the buck sweep series in the figure. Its base play is the sweep: the line blocks down (each blocker hits the defender inside him, from the side) while both guards pull, running sideways behind the line to lead the halfback around the wing. When the defense starts chasing the sweep, the same first second becomes the fullback’s buck up the middle or a trap (a defensive lineman is let through unblocked and then hit from the side by a pulling lineman), a wingback counter back the other way, or the waggle, the Wing-T’s bootleg: the quarterback fakes, rolls out the other way behind a pulling guard and runs or throws. Defenders who flow (run toward the side the play seems to be going) with the first thing they see get punished; defenders who wait get blocked.

The trade is complexity and, again, width. A misdirection offense needs a lot of practice time to run its fakes well, and its backs are runners and blockers, not receivers. You’ll rarely see a true Wing-T on Sunday, but you’ll see its children every week: the jet sweep, in which a receiver sprints across behind the line just as the ball is snapped, the down-block-and-pull schemes of the NFL’s gap runs, and the NFL bootleg are Wing-T ideas in an NFL uniform (Perimeter Runs and Misdirection traces them).

The goal-line formation: everyone in close

Near the goal line the field gets short and narrow at the same time. There’s no room behind the defense to throw deep, and the offense needs a yard or two, not twenty. So every team keeps a goal-line formation: the quarterback under center, two backs stacked behind him, three tight ends (or a sixth lineman, the jumbo grouping), and no wide receivers at all. Everyone lines up tight to the ball. In personnel terms that’s 23, or 14 with a tight end standing at fullback, or a jumbo grouping.

Why it exists: shrink the field, pick the spot

Do the count from the iso. The offense has nine blockers: five linemen, three tight ends and the fullback (the quarterback and the tailback don’t block). The defense has eleven players, and at the goal line all eleven are in the box, because there’s no deep field left to protect. Eleven against nine: on paper the defense wins the count by two, and down here it always will. So the goal-line formation isn’t trying to win the count. It’s trying to make the count matter less, in two ways.

First, it shrinks the space. Runs are aimed at gaps, the spaces between blockers, and every defender near the ball is assigned one to fill. Tight splits (here about a yard and a quarter between linemen, against the usual yard and a half or more) and everyone close to the ball leave narrow gaps with little room to slip through. Count them in the figure: the seven players on the line make eight gaps, and the wing makes a ninth. The defense needs a player in each of the nine before it has anyone left over. Second, the offense picks the spot. It needs one yard, not ten, so it can pour extra bodies into a single gap: a double team on the line, the fullback leading into the hole, and a tailback who only has to fall forward. The defense’s two spare players are somewhere on a crowded line, and they may never reach the one gap that matters. The defense answers by bringing its own heavy package and crowding the line, so the goal line is the one place in football where both sides deliberately give up all their space.

The price is that a defense that knows you’ll run can commit everyone to stopping it. That’s why the goal-line formation is also where a lot of short play-action touchdowns come from: the quarterback fakes the handoff, every defender bites, and one of the three tight ends, who looked like a blocker, slips into the end zone alone. In 2022–2025, about three of every four NFL plays inside the 2-yard line that came from under center with two or more backs were runs; when they were passes, about four in five used a play fake.17 (The quarterback sneak starts here too, and so does its push-assisted version, the tush push, in which teammates shove the quarterback forward from behind, still legal in 2026.) Red Zone, Goal Line, and Short Yardage is where that fight gets its full treatment.

  • No wide receivers and three tight ends: goal-line.
  • Quarterback in the shotgun: you’re in the next chapter’s world, even with two backs.
CautionCommon misconception

“Formation names are universal.” They’re universal enough for broadcasts, but not in playbooks. One team’s “ace” is another’s “Deuce” or “Tiger”; Lombardi called split backs “Red,” and other systems call two-back looks “Near,” “Far” and “Split.” Even the broadcast names blur: an offset I with a tight end at fullback, or a one-back set with the quarterback in the pistol, may be called three different things by three analysts. Learn the shapes, not just the words: under center or not, how many backs, where they stand, where the tight ends are. The shape is what tells the defense, and you, what’s possible.

In 2006 offenses were in the shotgun on about one play in five. By 2023 it was about 72%. Over the same years the I-formation’s share of plays fell from about 11% in 2016 to about 6% in 2022, as far as the formation labels in the public data go.18 The causes overlap, but there are five worth knowing.

  1. Passing became more valuable than running. After the 1978 rule changes, and again as offenses learned to throw efficiently, a pass gained more on average than a run. In the EPA terms you met in Downs, Distance, and Field Position, NFL dropbacks (sacks and scrambles included) averaged about +0.04 expected points per play in 2025 and designed runs about −0.05; across 2006–2025 it was about +0.03 against −0.07, and dropbacks came out ahead in every one of those seasons.19 Expected Points and Success Rate shows how analysts measure that gap, and why it’s smaller than it looks (runs are less risky, and they set up the pass). An offense that passes more wants more receivers, and the second back is the first player to go.
  2. The shotgun is a better place to pass from. Five yards back, the quarterback takes a shorter drop and never has to turn his back on the defense. He sees the coverage from the snap and has more time before the rush arrives. The shotgun was used in the NFL as early as 1960 and revived by Tom Landry’s Cowboys in 1975, but for decades it was a third-down formation;20 the history chapters follow it.
  3. Spreading the defense makes running easier, too. Three or four receivers pull defensive backs out wide, which leaves fewer defenders near the ball. An offense in the shotgun with 11 personnel can often run against six defenders in the box instead of seven or eight: the same count you did for the iso, now tilted toward the offense. Spread formations are the next chapter.
  4. College stopped producing under-center quarterbacks. By the 2010s most college offenses, and many high schools, played almost entirely from the shotgun. Quarterbacks arrived in the NFL having rarely taken a snap under center, and NFL coaches built around what their quarterbacks did best.
  5. A fullback is an expensive specialist. A roster spot used on a lead blocker is one not used on a receiver or tight end who can play every down. Many NFL teams stopped carrying a true fullback at all.
ImportantWhy it exists

The shotgun won the passing game; the classic formations kept the running game’s advantages. Under center, the quarterback can hand off to a back already running downhill, toward the line, and fake a handoff with his back to the defense. In the shotgun, many runs start sideways, and a play fake is less convincing because the defense can see the ball the whole time. Each formation is a bundle of trade-offs, and the NFL’s long move to the shotgun traded the run game’s angles for the passing game’s vision. The 2025 turn back toward under center is offenses deciding they gave up too much.

Three things stand out, and they’re where the classic formations live now: two situations (short yardage and play-action) and one kind of team (the few that still build around a fullback).

Short yardage. With a yard or less to go, NFL offenses lined up under center on about 63% of plays in 2025, nearly twice their overall rate. When the offense needs one yard, the quick snap, the downhill handoff and the extra blockers of the I and the goal-line formation are worth more than the quarterback’s view of the coverage. (The sneak and the tush push start from under center too.)

Play-action. About four of every five dropbacks from under center in 2025 used a play fake, against about one in ten from the shotgun. That’s no coincidence. A quarterback who turns his back to the defense and puts the ball in a running back’s belly is selling the run with his whole body, and a two-back formation that usually runs is the best advertisement for a fake. The formation’s predictability is the bait. Play-Action, Bootlegs, and Screens shows what the fake does to the defense.

Notice the middle bar in each panel, too. The pistol isn’t under center, so the next chapter files it with the shotgun, but in 2025 it behaved like a cousin of this chapter’s formations: offenses passed on only about 29% of pistol snaps, and about half of those dropbacks used a play fake. That’s what you’d expect from an alignment built to keep the running back directly behind the quarterback: a downhill run and a believable fake, from a few yards back.

NoteFilm room: San Francisco 49ers, 2017–2025

Kyle Shanahan’s 49ers have been the NFL’s leading 21-personnel team for most of his tenure, with fullback Kyle Juszczyk (Personnel has the numbers). Much of it comes from the I and offset I: Juszczyk lines up behind the quarterback, leads on a linebacker or the edge defender on running plays, and on play-action slips out into the flat as the defense crashes on the run. In 2025 the 49ers were under center on about 46% of plays, well above the league’s 34%, and had two backs on the field on about a quarter.21

The 49ers’ version of the I isn’t the 1970s version, though. Juszczyk motions out of the backfield, lines up as a wing or in the slot, and returns to the I on the next play, so the defense can’t key on where he stands. The formation is classic; the way it’s used is modern.

What to notice: when the 49ers line up in the I, watch where the fullback’s path ends. His first steps are designed to look the same on the run and on the fake. If he hits someone, it’s a run; if he crosses the formation as if to block the backside end, brushes past him and keeps going to the flat, it’s the bootleg, and the defense has been fooled by its own expectations.

NoteFilm room: Baltimore Ravens 2024, Patrick Ricard and Derrick Henry

The 2024 Ravens (head coach John Harbaugh, offensive coordinator Todd Monken, an era that ended after 2025) paired Lamar Jackson with Derrick Henry, who ran for 1,921 yards, second in the league, behind Patrick Ricard, an undrafted defensive tackle from Maine turned 300-pound fullback.22 Baltimore had two or more backs on the field on about a quarter of its 2024 plays, against a league rate of about 18%. A defense built to stop Jackson’s passing sits in nickel; the Ravens line up with two backs, and the nickel back suddenly has a 300-pound fullback coming at him. If the defense answers with linebackers, Jackson throws over them: the personnel dilemma, delivered from a formation that has been around since the 1950s.

What to notice: in a classic two-back formation, find the fullback and do the count: blockers against defenders in the box. If the defense matched the count with a defensive back where a linebacker should be, the offense has the matchup it wanted; if it matched it with a linebacker, look for the throw over him.

The 2025 under-center revival

The shotgun’s share fell from 72% in 2023 to 66% in 2025, and FTN’s charting has offenses under center on 33.6% of plays in 2025, up from 27.3% in 2023.23 The reason is the same arms race that sent personnel heavier (Personnel told that half): defenses spent the early 2020s getting lighter and playing more two-deep safety looks to stop the pass, which left fewer big bodies near the line. Offenses answered by getting bigger and putting the quarterback back under center, where the run game has its best angles and play-action its best disguise. The offensive side of the ledger pushed the same way. Wide zone and play-action from under center, the Shanahan and McVay style, kept producing: in 2024 and 2025, dropbacks from under center averaged about +0.14 expected points per play against about +0.03 from the shotgun.24 (That gap flatters under center, whose passes come on downs when the defense is braced for a run, but that is exactly the point of the formation.) And the quarterbacks who run those offenses are comfortable under center. Sean McVay’s Rams were under center on about 59% of their 2025 snaps, the most in the league, and their quarterback, Matthew Stafford, was the league’s MVP. The champion Seahawks, under offensive coordinator Klint Kubiak in 2025, were second, at about 53%.

Whether the turn lasts depends on the defense’s answer, and in 2025 defenses were already answering by playing more base. The old formations didn’t come back because they were old. They came back because, for now, they’re the right answer to what defenses are doing. That’s the pattern of this entire book.

What the backfield looks like on TV

Every diagram in this chapter is drawn from above, the way coaches draw them and the way the All-22 film shows them. The broadcast mostly isn’t. The main sideline camera sits high on one side of the field, looking across it, so the direction the offense is moving runs left to right across your screen and the width of the field is squashed into the up-down direction. That flips the picture in a way worth knowing before you try to name a backfield live.

CAM = np.array([-1.0, -52.0, 24.0])      # (d, w, height) in yards: high on the offense's left sideline
LOOK = np.array([-2.5, 0.0, 0.0])


def tv_project(pts):
    """Pinhole projection of (d, w, z) points onto the camera's screen; returns screen (x, y), depth."""
    f = LOOK - CAM
    f /= np.linalg.norm(f)
    r = np.cross(f, [0.0, 0.0, 1.0])
    r /= np.linalg.norm(r)
    u = np.cross(r, f)
    v = np.atleast_2d(pts) - CAM
    depth = v @ f
    return np.c_[(v @ r) / depth, (v @ u) / depth], depth

TipWatch for it: name that formation

For ten snaps in a row, before the ball is snapped, say three things out loud and then a name.

  • Under center or gun? Look at the quarterback’s hands. Under center means this chapter’s formations; pistol or shotgun means the next chapter’s.
  • One back or two (or three)? Count the players strung out behind the quarterback (on the sideline camera, a row is an I and a stack is split backs). Name the shape by where they stand: a tight end directly behind the quarterback still makes it an I. A tight end off the line beside a tackle (an H-back) is a tight end for the personnel count.
  • Stacked, split or offset? Two backs in a line is an I; side by side is a pro set; one a yard to the side is an offset I. Three backs is a Power I, a wishbone or a T, and in the NFL that means short yardage.
  • Where are the tight ends? One on each side with one back is an ace. Three tight ends and no receivers is goal line.
  • Count. Blockers (linemen, tight ends and any back who isn’t getting the ball) against the defenders in the box. One more defender than blockers says the defense is braced for the run.
  • Name it, then guess. Under center with two backs: lean run (about three plays in four), and if it’s a pass, expect a play fake. Under center with one back: still run-leaning (about three passes in eight in neutral situations) and still play-action heavy. Note which way an offset fullback points and whether the play went that way.
  • Log it. Add a formation column to your prediction log next to personnel: “21, I, UC” or “12, ace, UC.” After ten snaps, check how often your run/pass guess was right from under center versus from the gun.

What formation and personnel is this? Run or pass? And which side is the fullback set up to lead to?

TipAnswer

An offset I, offset weak (a “Weak I”), with 21 personnel. Two backs (fullback and tailback) and one tight end, so two receivers. The fullback is offset to the left, away from the tight end: that’s “weak.”

Run or pass: lean run. Two backs stacked behind an under-center quarterback ran about three times in four in neutral situations, and second-and-4 is a comfortable running down. If it’s a pass, expect a play fake.

Which side: the left. The offset fullback is already behind the left guard, in position to lead on a weak-side run (an iso into the bubble over the Will, as in the iso diagram) or to pick up a blitzing linebacker from that side on play-action. Whether this offense actually runs toward its offset is a tendency, not a rule, and smart offenses break it on purpose. There’s no league-wide number to look up: the public charting data records how many backs there are, not which way the fullback is offset; that takes tracking data (Tracking Data and the Big Data Bowl works with it). Log it for the team you’re watching instead.

TipPredict the play 2: second-and-goal at the 1

The offense has the ball on the defense’s 1-yard line, second down. It sends in three tight ends, a fullback and a tailback; the defense answers with its goal-line package.

Count the blockers and the defenders in the box. Who wins the count, and so how does the offense try to score? If it throws, who is the likeliest target?

TipAnswer

The defense wins the count, eleven to nine, and at the goal line it always will: nobody has to stay deep. The offense’s blockers are the five linemen, the three tight ends and the fullback; the quarterback and the tailback don’t block.

So the offense doesn’t try to block everyone. It tries to win one gap. Expect a run aimed at a single spot with extra bodies there (a double team and the fullback leading into the hole) or a quarterback sneak behind the center. A tailback who needs one yard only has to fall forward.

If it throws: a play fake and a tight end, because eleven defenders committed to the run is exactly what the fake exploits. The likeliest target is the wing (H) or the tight end on the side away from the fake: he blocks for a count, as he would on a run, then slips into the end zone while the linebackers and safeties step up. Watch the two safeties: if both crash at the snap, the tight end is alone.

TipPredict the play 3: the I against eight in the box

First-and-10 at midfield, second quarter. The offense is in the I with 21 personnel. The defense, in its base 4-3, has walked the strong safety down beside the linebackers, leaving one safety deep.

Do the count. Why did the defense bring the safety down, and what is the offense likely to try?

TipAnswer

The count is eight against seven. The offense has seven blockers (five linemen, the tight end and the fullback). In the box the defense has four linemen, three linebackers and the strong safety: eight. The I’s whole advantage was a hat on a hat with a lead blocker; the safety takes that away by giving the defense one more player than the offense can block. It came down because the formation says run.

What the offense is likely to try: the run is still the most common call. If it runs, it often runs away from the extra man, to the left here: the safety walked down on the tight end’s side, so a run the other way makes him chase across the formation, and he arrives late. But this is also the picture play-action was invented for. Single-high, with the safety in the middle, means each cornerback is alone on his side, with no safety beside him. If the defense is playing man coverage behind the safety (Cover 1), that’s a one-on-one outside; if it’s zone with the corners dropping deep (Cover 3), the fake pulls the linebackers up and opens the window behind them for a crossing route. Either way, the run fake is what moves the eight defenders.

Takeaways

  • The classic formations are defined by the backfield. Read them back to front: quarterback under center or not, how many backs, and whether they’re stacked (I), side by side (pro set), offset (offset I) or three wide (T, wishbone, Power I).
  • Each formation is an answer to a problem, and every answer has a price. The I gives the tailback a lead blocker and a running start but is slow to the edge and obvious; the pro set is balanced and quick-hitting but has no deep tailback; the singleback and ace trade the fullback for a tight end or receiver; the wishbone and flexbone are built to read a defender instead of blocking him; the Wing-T lives on misdirection; the goal-line formation trades all its space for blockers.
  • They’re one family. The T’s three backs were stacked (I), split (pro set), spread (flexbone), scattered for misdirection (Wing-T) or traded for receivers (singleback). Knowing the tree helps you name the shape.
  • Count, every time. Blockers (everyone but the quarterback and the ball carrier) against defenders in the box. A hat on a hat favors the run; one extra defender is the defense’s answer; at the goal line the defense always wins the count, so the offense picks one gap instead.
  • Formation follows personnel. 21 usually means an I, offset I or pro set; 12 an ace; 11 a singleback or a shotgun look; three backs or three tight ends mean short yardage. Name the shape by where players stand, the personnel by who they are.
  • Two-back, under-center football faded because passing won, and it survives where running and faking matter most. In 2025, under center meant run about 70% of the time, play-action on about four of five dropbacks, and it was the default on short yardage. The under-center share rose to about a third of plays as offenses got bigger against lighter defenses.
