def dashed_motion(p, key, to, head=0.5):
    """Pre-snap movement to the spot `to` (absolute d, w), drawn dashed with a short solid head.
    (gridiron draws the last part of a path's final segment solid, so one long segment would
    look like a route.)"""
    d0, w0 = p._cur(key)
    dd, dw = to[0] - d0, to[1] - w0
    f = max(0.0, 1 - head / float(np.hypot(dd, dw)))
    p.motion(key, pts=[(dd * f, dw * f), (dd, dw)])


def schedule(p, plan):
    """Re-time pre-snap movement. plan = {key: (t_start, t_end)} in seconds from the snap.
    gridiron ends every pre-snap motion exactly at the snap and starts it whenever its length and
    speed require, so two players who should move together drift apart. This stretches each listed
    player's pre-snap path into his window (and holds him still until the snap if it ends early).
    Implemented here rather than in gridiron; call it after every action has been added."""
    kf, t0, t1 = p._compile()
    for key, (a, b) in plan.items():
        arr = kf[key]
        pre, post = arr[arr[:, 0] <= 1e-9], arr[arr[:, 0] > 1e-9]
        T0, T1 = pre[0, 0], pre[-1, 0]
        new = pre.copy()
        new[:, 0] = a + (pre[:, 0] - T0) / ((T1 - T0) or 1.0) * (b - a)
        rows = [new] + ([np.array([[0.0, *new[-1, 1:]]])] if b < 0 else []) + [post]
        kf[key] = np.vstack(rows)
    p._compiled = (kf, min(float(f[0, 0]) for f in kf.values()), t1)
    return p

def mpath(ax, P, pts, kind="pre"):
    """A path through (d, w) points drawn by hand. 'pre' = pre-snap motion (dashed, blue);
    'post' = what he does after the snap (thin, solid, grey)."""
    col = style.OFFENSE if kind == "pre" else style.INK_2
    xy = [P(d, w) for d, w in pts]
    xs, ys = zip(*xy)
    lw = 1.6 if kind == "pre" else 1.0
    cut = (xs[-2] + 0.75 * (xs[-1] - xs[-2]), ys[-2] + 0.75 * (ys[-1] - ys[-2]))
    ax.plot(list(xs[:-1]) + [cut[0]], list(ys[:-1]) + [cut[1]], color=col, lw=lw,
            ls=(0, (3, 2)) if kind == "pre" else "-", zorder=5, solid_capstyle="round")
    head = (xs[-2] + 0.7 * (xs[-1] - xs[-2]), ys[-2] + 0.7 * (ys[-1] - ys[-2]))
    ax.add_patch(FancyArrowPatch(head, xy[-1], arrowstyle="-|>", mutation_scale=8, color=col, lw=lw,
                                 shrinkA=0, shrinkB=0, zorder=5))

def snap_dot(ax, P, at, dd=-1.4, dw=0.0):

def snapshot(play, t, ax=None, lateral=(-21, 21), window=(-8.5, 15.5), ghosts=(), ghost_t=None, notes=()):
    """Everyone where he is at time t, a faint trail behind each mover since the first frame, and a
    hollow 'ghost' where each listed player stood at `ghost_t`."""
    pos = {k: play.position(k, t) for k in play.players}
    moved = [pl.moved(d=float(pos[k][0]), w=float(pos[k][1])) for k, pl in play.players.items()]
    q = Play([m for m in moved if m.side == "off"], [m for m in moved if m.side == "def"],
             los=play.los, ball_y=play.ball_y, to_go=play.to_go)
    q.highlights = dict(play.highlights)
    for n in notes:
        q.note(*n)
    fld = q.draw(ax=ax, lateral=lateral, window=window, title=False)
    ax, ppy = fld.ax, fld.pts_per_yard()
    no_numbers(ax)
    P = P_of(fld, play)
    start = play._compile()[1]
    for k, pl in play.players.items():
        if pl.pos in ("LT", "LG", "C", "RG", "RT", "DE", "DT", "NT"):
            continue
        pts = np.array([play.position(k, tt) for tt in np.linspace(start, t, 40)])
        if np.ptp(pts[:, 0]) + np.ptp(pts[:, 1]) < 0.4:
            continue
        xs, ys = fld.xy(*play.to_abs(pts[:, 0], pts[:, 1]))
        col = style.OFFENSE if pl.side == "off" else style.DEFENSE
        ax.plot(xs, ys, color=col, lw=1.2, alpha=0.55, ls="-" if pl.side == "off" else (0, (3, 2)), zorder=3)
    if ghost_t is not None:
        for k in ghosts:
            d0, w0 = play.position(k, ghost_t)
            d1, w1 = play.position(k, t)
            if abs(d1 - d0) + abs(w1 - w0) > 1.0:
                ghost_marker(ax, *P(d0, w0), play.players[k].label, ppy, play.players[k].side)
    bd, bw = play.ball_position(t)
    carried = any(np.hypot(*(play.position(k, t) - np.array([bd, bw]))) < 0.3
                  for k, pl in play.players.items() if pl.side == "off" and k != "C")
    bx, by = P(bd + (0.95 if carried and t > 0 else 0.0), bw)   # a carried ball sits just in front of him
    ax.add_patch(Ellipse((bx, by), 0.75, 1.1, facecolor=style.BALL, edgecolor="white", lw=0.6, zorder=9.5))
    return fld


def strip(play, times, labels, ncols=2, **kw):
    """The print version of an animation: one panel per chosen moment, each with a caption."""
    lateral, window = kw.get("lateral", (-21, 21)), kw.get("window", (-8.5, 15.5))
    n = len(times)
    nrows = int(np.ceil(n / ncols))
    pw = 6.5 / ncols
    h = nrows * (pw * (window[1] - window[0]) / abs(lateral[1] - lateral[0]) + 0.36) + 0.35
    fig, axes = plt.subplots(nrows, ncols, figsize=(6.5, h), squeeze=False)
    for ax in axes.flat[n:]:
        ax.axis("off")
    for ax, t, lab in zip(axes.flat, times, labels):
        snapshot(play, t, ax=ax, **kw)
        ax.set_title(lab, fontsize=8.2, color=style.INK, loc="left", pad=3)
    fig.tight_layout(h_pad=0.5, w_pad=0.5, rect=(0, 0.05, 1, 1))
    hs = [Line2D([], [], color=style.OFFENSE, lw=1.2, alpha=0.7, label="Offense: path so far"),
          Line2D([], [], color=style.DEFENSE, lw=1.2, ls=(0, (3, 2)), label="Defense: path so far"),
          Patch(facecolor="none", edgecolor=style.DEFENSE, lw=1.1, ls=(0, (2, 1.2)), label="Dashed outline: where he started")]
    fig.legend(handles=hs, loc="lower center", ncol=3, fontsize=7.2, frameon=False, bbox_to_anchor=(0.5, 0.0))
    return fig


def show_play(play, times, labels, web_caption, **kw):
    """Web: an animation with a caption that changes with the phase of the play.
    Print: a labelled frame strip (panel titles = `labels`)."""
    if PDF:
        strip(play, times, labels, **kw)
        plt.show()
        return
    df = play.to_tracking()
    tmap = df.groupby("frameId")["time"].first()
    anim = animate(df, to_go=play.to_go)
    ax = anim._fig.axes[0]
    cap = ax.text(0.01, 0.02, "", transform=ax.transAxes, ha="left", va="bottom", fontsize=9, color=style.INK,
                  zorder=21, bbox=dict(boxstyle="round,pad=0.3", fc=style.SURFACE, ec=style.THIRD, lw=1.0))
    base = anim._func

    def upd(fid, *a):
        out = base(fid, *a)
        cap.set_text(web_caption(float(tmap.get(fid, 0.0))))
        return out
    anim._func = upd
    if shutil.which("ffmpeg"):
        html = anim.to_html5_video(embed_limit=40).replace("<video ", '<video style="max-width:100%;height:auto" ')
    else:
        html = anim.to_jshtml(default_mode="once")
    plt.close(anim._fig)
    display(HTML(html))


def phases(cuts):
    """[(t_upper, caption), ...] -> function of t (seconds from the snap) for show_play."""
    def f(t):
        for up, txt in cuts:
            if t < up:
                return txt
        return cuts[-1][1]
    return f

First-and-10, early in the second quarter. The receiver split wide on the right turns and jogs toward the ball, then breaks into a sprint, running flat along the line, across the whole formation. The cornerback who had been standing seven yards in front of him turns and runs too, step for step, all the way across the field. “There’s your man coverage,” says the analyst, and before he finishes the sentence the ball is snapped, the receiver takes a little flip from the quarterback without breaking stride, and he is ten yards upfield before the corner chasing him can turn the corner. How did the analyst know? Why did the offense want him to know? And if the run was coming anyway, why didn’t the defense simply let the receiver go? This chapter is about the most common thing an NFL offense does before the snap, moving somebody, and about what each kind of movement is for.

NoteYou’ll need
  • The pre-snap rules (The Pre-Snap Phase): after a shift (two or more players moving at once) everyone must be set for a full second; in motion only one backfield player may be moving at the snap, parallel to or away from the line, never toward it.
  • Formations (Spread and Modern Formations): 2x2 is two receivers on each side, 3x1 (trips) three to one side and one to the other, a bunch is three receivers squeezed into a tight triangle, and an empty set has no back beside the quarterback. The side with more players (usually the tight end’s side) is the strength.
  • A first look at defense (A First Look at Defense): in man coverage each defender is assigned a receiver; in zone coverage each is assigned an area. The safeties’ picture is one-high (one deep in the middle) or two-high (two deep, one per side), and the box is the area near the ball where run defenders stand. Counting blockers against box defenders tells you whether the run has the numbers.

Motion has a job

The Pre-Snap Phase gave you the rules (who may move, which way, and when everyone has to freeze) and three reasons motion exists: it asks the defense a question, it gives a player a running start, and it changes the formation late. The rules set the shape: a receiver running across the formation moves flat, parallel to the line, because he may not be angling toward it at the snap, and two receivers never sprint at the snap together, because only one player may be moving. This chapter splits those three reasons into five jobs, gives every kind of motion a name, and shows you how to read the defense’s answer.

Here is the idea to hold on to: no good offense moves a player for decoration. Kyle Shanahan, whose 49ers have been among the league’s heaviest motion users since 2017, put it plainly: “You do it for a reason. You do it to help one of your players or you do it because you see something that you think will affect the defense.” Sean McVay of the Rams added the warning every coach repeats: “It’s not just to look cool.”1

Every motion does one or more of five jobs:

  1. It asks a question: man or zone? Someone on the defense has to deal with the moving player, and how they deal with him is a clue to the coverage.
  2. It changes the strength and the numbers. A player who moves from one side to the other changes how many receivers (or blockers) are on each side, and the defense has to rebalance in a second or two.
  3. It gives a running start. A player moving at the snap is already at speed when the play begins, and nobody on the defense is.
  4. It creates leverage. A defender’s position relative to his receiver, inside or outside, in front or behind, is planned against a standing receiver. A moving one changes it.
  5. It stresses communication. Each time the formation changes, the defense has to re-declare its calls, and every re-declared call is a chance for two defenders to take the same man and none to take another.

The costs from The Pre-Snap Phase still apply (play-clock time, precise timing, and the risk of tipping the play), and this chapter adds one more when it reaches motion at the snap. First, the vocabulary: the motions themselves.

CautionCommon misconception

“Jet motion” sounds like a technical term with one meaning. It isn’t. Motion names are playbook dialect: each staff names its motions in its own language, often after the letter of the player who moves (“zip” and “zoom” for the Z receiver, “yo-yo” for Y), and the same word can mean different things in two buildings. Some teams use jet and fly for the same motion; some use orbit only for a running back. The names below are the common ones in NFL film breakdowns and coaching clinics, and the glossary records them.2 Learn the shapes, and you’ll recognise the motion whatever the broadcast calls it.

Motions that cross the formation. These are the big, fast ones, almost always timed so the ball is snapped while the player is still moving.

  • Jet motion: a receiver sprints across the formation, flat along the line, timed so that the ball is snapped just before he reaches the quarterback. In the shotgun he passes between the quarterback and the line, close enough to take a handoff or a little forward flip (the jet sweep, a run to the edge by the motion man himself). That little flip travels forward, so in the rules it is a forward pass: if it hits the ground it is an incomplete pass, not a fumble.3 Most of the time he doesn’t get the ball at all: the jet is a threat, and the defense has to respect it either way.
  • Fly motion: the older name for the same idea from the Wing-T, a run-heavy offense of the 1950s onward. A wingback (a back lined up just outside and behind the tight end) sprints across behind an under-center quarterback, who turns and hands him the ball on the “fly sweep.” Many staffs say jet and fly interchangeably. Where they distinguish them, fly is usually the under-center version, with the runner passing behind the quarterback.
  • Orbit motion: a receiver comes in toward the ball and swings behind the quarterback (and behind the back), like a moon around a planet, and is still moving toward the far side at the snap. Going behind the quarterback means he can take a pitch or a shovel pass (a short underhand toss), or turn into a receiver out of the backfield, and he is harder for the defense to track because he disappears behind the offensive line for a moment.
  • Zoom motion: a receiver goes all the way across the formation and then sets on the other side, usually as the new inside receiver (#3, counting from the sideline in). It isn’t about speed; it is about the new formation. A 2x2 becomes a 3x1, and the defense has to rebalance (Job 2, below).

Motions out of the backfield.

  • Rocket motion (also rock motion): a back sprints from his spot in the backfield straight out toward the sideline, staying deep, so he is already running wide when the ball is snapped. In option offenses such as the flexbone the same name is used for a slotback (a back set just behind the line, outside the tackle) sprinting on a deep path behind the quarterback.4 The play it sets up is the rocket toss, a pitch to the back at the edge, and the rocket motion is a fine way to get a back into a pass route with a head start, too.

Motions that come back, or come in. These are subtler, and they matter most in the passing game.

  • Return motion (also yo-yo motion or yo-yo): a receiver motions one way, then reverses back toward where he started, usually right at the snap. Its targets are the defenders who follow motion: whoever chased the first move is now running the wrong way.
  • Glide motion: a wide receiver motions in toward the ball at a controlled pace, and the snap comes when he is about halfway to the formation, and he is still moving when the ball is snapped (that is the difference from short motion, which stops and sets). He starts the play already moving inside, which helps a crossing route, a quick break back outside, or a crack block (a receiver blocking back inside on a linebacker or safety, taught with the perimeter runs in Perimeter Runs and Misdirection).
  • Short motion (also rip/liz motion, after the “right” and “left” tags some playbooks use): a few yards and no more, usually a receiver tightening his split, or tucking in beside or behind a teammate to form a stack, and usually set again before the snap. Short motion looks like nothing, which is part of the point: a receiver who starts closer to the ball has more room to the outside, and a defender who lined up against his old spot is a step out of place. (Don’t confuse these rip/liz tags with the Rip/Liz strength call from Formation Rules: the same two words are a direction tag in some offensive playbooks and the defense’s strength call in others.)

Moves that change the whole formation. These are usually shifts, because more than one player moves.

  • A trade (a TE trade): the tight end changes sides. Because he leaves the line, someone else has to keep the offense legal (seven on the line, with only the end men eligible): usually a receiver on the new side steps off the line and one on the old side steps on, all at once. The strength flips, and the defense has to flip its front and its coverage calls with it. Because a trade is a shift, the offense must then hold still for a full second, which guarantees the defense that beat to re-set; offenses often follow the trade with a motion to take it back. (You saw a trade, and the defense flipping in answer, in The Pre-Snap Phase.)
  • The shift to bunch: three receivers squeeze into a tight triangle at the last moment. Against man coverage a bunch is a traffic jam that defenders have to fight through (the crossing routes it makes possible are the rub routes of the passing chapters), and creating it late means the defense can’t line up in its planned answer to bunch until it’s almost too late to communicate it.

And the timing word: motion at the snap. Motion at the snap isn’t a path. It means the player is still moving when the ball is snapped. Jet, fly, rocket, orbit, glide and return motion are almost always run this way; zoom and short motion usually aren’t. The distinction matters so much in the modern game that it gets its own section later in the chapter.

NoteMadden vs. real life

In Madden, you can send almost any player in motion from a menu before any play, in any direction you like. Real NFL motion is almost never improvised. It is written into the play call (“Gun Trips Right, Z Jet, 26 Zone”, where 26 Zone is a zone run to the right), rehearsed dozens of times in practice so that the snap arrives when the motion man reaches an exact spot, and paired with the play it is meant to help. The quarterback’s job isn’t to choose the motion. It’s to read what the defense does with it, and to snap the ball at the right instant.

Job 1: asking man or zone

Start with the question in the opening scene. When a receiver goes in motion, the defense has three basic ways to deal with him, and each one looks different.

Travel: somebody goes with him

In man coverage each defender has a receiver, so the simplest answer to motion is for that defender to go with him. A defender who follows a man in motion across the formation is said to travel with him.

To the quarterback, a corner running across the formation is about the loudest signal a defense can send. A zone defender has no reason to leave his area to follow one receiver across the field, so a traveller almost always means man coverage.

Why travel at all, if it’s so obvious? Because it keeps the defense simple. The matchups the coaches planned all week (our best corner on their best receiver) survive any motion, and nobody has to make a new call in the last two seconds. That’s a real advantage against a team that motions on most snaps.

The price shows in the last two panels. At the snap the traveller is trailing a receiver who is already running at full speed, and he himself is moving sideways, so the moment Z turns upfield the corner has to stop, redirect and chase; by panel 4 he is a step inside and behind. Anything that goes toward the side the motion is heading (a jet sweep, a flat route, a wheel up the sideline) is a race the defender begins already losing. And when the corner crosses the formation he ends up somewhere he wasn’t built to play, near the box, where a run might come straight at him. That’s why many man defenses don’t travel on every motion: on a full-speed jet across the formation, they exchange instead, passing him to a teammate on the other side (more on that in a moment).

Bump and spin: the zone answer

A zone defense doesn’t care who the receiver is; it cares where he is. So when a receiver crosses the formation, nobody follows him. The defenders adjust their spacing, each moving over one spot toward the motion, and the safeties may rotate so that the side the motion is heading to gets an extra man.

Two things happened, and both have names.

  • The bump. The linebackers each slid a step toward the motion. A bump is defenders shuffling one spot toward the motion instead of following it, each taking over a new area or a new receiver. Nobody crossed the formation; everyone just re-spaced. Coaches also call it a slide or a shuffle.
  • The spin. The two safeties rotated. A spin is one safety coming down while the other moves to the deep middle, so two deep safeties become one. The safety who comes down is usually the one on the side the motion is heading to, as here, but some calls rotate the other way on purpose, and the safety who comes down may stop in the middle of the field rather than out wide. (The rotation itself, and every variety of it, is taught in Disguise, Rotation, and Split-Field Coverage.)

Why answer motion this way? Because it keeps the numbers right without sending anyone on a long run. Once the motion is over there will be more receivers on the left, so the defense moves more men to the left. Notice the timing in the strip: at the snap Z is still just right of the ball, yet the defense has already shifted. It moves to where the jet is heading, not where he is, because a defender who waits for Z to arrive is too late. The safety who spins down becomes an extra defender on the motion side, the man who stops the jet sweep in the fourth panel. Every receiver is still inside somebody’s area, and no defender ended up on the wrong side of the formation.

The price: the defense has shown its hand. Nobody travelled, so it’s probably zone. The safeties spun, so it’s probably one-high now, with three deep defenders (the two corners and the middle safety) and four or five underneath, the shape of Cover 3. The side the motion left is thinner than it was, with a corner standing alone over empty grass. And the spin took time: a safety who must cover the deep middle had to run eight or nine yards before the snap, which is exactly why offenses love to snap the ball while he is still running.

The man/zone indicator, and the running back version

Put the two pictures together and you have the man/zone indicator: reading the defense’s reaction to motion as a clue to its coverage.

The cleanest version of the test doesn’t use a receiver at all. Motion the running back out of the backfield to a wide spot, leaving the quarterback alone (an empty formation). In man coverage, someone is assigned to that back, usually a linebacker, and that linebacker has to walk out of the box with him. In zone, the defender already standing out there (a nickel back or an outside linebacker) simply takes him, and the linebackers stay home. A linebacker walking out to the numbers with a running back is one of the most reliable pre-snap tells in football (and when it doesn’t happen, the next section shows you what to check before you call zone). It also tells the quarterback that the middle of the field has one fewer defender in it, which is why quarterbacks who can run love to see it.

The same reading works when the tight end motions. In man, whoever has him (often a safety or a linebacker) goes with him; in zone, nobody follows and the defense re-declares its strength toward his new side. Because the tight end is also the run strength, expect the front to shift with him as well as the coverage.

These are this book’s words. Defensive staffs have their own: following the motion man is also called locking or tracking him, the man-coverage hand-off a switch, and the safety rotation a rotate; when two defenders swap two receivers who cross, the call is a banjo.5 If you read a breakdown that uses different words, match the shapes, not the labels.

ImportantWhy it exists

The man/zone indicator exists because coverage is the defense’s best-kept secret, and motion is the cheapest way to make it leak. A defense can hide its coverage while everyone stands still. It can’t avoid reacting to a player who moves, and each way of reacting has a different shape. For the offense the information is worth a lot: man coverage invites crossing routes, rubs and quarterback runs; zone invites throws into the gaps between defenders. A concrete version: the call has two receivers crossing on one side and a seam route between the linebackers; if the motion says man, the quarterback expects the crossers to rub their defenders off and throws to the one who comes free; if it says zone, he looks for the seam in the hole between the linebackers. When the motion sets early, the quarterback has time for more: with a kill call or a built-in check he can swap the play for one that suits the answer. When the motion is at the snap, the answer arrives too late to change the play; it feeds only his read after the snap. Motion that sets early buys a check; motion at the snap buys a head start.

When the indicator lies

Defensive coordinators know exactly what you just learned, and they spend their careers making it unreliable. Three counters are worth knowing now; the full treatment is in Disguise, Rotation, and Split-Field Coverage.

Man defenses that exchange. A man defense doesn’t have to travel. Many play man with a rule that says, in effect, “if he crosses the formation, pass him on.” That is an exchange: nobody follows the motion man across; instead the defenders slide one spot over and swap receivers, so the defender on the side he arrives at picks him up and the one he left behind takes a new job. The coverage is still man, but nobody ran across the field, so from the stands it looks exactly like a zone bump. Defenses often decide this by the kind of motion: they might travel on a short motion (cheap to follow) and exchange on a full-speed jet (expensive to chase).

How do you tell them apart? Not by the slide itself, which is the same, but by where the slid defenders end up:

  • Distance and leverage. In man, the defender who picked up the motion man stops close to him (often two to four yards off) and on one side of him, usually inside, squared up to that receiver. In zone, he stops at area depth (five to seven yards) in the space between receivers, and doesn’t seem to care which one is nearest.
  • Eyes. A man defender looks at his receiver; a zone defender looks into the backfield at the quarterback. On a tight shot you can often see which.
  • Everyone else. Man teams tend to show it elsewhere too: corners pressed or tight with inside leverage, a single deep safety. If one defender exchanged and the rest look like man, believe man.
  • After the snap. The final answer: man defenders turn and run with receivers, zone defenders drop to spots. That is a replay check, not a pre-snap one, but it is how you keep score.

One practical limit: the standard broadcast angle, the high sideline camera behind the offense that How to Watch a Broadcast described, usually shows the box and the slot defenders at the snap, but often not the far corner or the deep safeties. A linebacker walking out, or a nickel travelling, you can read live. A spin, or a corner on the far side, you usually confirm on the replay or the All-22.

Zone defenses that follow. Modern zones are pattern-matching coverages: each defender starts with an area but plays the receiver who comes into it like man coverage. Under some of those rules a defender follows the motion man a few steps before passing him on. The quarterback sees a defender move with the motion and thinks man; the coverage is zone.

Fake spins. A safety spins down with the motion, the quarterback concludes “one-high,” and at the snap the safety backs out again into a two-high shell. Defenses that disguise heavily rehearse every rotation from several starting pictures for exactly this reason.

That last counter is why offenses have pushed motion closer and closer to the snap. If the ball is snapped while the motion man is still running, the defense has to react once, at full speed, with no time for a second move. A defender who travels at the snap really is chasing; a safety who spins at the snap really is rotating. Motion at the snap doesn’t make the defense tell the truth, but it leaves much less time to lie.

Is the indicator wearing out, now that almost every offense motions on most plays and every defense has drilled its answers for years? Probably somewhat, but there is no clean public measure of it yet. Disguise, Rotation, and Split-Field Coverage shows how to measure a defense’s lie with tracking data, and Tendencies: PROE, Personnel, Motion, and Coverage how to check a team’s motion habits against its coverage. For watching, treat the indicator as one strong clue to combine with the shell and the alignment, never as the answer.

This is the arithmetic every motion forces. The defense has eleven players, and each one it moves toward the motion comes from somewhere. If a linebacker walks out, the box gets lighter, and the run game gets the numbers advantage that A First Look at Defense taught you to count. If a safety rotates over, the deep coverage changes and the lone receiver on the other side may be in a one-on-one with no help over the top. If the corner follows, the coverage is man and the quarterback knows it. Offensive coordinators design motions to make the defense choose among those three, then call plays that attack whichever spot the defense gave up.

The trade does the same thing to the defensive front. Defensive linemen and linebackers set their alignment to the tight end: the bigger end lines up across from him, the strong-side linebacker walks out to his side, and the strong safety cheats toward him. When the tight end trades sides, all of that has to flip in the two or three seconds before the snap, which is why offenses that like to run at the strength love to trade at the last moment. A defense that flips late and loses a gap in the confusion is a run waiting to happen; a defense that refuses to flip has its strength on the wrong side. The defense does get one guaranteed beat: a trade is a shift, so the offense must then stand still for a full second. Defenses use it by declaring the strength late, not setting the front until the shift is over, and offenses answer by following the trade with a motion, the “shift, freeze, motion” sequence from The Pre-Snap Phase.

The shift to bunch attacks the coverage the same way. Most defenses have a pre-planned answer to a bunch, because man coverage against three receivers in a tight triangle invites collisions and picks. If the bunch appears with two seconds left, that answer has to be called, heard and applied by four or five defenders at once.

Motion changes the numbers for the run game too. A receiver or a fullback who motions toward the ball can become an extra blocker: a lead blocker through the hole, or a receiver arriving with momentum to crack back on the defender who has to set the edge. Either way the offense has added a blocker, or a better blocking angle, on the side of the run, and every player who arrives in the box adds a gap the defense must fill. The blocks themselves are taught in Zone Running and Perimeter Runs and Misdirection.

Why shift instead of motion?

The trade and the bunch are shifts, not motions, and a shift buys different things. Several players can move at once: the tight end, a back and a receiver can all change spots, so the offense can build a formation no single motion could reach (the strength flipped, the back on the other side, a receiver now on the line), and the defense has to re-declare almost everything. It also lets the offense line up in one look, watch the defense align to it, and then move into the formation it really wants. The price is the one-second freeze: after a shift everyone must be set for a full second, so the shift itself gives nobody a running start, and the defense gets a full beat to re-set. That is why the modern pattern is so often both: shift to change the formation, freeze, then motion one man to get the running start and ask the man-or-zone question.

Jobs 3 and 4: the running start and leverage

The third and fourth jobs are about the moment the ball is snapped, and they are really one idea: a player who is moving at the snap begins the play somewhere the defense didn’t plan for, at a speed the defense doesn’t have.

The running start

How big is a running start? You can estimate it with simple physics. A football player accelerates roughly like a sprinter: hard at first, then levelling off near his top speed. The chart below compares two receivers with the same top speed, one starting from a standstill and one already running at jet-motion pace when the ball is snapped.

Three yards in a second is enormous. It is the difference between a jet sweep that reaches the edge before the defensive end can widen and one that is tackled for no gain; between a receiver who is covered and one who is open. A defender can match a receiver’s speed. He can’t match a head start, because he isn’t allowed to have one: he must react to the snap like everybody else.

One catch: the head start points in the direction of the motion, which the rules make sideways or backward. A play that keeps going that way gets nearly all of it: a jet sweep, a flat route, a wheel up the sideline, a crack block. A receiver who has to turn upfield, or cut back the other way, spends some of it on the turn, and a return motion deliberately throws it away to punish the defender who matched it. That is why offenses kept looking for motions that end pointed nearly upfield at the snap, which is exactly what the “cheat motion” later in this chapter was about.

In the model, the lead depends on two things: how fast the motion is run, and how long the standing player still needs to reach top speed. That is why offenses with very fast receivers motion so much: a head start is worth most to receivers who run their motion at a high speed and still have a long acceleration phase ahead of them, and very fast receivers tend to do both. It’s also why defenses hate pressing a receiver who is in motion: press coverage (a defender lined up right on the receiver to jam him at the line) depends on getting hands on a man who is standing still. You can’t jam somebody who is already running past you.

Leverage: start where he didn’t plan for

Every defender aligns with leverage, a deliberate position inside or outside his receiver that tells him which way to force him and where his help is. That alignment is planned against a receiver who is standing still. Motion changes it in a way the defender can’t fully fix:

  • A defender who travels with motion is, by definition, behind the receiver in the direction he is moving. A receiver moving inside at the snap (glide motion) is already inside his defender, which helps routes that break inside. A receiver moving outside at the snap (orbit, rocket) is already outside his.
  • A defender who bumps is standing in a spot chosen for the old formation, a step or two from where he would have lined up if the offense had started in the new one.

The most famous way to exploit the first of these is the return motion.

Look at the defender’s problem in the third panel. A man defender’s whole technique is to mirror the receiver, and here the receiver has reversed direction at the one instant the defender can’t anticipate. Every step the corner took following the motion is now a step in the wrong direction. Against zone the same motion does little (the zone defenders bump and then sit in their areas), which is why return motion is called most often when the offense expects man coverage: near the goal line, on third-and-short, against blitzes.

The jet as a fake

Most jet motions end with the ball going to somebody else. That’s not wasted motion. A receiver sprinting across the formation at the snap is a threat to run a sweep, so the defenders responsible for the edge on that side must take a step toward him, and the linebackers’ eyes go with him. A run the other way, or a play-action pass behind the flowing linebackers, now has one fewer defender in its way, or one who is a step late. Offenses that run a lot of jet motion build their plays in pairs: the jet sweep itself, and the plays that take advantage of the defense respecting the jet sweep. That pairing, and why the sweep has to be run often enough to stay believable, is the subject of Perimeter Runs and Misdirection.

Job 5: stressing the defense’s communication

The fifth job doesn’t show up on a diagram, but defensive coaches talk about it constantly. Every change in the formation obliges the defense to re-communicate: the linebacker who makes the front call (usually the Mike, who tells the linemen which way to align) re-declares the strength; the safeties re-call the coverage check (the pre-planned adjustment of the called coverage to the formation in front of them, since many coverages change when a 2x2 becomes a 3x1); the corners confirm who has whom; and in a man defense the players involved have to agree in one word whether they’re travelling or exchanging.

All of that happens with the offense deciding when the ball moves. If the motion starts with ten seconds on the play clock, the defense has time. If it starts with three, and the ball is snapped as the motion man passes the quarterback, the defense has one chance to get every call right, often in a loud stadium. The typical failure isn’t dramatic: two defenders take the same receiver and nobody takes another, or a safety rotates the wrong way. But one blown assignment is usually all an NFL offense needs for a big play.

The stress runs both ways. The offense, too, has to execute the motion perfectly: the motion man has to start on the right signal, run at the right pace and be at the right spot when the ball is snapped, and the center has to snap it at precisely that moment. You can often see the machinery on the broadcast. In the shotgun, the quarterback commonly starts the motion with a leg lift or a tap of his heel or hand, with no words at all, and the snap then comes on a count the center and quarterback have timed to the motion man’s landmark.6 A motion that arrives early leaves the receiver standing and waiting; one that arrives late leaves the quarterback holding the snap while the defense sorts itself out. And a rhythm that never varies is one the defense can learn, which matters in the next section.

Motion at the snap

Everything above points to one conclusion: the later the motion, the more it’s worth. A motion that ends two seconds before the snap gives the defense two seconds to adjust, disguise and re-communicate. A motion that is still going when the ball is snapped gives it none. Through the late 2010s and 2020s, NFL offenses learned that lesson fast.

ESPN Analytics, which charts specifically whether a player is moving at the snap, found that the average team did it on about 4% of its plays in 2017 and about 22% in 2023.7 The 2023 Miami Dolphins, coordinated by head coach Mike McDaniel, were the extreme: early that season ESPN had them with a player in motion at the snap on 59% of plays, by far the most of any team since ESPN began tracking in 2017.8 By 2025 other offenses had caught up: according to Next Gen Stats, the Atlanta Falcons under offensive coordinator Zac Robinson led the league in 2025 with motion at the snap on 60.8% of their plays (and used some pre-snap motion on 70.1%, sixth in the league).9

Why is motion at the snap so valuable? Because it does jobs 3, 4 and 5 at full strength at once: the running start is at full value, the leverage can’t be fixed after the snap, and the defense’s adjustment, with its communication, has to happen in real time, with no time left to disguise.

And what does it cost? The motion man can’t be a pass blocker, and his route has to start from wherever the motion takes him. The timing has to be perfect on every snap. The quarterback gives up the chance to check the play after seeing the defense’s reaction: the information arrives in time for his read after the snap, not for his call before it. And it hands the defense one piece of information that defensive coordinators have learned to use: if the snap comes when the motion man reaches the quarterback, the defense knows when the snap is coming. A defense that times its own movement (a blitzer creeping up, a safety rotating) to the motion man’s arrival can move at the last instant without risking an offside penalty. Writing in January 2024, Ben Solak of The Ringer pointed out that the two defenses then meeting in the AFC Championship Game, Steve Spagnuolo’s Chiefs and Mike Macdonald’s Ravens, ranked first (Baltimore) and sixth (Kansas City) in EPA per play allowed on snaps with motion, according to Next Gen Stats, and that both made a habit of changing the whole picture as the motion triggered the snap.10 The arms race moves to the last half-second.

The “cheat motion” and the 2024 rule language

Motion at the snap reached the edge of the rulebook in 2023. Instead of sending a fast receiver all the way across the formation, the Dolphins (and soon the 49ers and Rams) lined him up tight, close to the tackle, and had him take a short, quick motion outward, toward his own sideline, just before the snap, so that he could turn upfield at full speed the instant the ball moved: a sprinter’s rolling start. Kyle Shanahan called it a “little cheat motion” and admitted the 49ers had copied it “in Week 2 after we saw it.”11 Reporters noticed that Tyreek Hill sometimes seemed to be turning upfield a fraction early, but he was never flagged for illegal motion all season.

Was it legal? The rule said a player may not be moving toward the line at the snap, and the officials judged that these motions were sideways at the moment the ball moved. For 2024 the league added language to the false-start rule that was widely reported as aimed at the trick. An eligible receiver who is on the line and moves backward or changes his stance (from a two-point to a three-point stance, or back) now has to come to a complete stop before the snap, or it’s a false start; a player in the backfield who changes his stance doesn’t have to stop, as long as his movement isn’t abrupt (a false start) or forward (illegal motion).12

Read that language next to the trick and you’ll notice they don’t quite meet. The cheat motion was run by a receiver who had to be off the line to go in motion at all, moving sideways; the new sentence is about receivers on the line stepping back or changing stance. League officials later told reporters exactly that: it was a clarification, not a rule change, aimed at abrupt stance changes, and it did not outlaw sideways motion that turns upfield at the snap. So the honest summary is that the cheat motion itself survived. What the clarification ruled out was a neighbouring way to steal a start: a receiver on the line rocking back or dropping into a stance at the last instant.

The rulebook: NFL, college and Canada

You have the NFL’s rule from the “You’ll need” box: one backfield player, moving parallel to or away from the line. Two details are worth adding.

First, a player who is on the line can’t go in motion at all without first getting off it, and that has real consequences. In Super Bowl LVII, when the Chiefs wanted to run their return-motion play a second time, Patrick Mahomes signalled for Kadarius Toney to go in motion, but Toney was lined up on the line of scrimmage and couldn’t. Toney pointed across the field at Skyy Moore, who was in the backfield, and Moore went instead; he scored the touchdown.13

Second, the rule’s purpose is the same as the one-second set’s: the offense may move before the snap, but it may not get a running start at the line of scrimmage. A sideways head start is allowed; a forward one isn’t.

College football’s rule is essentially the same as the NFL’s: one player, not moving toward the opponent’s goal line at the snap, and a player who starts on the line has to become a back and come to a complete stop before he can go in motion.14 The big outlier is Canadian football. In the CFL, every backfield player except the quarterback may be in motion at the snap, in any direction, including forward.15 Canadian receivers routinely hit the line of scrimmage at full speed as the ball is snapped (a “waggle”), which is one reason the Canadian game, on a bigger field with twelve players a side, is so pass-heavy. If you’ve wondered what would happen if the NFL allowed forward motion, the CFL is the experiment: receivers that are nearly impossible to jam, and defenders who line up far off the ball to avoid being run past.

The motion boom, by the numbers

Motion is as old as the T formation (the Bears’ man in motion in the 1940 championship game is in The Pre-Snap Phase), and some coaches made pre-snap movement a signature long before it was fashionable. Tom Landry’s Cowboys of the 1960s to 1980s famously had their offensive linemen stand up in unison and then drop back into their stances while the backs shifted behind them, the “Landry shift,” partly to hide the new backfield alignment behind a wall of standing linemen.16 But for most of NFL history motion was a tool for particular plays, not a habit.

That changed in the late 2010s. Kyle Shanahan (49ers head coach from 2017) and Sean McVay (Rams head coach from 2017) built offenses in which motion was part of almost every call, and as their assistants became coordinators and head coaches around the league, the habit spread. By ESPN’s count, the 49ers used pre-snap motion on 69.1% of their plays from 2017 through 2021, the most in the league, and on 76.9% in 2021 alone, while the league as a whole rose from 34.9% (2017) to 46.8% (2021).17 PFF’s longer shift-or-motion series, which The Pre-Snap Phase quoted, shows the same climb to a record 63.9% in 2025, when every team except the Giants moved someone before at least half of its snaps.18

Why the boom? Three reasons, each from earlier in this chapter. Defenses started hiding their coverage better: Next Gen Stats measured two-high (split-safety) coverage on 32.9% of plays in 2018 and 42.0% in 2025, the highest since its tracking began,19 and two-high shells are built to rotate, so offenses needed a way to make them commit. The head start became more valuable as offenses filled with very fast receivers. And it worked visibly for the offenses that did it most, so it was copied.

Four things make the numbers differ, and each is worth knowing before you quote one:

  1. What counts as movement. PFF counts shifts and motion. FTN’s is_motion flag means “motion occurred on the play before or at the time of the snap.” ESPN’s headline series counts only a player still moving at the snap, a much smaller set. Next Gen Stats publishes yet another pair, total pre-snap motion and motion at the snap: Atlanta’s 70.1% and 60.8% in 2025, quoted above.
  2. Who or what does the charting. PFF and FTN use human charters watching video; Next Gen Stats uses the player-tracking chips. A human charter and a tracking threshold can disagree about whether a receiver’s two-step shuffle “counts.”
  3. Which plays are in the denominator. All snaps, or only runs and passes? Are kneel-downs, spikes and penalty-only plays included? Playoffs? These shift a rate by a point or two.
  4. Which seasons exist. FTN’s charting in nflverse starts in 2022, ESPN’s motion-at-the-snap series in 2017, PFF’s runs back further.

The rule this book follows: every motion number carries its source, and two sources never share an axis. When you compare teams or seasons, compare them inside one series.

Does motion work?

The honest answer is “yes, but not in the way a simple average suggests.” In FTN’s data, plays with motion and plays without it look surprisingly alike. On first and second downs in 2025, passes with motion averaged about +0.10 EPA per play against +0.06 without, and runs about −0.05 against −0.08; in 2024 the two were nearly identical, and in 2022 passes without motion did better.20 Teams that motion more did have slightly better offenses in each of the four seasons, but the relationship is loose.

CautionCommon misconception

“Motion plays gain more, so offenses should motion on every play.” The averages above don’t show that, and even if they did, the comparison wouldn’t prove it. Offenses choose when to motion: some save it for their best-designed plays, some motion on almost everything, and defenses adjust to teams that motion a lot. The plays with and without motion aren’t a controlled experiment. Other comparisons find more: ESPN’s numbers for 2017–2023, as reported by The Ringer in 2024, had runs with motion at the snap gaining about 0.11 EPA per play more than runs with no motion, and dropbacks about 0.08 more.21 That is a simple with-and-without comparison too, though, so it carries the same caution. Part of motion’s value is also invisible in its own plays: a motion that always meant “run” would be a tell, so offenses use the same motions on different plays to keep the defense from reading them. Measuring that kind of value is part of Tendencies: PROE, Personnel, Motion, and Coverage.

Film room

NoteFilm room: Miami Dolphins 2023, motion at the snap

No offense in the tracking era moved like Mike McDaniel’s 2023 Dolphins. In FTN’s charting they used motion on 79.4% of their runs and passes, the highest team-season from 2022 to 2025; ESPN had them with a player in motion at the snap on 59% of plays early that season, a record. They scored 70 points against Denver in Week 3, and Tyreek Hill led the NFL with 1,799 receiving yards.22

What to notice when you watch the film: the motion is short and timed. Hill or Jaylen Waddle moves a few yards, sometimes in toward the formation, sometimes out toward the sideline (the “cheat” version), and the ball is snapped as he gets going. The defense rarely gets a clean look at its own assignment, and Hill starts every route already running. Then watch the defenses that slowed Miami down. The Chiefs, who beat the Dolphins twice that season (in Frankfurt in November and in the wild-card round), mostly refused to chase: in the November game they played zone on a season-high 61% of Miami’s dropbacks, by ESPN’s count, with Steve Spagnuolo’s instruction to keep “22 eyeballs” on Hill, and L’Jarius Sneed matched Hill when they did play man.23 McDaniel was let go by Miami in January 2026 and is the Chargers’ offensive coordinator for the 2026 season.

NoteFilm room: Kansas City Chiefs, Super Bowl LVII (February 12, 2023)

Two of the most famous motion plays of the decade came less than three minutes of game clock apart. In the fourth quarter against the Eagles, Kansas City ran a play its coaches call “Corn Dog”: a receiver motions in toward the formation, then turns and runs back toward the sideline at the snap, a return motion built to beat man coverage. Kadarius Toney caught a 5-yard touchdown on it with 12:04 left to give the Chiefs the lead, and Skyy Moore scored from 4 yards on a mirrored version at 9:22. Both times the Eagles were in man coverage, the defender went with the first move, and the receiver was alone when he turned around. Kansas City won 38–35. A year later in Super Bowl LVIII the Chiefs ran a renamed relative of the play for Mecole Hardman’s winning touchdown in overtime.24

What to notice: the motion does two jobs on one play. First it asks the question (did the defender follow?), and then it attacks the answer (the defender who followed is moving the wrong way). Andy Reid’s offense has used motion that way for years, including orbit motion that loops a receiver behind the quarterback and out to the flat, pulling defenders with him while the real target comes open elsewhere.25

NoteFilm room: San Francisco 49ers 2021, Deebo Samuel

In 2021 the 49ers motioned on 76.9% of their plays, the most in the league by ESPN’s count. Much of it went through Deebo Samuel, a wide receiver who spent part of the season lining up and motioning into the backfield. Motion let Shanahan move Samuel from receiver to ball carrier and back again before the snap, so the defense never knew which position it was defending. He caught passes for 1,405 yards and ran for 365 more, with eight rushing touchdowns.26

What to notice: the motion creates the matchup. When Samuel motioned from a receiver spot into the backfield, the defender assigned to him in man coverage had to follow him into the box, where a cornerback or safety is suddenly playing run defense against a 215-pound receiver with a head start. Against zone, the defense now faced a backfield with two ball carriers in it, one of them very fast, and had to respect both.

NoteFilm room: Los Angeles Rams 2017–2025, the same look, different plays

McVay’s Rams were not the league’s most frequent motion team in his first seasons (ESPN had them eighth from 2017 to 2021), but they made the jet motion famous, and by 2024 FTN charted them with motion on 76.7% of their plays, the highest in the league that season. The Rams’ trademark is pairing: over a game, the same jet look comes before an outside-zone run, a jet sweep and a play-action pass, so the motion tells the defense nothing about the play. What to notice: the play was called in the huddle, so the defense’s reaction doesn’t change this play; it tells you which one should work. Watch the linebackers when the jet man crosses the quarterback. If they jump toward him, the run the other way should hit; if they stay home, expect the Rams to give it to the jet man on a later snap.

Two habits make the card work. Say each answer out loud, or write it down, before the snap. A guess you didn’t commit to can’t teach you anything. And read the lines in order. Watch for it on Sunday

TipWatch for it: follow the motion

For one quarter, every time an offensive player moves before the snap, do four things.

  • Name it. Jet, orbit, return, zoom, glide, short, a trade, a shift? If you aren’t sure, describe the path (“across, in front of the quarterback”).
  • Watch who follows. Pick up the defender who was across from the motion man before he moved. Did that defender travel across (T), did the defenders bump over (B), did a safety spin (S)? Write the letters. If it was a bump, look at where the slid defender stopped: tight on one man (an exchange, so man) or in a space between receivers (zone)?
  • Call it. Man or zone? One-high or two-high at the snap? Write your call before the ball is snapped.
  • Note the timing. Was the ball snapped while he was still moving (motion at the snap), or did he set first?

Then check your calls on the replay, or better, on the All-22 film, which shows the whole secondary. Add a column for it in your prediction log from How to Watch a Broadcast. After a few games you’ll notice two things: travellers are rarer than you expect (many man defenses exchange instead), and your calls get better when the motion is at the snap.

Man coverage, with one deep safety: Cover 1, almost certainly. Two things happened at once. Takeaways

  • Motion is never decoration. Every motion does at least one of five jobs: it asks man or zone?, changes the strength and the numbers (receivers or blockers), gives a running start, creates leverage, or stresses the defense’s communication. A shift moves several players and rebuilds the formation, but it costs a one-second freeze and gives nobody a running start.
  • Learn the shapes, not just the names. Jet, fly and orbit cross the formation (in front of, behind, and looping behind the quarterback); rocket sprints a back to the edge; return goes and comes back; zoom crosses and sets; glide comes in and is still moving at the snap, short motion comes in and sets; a trade or a shift to bunch rebuilds the formation. Names vary by playbook.
  • Read the answer, then check it. A defender who travels suggests man; defenders who bump suggest zone; safeties who spin turn two-high into one-high. But a man defense can exchange, sliding over just like a zone, so look at where the slid defenders stop: tight on one receiver means man, sitting in a space between receivers means zone. Pattern-matching zones can follow, and safeties can spin back. This read is line 4 of the checklist so far: situation → personnel → formation and strength → the motion response → box → shell, then log it.
  • The later the motion, the more it’s worth. Motion at the snap gives a running start worth about three yards in the first second (in the direction of the motion) and leaves the defense no time to disguise, but it also leaves the quarterback no time to check the play. NFL use of it rose from about 4% of plays in 2017 to over 20% by 2023 (ESPN).
  • The rules shape the motion. American rules allow one backfield player moving sideways or back at the snap; Canadian rules let backs move forward. The 2024 NFL language on receivers who change stance on the line was a clarification that left the sideways “cheat motion” legal.
  • Motion numbers depend on the source. PFF (any shift or motion: 63.9% in 2025), FTN (motion before or at the snap: 55.1%), ESPN and Next Gen Stats (motion at the snap) all measure different things. Quote one source at a time.
