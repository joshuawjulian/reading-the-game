UC, GUN, TB_D = -1.9, -5.0, -7.0          # QB under center, QB in the shotgun, deep tailback (yards)
TE_RING = style.LINE_TO_GAIN               # gold ring: a tight end (as in the personnel chapter)
RB_RING = style.THIRD                      # aqua ring: a running back
SIDELINE = 26.67                           # yards from the middle of the field to either sideline


def line5():
    """The five offensive linemen, unlettered. (In this chapter "C" always means a cornerback, so the
    center, always the middle lineman, carries no letter.)"""
    return [Player(k, k, "off", LINE_D, w, label="")
            for k, w in [("LT", -2 * OL_SPLIT), ("LG", -OL_SPLIT), ("C", 0.0),
                         ("RG", OL_SPLIT), ("RT", 2 * OL_SPLIT)]]


def P(key, pos, d, w, label=None):
    """An offensive player at (d, w): d = yards from the ball (negative = backfield), w = yards to the
    offense's right."""
    return Player(key, pos, "off", d, w, label=label or key)


def D(key, pos, d, w, label=None):
    return Player(key, pos, "def", d, w, label=label)


def check_legal(players):
    """Seven men on the line, and only the two at its ends may be eligible receivers."""
    off = [p for p in players if p.side == "off"]
    line = sorted([p for p in off if abs(p.d - LINE_D) < 0.05], key=lambda p: p.w)
    assert len(off) == 11, f"{len(off)} offensive players"
    assert len(line) == 7, f"{len(line)} on the line: {[p.key for p in line]}"
    for p in line[1:-1]:
        assert p.pos not in ("WR", "TE", "RB", "FB"), f"covered receiver {p.key}"
    return players


def ring_bodies(p):
    """Ring the backs (aqua) and tight ends (gold), so the personnel stays countable."""
    for k, pl in p.players.items():
        if pl.side == "off" and pl.pos in ("RB", "FB"):
            p.highlight(k, color=RB_RING)
        elif pl.side == "off" and pl.pos == "TE":
            p.highlight(k, color=TE_RING)
    return p


def hide_yard_numbers(ax):
    """Drop the faint painted field numbers (they sit where the receivers stand in these zoomed views)."""
    for t in list(ax.texts):
        if t.get_text().isdigit() and t.get_alpha() is not None and abs(t.get_alpha() - 0.55) < 1e-6:
            t.remove()


def pxy(play, fld, d, w):
    x, y = fld.xy(*play.to_abs(d, w))
    return float(x), float(y)


def put(ax, play, fld, text, d, w, size=7.5, color=style.INK, ha="center", va="center", weight="normal",
        boxed=True, italic=False):
    """A short on-field label at (d, w)."""
    x, y = pxy(play, fld, d, w)
    kw = dict(bbox=dict(boxstyle="round,pad=0.22", fc=style.SURFACE, ec=style.BASELINE, lw=0.5)) if boxed else {}
    ax.text(x, y, text, fontsize=size, color=color, ha=ha, va=va, fontweight=weight, zorder=12,
            style="italic" if italic else "normal", linespacing=1.1, **kw)


def shade(ax, play, fld, pts, color, alpha=0.12, ls="-", lw=0.8, z=1.5):
    """Shade a polygon given as (d, w) points."""
    xy = [pxy(play, fld, d, w) for d, w in pts]
    ax.add_patch(Polygon(xy, closed=True, fc=color, ec="none", alpha=alpha, zorder=z))
    ax.add_patch(Polygon(xy, closed=True, fc="none", ec=color, alpha=min(1.0, alpha * 5), lw=lw, ls=ls, zorder=z))


BOX_W, BOX_D = 6.0, 7.5        # the drawn "box": about six yards either side of the ball, 7.5 deep


def shade_box(ax, play, fld):
    shade(ax, play, fld, [(0.3, -BOX_W), (BOX_D, -BOX_W), (BOX_D, BOX_W), (0.3, BOX_W)], style.DEFENSE,
          alpha=0.07, ls=(0, (3, 2)))


def box_count(play):
    return sum(1 for p in play.players.values()
               if p.side == "def" and abs(p.w) <= BOX_W and 0 < p.d <= BOX_D)


def branch(ax, play, fld, pts, color=style.OFFENSE, ls="-", lw=1.7, z=4):
    """A route-style arrow drawn by hand through absolute (d, w) points (used for 'either/or' branches)."""
    xs, ys = zip(*[pxy(play, fld, d, w) for d, w in pts])
    ax.plot(xs[:-1] + (xs[-1],), ys[:-1] + (ys[-1],), color=color, lw=lw, ls=ls, zorder=z,
            solid_capstyle="round")
    ax.add_patch(FancyArrowPatch((xs[-2] + 0.8 * (xs[-1] - xs[-2]), ys[-2] + 0.8 * (ys[-1] - ys[-2])),
                                 (xs[-1], ys[-1]), arrowstyle="-|>", mutation_scale=11, color=color, lw=lw,
                                 shrinkA=0, shrinkB=0, zorder=z))


def receiver_numbers(ax, play, fld, d_lab=1.45, size=7.0, at=None):
    """Write #1, #2, #3 ... above each receiver, counting from the sideline in on each side.
    Backs in the backfield (within 3 yards of the ball, behind the line) are not counted. When two
    receivers share a width (a stack), the one on the line counts first. `at` = {key: (d, w)} moves
    a label (for clusters, where the default spots would collide)."""
    at = at or {}
    rec = [p for p in play.players.values() if p.side == "off" and p.pos in ("WR", "TE", "RB", "FB")
           and not (abs(p.w) < 3.0 and p.d < -2.0)]
    for sgn in (-1, 1):
        side = sorted([p for p in rec if p.w * sgn > 0], key=lambda p: (-round(abs(p.w), 1), -p.d))
        for i, p in enumerate(side, 1):
            d, w = at.get(p.key, (d_lab, p.w))
            put(ax, play, fld, f"#{i}", d, w, size=size, color=style.INK_2, boxed=False, weight="bold")


def grid(nrows, ncols, lateral, window, width=6.6, extra=0.38):
    pw = width / ncols
    ph = pw * (window[1] - window[0]) / (lateral[1] - lateral[0]) + extra
    return plt.subplots(nrows, ncols, figsize=(width, nrows * ph), squeeze=False)


def title(ax, text, sub=None):
    ax.set_title(text, loc="left", fontsize=9.5, fontweight="bold", color=style.INK, pad=3)
    if sub:
        ax.text(1.0, 1.0, sub, transform=ax.transAxes, ha="right", va="bottom", fontsize=7.5,
                color=style.INK_2)

def form(name):

    f = {
        "2x2": [q, rb, P("X", "WR", LINE_D, -15.5), P("H", "WR", OFF_BALL, -8.5),
                P("Y", "TE", OFF_BALL, 8.5), P("Z", "WR", LINE_D, 15.5)],
        "3x1": [q, rb, P("X", "WR", LINE_D, -15.5), P("Y", "TE", OFF_BALL, 7.0),
                P("H", "WR", OFF_BALL, 11.2), P("Z", "WR", LINE_D, 15.5)],
        "trey": [q, rb, P("X", "WR", LINE_D, -15.5), P("Y", "TE", LINE_D, TE_W),
                 P("H", "WR", OFF_BALL, 10.0), P("Z", "WR", OFF_BALL, 15.5)],
        "empty": [q, P("X", "WR", LINE_D, -15.5), P("H", "WR", OFF_BALL, -8.5),
                  P("RB", "RB", OFF_BALL, 6.0, "R"), P("Y", "TE", OFF_BALL, 10.5), P("Z", "WR", LINE_D, 15.5)],
        "bunch": [q, rb, P("X", "WR", LINE_D, -15.5), P("H", "WR", LINE_D, 8.6),
                  P("Y", "TE", -2.2, 7.2), P("Z", "WR", -2.2, 10.0)],
        "stack": [q, rb, P("X", "WR", LINE_D, -15.5), P("H", "WR", OFF_BALL, -8.5),
                  P("Z", "WR", LINE_D, 11.0), P("Y", "TE", -2.5, 11.0)],
        "nub": [q, P("RB", "RB", GUN, -2.2, "R"), P("Y", "TE", LINE_D, -TE_W), P("H", "WR", OFF_BALL, 7.0),
                P("Z", "WR", OFF_BALL, 11.2), P("X", "WR", LINE_D, 15.5)],
        "quads": [q, P("X", "WR", LINE_D, -15.5), P("RB", "RB", OFF_BALL, 5.8, "R"),
                  P("Y", "TE", OFF_BALL, 9.1), P("H", "WR", OFF_BALL, 12.4), P("Z", "WR", LINE_D, 15.8)],
    }[name]

def plate(ax, name, head, sub, numbers=True, lateral=PLATE_LAT, window=PLATE_WIN, at=None):

    if numbers:

    title(ax, head, sub)
    return p, fld

# --- Frame strips (PDF) and videos (web) for the two animations. ------------------------------------------
MARK_R = 0.62


def tracking(play):
    df = play.to_tracking()
    df["_id"] = df["displayName"].astype(str)
    return df


def panel(ax, play, df, t, lateral, window, rings=()):
    """One moment of a simulated play: markers, trails, rings."""
    tmap = df.groupby("frameId")["time"].first()
    fid = int((tmap - t).abs().idxmin())
    ylo, yhi = sorted((play.ball_y - lateral[0], play.ball_y - lateral[1]))
    fld = Field(ax=ax, orient="vertical", los=play.los, window=window, numbers=False, hashes=True,
                lateral=(ylo, yhi), to_go=play.to_go)
    cur = df[df["frameId"] == fid]
    shown = cur[cur["x"].between(play.los + window[0] + 0.5, play.los + window[1] - 0.5)
                & cur["y"].between(ylo + 0.5, yhi - 0.5)]
    trails = df[(df["frameId"] <= fid) & df["_id"].isin(set(shown["_id"])) & (df["side"] != "ball")]
    for (_, side), g in trails.groupby(["_id", "side"]):
        xs, ys = fld.xy(g["x"].values, g["y"].values)
        ax.plot(xs, ys, color=style.OFFENSE if side == "off" else style.DEFENSE, lw=0.9, alpha=0.4, zorder=3)
    ppy = fld.pts_per_yard()
    for key, col in rings:
        rr = cur[cur["displayName"] == key]
        if len(rr):
            x, y = fld.xy(rr["x"].iloc[0], rr["y"].iloc[0])
            ax.add_patch(Circle((float(x), float(y)), MARK_R * 1.65, fc="none", ec=col, lw=1.7, zorder=7))
    for _, row in shown.iterrows():
        x, y = (float(v) for v in fld.xy(row["x"], row["y"]))
        if row["side"] == "ball":
            ax.add_patch(Ellipse((x, y + 0.5), 0.75, 0.45, fc=style.BALL, ec="white", lw=0.5, zorder=11))
            continue
        draw_marker(ax, x, y, row["side"], row["label"], ppy, r=MARK_R)
    return fld


def strip(play, times, notes, lateral, window, rings=(), ncols=2, width_in=6.5):
    df = tracking(play)
    nrows = int(np.ceil(len(times) / ncols))
    pw = width_in / ncols
    span_w, span_d = lateral[1] - lateral[0], window[1] - window[0]
    fig, axes = plt.subplots(nrows, ncols, squeeze=False, figsize=(width_in, nrows * (pw * span_d / span_w + 0.55)))
    for ax in axes.flat[len(times):]:
        ax.axis("off")
    for i, (ax, t) in enumerate(zip(axes.flat, times)):
        panel(ax, play, df, t, lateral, window, rings)
        ax.set_title(f"{i + 1}.  " + ("snap" if abs(t) < 1e-6 else f"{t:+.1f} s"), fontsize=8.5,
                     color=style.INK_2, loc="left")
        ax.text(0.5, -0.02, notes[i], transform=ax.transAxes, ha="center", va="top", fontsize=7.4,
                color=style.INK, linespacing=1.15)
    fig.tight_layout(h_pad=2.2, w_pad=0.8)
    return fig


def video(play, times, notes, lateral, window, rings=(), width_in=5.6):
    df = tracking(play)
    span_w, span_d = lateral[1] - lateral[0], window[1] - window[0]
    fig, ax = plt.subplots(figsize=(width_in, width_in * span_d / span_w + 0.45))
    tmap = df.groupby("frameId")["time"].first()

    def update(fid):
        ax.clear()
        t = float(tmap[fid])
        panel(ax, play, df, t, lateral, window, rings)
        cur = [n for tt, n in zip(times, notes) if tt <= t + 1e-6]
        txt = ("snap" if abs(t) < 0.05 else f"{t:+.1f} s") + (("   " + cur[-1].replace("\n", " ")) if cur else "")
        ax.set_title(txt, fontsize=8.5, color=style.INK, loc="left")
        return []

    anim = animation.FuncAnimation(fig, update, frames=sorted(df["frameId"].unique()), interval=100)
    html = (anim.to_html5_video(embed_limit=40).replace("<video ", '<video style="max-width:100%;height:auto" ')
            if shutil.which("ffmpeg") else anim.to_jshtml(default_mode="once"))
    plt.close(fig)
    display(HTML(html))


def animated(play, times, notes, lateral, window, rings=()):
    """Video on the website; a numbered, captioned frame strip in the PDF."""
    if PDF:
        strip(play, times, notes, lateral, window, rings)
        plt.show()
    else:
        video(play, times, notes, lateral, window, rings)

Third-and-6, midway through the second quarter. The offense breaks the huddle and the running back doesn’t stop behind the quarterback. He keeps jogging, out past the right tackle, and settles in the slot just inside the tight end, who is himself standing ten yards from the ball. Nobody is left beside the quarterback. Five players are spread across the width of the field, and the defense starts to move: a linebacker drifts out after the running back, a safety creeps down toward the line, a cornerback points and shouts. The analyst says “empty set, they’re going to make the defense show its hand.” Two series later the same offense lines up with all three of its wide receivers squeezed in tight, the nearest one barely four yards outside the tackle, and the analyst says “look how condensed they are.” Same team, same players, opposite pictures. This chapter explains both: what the spread formations are, why spreading players across the field makes a defense commit, and why so many of the best NFL offenses then pulled their receivers back in.

Three ideas from Formation Rules and the Language of Alignment run through this chapter, so here they are in one line each. Seven players must be on the line of scrimmage, and only the two at the ends of that line can catch a pass; everyone else is a yard or more behind it, “off the line.” The letters name spots, not body types: X is the receiver on the line on one side, Z the receiver off the line on the other, Y usually the tight end, H a slot receiver or second back. A split is how far a receiver lines up from the ball or the nearest lineman, and a slot receiver is one who lines up between the outside receiver and the line. And remember the last chapter’s rule: personnel is who is on the field. Formation, this chapter’s subject, is where they stand.

What makes a formation “spread”

A formation is a picture of where the eleven offensive players stand at the snap. For most of football history that picture was compact: the receivers lined up a few yards outside the tackles, the quarterback was under center, and two backs stood behind him. The classic formations, the I, the pro set and the singleback, are all built that way, and they are built to run the ball with as many blockers near it as possible.

A spread formation does the opposite. It takes the players who can catch passes and spreads them out across the width of the field, toward both sidelines, usually with the quarterback a few yards back in the shotgun so he can see it all. Nothing about the personnel has to change. The same eleven players can line up either way:

Look at what the spread did to the defense without anybody running a step. On the left, the defense can keep seven players in the box: the area near the ball, roughly between the ends of the offensive line and within about seven yards of it, where the defenders who stop runs stand. (The full definition, and the habit of counting the box before every snap, belong to A First Look at Defense.) On the right, the offense has put four players out toward the sidelines. Each of them is a threat to catch a pass, so each needs a defender somewhere near him, and those defenders have to come from somewhere. They come out of the box. Now there are six.

That is the first and most basic reason spread formations exist. The field is 53⅓ yards wide, and a spread formation makes the defense defend all of it. Every defender who leaves the middle to cover a receiver out wide is a defender who isn’t there to tackle a runner or to rush the quarterback.

There are three more reasons, and they are what the rest of this chapter is about.

Spreading out makes the defense declare itself. When receivers stand close together, defenders can stand close together too, in a clump that hides what they plan to do. When the receivers spread out, the defenders have to spread with them, and where they go is information. A cornerback who follows a receiver all the way across the field is probably playing man-to-man, each defender assigned to one receiver, rather than zone, each defender assigned to an area of the field (both are introduced properly in A First Look at Defense). A safety who walks down toward a slot receiver has told you he isn’t going to be deep. A linebacker who stays in the box has told you somebody outside is uncovered or covered by someone farther away. The quarterback reads all of that before the snap. So can you.

Spreading out creates space. A receiver near the sideline with one defender in front of him and nobody else within fifteen yards is in a one-on-one race. A quick throw to him, or a run through a box with six defenders instead of eight, gives the ball carrier room to make one man miss.

Spreading out simplifies the quarterback’s job. With receivers spaced widely, each pass-catcher occupies his own patch of grass, and the quarterback can see who is open by seeing where the defenders aren’t.

ImportantWhy it exists

Spread formations exist to make the defense cover the whole field and tell you how it plans to do it. They trade blockers near the ball for space away from it. What the offense gives up is the old advantages of tight formations: extra blockers at the point of attack (the spot where the run is aimed), a quarterback under center who can hand off and fake quickly, and a run game that can simply outnumber the defense. A spread offense can still run, often very well, but it runs by making the box lighter, not by bringing more bodies to it.

Formation, not philosophy

A spread formation is a picture at the snap: receivers out wide. A spread offense is a whole philosophy, born mostly in college, built on those pictures. Every NFL team uses spread formations every week; only some would call their offense a spread offense. This chapter is about the pictures. The philosophy has its own chapter in The College Laboratory, and the systems that grew out of it, like the Air Raid, are installed in Part 4.

CautionCommon misconception

“Spread means pass.” Spreading out makes the defense lighter in the box, which is good for running. Many college spread offenses were built to run first, and in the NFL, 11-personnel formations with three receivers spread out are only slightly pass-leaning (about 56% pass in neutral situations, as Personnel showed). The one spread formation that really does mean “pass,” the empty backfield, is the exception, and we’ll measure it below.

On the right, the slot receiver is the tight end, Y. The defense has a cornerback on each outside receiver, a fifth defensive back (the nickel back, $) on the left slot, and two safeties deep. If the linebacker stays inside, the box is solid against the run but the tight end is alone, and a five-yard throw to him is as easy as a handoff. If he walks out, he takes the throw away and leaves five defenders in the box to face five blockers. Every lineman now has someone to block, and the first defender the running back has to beat is a safety twelve yards away. The defense can’t be right both ways, because one player can’t stand in two places. The offense watches where he stands and does the other thing.

That defender, the one stuck between the box and a slot receiver, is so important that he has his own names. A defender who stands halfway between the end of the line and the slot is “at the apex” (the point of the triangle the three of them make), and the man outside the box but inside a receiver is the overhang defender, because he hangs over the edge of the box. The passing game builds whole plays around reading him. (The name for a defender with two jobs and time for only one is a conflict defender; you’ll meet him again in RPOs, run-pass options in which the quarterback decides after the snap whether to hand off or throw, and which turn this exact question into a play.) For now, notice that the formation created the question. In the tight formation there was no slot receiver, so there was no question.

The defense isn’t helpless, of course. It can bring a safety down to help, which leaves less help deep. It can play a coverage in which the outside cornerback and the linebacker trade responsibilities. It can disguise its answer and change it at the snap. Those are the defense’s moves, and Parts 5 to 7 are full of them. But every one of them costs something somewhere else on the field, and the spread’s whole idea is to make the defense pay that cost on every snap.

NoteMadden vs. real life

In Madden, the defense lines itself up: you pick a play, and the computer slides every defender into place against whatever formation the offense shows. Real defenses have to do that themselves, in about the ten seconds between the offense breaking the huddle and the snap. They do it with rules agreed in advance: “against 2x2, the nickel takes the slot to the field and the Will takes the other one; against trips, the safety rotates down.” The middle linebacker and the safeties shout those adjustments to everyone else, and they all have to hear and agree. That’s why offenses love formations the defense hasn’t prepared for, and why a defense that sees a formation it didn’t practice sometimes ends up with two players covering one receiver and nobody on another.

Counting receivers: #1, #2, #3

You met the coaches’ counting system in Formation Rules, and every formation name in this chapter is built on it, so here it is again with a name. Receiver numbering counts the receivers from the sideline in, separately on each side of the ball. The widest receiver on a side is #1, the next one in is #2, the next is #3. It doesn’t matter whether he’s a wide receiver, a tight end or a running back who has lined up out wide: if he’s out of the backfield, he gets a number. A back standing beside or behind the quarterback isn’t counted until he leaves.

Why number them instead of using names? Because the defense doesn’t care who the receivers are; it cares where they are. A coverage rule like “the nickel back takes #2” works against any formation and any personnel. The numbers also give a formation its name: two on each side is “2x2,” three on one side and one on the other is “3x1.” The side with more receivers is the passing strength you met in the last chapter.

Two wrinkles, because defenses don’t all count the same way. Many coverage systems give an offset back (one standing to one side of the quarterback) the next number on his side before the snap, #3 beside two receivers, because he can release there in a step. And the count is redone after every motion: a receiver who motions across the formation changes both sides’ numbers, so the player who was “#2 right” in the huddle can be “#3 left” at the snap. That re-counting, under a play clock, is one reason offenses motion so much.

2x2: balance

A 2x2 formation, also called doubles, has two receivers on each side of the ball. In the most common NFL version the outside receivers (#1s, X and Z) stand on the line near the sidelines and the inside receivers (#2s, the slots) stand off the line, a few yards inside them. The slots must be off the line: if a slot receiver stood on the line, the outside receiver would be “covered” (no longer at the end of the line) and couldn’t legally catch a pass. That’s the covered receiver rule, and it’s why you’ll almost always see a slot receiver a step behind the receiver outside him.

The 2x2’s virtue is balance. The defense can’t tell from the alignment which side the play is going to, because the two sides look alike. The offense can run the same route combination to both sides, called a “mirrored” concept, and let the quarterback pick whichever side the defense covers worse. Or it can run two different combinations and pick based on what the defense shows. Against a defense with two deep safeties, a 2x2 asks each safety to help on his own side, two receivers apiece, and a defense that wants to bring an extra defender into the box has to decide which side gets less help.

What 2x2 gives up is concentration. Because the formation is balanced, the defense can be balanced too: it can play the same coverage to both sides with the same number of defenders, and nothing about the alignment forces it to shift. The formations that follow all trade some of that balance for an advantage somewhere specific.

3x1: the overload

Put three receivers on one side and one on the other, and you have a 3x1 formation, almost universally called trips. It’s the formation of the opening play in What a Play Really Is, “Gun Trips Right,” and it’s one of the most common pictures in the NFL.

Trips does two things at once. On the three-receiver side, the trips side, it puts more receivers than the defense normally has defenders, so the defense has to send help. On the other side, the backside, it leaves one receiver, often the offense’s best, alone with one cornerback.

To see the problem, you need a way to count a side. Here’s the one this book uses, every time:

  1. Count the receivers on that side (#1, #2, #3).
  2. Count the underneath defenders who could cover them: everyone on that side who stands outside the box, including an apex or overhang defender walked out toward a slot. Defenders inside the box don’t count; they’re there for the run.
  3. Note the deep safety on that side separately, as help over the top (“+ safety”). He’s too deep to stop a quick throw, but he matters on anything thrown down the field.

If step 1 beats step 2, the offense has a numbers advantage underneath, and a quick throw to that side is waiting. The defense has to choose which side’s problem to solve:

If the defense stays balanced, the trips side is three receivers against three defenders underneath and one safety, and the offense can attack it with combinations designed to put two receivers on one defender. If the defense rotates, it gets an extra man to the trips side, but the backside receiver is now alone on what coaches call an island, one-on-one with nobody behind his cornerback. That’s why offenses so often put their best receiver at the single-receiver spot in 3x1. They’re betting he wins one-on-one, and if the defense refuses that bet by keeping a safety on his side, the trips side is short-handed.

Many modern defenses have a third answer. They keep both safeties deep before the snap, so the picture still looks balanced, and then let the backside safety leave X to his cornerback and drift across to help on the trips side’s #3 if he runs deep. Coaches call it poach, and it and its cousins are the subject of Disguise, Rotation, and Split-Field Coverage. It looks like a better deal, but count it: X is still on an island. Every answer to trips spends the backside safety somewhere.

The trade-off for the offense is that trips announces its strength. The defense knows which side has the numbers and can tilt its coverage, its front and its blitzes (extra pass rushers; see A First Look at Defense) toward it. Trips also leaves the backside thin for the run game: with only one receiver over there, the defense can squeeze its run defenders toward the trips side without worrying much about a quick throw the other way. Offenses answer by running away from trips, toward the single receiver, where the defense has thinned out; the run game chapters show how.

Trips also interacts with the hash. With the ball on a hash mark, one side of the field is about six yards wider than the other (the field side and the boundary side). Most offenses put trips to the field side, where three receivers have room to spread out, and leave the single receiver to the short side. When an offense puts trips into the boundary, squeezing three receivers into the narrow side, it’s usually a deliberate surprise: the defense’s rules for the field and the boundary may not fit.

Empty: everyone out

An empty formation has nobody in the backfield but the quarterback. All five eligible receivers are out, most often three on one side and two on the other, so it’s also called 3x2 (and “5-wide” on broadcasts). As the plates above showed, empty usually isn’t a special personnel grouping. The running back and tight end simply line up as receivers. In 2025, about 72% of NFL empty snaps came from ordinary 11 personnel.1

Empty is the spread taken to its limit, and it does three things nothing else does.

It shows the quarterback everything. With five receivers spread out, the defense has to put someone near each of them, and the quarterback can often make a good guess at man coverage or zone before the snap by watching who goes where. Where a defender lines up isn’t enough on its own (a linebacker walked out over the back can be playing a zone area just as easily as the back himself), but how he lines up and moves is. A defender who presses the receiver or sits on his inside shoulder with his eyes on him is usually in man; one who stands off with his eyes on the quarterback is usually in zone. And if the back motions and a linebacker has to run across the field with him, that’s a strong man tell, the trick Motion and Shifts builds on.

It isolates the backs and tight ends. A running back split out wide is often covered by a linebacker, who is bigger and slower than the defensive backs. If the defense brings in more defensive backs to cover him, it is smaller everywhere else.

It exposes the blitz. Five receivers out means only the five offensive linemen are left to block. If the defense shows six pass rushers, everyone in the stadium can count: five blockers, six rushers, one man free. The protection decides which man: it usually blocks from the inside out and leaves the widest rusher, because he has the longest way to run. The quarterback knows it in advance, so the answer is built into the play: get the ball out before the free man arrives, to a receiver running into the space a blitzer just left. Coaches call that a hot route, and a receiver who sees the blitz and cuts his route short for it is making a sight adjust; you’ll hear both words on broadcasts.

Two numbers tell the story. First, empty is a specialty formation: between about 7 and 9.5 percent of plays, every season for a decade, with no steady trend either way. Second, it is overwhelmingly a passing formation: in 2023–2025, more than 94% of NFL plays from empty were dropbacks, against about 58% from everything else.3 Offenses save it for third down, and especially third-and-long, when they need a completion and a run would rarely get the yards anyway.

What about the dip to a decade low in 2025? It came in the same season that 11 personnel, the grouping most empty snaps come from, fell to a decade low of its own (Personnel tells that story). But the mix explains only a little of it. Offenses also went empty less often from 11 personnel: about 10.6% of 11-personnel snaps were empty in 2024, and 8.6% in 2025.4 One season is a blip until the next one confirms it.

That predictability is the price, and it’s why empty is a tool rather than a base formation. If empty is almost always a pass, the defense can rush the passer without worrying about the run. Offenses that use it a lot find ways to make it less predictable, and the best way is a quarterback who is dangerous running the ball. The Buffalo Bills under Josh Allen have gone empty on about 10% of their plays since 2019, against a league rate of about 8%, and in 2023 about one in eight of their empty snaps was a designed run, nearly all by Allen, more than twice the rate for the rest of the league.5 Why that works is the second drill at the end of the chapter.

Clusters: bunch and stack

Spreading receivers apart is one idea. Putting them close together on purpose is another, and it’s just as old. Here are the four variations you’ll see most, again all from one personnel group:

A bunch is three receivers clustered in a tight triangle, usually within two or three yards of each other: one on the line (the “point”), the other two just off it, one inside and one outside. A stack is two receivers lined up one directly behind the other, the front man on the line and the back man a yard or two behind. Both are legal for the same reason as everything else in this chapter: only one receiver in each cluster is on the line, and he’s the end of it.

Clusters are where numbering gets awkward, so it’s worth a rule. Count by width, as always: in the bunch above, Z is the widest (#1), the point man H is #2 and Y, tucked inside, is #3. In a stack the two receivers share a width, and most staffs count the man on the line as #1 and the man behind him as #2. Don’t expect the numbers to hold for long, though. Many defenses facing a cluster number it after the snap, by who releases where (“whoever goes out is #1”), which is exactly the kind of on-the-fly agreement the cluster is designed to strain.

Why would an offense that wants space squeeze its receivers together? Three reasons, and they all have to do with defenses that play man-to-man.

Traffic. In man coverage, each defender follows one receiver wherever he goes. When three receivers start within two yards of each other and then run in different directions, their paths cross, and the defenders chasing them have to get around each other and around the other receivers. A defender who is a step late is a receiver who is a step open. Watch the ringed safety:

The receivers never touched the defenders, and that matters. Running into a defender on purpose to knock him off a receiver, after the snap and before the pass is touched, more than a yard past the line of scrimmage, is offensive pass interference.6 Read that rule closely and you can see how bunch plays are built. Contact within a yard of the line isn’t pass interference, which is one reason the crossings in a bunch release happen right at the line, as they do in the strip above. Past that yard, the receivers’ own paths have to do the work: each one runs his route exactly as drawn, and if a defender has to go around him, that’s the defender’s problem. Officials have to judge the difference between a receiver running his route (“natural pick,” or “rub”) and a receiver who goes looking for a defender to block, and the second one draws a flag. The full family of rub routes is in Dropback Concepts.

Free releases. A cornerback who wants to jam a receiver at the line (to press him) has to line up right in front of him. In a stack, only the front receiver can be pressed; the man behind him starts with a free run. In a bunch, the two back receivers are shielded in the same way. That makes clusters a favorite answer to defenses that press.

Communication. Clustered receivers force defenders to decide, at the snap, who takes whom. Many defenses don’t try to follow their man through the traffic; they agree to switch (coaches also say “swap”), so the defender nearest the receiver going out takes him and the defender nearest the receiver going in takes that one. Switching works, but it takes two defenders who see the same thing at the same time and talk. Every time a defense has to communicate under stress, it can get it wrong.

What clusters give up is width. Three receivers within three yards of each other can be covered by fewer zone defenders than three receivers spread across the field, because a zone defender standing in the right spot can cover two of them. That’s why bunch and stack are at their best against man coverage, and why you’ll see them most in the red zone and on third-and-short, where defenses play the most man. They aren’t useless against zone, though: three receivers released from one spot can flood one defender’s area at three depths (short, middle and deep), and he can only cover one of them.

Andy Reid’s Kansas City Chiefs are the best-known modern users, stacking and bunching receivers (often with Travis Kelce in the cluster) as an answer to man coverage; both of their fourth-quarter touchdowns in Super Bowl LVII came from a stack with a receiver tucked behind another, though the trick that beat the man coverage was the return motion you’ll meet in Motion and Shifts.7

Lopsided: nub and quads

Two more names describe formations by what’s missing.

A nub side is a side of the formation where a tight end is the only receiver, attached to the line with no wide receiver outside him. “Trips nub” (the bottom-left plate above) puts three receivers to one side and only the tight end on the other. It’s a clever shape, because it splits the formation’s identity in two. The tight end side is the strong side for the run game (it has the extra blocker), while the trips side is the receiver strength (it has the extra pass-catchers). Remember the common defensive rule from Formation Rules: front to the tight end, coverage to the passing strength. Trips nub is built to make that rule hurt. The front leans one way and the coverage the other, so a defense that sets its front to the tight end is under-manned against the receivers, and a defense that rotates its coverage to the receivers is light against a run to the tight end.

The defense’s usual answer is to keep the backside cornerback on the nub side anyway, lined up over or just outside the tight end five to seven yards deep. From there he can come up and force a run to that side or run with the tight end if he goes deep. Other defenses send that cornerback across to the trips side and give the tight end to a linebacker or safety. Either way the defense has had to decide, which is the point of the formation.

Quads puts four receivers on one side. Only five players can be eligible, and the line needs an eligible end on each side if the offense wants to be able to throw to both, so quads almost always means empty: four receivers on one side, one on the line on the other, nobody beside the quarterback (the 4x1 above). Keep a back beside the quarterback and the arithmetic breaks: with nobody on the other side, the tackle becomes that end of the line, both remaining players on the line have to stand on the quads side, and the inner one is covered and ineligible. It’s the most extreme overload in football, and it’s rare: you’ll see it in two-minute drills, on trick plays, and from offenses testing whether the defense has a rule for it. Its value is the question it asks. A defense that hasn’t practiced against four receivers on one side has to improvise, and improvised coverage is how receivers get open.

The condensed revolution

Everything so far might suggest the answer is always “spread out more.” For about a decade, that was roughly where NFL offenses were heading. Then, in the late 2010s, some of the most successful offenses in the league started doing something that looked like going backward: they pulled their receivers in.

A condensed formation lines its receivers up close to the offensive line, with reduced splits (inside the painted numbers) or even tight ones (a few yards outside the tackle or tight end), instead of out near the sidelines. The personnel is the same; the receivers just stand closer to the ball. (Coaches have a dialect for this. Besides the plus and minus splits you met in Formation Rules, you’ll hear “reduced,” “squeeze” and “nasty,” the last one for a receiver only two or three yards outside the tackle or tight end.) Sean McVay’s Los Angeles Rams made it their signature in the late 2010s, Kyle Shanahan’s San Francisco 49ers took it furthest, and much of the league followed: by Next Gen Stats’ count, condensed alignments went from about 11% of NFL plays in 2016 to outnumbering wide ones in 2023.8

That raises a question of words. Is a condensed trips formation a “spread” formation? The cleanest way to think about it is as two separate dials. One dial is the count: how many receivers stand on each side (2x2, 3x1, empty). The other is the width: how far from the ball they stand (spread or condensed). Any count can be set at any width:

So a condensed 3x1 is still a 3x1, and still part of the shotgun, three-receiver family the NFL lives in; it just isn’t wide. Broadcasters will call all four of these pictures “spread” at some point. When this book says spread or condensed, it means the width dial.

If the condensed pictures remind you of the tight formation in the very first figure of this chapter, they should. On paper the alignments are close. The difference is in who is standing there and what they’re for:

  • Different bodies. The old tight formations got their blockers by bringing in a fullback or a second tight end, and the defense answered with linebackers. A condensed formation is usually 11 personnel: three wide receivers. The defense still has to stay in nickel to cover them, so the extra defender near the ball is a nickel back (about 195 pounds) rather than a linebacker. (In 2018, defenses had five or more defensive backs on the field for about 94% of the Rams’ snaps, as Personnel showed.)
  • Different threats. A fullback or second tight end in a tight formation is mostly a blocker. A wide receiver in a condensed split can block, but he can also run past the defender in front of him or across the field, so every defender near him has to respect both.
  • One picture for everything. The point of condensing isn’t to run the ball more; it’s to run the ball and throw it from a picture that doesn’t tell the defense which. The four reasons below are all versions of that.

1. Room to the outside

You met the basic trade in Formation Rules, as a tell: a reduced split often means a route toward the sideline or a block on the inside. Here’s why it works. A receiver lined up near the sideline has little grass outside him, and the cornerback covering him knows it. The cornerback can line up inside the receiver and let the sideline guard the outside, like a twelfth defender. A throw to that receiver on an outside-breaking route also has to travel across most of the field. Move the receiver in toward the ball and the grass outside him becomes his, and the same route turns into a shorter, safer throw:

2. The two-way go

That leads to the second reason. A receiver with room on both sides of his defender has what coaches call a two-way go: he can break inside or outside, and the defender has to guess. Against a receiver near the sideline, the defender doesn’t guess; he takes away the inside and lets the sideline squeeze everything that breaks outside. Against a receiver with a two-way go, a defender who leans one way gets beaten the other.

The defender’s position relative to the receiver, inside him, outside him or straight in front, is called his leverage, and the quick passing game is largely a contest of reading it. The point here is about the formation: where a receiver lines up decides how many ways he can go, and so how much the defender has to guess. Slot receivers have always had two-way gos. Condensed splits give one to the outside receivers too.

The two-way go matters most against man coverage with no help nearby, which is why you’ll hear it so much on third down. If the defense has a safety or a linebacker helping inside, the cornerback can play outside leverage, give up the inside to his helper and stop guessing. Taking that help away, by spreading the other defenders out or pulling them toward the run, is the offense’s job.

3. Crack blocks

The third reason is the run game. A receiver who lines up close to the formation is already inside the cornerback and right next to the defenders he wants to block, close enough to hit them from the side. A crack block is a wide receiver blocking back toward the ball, on a nickel back, linebacker or safety who is watching the play and not him. It’s one of the most effective blocks in football, because the defender is looking the other way and the receiver gets a running start. From a wide split it takes too long: by the time the receiver gets back inside, the play is over. From a condensed split he’s there in a step or two.

If that sounds like a cheap shot, the rulebook agrees with you halfway. A crack block is legal, but only if it’s above the waist and not to the head or neck: a receiver who lines up more than two yards outside the tackle and blocks back toward the ball near the line may not go low, and doing it costs 15 yards.9 So the receiver needs to get his eyes and hands on the front of the defender, chest-high, which is one more reason he wants to start close to him. 4. Everything looks the same

The last reason is the subtlest, and for many coaches the most important. When receivers stand close to the formation on every play, a run and a pass start from the same picture. The receiver who cracks on a run is lined up exactly where he lines up to run a route on play-action (faking the handoff, then throwing). The defense can’t read the play from the splits. The condensed formations of the late-2010s Rams and 49ers were built around exactly that: outside zone runs (on which the whole line steps sideways together and the back aims for the edge), crack blocks and play-action passes, all from one look. The condensed split is also the ideal launch point for a favorite play-action pass of those offenses: the receiver starts inside the cornerback and just outside the linebackers, the linebackers step up to stop the fake run, and he crosses the field behind them. A linebacker or safety who guessed “run” because the receivers were tight was guessing wrong half the time.

Two more things come with the tight picture. A receiver eight yards from the ball can motion across the formation in about a second, so condensed splits and pre-snap motion go naturally together. And the more often the offense lines up this way, the less any single snap gives away.

What condensed gives up

Nothing in football is free, and condensing has two clear costs.

It brings the defense in too. Defenders line up on receivers, so when the receivers move in, the defenders move in, and the box can get more crowded, not less. The spread’s first benefit, thinning the box, runs backward. Condensed offenses accept that because the receivers now block those extra defenders, but a condensed formation against a defense that plays it well can feel very crowded.

It gives up horizontal stretch. A spread formation makes defenders cover sideline to sideline; a condensed one lets them stay closer together, so one defender can sometimes cover two receivers, and quick throws to the outside, like screens to a receiver near the sideline, have less room. The horizontal stretch, making defenders cover too much width, is one of the passing game’s basic weapons, and condensing trades some of it for the other advantages.

ImportantWhy it exists

Condensed formations exist because the spread taught defenses to play wide. Once defenses learned to spread out and play the receivers from the inside, using the sideline as help, the open space moved: it was now outside the receivers and in the run fits near the ball (each run defender’s assigned gap, which a block from an unexpected angle can wipe out). Condensing puts the receivers where the new space is, and where they can block the defenders who are watching the ball. It trades width for angles. It’s the same idea as the spread, making the defense defend space it would rather not, aimed at a different patch of grass.

Specialty looks

Three more formations deserve names, not because you’ll see them every drive but because when you do see them, they’re asking a specific question, and you can answer it before the snap.

Why snap to a running back? Think about a normal handoff. The quarterback takes the snap, hands the ball off, and then does nothing useful for the run: he’s one of the offense’s eleven players, but he isn’t blocking anybody. The defense, meanwhile, has all eleven of its players available to tackle. In the wildcat, the player who takes the snap is himself a running threat. And the quarterback, now split out wide, still needs a defender on him, because he is still an eligible receiver: leave him alone and the offense can throw to him, or toss him the ball back for a pass of his own. So the offense has one more player who matters to the run and the defense has one fewer to stop it. The motion man adds a second runner, so the defense can’t key on one ball carrier.

The wildcat’s ancestor is the single wing, the direct-snap formation that ruled football before the T-formation. Its modern NFL moment came in 2008, when the Miami Dolphins used it to stun the New England Patriots (the story is in the film room below), and for a season or two teams all over the league copied it. Then it faded, for a reason worth remembering: the player taking the snap usually couldn’t throw. Once defenses stopped respecting the pass, they put an extra player in the box, and the wildcat’s one-player advantage disappeared. It survives today in short bursts, as a short-yardage or goal-line package, or built around a player who really can both run and throw. In Next Gen Stats’ formation labels it never reached even one play in a hundred in any season from 2016 to 2022.10

The point is arithmetic. A defense lines up its front by counting blockers on each side of the center. If one side suddenly has an extra lineman and the defense doesn’t notice or doesn’t shift, the offense has a blocker for every defender on that side, and it runs there. The defense’s simplest answer is to declare the extra-lineman side its strong side and slide its whole front a gap over, which works if everyone sees it in time, and can open a hole somewhere else or leave a pass-rusher out of position if they don’t.

Moving the left tackle isn’t the only way to get there. An offense can also bring in a sixth offensive lineman to stand next to a tackle, which gives the same extra blocker without weakening the other side of the line.

Two rules keep the look honest. Linemen wear numbers 50 to 79 and are ineligible, so a tackle who moves to the end of the line is still just a blocker, which is why a tight end usually lines up outside him if the offense wants to threaten a pass that way. And the line still needs seven men, so the side left with only a guard needs a receiver on the line at its end, like X in the picture. (A lineman can become eligible by reporting to the referee, a different trick.)

The gate asks the defense a yes-or-no question. If the defense lines up to block the kick and ignores the group out wide, the offense snaps the ball to the holder, who throws to or pitches to the gate, where there are more offensive players than defenders: in the picture, six blockers for four defenders, and a seventh man to carry the ball. If the defense sends people out to cover the gate, the gate “swings shut”: the linemen run back to their normal spots and the team kicks, or the offense uses the defense’s confusion to run a play from the normal formation. Either way the defense has to recognize it, count it, and adjust in a few seconds, which is exactly the kind of pressure every formation in this chapter is designed to create.11

Where spread formations came from

The full story belongs to Part 11, but three sentences explain why the NFL looks the way it does. Spreading receivers out is almost as old as the forward pass: Dutch Meyer’s TCU teams of the 1930s and 1940s used what he called the “spread formation,” and from the late 1950s on, the run and shoot and later the Air Raid turned the idea into whole offenses, mostly in high school and college.12 The NFL came to it slowly, then all at once, as a generation of quarterbacks and receivers arrived having played nearly every snap in the shotgun with three or four receivers out, and as offenses like the 2007 Patriots (see the film room) showed what it could do; by the late 2010s the shotgun 11-personnel spread was the league’s default, and the condensed formation was the next move in the same back-and-forth. The College Laboratory and The Modern Era tell it in full.

Film room

NoteFilm room: New England Patriots 2007

The 2007 Patriots went 16–0 in the regular season and scored 589 points, at the time an NFL record. Tom Brady threw 50 touchdown passes (also a record then), Randy Moss caught 23 of them (a single-season record), and Wes Welker caught 112 passes.13 The offense, coordinated by Josh McDaniels, lived in spread formations: by the play-by-play data’s shotgun flag, the Patriots were in the shotgun on about half their plays, more than any other team and nearly twice the league rate of about 27%.

The spread was built around two very different receivers, and it shows the two halves of this chapter. Moss was the deep threat outside, exactly the kind of receiver a 3x1 leaves on the island: a defense that rotated its safety away from him left him one-on-one with a cornerback, and a defense that kept a safety over him was short-handed everywhere else. Welker worked from the slot, where he had a two-way go on most snaps and could break inside or outside depending on how the defender played him.14

What to notice: when an offense puts one receiver alone on one side and three on the other, find the safeties. Where they go before the snap tells you which bet the defense is making, and which receiver the quarterback will look at first.

NoteFilm room: Los Angeles Rams 2018

Sean McVay’s 2018 Rams went 13–3 and reached Super Bowl LIII, and as Personnel showed, they used 11 personnel on about 93% of their plays. With the personnel never changing, the variety had to come from formation and motion, and the Rams’ signature was the condensed formation: receivers lined up close to the formation, so tight that about 60% of the Rams’ formations were less than 20 yards wide at the snap, against 37% for the next-tightest team. (For scale: a 2x2 with both outside receivers on the numbers and the ball in the middle is about 25 to 30 yards wide from one outside receiver to the other.)

The tight splits made the run game and the passing game look alike. On outside zone runs, the receivers were close enough to block the linebackers and safeties at the edge; on play-action and bootlegs, the same receivers released into routes from the same spots, and defenders who had been hit by those receivers on the previous run had to guess.15

What to notice: before the snap, measure the outside receiver against the numbers painted on the field. If he’s well inside them, look for a crack block on a run or an outside-breaking route on a pass, and watch the cornerback: if he widens to protect the outside, the inside is open.

NoteFilm room: Miami Dolphins 2008, the Wildcat

On September 21, 2008, the Miami Dolphins, who had gone 1–15 the year before, traveled to New England and beat the Patriots 38–13. Running back Ronnie Brown took direct snaps in a formation the Dolphins called the Wildcat and scored four rushing touchdowns, and threw a touchdown pass from it as well. Miami stacked two of this chapter’s tricks: the Wildcat line was unbalanced, and the handful of plays (a power, a counter, a sweep to Ricky Williams coming across in motion, a reverse and a couple of passes) all started from the same picture. The loss ended New England’s 21-game regular-season winning streak, and Miami went on to win the AFC East at 11–5.16

The idea came to Miami through quarterbacks coach David Lee, who had run a version of it at the University of Arkansas with Darren McFadden. The name has a longer and disputed history: several high school and college coaches ran direct-snap formations under the same name in the late 1990s and 2000s, and all of them were rediscovering the single wing.17

What to notice: when a running back walks up to the shotgun spot and the quarterback jogs out wide, count the box. The wildcat’s edge is one extra runner; if the defense has one extra defender inside, it has taken that edge away, and the offense’s best play may be to shift back out of it.

Watch for it on Sunday

TipWatch for it: formation and leverage
  • Name the formation. As the offense breaks the huddle, count receivers on each side, from the sideline in: 2x2, 3x1, empty (3x2), or a variation (bunch, stack, nub, quads). Is anybody beside the quarterback?
  • Find the strength. Which side has more receivers? Is the tight end on the same side or the other one (a nub)? A split strength means the defense has to pick one.
  • Check the splits. Use the painted numbers as your ruler. An outside receiver on or outside the numbers is a normal or wide split; one clearly inside the numbers (a high camera shows it best) is condensed, and one within a few yards of the tackle or tight end is tight. Condensed splits mean crack blocks, outside-breaking routes, play-action crossers and rubs; wide splits mean the offense wants the defense spread thin.
  • Count each side, three steps. Receivers on that side; underneath defenders outside the box on that side (an apex defender counts); then the deep safety as “+ safety.” More receivers than underneath defenders is a quick throw waiting to happen; a lone receiver with no safety on his side is on an island.
  • Count the box. Did the spread pull defenders out? Six or fewer in the box against a spread formation is an invitation to run.
  • Look for specialty clues. The quarterback split out wide: wildcat. Three linemen on one side of the center: unbalanced. Linemen standing far out on an extra point: swinging gate.
  • Log it. Add a “formation” column to your prediction log beside personnel: for example “11, 3x1 right, condensed.” After a half, check how often empty meant pass.

The open grass is outside the receivers: about 18 yards between each outside receiver and his sideline, and the cornerbacks, lined up inside them, aren’t protecting any of it.

What it sets up right now:

  1. Outside-breaking routes: an out (straight up, then a sharp cut to the sideline), a corner (up, then angling toward the back corner of the end zone) or a fade (a deep throw toward the sideline). With the cornerback inside him, the receiver doesn’t even need his two-way go; he has a one-way go to the side the defender isn’t protecting.
  2. Play-action. It’s first down, the quarterback is under center and the receivers are tight: the defense has to respect the run. A fake handoff pulls the linebackers forward, and the receivers can release behind them or outside the cornerbacks.

What the corners should do: widen and play outside leverage, protecting the grass. (Good defenses do this before the offense can punish them; the drill shows a defense that hasn’t.) But look at what that creates. With the corners outside the receivers, X and Z are now outside the nickel and the linebackers and inside the cornerbacks, exactly the picture the crack toss started from. X can crack down on the nickel while a pulling lineman takes the cornerback. And with the corners guarding the outside, the inside opens for slants and digs (a dig is a route that goes straight up the field, then cuts flat across it). The defense can take away one thing; the formation has set up the other.

Takeaways

  • A formation is where the players stand, and a spread formation puts the pass-catchers across the width of the field. It makes the defense defend all 53⅓ yards, thins the box and forces defenders to show where they’re going. Personnel doesn’t change; one grouping can line up a dozen ways.
  • Count receivers from the sideline in, side by side: #1, #2, #3. That count names the formation (2x2, 3x1, empty 3x2, quads 4x1) and sets the defense’s coverage rules. Then count each side in three steps (receivers, underneath defenders outside the box, the deep safety) to see where the offense has the numbers.
  • Every spread formation asks the defense a question it can’t answer both ways. Cover the slot or stay in the box; help the trips side or the lone receiver; match an empty set with defensive backs or keep a sixth man inside. The offense does whatever the defense didn’t.
  • Clusters and lopsided formations have specific jobs. Bunch and stack beat man coverage and press with traffic and free releases; nub splits the formation’s strength; quads asks a question most defenses haven’t practiced. Empty shows the quarterback everything but is more than 94% pass.
  • Count and width are two separate dials. 2x2, 3x1 and empty describe the count; spread and condensed describe the width. Condensed splits are the modern counter-move: pulling receivers in gives them room outside, a two-way go, angles for crack blocks and a way to make runs and passes look alike, at the cost of some width and a more crowded box. It differs from the old tight formations because the bodies are wide receivers, so the defense has to stay small.
  • Specialty looks (wildcat, unbalanced, swinging gate) are arithmetic tricks. Each one adds a blocker or a runner somewhere the defense hasn’t counted. Count the defense’s answer before the snap.
