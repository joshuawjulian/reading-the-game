def O(key, pos, d, w, label=None):

def D(key, pos, d, w, label=None):

def line(labels=False):
    """Five offensive linemen; unlabeled except the center unless labels=True."""
    names = ["LT", "LG", "C", "RG", "RT"]
    return [O(k, k, -0.7, w, k if (labels or k == "C") else "")
            for k, w in zip(names, (-3.2, -1.6, 0.0, 1.6, 3.2))]


def trips_y():
    """11 personnel, shotgun 3x1: the tight end (Y) detached as the inside receiver of three on the right.
    Seven on the line: the five linemen, X and Z. Y and H are off the ball, R beside Q."""
    return line() + [O("QB", "QB", -5.0, 0.0, "Q"), O("RB", "RB", -5.0, -1.6, "R"),
                     O("X", "WR", -0.7, -15.0, "X"), O("Y", "TE", -1.5, 6.5, "Y"),
                     O("H", "WR", -1.5, 11.0, "H"), O("Z", "WR", -0.7, 16.0, "Z")]


def dl_nickel():
    """Four down linemen against a formation with no attached tight end."""
    return [D("LDE", "DE", 1.0, -5.0, "E"), D("LDT", "DT", 1.0, technique("1", -1), "T"),

def go_to(p, key, end, kind="path", delay=0.0, speed=None, via=()):
    """Move `key` through absolute (d, w) points `via` to `end`, as a block, path, run or route."""
    start = np.array(p._cur(key))
    pts = [tuple(np.array(q) - start) for q in (*via, end)]
    if kind == "block":
        return p.block(key, pts=pts, delay=delay, speed=speed)
    if kind == "route":
        return p.route(key, pts, frame="field", delay=delay, speed=speed)
    if kind == "rush":
        return p.rush(key, pts=pts, delay=delay, speed=speed)
    return {"path": p.path, "run": p.run}[kind](key, pts, delay=delay, speed=speed)


def _P(fld, p, d, w):
    return fld.xy(*p.to_abs(d, w))


def say(fld, p, d, w, text, color=style.INK_2, fontsize=7.0, ha="center", weight="normal"):
    """A short on-field label at (d, w)."""
    x, y = _P(fld, p, d, w)
    fld.ax.text(x, y, text, fontsize=fontsize, color=color, ha=ha, va="center", zorder=11, fontweight=weight,
                linespacing=1.1, bbox=dict(boxstyle="round,pad=0.18", fc=style.SURFACE, ec="none", alpha=0.92))


def point_to(fld, p, text_at, target, text, color=style.INK_2, fontsize=6.8):
    """A label with a thin leader line to the player or spot it names."""
    (xt, yt), (xp, yp) = _P(fld, p, *text_at), _P(fld, p, *target)
    fld.ax.annotate(text, (float(xp), float(yp)), xytext=(float(xt), float(yt)), fontsize=fontsize,
                    color=color, ha="center", va="center", zorder=11, linespacing=1.1,
                    arrowprops=dict(arrowstyle="-", color=color, lw=0.7, shrinkA=2, shrinkB=7),
                    bbox=dict(boxstyle="round,pad=0.18", fc=style.SURFACE, ec="none", alpha=0.92))

def ghost(fld, p, pts, color=style.DEFENSE, lw=1.4, alpha=0.6, zorder=5):
    """A faint dashed arrow through absolute (d, w) points: what a player did (or will do) off the play."""
    xs, ys = _P(fld, p, np.array([q[0] for q in pts]), np.array([q[1] for q in pts]))
    fld.ax.plot(xs[:-1], ys[:-1], color=color, lw=lw, ls=(0, (3, 2)), alpha=alpha, zorder=zorder)
    fld.ax.add_patch(FancyArrowPatch((xs[-2], ys[-2]), (xs[-1], ys[-1]), arrowstyle="-|>", mutation_scale=10,
                                     color=color, lw=lw, alpha=alpha, ls=(0, (3, 2)), zorder=zorder))


def clean(ax):
    """Zoomed diagrams: drop hash ticks and yard numbers, and clip text to the window."""
    for ln in list(ax.lines):
        xs, ys = ln.get_data()
        if len(xs) == 2 and ln.get_linewidth() == 0.5 and np.hypot(xs[1] - xs[0], ys[1] - ys[0]) < 1.0:
            ln.remove()
    for t in list(ax.texts):
        if t.get_text().isdigit() and t.get_alpha() == 0.55:
            t.remove()
        else:
            t.set_clip_on(True)


def draw(p, ax=None, width_in=5.8, **kw):
    fld = p.draw(ax=ax, width_in=width_in, title=False, **kw)
    clean(fld.ax)
    return fld

# ---- frame strips (print) and videos (web) for simulated plays
MARK_R = 0.6    # frame-strip marker radius in yards, smaller than the static diagrams'


def tracking(play):
    df = play.to_tracking(t_after=1.6)   # run past the last route so a late throw can land
    df["_id"] = df["displayName"].astype(str)
    return df


def panel(ax, play, df, t, lateral, window, rings=(), ball=1.0):
    """One moment of a simulated play: markers, trails of on-screen players, rings (ball = ball size factor)."""
    tmap = df.groupby("frameId")["time"].first()
    fid = int((tmap - t).abs().idxmin())
    ylo, yhi = sorted((play.ball_y - lateral[0], play.ball_y - lateral[1]))
    fld = Field(ax=ax, orient="vertical", los=play.los, window=window, numbers=False, hashes=False,
                lateral=(ylo, yhi), to_go=play.to_go)
    cur = df[df["frameId"] == fid]
    shown = cur[cur["x"].between(play.los + window[0] + 0.4, play.los + window[1] - 0.4)
                & cur["y"].between(ylo + 0.4, yhi - 0.4)]
    ids = set(shown["_id"])
    trails = df[(df["frameId"] <= fid) & df["_id"].isin(ids) & (df["side"] != "ball")]
    for (_, side), g in trails.groupby(["_id", "side"]):
        xs, ys = fld.xy(g["x"].values, g["y"].values)
        ax.plot(xs, ys, color=style.OFFENSE if side == "off" else style.DEFENSE, lw=0.9, alpha=0.4, zorder=3)
    ppy = fld.pts_per_yard()
    for _, rr in shown.iterrows():
        x, y = (float(v) for v in fld.xy(rr["x"], rr["y"]))
        if rr["side"] == "ball":
            ax.add_patch(Ellipse((x, y + 0.45 * ball), 0.7 * ball, 0.42 * ball, fc=style.BALL, ec="white",
                                 lw=0.5, zorder=9))
            continue
        lab = str(rr.get("label", "") or "")
        if rr["side"] == "off":
            ax.add_patch(Circle((x, y), MARK_R, fc=style.OFFENSE, ec="white", lw=0.6, zorder=8))
            tc = "white"
        else:
            ax.add_patch(RegularPolygon((x, y), 4, radius=MARK_R * 1.3, orientation=np.pi / 4,
                                        fc=style.DEFENSE_TINT, ec=style.DEFENSE, lw=1.0, zorder=8))
            tc = style.INK
        fs = max(4.0, min(8.0, ppy * (0.8 if len(lab) <= 1 else 0.6)))
        ax.text(x, y, lab, ha="center", va="center", fontsize=fs, color=tc, fontweight="bold", zorder=10)
    for key, col in rings:
        rr = cur[cur["displayName"] == key].iloc[0]
        x, y = fld.xy(rr["x"], rr["y"])
        ax.add_patch(Circle((float(x), float(y)), 1.05, fc="none", ec=col, lw=1.7, zorder=7))
    return fld


def strip(play, times, notes, lateral, window, ncols=2, rings=(), width_in=6.4, frame_rings=None):
    """Numbered, captioned frame strip: the PDF stand-in for the video.
    frame_rings: optional {frame index: ((key, color), ...)} drawn on that frame only, on top of `rings`."""
    df = tracking(play)
    nrows = int(np.ceil(len(times) / ncols))
    pw = width_in / ncols
    span_w, span_d = lateral[1] - lateral[0], window[1] - window[0]
    fig, axes = plt.subplots(nrows, ncols, squeeze=False,
                             figsize=(width_in, nrows * (pw * span_d / span_w + 0.55)))
    for ax in axes.flat[len(times):]:
        ax.axis("off")
    for i, (ax, t) in enumerate(zip(axes.flat, times)):
        panel(ax, play, df, t, lateral, window, tuple(rings) + tuple((frame_rings or {}).get(i, ())), ball=2.4)
        ax.set_title(f"{i + 1}.  " + ("snap" if abs(t) < 1e-6 else f"{t:+.1f} s"), fontsize=8.5,
                     color=style.INK_2, loc="left")
        ax.text(0.5, -0.03, notes[i], transform=ax.transAxes, ha="center", va="top", fontsize=7.5,
                color=style.INK, linespacing=1.15)
    fig.tight_layout(h_pad=1.6, w_pad=0.6)
    return fig


def video(play, times, notes, lateral, window, rings=(), width_in=5.6):
    """Web version of the strip: every frame, with the strip's notes as running captions."""
    from matplotlib import animation
    from IPython.display import HTML, display
    df = tracking(play)
    span_w, span_d = lateral[1] - lateral[0], window[1] - window[0]
    fig, ax = plt.subplots(figsize=(width_in, width_in * span_d / span_w + 0.4))
    tmap = df.groupby("frameId")["time"].first()

    def update(fid):
        ax.clear()
        t = float(tmap[fid])
        panel(ax, play, df, t, lateral, window, rings, ball=1.5)
        cur = [n.replace("\n", " ") for tt, n in zip(times, notes) if tt <= t + 1e-6]
        ax.set_title(("snap" if abs(t) < 0.05 else f"{t:+.1f} s") + (("   " + cur[-1]) if cur else ""),
                     fontsize=8.5, color=style.INK, loc="left")
        return []

    anim = animation.FuncAnimation(fig, update, frames=sorted(df["frameId"].unique()), interval=100)
    html = (anim.to_html5_video(embed_limit=40).replace("<video ", '<video style="max-width:100%;height:auto" ')
            if shutil.which("ffmpeg") else anim.to_jshtml(default_mode="once"))
    plt.close(fig)
    display(HTML(html))


def animation_or_strip(play, times, notes, lateral, window, rings=(), ncols=2, frame_rings=None):
    """Video on the website; a numbered, captioned frame strip in the PDF."""
    if pdf_build():
        strip(play, times, notes, lateral, window, ncols=ncols, rings=rings, frame_rings=frame_rings)
        plt.show()
    else:
        video(play, times, notes, lateral, window, rings=rings)


def pct(x, nd=0):
    return f"{100 * x:.{nd}f}%"

# ---- 1. The tendency report: Seattle's 2025 defense against dropbacks, by situation
TR_COLS = ["season", "season_type", "game_id", "play_id", "posteam", "defteam", "play_type", "down", "ydstogo",
           "yardline_100", "qb_dropback", "epa"]
SITS = ["1st & 10", "2nd & 7+", "2nd & 1-6", "3rd & 1-3", "3rd & 4-6", "3rd & 7+", "Red zone"]
TWO_HIGH = ["COVER_2", "2_MAN", "COVER_4", "COVER_6"]
COV_NAME = {"COVER_0": "Cover 0", "COVER_1": "Cover 1", "COVER_2": "Cover 2", "COVER_3": "Cover 3",
            "COVER_4": "Cover 4", "COVER_6": "Cover 6", "2_MAN": "2-Man", "COVER_9": "Cover 9"}


def situation(down, togo, yl):
    if yl <= 20:
        return "Red zone"
    if down == 1:
        return "1st & 10" if togo == 10 else None
    if down == 2:
        return "2nd & 1-6" if togo <= 6 else "2nd & 7+"
    if down == 3:
        return "3rd & 1-3" if togo <= 3 else ("3rd & 4-6" if togo <= 6 else "3rd & 7+")
    return None


def dropbacks(season):
    """Regular-season dropbacks with the defense's charted rushers and coverage (participation file)."""
    d = pbp([season], columns=TR_COLS)
    d = d[(d["season_type"] == "REG") & d["play_type"].isin(["pass", "run"]) & (d["qb_dropback"] == 1)
          & d["down"].notna()].copy()
    part = nfl().load_participation([season]).to_pandas()[
        ["nflverse_game_id", "play_id", "number_of_pass_rushers", "defense_man_zone_type", "defense_coverage_type"]]
    d = d.merge(part, left_on=["game_id", "play_id"], right_on=["nflverse_game_id", "play_id"], how="left")
    d["sit"] = [situation(a, b, c) for a, b, c in zip(d["down"], d["ydstogo"], d["yardline_100"])]
    d = d[d["sit"].notna() & d["defense_coverage_type"].notna()
          & ~d["defense_coverage_type"].isin(["BLOWN", "COMBO"])]
    d["rush5"] = d["number_of_pass_rushers"] >= 5
    d["man"] = d["defense_man_zone_type"] == "MAN_COVERAGE"
    d["two_high"] = d["defense_coverage_type"].isin(TWO_HIGH)
    return d


def tendency_report(d, team=None):
    t = d[d["defteam"] == team] if team else d
    g = t.groupby("sit").agg(n=("epa", "size"), rush5=("rush5", "mean"), man=("man", "mean"),
                             two_high=("two_high", "mean"),
                             top=("defense_coverage_type", lambda s: COV_NAME.get(s.value_counts().index[0], "?")),
                             top_share=("defense_coverage_type", lambda s: s.value_counts(normalize=True).iloc[0]))
    return g.reindex(SITS)

# ---- 2. Does the opening script gain more? Efficiency by the offense's snap number, 2022-2025
SC_COLS = ["season", "season_type", "week", "game_id", "play_id", "posteam", "play_type", "down", "wp",
           "half_seconds_remaining", "pass", "epa", "success"]
SEASONS = [2022, 2023, 2024, 2025]
BINS = [0, 5, 10, 15, 20, 25, 30, 35, 40, 45, 50, 200]
BIN_LABELS = ["1-5", "6-10", "11-15", "16-20", "21-25", "26-30", "31-35", "36-40", "41-45", "46-50", "51+"]


def personnel_code(s):
    """'1 C, 2 G, 1 QB, 1 RB, 2 T, 1 TE, 3 WR' (or pre-2023 '1 RB, 1 TE, 3 WR') -> '11' (+OL if 6+ linemen)."""
    if not isinstance(s, str):
        return None
    n = {k: int(v) for v, k in re.findall(r"(\d+) ([A-Z]+)", s)}
    ol = sum(n.get(k, 0) for k in ("C", "G", "T", "OL")) or 5
    return f"{n.get('RB', 0) + n.get('FB', 0)}{n.get('TE', 0)}" + ("+OL" if ol > 5 else "")


def snaps(seasons):
    d = pbp(seasons, columns=SC_COLS)
    d = d[(d["season_type"] == "REG") & d["play_type"].isin(["pass", "run"]) & d["down"].notna()]
    d = d.sort_values(["game_id", "play_id"]).copy()
    d["snap_no"] = d.groupby(["game_id", "posteam"]).cumcount() + 1     # the offense's nth snap of the game
    part = nfl().load_participation(seasons).to_pandas()[["nflverse_game_id", "play_id", "offense_personnel"]]
    ftn = nfl().load_ftn_charting(seasons).to_pandas()[["nflverse_game_id", "nflverse_play_id", "qb_location",
                                                        "n_offense_backfield"]]
    d = d.merge(part, left_on=["game_id", "play_id"], right_on=["nflverse_game_id", "play_id"], how="left")
    d = d.merge(ftn, left_on=["game_id", "play_id"], right_on=["nflverse_game_id", "nflverse_play_id"], how="left")
    d["pers"] = d["offense_personnel"].map(personnel_code)
    # a "look" = personnel + QB alignment + number of backs; "new" = not used by this team in an earlier game
    ok = d["pers"].notna() & d["qb_location"].notna()
    d["look"] = np.where(ok, d["pers"] + "|" + d["qb_location"].astype(str) + "|"
                         + d["n_offense_backfield"].astype(str), None)

What a game plan is, and why there has to be one

Start with arithmetic. An NFL playbook holds hundreds of plays once you count every formation, motion and protection they can be run from. A team gets three real practices a week during the season, and only about one practice a week in pads: fourteen padded practices in the whole regular season.1 A player can learn and rehearse only so many new things in three days, and every snap of practice spent on one play is a snap not spent on another. So each week a staff has to choose, and what it chooses is the game plan: the slice of the playbook picked for one opponent, plus the rules for using it. Which runs and passes, from which formations, in which situations. Which coverages and pressures. Which of the opponent’s players to attack, and which to avoid.

What a Play Really Is drew the funnel: playbook, then game plan, then call sheet. This chapter is the middle step, and it is most of a coach’s working life. A game plan answers four questions, and the rest of the chapter takes them one at a time.

  1. What do they do? The opponent’s habits, by situation. That’s the tendency report, built by the quality control staff from film.
  2. What do they do best, and who does it? The thing the opponent most needs to work: a run, a receiver, a pass rusher, a blitz. The plan either takes it away or protects against it.
  3. Where can we win? The matchups: which of our players should beat which of theirs, and how to arrange the formation so those pairings happen.
  4. What do they think we’ll do? The opponent has scouted us. The plan includes which of our own tendencies to keep, hide or break (the self-scout from Constraint Theory).
ImportantWhy it exists

A game plan exists because time is scarce and every opponent is different. Practicing everything means practicing nothing well; practicing the same thing every week means the opponent knows what’s coming. The plan is the compromise: a stable core of plays the team runs every week (so they are sharp) dressed and aimed differently for each opponent (so they are hard to scout). The trade is real. Every play in the plan gets practice time and can be called with confidence; every play left out is unavailable on Sunday, however well it might have worked. A coach who loves a play that isn’t in this week’s plan can’t call it. That is why what’s left out of a game plan matters as much as what goes in.

NoteMadden vs. real life

In a video game, every play in the book is available on every snap, and your players execute a play you’ve never called as well as one you call every week. Neither is true in real life. A play-caller chooses from a few dozen calls per situation that his players rehearsed on Wednesday, Thursday and Friday, and a play the players haven’t practiced this week is a play somebody is likely to get wrong. The coach’s real menu is not the playbook. It’s the practice schedule.

Monday: close the last game, open the next one. The morning after a game belongs to the last game. Coaches grade every player on every snap, the players come in for treatment, a lift and a film session on what went wrong, and the staff decides what to fix. By the afternoon the coordinators have turned to the next opponent, and they are not starting from scratch. The quality control coaches have been breaking that opponent down for a week or more (more on that below), so the coordinators spend the day watching film with a stack of reports already beside them.

Tuesday: the coaches’ day. At most teams Tuesday is the players’ day off, and the longest day of the week for the coaches.2 This is when the first-and-second-down game plan gets built: which runs, which passes, from which formations, against which of the opponent’s fronts and coverages. It’s when the first draft of the opening script appears, and when the plays the players will see on Wednesday morning are drawn, named and loaded onto their tablets. Some coordinators sleep at the facility. John Wolford, then a Rams quarterback, described getting the week’s play calls on Tuesday night, “though oftentimes we do not get these until Wednesday morning.”3

Wednesday: install one, first and second down. The first practice day teaches the plays that will be called most. First and second down are roughly three of every four snaps an offense runs, so they get the first meetings, the first walk-through and the most practice repetitions. The defense mirrors it: its base fronts and coverages against this opponent’s base runs and passes. A scout team of backups runs the opponent’s plays, drawn from the quality control coaches’ cards, so the starters see what they’ll see on Sunday. A scout card is a drawing of one opponent play on a sheet of poster board: the scout quarterback holds it up in the huddle, each backup finds his assignment on it, and they line up and run it.

Thursday: install two, third down. Third down is its own game (see Third Down): the defense’s pressures and disguises come out, and the offense needs answers to each. Thursday adds the third-down plan, the protections against the opponent’s blitzes, and usually some changes to Wednesday’s plan after the staff has watched Wednesday’s practice film. A quarterback calls it the hardest day of the week: “Bodies are sore from yesterday, practice is more intense, and more information is thrown at you.”

Friday: install three, the rare situations. Red zone, goal line, two-minute drill and the four-minute offense (running out the clock with a lead) come last, in a shorter practice. These are the situations with the fewest snaps and the highest stakes per snap. The call sheet is finished, and the script is set. The third phase gets its plan too: the special-teams coordinator’s return and coverage schemes for this opponent’s kicker and punter, and any fake or punt-block chance he found on film.

Saturday: walk-through and travel. A light, helmets-only rehearsal, special situations, travel for road games, and meetings at the team hotel at night, where the players often walk through the opening script. Many head coaches also hold a game-management meeting late in the week: timeouts, challenges, the end of each half and the analytics staff’s decision card, so that the clock decisions on Sunday are rehearsed rather than improvised.

Sunday: the sheet. Everything above is now a few hundred calls on one or two laminated pages in the play-caller’s hand, plus a booth full of people charting what the opponent actually does. Only a fraction of those calls get used: in 2025 an offense ran about 60 plays a game (runs and passes, not counting kicks or plays wiped out by penalties), and many of those were the same favorite calls more than once.4 The sheet is big because nobody knows in advance which sixty will be needed.

Who decides? The coordinators build their sides of the plan and usually call them on Sunday, but the head coach signs off on the whole thing, and on the decisions that belong to the whole team (fourth downs, two-point tries, timeouts, how aggressive to be) he has the final word. A head coach who calls one side himself, as several do, is also that side’s coordinator in all but name.

ImportantWhy it exists: why the installs go in that order

The week runs from the most common situations to the rarest. First and second down get Wednesday, the freshest day with the most practice time, because they’re most of the game; red zone and two-minute get Friday’s short practice because they are a small share of snaps. The defense has the same calendar, for the same reason. The trade is plain: the situations that decide close games (a fourth-and-goal, the last two minutes) get the least rehearsal. Coaches cover that gap by keeping their red-zone and two-minute plans small and familiar, built from the same few concepts week after week, with a new wrinkle or two for the opponent.

Two weeks break the pattern. A short week before a Thursday game compresses everything into about three days. The players come in sore on Monday, there is no time for a real practice in pads, and the install happens mostly in meetings and walk-throughs, so game plans on Thursday nights are usually smaller and simpler. A bye week goes the other way: the players get several days off, and the staff turns its film study on itself. That’s when most teams do the formal self-scout from Constraint Theory, and it is often when they plan the tendency breakers for the rest of the season.

Film first: the breakdown and the tendency report

Every game plan starts with film, and the film is broken down long before the coordinators see it. The quality control coaches (QC coaches) from What a Play Really Is take the opponent’s recent games (often the last four or five, plus any game against an offense or defense like their own, and last season’s meeting if the coordinator hasn’t changed) on coaches film, and tag every snap. A tag is a row in a database: down, distance, yard line, hash, quarter, score, personnel, formation, motion, the play (run direction, pass concept, protection), the defense’s front, coverage and pressure, and the result. A season of one team’s snaps is a thousand-plus rows, and a QC coach may tag several opponents at once, a week or two ahead.

Those rows become the tendency report: the written breakdown of what the opponent does in each situation, personnel group and formation. On offense it reads like “from 12 personnel, under center, on first down: run 68%, mostly outside zone to the tight end side; play-action off it when the strong safety walks down.” On defense it reads like “third-and-4 to 6: man coverage half the time, five-man pressure from the slot when we’re in empty.” The report comes with cut-ups: video playlists of every snap in one category (“all their third-and-medium blitzes,” “every red-zone snap against 3x1”) so a coordinator can watch the twenty plays behind a number rather than trust the number alone.

Two deep safeties mean fewer defenders near the line, so the first-down plan leans on runs and on the throws that beat two-high shells: the smash and dagger family from What Beats What.
  • Second-and-long and third-and-4 to 6: five or more rushers on only 14% and 11% of dropbacks, well under the league’s 22% and 28%. Seattle was content to rush four and cover. So the third-and-medium plan can lean on routes that take a beat longer to develop, because the extra rusher usually isn’t coming. But four rushers isn’t the same as a plain four-man rush. Macdonald’s defense is built on simulated pressure (a linebacker or defensive back rushes while a lineman drops into coverage; see Zone Blitzes and Simulated Pressure), and this column can’t tell the two apart. The protection still has to find the fourth rusher, and the hot throw still has to be ready.
  • Third-and-medium: man coverage on about half the snaps, the same as the rest of the league (third-and-short ran a little higher, on a handful of snaps). A tendency only matters where a team differs from what everyone does. Here the report says “expect man about half the time, like everyone else,” so the plan carries a man beater and a zone beater in every third-down box (you’ll see that pairing on the call sheet below).
  • Third-and-long: more five-man pressure than average (37%), and the loudest cell on the page: man coverage on only 19% of those snaps, against 30% for the league. Put the two together and Seattle’s third-and-long answer is pressure with zone behind it, the fire zone family from Zone Blitzes and Simulated Pressure: five come, and the coverage behind them sits in zones with a hole where a rusher left. So the call is not a man beater that needs time. It’s a quick zone beater thrown into the vacated spot, a screen that lets the rush run past, or a deeper zone beater only if the protection can pick up all five. That’s why the call sheet’s third-and-long box carries a “vs 5+” line.

Three cautions keep a report honest, and good staffs say them out loud. Sample size: the third-and-short row rests on 26 snaps, and a couple of plays either way would move it ten points. Context: a defense’s numbers depend on whom it played; a coverage rate built against pass-happy opponents is not the same as one built against run-first teams. Coding: these are charters’ labels, and different charting services code the same snap differently (in the public data, for example, the league’s man-coverage share jumps from season to season in a way that looks like a change in how snaps were charted, not in how teams played). The coverage column is also what the defense played after the snap. The tendency a staff would most like to know, how often a two-deep look before the snap turns into one deep safety after it (the disguise of Disguise, Rotation and Split-Field), isn’t in this table at all; the QC coaches chart it by hand. That is why the cut-ups come with the report. The number tells the coordinator where to look. The film tells him what he’s looking at.

CautionCommon misconception

“The tendency report tells you what they’re going to call.” It tells you what they have called, in situations like this, against opponents like the ones they’ve faced. The opponent knows its own report too: it ran a self-scout over its bye, and its coordinator may break the tendency on purpose this week, in this game, because he knows you have the report. A report shifts the odds; it doesn’t read minds. Its best use is to prepare the players for the likeliest looks so they recognize them instantly, while the plan keeps an answer ready for the tendency breaker.

Who builds what: QC coaches and the analytics staff

The staff map in What a Play Really Is showed where people sit on game day. The game week shows what they produce, and the two jobs at the bottom of that map do very different work.

The quality control coaches produce the descriptive layer: what the opponent did, tagged and counted. They build the tendency report, the cut-ups, the scout-team cards that show the backups what to run in practice, and often the first draft of the call sheet’s formatting. It is detail work at enormous volume, which is why it’s the traditional first job for a young coach, and why so many head coaches started there.

The analytics staff produces the decision layer: numbers that tell the staff what to do, not just what happened. Three of its products show up in a game week:

  • The fourth-down and two-point chart. Before the game, the analysts work out (for this opponent, this stadium, this kicker, the expected weather) when the numbers favor going for it on fourth down and when to try for two. The head coach sees the chart on Wednesday or Thursday, argues with it, and carries a version of it on Sunday, so that the decision on fourth-and-2 at the opponent’s 38 was mostly made calmly three days earlier. The math behind it is in Win Probability and Fourth Downs.
  • Efficiency splits. Where is this opponent good and bad once you adjust for situation: by down, by field zone, by kind of play, by the position of the player being targeted. You’ll see one of these in the matchup section.
  • Opponent models. How often this offense passes in each situation compared with an average team (the pass-rate-over-expected idea in Tendencies, PROE and Charting), which turns “they like to throw” into a number the defensive coordinator can plan around.

The two jobs overlap more every year: QC coaches use data tools, and analysts watch film. The difference that remains is in the question each answers. QC asks what did they do? Analytics asks what should we do about it?

Scouting the play-caller and the quarterback

A tendency report describes a team. A good game plan also scouts two people: the man calling the opponent’s plays and the man taking the snaps. Each has habits that travel with him.

The play-caller. A coordinator’s habits survive from week to week and often from team to team. Staffs study questions like these:

  • What does he call when he needs it? His favorite third-and-medium concept, the pressure he trusts on a big third down, his go-to red-zone call. Under stress, play-callers fall back on what they believe in, which makes the biggest snaps the most predictable ones.
  • What does he call after something happens? After a timeout, after a turnover, on the first play after halftime, after a big play against him. These “reset” moments are when coordinators reach for prepared calls, and the calls show up on film.
  • How does he script? His opening scripts from previous games show what he likes to test early: motions to find man coverage, a quick shot downfield, a new formation. A defense that knows his early habits can plan its own answers for the first two series.
  • Who actually calls the plays? The title can mislead. In 2026 Kyle Shanahan (49ers), Matt LaFleur (Packers) and Kevin O’Connell (Vikings) call their offenses as head coaches, and the Jets’ Aaron Glenn calls his defense, so the coordinator’s name on the depth chart is not the man to scout.6

When a play-caller changes teams, the staff simply follows him to his old film. A defense facing the 2026 Lions studies Drew Petzing’s Arizona offenses as well as Detroit’s; an offense facing the 2026 Ravens watches the Miami defenses that Anthony Weaver called, and the Chargers defenses that Jesse Minter coordinated before he became Baltimore’s head coach. That last one is the “who actually calls it” question again: with a defensive head coach and a defensive coordinator who have both run defenses elsewhere, whose habits Baltimore’s defense will show is itself a scouting question until the 2026 film answers it. The players are different, but a coordinator’s sequencing, his favorite pressures and his third-down habits usually aren’t.

Here’s what that looks like done with public data, and why the film still matters. Petzing called Arizona’s offense from 2023 through 2025.7 Measure how often his offenses threw on early downs in close games, compared with what an average team would have done in the same situations (pass rate over expected, from 13-04), and you get -7 percentage points in 2023 (29th in the league), +2 in 2024 and +8 in 2025 (1st).8 Same play-caller, opposite ends of the league. The season averages followed Arizona’s roster and its quarterback situation, not Petzing’s character, so a defense preparing for his Detroit offense learns little from them. What travels is smaller and lives on film: what he calls after a timeout, how he likes to open a second half, which concept he trusts on third-and-5. That is why the QC coaches cut up his Arizona third downs rather than quoting his pass rate.

The same problem makes Week 1 the hardest game to plan for: a new coordinator, new players and an off-season of changes, with nothing yet on film from this team in this system, and preseason games in which starters barely play and nobody shows his real plan. So staffs plan Week 1 from the coordinator’s old film and last season’s tape, keep their own plan simpler than usual, and expect to learn the real answers in the first quarter, which puts extra weight on the opening script.

Scouting has limits, and the line is clear: film, public data and what a team shows on the field are fair game; spying on the opponent is not. The league bans videotaping an opponent’s signals from the sideline or booth, and when the Patriots were caught taping the Jets’ defensive signals in Week 1 of 2007 (“Spygate”), the league fined Bill Belichick $500,000 and the club $250,000 and took away its 2008 first-round pick.9

The quarterback. The opposing quarterback is studied like a coverage. A defensive staff wants to know:

  • Where does he go under pressure? Does he climb the pocket or drift back, escape to his right or his left, throw it away or take the sack?
  • How does he handle disguise? Does he hold the ball when the safeties rotate late (the disguise from Disguise, Rotation and Split-Field), or throw on rhythm to where the coverage was?
  • What’s his favorite answer? The receiver he looks for on third down, the throw he trusts against the blitz, the throws he avoids (some quarterbacks rarely throw the deep out to the far side of the field).
  • What does he change at the line? Which checks he’s allowed, and what he checks to against which looks. If he always checks to a run against two-deep safeties, the defense can show two deep and play one.

The answers become the defense’s rules for the week: “make him hold it” (disguise late, cover longer), “keep him in the pocket” (rushers stay in their lanes rather than looping inside), “make him throw left,” “take away his first read on third down and see if he gets to his second.” An offense does the same to the defense’s best pass rusher: which side he rushes from, what move he wins with, whether he drops into coverage on pressures, and therefore whether the plan needs a tight end or a back to help block him, a protection that slides toward him, or quick throws that get the ball out before he arrives.

Matchup hunting

Football is eleven against eleven, but it’s decided one pairing at a time. A matchup is the pairing of one offensive player with the defender responsible for him, and a mismatch is a pairing one side should win clearly: a fast tight end covered by a slower linebacker, a big receiver against a short cornerback, a running back out of the backfield against a linebacker who can’t turn and run. Matchup hunting is the offense’s half of the game plan that asks where can we win? and then arranges formations, motions and personnel so that those pairings actually happen on the field.

Staffs build the answer as a grid: every one of their receivers against every defender who might cover him, graded from scouting (speed, size, how each defender has fared on film against players like ours). Figure 44.3 is a made-up example for a sample game.

The notes on the right of Figure 44.3 are the important part, because a mismatch on paper is worth nothing until the formation produces it. Offenses have three tools for that.

  • Formation. Defenses assign coverage by rules: in a typical Cover 1 (man coverage with one deep safety), the linebackers take the backs and tight ends, the nickel takes the slot. So an offense that knows the rule can place its tight end where the rule hands him to a linebacker, such as the inside receiver in a three-receiver set. Detaching a tight end from the line, or putting him in the slot, is matchup hunting by alignment. It’s a question, not a guarantee: against a tight end who can really run, many defenses now answer with a safety on him and give the linebacker the back. Then the offense has learned something else, such as who the robber is this week, or that there isn’t one.
  • Motion. Motioning a player across the formation or out of the backfield does two jobs at once: it shows whether the defense is in man (a defender runs with him) and it moves the matchup. Send the running back out wide and, in man coverage, a linebacker has to follow him into open space.
  • Personnel. This is the oldest version. If the offense sends in two or three tight ends, the defense usually answers with bigger players (base or heavy personnel); then the offense spreads those tight ends out and throws against linebackers. If the defense answers with smaller, faster defensive backs, the offense keeps the tight ends in tight and runs at them. A receiving tight end who can also block is valuable precisely because he forces the defense to pick its poison, which is the conflict you met in Personnel Groupings.

The defense hunts too, in reverse. It looks for the offense’s weak blocker to rush against, the receiver who can’t beat press coverage, the back who can’t pick up a blitz. And it protects its own weak links by substitution (a sixth defensive back for a slow linebacker), by rules that hand the dangerous receiver to the right defender (a shadow corner), or by sending help.

Two lessons hide in the whiskers. The first is noise: most defenses’ intervals cross zero, so the middle of the chart is mostly luck, and a staff that built a plan on a 0.1 difference would be chasing ghosts. The second is change. Take away their best thing

The defensive half of the second question, what do they do best?, produces the most famous principle in game planning: take away the thing the opponent most needs, and make it beat you some other way. If an offense lives on one runner, stop the run and dare the quarterback to win the game. If it lives on one receiver, double him and make the other four beat you. If it lives on one concept, take away that concept’s landing spot.

The principle is associated above all with Bill Belichick, whose defenses, as the Giants’ defensive coordinator and then as head coach of the Browns and the Patriots, were built anew for each opponent.10 Michael Lombardi, who worked for him for years, added an important refinement: Belichick “doesn’t take away what the opponent does best, but what their individual players do best.”11 The target is often a person (the one receiver the quarterback trusts, the back the offense is built around), not a play. Belichick’s best-known plans (the Buffalo plan in the film room below, Super Bowl XXXVI against the Rams’ Marshall Faulk, Super Bowl LIII against the Rams again) are drawn and taken apart in Case Study: Belichick and Game-Plan Football.

The most famous “take it away” plan sits in the Pro Football Hall of Fame. Before Super Bowl XXV, on January 27, 1991, Bill Belichick was the Giants’ defensive coordinator, facing a Buffalo offense that ran a fast no-huddle attack built on Jim Kelly’s passing. His plan used extra defensive backs to take away Kelly’s deep throws and told the defenders to hit the receivers hard to limit the short passing game; the handwritten plan, with sections such as “no-huddle adjustments,” went to the Hall afterwards.12 The Giants won 20–19.

What to notice: the plan conceded something on purpose. With more defensive backs on the field, the Giants were lighter against the run, and Buffalo’s running game had a good day. Belichick’s bet was that the Bills couldn’t run the clock and the scoreboard the way they could throw it, and that the Giants’ own ball-control offense would keep Kelly on the sideline. The plan in the Belichick case study shows how the alignments worked. The lesson for every game plan is the one in Figure 44.5: decide what you’ll give up, before the game, rather than finding out during it.

Offenses play the same game against the defense’s best player. If the opponent’s best player is a pass rusher, the plan “takes him away” with help: a tight end or back chipping him before releasing, a protection that slides toward him, quick throws on his side, runs straight at him (a rusher who is flying upfield is often vulnerable to a run right where he lined up), and screens that turn his speed against him. The cost: every helper on the rusher is one fewer receiver in the pattern.

A game plan can also be shaped by things that have nothing to do with the opponent’s roster. On a Monday night in December 2021, in winds strong enough to make passing a coin flip, Belichick’s Patriots beat the Bills in Buffalo 14–10 while throwing just three passes all game; the plan for that night was simply to run the ball and leave the throwing, into the wind, to Buffalo.13 Nothing was taken away that night; the whole plan was built around the weather, which neither roster could change.

Walk the numbers.

(1) The openers. The first box is the opening script, which gets its own section next. Notice two things about it: it covers only first and second down (third downs and red-zone snaps go to their own boxes, and the script resumes afterwards), and some of its plays are marked as tests.

(2) Ranked lists. Each situation box holds a handful of calls, and the order matters. The first call is the one the staff expects to work best against what the tendency report says the defense will do. The others are the answers if the defense adjusts: the counter if the linebackers start flowing, the play-action if the safety comes down. A play-caller who has used the first call twice moves down the list, which is sequencing in printed form. Real boxes are split much further than this mock-up’s: by personnel group or formation, and often by hash (one list for the ball on the left hash, another for the right, because the wide side of the field changes what works). That’s why real sheets are so dense.

(3) Paired calls. Where the report says the defense is a coin flip (man or zone on third-and-medium), the sheet carries one coverage beater for each answer, and the play-caller picks by what the booth has seen today, or gives the quarterback both and lets him choose at the line once he sees the defense: the kill call from The Pre-Snap Phase.

(4) Shots with triggers. “Shot” plays are deep throws that need a specific look to work, so the box says when to call them: when the strong safety walks down into the box, when the defense has been crowding the line on second-and-short. A shot called into the wrong look is just a long incompletion.

(5) The package. A box built for this opponent only, which the Game-plan packages section below explains. “Viper” is not a football term you need to look up: in this example it’s the sample opponent’s own name for its favorite third-down blitz, the name the offense’s QC coaches heard or read on film and kept, so that everyone on the offense means the same blitz when they say it.

(6) The decisions already made. The analytics staff’s fourth-down and two-point chart lives on the sheet so the head coach isn’t doing arithmetic with 70,000 people screaming. Decoded, the box says: on fourth-and-1, go for it anywhere from our own 35-yard line on; on fourth-and-2 or 3, from the opponent’s 45 on; and after a touchdown, go for two if the score (before the try) has us down 2 or 5 or up 1 or 5. Down 2, a two-pointer ties; down 5, it makes a field goal enough to tie; up 1 or 5, it stretches the lead to 3 or 7, so a field goal or a touchdown-and-kick no longer beats us. (The numbers in the mock-up are invented; real charts change with the opponent, the score, the clock and the kicker.)

The unnumbered boxes are the situations every sheet carries: two-minute (hurry, with a spike to stop the clock), four-minute (runs that keep the clock moving when protecting a lead), backed up (inside our own 5, where a safety or a turnover is a disaster, so the list is runs and one max-protection shot), and sudden change (the first call after a timeout or a takeaway, which the scouting section said opponents chart). The play names are shorthand for the formation, motion and play, in an invented language built on the course’s numbering: “Gun Trey Rt Zoom Y Opt” is shotgun, three receivers to the right with the tight end inside, the back motioning out (here called “zoom”), and the tight end’s option route; “Post-Whl” is a post with a wheel route under it, “Go-Whip” a go route paired with a whip route that breaks in and then snaps back out, “Fly” a jet-style motion across the formation, and “vs 5+” means “if they rush five or more.”

The sheet is offensive. The defensive coordinator’s sheet across the field looks different: it is filed by situation and by the offense’s personnel and formation, because the defense has to answer what it sees. A box might read “3rd & 4-6 vs 11 personnel, 3x1: Cover 1 robber / sim pressure / 2-Man,” with the same ranked and paired logic underneath. Defensive coordinators plan their opening series too, though rarely as a strict list: a few fronts, coverages and pressures they want to show early, partly to see how the offense’s protection and quarterback answer them.

Sheets differ as much as the coaches who hold them. Andy Reid’s is famously big: during one of the Chiefs’ Super Bowls, Waffle House posted his play card next to its own menu.14 In Kansas City the call has traveled in two steps: Reid says a short name for the play into his headset, and his offensive coordinator finds it on his own copy of the sheet and relays the full call, about a dozen words of formation, motion and play, to the quarterback, ideally with 20 seconds left on the play clock. The rush is the radio: the coach’s voice in the quarterback’s helmet cuts off with 15 seconds left (see What a Play Really Is), so a call that isn’t in by then doesn’t arrive.15 Kyle Shanahan, by contrast, has refused to number his plays for a wristband system: a former quarterback of his said Shanahan’s view was that he “can’t put it numerically because he needs to watch the defense” before choosing the words.16

The opening script

The opening script is a list of plays, usually about fifteen, that the play-caller decides before the game to call in order at the start. It’s the most visible fingerprint of the game week on a broadcast, and it’s older than most of the people who use it.

Where it came from. Bill Walsh, the 49ers’ head coach from 1979 to 1988, is generally credited with popularizing the practice of scripting the first fifteen offensive plays of a game, and his scripts often ran into the twenties. Walsh was a former high school coach and an assistant to Paul Brown (the founding coach of both the Browns and the Bengals). He planned everything, and he gave several reasons for the script, all of them about the gap between a quiet room on Friday and a loud sideline on Sunday.17 Decisions made calmly during the week are better than decisions made in the emotion of the first quarter. A script guarantees that the plays the staff liked best actually get called, rather than forgotten when the game starts fast. Players who know the first plays in advance can rehearse them, and they start the game calmer. And a script lets the offense ask the defense questions in a deliberate order and watch the answers.

What’s on one. A modern script is still mostly first-and-second-down plays that the team runs well, chosen against the opponent’s tendencies, plus some deliberate questions. Each test play is there to find out how the defense will handle something the offense wants to use later:

  • A new formation, to see how the defense aligns to it (in the opening scene, two detached tight ends: does the defense keep its linebackers on the field and walk them out, or substitute?).
  • A motion, to see who runs with it (man or zone) and whether the safeties rotate.
  • A personnel group, to see what the defense sends on in response.
  • A shot, often early, to see whether the deep safety is disciplined, and to plant the idea that he has to be.

That’s the answer to the opening scene’s last question. The first play was a test, and it got its answer: this defense keeps its base personnel on the field against two split tight ends and walks two linebackers out to cover them. Running the same play again on the second snap would have spent the answer while the defense was still thinking about it. It’s worth more saved, for a third-and-5 when a tight end against a linebacker in space is exactly what the offense needs. The answer has a second half too: count the box. With two linebackers walked out, five defenders are left near the ball against five offensive linemen, so the same formation is also a good run look, and many offenses carry a run-pass option from it that reads the walked-out linebacker (throw to the tight end if he stays inside, hand off if he widens).

The booth charts every answer. By the end of the first quarter, the play-caller knows things that no amount of film could have told him, because the film showed this defense against other offenses and the script showed it against his. Those answers drive the calls for the rest of the game, which is the subject of In-Game Adjustments, Tempo, and the Sideline Battle.

What a script is not. It is not a promise to call fifteen plays in a row no matter what. Third downs, red-zone snaps, penalties and field position all interrupt it (many scripts carry “backed-up” alternatives, because a scripted shot or reverse is not what you call from your own 2): the play-caller goes to that situation’s box, then comes back to the next unused play on the script. And a play-caller who sees something he likes, or something that’s going badly, can leave the script at any time; most stay with it through the first couple of series because that’s the point of having one.

NoteFilm room: Kansas City Chiefs, 2019 season — Andy Reid’s index card

Before Super Bowl LIV, reporters asked how Andy Reid builds his openers. The answer was a 3-by-5 index card. Reid writes his first fifteen plays on it in red pen, first and second downs only (third down and red zone come from the sheet), and on Friday evening he texts a picture of it to his offensive coordinator with one word: “Thoughts?” They talk it through on Saturday morning, and the quarterbacks have their say. Matt Nagy, a longtime Reid assistant who had been his offensive coordinator in Kansas City, described how the list settles: “He’ll know there’s a handful of plays that everybody really likes, and those are the ones that end up in the first 15.”18 Reid also draws new plays on index cards early in the week and hands them to the quarterbacks, some of them with three players in motion at once, and the group picks a favorite or two to try in that week’s install; one former assistant remembered him sending more on a Friday night: “I’ll put on another 15, I’m feeling it.”19

What to notice: the script is a collaboration that ends with the quarterback’s comfort. A play the quarterback doesn’t like rarely survives to Sunday, because the opening plays are the ones the whole offense has rehearsed most and believes in. Watch the Chiefs’ first two drives for plays you haven’t seen from them before. Many of them come from those early-week index cards.

The right panel shows where the script’s fingerprint does appear. About 4.0% of early snaps came from a look the team had never used before that season, against 3.6% in the middle of the game and 2.9% late. That’s a modest difference, and the public data sees only the coarsest version of “new” (personnel, where the quarterback stands, how many backs). The novelty a defense notices, a motion it hasn’t seen, a tight end split out, a route combination from a familiar formation, is invisible in these columns.

Why doesn’t all that preparation buy yards? Three reasons, and they explain what the script is really for.

  1. The defense has a script too. Defensive coordinators plan their opening series as carefully as offenses do, with their own new wrinkles and their own questions. Prepared meets prepared.
  2. Test plays are asked, not won. A play called to see who follows the motion is chosen for what it reveals, not for its expected gain.
  3. The payoff comes later, and averages can’t see it. What the script buys is information, and the offense cashes it in after the script is over, when the play-caller calls the plays the answers pointed to. You might expect that to show up as better efficiency in snaps 16 to 60, and the left panel shows it doesn’t. That isn’t proof the information is worthless. League-wide averages can’t see it, because the defense is learning from the same snaps and adjusting too (that’s In-Game Adjustments), and one offense’s edge is the other side’s loss in the same game. The value of a good question shows up in single games and single calls, not in a 32-team average.

So the honest summary is: the script is a tool for asking good questions and starting calmly, not a magic list of plays that work better. That fits what Walsh said it was for.

Game-plan packages

Most of a game plan is the team’s stable core, aimed at this opponent. A small part is new: a game-plan package, a group of plays, formations or calls built for one opponent and one purpose, taught that week, and often retired after the game. The “vs Viper pressure” box on the call sheet is one: three answers to the opponent’s favorite third-down blitz, which the QC coaches flagged in their blitz report, drawn up on Wednesday night as part of the third-down plan and installed Thursday.

Packages come in many sizes:

  • An offensive package against one defensive habit: a set of motions to identify a man-heavy defense’s coverage, a screen built to punish a defense that blitzes the slot defender, a run designed against one defensive line’s stunt.
  • A matchup package: the formations that create the hunted pairing from Figure 44.3, such as the 3x1 set with the tight end inside, plus two or three plays from it that feature him.
  • A defensive package against one offensive threat: the dime bracket in Figure 44.5, a special pressure aimed at a protection weakness found on film, a disguise built for a quarterback who throws on rhythm.
  • A trick play saved for one opponent: a reverse or a throwback designed against a defense that pursues hard.

Packages are small for the reason this chapter started with: practice time. A package that needs new rules for every player won’t be ready by Sunday, so the best ones are built from things the team already knows: the same play in a new formation, the same coverage with one player’s job changed. That’s the illusion of complexity again, used for one week. And packages are retired because they go on film. The moment a package works, every future opponent’s QC coaches tag it, and the next defense will have a plan for it. Some come back later in the season, when the opponent least expects them, as the tendency breakers a self-scout recommends.

CautionCommon misconception

“Teams install a whole new offense every week.” They don’t, and they couldn’t. Players run their core plays hundreds of times over a season, and that’s what makes them fast and confident. What changes each week is the selection, the packaging (formations, motions, personnel) and the aim: which matchups, which tendencies, which situations. A few genuinely new plays appear each week, in packages small enough to learn in three days.

TipWatch for it on Sunday: the script detector

Pick one offense and watch its first fifteen offensive plays with a notepad. You’re looking for the parts of the plan that were written on Tuesday.

Before each snap

  • Write down the personnel group and formation. Is it something you haven’t seen from this team before? Mark it NEW.
  • Is there a motion? Watch who goes with it. That’s a test question: man or zone?
  • Are the plays alternating in a way that ignores the down? Scripts are built for ordinary downs, so a second-and-9 shot play or a first-down screen out of nowhere is often the script talking.

After each snap

  • Did the play seem designed to find something out (a quick throw to a detached tight end, a motion followed by a simple run)? Note what the defense did.
  • After the first two drives, look at your list. Every NEW mark is a part of this week’s game plan. Watch for those looks to come back later, especially on third down and in the red zone, with a different play attached.
  • Count when the script ends: when the offense starts repeating looks and the calls start following the down and distance closely, the play-caller has moved from the script to the sheet.

And on the sideline: watch the play-caller hold the sheet over his mouth while he talks into the headset. He is hiding his lips from anyone across the field, in the stands or on a broadcast close-up who might read the call, and that sheet in front of his face is the week you just read about.

So Y’s option route (he reads his defender and breaks away from him at five or six yards) goes out, toward the sideline, where neither the Will nor the robber can get to it in time. It’s the first call in the “3rd & 3-6, vs man” box of the call sheet, and it’s built for exactly this picture.

A caution about the tell. A linebacker widening with a motioning back suggests man, but zone defenses also “bump” a defender out with motion. The real tell is whether he runs all the way with the back and whether the rest of the defense rotates (see Disguise, Rotation and Split-Field), and on third down smart defenses disguise this on purpose.

Partial credit if you said the back: a running back matched on a linebacker in space is the other pairing the grid liked. And notice what the defense could have done instead: if this team had built the dime bracket from Figure 44.5 into its game plan, a defensive back would be on Y and a safety over him, and the right answer would be the other side of the field. Which picture you see before the snap tells you which game plan you’re watching.

The defense has taken away the offense’s favorite run. What should the offense call, and what did the defense pay for this alignment?

TipAnswer

The best call is play-action off the outside-zone look, with a shot behind the safety who came down. Start with what the defense did. The end is set inside the tight end, so Y can’t reach him and the edge is set hard; the strong safety is an extra run defender outside him; and the front is leaning that way. That’s a defense taking away the best thing. (Notice it did not do it with a wide-9 end: an end lined up far outside the tight end makes outside zone easier, because the tight end just pushes him wider and the back cuts up inside him, the trade from Even Fronts.)

Now find the bill. There’s one deep safety, and he has the whole deep middle. The Will has been pulled out of the box to cover H, so a linebacker is matched on a slot receiver. And everything is shaded right, so the left side of the line has fewer defenders than usual. Play-action that looks exactly like outside zone right attacks the biggest of those bills: the strong safety and the linebackers have spent all week being told to fly at that run, so they bite on the fake, and a receiver runs behind them into the space the one deep safety can’t cover alone. This picture is the trigger printed on the call sheet’s SHOTS box (“when SS walks down”). The other answers are good alternatives: a quick throw to H on the linebacker, the safest completion; or a run to the left, away from the crowd, such as inside zone or a counter. The constraint tree from Constraint Theory is a list of exactly these answers.

On second-and-3 the offense has a free down: even an incomplete pass leaves a manageable third-and-3. That’s why this is the classic moment for the play-action shot, and why the call sheet’s “2nd & 1-3 (shot down)” box exists. If you said “outside zone right anyway,” you’ve found the call this defense spent its week preparing for.

Takeaways

  • A game plan is a choice under a time limit. Three practices a week can’t cover a playbook, so each week a staff picks the slice aimed at one opponent: its habits, its best thing, the matchups available, and what it expects from us.
  • The week runs from common to rare. First and second down on Wednesday, third down on Thursday, red zone and two-minute on Friday, with the coaches’ Tuesday in front of it all and the QC coaches’ breakdown a week or more ahead.
  • Quality control describes; analytics decides. The QC staff’s tendency report and cut-ups say what the opponent did; the analytics staff’s fourth-down charts and efficiency splits say what to do about it. Both shift odds; neither reads minds, and the opponent has scouted itself too.
  • Matchup hunting is arranged, not found. A good pairing on paper only matters once formation, motion or personnel forces it onto the field; defenses avoid bad pairings by substitution, shadowing and help.
  • Taking away the best thing always sends a bill. Every defender added against the opponent’s strength is taken from somewhere else, and the best game plans decide in advance where they’d rather be beaten.
  • The opening script buys information and composure, not yards. In 2022–2025 neutral situations, scripted snaps were no more efficient than later ones; they were somewhat more likely to show something new, and the answers they get pay off later in the game.
