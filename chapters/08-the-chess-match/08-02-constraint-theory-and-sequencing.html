def O(key, pos, d, w, label=None):
    return Player(key, pos, "off", d, w, label=key if label is None else label)


def D(key, pos, d, w, label=None):
    return Player(key, pos, "def", d, w, label=label)


def line(labels=False):
    """Five offensive linemen; unlabeled except the center unless labels=True."""
    names = ["LT", "LG", "C", "RG", "RT"]
    return [O(k, k, -0.7, w, k if (labels or k == "C") else "")
            for k, w in zip(names, (-3.2, -1.6, 0.0, 1.6, 3.2))]


def offset_i(labels=False):
    """21 personnel, offset I, tight end right, receivers in reduced (tight) splits. Seven on the line:
    five linemen, Y and X; Z is off the ball, and Q, F and R are backs."""
    return line(labels) + [O("QB", "QB", -1.9, 0.0, "Q"), O("FB", "FB", -4.5, 1.5, "F"),

def threat(fld, p, *keys):
    """Dashed brown rings: defenders threatening to blitz before the snap (not the play's target)."""
    for k in keys:
        x, y = _P(fld, p, p.players[k].d, p.players[k].w)
        fld.ax.add_patch(Circle((float(x), float(y)), 1.25, fc="none", ec=style.BALL, lw=1.5, ls=(0, (2.2, 1.4)),
                                zorder=7))


def go_to(p, key, end, kind="block", delay=0.0, speed=None, via=()):
    """Move `key` through absolute (d, w) points `via` to `end`, as a block, path, run, route or pull."""
    start = np.array(p._cur(key))
    pts = [tuple(np.array(q) - start) for q in (*via, end)]
    if kind == "block":
        return p.block(key, pts=pts, delay=delay, speed=speed)
    if kind == "pull":
        return p.pull(key, pts=pts, delay=delay, speed=speed)
    if kind == "route":
        return p.route(key, pts, frame="field", delay=delay, speed=speed)
    return {"path": p.path, "run": p.run}[kind](key, pts, delay=delay, speed=speed)


def oz_first_second(p, fake=False):
    """The shared first second of the outside-zone family: every lineman steps right, Q opens right
    (and on a fake the back runs the outside-zone track without the ball)."""
    for k in ("LT", "LG", "C", "RG", "RT"):
        p.block(k, pts=[(0.2, 1.4)])
    go_to(p, "QB", (-3.3, 1.7), "path", speed=3.0)
    if fake:
        go_to(p, "RB", (-1.2, 4.6), "path", via=[(-5.6, 1.6), (-2.6, 3.8)])


def _P(fld, p, d, w):
    return fld.xy(*p.to_abs(d, w))


def say(fld, p, d, w, text, color=style.INK_2, fontsize=7.0, ha="center", weight="normal"):
    """A short on-field label at (d, w)."""
    x, y = _P(fld, p, d, w)
    fld.ax.text(x, y, text, fontsize=fontsize, color=color, ha=ha, va="center", zorder=11, fontweight=weight,

def point_to(fld, p, text_at, target, text, color=style.INK_2, fontsize=6.8):

def badge(fld, p, d, w, text, color=RED, fontsize=7.0):

def ghost(fld, p, pts, color=style.DEFENSE, lw=1.4, alpha=0.55, zorder=5):
    """A faint dashed arrow through absolute (d, w) points: what a player did on earlier snaps."""
    xs, ys = _P(fld, p, np.array([q[0] for q in pts]), np.array([q[1] for q in pts]))
    fld.ax.plot(xs[:-1], ys[:-1], color=color, lw=lw, ls=(0, (3, 2)), alpha=alpha, zorder=zorder)
    fld.ax.add_patch(FancyArrowPatch((xs[-2], ys[-2]), (xs[-1], ys[-1]), arrowstyle="-|>", mutation_scale=10,
                                     color=color, lw=lw, alpha=alpha, ls=(0, (3, 2)), zorder=zorder))


def was_here(fld, p, d, w, to=None):
    """A faint dashed outline where a defender started, and (optionally) a dashed arrow to where he is now."""
    x0, y0 = _P(fld, p, d, w)
    fld.ax.add_patch(RegularPolygon((float(x0), float(y0)), 4, radius=0.95, orientation=np.pi / 4, fc="none",
                                    ec=style.DEFENSE, lw=0.9, ls=(0, (2, 1.5)), alpha=0.8, zorder=6))
    if to is not None:
        ghost(fld, p, [(d, w), to], alpha=0.9, lw=1.5)


def shifted(defense, **spots):
    """The defense with some players drawn where they have moved to: shifted(base_43(), WILL=(4.6, 0.2))."""
    return [q.moved(d=spots[q.key][0], w=spots[q.key][1]) if q.key in spots else q for q in defense]


def clean(ax):
    """Zoomed diagrams: drop hash ticks and yard numbers, and clip text to the window."""
    for ln in list(ax.lines):
        xs, ys = ln.get_data()
        if len(xs) == 2 and ln.get_linewidth() == 0.5 and np.hypot(xs[1] - xs[0], ys[1] - ys[0]) < 1.0:
            ln.remove()
    for t in list(ax.texts):
        if t.get_text().isdigit() and t.get_alpha() == 0.55:
            t.remove()
        else:
            t.set_clip_on(True)


def draw(p, ax=None, width_in=5.8, **kw):
    fld = p.draw(ax=ax, width_in=width_in, title=False, **kw)
    clean(fld.ax)

MARK_R = 0.6    # frame-strip marker radius in yards, smaller than the static diagrams'


def tracking(play):
    df = play.to_tracking()
    df["_id"] = df["displayName"].astype(str)
    return df


def panel(ax, play, df, t, lateral, window, rings=()):
    """One moment of a simulated play: markers, trails of on-screen players, rings."""
    tmap = df.groupby("frameId")["time"].first()
    fid = int((tmap - t).abs().idxmin())
    ylo, yhi = sorted((play.ball_y - lateral[0], play.ball_y - lateral[1]))
    fld = Field(ax=ax, orient="vertical", los=play.los, window=window, numbers=False, hashes=False,
                lateral=(ylo, yhi), to_go=play.to_go)
    cur = df[df["frameId"] == fid]
    shown = cur[cur["x"].between(play.los + window[0] + 0.4, play.los + window[1] - 0.4)
                & cur["y"].between(ylo + 0.4, yhi - 0.4)]
    ids = set(shown["_id"])
    trails = df[(df["frameId"] <= fid) & df["_id"].isin(ids) & (df["side"] != "ball")]
    for (_, side), g in trails.groupby(["_id", "side"]):
        xs, ys = fld.xy(g["x"].values, g["y"].values)
        ax.plot(xs, ys, color=style.OFFENSE if side == "off" else style.DEFENSE, lw=0.9, alpha=0.4, zorder=3)
    ppy = fld.pts_per_yard()
    for _, rr in shown.iterrows():
        x, y = (float(v) for v in fld.xy(rr["x"], rr["y"]))
        if rr["side"] == "ball":
            ax.add_patch(Ellipse((x, y + 0.45), 0.7, 0.42, fc=style.BALL, ec="white", lw=0.5, zorder=9))
            continue
        lab = str(rr.get("label", "") or "")
        if rr["side"] == "off":
            ax.add_patch(Circle((x, y), MARK_R, fc=style.OFFENSE, ec="white", lw=0.6, zorder=8))
            tc = "white"
        else:
            ax.add_patch(RegularPolygon((x, y), 4, radius=MARK_R * 1.3, orientation=np.pi / 4,
                                        fc=style.DEFENSE_TINT, ec=style.DEFENSE, lw=1.0, zorder=8))
            tc = style.INK
        fs = max(4.0, min(8.0, ppy * (0.8 if len(lab) <= 1 else 0.6)))
        ax.text(x, y, lab, ha="center", va="center", fontsize=fs, color=tc, fontweight="bold", zorder=10)
    for key, col in rings:
        rr = cur[cur["displayName"] == key].iloc[0]
        x, y = fld.xy(rr["x"], rr["y"])
        ax.add_patch(Circle((float(x), float(y)), 1.05, fc="none", ec=col, lw=1.7, zorder=7))
    return fld


def strip(play, times, notes, lateral, window, ncols=2, rings=(), width_in=6.4):
    """Numbered, captioned frame strip: the PDF stand-in for the video."""
    df = tracking(play)
    nrows = int(np.ceil(len(times) / ncols))
    pw = width_in / ncols
    span_w, span_d = lateral[1] - lateral[0], window[1] - window[0]
    fig, axes = plt.subplots(nrows, ncols, squeeze=False,
                             figsize=(width_in, nrows * (pw * span_d / span_w + 0.5)))
    for ax in axes.flat[len(times):]:
        ax.axis("off")
    for i, (ax, t) in enumerate(zip(axes.flat, times)):
        panel(ax, play, df, t, lateral, window, rings)
        ax.set_title(f"{i + 1}.  " + ("snap" if abs(t) < 1e-6 else f"{t:+.1f} s"), fontsize=8.5,
                     color=style.INK_2, loc="left")
        ax.text(0.5, -0.03, notes[i], transform=ax.transAxes, ha="center", va="top", fontsize=7.5,
                color=style.INK, linespacing=1.15)
    fig.tight_layout(h_pad=1.4, w_pad=0.6)
    return fig


def video(play, times, notes, lateral, window, rings=(), width_in=5.6):
    """Web version of the strip: every frame, with the strip's notes as running captions."""
    from matplotlib import animation
    from IPython.display import HTML, display
    df = tracking(play)
    span_w, span_d = lateral[1] - lateral[0], window[1] - window[0]
    fig, ax = plt.subplots(figsize=(width_in, width_in * span_d / span_w + 0.4))
    tmap = df.groupby("frameId")["time"].first()

    def update(fid):
        ax.clear()
        t = float(tmap[fid])
        panel(ax, play, df, t, lateral, window, rings)
        cur = [n.replace("\n", " ") for tt, n in zip(times, notes) if tt <= t + 1e-6]
        ax.set_title(("snap" if abs(t) < 0.05 else f"{t:+.1f} s") + (("   " + cur[-1]) if cur else ""),
                     fontsize=8.5, color=style.INK, loc="left")
        return []

    anim = animation.FuncAnimation(fig, update, frames=sorted(df["frameId"].unique()), interval=100)
    html = (anim.to_html5_video(embed_limit=40).replace("<video ", '<video style="max-width:100%;height:auto" ')
            if shutil.which("ffmpeg") else anim.to_jshtml(default_mode="once"))
    plt.close(fig)
    display(HTML(html))


def animation_or_strip(play, times, notes, lateral, window, rings=(), ncols=2):
    """Video on the website; a numbered, captioned frame strip in the PDF."""
    if pdf_build():
        strip(play, times, notes, lateral, window, ncols=ncols, rings=rings)
        plt.show()
    else:
        video(play, times, notes, lateral, window, rings=rings)

from gridiron.data import pbp, nfl

SEASONS = [2022, 2023, 2024, 2025]
PCOLS = ["season", "season_type", "game_id", "play_id", "posteam", "play_type", "down", "ydstogo", "wp",
         "half_seconds_remaining", "pass", "epa", "success", "drive", "yards_gained"]
GROUPS = ["11", "12", "13", "21", "22"]
ALIGN = {"U": "Under center", "S": "Shotgun", "P": "Pistol"}


def backs_tes(s):
    """'1 C, 2 G, 1 QB, 1 RB, 2 T, 1 TE, 3 WR' (or the pre-2023 '1 RB, 1 TE, 3 WR') -> '11'."""
    if not isinstance(s, str):
        return None
    n = {k: int(v) for v, k in re.findall(r"(\d+) ([A-Z]+)", s)}
    return f"{n.get('RB', 0) + n.get('FB', 0)}{n.get('TE', 0)}"


def n_linemen(s):
    """Offensive linemen on the field ('... 1 C, 2 G, 2 T ...' or a pre-2023 '6 OL'); 5 if not listed."""
    if not isinstance(s, str):
        return 5
    n = sum(int(v) for v, k in re.findall(r"(\d+) ([A-Z]+)", s) if k in ("C", "G", "T", "OL"))
    return n if n else 5


def neutral_plays(seasons):
    """Regular-season runs and dropbacks on first and second down, win probability 20-80%, outside the
    last two minutes of each half, with personnel (participation) and QB alignment (FTN charting)."""
    d = pbp(seasons, columns=PCOLS)
    d = d[(d["season_type"] == "REG") & d["play_type"].isin(["pass", "run"])].sort_values(["game_id", "play_id"])
    # was the previous snap of the same drive a run of 10+ yards? (for the "after the big run" check)
    prev = d.groupby(["game_id", "drive"])
    d["after_big_run"] = (prev["play_type"].shift(1) == "run") & (prev["yards_gained"].shift(1) >= 10)
    d["first_of_drive"] = prev["play_type"].shift(1).isna()
    d = d[d["down"].notna()]
    part = nfl().load_participation(seasons).to_pandas()[["nflverse_game_id", "play_id", "offense_personnel"]]
    ftn = nfl().load_ftn_charting(seasons).to_pandas()[["nflverse_game_id", "nflverse_play_id",
                                                        "qb_location", "is_play_action"]]
    d = d.merge(part, left_on=["game_id", "play_id"], right_on=["nflverse_game_id", "play_id"], how="left")
    d = d.merge(ftn, left_on=["game_id", "play_id"], right_on=["nflverse_game_id", "nflverse_play_id"],
                how="left")
    d["pers"] = d["offense_personnel"].map(backs_tes)
    d["ol"] = d["offense_personnel"].map(n_linemen)
    d = d[(d["down"] <= 2) & d["wp"].between(0.2, 0.8) & (d["half_seconds_remaining"] > 120)
          & d["qb_location"].isin(list(ALIGN)) & d["pers"].notna()].copy()
    d["align"] = d["qb_location"].map(ALIGN)
    d["group"] = d["pers"].where(d["pers"].isin(GROUPS), "other")
    d["look"] = d["group"] + " | " + d["align"]
    return d


def tendency_table(d, team=None, season=2024, groups=("11", "12", "13", "21")):
    """Pass rate and snap count by personnel group (rows) and quarterback alignment (columns)."""
    x = d[(d["season"] == season) & d["group"].isin(groups)]
    if team:
        x = x[x["posteam"] == team]
    cols = ["Under center", "Shotgun", "Pistol"]
    rate = x.pivot_table(index="group", columns="align", values="pass", aggfunc="mean")
    n = x.pivot_table(index="group", columns="align", values="pass", aggfunc="size")
    return (rate.reindex(index=list(groups), columns=cols),
            n.reindex(index=list(groups), columns=cols).fillna(0).astype(int))


def guess_from_film(d, min_other=8):
    """For every snap, what a scout would have predicted from the team's OTHER games that season:
    the pass rate of that personnel-and-alignment look (if seen at least `min_other` times elsewhere),
    and the team's overall pass rate. Leaving the game out keeps the guess honest."""
    out = d.copy()
    for name, keys in (("xp_look", ["season", "posteam", "look"]), ("xp_team", ["season", "posteam"])):
        g, gg = out.groupby(keys)["pass"], out.groupby(keys + ["game_id"])["pass"]
        n_other = g.transform("size") - gg.transform("size")
        p_other = (g.transform("sum") - gg.transform("sum")) / n_other.clip(lower=1)
        out[name] = p_other.where(n_other >= min_other)
    out["xp_look"] = out["xp_look"].fillna(out["xp_team"])
    out = out[out["xp_team"].notna()]
    out["hit_look"] = (out["xp_look"] > 0.5).astype(int) == out["pass"]
    out["hit_team"] = (out["xp_team"] > 0.5).astype(int) == out["pass"]
    out["said"] = np.select([out["xp_look"] < 0.45, out["xp_look"] > 0.55], ["run", "pass"], "toss-up")
    return out

Every play is an answer to a question

Start with what a defense is trying to do on every snap. It has eleven players and the offense has eleven, but the offense knows where the ball is going and the defense doesn’t. So the defense guesses, in an educated way. It studies the offense all week, learns what the offense likes, and on Sunday it leans. A linebacker who knows the offense runs outside zone half the time from this formation takes his first step toward the play side a little faster. A safety who has seen the tight end block on every snap creeps down to help against the run. A defensive end on the far side, told the cutback is where this offense hurts people, chases hard down the line.

Each of those leans makes the defense better against what the offense usually does. Each one also makes it worse against something else. The linebacker flying to the play side has left the middle of the field. The safety creeping down has nobody behind him. The end chasing down the line is no longer outside the quarterback. An offense that can see those leans, and that has a play aimed at each one, wins the argument.

That is the core of what coaches call constraint theory. Every offense has a small number of base plays: the handful of runs and passes it builds itself around, calls most often, and wants to be good at whatever the defense does. For a Shanahan-family offense the base play is outside zone; for a gap-scheme team it might be power; for a spread passing team it might be a quick-game concept or the inside-zone RPO. Then it has constraint plays: plays whose main job is to punish the defense for over-playing the base play. A constraint play works best exactly when the defense cheats against the base play, and its mere existence is what keeps the defense from cheating too much. The word comes from what it does to the defense: it constrains how far the defense can lean.1

ImportantWhy it exists

A defense that knows what’s coming wins. Constraint plays exist so that the defense can never fully commit to stopping the base play: every lean has a price, and the offense collects it. The trade is that a constraint play is a specialist. It is usually a worse play than the base play when the defense hasn’t cheated (a bootleg into an end who stayed home is a sack waiting to happen, a counter into linebackers who didn’t flow runs straight into them), so it can’t become the offense’s main course. It lives on the base play’s reputation, and the base play lives on its protection. (If that sounds like it contradicts the receiver-run and play-action numbers from earlier chapters, where these plays win on average, hold the thought: the section “Why not just run the constraint play more?” below shows why their averages should look better anyway.)

The two kinds of play need each other. Without the constraint plays, the defense cheats freely and the base play dies. Without the base play, the constraint plays have nothing to punish: nobody chases a fake of a play that’s never run. This is why coaches talk about plays “talking to each other.” A good offense is not a list of good plays. It’s a set of plays in which each one’s weakness is another one’s strength.

You won’t often hear “constraint play” on a sideline or in a clinic talk. Coaches say the same thing in their own dialects: complementary plays, answers, if-then calls (“if the Will flows, then counter”), or the “play-pass off” a run. When a coordinator says a play “keeps them honest,” he is describing a constraint play.

NoteMadden vs. real life

In a video game you can find a play that works and call it until it stops working, and often it never does, because the computer’s defense doesn’t keep a scouting report on you. Real defenses do. They watch every snap you have run this season, they know what you like from every formation, and they adjust between series, at halftime and during the week. There is no best play in real football, only a best play given what the defense is doing about the others. That is also why the answer to “if the bootleg is so good, why not run it every time?” is the same as the answer to “why not throw deep every time?”: the defense would stop leaning, and the play would turn back into an ordinary one, or worse.

CautionCommon misconception

“Constraint plays are trick plays.” A few are (a flea-flicker, in which the back takes a handoff and pitches the ball back to the quarterback, who throws deep, is a constraint on safeties who come up to stop the run), but most are ordinary, bread-and-butter football: the bootleg, the screen, the draw, the counter, the quick throw behind a blitzing linebacker. What makes a play a constraint is not that it’s unusual. It’s that its job is defined by another play.

The constraint tree: outside zone and its family

The cleanest place to see constraint theory is the offense that has made it a way of life: the outside-zone system of Mike Shanahan’s Broncos in the 1990s (built with his offensive line coach Alex Gibbs, whose story is in Zone Running) and of his son Kyle’s 49ers today, and the many coaches who came out of it. Its base play is outside zone (wide zone), the run you met in that chapter: every lineman steps hard to one side, the back aims at the tight end’s outside leg, and the defense has to run sideways with the play or get outflanked. Outside zone trains a defense, snap after snap, to flow: to run hard to the play side. Every constraint play in the system is designed around a defender who has learned that lesson too well. How many plays is that? Fewer than a Madden menu suggests: a few base runs and a few base passes, each with a small family of answers like the one below. The exact count varies by staff, but the point of the system is that it is small enough to practice until every play looks the same for its first second.

Read Figure 43.1 one branch at a time, starting from what the defense does wrong.

(1) Linebackers flow hard: the bootleg. Outside zone punishes slow linebackers; a linebacker who doesn’t beat his blocker to the play side gets sealed inside and the back runs past him. So linebackers learn to fly. The bootleg fakes outside zone one way and rolls the quarterback out the other way, behind a blocker (a pulling guard or the fullback) who takes care of the defensive end on that side; that blocker is why 04-05 also called it the protected boot. Receivers cross the field to meet him at three depths, and the crossing routes come open behind the linebackers who just ran the other way. The defense’s lesson (“flow fast”) is the very thing the bootleg cashes in.

(2) The backside end chases: the naked boot. On outside zone, the defensive end on the far side (the backside) has two jobs that pull against each other. He is the man who stops the cutback: if he stays put, the back has a lane back to his side. But he is also the last defender on that edge, so his coaching, which you met in 03-05, is to squeeze while checking boot, counter, reverse: shuffle down the line a yard or two to shrink the cutback lane, keep his shoulders square and his eyes on the quarterback, and stay at the edge until he is sure the ball has gone the other way. That is the disciplined version, and it is hard to play. When the cutback has been hurting the defense, the end starts to crash instead: he turns his shoulders, flattens down the line and chases the back. Figure 43.2 shows the difference. Crashing is the cheat. The naked bootleg sends the quarterback right at the spot the crashing end just left, with no blocker for him (that’s the “naked”). The end who crashed is now behind a quarterback who is running away from him, and nobody else is close. If the end squeezed and stayed home instead, he is standing on the quarterback’s launch point, and the quarterback is in trouble, which is the price of a play that only works against a cheat.

The keeper off the naked. Once the quarterback is outside with the ball, somebody on that side has to cover the receivers coming across, and somebody has to stop the quarterback from running. If the defender on that edge runs with the flat route, the quarterback keeps it and runs, often for a first down, because the rest of the defense is still on the other side of the field chasing the fake. The keeper is a constraint on a constraint: it punishes the defense’s answer to the naked boot.

(3) The backside linebacker over-runs: the counter. The back on outside zone has three choices (bang, bend, bounce), and the cutback lane is always there for the backside linebacker to fill. If that linebacker starts running with the flow too, to get to the play side faster, the backside goes empty. The counter starts exactly like outside zone (the line’s first steps, the back’s first step) and then brings the ball back the other way behind a pulling lineman. It’s the run version of the bootleg: it punishes everyone who went with the first step.

(4) The safety races down: the play-action shot. A safety in the box is an extra run defender, and against an outside-zone team the safety on the play side learns to come down fast: a 6-yard run turns into a 2-yard run if he arrives in time. The play-action shot fakes the same run, keeps seven or eight players in to block, and sends two receivers deep, often in a yankee combination (a post from one side, a deep crosser from the other). A safety who has come forward for the run can’t get back to either.

Notice what the arrows in Figure 43.1 all have in common: they start at the defense. A constraint play is defined by the defender it punishes, not by the offense’s drawing. That gives you a practical way to watch: find the defender who is cheating, and you’ve found the play that’s coming.

NoteGo deeper: the RPO as a built-in constraint

The RPO takes this idea to its logical end. Instead of calling the constraint play on a later snap, after the defense has shown its lean, it attaches the constraint to the base play on the same snap: the line blocks the run, and the quarterback watches one defender (the conflict defender) and throws behind him if he leans toward the run. The constraint is no longer a separate call that has to be timed; it’s a branch of the base play that fires automatically. That’s why RPOs are sometimes described as “constraint theory in one play.” The cost is that the read has to be fast and the pass has to be short (the ineligible-downfield rule sees to that), so RPOs don’t replace the deep constraints, like the shot play, that punish the safety.

Same look, four plays: the signature plate

Constraint plays only work if, for the first second, they look like the base play. A linebacker reads a guard’s helmet and the running back’s first two steps; if those match what he has seen on every outside zone, he moves. This is why Shanahan-family offenses are obsessive about one formation, many plays: the same personnel, the same alignment, the same line steps and the same back’s track, with the decision about which play it really is hidden until the defense has already reacted.

In Figure 43.3 below, cover the bottom three panels with your hand and look only at the top-left one. That is what the defense sees for the first half-second of every play in the figure: the same alignment, the same line stepping right (all but one guard, as you’ll see), the same back heading toward the tight end. A defender who waits to find out which play it is gets beaten by outside zone, because outside zone is a race to the edge. A defender who reacts at once gets beaten by one of the other three. The offense has made waiting and reacting both wrong.

How long does a defender have? Not long. A linebacker’s read happens in his first step or two, well under a second, and by the time he is sure, outside zone has already reached the edge. That is why “wait and see” is not an option for him.

Look closely and the plays aren’t perfectly identical, which matters for the arms race. On the counter the right guard pulls left instead of stepping right, a clue for a linebacker disciplined enough to read the guard instead of the back. On the naked boot nobody blocks the backside end, and that is a louder clue than it sounds: on a real outside zone somebody, the backside tackle or a tight end, cuts or cuts off that end (the block the tight end showed and then abandoned on 04-05’s naked boot), so a well-coached end’s rule is “no cut, no block: think boot.” Defenses coach to those clues (“read the guard,” “if nobody blocks you, think boot”), and offenses answer by disguising the clue: the protected boot puts a blocker back on the end, and some counters pull a tight end instead of a guard. The game never ends; it just moves to smaller details.

Why not just run the constraint play more?

Here is the puzzle that comes with every constraint play. In Perimeter Runs and Misdirection you saw that an NFL receiver’s carry (jet sweeps, end-arounds, reverses) has been worth far more on average than a running back’s, every season from 2016 to 2025. In Play-Action, Bootlegs, and Screens you saw that play-action dropbacks beat ordinary dropbacks by a wide margin, season after season. If the constraint plays are that much better, aren’t coaches leaving points on the table by calling them so rarely?

Partly, maybe. But a simple model shows why a constraint play should look better on average than the base play even when a coach is calling it exactly the right amount. Imagine the offense chooses what share of its snaps to spend on the constraint play. As that share rises, the defense sees it more on film and respects it more: it cheats less against the base play. So the constraint play gets worse the more it’s called (fewer defenders are cheating for it to punish) and the base play gets better (fewer defenders are cheating against it).

The dashed line in Figure 43.4 is what the offense actually cares about: its expected points per snap, counting every snap. It peaks at about 13% here, and at that peak the constraint play is still worth clearly more per snap than the base play. Why stop there? Work through what one more percentage point of constraint calls would do, with the toy numbers at the peak:

  • The gain. One snap in a hundred switches from the base play (worth +0.06) to the constraint play (worth +0.21). That adds +0.0015 EPA per snap, averaged over all snaps.
  • The cost. The defense, seeing the constraint a little more often, respects it a little more. Every constraint call the offense was already making (13% of snaps) loses about 0.03: -0.0039 per snap.
  • The side benefit. A defense respecting the constraint more cheats less against the base play, so every base call gains about 0.003: +0.0026 per snap.

Add them up and the net is roughly zero. That is what “the peak” means: the value of one more call (economists say “at the margin”) has fallen to nothing, even though the average constraint call is still worth far more than the average base call. The constraint play’s average includes all the early, profitable calls; the next call only gets the margin.

So “the constraint play averages more than the base play” is not, by itself, evidence that coaches are under-using it; a well-run offense should show exactly that gap.2 What the gap can’t tell you is whether a particular team is at its peak or short of it, and economists who study play-calling have argued both ways. A second reason the gap flatters constraint plays is selection: coaches call them when they see the defense leaning, so their average is built from good moments. That’s not cheating the numbers. That’s the job.

Constraint theory says the plays must look alike. Why does this work so well? Because the burden falls unevenly. A defense doesn’t line up against a play; it lines up against a formation, with rules for who aligns where against two backs, against three receivers, against a wing, against motion. Every new look is something the defense has to install, practice and get right in a few seconds before the snap. The offense, meanwhile, practices one set of blocking rules. In Figure 43.5 the offense learned one play, and the defense had to learn four answers. Multiply by a game plan’s worth of plays and formations and the arithmetic gets lopsided fast: eight formations with three motions each is 24 pictures the defense must have an alignment for, and behind each picture could be any of a dozen core plays, so the defense prepares for close to three hundred combinations while the offense still practices a dozen plays.

The second payoff is that the outfits hide tells, the pre-snap clues you met in How to Watch a Broadcast. If an offense runs outside zone only from the offset I, the offset I is a tell. Spread the same play over four looks and every look carries several plays, so none of them gives the play away. This is a big part of why pre-snap motion has spread so fast: PFF charted a shift or motion on 63.9% of NFL plays in 2025, up from 38.4% in 2016.4 Motion is cheap dressing. It costs the offense almost nothing to learn and the defense a great deal to handle.

What it costs. The illusion has limits. A play still needs the right players to work: outside zone from an empty backfield is not outside zone. Every new outfit adds a little to the players’ memory load and to the chance of a busted alignment. And the outfits only stay an illusion if the plays inside them are mixed: a formation used for one play is just a tell with extra steps.

NoteFilm room: Los Angeles Rams 2018

Sean McVay’s 2018 Rams took the “one look” half of the illusion to an extreme. They used 11 personnel (one back, one tight end, three receivers) on about 93% of their plays, against a league median of about 65%, by this book’s count from nflverse data.5 A defense facing them could learn almost nothing from the substitution before each snap, because there was almost never a substitution: runs, play-action passes and dropbacks all came from the same eleven players, usually in a tight, condensed alignment, often with a receiver in motion. The constraint plays in 03-05’s film room (the jet sweeps that went for big gains) worked partly because they came out of the same look as everything else.

What to notice: when a team never changes personnel, stop looking at personnel and start looking at the details inside the look: the back’s depth, the receivers’ splits, which way the motion goes. That’s where a one-look offense hides, and where it sometimes leaks.

Sequencing: plays in an order

Constraint theory tells you which plays belong together. Sequencing is the order you call them in: show the defense a look, watch how it reacts, then call the play that beats that reaction. It turns a playbook into a conversation.

The idea is old. The Delaware Wing-T that you met in 03-05 was built as series football, and its coach, Harold “Tubby” Raymond, put the point bluntly: “The Wing-T is more than a formation. It is sequence football.”6 In the series the buck sweep comes first; the buck trap, the waggle and the counter are what the offense calls when the defense starts stopping the sweep. The mechanism is the defense’s key. Both guards pull on the buck sweep, so linebackers learn to read the guards and chase them; the rest of the series starts with the same backfield action and puts the ball where the chasing linebackers just left: inside on the trap, back the other way on the counter, and, on the waggle, a bootleg away from the sweep fake. To the linebacker reading the backfield, the constraint looks like the base play, which is the whole idea. Bill Walsh’s habit of scripting his first fifteen or so plays (the opening script that the next chapter takes apart) was partly a sequencing tool too: he planned in advance which looks to show early so he could see how the defense answered them and call the answers later.7

Sequencing comes in three shapes you can watch for.

1. Setting up the counter. Run the base play until the defense flows hard to it, then call the play that goes back against the flow. You don’t call the counter because the counter is good; you call it because the defense’s last three reactions say it will be.

2. Play-action off the base run. After the base run has gained yards from a look, call play-action from the same look within the next few snaps. This is the most common sequence in football, and the most watchable, because you can see it coming. Notice what 04-05 showed, though: linebackers bite on a good fake whether or not the run has been working. So coaches describe the sequence as being less about “earning” the fake with yards than about timing: a defense that has just been gashed is primed to come forward hard, its coaches have just told it to, and the offense knows that.

Is there any sign of that in the data? A little. On neutral first-and-10 snaps from 2022 to 2025, play-action passes called right after a run of 10 or more yards on the same drive averaged +0.21 EPA, against +0.11 for play-action at other times, and the gap showed up in 3 of the 4 seasons. Offenses did not call much more play-action at those moments (51% of dropbacks, against 47%), and the sample is small (581 snaps, so the gap is under two standard errors), so read it as consistent with the coaches’ timing story rather than as proof of it.8

3. The tendency breaker. The third shape needs two new words. A tendency is a pattern in what a team does in a given situation, formation or personnel group (running on four snaps in five from one look, throwing to the tight end on most third-and-shorts) that an opponent can find on film or in data and play against. A tendency breaker is a play called on purpose against your own tendency: a pass from the run look, a run from the empty backfield, a screen from the formation you always use for deep shots. It punishes an opponent who studied you, and it keeps the pattern honest for next week.

CautionCommon misconception

“A balanced offense alternates run and pass.” Alternating is a pattern too, and patterns are what defenses hunt. An economic study of NFL play-calling found that teams were more likely to switch between run and pass from one play to the next than a truly unpredictable mix would be, which makes the next call easier to guess, not harder.9 Balance means the defense can’t tell from the look or the last play what comes next, not that the offense takes turns.

Andy Reid’s offense is famous for the second half of the illusion of complexity, a huge menu of formations, shifts and motions dressing a stable core of concepts, and for sequencing that plays on what a defense did a few minutes earlier. The best-known example is the pair of fourth-quarter touchdowns in Super Bowl LVII against the Eagles, built on a return-motion play the team called “Corn Dog” (the full story is in Motion and Shifts): Kadarius Toney scored on it, and less than three minutes of game clock later Skyy Moore scored on a mirrored version to the other side. If the defense hasn’t fixed what beat it, the right next call is the same question again, and a staff that’s watching closely knows when a defense hasn’t adjusted. Tendencies and the self-scout

Every play an offense calls goes on film, and every opponent’s quality control coaches (the entry-level assistants from 01-04 who break film down into numbers) log it: down, distance, field position, personnel, formation, motion, run or pass, direction, result. From those logs come the tendencies a defense plays against on Sunday. A defense that knows an offense runs 80% of the time from a particular look can set its front, its linebackers’ keys and its safety’s depth to stop the run before the snap, and be right four times in five.

So offenses do the same thing to themselves. A self-scout is a team’s study of its own film and data, the way an opponent would study it, to find the tendencies it is giving away. Many teams do a formal one during the bye week, and the same questions come up in smaller ways all season.11 The questions are the opponent’s questions: What do we do from each personnel group and formation? Which way do we run when the back is offset left? What do we do after a motion? What do we call on second-and-long after a first-down run? Where are we a coin flip, and where are we a giveaway?

What a team does with the answers is a judgment call, and there are three choices:

  • Break it. Call a few tendency breakers from the giveaway look, so the next opponent’s numbers are muddier.
  • Hide it. Spread the giveaway play across more outfits (the illusion of complexity again) so no single look carries it.
  • Keep it as bait. Leave the tendency in place on purpose, let the opponent load up for it, and save the breaker for the moment it will hurt most: a big third down, the red zone, the playoffs.

Building a tendency table

You can run the opponent’s first question yourself. A tendency table cross-tabulates two things a defense can see before the snap (here, the offense’s personnel group and the quarterback’s alignment) and shows how often each combination produced a pass. Why the quarterback’s alignment and not the formation? Because it is the only formation information in the public data: nflverse records whether the quarterback was under center, in the shotgun or in the pistol, but not the receivers’ splits or the backfield set. A real tendency report charts far more: dozens of formation codes (2x2, 3x1, bunch, nub), the back’s offset, the motion and its direction, the hash and the field zone (“runs to the field,” “runs to the tight end,” “runs to the side the back is offset”). So whatever this table finds is a floor on what a pro staff gets from the look. To keep the situation from doing the talking (everyone throws on third-and-long), the table uses only neutral snaps: first and second down, the game reasonably close, outside the last two minutes of each half. The formal version of this filter, and expected-pass models that adjust for the situation play by play, are in Tendencies, PROE and Charting.12

The recipe, if you want to build one yourself (the code is under “Show the Python” on the website): load a season’s play-by-play with nflreadpy; join the participation file for personnel and the FTN charting file for the quarterback’s alignment, matching on game and play; keep the neutral first- and second-down snaps; then make a pivot table of the pass flag’s mean by personnel group and alignment, with a second pivot of counts so you can ignore cells with only a handful of snaps.

How much does the look give away?

A table shows the pattern; the question a defensive coordinator cares about is how much it’s worth. So run the scout’s experiment honestly. For every neutral snap from 2022 to 2025, predict run or pass using only that team’s other games that season (as a scout would, from film he has already seen): if the team passed on more than half of its snaps from this personnel-and-alignment look elsewhere, guess pass. Compare that with a dumber guess that ignores the look and uses only the team’s overall lean.

League-wide, the overall lean alone called about 53% of neutral snaps correctly, barely better than a coin. (For a team close to 50/50 overall, it can even do worse than a coin: if the team passed on 52% of snaps in its other games but ran more often in this one, every guess in this game is “pass” and most are wrong.) The look called about 67%. Two pre-snap facts, personnel and where the quarterback stands, turn a coin flip into two right guesses in three. Figure 43.8 shows how much that varied by team.

Two cautions keep this honest. Across the 128 team-seasons from 2022 to 2025, how well the look predicted an offense and how many points per play that offense produced were essentially unrelated (correlation -0.02); the fuller version of that question is in 13-04. NoteFilm room: Philadelphia Eagles 2024 — a tendency they could afford

The 2024 Eagles show the other side of the self-scout. In neutral situations they lined up in the pistol with 11 personnel 46 times and passed on 7% of them: about as loud a tell as an offense gives. Knowing “run” also didn’t tell a defense which run. The Eagles’ runs carried their own built-in answers: quarterback read keeps, RPO tags, and a mix of schemes and directions, so a defense that loaded up for “run” still had to guess where.

What to notice: a tendency costs you only if the defense can do something with it. Knowing the run is coming helps a defense that can win at the line; it doesn’t help much against a line and a back that win anyway. A self-scout doesn’t break every tendency. It asks which ones the next opponent can actually exploit, and spends its tendency breakers there.

The right panel shows why. Split the passes by whether they started with a run fake, and the difference between the looks disappears: a dropback without a fake was worth about the same from a running look as from a passing look, and a play-action pass did well from both. What made passes from run looks better is that 75% of them were play-action, against 21% from passing looks. In other words, the profitable tendency breaker in the NFL is not “surprise, it’s a pass.” It’s “surprise, it’s a pass that looks like the run you expected.” That’s constraint theory and the illusion of complexity again, showing up in the averages.

Treat this as what it is, an average over four seasons and every team, not a controlled experiment. Coaches choose when to break tendencies, and they pick good moments, so some of the gain is selection. But the asymmetry is striking, and it fits what coaches say: the run-look pass is the tendency breaker worth having.

The defense’s version: one look, several answers

Defenses play the same game from the other side. A defense has base calls too, the coverage and front it wants to be good at and calls most, and its own constraint plays: the blitz that punishes an offense for holding the ball (long play-action shots, slow-developing dropbacks, a protection that always slides the same way), and the drop that punishes a quarterback who expects a blitz and throws hot. It has tendencies the offense’s quality control coaches are hunting (does it blitz on third-and-medium? does the safety rotate toward the motion?), and it self-scouts and breaks them too.

The defensive form of the illusion of complexity is the one you met in Disguise, Rotation and Split-Field and Zone Blitzes and Simulated Pressure: show one picture before the snap and play several things after it. The offense has to decide its protection and its hot throws from the picture, so a picture that carries several defenses turns every one of those decisions into a guess.

Read Figure 43.10, below, as a constraint tree turned around. The defense’s base threat here is the Cover 0 blitz: six rushers, as many as the offense can block even with the back staying in, and nobody deep, so one missed assignment is a free rusher and the quarterback must get the ball out right now, often hot, to whoever the blitz left uncovered.

Count the threats in its top-left panel. Four linemen, two mugged linebackers and two walked-up defenders make eight possible rushers, and the offense has six blockers: five linemen and the back. It cannot block all eight, and it doesn’t try. That is the point of the look. The offense has to guess which six are coming, block those, and have a hot throw ready for anyone extra. An offense that has seen the Cover 0 version a few times learns to answer the pressure look the standard way: the quarterback identifies the Mike, the center and the back take the two mugged linebackers, and the quarterback is “hot” off any seventh rusher, throwing quick to the receiver that rusher left.

The fire zone and the simulated pressure are the defense’s constraint plays against that answer. They drop players into the windows where the hot throws go and bring rushers from where the protection isn’t looking, so the offense’s answer to the blitz becomes the defense’s opportunity.

The same structure holds for coverage shells. A two-high shell that can rotate to Cover 3 at the snap, or a single-high look that spins back to two deep, is one picture carrying several coverages, and it constrains the quarterback’s pre-snap read the way a run-and-boot look constrains a linebacker’s first step. The defenses that lean hardest on this, such as Brian Flores’s Vikings in 06-06’s film room, are the hardest to scout, because their tendencies are built to be coin flips.

The defense also has to self-scout. A defense that always rotates the safety toward the motion, or always blitzes on third-and-6, has handed the offense a free play: motion to set the rotation, or a screen against the blitz. The cure is the same as on offense: break it, hide it, or keep it as bait for one big moment.

TipWatch for it on Sunday: call the follow-up

Pick one offense for a quarter and keep a tiny log: the look (personnel and where the quarterback stands) and the play.

Before the snap

  • Is this a look you’ve already seen today? Say what it was used for last time.
  • If the last play from this look was a big run, say “play-action” out loud. It’s the most common follow-up in football, usually within the next few snaps.
  • Pick out the cheater: the linebacker flowing early, the backside end crashing down the line instead of squeezing, the safety creeping down. The next constraint play will be aimed at him.

After the snap

  • First step: did the line go the same way as the base play? If so, watch the ball, not the line.
  • Did the ball come back against the flow (counter), the quarterback roll away from it (boot), or the throw go over the safety who came down (shot)? Name the defender it punished.
  • On defense: when you see a pressure look on third down, count rushers at the snap. If a mugged linebacker drops, you’ve seen the defense’s constraint play against the hot throw.

A counter to the left (or another run back against the flow), not the naked boot. Two things have changed. The linebackers are flowing even harder, which is the counter’s cue: it starts like outside zone and brings the ball back to the side they just left. And the backside end has stopped crashing. He now plays boot, counter, reverse: he squeezes a yard, keeps his eyes on the quarterback and stays at the edge. That kills the naked boot, because he is standing on the spot the quarterback would roll to, facing him.

Is the end a problem for the counter, too? Partly: a disciplined boot-counter-reverse end is also a counter defender (it’s the C in his checklist). But a squeezing end is exactly the man the pulling guard is built to kick out, and the guard comes at him from inside, with a good angle. (If he squeezes so hard that he crosses the guard’s face, the guard logs him, sealing him inside, and the back bends outside instead.) The counter still wins here because the linebackers’ over-flow is the bigger cheat, and nothing else on the tree punishes it as directly. Expect the back to take one step right, then the right guard (or the tight end, in some versions) to pull back across, kick the end out, and the fullback to lead up onto the Will, who has to stop, turn and come back.

What about the protected bootleg, branch (1) on the tree? The linebackers are flying, and it puts a blocker on the end, so it’s a reasonable second answer, and partial credit if you said it. It is worse than the counter here because of where this end is: he is waiting at the quarterback’s launch point with his eyes on the quarterback, so the boot’s lone blocker meets a defender who expects him, the quarterback has to pull up early, and the crossing routes don’t get time to come open behind the linebackers.

Run or pass? If pass, what kind, and what is the defense afraid of?

TipAnswer

This is the snap for a play-action shot, and if it’s a pass it’s likely to go deep, though run is still the single most likely call. Start with the numbers: from this look the offense passes about three times in ten, so a pure tendency guess says run. That’s what the defense’s tendency report says too, and after a 23-yard run its coaches will have told it to stop the run, which is why the strong safety has come down to seven yards. That’s the lean.

Now think like the play-caller. The best time to break a tendency is when the defense is most committed to it. A play-action shot from the same look, with the same line steps and the same back’s track, attacks the one deep safety: one receiver deep down the middle to hold him, another on a deep crossing route behind the linebackers who come forward on the fake, seven or eight players blocking. Figure 43.9 is the case for it: passes from looks that usually mean run have been worth about twice as much as ordinary passes, almost entirely because they’re play-action.

A good defense knows this too, which is why many coordinators keep a safety deep for a snap or two after a big run even when the scouting report says run. That matters for the offense in two ways. If the quarterback has a check at the line and sees the defense back out of the eight-man look before the snap (the strong safety drifting back to two deep), he can check to the run, because the shot has lost its target. And if the defense disguises it and the free safety bails at the snap, the shot is covered on this play, but the offense has learned something for the next snap from this look: a defense that is guarding the deep middle has fewer men in the box, and the run is back on.

Six rushers again? What should the quarterback expect?

TipAnswer

Probably not six this time: expect a simulated pressure or a fire zone, with a defender dropping into H’s window. The defense’s base threat has been beaten twice in the same way: the offense now expects six, puts the center and the back on the two mugged linebackers and throws hot to the slot. That answer is exactly what the defense’s constraint plays are built to punish. The likeliest call drops one of the mugged linebackers (most likely the Will, on H’s side) straight into the short window the hot throw has been going to, and brings a rusher from somewhere the protection isn’t looking, such as the nickel off the edge (he’s walked up, so he can) or a lineman looping. Four or five rush; six or seven cover.

For the quarterback, the danger is throwing the same hot route on rhythm into a defender who is waiting for it. The better plan is to hold the ball a beat to see who actually comes, or to have a route behind the dropper. The defense can also bluff the other way and blitz six a third time, betting that the offense has adjusted already; that’s the coin flip the defense wants the quarterback to face. But after two clean hot completions, the sequencing logic points to the drop.

Takeaways

  • Base plays are what an offense wants to be good at; constraint plays punish the defense for stopping them. Every constraint play is defined by the defender it punishes: find the defender who is cheating and you’ve found the play that’s coming.
  • The two need each other. The base play keeps the defense leaning, the constraint plays limit the lean, and a constraint play’s average should look better than the base play’s even when it’s called the right amount, because each extra call costs a little on every other snap.
  • The illusion of complexity works from both ends. One play in many outfits gives the defense more to prepare; many plays from one look gives it nothing to read. Both hide tells.
  • Sequencing is the order of the conversation: the counter after the flow, play-action after the big run, the tendency breaker after a stretch of predictability. Sometimes the right follow-up is the same play again.
  • Tendencies are real and measurable. In neutral situations from 2022 to 2025, personnel plus the quarterback’s alignment called about two snaps in three correctly; the hardest offenses to read came mostly from the Shanahan family of offenses. Passes from run looks paid off, mostly because they were play-action.
  • Defenses play the same game: one pressure look or one shell carrying several calls, and their own self-scout.
