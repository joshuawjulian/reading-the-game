# --------------------------------------------------------------- players and alignments
def O(key, pos, d, w, label=None):
    return Player(key, pos, "off", d, w, label=key if label is None else label)


def D(key, pos, d, w, label):
    return Player(key, pos, "def", d, w, label=label)


def line():
    """The five offensive linemen, 1.6 yards apart; only the center is lettered."""
    keys = ["LT", "LG", "C", "RG", "RT"]
    return [O(k, k, -0.7, (i - 2) * SPLIT, "C" if k == "C" else "") for i, k in enumerate(keys)]


def spread(rb_w=-1.6, **moves):
    """11 personnel, shotgun 2x2: X and Z on the line outside (seven on the line with the five
    linemen), H (left slot) and Y (a tight end flexed into the right slot) a step off it."""
    off = line() + [O("QB", "QB", -5.0, 0.0, "Q"), O("RB", "RB", -5.0, rb_w, "R"),
                    O("X", "WR", -0.7, -15.0), O("H", "WR", -1.5, -9.0),
                    O("Y", "TE", -1.5, 9.0), O("Z", "WR", -0.7, 15.0)]
    return [p.moved(d=moves[p.key][0], w=moves[p.key][1]) if p.key in moves else p for p in off]


SHELLS = {
    # two-high: safeties 13 deep a few yards outside the hashes, corners 7 off with outside leverage
    "two": {"FS": (13.0, -8.0), "SS": (13.0, 8.0), "LCB": (7.0, -15.8), "RCB": (7.0, 15.8)},
    # Cover 2 look: squat corners 5 off, safeties a little wider
    "c2": {"FS": (13.0, -8.5), "SS": (13.0, 8.5), "LCB": (5.0, -16.3), "RCB": (5.0, 16.3)},
    # one-high: free safety in the middle, strong safety down over Y
    "one": {"FS": (13.5, 0.0), "SS": (7.0, 8.4), "LCB": (7.0, -15.8), "RCB": (7.0, 15.8), "NB": (5.0, -8.6)},
    # man-free: pressed corners, nickel tight on H, strong safety over Y, free safety deep
    "man": {"FS": (14.0, 0.0), "SS": (5.0, 8.6), "LCB": (1.2, -14.4), "RCB": (1.2, 14.4), "NB": (1.6, -8.4)},
}


def nickel(shell="two", **moves):
    """A 4-2-5 nickel against the 2x2: ends outside the tackles, a 1-technique and a 3-technique,
    Mike and Will at 4.5 yards, the nickel ($) over H; the secondary from SHELLS[shell].
    `moves` = {key: (d, w)} overrides any spot."""
    base = {"LDE": (1.0, -5.5), "LDT": (1.0, -0.7), "RDT": (1.0, 2.35), "RDE": (1.0, 5.5),
            "WILL": (4.5, -2.6), "MIKE": (4.5, 1.8), "NB": (5.0, -8.6)}
    base.update(SHELLS[shell])
    base.update(moves)
    lab = {"LDE": ("DE", "E"), "LDT": ("DT", "T"), "RDT": ("DT", "T"), "RDE": ("DE", "E"),
           "WILL": ("WILL", "W"), "MIKE": ("MIKE", "M"), "NB": ("NB", "$"), "LCB": ("CB", "CB"),
           "RCB": ("CB", "CB"), "FS": ("FS", "FS"), "SS": ("SS", "SS")}
    return [D(k, lab[k][0], *base[k], lab[k][1]) for k in lab]


def keep(players, lo, hi):
    """Only the players whose width is inside (lo, hi): half-field plates leave the rest out, because
    gridiron draws a marker even when it falls outside a zoomed window."""
    return [q for q in players if lo <= q.w <= hi]


# --------------------------------------------------------------- scripting movement
def timed(p, key, way, kind="path"):
    """Script a player: `way` = [(t, d, w), ...] absolute spots reached at time t (s after the snap).
    Standing still between two times is allowed. (gridiron's actions are speed-based; this turns
    a timetable into them.)"""
    cur = np.array(p._cur(key), dtype=float)
    tc, wait = 0.0, 0.0
    for t, d, w in way:
        nxt = np.array([d, w], dtype=float)
        dist = float(np.hypot(*(nxt - cur)))
        if dist < 1e-3:
            wait += t - tc
        else:
            p._add(kind, key, [tuple(cur), tuple(nxt)], speed=dist / (t - tc), delay=wait)
            wait = 0.0
            cur = nxt
        tc = t
    return p


def route_to(p, key, *pts, speed=None, delay=0.0):
    """A route through absolute (d, w) points."""
    return p.route(key, list(pts), absolute=True, speed=speed, delay=delay)


def protect(p, t=0.7):
    """Five linemen set, four rushers engage them: enough to show it is a pass, no more."""
    for k, w in zip(["LT", "LG", "C", "RG", "RT"], (-3.9, -1.7, 0.0, 1.7, 3.9)):
        if k in p.players:
            timed(p, k, [(t, -1.7, w)], "block")
    for k, (d, w) in {"LDE": (-1.4, -4.7), "LDT": (-0.6, -1.0), "RDT": (-0.6, 2.0), "RDE": (-1.4, 4.7)}.items():
        if k in p.players:
            timed(p, k, [(t, d, w)], "rush")


def presnap(p):
    """gridiron starts the clock at the first pre-snap movement: a tiny Mike shuffle gives the play
    pre-snap frames to draw, while nobody else moves."""
    m = p.players["MIKE"]
    p.motion("MIKE", to=(m.d, m.w - 0.2), speed=0.12)


# --------------------------------------------------------------- drawing helpers
def _P(fld, p, d, w):
    return fld.xy(*p.to_abs(d, w))


def no_numbers(ax):
    """Remove the painted yard-line numbers (they sit where receivers and safeties stand)."""
    for t in list(ax.texts):
        if t.get_alpha() == 0.55:
            t.remove()


def say(fld, p, d, w, text, color=style.INK_2, fontsize=7.0, ha="center", weight="normal", box=True):
    """A short on-field label at (d, w)."""
    x, y = _P(fld, p, d, w)
    fld.ax.text(float(x), float(y), text, fontsize=fontsize, color=color, ha=ha, va="center", zorder=12,
                fontweight=weight,
                bbox=dict(boxstyle="round,pad=0.18", fc=style.SURFACE, ec="none", alpha=0.92) if box else None)


def point_to(fld, p, text_at, target, text, color=style.INK_2, fontsize=6.8, shrink=7, ha="center"):
    """A label with a thin leader line to the player or spot it names."""
    (xt, yt), (xp, yp) = _P(fld, p, *text_at), _P(fld, p, *target)
    fld.ax.annotate(text, (float(xp), float(yp)), xytext=(float(xt), float(yt)), fontsize=fontsize,
                    color=color, ha=ha, va="center", zorder=12,
                    arrowprops=dict(arrowstyle="-", color=color, lw=0.7, shrinkA=2, shrinkB=shrink),
                    bbox=dict(boxstyle="round,pad=0.18", fc=style.SURFACE, ec="none", alpha=0.92))


def opening(fld, p, center, size, color=style.THIRD, alpha=0.22):
    """A green patch: the window the throw goes into. size = (depth, width) in yards."""
    cx, cy = _P(fld, p, *center)
    fld.ax.add_patch(Ellipse((float(cx), float(cy)), size[1], size[0], fc=color, alpha=alpha, ec=color,
                             lw=0.9, ls=(0, (2, 1.5)), zorder=2.2))


def num(fld, p, n, at, color=GREEN, fs=7.4):
    """A progression number in a small circle."""
    x, y = _P(fld, p, *at)
    fld.ax.text(float(x), float(y), str(n), fontsize=fs, color="white", ha="center", va="center", zorder=13,
                fontweight="bold", bbox=dict(boxstyle="circle,pad=0.18", fc=color, ec="none"))


def ruler(fld, p, fs=6.0, top=30):
    """Depth labels (yards past the line of scrimmage) down the left edge of the window."""
    W = 160 / 3
    x_left = W - fld.y1 + 0.35
    for d in range(5, top + 1, 5):
        y = p.los + d
        if fld.x0 + 0.8 <= y <= fld.x1 - 0.8:
            fld.ax.text(x_left, y, f"{d}" + (" yd" if d == 5 else ""), fontsize=fs, color=style.INK_2,
                        ha="left", va="center", zorder=12,
                        bbox=dict(boxstyle="round,pad=0.12", fc=style.SURFACE, ec="none", alpha=0.85))


def eyes(fld, p, frm, to, frac=0.62, color=GREEN, label="eyes", fs=5.6, label_off=(1.4, 0.0)):
    """A dotted arrow from the quarterback toward where he is looking (gridiron's read line runs
    player to player; this one can point at an area)."""
    a, b = np.array(frm, float), np.array(to, float)
    tip = a + (b - a) * frac
    (x0, y0), (x1, y1) = _P(fld, p, a[0] + 0.9, a[1]), _P(fld, p, *tip)
    fld.ax.annotate("", (float(x1), float(y1)), xytext=(float(x0), float(y0)), zorder=8,
                    arrowprops=dict(arrowstyle="-|>", color=color, lw=1.3, ls=(0, (1, 1.4))))
    if label:
        say(fld, p, tip[0] + label_off[0], tip[1] + label_off[1], label, color=color, fontsize=fs)


def ghost_marker(ax, x, y, label, ppy, color=style.DEFENSE):
    """A hollow, dashed defender: where he stood before the snap."""
    ax.add_patch(RegularPolygon((x, y), 4, radius=MARKER_R * 1.25, orientation=np.pi / 4,
                                facecolor="none", edgecolor=color, lw=1.1, ls=(0, (2, 1.2)), zorder=6))
    if label:
        ax.text(x, y, label, ha="center", va="center", color=color, fontweight="bold", zorder=6,
                fontsize=max(4.0, min(8.0, ppy * 0.7)))


def draw(p, ax=None, lateral=(-21, 21), window=(-8.0, 24.0), width_in=6.5, rule=True, legend=False):
    """A static plate: the Play's own diagram, without the painted numbers, with a depth ruler."""
    fld = p.draw(ax=ax, lateral=lateral, window=window, width_in=width_in, title=False, legend=legend)
    no_numbers(fld.ax)
    if rule:
        ruler(fld, p)
    return fld


# --------------------------------------------------------------- frame strips for print
def snapshot(play, t, ax=None, ghosts=(), ghost_t=0.0, lateral=(-21, 21), window=(-9.0, 21.0),
             width_in=6.5, notes=(), look=None, look_off=(1.4, 0.0), look_label="eyes", opens=(), trail_from=None):
    """Everyone where he is at time t, with faint trails, hollow markers where `ghosts` stood at
    `ghost_t`, an optional QB eye-line `look`=(d, w), green windows `opens` [(center, size)] and
    `notes` [(text, (d, w)) or (text, (d, w), (d, w) of the player it names)]."""
    pos = {k: play.position(k, t) for k in play.players}
    moved = [pl.moved(d=float(pos[k][0]), w=float(pos[k][1])) for k, pl in play.players.items()]
    q = Play([m for m in moved if m.side == "off"], [m for m in moved if m.side == "def"],
             los=play.los, ball_y=play.ball_y, to_go=play.to_go)
    q.highlights = dict(play.highlights)
    fld = q.draw(ax=ax, lateral=lateral, window=window, width_in=width_in, title=False)
    ax, ppy = fld.ax, fld.pts_per_yard()
    no_numbers(ax)
    P = lambda d, w: fld.xy(*play.to_abs(d, w))  # noqa: E731
    for c, s in opens:
        opening(fld, play, c, s)
    start = trail_from if trail_from is not None else play._compile()[1]
    for k, pl in play.players.items():
        if pl.side == "off" and pl.pos in ("LT", "LG", "C", "RG", "RT"):
            continue
        pts = np.array([play.position(k, tt) for tt in np.linspace(start, t, 30)])
        if np.ptp(pts[:, 0]) + np.ptp(pts[:, 1]) < 0.3:
            continue
        xs, ys = P(pts[:, 0], pts[:, 1])
        if pl.side == "off":
            ax.plot(xs, ys, color=style.OFFENSE, lw=0.9, alpha=0.45, zorder=3)
        else:
            ax.plot(xs, ys, color=style.DEFENSE, lw=1.1, ls=(0, (3, 2)), zorder=3)
    if t > ghost_t + 0.05:
        for k in ghosts:
            x0, y0 = P(*play.position(k, ghost_t))
            ghost_marker(ax, float(x0), float(y0), play.players[k].label, ppy)
    if look is not None:
        eyes(fld, play, play.position("QB", t), look, label=look_label, label_off=look_off)
    bx, by = P(*play.ball_position(t))
    thrown = False
    for kind, _, t_rel in play.ball_events:
        if kind == "pass" and t > t_rel:
            thrown = True
            sx, sy = P(*play.ball_position(t_rel))
            ax.add_patch(FancyArrowPatch((float(sx), float(sy)), (float(bx), float(by)), arrowstyle="-|>",
                                         mutation_scale=8, color=style.BALL, lw=1.3, ls=(0, (2, 1.5)), zorder=9))
    if thrown or t <= 0:          # in the quarterback's hands the ball would hide his label
        draw_marker(ax, float(bx), float(by), "ball", "", ppy)
    for note in notes:
        text, (d, w) = note[0], note[1]
        x, y = P(d, w)
        box = dict(boxstyle="round,pad=0.2", fc=style.SURFACE, ec=style.BASELINE, lw=0.5)
        if len(note) > 2:
            xp, yp = P(*note[2])
            ax.annotate(text, (float(xp), float(yp)), xytext=(float(x), float(y)), fontsize=5.8, color=style.INK,
                        ha="center", va="center", zorder=13, bbox=box,
                        arrowprops=dict(arrowstyle="-", color=style.INK_2, lw=0.7, shrinkA=1, shrinkB=5))
        else:
            ax.text(float(x), float(y), text, fontsize=5.8, color=style.INK, ha="center", va="center", zorder=13,
                    bbox=box)
    return fld


STRIP_KEY = [Patch(facecolor="none", edgecolor=style.DEFENSE, lw=1.1, ls=(0, (2, 1.2)),
                   label="Dashed outline: where he stood before the snap"),
             Line2D([], [], color=GREEN, lw=1.3, ls=(0, (1, 1.4)), label="The quarterback's eyes"),
             Line2D([], [], color=style.DEFENSE, lw=1.1, ls=(0, (3, 2)), label="Defender's path so far"),
             Line2D([], [], color=style.OFFENSE, lw=1.0, alpha=0.6, label="Offensive player's path so far"),
             Line2D([], [], color=style.BALL, lw=1.3, ls=(0, (2, 1.5)), label="The pass"),
             Patch(facecolor=style.THIRD, alpha=0.3, edgecolor=style.THIRD, label="The window the ball goes to")]


def strip(play, times, labels, per_panel=None, ncols=2, width_in=6.5, lateral=(-21, 21), window=(-9.0, 21.0),
          legend=STRIP_KEY, **kw):
    """The print version of an animation: panels at chosen moments of one simulated play.
    `per_panel` = {i: dict(notes=..., look=..., opens=...)}."""
    n = len(times)
    nrows = int(np.ceil(n / ncols))
    pw = width_in / ncols
    h = nrows * (pw * (window[1] - window[0]) / abs(lateral[1] - lateral[0]) + 0.34)
    fig, axes = plt.subplots(nrows, ncols, figsize=(width_in, h), squeeze=False)
    for ax in axes.flat[n:]:
        ax.axis("off")
    for i, (ax, t, lab) in enumerate(zip(axes.flat, times, labels)):
        extra = (per_panel or {}).get(i, {})
        snapshot(play, t, ax=ax, lateral=lateral, window=window, **kw, **extra)
        ax.set_title(lab, fontsize=8.0, color=style.INK, loc="left")
    fig.tight_layout(h_pad=0.6, w_pad=0.4)
    if legend:
        fig.legend(handles=legend, loc="upper center", bbox_to_anchor=(0.5, 0.0), ncol=2, fontsize=7.2,
                   frameon=False)
    return fig


def animation_or_strip(play, times, labels, web_times=None, **kw):
    """A video on the website; the numbered frame strip in print."""
    if PDF:
        strip(play, times, labels, **kw)
        plt.show()
    else:
        show_animation(play, times=web_times or times)

The question under every pass

Every pass play starts with a question the quarterback has to answer faster than any other decision in sport: what coverage is this? Not because coaches love labels, but because the coverage decides where the empty grass will be. Cover 2 leaves the sideline between its squatting corner and its deep safety; Cover 3 leaves the flats and the seams; quarters leaves the space behind its read safeties’ eyes; man leaves whoever loses his one-on-one. Know the coverage and you know, roughly, where the ball should go before the receivers have finished their first steps.

Quarterback Play showed the quarterback’s clock: a drop timed to the depth of the routes, a progression run one look at a time, a key defender answering each look. It left one thing out on purpose, because Part 6 had not yet taught the coverages: how he decides which defender is the key, and which side to look at first. That is the coverage read. Coaches teach it in many dialects, but nearly all of them reduce to four steps, in order.

  1. Shell, before the snap. First, though, the protection: how many defenders could rush, and who is his hot receiver if one more comes than he can block (Pass Protection). Then the coverage: one deep safety or two? Corners pressed or off? Is anyone creeping? Which side is the field side (the wide side of the field, away from the hash the ball sits on), since many defenses set their rotation and their coverage by it? This produces a hypothesis: “two-high, corners squatting, probably Cover 2.”
  2. Rotation, at the snap. Did the safeties stay where they were, or did one spin to the middle and the other come down? Is the middle of the field open or closed? Did anyone extra rush? This confirms or overturns the hypothesis.
  3. Key defender, after the snap. Given the coverage, which single defender decides whether the called concept works? For smash against Cover 2 it is the cornerback; for flood the curl-flat defender; for dagger the safety and then the linebacker; for four verticals the deep middle safety.
  4. Throw. Read that defender and throw where he isn’t. If the answer is “nowhere,” move to the next read, and eventually to the checkdown.

Notice what the quarterback did not do. He did not decide where to throw before the snap and then throw there; that is how the rotation in Disguise, Rotation, and Split-Field Coverage produced its interception. He also did not wait to see a receiver come open, which would have been too late: Z’s hitch is only open for the half-second before the strong safety arrives. He let one defender make the decision for him. Against Cover 2 that would have been the corner; once the coverage rotated to Cover 3, it became the strong safety rolling down to the flat. Either way the concept was built so that whatever that defender did, one of the two routes would be on the wrong side of him.

That is also why smash survived the rotation. Smash is built first of all to beat Cover 2, but its hitch sits under a Cover 3 corner who must bail and outside a flat defender who has to come from the middle of the field to reach it. The quarterback didn’t need to change the play. He needed only to know which half of it the coverage had handed him, and which defender now held the answer. Step 1: the shell, and what it is worth

The pre-snap read is worth more than its reputation suggests, even in an era of disguise. First, it narrows the menu. A defense showing two deep safeties and two off corners is very unlikely to be in Cover 0 one second later; a defense showing one safety in the middle and pressed corners is unlikely to end up in quarters. Each shell makes some coverages easy to reach and others a long run away, and as Disguise, Rotation, and Split-Field Coverage showed, every defender has to stand where his real job is still reachable. Second, the shell sets up everything that has to be decided before the ball moves: the protection, which side the progression starts on, whether to take a shot (the alert, later in this chapter) and whether to change the play. How much the quarterback may change varies by team. In most NFL offenses the play-caller chooses the concept, and the quarterback’s freedom at the line is to flip it to the other side, to pick between two plays called together, or, in a check-with-me system, to choose from a short menu the coaches built for that week.

What the shell can’t do is name the coverage. The quarterback treats it as a short list ranked by likelihood: “two-high: Cover 2, quarters or 2-Man, unless they rotate.” The rest of the read decides between them.

Step 2: rotation, and the one-second answer

The single most useful question after the snap is the one you can ask from your couch: one second after the snap, how many safeties are deep in the middle of the field? One, and the middle is closed: Cover 1 or Cover 3 (or a fire zone, which is Cover 3 behind a blitz). Two, splitting the field, and it is open: Cover 2, Tampa 2, 2-Man, quarters or Cover 6. None: Cover 0, and the ball should already be on its way. Then the corners separate the members of each family: in the closed family, corners who turn and run with their receivers mean Cover 1 and corners who bail with their eyes on the quarterback mean Cover 3; in the open family, corners who squat in the flats mean Cover 2 (or the Cover 2 side of Cover 6), corners who backpedal and read mean quarters, and corners who trail their man mean 2-Man.

The quarterback answers that question with his first glance after the snap, often while he is still dropping. He does not need to see every defender. He needs the two safeties, because they are the ones a disguise moves, and the corner on the side he is reading. He can see them over linemen who are taller than he is because the safeties stand 10 to 15 yards deep, far above the line of helmets, and because from the shotgun he starts 5 yards behind the line with his eyes already up the field, instead of turning his back on the defense to drop from under center.

Against split-field coverage, Cover 6 for example (quarters on one half, Cover 2 on the other), “the” coverage doesn’t exist. The quarterback reads each half as its own coverage and works the half his concept is on: smash to the Cover 2 side, dagger or a quick out by #2 to the quarters side. Offenses that expect split-field coverage often call a different concept to each side and let the quarterback pick a half from the safeties’ alignment before the snap, which is the “man side, zone side” idea later in this chapter applied to two zones.

Step 3: the key defender for each coverage

Once the coverage has a name, the key defender follows from the concept and the coverage together. The table pairs each coverage with the defender quarterbacks are usually taught to read against it and where the ball tends to go. It is a summary of Part 6 from the offense’s side.

The key is almost always one of three kinds of defender: the conflict defender the concept was built against (the corner in smash, the curl-flat defender in flood), the safety who reads the quarterback (the post safety against four verticals, the robber against in-cuts), or, against man, the matchup itself, the receiver’s defender, read by his leverage and the help behind him.

Moving a defender with your eyes: the look-off

Every defender in the second category has a weakness built into his job. A post safety, a robber, a Tampa 2 Mike and most zone linebackers are taught to watch the quarterback, because the quarterback’s eyes and shoulders tell them where the ball is going before it leaves his hand. That makes them movable.

A look-off is the quarterback deliberately looking at one receiver or area, often turning his shoulders that way too, to pull a defender who reads his eyes toward it, and then throwing somewhere else, into the space that defender left. It is the quarterback’s half of the disguise contest: the defense lies with its alignment before the snap; the quarterback lies with his head after it.

Why does the robber bite? Because his job is to be right about in-breaking throws, and the best evidence he has is the quarterback. A robber who ignores the quarterback’s eyes and simply sits in the middle is easy to throw around; one who reads them gets interceptions. The look-off turns his best evidence into bad evidence. The same logic works on the deep middle safety against four verticals (Cover 3 drew it: the quarterback looks at one seam, the safety leans, the ball goes to the other) and on a Tampa 2 Mike deciding which seam to carry. The play in this chapter’s opening is the two-high version. A read safety in quarters or Cover 2 also helps toward wherever the quarterback is looking, so eyes on the left seam pull the right-hand safety a step inside, and the throw goes to the right, into the space he left. The robber is drawn here because with only one eye-reader on the field the mechanism is easiest to see.

What a look-off costs. Time, mostly. Holding a defender with your eyes takes a beat, perhaps three to five tenths of a second, and every beat is one more for the pass rush. So the look-off is usually built into the play’s timing rather than added to it: the first read of a progression doubles as the look-off for the second, as in the figure, where the quarterback had a real reason to look at Z’s dig before coming back to H. It also takes a quarterback who can reset his feet and throw accurately to the other side of the field without a wasted step. A look-off by a quarterback whose feet don’t follow his eyes is just a late throw.

The defense’s answer. Coaches teach the eye-readers to “read through” the quarterback: to take their cue from his shoulders and the way his feet are pointing rather than his helmet, because a quarterback can turn his head cheaply but has to set his feet to throw. Defenses also mix in defenders who don’t read the quarterback, such as a robber who reads a receiver instead, so a look-off aimed at the wrong defender moves nobody. And they disguise who the eye-reader is. A quarterback who looks off a safety who wasn’t reading him has wasted his beat.

NoteMadden vs. real life

In a video game, defenders mostly react to the ball and to their assignments, and where your quarterback’s helmet points is rarely part of the contest. On a real field it is one of the most important things he does. NFL coaches grade a quarterback’s eyes on film the way they grade his footwork, and the most-praised quarterbacks are the ones whose eyes move defenders. A quarterback who stares at his first read is said to be “staring down” the receiver, and safeties feast on it.

Watching a look-off from the couch

The look-off is one of the few post-snap skills you can see on a broadcast without the All-22 film. Watch the quarterback’s helmet, not the ball. If his head goes to one side and the ball comes out to the other, ask who moved while his head was turned. On the replay from behind the offense, find the safety or linebacker who took a step toward where the quarterback looked. That step is what the look-off bought.

What beats what: the matrix

Now the question the chapter’s title promises. A coverage beater is a pass concept whose routes are arranged to attack the known weak spots of a particular coverage, so that if the defense plays that coverage, at least one receiver should come open. Smash against Cover 2 is the textbook case. The phrase is used loosely on broadcasts, often with a confidence the football doesn’t support, so here is a tighter version: a beater raises the odds against its coverage. It doesn’t guarantee anything, and it is usually worse against other coverages. That trade is the subject of the next section; first, the map.

The reasoning is the one Cover 2 and Tampa 2 built move by move: the corner’s rule (squat on anything short in the flat) and the safety’s rule (stay deeper than anything in your half) are both sound, and smash places one route on each side of the line between them. The corner can’t be under the hitch and over the corner route at once, and the safety, starting near the hash, is a long way from the sideline. Flood earns a “+” here for the same reason, one step weaker: its sail route lands in the same hole, but its third route, in the flat, is exactly what the squatting corner is waiting for, so the defense covers one of flood’s three levels for free.

Four verticals attacks the other thin spot. With both seams running straight up the middle third of the field, each deep-half safety has a seam inside him and a go route outside him, and he can’t honour both; a quarterback who reads one safety throws to whichever receiver that safety doesn’t take. Tampa 2 exists to fix exactly this (the Mike runs down the middle and carries one seam), which is why four verticals is only a “+” against Cover 2 in the matrix: the Tampa version takes away one of the throws. NFL versions also build a choice into each seam (a route conversion): against two deep safeties the seam runners bend away from the safeties or split them, and against one they stay straight, which is part of why the concept survives a disguised coverage. The hole shot, a go route caught along the sideline behind the squatting corner, is the outside version of the same attack.

Against Cover 2 the offense gives up two things: the quick game to the outside, because the corners are sitting on it, and the screen, because the corner is also the force player and is standing right where a screen to the outside wants to go. Hence the grey and orange in those cells.

Against Cover 3: three under the deep ones, four across the front

Cover 3 is the opposite shape: three deep and four underneath. Its weaknesses, taught in Cover 3, are horizontal and vertical at once. Four underneath defenders can’t cover five short zones across the field, so the flats are thin; three deep defenders can’t cover four vertical routes, so the seams are thin. Flood attacks the first, four verticals the second.

Flood is the purest Cover 3 beater because it uses Cover 3’s own rules against it. The corner’s rule (never let anyone behind you) means the go route takes him out of the play without a fight. The curl-flat defender’s rule (cover the curl, break to the flat) gives one man two receivers. The only help could come from the hook defender inside him or the post safety, and both are watching other routes. Four verticals works on the same arithmetic one level deeper: four vertical routes, three deep defenders, and a post safety who can be moved with the quarterback’s eyes. Y-cross and yankee attack the same single deep safety from across the field. In Y-cross the tight end runs a deep crosser behind the hook linebackers, who have dropped to 8 or 10 yards facing the quarterback and can’t see him go by, while a post or a go route holds the safety in the middle. The crosser comes out on the far side, 15 to 18 yards deep, in the space between the curl-flat defender (too shallow) and the corner (too deep, and occupied by the outside receiver). Every Cover 3 defender is where his rule says, and none of them has the crosser. Yankee does the same thing off play-action, with a post and a deep over route and a run fake that pulls the linebackers even further forward.

What Cover 3 does well doesn’t show in the grid at all, which is why its column has no orange. The grid grades pass concepts, and Cover 3’s strengths lie elsewhere: an eighth defender near the line against the run, and three deep defenders who never let anyone behind them, so it gives up very few explosive passes. What it concedes is the short and intermediate throw. In the charting data later in this chapter, every route type succeeds slightly more often than its own average against Cover 3. The defense’s bet is that a steady diet of 6-yard completions costs less than a light box or a busted deep ball.

Against quarters: make the safety’s eyes cost him

Quarters is built to stop most of what beats Cover 2 and Cover 3. It has four defenders who can play deep, so four verticals runs into four men; its safeties read the inside receiver and come down hard on anything short, so slants and short in-cuts thrown inside are dangerous. (The quick game still grades “+” against it because of the quick out by #2, thrown outside, away from those safeties, before the defender over the slot can widen.) Its weakness is the reverse of its strength. A read safety’s job is decided by one receiver, #2, and a route combination that tells him the wrong thing takes him out of the play.

The free safety did exactly what quarters asks: #2 went vertical, so he carried #2. The Will did what he was taught: wall the seam, then sink to his landmark. The corner played the outside receiver from outside, which is what lets quarters take away the deep outside. Every rule was obeyed, and together they left the middle of the field empty at 16 yards for a receiver breaking in from the outside. That is the general shape of attacking quarters: make #2 tell the safety something, then throw where the safety’s answer took him from. It is also why the backside matters: a defense that expects dagger can rob the dig with the other safety, if his own #2 stays short and frees him to look inside.

Scissors makes the safety’s answer cost him in a different way. Here #1 runs a post and #2 a corner route, so their paths cross at 10 to 12 yards. Each defender started with leverage, on the side his help isn’t: the safety inside #2, the corner outside #1. Once #2 goes vertical the safety must carry him, and when #2 breaks for the sideline the safety has to chase him outward, across the path of the post. The corner, playing #1 from outside, has to cross inward to follow the post. Both defenders are now chasing receivers who broke away from their leverage, and they have to run through each other to do it, a rub at depth. Well-coached quarters teams answer by trading the routes as they cross (the corner takes the corner route, the safety the post), but that takes a word and a split-second decision at full speed, and the hesitation is the window. The quick out by #2 attacks quarters underneath, in the first second, before the apex defender can widen (Quarters and the Two-High Match Family); the running back’s wheel and the bootleg flood do it at the edge.

The concepts quarters likes follow from the same rule. Levels and drive send two receivers from one side across the middle at two depths, the deeper one a dig at about 12 yards, and in both #2 breaks inside instead of going up the seam. In levels #2 runs the dig himself, straight at the read safety who was watching him all along. In drive #2 runs the shallow cross, which by rule frees the safety to stop backing up and drive on the next route coming inside: #1’s dig. Either way the deep in-cut meets a safety coming downhill with his eyes on it. That is the “−” in the quarters column. The same concepts grade “+” against Cover 0, Cover 1 and Cover 2 because their crossers and in-cuts run through traffic against man and settle between spot-dropping linebackers against Cover 2.

What quarters gives the offense for free, if it chooses to take it, is not in the matrix at all: runs. With two safeties at 10 to 12 yards and only six or seven defenders near the line, the box is light, and Run Fits and Box Math showed why a light box invites the offense to hand the ball off. Some of the best answers to two-high coverage aren’t passes.

Against man: run away, run through traffic

Man coverage has no zones to stretch, so the stretches of the zone concepts don’t apply. Smash against Cover 1 is just two one-on-ones, and both favour the defense. The slot defender on Y plays with outside leverage, because his help (the free safety) is inside, so the corner route breaks straight into the side he is protecting. And the corner on Z, with nobody running deep behind him to worry about, can sit on the hitch. Flood against man is three one-on-ones near a sideline, which crowds the space the receivers need. Man coverage loses in three different ways: traffic (receivers whose paths cross, so a defender chasing one runs into the other: mesh and every rub), leverage (routes that break away from the side the defender is protecting: the out against inside leverage, the slant against outside leverage), and talent (a receiver who simply beats his man). Mesh, the purest man-beater, gets the double “+” in the Cover 0 and Cover 1 columns; the figure in the two-way section below shows it.

2-Man is the exception that explains the rule. With two deep safeties over the top, the underneath defenders can trail their receivers with inside leverage and give up nothing deep, so the vertical concepts (four verticals, smash’s corner route, yankee) all run into a defender underneath and a safety above. It is the most orange column of the matrix for that reason: almost every concept built to stretch a zone is worse against it. Its weaknesses are short and outside, away from the trailing defenders: outs, stops and the flat route break away from their inside leverage, while slants break straight into it, which is why the quick game, a mix of both, grades only even. The other weakness is the quarterback’s legs: with every defender’s back turned to him, a scramble often gains more than any throw.

Against pressure: get it out, or throw it where they left

The last column is the one that answers Zone Blitzes, Fire Zones, and Simulated Pressure. Against any blitz the first answer is time: the deep concepts (dagger, scissors, yankee) need three seconds or more, so dagger and scissors grade orange against both Cover 0 and fire zones. Yankee grades even against Cover 0 for one reason: it is a play-action call with seven or eight blockers, so it can block an all-out blitz more often than a five-man dropback protection can, and a post and a deep over route against man with no safety behind them are a touchdown if the ball gets out. If the protection breaks, the quarterback, with his back half-turned on the fake, is hit before he can throw: grey means “depends.” The second answer depends on what is behind the blitz.

  • Cover 0 has nobody deep and everybody in man. The answer is the quick man-beater (mesh, slants, a rub) or the hot throw to the receiver whose defender blitzed, thrown before the extra rusher arrives. A one-on-one go route with no safety behind it is a touchdown if the ball can get out, which is why the cell for four verticals is grey rather than orange.
  • A fire zone has three deep and three underneath. It is built to take away the hot throw: the dropper lands in the window a hot slant or a quick in-cut wants. So the answers are the throws it can’t cover with three underneath: the seams, which the seam-flat defenders have to carry alone (four verticals and flood grade “+”), the window behind the dropper (a lineman can’t get much deeper than 5 or 6 yards in time), and screens, thrown behind rushers who have committed to the quarterback. Why does four verticals get a “+” here while dagger, which also sends a seam, gets a “−”? Timing. Four verticals throws the seam on rhythm, at the top of the drop, before the fifth rusher arrives, and it reads one defender who is alone. Dagger’s dig breaks at 15 or 16 yards behind the seam and needs a beat longer, and the middle dropper and the hook defender are sitting exactly where the dig crosses.

If the grid is this one-sided, why play Cover 3 or 2-Man?

Read the matrix literally and two coverages look like mistakes. Cover 3 is beaten by flood, four verticals, Y-cross and yankee and has no orange cell at all. 2-Man is orange almost everywhere, so why doesn’t every defense play it on every down? Yet in the charting data Cover 3 is the most common coverage in the league, on 32% of charted dropbacks from 2018 to 2022, and 2-Man is one of the rarest, on 2%.1

The grid grades pass concepts against coverages and nothing else, and each coverage pays its way somewhere the grid can’t see.

  • Cover 3 is cheap and sound. With one safety deep, the other can stand near the line as an eighth defender against the run, and three deep defenders who never let anyone behind them mean few explosive passes. It concedes completions underneath and dares the offense to string them together without a mistake. It is simple, so it is rarely blown, and it is the base that Cover 3 match, the fire zones and the rotations of Part 6 are built on.
  • 2-Man is expensive. It needs underneath defenders who can trail receivers all over the field, it leaves only five or six defenders near the line against the run, and with every back turned to the quarterback it invites scrambles. So it is a passing-down call. On third-and-7 or longer in those seasons it was 6% of charted dropbacks, still a small share. Played on every down, it would be run on and scrambled against; played rarely, it ambushes the vertical concepts offenses like to call on third-and-long.

A coverage’s real value is its whole column, plus the run game, plus how often it gets blown, weighted by the situations in which the defense chooses to play it. That is the defense’s half of the bet.

A short history: why the matrix keeps changing

The idea of choosing a pass play to beat a particular coverage is as old as coverages that could be studied. Sid Gillman, who coached the Los Angeles Rams in the late 1950s and the AFL’s Chargers in the 1960s, is usually credited with two of its foundations: studying opponents on film, cut up and spliced by play, and designing pass patterns to stretch a defense across the whole field, vertically and horizontally, so that zone defenders faced more receivers than they could cover. Bill Walsh, who said that much of what he did derived from Gillman, built those ideas into the West Coast offense’s progressions.2 The rules then tilted the field. In 1978 the NFL allowed defenders to contact a receiver only within the first five yards past the line of scrimmage, which made it far harder for defenses to disrupt routes and far more valuable for offenses to read coverage and throw on time (1978 and the Passing Revolution tells that story).3

The defense’s side of the history is the story of Part 6 told backwards. Four verticals against spot-drop Cover 3, a staple of the run-and-shoot and BYU passing games by the 1980s, helped produce pattern-matching Cover 3, usually traced to Nick Saban and Bill Belichick in Cleveland in the early 1990s (Cover 3).4 Smash and the quick out by #2 helped produce Palms, the quarters check that traps them. And the spread offenses of the 2000s and 2010s, which could create a one-on-one or a numbers advantage against any single coverage, helped push NFL defenses toward the disguised two-high shells that spread after 2018 (Fangio’s Two-High Revolution and Macdonald’s Answer). The matrix in this chapter is a snapshot of the mid-2020s. Its rows and columns will look different in ten years, because both sides are always working on the cells that hurt them.

Why “beaters” are bets

If offenses know which concept beats which coverage, why doesn’t every pass work? Because a beater only beats the coverage the defense actually plays, as it actually plays it, and four things stand between the call sheet and that coverage.

1. The defense chose its coverage too, and it hid it. The play-caller picks a concept before the snap, often before the huddle breaks, from a guess about the coverage. The defensive coordinator picks the coverage at the same time, from a guess about the concept, and then disguises it. Neither sees the other’s card. A smash call that meets Cover 2 looks brilliant on the replay; the same call against 2-Man looks like a mistake, and the play-caller is the same man with the same information. The rotation at the start of this chapter is the everyday version: the offense called a Cover 2 beater and got Cover 3. Keep that in mind when a broadcast calls a completion “a great call against that coverage.” The analyst is grading the call with the answer key in hand; the honest version is “that was a good bet, and it paid.”

2. Coverage labels hide different rules. “Cover 3” names a family, not a set of instructions. Against spot-drop Cover 3, four verticals is a math problem the defense loses; against Cover 3 match, whose seam-flat defenders carry the slot receivers, it meets four defenders deep and the math is gone. The label in the matrix is the same; the behaviour after the snap is not. Quarters, Cover 6 and Palms look identical before the snap and play completely differently against the same concept. Every match coverage is a set of if-then rules (Quarters and the Two-High Match Family), and the “beater” depends on which rules this staff teaches.

3. Execution and talent. A beater produces a window; somebody still has to hit it. The window in the smash plate is about five yards wide and opens for perhaps half a second. A protection that breaks at 2.2 seconds, a corner route run a yard too flat, a ball thrown a beat late, a defender who is simply faster than the diagram assumed: each closes it. And the reverse happens constantly. Great receivers beat good coverage with bad matchups, and great quarterbacks complete throws into coverages the play was not designed for. That is why the best players change the matrix, which describes average players.

4. Situation and the second read. The defense doesn’t play coverages at random; it plays them by down, distance, field position, personnel and score (Tendencies, PROE, and Charting). On third-and-long, a defense can sit back in deep coverage and concede the short throw, so a concept can “beat” the coverage and still lose the down if its open receiver is two yards short of the sticks. Beating the coverage and moving the chains are different things.

Read it the way an analyst would. Across 63,058 targeted throws, the average success rate is 52%. The best cell is 63% and the worst 28%, but most of that spread is between routes, not between coverages: go routes are orange in every column because deep throws fail more often whatever the coverage, and hitches are blue in most. Along the rows, some of the matrix’s logic shows clearly. A hitch or curl thrown against Cover 2 succeeds 63% of the time and against 2-Man only 38%: against Cover 2 the receiver settles in a window between underneath defenders who dropped to landmarks, while against 2-Man a defender is trailing on his hip with a safety over the top. Go routes do best against Cover 2 (43%), the hole shot behind a squatting corner, and worst against quarters (31%), which has a deep defender for every vertical route (only Cover 0, at 28%, is lower). Corner routes are a poor bet against Cover 1 (40%).5

Other cells seem to contradict the matrix, and each one teaches something about reading data like this.

  • The smash corner against Cover 2 succeeds only 49% of the time, below the chart’s average. But a corner route is a 20-yard throw toward the sideline, caught about half the time against most coverages, and success rate counts every incompletion as a failure. The corner route’s own average is 45%; against that baseline, Cover 2 is one of its better columns.
  • Slants and digs against Cover 3 succeed more often than average (61% and 58%), although four underneath defenders should make inside throws risky. Every route does a little better than its own average against Cover 3, the coverage that concedes underneath to protect against the deep ball, and a quarterback throws the slant against Cover 3 only after he has seen the hook defender leave its window. The cell records the result of a read, not of a call.
  • Crossers against man (Cover 1 53%, Cover 0 50%) succeed at about the crosser’s usual rate, with no sign of mesh’s double plus. The label lumps shallow crossers together with deep ones, and success rate calls a 6-yard catch on third-and-8 a failure however open the receiver was. It is also possible that the man-beater’s edge is simply smaller than coaching lore says, which is this section’s point.

Two cautions apply to every cell. First, the chart shows only the route thrown to, and the quarterback chooses that route after seeing the coverage, so every cell is already the result of someone’s read. Second, success rate punishes deep routes, which fail more often and win bigger. Neither caution changes the headline: even the best matchups fail more than a third of the time. A coverage beater moves the odds by a few points per throw. That still matters. Over a season of about 600 dropbacks, three points per throw is about 18 more successful plays, roughly one a game, which is why coordinators spend their week on it (The Game-Planning Week).

Calling against a menu, not a coverage

So a play-caller doesn’t pick the beater for the coverage; he picks the concept with the best expected result against the coverages this defense is likely to play in this situation. That is a weighted average, and you can do it with the matrix. Suppose that on second-and-7, film study says a defense plays Cover 3 35% of the time, Cover 1 25%, quarters 25% and Cover 2 15%. Score each grade as a number from −2 to +2, weight it by how often the coverage comes, and add. (This treats ordinal grades as if they were measurements: nobody knows that a “++” is exactly twice a “+”. It is a heuristic for organising a judgment, not a forecast of yards, and the ranking it produces matters more than the decimals.)

Flood wins on average, because the most common coverage is the one it beats. But notice the last column. Flood is bad against man, and one snap in four is man. Mesh and dagger never go below zero. If the down is third-and-7 and a bad play ends the drive, a careful coordinator may prefer the call with no bad outcome to the one with the best average. That is the logic of the next section.

Two-way concepts: an answer for man and an answer for zone

A two-way concept has an answer built in for both man and zone coverage, so it isn’t a bad call whatever the defense plays. There are two ways to build one. The routes themselves can work against both (crossing routes rub against man and settle into holes against zone), or a receiver can change his route after the snap according to the coverage, the option route of Passing Game Fundamentals, whose runner breaks away from his defender against man and sits down in the window against zone. Mesh is the classic of the first kind.

Notice what the two-way concept costs. Against zone, mesh’s crossers are fine, settling for five or six yards in front of defenders who rally to them; they are not a zone beater the way flood is. In the matrix, mesh is grey across the zone columns: never bad, rarely great. That is the trade. A two-way concept gives up some of the ceiling of a pure beater to raise the floor. On third-and-3, where five or six yards is a first down and a bad call ends the drive, that is a bargain. On first-and-10, where a coordinator can afford a swing for a big gain, he may prefer the pure beater and accept that it will sometimes meet the wrong coverage.

There are three common ways offenses build two-way plays:

  • Routes that work against both. Mesh, drive and the shallow cross rub against man and find holes against zone. So does a slant run inside a defender with outside leverage.
  • Choices inside one route. The option route, a seam route that stays straight against one deep safety and bends into the open middle against two (a route conversion), and the hitch that turns into a fade against press: each receiver reads the coverage after the snap, and the quarterback reads the same thing.
  • A man side and a zone side. Put a zone beater on one half of the field (curl-flat, smash) and a man-beater on the other (a rub, a slant-flat), and let the quarterback choose his half from the man/zone read: the man/zone indicator on a motion before the snap, or the corners’ first steps after it. This is called a half-field or combination call, and many third-down playbooks are built from them.

All three ask more of the quarterback and the receivers than a single beater does, because now both of them must read the coverage the same way, at speed. When they disagree, the result is the interception where the receiver turned out and the ball went in. That is why two-way concepts are a veteran offense’s tool and why young quarterbacks are often given simpler, single-answer calls.

The defense answers back

If the offense’s job is to call the concept that beats the defense’s coverage, the defense’s job is to play a coverage that beats the offense’s concept, or at least doesn’t lose to it. Defenses fight the matrix in three ways.

Change the coverage’s rules against a known concept. This is the match idea from Part 6 seen as a counter. The clearest example is four verticals against Cover 3. Spot-drop Cover 3 loses to it by arithmetic; Cover 3 match changes one rule, the seam-flat defender’s, and the arithmetic is gone.

Why does the matrix call smash “even” against quarters but a loss against Palms, if a vertical #2 makes Palms play like quarters? Because plain quarters handles smash in much the same way, which is why that cell is grey, not blue. What Palms adds is the trap on the other branch: if the offense answers with a quick out by #2, the corner, reading #2 from the start, jumps it.

Take the beater’s target away directly. Brackets, robbers and lurkers put an extra defender where a known concept wants to throw (Man Coverage). A defense that expects mesh on third-and-short can sit a linebacker at the mesh point, at five yards over the ball, so the crossers run into him instead of past each other. A defense that expects dagger can rob the dig with a safety who ignores the seam: usually the backside safety, freed when his own #2 stays short, or a poach call that sends him across on purpose.

Make the coverage hard to identify. Disguise and late rotation don’t change the matrix; they hide which column the offense is in until it’s too late to use it. That is why a beater called before the snap is a bet and a beater chosen after the snap, by a quarterback reading the key defender, is much closer to a sure thing.

The alert: calling the shot before the snap

One more pre-snap tool turns the coverage read into a big play. An alert (many staffs say shot alert) is a pre-planned exception to the called play: if the quarterback sees a particular look before the snap, he throws to the alert receiver first, usually on a deep route; if he doesn’t see it, he runs the play as called. The play-caller builds it in; the quarterback decides at the line.

The classic trigger is a single deep safety with the offense’s best receiver isolated on the other side of the field, one-on-one against a corner with no help leaning his way. That is the situation in which a deep throw has the best odds and the highest payoff: the receiver only has to beat one man, and if he does, nobody is behind the corner. Other alerts are keyed to Cover 0 (any one-on-one deep with no safety at all), to a pressed corner whose leverage gives away the throw (one shaded inside with no deep help is conceding the fade or the back-shoulder throw outside; one shaded outside with no inside help is conceding the slant), or to a safety cheating down into the box against play-action. A shot play from Play-Action, Bootlegs, and Screens is called to take a deep shot; an alert is a shot that the defense has to give.

Why not just call the deep shot? Because the deep shot is a bad play against most coverages and a great one against a few, and the alert lets the offense take it only when the coverage shows that it is one of the few. The called play underneath is the insurance. What it costs is that the quarterback makes the decision before the snap, which is exactly the moment the defense is lying to him. Alerts are therefore built around looks that are hard to fake: a single safety who can’t get back to the alert side in time, whatever he does at the snap, or a corner pressed with no help. If the safety rotates over at the snap, a good quarterback abandons the alert and goes back to the play.

Putting it together: identify, then predict

Everything above reduces to a routine you can run on any passing play, live or on the replay: the quarterback’s four steps, minus the throwing. Call the shell and your two likeliest coverages; watch the safeties at the snap; let the corners name the coverage within its family; find the defender the routes are putting in conflict (two receivers to one sideline at two depths means a high-low on that side’s corner or curl-flat defender; a slot running up the seam means somebody’s rule about #2 is being tested); then say the open receiver out loud before the ball is thrown. What to notice: what a quarterback brings to the matrix. Two-high coverage is built to take away the deep throw and the easy read; it is beaten by a quarterback who identifies the coverage at the snap, moves a safety with his eyes and throws on time into a window between the levels: the dig behind a read safety, the seam the Mike didn’t carry. On a Rams passing down from 2025, watch Stafford’s helmet in the first second after the snap. It often goes where the ball isn’t going.

Watch for it on Sunday

TipWatch for it

Predict the open man. For one quarter, on every dropback pass, make the coverage call and then a prediction, before the ball is thrown.

  • Before the snap: shell (one-high or two-high), corners (press or off), and the two coverages you think are most likely. Is a receiver isolated against a single safety? That’s an alert candidate.
  • One second after the snap: how many safeties in the middle? Then the corners: squat (Cover 2), bail (Cover 3), read (quarters), trail (man). Say the coverage out loud.
  • Name the conflict defender on the side the routes are going: the corner (two receivers at two depths outside), the curl-flat defender (three at three levels), the read safety (a slot going up the seam), the matchup (crossers against man).
  • Predict the open receiver from what that defender does, before the throw. Then watch.
  • Watch the quarterback’s helmet. Did he look one way and throw the other? Who moved?

For your log: coverage called, receiver predicted, receiver thrown to. Score yourself on the predictions. Over a quarter you’ll find that you get Cover 3 and Cover 2 right far more often than quarters and match coverages, which is exactly the point of those coverages.

Before the snap: does the quarterback stay with the called play, or is there a better throw?

TipAnswer

This is the textbook alert: throw to X first. One deep safety, already leaning toward the trips, means X is one-on-one with no help on his side. The corner pressing him is shaded inside, so he is taking away the slant and conceding the outside: the alert throw is a fade or a back-shoulder throw up the left sideline. If X wins at the line, the throw has a high chance of a big gain, and the safety, 4 or 5 yards the wrong way and 14 deep, can’t get there in time.

The insurance: if the safety rotates back toward X at the snap, or the corner bails off the press, the alert is off and the quarterback works the flood to the trips side, where three receivers now face a defense that has shifted its help that way. And the defense knows all of this: a single safety cheating to trips is sometimes bait for exactly this throw, with a rotation back at the snap. That is why an alert is decided before the snap but confirmed in the first half-second after it.

What is the coverage, which defender should the quarterback read, and where will the ball go?

TipAnswer

Tampa 2. Two safeties splitting the field (middle open), corners squatting in the flats (Cover 2 corners), and the Mike running down the middle: that is the Tampa 2 Mike, whose job is to carry the most dangerous seam and close the gap between the safeties that four verticals is built to hit.

Read the Mike, then a safety. The Mike can only carry one seam. Whichever he takes, the other seam is a one-on-one high-low on that side’s safety: the seam inside him and the go route outside him. If the safety stays inside on the seam, the go route is open along the sideline behind the squatting corner (the hole shot); if he widens toward the go route, as these two already have, the seam splits the gap between him and the Mike. Here the Mike has opened toward Y’s seam on the right, so the throw is H’s seam on the left, between the Mike and the free safety. This is where the look-off earns its keep: a quarterback who wants that throw holds his eyes on Y’s seam for a beat to keep the Mike drifting right, and many offenses coach the seam runners to bend away from the Mike as well. If the quarterback can’t decide in time, the back underneath, past the two hook defenders who have dropped deep under the verticals, is the checkdown. This is why four verticals is only a “+” against Cover 2 in the matrix, not a “++”: the Tampa Mike takes away half of the concept.

You have four concepts on the call sheet: smash, flood, dagger and mesh. Using the matrix grades (reprinted below), which has the best expected result? Which has the safest floor? Which would you call, and why?

TipAnswer

Weight each grade by the coverage mix (Cover 2 30%, quarters 35%, 2-Man 20%, Cover 3 15%):

  • Smash: +2, 0, −2, +1 → 0.60 + 0 − 0.40 + 0.15 = +0.35, worst case −2 (2-Man).
  • Flood: +1, +1, −1, +2 → 0.30 + 0.35 − 0.20 + 0.30 = +0.75, worst case −1.
  • Dagger: +1, +2, −1, 0 → 0.30 + 0.70 − 0.20 + 0 = +0.80, worst case −1.
  • Mesh: 0, 0, +1, 0 → +0.20, worst case 0.

Dagger has the best average, because the most likely coverage, quarters, is the one it was built to beat, and it is decent against Cover 2 too. Flood is a close second. Mesh never loses but rarely wins big. Smash, the famous Cover 2 beater, is the worst choice here: this defense plays enough 2-Man to punish it.

What a coordinator would add. Third-and-7 is a long-ish third down, and route depth matters as much as the grade. Dagger’s dig, at 15 or 16 yards, is past the sticks. Mesh’s crossers settle at 5 or 6 yards against zone, short of the line to gain, so its “safe floor” is less safe than it looks: a completed mesh crosser that is tackled a yard short still ends the drive. But dagger needs about three seconds, so the call depends on the protection and on whether this defense blitzes on third down; against a pressure-heavy defense the slightly quicker flood, or a two-way call, might be safer. And the 20% of 2-Man is the danger in every choice except mesh: if the 2-Man share were higher, or the down shorter, mesh’s floor would start to look like the smart call. The matrix doesn’t make the decision; it organises it.

Takeaways

  • The quarterback reads coverage in four steps: the shell before the snap (a hypothesis), the rotation at it (the middle open or closed one second later), the key defender after it (the one the concept puts in conflict), and the throw that defender leaves. The look-off is the same read run in reverse: move the key defender with your eyes, then throw where he was.
  • Every cell of the concept-by-coverage matrix follows from two questions: where are this coverage’s thin spots, and does it give help to the defender this concept attacks? Smash beats Cover 2 at the sideline; flood and four verticals beat Cover 3’s arithmetic; dagger and scissors make quarters’ read safeties pay for their rule; mesh beats man with traffic; quick throws, seams and screens beat pressure. Against split-field coverage, read each half on its own. The grid leaves out the run game, which is where Cover 3 and the two-high shells earn their keep.
  • Beaters are bets. Disguise hides the coverage, match rules change what a label means, execution and talent widen or close windows, and the defense chooses by situation too. In charting data the edges are real but small: even the best route-against-coverage matchups fail more than a third of the time.
  • Two-way concepts trade ceiling for floor. Built-in answers for man and for zone mean the call is never bad, at the cost of rarely being great; they rule third down.
  • The matrix is never finished. Every defensive counter (Cover 3 match, Palms, Tampa 2, robbers) opens something new, and the offense’s next concept attacks it. The shot alert lets the offense take a deep shot only when the coverage gives it.
  • Watching: call the coverage one second after the snap, find the conflict defender, and predict the open man before the throw.
