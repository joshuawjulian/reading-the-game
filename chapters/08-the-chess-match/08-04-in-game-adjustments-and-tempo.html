def _P(fld, p, d, w):
    return fld.xy(*p.to_abs(d, w))


def go_to(p, key, end, kind="route", delay=0.0, speed=None, via=()):
    """Move `key` through absolute (d, w) points `via` to `end`, as a route, path, run, block or drop."""
    start = np.array(p._cur(key))
    pts = [tuple(np.array(q) - start) for q in (*via, end)]
    if kind == "route":
        return p.route(key, pts, frame="field", delay=delay, speed=speed)
    if kind == "block":
        return p.block(key, pts=pts, delay=delay, speed=speed)
    if kind == "drop":
        return p.drop(key, end, delay=delay, speed=speed)
    return {"path": p.path, "run": p.run}[kind](key, pts, delay=delay, speed=speed)


def say(fld, p, d, w, text, color=style.INK_2, fontsize=7.0, ha="center", weight="normal"):
    """A short on-field label at (d, w)."""
    x, y = _P(fld, p, d, w)
    fld.ax.text(x, y, text, fontsize=fontsize, color=color, ha=ha, va="center", zorder=11, fontweight=weight,
                linespacing=1.1, bbox=dict(boxstyle="round,pad=0.18", fc=style.SURFACE, ec="none", alpha=0.92))


def point_to(fld, p, text_at, target, text, color=style.INK_2, fontsize=6.8):
    """A label with a thin leader line to the player or spot it names."""
    (xt, yt), (xp, yp) = _P(fld, p, *text_at), _P(fld, p, *target)
    fld.ax.annotate(text, (float(xp), float(yp)), xytext=(float(xt), float(yt)), fontsize=fontsize,
                    color=color, ha="center", va="center", zorder=11, linespacing=1.1,

                    bbox=dict(boxstyle="round,pad=0.18", fc=style.SURFACE, ec="none", alpha=0.92))


def threat(fld, p, *keys):
    """Dashed brown rings: defenders threatening to blitz before the snap."""
    for k in keys:
        x, y = _P(fld, p, p.players[k].d, p.players[k].w)
        fld.ax.add_patch(Circle((float(x), float(y)), 1.25, fc="none", ec=style.BALL, lw=1.5, ls=(0, (2.2, 1.4)),
                                zorder=7))


def ghost(fld, p, pts, color=style.DEFENSE, lw=1.4, alpha=0.55, zorder=5):
    """A faint dashed arrow through absolute (d, w) points: what a player did on earlier snaps."""
    xs, ys = _P(fld, p, np.array([q[0] for q in pts]), np.array([q[1] for q in pts]))
    fld.ax.plot(xs[:-1], ys[:-1], color=color, lw=lw, ls=(0, (3, 2)), alpha=alpha, zorder=zorder)
    fld.ax.add_patch(FancyArrowPatch((xs[-2], ys[-2]), (xs[-1], ys[-1]), arrowstyle="-|>", mutation_scale=10,
                                     color=color, lw=lw, alpha=alpha, ls=(0, (3, 2)), zorder=zorder))


def clean(ax):
    """Zoomed diagrams: drop hash ticks and yard numbers, and clip text to the window."""
    for ln in list(ax.lines):
        xs, ys = ln.get_data()
        if len(xs) == 2 and ln.get_linewidth() == 0.5 and np.hypot(xs[1] - xs[0], ys[1] - ys[0]) < 1.0:
            ln.remove()
    for t in list(ax.texts):
        if t.get_text().isdigit() and t.get_alpha() == 0.55:
            t.remove()
        else:
            t.set_clip_on(True)


def draw(p, ax=None, width_in=5.8, **kw):

MARK_R = 0.6    # frame-strip marker radius in yards, smaller than the static diagrams'


def tracking(play):
    df = play.to_tracking()
    df["_id"] = df["displayName"].astype(str)
    return df


def panel(ax, play, df, t, lateral, window, rings=()):
    """One moment of a simulated play: markers, trails of on-screen players, rings."""
    tmap = df.groupby("frameId")["time"].first()
    fid = int((tmap - t).abs().idxmin())
    ylo, yhi = sorted((play.ball_y - lateral[0], play.ball_y - lateral[1]))
    fld = Field(ax=ax, orient="vertical", los=play.los, window=window, numbers=False, hashes=False,
                lateral=(ylo, yhi), to_go=play.to_go)
    cur = df[df["frameId"] == fid]
    shown = cur[cur["x"].between(play.los + window[0] + 0.4, play.los + window[1] - 0.4)
                & cur["y"].between(ylo + 0.4, yhi - 0.4)]
    ids = set(shown["_id"])
    trails = df[(df["frameId"] <= fid) & df["_id"].isin(ids) & (df["side"] != "ball")]
    for (_, side), g in trails.groupby(["_id", "side"]):
        xs, ys = fld.xy(g["x"].values, g["y"].values)
        ax.plot(xs, ys, color=style.OFFENSE if side == "off" else style.DEFENSE, lw=0.9, alpha=0.4, zorder=3)
    ppy = fld.pts_per_yard()
    for _, rr in shown.iterrows():
        x, y = (float(v) for v in fld.xy(rr["x"], rr["y"]))
        if rr["side"] == "ball":
            ax.add_patch(Ellipse((x, y + 0.45), 0.7, 0.42, fc=style.BALL, ec="white", lw=0.5, zorder=9))
            continue
        lab = str(rr.get("label", "") or "")
        if rr["side"] == "off":
            ax.add_patch(Circle((x, y), MARK_R, fc=style.OFFENSE, ec="white", lw=0.6, zorder=8))
            tc = "white"
        else:
            ax.add_patch(RegularPolygon((x, y), 4, radius=MARK_R * 1.3, orientation=np.pi / 4,
                                        fc=style.DEFENSE_TINT, ec=style.DEFENSE, lw=1.0, zorder=8))
            tc = style.INK
        fs = max(4.0, min(8.0, ppy * (0.8 if len(lab) <= 1 else 0.6)))
        ax.text(x, y, lab, ha="center", va="center", fontsize=fs, color=tc, fontweight="bold", zorder=10)
    for key, col in rings:
        rr = cur[cur["displayName"] == key].iloc[0]
        x, y = fld.xy(rr["x"], rr["y"])
        ax.add_patch(Circle((float(x), float(y)), 1.05, fc="none", ec=col, lw=1.7, zorder=7))
    # once a pass is in the air: the throw as a dashed brown arrow, and a brown ring on the target
    passes = [(k, tt) for kind, k, tt in play.ball_events if kind == "pass"]
    if passes and t >= passes[0][1] + 0.05:
        target, t_rel = passes[0]
        b = df[(df["side"] == "ball") & (df["time"] >= t_rel - 1e-6) & (df["time"] <= t + 1e-6)]
        if len(b) >= 2:
            xs, ys = fld.xy(b["x"].values, b["y"].values)
            ax.plot(xs[:-1], ys[:-1], color=style.BALL, lw=1.3, ls=(0, (3, 2)), zorder=8.5)
            ax.add_patch(FancyArrowPatch((xs[-2], ys[-2]), (xs[-1], ys[-1]), arrowstyle="-|>", mutation_scale=8,
                                         color=style.BALL, lw=1.3, zorder=8.5))
        rr = cur[cur["displayName"] == target].iloc[0]
        x, y = fld.xy(rr["x"], rr["y"])
        ax.add_patch(Circle((float(x), float(y)), 1.05, fc="none", ec=style.BALL, lw=1.5, zorder=7))
    return fld


def strip(play, times, notes, lateral, window, ncols=2, rings=(), width_in=6.4):
    """Numbered, captioned frame strip: the PDF stand-in for the video."""
    df = tracking(play)
    nrows = int(np.ceil(len(times) / ncols))
    pw = width_in / ncols
    span_w, span_d = lateral[1] - lateral[0], window[1] - window[0]
    fig, axes = plt.subplots(nrows, ncols, squeeze=False,
                             figsize=(width_in, nrows * (pw * span_d / span_w + 0.55)))
    for ax in axes.flat[len(times):]:
        ax.axis("off")
    for i, (ax, t) in enumerate(zip(axes.flat, times)):
        panel(ax, play, df, t, lateral, window, rings)
        ax.set_title(f"{i + 1}.  " + ("snap" if abs(t) < 1e-6 else f"{t:+.1f} s"), fontsize=8.5,
                     color=style.INK_2, loc="left")
        ax.text(0.5, -0.03, notes[i], transform=ax.transAxes, ha="center", va="top", fontsize=7.4,
                color=style.INK, linespacing=1.15)
    fig.tight_layout(h_pad=1.6, w_pad=0.6)
    return fig


def video(play, times, notes, lateral, window, rings=(), width_in=5.6):
    """Web version of the strip: every frame, with the strip's notes as running captions."""
    from matplotlib import animation
    from IPython.display import HTML, display
    df = tracking(play)
    span_w, span_d = lateral[1] - lateral[0], window[1] - window[0]
    fig, ax = plt.subplots(figsize=(width_in, width_in * span_d / span_w + 0.4))
    tmap = df.groupby("frameId")["time"].first()

    def update(fid):
        ax.clear()
        t = float(tmap[fid])
        panel(ax, play, df, t, lateral, window, rings)
        cur = [n.replace("\n", " ") for tt, n in zip(times, notes) if tt <= t + 1e-6]
        ax.set_title(("snap" if abs(t) < 0.05 else f"{t:+.1f} s") + (("   " + cur[-1]) if cur else ""),
                     fontsize=8.5, color=style.INK, loc="left")
        return []

    anim = animation.FuncAnimation(fig, update, frames=sorted(df["frameId"].unique()), interval=100)
    html = (anim.to_html5_video(embed_limit=40).replace("<video ", '<video style="max-width:100%;height:auto" ')
            if shutil.which("ffmpeg") else anim.to_jshtml(default_mode="once"))
    plt.close(fig)
    display(HTML(html))


def animation_or_strip(play, times, notes, lateral, window, rings=(), ncols=2):
    """Video on the website; a numbered, captioned frame strip in the PDF."""
    if pdf_build():
        strip(play, times, notes, lateral, window, ncols=ncols, rings=rings)
        plt.show()
    else:
        video(play, times, notes, lateral, window, rings=rings)

# ---------- tempo: game-clock seconds between consecutive snaps when the clock never stopped
TCOLS = ["season", "season_type", "game_id", "play_id", "posteam", "play_type", "down", "drive", "qtr",
         "game_seconds_remaining", "half_seconds_remaining", "wp", "no_huddle", "complete_pass", "out_of_bounds",

def snap_gaps(seasons):
    """Seconds of game clock from one snap to the next, on the same drive, when the clock kept running in
    between: the earlier play was a run or a completed pass that stayed in bounds, with no penalty, timeout,
    score, turnover or spike and nothing else logged between the two snaps; both snaps in a neutral game
    state (win probability 20-80%) and more than 2:30 before the end of the half."""
    d = pbp(seasons, columns=TCOLS)
    d = d[d["season_type"] == "REG"].sort_values(["game_id", "play_id"]).copy()

The game is re-planned every series

Start with a fact that surprises most fans: once the game starts, nobody installs anything. The plays a team will run on Sunday were chosen, taught and practiced from Wednesday to Friday (The Game Plan walks through that week). What changes during a game is the order of the menu, the rules players use inside each play, and the play-caller’s beliefs about what the opponent is doing. That is still a lot. It’s the difference between a chef inventing a dish mid-service, which never happens, and a chef noticing that table six hates cilantro, which happens all night.

Two kinds of information drive those changes, and they arrive in two places.

From above. The coaching booth is the glassed-in coaches’ room at press-box level, the place you met in What a Play Really Is. From up there you see all 22 players at once, including the safeties 15 yards deep whom nobody on the sideline can see. Typically one coordinator, or an assistant who isn’t calling plays, sits there with the quality control coaches and analysts. While the game goes on they keep running tallies: what coverage did the defense play against each personnel group, did the safety rotate toward the motion, which blitzes came on third down, which of our linemen is losing. The league even provides a booth dashboard for counting personnel groupings and play results in real time.1 It is a coach with a spreadsheet and a headset, not a genius with a flash of insight.

From the sideline. Since the 2014 season every team has had league-issued Microsoft Surface tablets on the sideline and in the booth. On them, within seconds of each snap, coaches and players get high-resolution still photos of the formation before the snap and the shape of the play just after it.2 Under the league’s equity rule each team may use only as many working tablets as its opponent has, so a failure on one side shuts down the same number on the other; the same rule switches off a team’s coach-to-player headsets if the other team’s fail (What a Play Really Is).3 The pictures are stills, not video; the point is to answer a specific question (“where was the safety when you released?”) with a picture instead of an argument.

ImportantWhy it exists: the booth and the tablets

Every play has two halves of information: what the defense showed before the snap and what it did after. Players see their own patch of grass; the sideline sees helmets. The booth and the tablets exist to give the people making the next decision the whole picture, fast. The cost is attention and trust. A coordinator who tries to absorb everything the booth sees stops thinking about the next call, so good staffs filter hard: the booth reports only what the play-caller has asked it to look for, plus anything that has changed. Which coordinator sits upstairs varies by team, which you can usually spot on a broadcast by who is shown wearing a headset in the press box.

NoteMadden vs. real life

In a video game, when a play stops working you scroll to a different one: the whole playbook is always available. A real team can only run what it practiced that week, and a play that hasn’t been repped against the opponent’s looks is a play somebody will run wrong. So “adjusting” in the NFL mostly means choosing differently from a small menu, and changing small rules inside plays (a protection that slides the other way, a corner who stops pressing a fast receiver), not drawing up something new in the dirt. Coaches do occasionally scribble a wrinkle on a whiteboard, but the bigger the change, the more likely it is to be run badly.

Halftime: thirteen minutes

The NFL’s halftime is 13 minutes by rule.4 For the staff, that is the one sustained block of the game in which every coach and player is in the same room, and the coaches from the booth come down to join them. A halftime adjustment is a change to the game plan made in that window.

Thirteen minutes is less than it sounds. Players need to walk in, get treatment and use the bathroom; they have to be back on the field in time to warm up; and the head coach usually wants a minute with the whole team. What’s left is a few minutes for coordinators to agree on what they’ve seen and a few minutes for each unit to hear the changes. That shapes what a halftime adjustment can be:

  • Re-ranking the sheet. The plays that worked move up, the ones that didn’t get scratched, and the constraint plays that the first half set up (Constraint Theory) get scheduled. This is the commonest and cheapest change.
  • Changing who does what inside the plays already installed. Sliding the protection toward a rusher who is winning, keeping a tight end in to help a struggling tackle, moving the best corner onto the receiver who is hurting you, playing more of the coverage that held up and less of the one that didn’t.
  • Changing the mix. More of one personnel group, more motion, more runs, fewer blitzes. The plays are the same; their frequencies change.
  • Fixing a tell. If the opponent seems to know what’s coming, a staff looks for what’s giving it away (a back’s alignment, a linebacker’s stance) and changes it.
  • Re-planning around injuries. The most common adjustment of all is forced, and it doesn’t wait for halftime. When the left tackle or the nickel back goes down, the plan bends around his backup: a tight end or back is assigned to help the backup tackle, the protection slides toward him, a backup corner gets safety help or is hidden in a package, and plays the backup hasn’t repped get scratched.

Notice what’s missing from the list: new plays. With five minutes per unit and no reps, there is no time to teach one.

CautionCommon misconception

“The second half was won in the locker room.” Sometimes the halftime change is real and visible: a team that was blitzed to pieces comes out keeping six in to protect, or a defense that was gashed by outside runs starts setting the edge differently. But a lot of what gets called a halftime adjustment is the scoreboard, the clock and plain luck moving in the other direction, as the last part of this chapter shows with data. When you hear the phrase, ask: what specifically changed?

Tempo: the clock between plays

Now the other weapon a staff can use during a game: speed. When a play ends, the 40-second play clock starts, and the offense must snap the ball before it reaches zero.5 What it does with those 40 seconds is its tempo: how fast it gets to the line and snaps. Most offenses, most of the time, use almost all of it: huddle, line up, shift or motion, look at the defense, snap with a few seconds left. A hurry-up offense does the opposite and snaps as fast as it can. Every team hurries when it’s short of time (the two-minute drill, the subject of The Clock, the Two-Minute Drill, and Fourth Down). The interesting question is the team that hurries when it doesn’t have to.

Three words get mixed up on broadcasts, so pin them down:

  • No-huddle is about communication: the call reaches the players at the line, by signals, words or wristbands, instead of in a huddle (What a Play Really Is).
  • Tempo is about speed: how much of the play clock the offense uses.
  • Hurry-up is tempo turned all the way up.

A team can go no-huddle and still be slow (line up fast, then look at the defense and wait for the sideline), and a team can huddle and still be fast, as you’ll see with the sugar huddle below.

How fast do NFL offenses actually play? The play-by-play data doesn’t record the play clock reliably, but it does record the game clock at every snap. On snaps where the clock kept running between plays (the earlier play was a run or a completed pass that stayed in bounds, with no penalty or timeout), the game-clock seconds between two snaps measure how long the play took plus how long the offense took to snap the next one.6 A team that snaps with the play clock nearly expired lands around 40 seconds or a bit more; a true hurry-up offense lands around 25 to 30.

Figure 45.2 says something worth absorbing before any history: the modern NFL barely hurries in normal game situations. The median team averaged about 39 seconds between running-clock snaps, which means it snapped near the end of the play clock, and only one team, Kellen Moore’s New Orleans Saints,7 averaged under 35. No-huddle snaps were faster (a median of 31 seconds against 40), but not all no-huddle teams are fast: Washington skipped the huddle on about 78% of these snaps and still had a median of 39 seconds. That is the “information” no-huddle you met in What a Play Really Is, and it has a recognizable rhythm: the offense hurries to the line, lets the defense settle into its look, and then everyone, quarterback included, turns to the sideline for the real call before snapping late in the clock.

Why go fast?

If almost nobody does it, why does tempo matter? Because when a team turns it on, at the right moment, it attacks the defense in five ways.

  1. Fatigue, and the end of the rotation. Defensive linemen are the biggest men on the field and the ones doing the most violent work, so defenses rotate them constantly. In 2025 the typical regular defensive lineman played only about 48% of his team’s defensive snaps, only about 3% averaged 80% or more, and a team used about 7 linemen a game.8 A rotation needs substitutions, and tempo takes substitutions away (the next section shows how), so the same four men keep rushing. At a normal pace a lineman gets 30 seconds or more between snaps (Figure 45.2: even New Orleans averaged about 34 seconds from snap to snap, which is roughly 28 seconds of rest after a five- or six-second play). A true hurry-up, snapping 20 to 25 seconds after the last snap, cuts that to 15 or 20 seconds. Tired linemen stop winning their blocks and tired defensive backs stop running with receivers.
  2. The substitution freeze. The rules let the defense match the offense’s personnel only when the offense substitutes. If the offense keeps the same eleven and lines up quickly, the defense is stuck with whoever is on the field. This is big enough to get its own section, next.
  3. Simplifying the defense. Disguise, pressure and rotation take time: the call has to be signalled in, relayed, and then the defenders have to walk to their spots and hold a fake picture. Snap fast and the defense has to line up where it will play, which means the quarterback sees the truth.
  4. More snaps. More plays per game means more chances for a better offense’s edge to show up, the same logic as a better poker player wanting to play more hands.
  5. Beating the replay. Scoring plays and turnovers are reviewed automatically, but most other close calls, a catch that might have hit the ground or a spot just short of the line to gain, need a coach’s challenge, and a challenge must come before the next snap.9 An offense that likes the call on the field can sprint to the line and snap before the other sideline has seen a replay (Officiating, Replay, and the Rules of the Ball).

And the costs, because there are always costs:

  • Your own substitutions get harder. Changing your own personnel invites the defense to change too and slows you down, so a tempo offense tends to stay in one or two groups and needs players who can do several jobs: a tight end who can block and run routes.
  • Your own players get tired too, and so do the call and the communication: a tempo offense needs short calls, simple rules and a quarterback who can manage the line.
  • The other team gets the ball back sooner. A fast three-and-out (three plays, no first down, and a punt) leaves your own defense on the field after barely sitting down. That last cost turns out to be big, as the Eagles film room below shows.

The substitution freeze

You met the rule that makes tempo a strategic weapon in Personnel Groupings: substitution matching. Before the two-minute warning, if the offense substitutes, the umpire stands over the ball and the offense may not snap until the defense has had a reasonable chance to substitute too.10 Here it is from the clock’s point of view.

Read the rule backwards and you have the tempo offense’s loophole: the protection exists only when the offense substitutes. If the offense sends out the same eleven players, nobody stands over the ball, and the offense may snap as soon as it is set. The defense can still change players, but at its own risk, and the risk comes in two forms. If the defense has twelve men in its formation when the snap is about to come, the officials blow the play dead before it starts: five yards. If the twelfth man is still jogging off when the ball is snapped, the play runs, and afterwards the offense chooses between the result and the five yards. That “free play” is why quarterbacks hurry the snap when they see a defender running for the sideline. Defenses get caught often enough to make the threat real: from 2016 to 2025, outside the last two minutes of each half, defenses were flagged for too many men about 25 times a season, and the foul was about 4 times as common on no-huddle snaps as on snaps after a huddle.11 The result is the substitution freeze: by keeping its personnel and moving fast, the offense freezes the defense in whatever personnel it had on the field.

Two things set the speed limit. The officials still have to spot the ball and the umpire has to get out of the way. Only about one no-huddle snap in twenty came within 21 seconds of the previous snap in the running-clock data, and after a five- to seven-second play that is about 15 seconds from tackle to snap, however fast the offense wants to go. And the defense’s call travels the same way the offense’s does: the green dot’s helmet radio goes silent at 15 seconds on the play clock, just as the quarterback’s does. Against a huddling offense the defensive play-caller has twenty seconds or more to decide and signal; against tempo he has maybe eight, starting from the moment his personnel caller (the assistant who watches the offense’s sideline and shouts the grouping) confirms that nobody came on.

Notice what the freeze does not do. It doesn’t guarantee a good play: base linebackers can cover tight ends, and many defenses have a call ready for exactly this. What it does is take away the defense’s best tool, choosing the right players, and force the defense to solve the problem with alignment and rules, which is harder and more error-prone. And the offense gets to choose its moment: tempo is usually a change-up, turned on after a big gain (when the defense is tired and wants to substitute), or when the offense sees a personnel group it likes on the field.

ImportantWhy it exists: the freeze is the matching rule’s other half

The matching rule’s bargain (Personnel Groupings explains why it exists) is that the offense chooses its players first and the defense gets a fair moment to answer. The freeze is that bargain run in reverse: give up your own right to substitute, and you take away the defense’s protection. The league keeps adjusting the edges of the trade. Since 2025 officials have been told to close the substitution window if a defense deliberately slow-walks its substitutes, and an offense that fakes a substitution (a player jogging onto the field and back off to draw a defensive response) now opens a substitution window, with a warning first and a 15-yard penalty after that. The trigger was a Packers fake substitution in Week 3 of the 2025 season.12

The sugar huddle

Between the huddle and the no-huddle sits a hybrid. In a sugar huddle the offense huddles quickly, a few yards from the ball instead of the usual seven or eight, breaks almost at once, sprints to the line and snaps fast. It gets the huddle’s strengths (everyone hears the whole call in one voice, the defense can’t read signals) without giving the defense its usual time to substitute, signal its call and set a disguise. Offenses save it for moments when they have a call they love and want to run it before the defense can reorganize. Ben Johnson’s Lions were known for it at key moments, and Chicago expected him to bring it with him when he became the Bears’ head coach in 2025.13

You can tell it from a team that merely huddles a little closer by the clock. A normal huddle breaks with 20-odd seconds left and the snap comes with about five on the play clock; after a sugar huddle the players are in their stances a second or two after it breaks and the snap comes with 25 to 30 seconds still showing (Figure 45.3, bottom). The huddle itself forms very close to the ball, sometimes with the quarterback facing the line. On the defensive side, watch for defenders looking to the sideline for a call that hasn’t arrived.

A short history of going fast

Tempo has a richer history than most schemes, because every time an offense used it, the league and the defenses argued about whether it was fair.

Cincinnati, 1988. The first NFL team to make the no-huddle a standard weapon was Sam Wyche’s Bengals, who called it their “attack offense”: shortened huddles very close to the line (what today would be called a sugar huddle), or none at all. In the 1988 playoffs the Seahawks answered it with a transparent tactic: their defensive tackle Joe Nash went down “injured” on third down after third down, stopping the clock so Seattle could bring in its nickel package. When Bills coach Marv Levy said he’d do the same in the AFC Championship Game, the league told Wyche, about two hours before kickoff, that the Bengals would be penalized 15 yards every time they ran the no-huddle. Wyche, a former NFL quarterback with a gambler’s nerve, demanded the commissioner and threatened to tell the world after the game that the league’s penalties had beaten him. The ruling was reversed within seconds, and Cincinnati won 21–10.14

Buffalo, 1990. The team that lost that game learned from it. The Bills built their own version, the K-Gun, a no-huddle offense built around three wide receivers and a tight end who could block and catch (Keith McKeller, whose nickname “Killer” gave the offense its K, by most accounts) and quarterback Jim Kelly calling the plays at the line.15 Under head coach Marv Levy and offensive coordinator Ted Marchibroda, Buffalo went 13–3 in 1990, led the league with 428 points, beat the Raiders 51–3 in the AFC Championship Game, and went on to play in four straight Super Bowls.16 The K-Gun was tempo as a system: the same personnel all game, the defense frozen in whatever personnel it started the drive with, the quarterback choosing the play after seeing the defense.

It also produced one of the most famous answers to tempo, which wasn’t a defensive scheme at all. In Super Bowl XXV, the New York Giants kept the ball for 40 minutes and 33 seconds, a Super Bowl record, by running the ball and sustaining long drives, and won 20–19. The best defense against a fast offense is an offense that keeps it on the bench. (Bill Belichick’s defensive plan for that game, which invited the Bills to run rather than throw, is drawn in The Belichick Patriots.)

Indianapolis and Denver, 2000s–2013. Peyton Manning’s no-huddle was the information version: line up early, read the defense, and change the play at the line while the defense, frozen in its personnel, could only disguise (What a Play Really Is tells his “Omaha” story, and The Pre-Snap Phase his check-with-me system). When Manning wanted to, he could also go fast: his 2012 and 2013 Broncos were among the quickest teams in Figure 45.5, alongside the 2012 Patriots, who had turned their no-huddle into a true hurry-up.

Philadelphia, 2013. Then came the most famous tempo experiment in NFL history.

How defenses answer tempo

A defense facing tempo has a problem with three parts: it can’t change players, it has little time to get a call in (maybe eight seconds instead of twenty), and it has no time to disguise. Its answers go after each part, and each has a price.

Substitute first. The cleanest answer to the freeze is to beat it to the punch: change players the instant the whistle blows, before the offense is set, as defenses already do with their third-and-long and short-yardage packages (Personnel Groupings). A defense that expects a hurry-up after a big gain can send its nickel on immediately. The cost is a guess: if the offense then substitutes, it gets to choose last, and if it snaps while the swap is still under way, the defense is the one with twelve men.

Simplify the call. Every defense that expects tempo has a short list of tempo calls: one word, or one hand signal, that tells all eleven players the front, the coverage and the rush, with nothing that requires a walk-up, a shift or a disguised rotation. The idea is to be sound rather than clever: a call that has an answer for every formation and every run, so that nobody can be caught out of position. The cost is plain in Figure 45.6: the defense that lines up where it will play gives the quarterback an honest picture, and gives up the disguise and pressure tricks of Disguise, Rotation, and Split-Field Coverage and Zone Blitzes and Simulated Pressure.

Let the players adjust by rule. The defensive version of a check-with-me is a call that isn’t finished until the offense lines up: the sideline sends a front and a coverage family, and the players finish it themselves according to the formation they see (set the strength to the tight end, play quarters against two receivers on a side and a different rule against three). The green-dot linebacker or the safeties communicate the finishing touches with one or two words. This moves the decision from the coordinator, who is now too slow, to the players. The price is practice time and brainpower: every rule has to be repped against every formation during the week, a single player who gets the rule wrong breaks the coverage, and a defense built on rules is predictable to an offense that has studied them.

Recruit players who don’t need to come off. The deepest answer is roster construction. If your “linebacker” can cover a slot receiver and your nickel back can fit a run like a linebacker, the freeze stops hurting, because your base defense and your nickel defense are the same eleven men. The league-wide rise of hybrid safety-linebackers and the big nickel (a third safety as the fifth defensive back) is partly a response to tempo and spread offenses: it’s the defense’s way of buying back the substitution the freeze took away. The price here is size and scarcity. A 210-pound hybrid who can cover a slot receiver gets moved by a 315-pound guard when the offense runs at him, and a linebacker big enough to take on that guard usually can’t run with a receiver. Players who can do both are rare, and expensive.

Slow the game down. And then there are the answers the league polices. The 1988 Seahawks’ fake injuries were not the last: in September 2011, after two Giants defenders went down at the same moment against the Rams’ no-huddle on Monday Night Football, the league sent every team a memo warning of fines, suspensions and even lost draft picks for faking injuries.21 The rules add a cost of their own: before the two-minute warning, a player whose injury stops the game must leave for at least one down,22 so a fake costs the defense the faker. Defenses can also take a timeout, the cleanest reset there is, if they’re willing to spend it. In college football, where tempo became far more extreme, the rules committee in 2014 proposed a ten-second window at the start of each play clock in which the offense couldn’t snap, to let defenses substitute. It drew loud opposition from tempo coaches and was shelved.23

TipWatch for it

When an offense goes fast, take your eyes off the ball for one snap and watch the defense’s sideline and its secondary. Is anyone running off the field as the ball is snapped? Is a defender still pointing or walking to his spot? Is the defense lining up in a plain look it hasn’t shown all day? Those are the costs of tempo being paid, and they tell you why the offense chose that moment.

Does tempo work, and why did it fade?

The obvious question about all of this: does going fast actually gain yards? The running-clock data can give a rough answer. Of the snaps in Figure 45.2’s sample from 2006 to 2025, call a snap “fast” if it came within 30 seconds of the previous one. Fast snaps averaged +0.053 EPA per play and a 48% success rate, against +0.011 and 45% for the rest (Figure 45.8, right). Part of that gap is the situation, not the speed: fast snaps are more often second- or third-and-short, where every offense does well. Comparing only snaps on the same down and a similar distance shrinks the edge to about +0.03 EPA per play and 1.7 points of success rate.24

So tempo works, a little, as it is used. That last phrase matters. Offenses go fast when they like what they see (a tired defense, a mismatch frozen on the field), so this measures tempo at the moments a play-caller chose, not what would happen if a team snapped fast every time. An edge of three hundredths of a point per play is worth having on the snaps you pick; it is not big enough to pay the bill Kelly’s defenses paid if you go fast all game.

That may be why the league tried tempo and then mostly put it back on the shelf. In the left panel, the share of fast snaps rose from about one in twelve in the late 2000s to about one in seven in Kelly’s years, then drifted back. Three things probably explain the retreat. The bill is real: a fast offense that stalls puts its own defense back on the field. Defenses found the answers in the previous section, above all the versatile players who make the freeze hurt less. And the style of offense that took over uses the play clock instead of racing it: pre-snap shifts and motion, which take seconds every snap, rose from 37.6% of plays in 2014 to 63.9% in 2025 by PFF’s charting,25 and an offense that changes personnel groups from play to play opens the matching window every time it does (Motion and Shifts). Going fast survives as a change-up, which is what the data says it is good for.

The call after the call

Everything so far has been about each side adjusting to what the other did. The best play-callers go one step further and call for what the other side is about to do. That is the coordinator-versus-coordinator contest that coaches sometimes call “the call after the call”: you don’t call the play that beats what the defense just ran; you call the play that beats what the defense will run in response to what you just did.

A few things make this contest more than rock-paper-scissors.

Both sides keep score. The booth’s tallies are not just for the play-caller’s own side. The offensive staff is also tracking the defense’s answers: when we showed trips, what did they do; when we motioned, who followed. After a couple of series, each staff has a little in-game tendency report on the other, the live version of the self-scouting and tendency work in Constraint Theory.

The second answer is often the money call. A play-caller can deliberately run a play he expects the defense to adjust to, in order to set up the play that beats the adjustment. That’s the sequencing idea from Constraint Theory applied across a game, and the reason the most dangerous moment for a defense is often right after it has fixed a problem.

Nobody can go infinitely deep. If the offense calls for the defense’s adjustment, the defense can skip the adjustment and stay where it was, and now the offense’s clever call is running into the very defense it was designed to avoid. In practice coordinators think about one or two moves ahead, and the deciding factor is usually what the players can execute. A defense with one coverage it plays brilliantly can ignore much of this contest; a defense with five it plays adequately is always one step from a bust.

Here is a single adjustment and its answer as frame pairs. In the first half the defense plays man coverage on third down, and the offense beats it repeatedly with mesh, two shallow crossers that run past each other so the defenders chasing them in man have to fight through traffic. At halftime the defense keeps the same look but changes the rule for the two inside defenders: instead of each chasing one crosser all the way across the field, each sits in his area and takes whichever crosser comes to him, a banjo-style “pass it off” rule (a combo coverage).

def mesh(second_half):

    # mesh: two crossers about a yard apart in depth (H under at 4.5-5, Y over at 5.5-6), meeting over the ball

    go_to(p, "Y", (6.1, -12.0), "route", via=[(5.6, 5.5)])
    go_to(p, "X", (15.0, -13.0), "route"); go_to(p, "Z", (15.0, 13.0), "route")
    p.dropback(1.5)
    for k_ in ("LT", "LG", "C", "RG", "RT"):
        p.block(k_, pts=[(-1.0, 0.0)])
    p.block("RB", pts=[(0.6, 1.0)])
    p.man("LCB", "X"); p.man("RCB", "Z")
    go_to(p, "FS", (15.0, 0.0), "drop", speed=1.5)
    for k_ in ("LDE", "LDT", "RDT", "RDE"):
        p.rush(k_, pts=[(-1.4, 0.0)], speed=1.6)
    go_to(p, "MIKE", (0.4, 2.4), "path", speed=3.0)     # right B gap (the tackle has the right A gap)
    go_to(p, "WILL", (0.6, -0.8), "path", speed=3.0)    # left A gap
    if not second_half:
        # man: each chaser trails his crosser by a step or two, on his hip, and loses ground in the traffic
        go_to(p, "NB", (4.9, 3.6), "path", via=[(4.6, -5.6), (4.5, -1.4)], speed=4.5)

Do halftime adjustments decide games? Testing the narrative

Back to the analyst’s line. If halftime adjustments decide games, three things should show up in the data: second halves should look different from first halves in a way that favors the team with the better staff; a team’s first-half performance shouldn’t tell you much about its second half (the adjustment breaks the pattern); and some coaches should be consistently better than others at turning a game around after halftime. The play-by-play data lets us check each one. All the numbers below use regular-season games from 2006 to 2025 and leave out the last two minutes of each half and snaps where one team’s win probability (a model’s estimate of each team’s chance of winning from that moment, built in Win Probability and Decision Analysis) was above 90%, so that two-minute drills and garbage time (the late minutes of a lopsided game, which analysts filter out) don’t dominate.26

First: second halves look exactly like first halves, on average. League-wide, offenses averaged -0.004 EPA per play in first halves and -0.006 in second halves. If defenses (or offenses) systematically out-adjusted the other side at halftime, you’d expect a gap. There isn’t one, which is what you’d expect if both sides adjust about equally well, and if most adjusting happens every series rather than once.

Second: the first half tells you very little about the second half, and that’s mostly not adjustment. Figure 45.11 (left) splits about 7,121 team-games by how well each offense played in the first half. Offenses that were terrible before halftime were much better after it, and offenses that were brilliant came back to earth. The correlation between a team’s first-half and second-half EPA per play in the same game is only 0.08, on a scale where 0 means the first half tells you nothing about the second and 1 means it tells you everything.

That looks like a story about adjustments, and it’s the classic trap of regression to the mean. A first half is about 25 to 35 snaps, and EPA per play over that few snaps is mostly noise: a fumble, a dropped pass, a 60-yard run. An offense that had an awful first half was usually an ordinary offense having an unlucky half, and its next half looks ordinary, whether or not anyone adjusted anything. The team that “came out a different team after halftime” was often just the same team without the bad bounces. The same arithmetic makes the hot first-half team cool off.

That is what Figure 45.12 shows: 32 of the 35 coaches sit inside that band. Reid’s gap is one of the few outside it, which is about as many as luck predicts, and if it means anything, it says his teams were unusually good before halftime, not bad after it.
  • Score effects, for teams that lead. Teams that lead at halftime play differently afterwards: they run more, take fewer risks and drain the clock, while teams that trail throw more and gamble. You can see it in Figure 45.13. A team leading at halftime is usually the better team, so you’d expect it to keep outscoring its opponent; instead, leaders by one to 13 points were outscored in the second half, and big leaders barely broke even. Measured against what their quality predicted, halftime leaders gave back about 0.7 points in the second half. That is real, but small next to the dots: even for the coaches who led most often, it moves the second-half margin by only about 0.2 of a point. Score effects change how a game looks (more runs, fewer deep shots, a comeback that comes up short) far more than they change the final margin.
  • Possessions, for single games. Second halves start with a kickoff to the team that didn’t receive the opening one, and most coaches who win the coin toss defer so they get the ball to start the second half (The Game in Fifteen Minutes).27 In one game that matters: a team that kicked off to start the game, then watched the other offense hold the ball for long drives, can come out after halftime with the ball and a fresh set of downs, and look like a new team. Over 100 or more games who gets the ball first evens out, so possessions explain individual turnarounds like Super Bowl LVII, below, not a coach’s dot.

NoteFilm room: Kansas City Chiefs, Super Bowl LVII (2022 season)

Super Bowl LVII (February 12, 2023) is everyone’s favorite example of a halftime turnaround. The Eagles led 24–14 at the half; Patrick Mahomes had re-aggravated his injured ankle late in the second quarter; and the Chiefs scored on all four of their second-half possessions to win 38–35, including two fourth-quarter touchdowns on the same motion play (the story of “Corn Dog” is told in Motion and Shifts and What Beats What).28 “Kansas City’s halftime adjustments” is the usual summary, and a Super Bowl halftime, stretched for the show, gives a staff far more than the usual 13 minutes to make them. What does the data say changed?

What to notice: start with the first panel of Figure 45.14. Kansas City’s offense barely played in the first half: 20 runs and passes against Philadelphia’s 44, because the Eagles’ offense held the ball for long drives. The Chiefs had been effective when they did have it, about +0.15 EPA per play, and the second half gave them 31 snaps, starting with the ball: Philadelphia had received the opening kickoff, so Kansas City received the second-half kickoff and drove for a touchdown. That is the possessions effect in a single game. Some of the change was real and visible: the run game went from 36 rushing yards to 126, with Isiah Pacheco and Jerick McKinnon running well and Mahomes scrambling on the bad ankle. What didn’t change much was the motion: by FTN’s charting the Chiefs used motion on 40% of first-half snaps and 45% after halftime.29 The two motion touchdowns were a game-plan call used at the right moment near the goal line (the second one set up by Kadarius Toney’s 65-yard punt return, the longest in Super Bowl history), not a new offense invented in the locker room.

One thing held in both halves. Philadelphia’s defense led the league with 70 sacks in the 2022 regular season and sacked Mahomes zero times in the Super Bowl, even on a bad ankle. Kansas City’s protection plan, built during the week, worked from the first snap. And Philadelphia’s offense didn’t collapse after halftime: it ran 28 plays at a better EPA per play (+0.22, up from +0.10), and scored 11 points. The Eagles lost the second half because their defense, after keeping the Chiefs’ offense off the field for most of the first half, couldn’t get a single stop.

So was it halftime adjustments? Partly: the Chiefs ran more and ran better, and the staff saved its best red-zone call for the moments that mattered. But the honest summary also includes the second-half kickoff, a punt return, a protection plan that never needed adjusting, and an Eagles defense that simply didn’t stop anything. (If anyone’s defense was worn down, it was Kansas City’s, which had been on the field for 44 first-half plays.) That’s the habit this chapter wants you to build: when someone credits halftime, ask what changed, and check how much of it was the clock and the bounces.

TipWatch for it on Sunday: the adjustment hunter

Pick one team for a game and keep a short list in two columns, first half and second half.

Before the snap

  • Personnel: which groups does the offense use, and does the defense still match them? A team that lived in 11 personnel and comes out in 12 has changed its plan.
  • Tempo: does the offense go fast after big plays? Does the defense look rushed, with players running off at the snap or a plain, undisguised look?
  • Motion: more or less of it after halftime, and does the defense still follow it the same way?
  • Coverage shell: two-high or one-high before the snap, and does it change after the half?

After the snap

  • Pressure: count rushers on third down in each half. Did a blitz that worked disappear, or one that got burned stop coming?
  • Protection: is a back or tight end staying in to help a struggling tackle after halftime?
  • The fix and the answer: when the defense changes something, watch the next series. Does the offense call the play that beats the change? That’s the call after the call.

Then check the box score: who had more snaps in each half, and who led at halftime? Before you credit an adjustment, ask whether the scoreboard and the clock explain the second half.

The offense wants to snap fast and throw at those linebackers. What happens before the snap, and what matchup will the offense actually get?

TipAnswer

The substitution kills the tempo. Because the offense changed a player, the umpire stands over the ball until the defense has had time to answer, and the defense sends its nickel on for a linebacker. The offense gets three receivers against a nickel defense: the ordinary matchup, not the one it wanted. The rule doesn’t care that it was only one player, or that the offense was in a hurry: any offensive substitution before the two-minute warning opens the defense’s window. If the quarterback snaps while the umpire is still over the ball, the first time costs a warning and a replayed down, and after that 15 yards.

Compare Figure 45.4. There the offense kept the same eleven and moved its tight ends into the slots instead of swapping one for a receiver, and the defense was stuck with linebackers on them. Same goal, and the difference is one player crossing the sideline. That is the cost listed earlier, “your own substitutions get harder,” in action: the tired tight end has to stay on the field if the offense wants the freeze, which is why tempo teams prize tight ends who can block and run routes, and why they pick the moments they go fast. (Why sub here at all? Maybe U was gassed, or hurt, or the coach simply wanted his receiver more than he wanted the freeze. Sometimes that’s the right trade.)

If the defense, for its own reasons, doesn’t bring the nickel on, the offense gets exactly what it wanted: H, a receiver, against a linebacker in the slot. That is the throw.

The offense has seen the adjustment. What does it call now, and who is it aimed at?

TipAnswer

A route that uses the sitting defenders’ new habit against them: most likely a pivot (whip) route by a slot receiver, or a crosser that keeps climbing behind them, with the target being the space the sitter just committed to leave. The second-half rule tells each inside defender to stop at five or six yards and take whoever comes into his area. A slot receiver who runs straight at that defender as if he’s crossing, then plants and snaps back the way he came, is working against a man who has just committed to the cross; the defender is standing still and has to turn and chase. Many offenses pair it with a crosser from the other side that, instead of flattening at five yards, keeps climbing to eight or ten, behind the sitters and in front of the deep safety.

This is the call after the call: the offense isn’t trying to beat the first-half man coverage any more, and it isn’t simply running mesh again. It’s punishing the specific way the defense fixed mesh. Expect the defense to answer that, too, perhaps by mixing the man coverage back in so the sitters can’t be read in advance. Another good answer is the running back: a back who releases late on a wheel or rail route up the sideline is running away from defenders who are looking inside for crossers. If you said “run the ball,” that’s reasonable partial credit: two defenders standing at five yards looking for crossers are a step slower to the run, but on third-and-6 most play-callers would rather attack the adjustment directly.

What is the offense trying to do, and when? Then switch sides: you are the defensive coordinator. Do you call a timeout, and what does each choice cost you?

TipAnswer

It’s a sugar huddle, and the offense wants to snap before the defense can get its short-yardage personnel on the field or set its own call: expect a fast snap, with twenty-five or more seconds still on the play clock, and a short-yardage run, most likely a quarterback sneak or an inside run behind the extra tight ends. The huddle lets the offense give a precise call to everyone at once (no signals for the defense to steal) and still get to the line almost immediately. Because the offense already has its heavy personnel on the field from third down and isn’t substituting now, the umpire won’t stand over the ball for the defense: if the defense starts sending in its goal-line players, it risks being caught with twelve on the field.

The defense’s choice. No timeout: play the down with the base players on the field and a simple short-yardage call (everyone low, fill the gaps). It keeps the timeout for the end of the game, but it is defending fourth-and-1 with lighter players than it wants. Timeout: it gets its short-yardage group on and a proper call in, but it spends one of its three second-half timeouts with a quarter to play, and it gives the offense the same minute to think, change its play or bring on its own short-yardage specialists. On a fourth down near midfield the timeout is often worth spending, because the down is worth more than the timeout; skipping it makes sense for a defense that trusts its call. Either way the sugar huddle has already won something: it has forced the defense to choose.

Takeaways

  • Adjusting is constant and small. Between series, the booth’s tallies and the tablet photos meet the coaches and players on the bench, and what comes out is a rule, an alignment, a protection call or a re-ranked call sheet, chosen from plays already practiced.
  • Tempo is a weapon about the rules, not just speed. The defense may match personnel only when the offense substitutes, so an offense that keeps its eleven and goes fast freezes the defense in the wrong players, tires it, and strips away its disguise. The sugar huddle gets the same benefits with a huddle’s clarity.
  • Tempo costs the offense too. Few personnel groups, simple calls, tired players and, above all, its own defense back on the field after a fast three-and-out. Kelly’s Eagles were the fastest offense in the league for three seasons and their defense faced the most plays each year.
  • Tempo works, a little, as a change-up. Fast snaps gain about three hundredths of a point of EPA per play more than slow ones in the same situation, at the moments play-callers choose them. That is why the league tried going fast all game in the mid-2010s and then went back to using it in spots.
  • Defenses answer tempo by being sound rather than clever: substituting at the whistle, one-word calls, rules the players finish themselves, and versatile players who never need to come off. Each has a price. The league polices the rest.
  • The call after the call is choosing the play that beats the defense’s next answer. Watch the series after a team fixes something.
  • “Halftime adjustments” mostly don’t survive the data as a decisive, repeatable skill. First and second halves look the same league-wide, bad halves regress to the mean, being better after halftime doesn’t carry over from season to season, and luck, the scoreboard and who had the ball explain most big swings. Ask what changed.
