# --- markers, labels and arrows on a drawn field ------------------------------------------------
def O(key, pos, d, w, label=""):
    """A kicking-team (blue) player at (d, w): d = yards downfield of the line, w = yards to its right."""
    return Player(key, pos, "off", d, w, label=label)


def D(key, pos, d, w, label=""):
    """A receiving-team (orange) player."""
    return Player(key, pos, "def", d, w, label=label)


def P_(fld, p, d, w):
    """(d, w) in the kicking team's frame -> plot coordinates on this drawn field."""
    x, y = fld.xy(*p.to_abs(d, w))
    return float(x), float(y)


def say(fld, p, d, w, text, color=style.INK_2, fs=6.6, ha="center", weight="normal"):
    """A short on-field label at (d, w)."""
    x, y = P_(fld, p, d, w)
    fld.ax.text(x, y, text, fontsize=fs, color=color, ha=ha, va="center", zorder=9, fontweight=weight,
                linespacing=1.1, bbox=dict(boxstyle="round,pad=0.15", fc=style.SURFACE, ec="none", alpha=0.9))


def lead(fld, p, frm, to, color=style.MUTED):
    """A thin leader line from a label at `frm` to a player at `to`."""
    (x0, y0), (x1, y1) = P_(fld, p, *frm), P_(fld, p, *to)
    fld.ax.plot([x0, x1], [y0, y1], color=color, lw=0.7, zorder=7.5)

def shade(fld, p, d0, d1, w0, w1, color, alpha=0.10):

# --- animations: gridiron tracking, with the ball's flight replaced by the kick's --------------
def with_ball(play, ball_fn, t_after=0.6):
    """Simulated tracking (BDB columns) for a kicking play. gridiron's ball model follows a QB;
    here `ball_fn(t)` gives the ball's (d, w): snap, hold or catch, then the kick."""
    df = play.to_tracking(t_after=t_after)
    m = df.side == "ball"
    dw = np.array([ball_fn(t) for t in df.loc[m, "time"]])
    x, y = play.to_abs(dw[:, 0], dw[:, 1])
    df.loc[m, "x"], df.loc[m, "y"] = x, y
    return df


def lerp(a, b, f):
    f = min(max(f, 0.0), 1.0)
    return (a[0] + (b[0] - a[0]) * f, a[1] + (b[1] - a[1]) * f)


def sxy(play, d, w):
    """(d, w) -> plot coordinates inside a gridiron frame strip (vertical view)."""
    x, y = play.to_abs(d, w)
    return float(FIELD_WIDTH - y), float(x)


def strip_text(ax, play, d, w, text, color=style.INK_2, fs=6.0):

def show_strip(df, times, annotate=None, title=None, lateral_pad=4.0):
    """HTML: a scrubbable video of the simulated tracking. PDF: a frame strip at `times`."""
    if PDF:
        fig = frame_strip(df, times=times, title=title, lateral_pad=lateral_pad)
        for ax in fig.axes:          # painted yard numbers land outside the cropped panels; drop them
            for t in list(ax.texts):
                if t.get_text().isdigit():
                    t.remove()
        fig.tight_layout(h_pad=1.6)
        if annotate:
            annotate(fig.axes[:len(times)])
        plt.show()
        return
    anim = animate(df, title=title)
    if shutil.which("ffmpeg"):
        html = anim.to_html5_video(embed_limit=40).replace("<video ", '<video style="max-width:100%;height:auto" ')
    else:
        html = anim.to_jshtml(default_mode="once")
    plt.close(anim._fig)
    display(HTML(html))

def kicking_plays():
    """Field goals, extra points, two-point tries, punts, fourth-down runs and passes, and runs/passes
    from kick formations."""
    out = []
    for s in SEASONS:
        df = nfl().load_pbp([s])
        df = df.filter(pl.col("play_type").is_in(["field_goal", "punt", "extra_point"])
                       | (pl.col("two_point_attempt") == 1)
                       | ((pl.col("down") == 4) & pl.col("play_type").is_in(["run", "pass"]))
                       | pl.col("desc").str.contains("Punt formation|Field Goal formation"))
        out.append(df.select([c for c in KEEP if c in df.columns]).to_pandas())
    return pd.concat(out, ignore_index=True)

The third phase in one minute

Every football team is really three teams. The offense and defense get the airtime; the third, special teams, takes the field for the kicking plays: kickoffs, punts, field goals and extra points. Those were 17.5% of NFL plays in 2025, about one snap in six (the count is in What a Play Really Is), and they are run by their own coach, the special teams coordinator. You met the specialists (kicker, punter, long snapper, holder, returner and gunners) in The Twenty-Two. This chapter puts them to work.

Two kinds of kicks are in this chapter. Both are scrimmage kicks: they start with a snap from the line of scrimmage, like any offensive play, and the offense chooses them, usually on fourth down. The field goal tries to score; the punt gives the ball away on purpose in exchange for field position. (The kickoff, a free kick from a tee that starts each half and follows every score, is a different animal with its own fast-changing rules; it has its own chapter, Kickoffs, Returns, and the Dynamic Kickoff.)

Why should a viewer who cares about schemes care about kicking? Because the kicking game sets the price of everything else. How far out a kicker can be trusted decides where a fourth down becomes a field-goal try instead of a gamble. How far a punter can pin the opponent decides how much a failed fourth-down try costs. Every fourth-down decision you will learn to judge in The Clock, the Two-Minute Drill, and Fourth Down starts with the numbers in this chapter.

The rule of thumb is add 18: from the opponent’s 30, a 48-yarder; from the 36, a 54-yarder. The 18 is the 10-yard end zone (the posts have stood at the back of it, on the end line, since 1974) plus the distance between the line of scrimmage and the spot where the holder sets the ball: about 7 to 8 yards on the field, and 8 by the scorers’ convention, which is where “add 18” comes from. In nflverse play-by-play since 1999, 87% of all field-goal attempts are recorded at exactly the line of scrimmage plus 18.1 That share reflects the scorers’ rule of thumb more than a tape measure: nobody measures where the holder’s finger actually was. College announcers often say “add 17,” because many college holders kneel a yard closer.

Why does the holder kneel so far back? Because of the rushers. A kicked ball rises steeply, but it needs a few yards of travel to climb above the outstretched hands of the defenders at the line. Move the spot closer and the ball is still low when it crosses the line, so it is easier to block. Move it farther back and the ball clears the line easily, but now the snap takes longer, the kick is longer and less accurate, and edge rushers have more time to get around the end to the spot. Seven to eight yards is where a century of trial and error settled.

The goal itself is a crossbar 10 feet off the ground and 18 feet 6 inches wide, with uprights reaching 35 feet above it.2 That width is no accident of design: the NFL’s hash marks are 18 feet 6 inches apart too, so a kick from either hash is lined up directly with one upright and needs only a mild angle. In college, where the hashes are much wider, kickers face real angles from the hash, which is why college teams sometimes waste a down running the ball to the middle of the field before a kick. In the NFL the hash matters much less, though many kickers still have a side they prefer.

What “field goal range” means

Field goal range is the yard line beyond which a coach believes his kicker makes the kick often enough, today, that kicking beats the alternatives. Notice the three parts of that definition. It is a coach’s belief, not a fact painted on the field. It depends on today: the wind, the cold, the roof. And it is relative to the alternatives: going for it on fourth down, or punting. A 55-yard try that goes in six times in ten might be a great call on fourth-and-9 and a poor one on fourth-and-1.

Coaches don’t guess at the first part. Before every game, kickers warm up from progressively longer distances toward both end zones and tell the special teams coordinator their lines: “the 35 going this way, the 31 going the other way, into the wind.” Those yard lines go on the call sheet next to the fourth-down plan, and they are why the coach in the opening scene looked at the flags.

The cost of a miss is the other half. A missed field goal isn’t like a punt. The other team takes over at the spot of the kick, where the holder put the ball down, or at its own 20 if the kick came from inside the 20.3 So a missed 54-yarder from the opponent’s 36 hands the other team the ball at its own 44: you didn’t score, and you gave them a short field. A punt from the same spot would have pinned them somewhere near their 10. Range is the yard line where those two prices balance. (How often is “often enough”? That depends on the score, the clock and the distance to go, and the arithmetic is built in The Clock, the Two-Minute Drill, and Fourth Down.)

One more way a miss can hurt. A field-goal try that falls short in the field of play or the end zone is a live ball, and once a defender touches it the special missed-kick rule no longer applies: it is treated like a punt, and he may catch it and run it back.4 That is why defenses station a returner under the crossbar on very long tries. On November 4, 2007, at Minnesota, San Diego’s Antonio Cromartie caught a missed 57-yarder nine yards deep in his own end zone and returned it 109 yards for a touchdown, at the time the longest play in league history.5

In a video game the kicker’s range is a meter: aim, press, and the leg is exactly as strong on a frozen December night in Buffalo as indoors in September. Why? Several things at once, and nobody can cleanly split the credit:

  • Specialization. Kickers used to be position players who also kicked.   • Competition for jobs. There are only 32 jobs. A kicker who makes 80% gets cut for one who makes 88%, and the pool of candidates (college, other leagues, soccer) has grown.
  • The ball. Through 2024 the special “K-balls” used for kicks were shipped to the officials and handed to each team only an hour before the game, to be prepared with towels and a brush. For 2025 the league let teams receive their kicking balls before the season and break them in like quarterbacks’ balls. Kickers had long complained that a ball fresh out of the box, not yet broken in, felt hard and dead off the foot. Long-range kicking jumped in 2025: 73 made kicks of 55-plus yards, against about 22 a season from 2016 to 2021.8

Wind, cold, altitude and roofs

The ball doesn’t know the score, but it knows the air. Four conditions move range:

  • Wind is the obvious one. A 15-mph wind in the kicker’s face can take five or more yards off his range; at his back it adds them. Crosswinds push the ball sideways, which is why kickers aim “off” the upright in a crosswind. That’s why kickers study the flags on top of the uprights before every kick.
  • Cold costs distance twice. Cold air is denser, so the ball meets more drag, and a cold ball is harder and less lively off the foot. Kickers talk about a cold ball feeling like a rock.
  • Altitude gives distance back. Denver’s stadium sits about a mile above sea level, where the air is roughly 15% thinner, so the ball carries farther. Coaches lean on it a little: in 2010–2025, 7% of field-goal attempts in Denver were from 55 yards or more, against 5% everywhere else.
  • Roofs remove the first two. A dome is warm and still every week.

The operation: snap, hold, kick

A field goal looks like one man’s moment. It is really a three-man relay with a stopwatch on it.

Snap-hold-kick is the name for that relay, and coaches time it from the snap to the moment the foot hits the ball, a number they call operation time. The professional target is about 1.3 seconds.11 Here is what has to happen inside it:

  1. The snap (roughly 0.4 seconds). The long snapper, head down and looking backward between his legs, fires a tight spiral seven or eight yards to the holder’s hands, ideally with the laces arriving in a predictable position.
  2. The hold (about half a second). The holder catches it, puts the point on the ground on his mark, tilts it as his kicker likes and spins the laces to face the goal posts, because a foot that strikes the laces sends the ball off line.
  3. The kick. The kicker has already started. He takes a small jab step as the snap leaves the snapper’s hands and two strides at an angle, so his plant foot lands beside the ball just as it touches down, and swings through.

Why does a tenth of a second matter so much? Because the rushers are running a race against it. An edge rusher who isn’t blocked needs roughly a second and a half to cover the eight or nine yards to the spot where he can get a hand on the ball. A 1.3-second operation beats him; a 1.5-second operation, caused by a snap that arrives high or a holder who bobbles it, doesn’t. That’s why the snapper, holder and kicker practice together every day, and why teams are reluctant to change any one of them in midseason. When a team signs a new long snapper in November, watch its kicker’s numbers.

Two habits of the operation follow from that race. Icing the kicker is the defense’s attempt to break the rhythm: its coach calls a timeout a moment before the snap, so the kicker has to walk back and wait. He can only do it once per kick, because the rules allow a team just one timeout in the same dead-ball period, and a second one called to “freeze” the kicker is unsportsmanlike conduct, a 15-yard foul.12 The whistle often comes too late to stop the play, and the kicker gets a free practice swing. Going the other way, when the snap is bad or the holder fumbles it, there is no point kicking a ball off the ground. The holder shouts a code word (“Fire!” in many systems), the ends and wings release as receivers, and he scrambles to run or throw. It rarely works, but it beats a certain block.

Field-goal protection runs on a few blunt rules, and every one of them has a reason:

  • Tight splits. The linemen stand almost foot to foot, much closer than on a normal play. There is no running lane to open, so there is no reason to leave gaps for rushers.
  • Step inside, protect inside. Each blocker is responsible for the gap to his inside, toward the snapper. The kick goes straight up the middle, so a rusher in the middle has the shortest route to the ball’s flight. Outside rushers have to travel farther and around bodies.
  • Don’t give ground. A blocker who gets pushed back a yard has moved the line of hands a yard closer to the rising ball. Interior rushers don’t need to get through; they need to push the wall back far enough that a raised arm meets the ball. Big defensive linemen on block units try to do exactly that: collapse the middle and reach.
  • Wings take the edge. The two wings set at an angle and meet the first rusher off each edge. Any extra rusher outside them is left free, because by the time he arrives the ball should be gone. That is the bet the 1.3-second operation is paying for.

The block unit on the other side has its own playbook: overload one side with as many rushers as the rules allow, crash a big man into the snapper’s gap, time the snap, and send the longest-armed players at the block point. The rules put limits on it, mostly to protect players who can’t defend themselves. The snapper is the clearest case: his head is down between his legs at the snap, so the rules treat him as a defenseless player until he has had a chance to protect himself, and no defender within a yard of the line may line up over him.13 On field goals and tries, no more than six defenders may be on the line on either side of the snapper, and nobody may push a teammate into the formation.14 Since 2017 a rusher may not take a running start and leap over the line to block a kick, a play that had become popular and that blockers, rising from their stances, could flip into a dangerous fall.15 Blocks are rare anyway: 1.9% of NFL field-goal attempts from 2021 to 2025. When one happens, check the replay for three things: a low kick (the kicker’s fault), a blocker pushed back (the protection’s fault), or a slow operation (the snap or the hold).

ImportantWhy it exists: the holder

Why not kick off a tee, as on kickoffs? The NFL’s rules don’t allow a tee on a field goal or extra point,16 so the ball has to be held, and the holder becomes a fourth specialist in the operation: usually the punter, who has the best hands among the specialists, already catches long snaps for a living, and practices with the kicker and snapper every day. The trade-off is one more pair of hands that can fail, and one more player who can run or throw on a fake. Both matter later in this chapter.

The extra point, before and after 2015

You met the try after a touchdown in The Game in Fifteen Minutes: kick the extra point for one point or run a two-point conversion from the 2 for two. Mechanically, the extra point is just a short field goal: same snap, same hold, same protection, same block units. What makes it worth a section is the experiment the league ran on it.

For decades the kick was snapped from the 2-yard line, a 20-yard kick. In 2014 kickers made 99.3% of them. A play that succeeds 99 times in 100 isn’t really a play, and the competition committee (the panel of team executives and coaches that proposes rule changes to the owners) wanted the try to involve a decision. The league tested a longer kick in the 2014 preseason, and in May 2015 the owners voted 30–2 to snap the kick from the 15 (a 33-yard kick), leave the two-point try at the 2, and let the defense score two points by returning a blocked kick or a turnover on the try the length of the field.17

  2. The two options became nearly equal. Since 2015 two-point tries have succeeded 48% of the time, worth about 0.95 points per try, against about 0.94 for the kick. When does the scoreboard say otherwise? Count the margin after the touchdown’s six points. If you now trail by 2, a two-pointer ties the game and a kick doesn’t. If you now trail by 5, a two-pointer makes it a 3-point game, so a field goal can tie. The best-known modern call comes when you trail by 14 late and score: you now trail by 8, and analytics favor going for two right away. Make it (about half the time) and the next touchdown plus a kick wins the game in regulation. Miss it and you need a two-pointer after the next touchdown just to tie, but you know that while there is still time to plan for it. Over the two tries, going for two first gives roughly a coin-flip chance of winning without overtime; kicking twice almost guarantees overtime. The punt: field position for sale

A punt gives the ball to the other team on purpose, as far down the field as possible. Downs, Distance, and Field Position explained the trade: you were probably going to lose the ball on fourth down anyway, so you sell possession for roughly 40 yards of field position. Now we look at how the sale is made, because a punt is a three-part play: protect the punter for two seconds, kick the ball where the coverage can reach it, and cover 40-plus yards of field before the returner can do anything with it. Every punt formation is a different answer to one tension: players who stay home to block can’t run downfield to cover.

The punt operation

The punter stands about 14 to 15 yards behind the line. The long snapper’s target is about 0.7 seconds to the punter’s hands; the punter takes a step or two and kicks, and the whole thing, snap to kick, is aimed at about 2.0 seconds. A good NFL punt then hangs in the air about 4.5 to 5 seconds.19 Those three numbers explain almost everything about punt formations. The blockers need to hold up rushers for about two seconds, and the coverage has about four and a half seconds after the kick to get 40 yards downfield.

Hang time (how long the ball is in the air) matters as much as distance. A 50-yard punt that hangs 3.8 seconds gives a returner a head start on the coverage; a 44-yarder that hangs 5 seconds lands with gunners standing next to him. The greatest punter in the sport’s history, Ray Guy of the Oakland Raiders (1973–86), was famous for exactly this: kicks so high that broadcasters timed them, and that made him the first pure punter elected to the Hall of Fame.20

The oldest answer is the tight punt: everyone packed in close to block, like a field-goal unit, then everyone runs. It is nearly unblockable and covers slowly, so today you see it mainly when a team is backed up inside its own 10, where a blocked punt is likely a touchdown or a safety for the other team. Backed up, the end line squeezes the punter too. With the ball at the 2, the back of the end zone is only 12 yards behind the line, so he can’t stand his usual 14 or 15 yards deep; he sets up a step in front of the end line, the snap arrives sooner, and the rushers have less ground to cover. If a snap sails over his head there, his first job is to fall on the ball: a punter tackled in his own end zone, or a ball that skids out of the back of it, is a safety. Late in a game a team sometimes concedes an intentional safety on purpose: two points for the opponent, in exchange for kicking the ball away from safer ground instead of punting from under its own goal posts.

Everywhere else, the NFL lines up in the spread punt (the right of the figure). The two outside men on the line, the gunners, split wide near the sidelines. Five interior linemen (the snapper, two guards and two tackles) stand on the line with a little more space between them than on a field goal. Two wings line up a yard behind the line, just outside the tackles, and one more man, the personal protector, stands five to seven yards deep. He is the punt team’s quarterback: he checks the return team’s alignment, calls the protection (“watch the overload left!”), blocks whoever comes free, and he is the most common ball carrier on a fake punt. Most units block with a zone or “slide” scheme: each lineman takes a gap, and the personal protector picks out the most dangerous rusher and slides the protection toward him, so that the man left unblocked is the one with the longest path to the ball.

Where does “shield” come in? The shield punt (also called the spread punt; the two names are used loosely and overlap) is named for its last line of protection: blockers standing a few yards in front of the punter who wall off any rusher well before the spot where the ball leaves his foot. In the NFL that shield is the two wings and the personal protector, the dotted triangle in the figure. The version with three blockers standing side by side a few yards behind the line, which you’ll see on college Saturdays, grew up in college football, and some NFL teams borrow it. The two levels differ because of one rule.

Under the NFL’s rules, only the two end men on the line (the gunners, as long as they are the widest players on each side) may run downfield before the ball is kicked; everyone else may not go more than a yard past the line until the kick.21 The rule dates from 1974, when the league, trying to revive punt returns, first barred every punt-team player from releasing early and then, after coaches complained, carved out the two men on the ends.22 College has no such restriction, so a college punt team can release its whole line at the snap and trust the shield to stop anyone who gets through. So what does the shield buy an NFL team, whose linemen must wait anyway? Protection with fewer bodies and shorter blocks. With the wings and the personal protector behind them, the linemen don’t have to win their blocks; they only have to slow a rusher for two seconds, then turn and run the instant they hear the foot hit the ball. NFL punt coverage is therefore built around the gunners.

Gunners and vices

The gunner is the punt team’s missile, and the return team’s answer is the vice (also called a jammer): a blocker, usually a cornerback, who lines up across from a gunner and tries to keep him from getting downfield, by jamming him at the line and then running with him and shoving him toward the sideline. Most return teams put two vices on each gunner, one on either side of him, so the gunner can’t simply run around one. That costs the return team four players, which is why the gunner is worth so much: one great gunner ties up two opponents on every punt. (Coaches usually spell it “vise,” like the tool, and strictly they use the word for the two-man bracket; a lone blocker on a gunner is a “jammer” or a “single jam.” The Big Data Bowl files spell it “vises.”)

Watch a gunner on the replay and you’ll see a little wrestling match on the sideline: the gunner trying to swim past one vice (a quick arm-over move) or “stack” him (get even with him and step directly in front of his path, so the vice is stuck behind), the vices trying to run him out of bounds. The vices can be this rough because the usual pass-defense limits are relaxed here: against a punt formation, contact on the end men that would normally be illegal contact or pass interference is allowed before the kick, as long as it isn’t holding.23 A gunner who voluntarily runs out of bounds to avoid a block commits a foul, and one who has been out of bounds, even if he was pushed, may not be the first player to touch the punt when he comes back in.24

  • A directional punt is aimed toward one sideline instead of the middle of the field. It gives up a few yards of distance to put the returner where he has room on only one side, so the coverage can squeeze him with the sideline acting as an extra tackler. Most NFL punts from a team’s own territory are directional in this sense; punters practice hitting the numbers on either side.
  • A coffin-corner punt is the old art of kicking the ball out of bounds just short of the goal line, near the corner. A ball kicked out of bounds is dead where it crosses the sideline, with no return and no touchback.25 The catch is precision: miss by a few yards one way and the ball goes into the end zone; miss the other way and it stays in bounds for the returner.
  • A pooch punt is the modern answer from near midfield: a deliberately short, very high kick, often struck so the ball turns end over end and “checks up” (bounces backward or dies) instead of rolling forward. It lands around the 10, the returner usually lets it go or fair-catches it, and the gunners down it before it reaches the end zone. From the opponent’s 40–50 in 2021–2025, 52% of NFL punts finished inside the 10 and 17% went for touchbacks (1,501 punts).

The pooch has largely replaced the coffin corner. In 2021–2025 only 2.0% of all punts went out of bounds inside the 10, against 6.1% that were downed there. A big reason is the Australian invasion: since Darren Bennett, an Australian rules football player, began punting for the San Diego Chargers in 1995, a steady stream of Australians (many trained by the ProKick Australia program) has brought end-over-end and spinning kicks that make the ball stop where they want it.26 The same tradition gave the NFL the rugby punt: the punter takes the snap, rolls out toward one side like a quarterback, and kicks on the run, usually a low, end-over-end ball that bounces and rolls. It buys time against the rush (the rushers aimed at the spot where a normal punt is kicked), it makes the ball miserable to field, and it carries a threat: if the edge is open, the punter can tuck it and run.

Measuring a punt: gross, net and where it lands

A punter’s gross average is how far his kicks travel from the line of scrimmage. The number that matters for field position is the net punting average: gross distance minus return yards, minus 20 yards for every touchback (because the ball comes back out to the 20), divided by the number of punts. One bookkeeping convention matters: a punt that goes into the end zone is credited with its distance to the goal line, not to wherever it landed. Net is what the punt team, kicker and coverage together, actually bought.

“A longer punt is a better punt.” Not past midfield, and not even always from deep. Judge a punt by where the receiving team starts, not by the number on the graphic.

The return team: return, block, or play it safe

Now flip to the other side. On every punt the receiving team makes a choice before the snap, and the alignment usually gives it away.

The return

Most of the time the return team wants a return, or at least a clean fair catch. Its two biggest jobs are to slow the gunners (the vices) and to hold up the interior coverage long enough to give the returner room. The most recognizable design is the return wall (or wall return): after a token rush to keep the punt team honest, most of the return team peels back toward one sideline and forms a line of blockers running parallel to it, about five yards apart. The returner catches the ball, heads for the wall, and runs up the field behind it while the blockers turn and seal the coverage players inside. A middle return does the opposite: blockers take on the coverage man in their lane and try to open a seam straight up the field.

The returner has the hardest decisions on the field, made in the four seconds the ball hangs: catch it and run, wave for a fair catch, or let it go. A fair catch is his insurance policy. While the ball is in the air he fully extends one arm above his helmet and waves it side to side.28 From then on, nobody may hit him; the ball is dead where he catches it, and he may not run with it. If he waves and then doesn’t catch it, he still can’t be touched while he tries, and he may not block anyone either.

ImportantWhy it exists: the fair catch

A returner looking up at a ball falling out of the sky can’t protect himself, and a coverage player arriving at full speed can hit him the instant it lands in his hands. The fair catch trades his right to advance for his safety. It also makes the punt a strategic choice for the returner: with coverage in his face he gives up a few yards to avoid a big hit and, more importantly, to avoid fumbling. The rules also protect him without a signal: the kicking team must give any receiver an unobstructed chance to catch a kick in the air, which is the next rule.

Kick-catch interference. A returner trying to catch a punt has the right of way to the ball, signal or no signal. A kicking-team player who touches him before or as the ball arrives, or gets in his path, has committed kick-catch interference: 15 yards from the spot of the foul.29 That’s why you see gunners pull up a yard or two short of a returner waiting for a punt, hands out to the side, and pounce only when the ball touches him or the ground. (A related oddity, covered in The Clock, the Two-Minute Drill, and Fourth Down: after a fair catch, the receiving team may try a free fair-catch kick for three points, a curiosity so rare that Cameron Dicker’s 57-yarder for the Chargers in December 2024 was the first one made since 1976.30)

Why returners don’t field punts inside the 10. A punt that lands inside the 10 will often bounce into the end zone for a touchback at the 20. A returner who catches it at the 7 and gets tackled at the 12 has lost eight yards against letting it go. So returners are taught to stay out of the 10 and let the ball decide, which is exactly what the pooch punt exploits: land it short enough that it doesn’t reach the end zone.

The punt block

The second choice is to go after the kick. A punt block call sends extra rushers, often seven or eight, usually overloading one side so that the protection runs out of blockers. The rushers’ target is not the punter. They aim at a spot about a yard in front of him, on the side of his kicking foot, where the ball leaves his foot. Hitting the punter is a foul: running into the kicker costs 5 yards, and roughing him (a harder hit, or contact with his plant leg) costs 15 and an automatic first down, which turns a failed block into a gift.31 There is no foul, though, if the rusher got a hand on the ball before the contact, if he was blocked into the punter, or if the punter ran with the ball out of the pocket without clearly setting himself to kick. That is why a rusher who “whiffs” right at the ball and then collides with the punter often draws no flag. A blocked punt is one of the biggest swings in football, because the ball usually ends up deep in the kicking team’s territory or in the end zone. They are rare: about 10 a season across the whole NFL in 2021–2025. The reason is the operation: a 2.0-second snap-to-kick behind a shield leaves very little time, and a block call costs the return.

Who gets a blocked kick? It depends on which side of the line it ends up. A blocked punt or field goal that stays behind the line is a loose ball that either team may pick up and run with. If the kicking team recovers it there, it is simply still its down: on fourth down it must carry the ball past the line to gain, or the other team takes over where it is stopped. If the ball goes beyond the line, the normal kick rules take over: the receiving team may advance it, the kicking team may not. (Touching the ball while trying to block it at the line doesn’t count as the receiving team touching the kick.)32

Punt safe

The third choice is defensive. When the punt team might fake (fourth-and-short, around midfield, a coach with a reputation, a punt team that showed a fake on film), the return team calls punt safe: it keeps its regular defenders on the field or brings them on, covers every eligible receiver man to man, keeps its front at the line to stop a direct snap, and leaves the returner deep in case the punt comes. Who is eligible on a punt? The same rules you learned in The Twenty-Two: the two ends of the line (here the gunners) and every player lined up off the line (the two wings, the personal protector and the punter) may catch a pass; the snapper, guards and tackles may not. That is six men to cover, which is why the right side of the figure has six dotted lines. Coaches call the linebacker who spies the punter the “rat” or the “spy.” It gives up most of the return, and in exchange makes sure the punt team doesn’t keep the ball. A return team in punt safe looks like a defense: corners in tight on the gunners, linebackers in the box, nobody turning to set a wall.

  1. Out of bounds or at rest. A punt that goes out of bounds is dead where it crossed the sideline, and it’s the receiving team’s ball there. One that rolls to a stop with nobody trying to pick it up is the receiving team’s ball where it stopped.
  2. The end zone. A punt that touches the ground in the end zone, or touches a kicking-team player who is standing on or behind the goal line, is a touchback: ball at the 20. That’s why you see a gunner leap from the field, bat a bouncing ball back out of the end zone while he’s still in the air, and only then land: as long as he hasn’t touched the goal line or the end zone before he touches the ball, the ball is still alive and can be downed short of the goal line.33
  3. First touching. A kicking-team player may not be the first to touch a punt beyond the line. If he does, it is first touching, which is a violation, not a foul: no yardage, the ball stays live, and the receiving team may take the ball at the spot of first touching or at the end of the play, whichever it prefers.34 The ball stays live only while nobody holds it, though. The moment a kicking-team player possesses the punt beyond the line (catches it or falls on it), the ball is dead, and if no receiver had touched it first, it’s the receiving team’s ball right there.35 That is what “downing” a punt is: first touching that ends in possession. The gunner catches the bouncing ball at the 4, play stops, and the receiving team takes the 4. The rule also gives the returner a cushion: once a coverage man has touched the ball, a returner can scoop it up and try to run, and if he is tackled behind the spot of first touching, his team still takes that spot. It isn’t a completely free play: the option is wiped out if possession changes, so a returner who then fumbles to the kicking team can lose the ball.
  4. Muff versus fumble. A muff is touching a loose ball without ever possessing it: the punt bounces off a returner’s hands, or off the leg of a blocker running past. A muffed punt is a live ball, and the kicking team may recover it, but it may not advance it: the ball is dead at the spot of the recovery, and the kicking team has a fresh first down there.36 A fumble is different: the returner caught the ball, possessed it, and then lost it. Then it’s an ordinary loose ball and either team may recover and run with it. The NFL sees around 46 muffed punts a season, and when the kicking team recovers one the momentum swing is enormous.
  5. The receiving team’s own players. A punt that touches any receiving-team player has been touched by the receiving team, even if he never meant to touch it. That is why returners shout a code word (“Peter!” is the classic) to warn teammates to get away from a ball they’re letting go: a punt that clips a blocker’s calf is a muff, and the coverage team can fall on it.
NoteMadden vs. real life

In a video game, a punt that you don’t catch is just a bouncing ball that eventually stops. In real life a loose punt sets off a chain of rules about who touched it first, and the returner is making a risk calculation on every kick: fielding the ball gains yards on average, but a muff turns a routine possession change into a turnover deep in his own territory. That’s why returners who look timid on TV (“Why didn’t he catch it?”) are often making the right call.

Fakes: why they work, and why they’re rare

The kicking team has one more option: don’t kick. A fake punt / fake field goal lines up as if to kick and runs or passes instead. The common versions: a direct snap to the personal protector, who runs through a gap; a punter’s pass to a gunner or wing who sneaks into the open while everybody watches the returner; and, on field goals, the holder standing up to throw or run, since he already has his hands on the ball.

Here’s the puzzle. In 2016–2025 the play-by-play tags about 20 fake punts a season across the whole league, 0.9% of punt-formation snaps, and they picked up a first down 63% of the time.37 Fake field goals have nearly disappeared: about 3 a season tagged in 2016–2025, against about 12 a season in 1999–2005. If fake punts convert more than half the time, why don’t coaches call them more?

  1. They work because they are rare. The conversion rate is high because coaches choose the moment: a look they scouted, a return team turned to set its wall, a short distance. Call fakes every week and return teams play punt safe every week, and the rate collapses. This is game theory in its plainest form: the right number of fakes is small but not zero, so that the other side has to respect the possibility.
  2. The downside is lopsided. A punt from your own 35 gives the opponent the ball somewhere around its own 25, while a failed fake gives it the ball at your 35, 40 yards closer, often with momentum and a home crowd behind it. A coach can survive a punt; he gets second-guessed for a week over a failed fake.
  3. The personnel aren’t built for it. The punt team is backups, linemen and specialists. A punter’s pass and a personal protector’s run are plays the unit practices a few minutes a week.
  4. Analytics took the edge off. Fourth-down models now tell coaches to go for it in the open, with the offense on the field, in many of the situations where a fake used to be the only way to try. Why risk a trick with backups when you can run your best short-yardage play with starters?

Fakes also reveal the deeper logic of the kicking game. Every punt-team formation is an offensive formation, with the same eligible receivers as any other: the two ends of the line and everyone lined up off it. A player wearing a lineman’s number (50–79) can still be one of them, but only if he reports to the referee before the play, who announces it to the defense; that is how a reserve tackle caught a touchdown in the film room below. And every return team is a defense that has to decide what it fears. When a punt team lines up, the return team is reading it, and vice versa, just as on any other down.

Film room

NoteFilm room: Ravens 2012–2024, Justin Tucker and the 66-yard kick

For most of his 13 seasons in Baltimore, Justin Tucker was the standard for accuracy: he made 89.1% of his 468 regular-season field-goal attempts. His most famous kick came on September 26, 2021, at Detroit. Baltimore trailed 17–16 with no timeouts; a fourth-and-19 completion got the ball to the Lions’ 48, and with three seconds left Tucker tried a 66-yarder, two yards longer than any field goal ever made. It hit the crossbar, bounced up and over, and Baltimore won 19–17.38 What to notice: the arithmetic (a snap from the 48 plus 18 yards is 66), the venue (indoors at Ford Field, no wind), and the operation: snapper Nick Moore and holder Sam Koch had to be perfect for a kicker to swing that hard. Tucker made 22 of 30 in 2024. Baltimore released him in May 2025, the league suspended him for ten weeks that summer, and he was still unsigned when the 2026 season began.39

NoteFilm room: Cowboys 2023–2025, Brandon Aubrey and the moving range line

Brandon Aubrey took the long way to the NFL: a first-round pick of Toronto FC in the 2017 MLS draft, then the USFL’s Birmingham Stallions, then a contract with Dallas in July 2023. He made the first 35 field-goal attempts of his NFL career, a record, and in 2024 he made 14 field goals of 50 yards or more (in 17 tries), the most in a season in NFL history.40 By the end of 2025 he had made 6 from 60-plus yards. What to notice: how a kicker like that changes the coach’s decisions. In 2023–2025 35% of Aubrey’s attempts came from 50 yards or more, against 24% for the league as a whole. For Dallas, field-goal range started near the opponent’s 42, so a drive that stalled at midfield still had a reasonable shot at three points, and fourth-and-long near the 40 became a kick instead of a punt.

NoteFilm room: Jaguars 2025, Cam Little goes to 68

On November 2, 2025, at Las Vegas, Jacksonville’s second-year kicker Cam Little ended the first half with a 68-yard field goal, breaking Tucker’s record by two yards (the Jaguars won 30–29 in overtime). In Week 18 he made a 67-yarder at home against Tennessee, the longest ever kicked in an open-air stadium.41 What to notice: the setting of the record (the dome at Allegiant Stadium, no wind) and the season it came in, the first with teams breaking in their own kicking balls. NoteFilm room: Seahawks, NFC Championship (2014 season), the fake that won a conference

On January 18, 2015, Seattle trailed Green Bay 16–0 in the third quarter of the NFC Championship Game and lined up for a field goal at the Packers’ 19. Holder Jon Ryan, the punter, took the snap, sprinted to his left, away from the rush, and threw a 19-yard touchdown pass to Garry Gilliam, a rookie offensive tackle wearing No. 79 who was playing the wing and slipped out of the formation unnoticed.42 Seattle went on to win in overtime. NoteFilm room: Dolphins and Colts, two direct-snap fake punts

The fake punt in this chapter’s figure, a direct snap to an up-back running through the middle, is the bread-and-butter version, and it works when the return team isn’t looking for it. In the AFC Wild Card game at Pittsburgh on January 8, 2017, Miami, down 17 in the second quarter, faced fourth-and-1 at its own 41. The snap went straight to safety Michael Thomas, who ran up the middle for three yards and a first down.43 For the opposite lesson, find the Colts’ fake punt against New England on October 18, 2015. Most of the punt team shifted to one side; wide receiver Griff Whalen became the snapper and safety Colt Anderson lined up behind him. The play was meant only to draw the Patriots offside, but Whalen snapped the ball anyway, and Anderson was dropped for a one-yard loss (with an illegal-formation flag the Patriots declined). It is remembered as one of the worst fakes ever run.44 What to notice: a fake needs surprise and a sound formation; the Dolphins had both, the Colts had neither.

Watch for it on Sunday

TipWatch for it on Sunday: the range finder

On every fourth down, call it before the offense lines up: go, punt or kick.

  • Do the arithmetic. Line of scrimmage plus 18 is the kick. Under 40 yards: a kick is nearly automatic. 50–55: depends on the kicker and the day. Beyond 58: only for a special leg, a roof or thin air, or the end of a half.
  • Check the conditions. Roof open or closed? Flags on the uprights (and which way they blow for this end)? A cold night? Did the broadcast show the kicker’s warmup line?
  • Read the distance and the score. Fourth-and-1 at the opponent’s 38 is a go in most modern playbooks; fourth-and-8 at the same spot is a kick for a good kicker, a punt for a shaky one.
  • When the punt team trots out, look at the return team. Two vices on each gunner and a deep returner: a return. Seven or eight on the line, crowding one side: a block. Defenders on the field, corners in tight on the gunners, no one turning back: punt safe, they fear a fake.
  • Watch where the punter aims. From midfield, a high pooch toward the 10 and gunners racing to the spot. From deep, a directional kick to the numbers.
  • Then watch the returner’s hands. One arm waving above the helmet is a fair catch. A returner backing away from a ball inside his 10 is letting it go for a touchback. Listen for the officials sorting out who touched it first.

It’s a 55-yarder (37 + 18), four yards beyond this kicker’s line into the wind: punt, or go for it, but don’t kick. The chapter’s weather numbers point the same way. Between the other two options it’s close, and teams differ. NFL offenses converted 39% of fourth-and-6 plays in 2016–2025, and a failed try hands the opponent the ball at its own 37. A punt from the 37 is the pooch scenario from the heatmap: into the wind it won’t carry into the end zone, so the punter can aim high at the 10 and the gunners can down it. A conservative coach punts and plays for field position in a tied game; an aggressive one goes for it. The kick is the one choice the conditions rule out.

Return, block or safe? And what should the punt team’s personal protector do about it?

TipAnswer

A punt block, overloading the punt team’s right side. The tells: eight on the line instead of the usual six, five of them on one side, and only one vice per gunner (strictly, a single “jammer”), because the return team needed the extra bodies up front and doesn’t care much about the coverage if it gets a hand on the ball. A team setting up a return would have two vices on each gunner and players ready to peel back for a wall; a punt-safe team would have defenders, not rushers, and nobody committing this many players to one side. (A side note on the rules: the limit of six defenders on one side of the snapper applies only to field goals and tries, not to punts.45) Eight on the line is a strong tell, not a guarantee: return teams sometimes show exactly this overload and then peel back into a return at the snap.

The personal protector’s job is exactly this moment. Before the snap he calls out the overload and slides the protection toward it, so the shield (the wings and the PP himself) and the right-side blockers take the inside rushers and the outermost man is left with the longest path. The punter may also take a slightly quicker drop or angle his kick away from the overload. The trade for the return team: with one vice per gunner, a punt that gets off cleanly will meet gunners arriving fast, so expect a fair catch. Fourth-and-9 from the 31 isn’t a fake situation, so the punt team just punts, quickly.

Whose ball is it, and where?

TipAnswer

The kicking team’s ball at the 7: first-and-goal. The ball touched a receiving-team player, so the receiving team “touched it first,” even though it was an accident and even though the returner never wanted it. That makes it a muff: the receivers touched it without possessing it. A muffed punt is a live ball and the kicking team may recover it. It may not advance it, so the ball is dead where the gunner fell on it, and the kicking team has a new set of downs there.

That’s why the returner should have been shouting his warning word (“Peter!”) as soon as he decided to let the ball go: every teammate near a bouncing punt is a liability. Order matters, too. If the ball had missed V and the gunner had fallen on it at the 7 untouched by any receiver, the kicking team would not have kept it: that is first touching (the gunner simply downed it), and it’s the receiving team’s ball at the 7. And if it had bounced into the end zone without touching anyone, it would have been a touchback at the 20.

Takeaways

  • Add 18. A field goal is the line of scrimmage plus about 8 yards to the holder and 10 for the end zone. Field goal range is a coach’s judgment about this kicker today, weighed against going for it and punting, and a miss costs the spot of the kick.
  • Kickers are far better than they were, so range has moved out about five yards in a generation. Wind and cold shorten it; altitude and roofs lengthen it. The data understates bad weather’s effect, because coaches don’t try the kicks they expect to miss.
  • A field goal is a 1.3-second snap-hold-kick operation behind a wall that steps inside and must not give ground. Block units overload, crash the middle and aim at the block point; the rules ban leaping, covering the snapper and more than six rushers on one side.
  • The 2015 extra-point change turned a 99% kick into a roughly 94% one and made the kick and the two-point try worth almost the same, so two-point tries doubled and the choice now follows the score.
  • A punt is protect, kick, cover. The NFL’s base is the spread punt, with the wings and personal protector forming the shield in front of the punter; only the gunners may release before the kick, so two vices on each gunner is the return team’s first priority. Judge punts by net and by where the opponent starts, not by gross distance.
  • The return team chooses before the snap: return, block or safe. The rules of the loose ball (fair catch, touchback, first touching, muff versus fumble, kick-catch interference) all follow from one question: who touched it first? Fake punts succeed more often than not precisely because they are rare.
