def O(key, pos, d, w, label=None):
    return Player(key, pos, "off", d, w, label=key if label is None else label)


def D(key, pos, d, w, label):
    return Player(key, pos, "def", d, w, label=label)


def line(labels=True):
    """The five offensive linemen. labels=False keeps only the C (wide figures overprint LT/LG/RG/RT)."""
    keys = ["LT", "LG", "C", "RG", "RT"]
    return [O(k, k, -0.7, (i - 2) * SPLIT, k if (labels or k == "C") else "") for i, k in enumerate(keys)]


def spread(labels=False, rb_w=-1.6):
    """11 personnel, shotgun 2x2: X and Z on the line outside, H (slot) and Y (tight end flexed as a
    slot) a step off it, the back beside the quarterback. Seven on the line: five linemen, X and Z."""
    return line(labels) + [O("QB", "QB", -5.0, 0.0, "Q"), O("RB", "RB", -5.0, rb_w, "R"),
                           O("X", "WR", -0.7, -15.0), O("H", "WR", -1.5, -9.0),
                           O("Y", "TE", -1.5, 9.0), O("Z", "WR", -0.7, 15.0)]


def nickel(**moves):
    """A 4-2-5 nickel against the 2x2: ends outside the tackles, a 1-technique on the center's left
    shoulder, a 3-technique outside the right guard, Mike and Will at 4.5 yards, the nickel ($) over H,
    corners 7 off, one deep safety (FS) and the strong safety (SS) at 8 over Y. `moves` = {key: (d, w)}."""
    dfn = [D("LDE", "DE", 1.0, -5.5, "E"), D("LDT", "DT", 1.0, -0.7, "T"), D("RDT", "DT", 1.0, 2.35, "T"),
           D("RDE", "DE", 1.0, 5.5, "E"), D("WILL", "WILL", 4.5, -2.4, "W"), D("MIKE", "MIKE", 4.5, 1.6, "M"),
           D("NB", "NB", 5.0, -8.5, "$"), D("LCB", "CB", 7.0, -15.5, "CB"), D("RCB", "CB", 7.0, 15.5, "CB"),
           D("FS", "FS", 13.0, 0.0, "FS"), D("SS", "SS", 8.0, 8.0, "SS")]
    return [p.moved(d=moves[p.key][0], w=moves[p.key][1]) if p.key in moves else p for p in dfn]


def to(p, key, end, kind="path", via=(), delay=0.0, speed=None):
    """Move `key` through absolute (d, w) points `via` to `end` as a block, rush, drop-style path or run."""
    start = np.array(p._cur(key))
    pts = [tuple(np.array(q) - start) for q in (*via, end)]
    if kind == "block":
        return p.block(key, pts=pts, delay=delay, speed=speed)
    if kind == "rush":
        return p.rush(key, pts=pts, delay=delay, speed=speed)
    if kind == "route":
        return p.route(key, pts, frame="field", delay=delay, speed=speed)
    return {"path": p.path, "run": p.run}[kind](key, pts, delay=delay, speed=speed)


def drop_to(p, key, end, via=(), delay=0.0, speed=None):
    """A zone drop through absolute (d, w) points (gridiron's .drop goes in a straight line only)."""
    start = p._cur(key)
    path = [tuple(start)] + [tuple(map(float, q)) for q in (*via, end)]
    return p._add("drop", key, path, speed, delay)


def timed(p, key, way, kind="path"):
    """Script a player: `way` = [(t, d, w), ...] absolute spots reached at time t (s after the snap).
    Standing still between two times is allowed."""
    cur = np.array(p._cur(key), dtype=float)
    tc, wait = 0.0, 0.0
    for t, d, w in way:
        nxt = np.array([d, w], dtype=float)
        dist = float(np.hypot(*(nxt - cur)))
        if dist < 1e-3:
            wait += t - tc
        else:
            p._add(kind, key, [tuple(cur), tuple(nxt)], speed=dist / (t - tc), delay=wait)
            wait = 0.0
            cur = nxt
        tc = t
    return p


def _P(fld, p, d, w):
    return fld.xy(*p.to_abs(d, w))


def say(fld, p, d, w, text, color=style.INK_2, fontsize=7.0, ha="center", weight="normal", box=True):
    """A short on-field label at (d, w)."""

    fld.ax.text(x, y, text, fontsize=fontsize, color=color, ha=ha, va="center", zorder=12, fontweight=weight,

def point_to(fld, p, text_at, target, text, color=style.INK_2, fontsize=6.8, shrink=7, ha="center"):

def dashed(fld, p, pts, color=RED, lw=1.8, dash=(0, (3, 1.8)), z=6):
    """A dashed arrow through absolute (d, w) points: a free rusher's path."""
    xs, ys = _P(fld, p, np.array([q[0] for q in pts]), np.array([q[1] for q in pts]))
    xs, ys = np.asarray(xs, float), np.asarray(ys, float)
    fld.ax.plot(xs, ys, color=color, lw=lw, ls=dash, zorder=z)
    fld.ax.add_patch(FancyArrowPatch((xs[-2] + 0.6 * (xs[-1] - xs[-2]), ys[-2] + 0.6 * (ys[-1] - ys[-2])),
                                     (xs[-1], ys[-1]), arrowstyle="-|>", mutation_scale=11, color=color,
                                     lw=lw, zorder=z))


def hole(fld, p, center, size, text=None, fontsize=6.6, text_at=None):
    """A red, hatched patch: a soft spot the coverage cannot reach in time."""
    cx, cy = _P(fld, p, *center)
    fld.ax.add_patch(Ellipse((float(cx), float(cy)), size[1], size[0], fc=RED, alpha=0.30, ec="none", zorder=2.2))
    fld.ax.add_patch(Ellipse((float(cx), float(cy)), size[1], size[0], fc="none", ec=RED, lw=1.1,
                             ls=(0, (2, 1.5)), zorder=2.3))
    if text:
        tx, ty = _P(fld, p, *(text_at or center))
        fld.ax.text(float(tx), float(ty), text, fontsize=fontsize, color=RED, ha="center", va="center",
                    zorder=12, fontweight="bold",
                    bbox=dict(boxstyle="round,pad=0.15", fc=style.SURFACE, ec="none", alpha=0.85))


def zone_patch(fld, p, center, size, color=style.DEFENSE, alpha=0.10):
    cx, cy = _P(fld, p, *center)
    fld.ax.add_patch(Ellipse((float(cx), float(cy)), size[1], size[0], fc=color, alpha=alpha, ec=color, lw=0.8,
                             ls=(0, (3, 2)), zorder=2))


def no_numbers(ax):
    """Remove the painted yard-line numbers (they sit where wide receivers and safeties stand)."""
    for t in list(ax.texts):
        if t.get_alpha() == 0.55:
            t.remove()


def clean(ax):
    """Box-zoomed diagrams: drop hash ticks and yard numbers, clip text to the window."""
    for ln in list(ax.lines):
        xs, ys = ln.get_data()
        if len(xs) == 2 and ln.get_linewidth() == 0.5 and np.hypot(xs[1] - xs[0], ys[1] - ys[0]) < 1.0:
            ln.remove()
    no_numbers(ax)


def ring(fld, p, d, w, color=PURPLE, r=1.2, lw=1.8):
    x, y = _P(fld, p, d, w)
    fld.ax.add_patch(Circle((float(x), float(y)), r, fc="none", ec=color, lw=lw, zorder=7.5))


def ghost_marker(ax, x, y, label, ppy, color=style.DEFENSE):
    """A hollow, dashed defender: where he stood before the snap."""
    ax.add_patch(RegularPolygon((x, y), 4, radius=MARKER_R * 1.25, orientation=np.pi / 4,
                                facecolor="none", edgecolor=color, lw=1.1, ls=(0, (2, 1.2)), zorder=6))
    if label:
        ax.text(x, y, label, ha="center", va="center", color=color, fontweight="bold", zorder=6,
                fontsize=max(4.0, min(8.0, ppy * 0.7)))

def snapshot(play, t, ax=None, ghosts=(), ghost_t=-0.6, lateral=(-21, 21), window=(-8.5, 17.5),
             width_in=6.5, notes=(), rings=(), holes=(), trail_from=None):
    """Everyone where he is at time t, with faint trails since `trail_from` (default: the first frame),
    hollow markers where `ghosts` stood at `ghost_t`, purple rings on `rings` (droppers) and red
    `holes` [(center, size, text)]."""
    pos = {k: play.position(k, t) for k in play.players}
    moved = [pl.moved(d=float(pos[k][0]), w=float(pos[k][1])) for k, pl in play.players.items()]
    q = Play([m for m in moved if m.side == "off"], [m for m in moved if m.side == "def"],
             los=play.los, ball_y=play.ball_y, to_go=play.to_go)
    ringed = {k for k, _ in rings}
    q.highlights = {k: c for k, c in play.highlights.items() if t > 0 or k not in ringed}
    fld = q.draw(ax=ax, lateral=lateral, window=window, width_in=width_in, title=False)
    ax, ppy = fld.ax, fld.pts_per_yard()
    clean(ax)
    P = lambda d, w: fld.xy(*play.to_abs(d, w))  # noqa: E731
    for c, s, txt in holes:
        hole(fld, play, c, s, txt, fontsize=5.6)
    start = trail_from if trail_from is not None else play._compile()[1]
    for k, pl in play.players.items():
        if pl.side == "off" and pl.pos in ("LT", "LG", "C", "RG", "RT"):
            continue
        pts = np.array([play.position(k, tt) for tt in np.linspace(start, t, 30)])
        if np.ptp(pts[:, 0]) + np.ptp(pts[:, 1]) < 0.3:
            continue
        xs, ys = P(pts[:, 0], pts[:, 1])
        if pl.side == "off":
            ax.plot(xs, ys, color=style.OFFENSE, lw=0.9, alpha=0.45, zorder=3)
        else:
            ax.plot(xs, ys, color=style.DEFENSE, lw=1.1, ls=(0, (3, 2)), zorder=3)
    if t > ghost_t + 0.05:
        for k in ghosts:
            x0, y0 = P(*play.position(k, ghost_t))
            ghost_marker(ax, float(x0), float(y0), play.players[k].label, ppy)
    for k, col in (rings if t > 0 else ()):      # a dropper is only a dropper once he drops
        x0, y0 = P(*play.position(k, t))
        ax.add_patch(Circle((float(x0), float(y0)), MARKER_R * 1.75, fc="none", ec=col, lw=1.6, zorder=7.5))
    bx, by = P(*play.ball_position(t))
    thrown = False
    for kind, _, t_rel in play.ball_events:
        if kind == "pass" and t > t_rel:
            thrown = True
            sx, sy = P(*play.ball_position(t_rel))
            ax.add_patch(FancyArrowPatch((float(sx), float(sy)), (float(bx), float(by)), arrowstyle="-|>",
                                         mutation_scale=8, color=style.BALL, lw=1.3, ls=(0, (2, 1.5)), zorder=9))
    if thrown or t <= 0:          # in the quarterback's hands the ball would hide his label
        draw_marker(ax, float(bx), float(by), "ball", "", ppy)
    for note in notes:              # (text, (d, w)) or (text, (d, w), (d, w) of the player it names)
        text, (d, w) = note[0], note[1]
        x, y = P(d, w)
        box = dict(boxstyle="round,pad=0.2", fc=style.SURFACE, ec=style.BASELINE, lw=0.5)
        if len(note) > 2:
            xp, yp = P(*note[2])
            ax.annotate(text, (float(xp), float(yp)), xytext=(float(x), float(y)), fontsize=6.0, color=style.INK,
                        ha="center", va="center", zorder=13, bbox=box,
                        arrowprops=dict(arrowstyle="-", color=style.INK_2, lw=0.7, shrinkA=1, shrinkB=5))
        else:
            ax.text(float(x), float(y), text, fontsize=6.0, color=style.INK, ha="center", va="center", zorder=13,
                    bbox=box)
    return fld


def strip(play, times, labels, notes_at=None, holes_at=None, ncols=2, width_in=6.5, lateral=(-21, 21),
          window=(-8.5, 17.5), legend=None, **kw):
    """The print version of an animation: panels at chosen moments of one simulated play."""
    n = len(times)
    nrows = int(np.ceil(n / ncols))
    pw = width_in / ncols
    h = nrows * (pw * (window[1] - window[0]) / abs(lateral[1] - lateral[0]) + 0.34)
    fig, axes = plt.subplots(nrows, ncols, figsize=(width_in, h), squeeze=False)
    for ax in axes.flat[n:]:
        ax.axis("off")
    for i, (ax, t, lab) in enumerate(zip(axes.flat, times, labels)):
        snapshot(play, t, ax=ax, lateral=lateral, window=window, notes=(notes_at or {}).get(i, ()),
                 holes=(holes_at or {}).get(i, ()), **kw)
        ax.set_title(lab, fontsize=8.2, color=style.INK, loc="left")
    fig.tight_layout(h_pad=0.6, w_pad=0.4)
    if legend:
        fig.legend(handles=legend, loc="upper center", bbox_to_anchor=(0.5, 0.0), ncol=2, fontsize=7.3,
                   frameon=False)
    return fig

                         "epa", "down", "ydstogo"])
DB = (_pl[(_pl["qb_dropback"] == 1) & (_pl["season_type"] == "REG")]
      .merge(_ftn, left_on=["game_id", "play_id"], right_on=["nflverse_game_id", "nflverse_play_id"], how="inner")
      .merge(_part, left_on=["game_id", "play_id"], right_on=["nflverse_game_id", "play_id"], how="left"))
DB = DB.dropna(subset=["n_pass_rushers", "n_blitzers"])
# Pressure includes sacks (FTN fills was_pressure for 2023-2025)
DB["press"] = DB["was_pressure"].astype(float).fillna(0).gt(0) | DB["sack"].eq(1)
# FTN's "blitzer" behaves like a rusher who is not one of the usual linemen and edge rushers (an off-ball
# linebacker or a defensive back). Four rushers with at least one blitzer means somebody who usually rushes
# did not: the closest thing to a simulated-pressure flag in public data.
DB["rush"] = np.select(
    [DB["n_pass_rushers"] <= 3, (DB["n_pass_rushers"] == 4) & (DB["n_blitzers"] == 0),
     (DB["n_pass_rushers"] == 4) & (DB["n_blitzers"] >= 1), DB["n_pass_rushers"] >= 5],
    ["three", "four", "sim", "blitz"], "other")
F = DB[DB["season"] >= 2023].copy()          # FTN-charted participation seasons (one source)
F["man"] = F["defense_man_zone_type"].eq("MAN_COVERAGE")
F["zone"] = F["defense_man_zone_type"].eq("ZONE_COVERAGE")


def _summ(g):
    lab = g[g["man"] | g["zone"]]
    return pd.Series({"n": len(g), "press": g["press"].mean(), "sack": g["sack"].mean(),
                      "epa": g["epa"].mean(), "epa_se": g["epa"].sem(), "man": lab["man"].mean()})

The blitzer’s dilemma

Start with the problem the zone blitz was invented to solve. A defense that wants to hit the quarterback has two ways to do it. It can win with four, beating blockers one on one (The Pass Rush), which needs great rushers and time. Or it can add rushers (Blitzing): send five or six, more than the offense can block, and get someone there before the ball is out.

The second way has a bill attached, and the bill is paid in coverage. Eleven defenders, minus six rushers, leaves five to cover five eligible receivers. With nobody to spare, the defense almost always plays man coverage behind a big blitz: each cover man takes one receiver, with one deep safety at most. That is the Cover 1 or Cover 0 blitz of the previous chapter, and it has three weaknesses offenses have known about for fifty years.

  1. The hot throw. A blitzer is a defender who isn’t covering, so the offense throws into the space he left before he arrives, and the receiver’s only defender is the man who just blitzed.
  2. One missed tackle is a touchdown. With no deep safety, a slant caught in stride can go the distance.
  3. Man coverage needs men who can cover. A linebacker on a fast slot receiver is a matchup offenses hunt.

So the question that drove a generation of defensive coaches was: can we get the confusion and the pressure of a blitz without playing man coverage behind it? Bill Arnsparger, the Dolphins’ defensive coordinator, had a two-word answer that Dick LeBeau would carry into his own career: safe pressure.

ImportantWhy it exists

A blitz trades coverage for pressure; the zone blitz tries to break the trade. A protection assigns blockers to the defenders most likely to rush, so what matters is not how many rush but which. Rush one it wasn’t counting, take away one it was, and the defense can create a free rusher or a mismatch with the same number of rushers, while everyone else plays a sound zone behind it.

The zone blitz: rush one they didn’t count, drop one they did

A zone blitz is a pressure that sends a defender the offense does not expect to rush (a linebacker or a defensive back) and drops one it does expect to rush (usually a defensive lineman) into a short zone, so the defense pressures the quarterback and still plays zone coverage behind the rush. That lineman-turned-coverage-player is the scheme’s signature, and coaches call him a dropper: a defensive lineman, or an edge rusher who normally rushes, who drops into a short zone at the snap. “Zone blitz” is the family name for any pressure built on that exchange; its two main forms, below, are the fire zone (five rush) and simulated pressure or the creeper (four rush). The opening play was a fire zone: five came, the end dropped.

On first hearing it sounds absurd. Why take a 290-pound defensive end, the man you pay to rush, and ask him to cover a receiver? There are three answers, and they explain the whole family.

He isn’t covering a receiver. He is standing in a lane. The dropper’s job is almost never “run with that slot receiver.” It’s “get five or six yards deep, into the spot where the quarterback’s hot throw is supposed to go, and turn around.” Why would the ball go there? Because of how a hot throw is really triggered. Pass Protection taught it as arithmetic: more rushers than blockers. But nobody can count eleven defenders in the half-second after the snap, so the arithmetic is done before the snap and turned into a rule about one defender: “we can’t block the Will if everyone comes, so if the Will comes, the ball goes to the slot, now.” Coaches say the quarterback is hot off the Will. At the snap his eyes go to that one man, and if he comes the ball is out in well under a second, usually before the quarterback could see a lineman on the other side turn and drop. Receivers have a version of the same rule, the sight adjust: a slot receiver whose defender blitzes cuts his route short and turns for the ball.

Both rules react to who rushes, not how many. Send the key man, drop a lineman into the space he left, and the defense gets exactly the throw it wants into a window it has just refilled. A slow, big man can make that play, because the ball is coming to him. If no hot throw comes, the dropper holds his spot and breaks on whatever enters it; the danger then is a receiver who settles behind him, which is where the holes later in this chapter come from.

His absence is the point. The offensive tackle across from him was counting on blocking him. When he drops, that tackle has nobody, and a blocker with nobody to block is a blocker the defense has taken out of the play. Meanwhile someone the protection didn’t count is coming from somewhere else.

The coverage stays zone. Because the defense swapped rushers instead of adding them, it still has six or seven players in coverage, watching the quarterback with deep help behind them.

What makes it work is who the five are. The protection was built to block the three down linemen and the two outside linebackers, the five men who usually rush. Instead, one of those (the strong-side end) drops, a second (the strong outside linebacker) drops, and the two replacements (the Mike and the strong safety) come from places the blocking rules may assign to a back, or to nobody. Extra blockers don’t help if no blocker’s rule includes the man who comes: an unassigned safety is free against seven blockers just as he is against five, and the quarterback has to throw hot off him or take the hit.

CautionCommon misconception

“A zone blitz is a blitz with zone coverage.” Close, but it misses the key move. Plenty of five-man pressures play zone behind them without dropping anybody unexpected. What makes a zone blitz a zone blitz is the exchange: a player the offense expects to rush drops, and a player it expects to cover rushes. Take away the exchange and you have a blitz with thin zone coverage behind it, which is the worst of both worlds: fewer cover men and rushers exactly where the protection expected them.

What a zone blitz gives up

Nothing is free, and a coach calling one knows exactly what he’s betting.

  • A pass rusher. The dropper is usually a good rusher, now jogging backward. If the replacement rusher is a 205-pound safety and the protection picks him up anyway, the defense has traded a good rusher for a worse one.
  • Underneath space. Three underneath defenders can’t cover the width of the field (more than 53 yards) the way four or five can. There are always gaps between them, and the dropper, slow and starting from the line, is the slowest to get to his spot.
  • Run defense. Linemen dropping and linebackers flying through gaps at the snap is a gamble against a run, because the run fits (Run Fits and Box Math) are all reassigned. That is why zone blitzes are mostly a passing-down call, though defenses also use them on early downs as run-and-pass pressures: the line slants into one gap, a linebacker comes through the next, and that wrecks zone-run blocking while three deep still guards against play-action.
  • Time to install. Every exchange is homework: an end learning a hook drop, a safety learning to take on a guard.

The fire zone: three under, three deep

The most common zone blitz has its own name. A fire zone is a five-man pressure with three defenders deep, each guarding a third of the field, and three underneath: a seam-flat defender on each side and one in the middle, usually with a defensive lineman as one of the three. Coaches also call it a 3-under 3-deep pressure, or “3 under, 3 deep,” which tells you its whole structure in four words.1

Why three deep? Because the deep thirds are where a mistake costs a touchdown, and the fire zone refuses that risk, the opposite of a Cover 0 blitz. It pays underneath, with three defenders doing the work of four or five, and bets that the rush will force the ball out before receivers find the gaps, so the throws it gives up are short ones in front of defenders who can rally to tackle.

Who plays the three underneath spots? The two outside ones are seam-flat players, and here is why the match version of Cover 3 matters (Cover 3). With only three underneath, nobody else can stop a slot receiver from running straight up the seam, between the deep thirds. So each seam-flat defender’s rule is: wall the #2 receiver’s vertical route (get in his path so he has to go around), carry him to about 10 to 12 yards until a deep defender can take over, and only then sink under the outside receiver or squeeze the flat if that receiver breaks out. That is why fire zones struggle against seam routes and do well against quick outs.

The middle player sits in the short middle in front of the deep middle safety, where the quick inside throw against a blitz wants to go; if the back releases, or a receiver crosses his face, he takes him. Coaches often call this spot the “hole” or “hot hole.” In this chapter we’ll call it the middle hook and keep the word hole for the soft spots a coverage can’t reach. A dropping tackle, who starts in the middle, usually takes the middle hook itself. A dropping end, who starts wider, usually works to the inside window on his own side, which is exactly where a slot receiver’s hot slant goes.

Count to eleven

Here is the most useful bit of arithmetic in this part of the book. Every defense has eleven players, and every one of them is a rusher, an underneath cover man, or a deep cover man. Fix any two of those numbers and the third is decided.

Call Rush Under Deep Behind it
Four-man rush, Cover 3 4 4 3 zone
Four-man rush, Cover 2 4 5 2 zone
Four-man rush, quarters (Cover 4) 4 3 4 zone (the safeties also fit the run)
Fire zone 5 3 3 zone (three under, three deep)

Before the snap you can usually count the deep defenders, the safeties 12 or more yards deep and any corner well off his receiver. That count caps everything else. A defense showing three deep players can rush at most five and still have three underneath; showing two deep safeties and two off corners (Cover 2 or quarters), it is very unlikely to bring six. Two-high doesn’t rule out a fire zone, though: one safety can rotate down at the snap while the other spins to the middle, just as in Disguise, Rotation, and Split-Field Coverage. And after the snap, counting rushers and deep defenders tells you what you just watched: five rush and three deep is a fire zone; four rush, with somebody unexpected among them, is the simulated pressure coming up below.

Where the holes are

Every fire zone has the same basic weakness: three underneath defenders spread across the field leave gaps between and behind them. But which gaps are biggest depends on who drops, because the holes live next to the slowest dropper. A defensive end who starts on the line of scrimmage can get five or six yards deep before the quarterback throws a quick pass; he cannot get to 12 yards and he cannot get to the sideline. ImportantWhy it exists: the fire zone’s bargain

Three deep makes the worst case a completed short pass rather than a long touchdown, and a dropper in the hot lane can turn the quick throw into an interception. The price, beyond the list above, sits underneath: three defenders for the width of the field, the slowest of them starting on the line of scrimmage, and a hole beside him for any quarterback who reads it after the snap.

Why the protection loses: rules, not people

Now the contest from the offense’s side, because the zone blitz is aimed less at players than at the rules the protection uses. Before the snap the offense has to make three decisions, all on the basis of what the defense shows (Pass Protection):

  1. Mike ID. The quarterback or center names one defender as the “Mike,” the reference point every blocker counts from.
  2. The slide. The line slides toward the biggest threat: the side with more possible rushers, or the most dangerous inside rusher. The man the slide leaves behind belongs to the back.
  3. The hot. If more defenders come than the protection can block, somebody is free. The call names the defender the quarterback is hot off and the receiver who gets the ball if that man comes.

Each of those decisions is a prediction about who will rush, made from where defenders stand. And that is the opening the defense attacks with a pressure look: a pre-snap alignment that shows more possible rushers than the offense can comfortably block (linebackers walked up into the A gaps, a safety or nickel creeping toward the line), whether or not they will rush. A pressure look’s job isn’t to rush. It is to make the offense commit its slide, its back and its hot throw before the snap, so that the defense can bring pressure from wherever the commitments aren’t.

Put the figure in motion and the contest has a rhythm you will see on nearly every third-and-long. The defense shows six or seven possible rushers, more than six blockers can handle, so the offense has to guess which of them matter. The offense commits: the three decisions above, plus any check, such as keeping a tight end in or changing a route so a hot throw is ready on the dangerous side. The defense answers at the snap: the threatening side drops, the quiet side comes, linemen block air, and the rusher nobody counted is free or facing a back instead of a lineman. The quarterback has about a second. If he sees it, he throws away from the droppers or slides in the pocket and finds the hole. If he doesn’t, he is sacked from a side he thought was safe, and the broadcast blames a lineman who did exactly what his rules told him.

NoteMadden vs. real life

In Madden the computer’s quarterback simply throws to whoever is open. A real offense agrees on its protection out loud, in a few seconds, and then every player follows his rule even when it is wrong for the rush that comes. A zone blitz is aimed at those rules, not at the receivers, which is why the same call can be a sack against one offense and a 15-yard completion against another whose rules differ.

Offenses fight back with the same weapons they use against disguise (Disguise, Rotation, and Split-Field Coverage). They change the Mike late to move the slide. They keep the back in with a dual read, checking two defenders in turn instead of one, or “fan” him to the edge so the line can take the inside rushers and the quarterback is hot off an inside linebacker instead. They give a tight end a check-stay rule: he stays in to block if the edge defender rushes and releases into his route if that defender drops, which answers the exchange directly. They use “hot” routes that read the coverage rather than the rush, so that a receiver breaks away from a dropper instead of into him. They empty the backfield to make the defense declare its rush (with no back to protect, the line slides and the quarterback alone handles the leftover man). Above all, they throw quickly to the holes: a 1.5-second throw to the flat or a hitch is over before any exchange has finished. Each answer is a little cheaper than the pressure it beats, which is why the contest never ends.

Simulated pressure: four that look like six

The fire zone still sends five. The modern form of the zone blitz goes one step further: send only four, from unexpected places, and keep seven in coverage. That is simulated pressure, a four-man rush built to look like a blitz: one or more players the protection expects to rush drop into coverage, and one or more it expects to cover rush instead, so the defense gets a blitz’s confusion with seven players still in coverage. You will hear it shortened to sim pressure or “sim,” and many coaches call it a creeper, because an off-ball defender “creeps” into the rush to replace the lineman who drops. Ted Nguyen of The Athletic puts the broadcast version simply: multiple defenders on the line showing blitz, but the defense rushes only four and drops seven.2

Why would four ever be enough against six blockers? Because a protection puts specific blockers on specific possible rushers. If the right four come, at least one blocker has nobody, and one of the four may face a back instead of a lineman. The defense hasn’t out-numbered the offense; it has out-positioned it, with seven men still covering five receivers.

Look at what the creeper did to each blocker. The center slid toward the Mike, and the Mike left: one blocker gone. The left tackle set up for the end, and the end dropped, so he did what his rule says and turned in to help the guard; that is exactly why the edge was empty. The back’s rule was “the Will, then the nickel.” On paper the nickel was covered. But the back had to wait to see whether the Will came, and by the time he turned outside the nickel was on top of him at full speed. Nobody’s rule was wrong; the defense simply chose the order in which the offense would find out. Four rushers against six blockers, and the offense ended up with one back against a running start, while the quick slant that should have punished the nickel’s blitz ran into the end.

That is the deal simulated pressure offers, and it explains why it spread through the NFL in the 2020s:

  • It keeps seven in coverage. A creeper has four underneath, or five with two deep safeties, so the fire zone’s holes mostly close.
  • It wins one-on-ones instead of numbers: a nickel against a back, a defensive tackle against a center who just helped somebody else.
  • It attacks the rules without the risk. If the protection guesses right, it is a normal four-man rush with a funny-looking front; if wrong, a free rusher or a hot throw into a dropper, at no extra cost.
  • It looks identical to the real thing. The same pressure look can hide a creeper, a fire zone or a Cover 0 blitz, so the offense can never safely ignore it.

Nothing about a sim requires Cover 3 behind it. With two deep safeties the arithmetic is four rush, five under, two deep, and the dropper becomes one of five underneath players instead of four:

The price of the creeper

If the exchange confuses no one, a creeper is just a four-man rush with a good rusher playing a hook zone, and a defensive tackle is a real liability if a receiver runs into his area with room to work. It is also hard to teach: the four rushers have to keep their rush lanes (The Pass Rush) from odd starting places, or the quarterback escapes through the gap the dropper left. The defenses that do it best have edge players who can drop, safeties and nickel backs who can rush, and linemen mobile enough to look at home in a hook zone.

What the charting data says

In public data the closest thing to a simulated-pressure flag comes from FTN, a football-data company whose charters watch every play and record, for every dropback, how many players rushed and how many of those were blitzers. FTN doesn’t publish a definition, but the counts behave as if a blitzer is a rusher who isn’t one of the usual linemen and edge rushers: an off-ball linebacker or a defensive back. Four rushers with at least one blitzer means somebody who usually rushes didn’t. It’s an imperfect proxy, and it errs both ways. It misses sims in which the dropper and his replacement are both linemen or edge rushers (say, an end drops and a stand-up outside linebacker beside him rushes, since neither counts as a blitzer). And it counts some plain four-man rushes as sims, such as a three-lineman front whose fourth rusher is simply an off-ball linebacker every snap, with no one dropping. With that caution, here is what each kind of rush bought in 2023–2025.3

Three things stand out. First, the proxy for simulated pressure got the quarterback under pressure on 36% of dropbacks, close to the 39% of a real blitz and well above the 27% of a normal four-man rush; its sack rate (7.9%) matched the blitz’s (8.0%). Second, it got there with zone coverage behind it most of the time: the coverage was man on 35% of these plays, about the same as behind a normal four-man rush (33%) and far below the 60% behind a blitz. If simulated pressure is this good, why isn’t it called more often? Partly because the proxy undercounts it, but mostly because it works only as a surprise: its value depends on the offense having to respect the blitz, which is why the teams that sim the most also blitz a lot, or at least show it.

Where it came from: Miami, Cincinnati, and “Blitzburgh”

Who invented the zone blitz depends on whom you ask, and the honest answer is that it grew in stages, with each coach building on the last.

Bill Arnsparger’s Dolphins. Arnsparger was Don Shula’s defensive coordinator in Miami in two stints (1970–73 and 1976–83), a quiet, meticulous coach whose units were famous for being greater than their parts: the “No-Name Defense” of the early 1970s and the “Killer B’s” of the 1982 Super Bowl team.4 In the early 1970s he used linebacker Bob Matheson, who could rush like an end or cover like a linebacker, as a defensive end in the 4-3; with Matheson on the line, the defense could send five and still cover with six without the offense knowing which five. The package was known as the “53” defense, after Matheson’s jersey number.5 “With Bob there, with linebacking skills, we were able to rush five guys and cover with six,” Arnsparger later explained. “That’s what you need to run a zone blitz.” His catchphrase for what he was after was “safe pressure.”6 Some histories therefore date the zone blitz to Arnsparger’s Miami defenses of the early 1970s. Others point instead to the blitzes and zone-based defenses he tinkered with in the late 1970s and early 1980s, in his second Miami stint, as the more direct ancestor of what came next.7

Dick LeBeau’s Bengals. Dick LeBeau had been a Hall of Fame cornerback for the Lions (1959–72, 62 interceptions) before he started coaching. He became Cincinnati’s defensive coordinator in 1984, facing Bill Walsh’s 49ers and the precise, quick passing games spreading through the league, offenses that ate man-coverage blitzes alive with hot throws.8 He had already been experimenting. Before the 1983 season, while still the Bengals’ defensive backs coach, he put in a package that showed blitz and then had veteran cornerback Ken Riley sit down in a zone with a safety rotating behind him. “We ran the zone blitz and the quarterback just threw the ball straight to Ken Riley,” he remembered of an exhibition game.9 Around the time he became coordinator he visited Arnsparger, by then the head coach at LSU, and came away with the “safe pressure” idea, which he built into a system: linebackers and defensive backs rushing from many angles, linemen dropping, zone coverage behind. The arrival of safety David Fulcher in 1986, a big man who could rush and cover, let him expand it.

Blitzburgh. The scheme became famous in Pittsburgh. Bill Cowher’s Steelers hired Dom Capers as defensive coordinator in 1992 (he stayed through 1994, then left to become the first head coach of the Carolina Panthers) and LeBeau as defensive backs coach; LeBeau was coordinator in 1995–96 and again from 2004 to 2014.10 Cowher called the scheme “a collaboration between myself and Dom Capers and Dick LeBeau.” With a 3-4 front whose outside linebackers could both rush and drop, the 1994 Steelers led the NFL in sacks with 55, a franchise record, and the city got a nickname.11 LeBeau’s own description, decades later, is still the best one-sentence summary of the idea: “It’s a zone area where you would blitz a guy that normally doesn’t blitz and make the quarterback figure out what’s going on.”

Both origin stories are true in their way; this book gives Arnsparger the idea and LeBeau the system. From Pittsburgh it spread with the coaches: Capers took it to Carolina, and in the 1990s and 2000s the fire zone became a standard call for almost every NFL defense, especially once Nick Saban and Bill Belichick’s pattern-matching rules, worked out in the early 1990s, gave the three underneath defenders a better way to handle routes than spot-dropping: the seam-flat and middle-hook rules described earlier in this chapter.12

NoteFilm room: Steelers, Super Bowl XLIII (Feb. 1, 2009) — the rusher who dropped

The most famous coverage drop in Super Bowl history came straight out of the zone-blitz mindset. Last seconds of the first half, Arizona at Pittsburgh’s 1-yard line. The Steelers, with LeBeau as coordinator, showed a big blitz, and outside linebacker James Harrison, that season’s Defensive Player of the Year and the Steelers’ most dangerous edge rusher, was one of the players the Cardinals had to account for. Kurt Warner, expecting pressure, threw a quick slant to Anquan Boldin, and Harrison, who had dropped instead of rushing, was standing in the lane. He caught it at the goal line and ran 100 yards for a touchdown. “We had a play call where we were basically on a max blitz,” Harrison said afterward. “Kurt had to get it out, so he had to throw a quick slant in or out, and I guessed on in.” Accounts differ on exactly what Harrison’s assignment was and how much of the drop was his own read.13

What to notice: the play is the zone blitz’s logic in pure form. The quarterback reacted to the pressure he expected, threw the hot route to the spot the blitz should have emptied, and a man he had counted as a rusher was standing in it.

The modern sim defenses

By the 2020s the fire zone was standard equipment, and offenses had answers to it: quick throws, empty sets, protections that counted the zone-blitz exchanges. The frontier moved to four-man simulated pressure, often with two-high coverage behind it, not just Cover 3. Three defenses of the 2020s are worth knowing.

NoteFilm room: Ravens 2023 — Mike Macdonald’s lie about who is coming

Mike Macdonald was Baltimore’s defensive coordinator in 2022 and 2023 before becoming the Seahawks’ head coach in 2024. His 2023 Ravens became the model modern pressure defense without being a heavy-blitz defense at all: they blitzed on about 21.5% of dropbacks, below the league average, by one count in December. What they did instead was lie. Macdonald “leaned into simulated pressures, making it look like they’re blitzing when they’re not, or disguising where exactly the pass rush is coming from,” and he was willing to drop 355-pound nose tackle Michael Pierce or 338-pound Travis Jones into coverage.14 Cody Alexander’s charting that October ranked Baltimore among the league’s heaviest users of simulated pressure, with “a deep bag of simulated pressures at their disposal” and a defense “not afraid to drop interior defensive linemen.”15 The Ravens finished 2023 with 60 sacks, the most in the NFL, and allowed 16.5 points a game, the fewest.16

What to notice: on a Ravens third down from 2023, find the biggest man in coverage after the snap. Kyle Hamilton, a 6-foot-4 safety, is often the rusher you didn’t expect; a defensive tackle is often the dropper. Macdonald’s Seattle defenses carried the same ideas into his head-coaching years: the 2025 Seahawks allowed the fewest points in the league (17.2 a game) and won Super Bowl LX, 29–13 over New England. How his defense combines this pressure with a two-high coverage shell is the second half of Case Study: Fangio’s Two-High Revolution and Macdonald’s Answer. As head coach he kept calling the defensive plays himself, with Aden Durde as his coordinator.17

NoteFilm room: Vikings 2023–2024 — Brian Flores and the blitz you have to respect

Brian Flores, Minnesota’s defensive coordinator since 2023 (and still in 2026), shows the other half of the bargain: a creeper only fools an offense that fears the blitz, and nobody had more reason to fear it than Flores’s opponents. In 2023 the Vikings rushed five or more on 51% of dropbacks in FTN’s charting, the most in the league. In 2024 they blitzed less and simmed more: by the FTN proxy, 11.3% of their dropbacks were four-man rushes including an off-ball rusher, the second-highest rate in the NFL behind Tampa Bay (12.1%), coached by Todd Bowles, another longtime pressure designer.18 Sports Info Solutions found that through Week 8 of 2024 the Vikings got pressure on 35% of dropbacks when they rushed four or fewer, sixth in the league, and on 34% when they blitzed, 27th: the four-man rushes from blitz looks were doing more work than the blitzes.19

What to notice: a Vikings passing down often starts with six, seven, even eight defenders near the line. Count them, then count the rushers. Every call starts from the same picture; that is the point.

Steve Spagnuolo, Kansas City’s coordinator since 2019 and the architect of the 2007 Giants’ pass rush, belongs on the list too: better known for his man-coverage blitzes (Blitzing), his 2023 Chiefs were also one of the league’s heavier sim users (sixth by the FTN proxy, at 6.8%) and finished with 57 sacks, second only to Baltimore.

Third down: when to blitz in zone and when in man

Fire zones and sims are calls, not a defense, and the cleanest pattern in when they’re called is on third down. On third-and-short a fire zone hands the offense exactly what it wants, a short throw in front of three underneath defenders, so a defense that blitzes there usually plays man. On third-and-long the short throw it concedes is short of the line to gain, and zone pressure becomes far more common.

Watch for it on Sunday

TipWatch for it

Before the snap: read the pressure look.

  • Count the players at or near the line. Six or more means a pressure look. Is it two linebackers mugging the A gaps, a safety walked down beside an end, a nickel creeping toward the slot?
  • Count the deep players. Two safeties deep and the corners off means at most five rushers in any sound call, and probably four. One deep safety leaves room for a fire zone or a Cover 1 blitz; none, with corners pressed, warns of Cover 0.
  • Find who could drop. Stand-up edge players and any lineman in a two-point stance are the likely droppers; a lineman with his hand down, weight forward, is probably rushing.
  • Watch the offense react. A center pointing and re-pointing, a back moving to the other side, a tight end staying in: the offense is changing its protection to answer what it sees.

After the snap: who dropped?

  • Find the lineman who dropped into coverage, and then find who replaced him as a rusher. That pair is the zone blitz. The broadcast’s high replay angle shows it best.
  • Count the rushers. Five with three deep is a fire zone; four, with one of them a linebacker or defensive back and a lineman in coverage, is a simulated pressure; five or six with man coverage is an ordinary blitz.
  • Follow the ball. Did the quarterback throw hot? Was a dropper in the lane? If the quarterback completed it, where was the hole: behind the dropper, over a dropping tackle, or the receiver a blitzing corner left?
  • Find the blocker who blocked nobody. On a sim pressure there is usually one: a center or tackle standing with his hands up and no one in front of him. He is the measure of how well the lie worked.

For your log: on every third-and-6-or-more for one quarter, write down the number at the line before the snap, the number who rushed, and whether a lineman dropped. A defense whose two numbers often differ is a sim defense.

How many will rush, who is most likely to drop, and what should the quarterback be careful about?

TipAnswer

Most likely four or five, not six, and at least one of the mugged linebackers will drop. Count to eleven. Two deep safeties and two off corners are four defenders playing deep or soft. A defense that sent six of the other seven would have one man underneath and soft corners: a Cover 0 blitz without the press coverage Cover 0 needs. So the likely calls are a simulated pressure (four rush, one lineman dropping, two deep behind) or a fire zone with a safety rotating down at the snap.

The likely droppers are one of the two mugged linebackers (they can’t both rush and leave the middle empty) and probably a lineman, who replaces him underneath.

The quarterback’s danger is the hot throw: a dropper is very likely to land in the short middle, right where the slant against a linebacker blitz goes. Better: quick hitches or outs against corners 8 yards off, or a route that sits in the hole behind a dropping lineman. One last look before the ball is snapped: if both safeties drive down late, it’s a Cover 0 blitz after all (a late spin some defenses keep for third-and-long), and now the hot throw is exactly right.

If this is a blitz, will the coverage behind it be zone (a fire zone) or man? And what’s the offense’s answer?

TipAnswer

Man, most likely a Cover 1 blitz. Two reasons. The situation: on third-and-short a fire zone gives up exactly what the offense needs, a short throw in front of three underneath defenders, so defenses that blitz on third-and-1 to 3 play man behind it about three times in four (the third-down chart above). The picture: the corners are pressed, a yard off and square to their receivers, and the nickel is tight on the slot. Those are man techniques (Disguise, Rotation, and Split-Field Coverage), and one deep safety is the Cover 1 shell.

The answer is the man-beater: quick throws that win against tight man coverage, such as a slant, a rub or pick concept where two receivers cross, or a back-shoulder fade (thrown to the outside shoulder, away from a defender who is running with the receiver), timed to come out before the extra rusher arrives, with the protection set to leave the free man (if there is one) on the side away from the throw. A smart defense knows the offense expects man here, which is exactly when a sim with zone behind it is worth calling.

Takeaways

  • A blitz trades coverage for pressure; the zone blitz tries not to. It rushes a defender the protection didn’t count and drops one it did (a lineman, the dropper), so the rush is confusing and the coverage behind it is still zone.
  • The fire zone is the standard form: five rush, three under, three deep. It refuses the deep ball and accepts short throws; its holes live next to the slowest dropper and move with each variant.
  • Protections and hot reads lose to rules, not people. The Mike call, the slide and the hot are predictions keyed to who rushes. A pressure look gets the offense to commit; the zone blitz brings rushers from wherever it didn’t, and drops a man into the hot throw’s lane.
  • Simulated pressure (the creeper) does it with four. Four rushers from unexpected places, seven in coverage: it wins one-on-ones and makes blockers block air without paying in coverage. In 2023–2025 charting it produced blitz-like pressure and sack rates with four-man-rush coverage behind it.
  • It only works if the blitz is real. Creepers fool offenses that must respect a blitz, so the best sim defenses (Macdonald’s Ravens and Seahawks, Flores’s Vikings) also show, and sometimes bring, everything.
  • Count to eleven. Rushers plus underneath plus deep equals eleven. Count the deep players before the snap to cap the rush, and count the rushers after it to name what you saw.
