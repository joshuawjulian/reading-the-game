def four_down(lbs=False):
    """A generic four-man rush: a wide end on each side, a 1-technique shaded on the center's left,
    a 3-technique on the right guard's outside shoulder; optionally two linebackers."""
    out = [D("LE", "DE", 1.0, -5.4, "E"), D("NT", "DT", 1.0, -0.7, "1"),
           D("DT", "DT", 1.0, 2.35, "3"), D("RE", "DE", 1.0, 5.4, "E")]
    if lbs:
        out += [D("MIKE", "MIKE", 4.6, 1.3, "M"), D("WILL", "WILL", 4.6, -2.2, "W")]
    return out


def to(p, key, end, kind="block", via=(), delay=0.0, speed=None):
    """Move `key` through absolute (d, w) points `via` to `end` as a block, rush, path or run."""
    start = np.array(p._cur(key))
    pts = [tuple(np.array(q) - start) for q in (*via, end)]
    if kind == "block":
        return p.block(key, pts=pts, delay=delay, speed=speed)
    if kind == "rush":
        return p.rush(key, pts=pts, delay=delay, speed=speed)
    return {"path": p.path, "run": p.run}[kind](key, pts, delay=delay, speed=speed)


def timed(p, key, way, kind="path"):
    """Script a player for the frame strips: `way` = [(t, d, w), ...] absolute spots reached at time t
    (seconds after the snap). Standing still between two times is allowed."""
    cur = np.array(p._cur(key), dtype=float)
    tc, wait = 0.0, 0.0
    for t, d, w in way:
        nxt = np.array([d, w], dtype=float)
        dist = float(np.hypot(*(nxt - cur)))
        if dist < 1e-3:
            wait += t - tc
        else:
            p._add(kind, key, [tuple(cur), tuple(nxt)], speed=dist / (t - tc), delay=wait)
            wait = 0.0
            cur = nxt
        tc = t
    return p


def _P(fld, p, d, w):
    return fld.xy(*p.to_abs(d, w))

def dashed(fld, p, pts, color=RED, lw=1.8, end="arrow", dash=(0, (3, 1.8)), z=6):
    """A dashed path through absolute (d, w) points, ending in an arrow or a block bar ('tee')."""
    xs, ys = _P(fld, p, np.array([q[0] for q in pts]), np.array([q[1] for q in pts]))
    xs, ys = np.asarray(xs, float), np.asarray(ys, float)
    fld.ax.plot(xs, ys, color=color, lw=lw, ls=dash, zorder=z)
    if end == "arrow":
        fld.ax.add_patch(FancyArrowPatch((xs[-2] + 0.6 * (xs[-1] - xs[-2]), ys[-2] + 0.6 * (ys[-1] - ys[-2])),
                                         (xs[-1], ys[-1]), arrowstyle="-|>", mutation_scale=11, color=color,
                                         lw=lw, zorder=z))
    elif end == "tee":
        v = np.array([xs[-1] - xs[-2], ys[-1] - ys[-2]])
        n = np.linalg.norm(v)
        perp = np.array([-v[1], v[0]]) / n * 0.75
        fld.ax.plot([xs[-1] - perp[0], xs[-1] + perp[0]], [ys[-1] - perp[1], ys[-1] + perp[1]], color=color,
                    lw=lw + 0.6, zorder=z)


def pushed(fld, p, a, b):

def arm(fld, p, a, b, lw=3.4, color=ARM):
    """An arm (or hands) as a thick stroke from a to b, with a dot for the hand: dark orange for the rusher,

def clean(ax):
    """Box-zoomed diagrams: drop hash ticks and yard numbers, clip text to the window."""
    for ln in list(ax.lines):
        xs, ys = ln.get_data()
        if len(xs) == 2 and ln.get_linewidth() == 0.5 and np.hypot(xs[1] - xs[0], ys[1] - ys[0]) < 1.0:
            ln.remove()
    for t in list(ax.texts):
        if t.get_text().isdigit() and t.get_alpha() == 0.55:
            t.remove()
        else:
            t.set_clip_on(True)


def draw(p, width_in=5.8, ax=None, ol_fs=None, **kw):
    """Play.draw, sized for the page and cleaned for a box-sized window. ol_fs shrinks the two-letter line
    labels (LT, LG, RG, RT) on wide figures, where the default size overprints."""
    fld = p.draw(width_in=width_in, ax=ax, **kw)
    clean(fld.ax)
    if ol_fs:
        for t in fld.ax.texts:
            if t.get_text() in ("LT", "LG", "RG", "RT"):
                t.set_fontsize(ol_fs)
    return fld

# --- frame strips (print) and video (web) for the simulated plays ---------------------------
MARK_R = 0.6
RING_R = 0.9


def tracking(play):
    df = play.to_tracking()
    df["_id"] = df["displayName"].astype(str)
    return df


def panel(ax, play, df, t, lateral, window, rings=(), links=()):
    """One moment of a simulated play: markers, faint trails, rings, and 'who has whom' links."""
    tmap = df.groupby("frameId")["time"].first()
    fid = int((tmap - t).abs().idxmin())
    ylo, yhi = sorted((play.ball_y - lateral[0], play.ball_y - lateral[1]))
    fld = Field(ax=ax, orient="vertical", los=play.los, window=window, numbers=False, hashes=False,
                lateral=(ylo, yhi), to_go=play.to_go)
    cur = df[df["frameId"] == fid]
    shown = cur[cur["x"].between(play.los + window[0] + 0.5, play.los + window[1] - 0.5)
                & cur["y"].between(ylo + 0.5, yhi - 0.5) & (cur["side"] != "ball")]
    ids = set(shown["_id"])
    trails = df[(df["frameId"] <= fid) & df["_id"].isin(ids)]
    for (_, side), g in trails.groupby(["_id", "side"]):
        xs, ys = fld.xy(g["x"].values, g["y"].values)
        ax.plot(xs, ys, color=style.OFFENSE if side == "off" else style.DEFENSE, lw=0.9, alpha=0.4, zorder=3)
    ppy = fld.pts_per_yard()
    for _, r in shown.iterrows():
        x, y = (float(v) for v in fld.xy(r["x"], r["y"]))
        lab = str(r.get("label", "") or "")
        if r["side"] == "off":
            ax.add_patch(Circle((x, y), MARK_R, fc=style.OFFENSE, ec="white", lw=0.6, zorder=8))
            tc = "white"
        else:
            ax.add_patch(RegularPolygon((x, y), 4, radius=MARK_R * 1.3, orientation=np.pi / 4,
                                        fc=style.DEFENSE_TINT, ec=style.DEFENSE, lw=1.0, zorder=8))
            tc = style.INK
        fs = max(4.0, min(7.5, ppy * (0.8 if len(lab) <= 1 else 0.55)))
        ax.text(x, y, lab, ha="center", va="center", fontsize=fs, color=tc, fontweight="bold", zorder=10)
    for key, col in rings:
        r = cur[cur["displayName"] == key].iloc[0]
        x, y = fld.xy(r["x"], r["y"])
        ax.add_patch(Circle((float(x), float(y)), RING_R, fc="none", ec=col, lw=1.7, zorder=7))
    for a, b, col in links:
        pa = cur[cur["displayName"] == a].iloc[0]
        pb = cur[cur["displayName"] == b].iloc[0]
        xs, ys = fld.xy(np.array([pa["x"], pb["x"]]), np.array([pa["y"], pb["y"]]))
        ax.plot(xs, ys, color=col, lw=1.5, ls=(0, (1, 1.4)), zorder=9)
    return fld


def strip(play, times, notes=None, lateral=(-9, 9), window=(-9, 6), ncols=3, rings=(), width_in=6.5,
          final_links=()):
    """Numbered, captioned frame strip with a fixed box-sized window, so markers stay legible in print."""
    df = tracking(play)
    nrows = int(np.ceil(len(times) / ncols))
    pw = width_in / ncols
    span_w, span_d = lateral[1] - lateral[0], window[1] - window[0]
    fig, axes = plt.subplots(nrows, ncols, squeeze=False,
                             figsize=(width_in, nrows * (pw * span_d / span_w + (0.6 if notes else 0.25))))
    for ax in axes.flat[len(times):]:
        ax.axis("off")
    for i, (ax, t) in enumerate(zip(axes.flat, times)):
        panel(ax, play, df, t, lateral, window, rings, final_links if i == len(times) - 1 else ())
        ax.set_title(f"{i + 1}.  " + ("snap" if abs(t) < 1e-6 else f"{t:+.1f} s"),
                     fontsize=8.2, color=style.INK_2, loc="left")
        if notes:
            note_below(ax, notes[i], size=6.9)
    fig.tight_layout(h_pad=1.0 if notes else 0.5, w_pad=0.5)
    return fig


def video(play, times, notes=None, lateral=(-9, 9), window=(-9, 6), rings=(), width_in=5.4):
    """Web version of the strip: every frame, with the strip's notes as running captions."""
    import shutil
    from matplotlib import animation
    from IPython.display import HTML, display
    df = tracking(play)
    span_w, span_d = lateral[1] - lateral[0], window[1] - window[0]
    fig, ax = plt.subplots(figsize=(width_in, width_in * span_d / span_w + 0.4))
    tmap = df.groupby("frameId")["time"].first()

    def update(fid):
        ax.clear()
        t = float(tmap[fid])
        panel(ax, play, df, t, lateral, window, rings)
        txt = "snap" if abs(t) < 0.05 else f"{t:+.1f} s"
        if notes:
            cur = [n.replace("\n", " ") for tt, n in zip(times, notes) if tt <= t + 1e-6]
            txt += ("   " + cur[-1]) if cur else ""
        ax.set_title(txt, fontsize=8.5, color=style.INK, loc="left")
        return []

    frames = [f for f in sorted(df["frameId"].unique()) if float(tmap[f]) >= -0.3]
    anim = animation.FuncAnimation(fig, update, frames=frames, interval=100)
    html = (anim.to_html5_video(embed_limit=40).replace("<video ", '<video style="max-width:100%;height:auto" ')
            if shutil.which("ffmpeg") else anim.to_jshtml(default_mode="once"))
    plt.close(fig)
    display(HTML(html))


def rush_animation(play, times, notes=None, **kw):
    """Video on the website; a numbered, captioned frame strip in the PDF."""
    if pdf_build():
        strip(play, times, notes=notes, **kw)
        plt.show()
    else:
        video(play, times, notes=notes, **{k: v for k, v in kw.items() if k in ("lateral", "window", "rings")})

def ts(team, season):
    return FTN[(FTN["defteam"] == team) & (FTN["season"] == season)].iloc[0]


CLE24, CLE25, DEN25, SF25 = ts("CLE", 2024), ts("CLE", 2025), ts("DEN", 2025), ts("SF", 2025)

# Year-to-year carry-over, within one charting source only (no 2022 -> 2023 pair: the source changes)
PAIRS = [s for s in SEAS[:-1] if s != 2022]


def carry(frame, key, a_col, b_col=None, min_n=None):
    b_col = b_col or a_col
    out = []
    for s in PAIRS:
        a = frame[frame["season"] == s].set_index(key)
        b = frame[frame["season"] == s + 1].set_index(key)
        if min_n is not None:
            a, b = a[a["n"] >= min_n], b[b["n"] >= min_n]
        j = a[[a_col]].join(b[[b_col]], rsuffix="_next", how="inner").dropna()
        out.append(j.iloc[:, 0].corr(j.iloc[:, 1]))
    return np.array(out)


QS = DB[DB["press"]].groupby(["season", "passer_id"]).agg(n=("sack", "size"), p2s=("sack", "mean"),
                                                          name=("passer", "first")).reset_index()
CARRY = {
    "Defense: pressure rate": carry(TS, "defteam", "pr"),
    "Defense: sack rate": carry(TS, "defteam", "sr"),
    "Defense: pressure rate\n(predicting next year's sack rate)": carry(TS, "defteam", "pr", "sr"),
    "Defense: pressure-to-sack rate": carry(TS, "defteam", "p2s"),
    "Quarterback: pressure-to-sack rate": carry(QS, "passer_id", "p2s", min_n=100),
}

The rusher’s problem: a race and a fight

The quarterback in today’s NFL releases the ball a little under three seconds after the snap on average (Pass Protection has the numbers), and on a quick throw in well under two. An edge rusher who lines up outside the tackle has to cover an arc of about thirteen yards to reach the spot where the quarterback sets his feet; a defensive tackle in the middle has nine or so, straight ahead, but through the thickest traffic on the field. Unblocked, an edge rusher would cover his arc in roughly two seconds. He is never unblocked. A man who weighs about 315 pounds, who knows where the quarterback will be and only has to stay in the way, is standing on the path.

So a pass rush is two contests at once. It is a race: the rusher has to get to the quarterback before the ball leaves, and a rusher who arrives at 3.4 seconds against a quarterback who threw at 2.6 has done nothing, however good his move was. And it is a fight: to win the race he has to beat a blocker who is bigger, who has the inside position, and who is allowed to use his hands too. The rusher’s advantages are the two the blocker lacks. He knows the snap is coming and where he is going, and he moves forward, while the blocker moves backward.

There is a third fact, and it explains why defenses pay pass rushers so much: most of the time, the defense rushes only four. In FTN’s charting of the 2023–2025 regular seasons, 69% of dropbacks faced exactly four rushers and 26% faced five or more.1 A blitz can manufacture a free rusher by sending more men than the offense can block, but every extra rusher is one fewer defender in coverage, and the quarterback knows it. Four rushers against five or six blockers leaves seven defenders to cover at most five receivers. If those four can create pressure by themselves, the defense gets to the quarterback and covers everybody. That is the holy grail of defensive football, and this chapter is about how it is done: by moves, by plans, and by stunts.

ImportantWhy it exists

A four-man rush is the balance point between coverage, which is expensive, and pressure, which is valuable. A team whose four can win their fights gets pressure without paying for it in coverage, which is why an edge rusher who wins them is usually the highest-paid player on his defense.

Get-off: the first step decides more than you think

The fight begins before anyone touches anyone. A rusher’s get-off is his first movement at the snap: how quickly he reacts to the ball moving and gets across the line of scrimmage.

Defenders are not allowed to move into the neutral zone before the snap, and unlike the offense they do not know the snap count. So a rusher does not listen to the quarterback’s cadence; he watches the ball, or the center’s hand on it, and goes the instant it moves. The offense’s weapons against this are a varied snap count (a “hard count” to draw a rusher offside, a quick count to catch him flat-footed) and, in a loud stadium, a silent count, which takes away the hard count but can leave the snap on a rhythm (the center looks back, the guard taps him, the ball comes on a set beat) that a good rusher learns to time. The rusher’s weapon is a stance with his weight already tilted forward and a first step that covers ground without popping up.

def along(pts, s):
    """The point a distance s (yards) along a polyline."""
    for a, b in zip(pts[:-1], pts[1:]):

        if s <= seg:
            f = s / seg
            return (a[0] + f * (b[0] - a[0]), a[1] + f * (b[1] - a[1]))
        s -= seg
    return pts[-1]

At full speed an edge rusher covers about seven yards a second, so a tenth of a second at the snap is most of a yard on the arc. In Figure 39.1 the two rushers differ by three-tenths, and by 1.2 seconds that has become a 2.2-yard difference in where they meet the tackle. The quick one meets him level, at the tackle’s own depth, where the tackle has to turn his hips to run with him, and a blocker who turns his hips has given up the inside and lost most of his power. The slow one meets him in front, where the tackle is square, balanced and ready to punch. Same athlete, same path, a completely different fight.

That is why scouts time rushers’ first steps at the Combine (the league’s pre-draft testing event), why defensive line coaches drill get-off every day, and why the best rushers are often the first player on the field to move. Tracking data now measures it directly: Next Gen Stats times how long a rusher takes to cross the line of scrimmage after the snap and quotes it in its broadcasts and analysis.2 For a scale: the quickest single snaps it has highlighted are around a third of a second, best-ever reps. A lineman whose body starts about a yard behind the ball needs roughly half a second to cross it from a standstill, so most rushes take longer; the simulated rushers at the end of this chapter take half a second to eight-tenths.

The arc and the corner

Before the moves, the geometry they are all trying to bend. An edge rusher’s basic path is the arc: a curve up the outside of the tackle and around him toward the quarterback. He aims it at a landmark, not at the blocker: the quarterback’s launch point, about seven or eight yards behind the line from the shotgun, which is why the diagrams put Q there. The tackle’s answer, as Pass Protection showed, is to kick-slide back and out, staying between the rusher and that spot, and to push the rusher past the quarterback if he keeps running. The pocket is a cup, and a rusher run around the rim of the cup ends up behind it, harmless.

How wide the rusher lines up is the first trade. A wide alignment like the wide-9 gives him a running start and room to build speed, at the cost of a longer arc and a soft spot inside him that draws and screens are built to punish; a tight alignment over the tackle gives him a shorter path and power, but no space to work.

The rusher wins the arc at one spot: the corner, the moment his hips draw level with the tackle’s. Before that, the tackle is in front of him and can punch and steer. Once the rusher’s hips are past, the tackle is beside or behind him, can only push, and any push sends the rusher where he wanted to go. (His last resort is to grab: a hold, ten yards if the official sees it.) That is why the last part of every good edge rush is a tight turn back toward the quarterback, flattening the arc once he is past the tackle rather than continuing upfield.

Two physical skills decide whether a rusher can do this, and you can see both on any broadcast replay. The first is bend: the ability to lean his whole body toward the inside, ankles flexed, shoulder almost at the turf, while running at full speed, so he can turn the corner tightly instead of drifting wide. The second is dip: dropping the shoulder nearest the tackle so the tackle’s reaching hands slide off it. Watch a great edge rusher win and you will see a body tilted at an angle that looks like it should fall over.

One coaching rule sits under all of it: rush half a man. A rusher never attacks the blocker’s whole chest, where the blocker is strongest; he attacks one half of him, a shoulder, so the blocker has to turn to stay in front. Every move below starts on a shoulder.

The tackle, of course, knows all this. If he gets deep enough fast enough, there is no corner to turn. He can also get it wrong in the other direction and overset: retreat wider or deeper than the rush in front of him needs, opening his hips to the outside too early, which leaves the space inside him empty. And so the rusher needs other ways to win: through the tackle, or back inside him. That is what the moves are.

The speed rush. The speed rush is the move everything else is built on: beat the tackle to the corner before he can get in front. Fast get-off, a curved path up the arc, a dip of the near shoulder as the tackle reaches for him, and a bend back toward the quarterback. It beats a tackle who sets too slowly or not deep enough, and it is the move tackles fear most, which is exactly why it makes every other move work: a tackle who is worried about speed retreats fast, and a man retreating fast is off balance in other ways. Its weakness is the one the pocket is built around: a tackle who gets deep enough simply runs the rusher past the quarterback.

The bull rush. The bull rush is the opposite idea. Instead of going around the blocker, the rusher goes through him: hands inside on the chest, hips low, legs churning, driving him straight back into the quarterback. A bull rush rarely produces a sack by itself; what it produces is a collapsing pocket. A quarterback who has a tackle or guard backing into his feet cannot step up, cannot finish his throwing motion and often cannot see. It beats a blocker who sets too high (pads up, so the rusher gets under him) or one whose weight is already going backward because he retreated too fast, since a man moving backward cannot plant and push. The blocker’s answer is the anchor: sink the hips, widen the base, and stop the slide. A bull rush against a blocker who anchors is a stalemate, and a stalemate is a win for the offense.

The long arm. The long arm is the bull rush with an escape hatch. The rusher locks one arm, fully extended, into the blocker’s chest (usually his inside arm into the outside half of the blocker’s chest, the near breastplate). Why can the blocker not just reach back? Because a punch is thrown with both hands and square shoulders, which keeps the elbows bent; one arm locked straight, with the rusher’s shoulder turned into it, reaches several inches farther. So the blocker’s hands fall short of the rusher’s body, the rusher controls the distance between them and drives the blocker back with it, and his other hand stays free to knock the blocker’s hands away and slip off the block when the moment comes. It has become one of the signature moves of the modern NFL edge rusher, because it works for long-armed athletes who are not giant men, and because it lets a rusher generate power without getting tied up chest to chest.

The swim. The swim move clears the blocker’s hands with one hand and throws the other arm over the top of his shoulder, like a swimmer’s overhand stroke, to get past him. On the edge it is usually an inside move: the rusher fakes outside to make the tackle widen, then swims back inside. In the middle it is a staple of quick, compact defensive tackles who meet guards almost at once. It needs timing and a blocker whose weight has gone the wrong way, and its risk is visible in the drawing: for a moment the rusher’s arm is up and his chest is open, and a blocker who is not fooled will drive right into it.

The rip. The rip move is an uppercut: the rusher drives his inside arm up under the blocker’s arm and through his armpit, lifting the arm off him and turning his own shoulder through the gap so his hips get past. It is the usual way a speed rush is finished, because at the corner the tackle’s outside hand is the last thing between the rusher and the quarterback, and the rip takes that hand away. A dip and a rip together are the most common sight on an edge sack.

The spin. The spin move is the showpiece. When the tackle has committed to the outside, the rusher plants his outside foot, turns his back to the blocker and spins all the way round, coming out back inside him toward the quarterback. Dwight Freeney made it famous with the Colts in the 2000s, and it carried him to the Hall of Fame.3 Done at the right moment, against a tackle who has overset, it leaves the blocker grabbing air. Done at the wrong one, against a blocker who stayed square, it spins the rusher out of the pocket, and for half a second he cannot even see the quarterback: big reward, big risk.

Telling them apart on the broadcast. The speed rush and the rip look like one motion: the rusher’s outside shoulder dips, his inside arm swings up through the tackle’s armpit, and he flattens toward the quarterback. In a bull rush the two men are chest to chest and the tackle’s helmet is snapped back; in a long arm there is daylight between them, one arm locked straight across it, and the tackle’s arms flailing short. A swim shows an arm going over the tackle’s shoulder, like a freestyle stroke, as the rusher goes inside; a spin shows the rusher’s back to the camera for an instant.

CautionCommon misconception

“Pass rushers win with strength.” Strength helps, especially in the bull rush and the long arm, but most rush wins are wins of timing and position: a quicker first step, hands that land before the blocker’s do, a move chosen for the set the blocker just showed. It is the mirror of Pass Protection’s point that most protection losses are losses of position.

Moves are counters to hands

Watch a pass rush closely and you will see that the rusher’s hands are busy before his feet have done anything clever. The blocker’s main weapon is his punch, a two-handed strike into the rusher’s chest to stop his momentum and steer him. Many moves are built specifically to beat that punch.

The two oldest are power moves. The club is a forearm or open-hand blow to the blocker’s shoulder or elbow that knocks his punch aside, usually to set up a rip or a swim. The push-pull (also called a snatch or a jerk) grabs the blocker, pulls him forward past the rusher just as he leans in, and lets go: the interior rusher’s signature move, because guards meet rushers almost at once and lean into them.

The cross-chop is the clearest example (Figure 39.3). As the tackle punches, the rusher swings his outside arm across his own body and chops down on the tackle’s hands or forearms, knocking the punch down and away, then rips or dips past while the tackle’s hands are out of position. It turns the blocker’s best tool into the reason he loses. On a replay it looks like a karate chop: the rusher’s arm sweeps across his chest and the tackle’s arms drop.

The ghost (rush move) goes one step further: no hand contact at all. A dip lowers the shoulder under hands that are already reaching; a ghost is timed to the punch. At the instant the tackle strikes, the rusher turns his inside shoulder away, so the punch lands on nothing and the tackle, having thrown his weight into a strike that hit air, lunges forward off balance as the rusher bends around him. On a replay, look for the tackle reaching at empty space, often falling forward. A ghost works best against a tackle who punches early and hard; a patient tackle who “catches” the rusher and strikes late gives it nothing to avoid, which is one reason many NFL tackles have stopped throwing one textbook punch at every rusher.

Rules shape this hand fight too. A rusher may use his hands on a blocker’s body, but thrusting them up into the blocker’s head or neck is a foul (an illegal use of hands, five yards and an automatic first down when the defense does it).4 That rule has a history you will meet below, and it is why the modern hand fight is aimed at the blocker’s hands, arms and chest.

The rush plan: setting up the move you want

Here is the real secret of pass rushing, and the reason the moves above are only half the story. A tackle facing the same rusher thirty-odd times in a game is not guessing blindly; he is studying the rusher too, and he adjusts. So the best rushers do not just have moves. They have a rush plan: a planned sequence across the game that shows the blocker one move until he adjusts to stop it, then uses the counter his adjustment has opened.

The logic in Figure 39.4 is a triangle. Speed threatens the corner, so the tackle must get depth, and there are only two ways to get it fast. Get it by widening, and the inside opens for the spin or swim. Get it by retreating straight back at full speed, and his weight is going backward, which is exactly when power works. So he must stay balanced and patient, which slows his set, which brings speed back. A rusher with one great move can be planned for. A rusher with two that look the same for the first two steps cannot.

That is the job of speed-to-power, probably the single most important move in the modern edge rusher’s kit. It starts exactly like a speed rush: the same get-off, the same first steps up the arc. The tackle, seeing speed, kicks back fast. Then, two or three steps in, the rusher turns straight into him and converts all that speed into a bull rush, hitting him while his weight is still going backward, when he cannot anchor. It is the move in this chapter’s opening, and it works because of the speed rush: the more a tackle fears the corner, the faster he retreats, and the more his own momentum helps the rusher push him.

You can follow a rush plan from the couch. Early, many rushers show speed again and again, to make the tackle think about the corner. Once he starts bailing, the counters come: the inside spin, the speed-to-power, the long arm. A tackle who has been beaten inside starts to sit, and the speed rush is back on. The plan is built during the week from film of that specific tackle (his set depth, whether he punches early, whether he oversets after being beaten), so the same rusher may use a completely different plan the following Sunday.

NoteMadden vs. real life

In Madden a pass-rush move is a button press, and whether it works is mostly a matter of ratings. Real moves work or fail depending on what the blocker is afraid of, which depends on what the rusher showed him on the previous ten snaps. The spin that beats a tackle in the fourth quarter would have been stuffed in the first, before he had a reason to overset. Pass rushing is a conversation across a game, not a series of independent dice rolls.

Inside rushers: less room, shorter path

Everything above is described from the edge, where most sacks are recorded. Defensive tackles play a different version of the same game. An interior rusher has the shortest path to the quarterback, and pressure up the middle is the most destructive kind: the quarterback cannot step up into it and cannot see over it. But he also has the least room to work. A guard is on him almost immediately (there is no long arc to set up a move, only a yard or so of space), and he often faces two blockers. That is the four-against-five arithmetic from the start of the chapter: five linemen against four rushers leaves one blocker spare, and the spare is usually the center, who sits in the middle and can turn either way to help a guard. Interior rushers therefore win with fast, compact hands: a club or a push-pull the instant they meet the guard, a swim or a rip past him, a bull rush to push the pocket. When the ball is coming out too quickly to get home, they do the next-best thing and get their hands up into the throwing lane; a batted pass is an interior rusher’s win. And they are measured differently. A 3-technique who drives the guard back into the quarterback’s lap, forcing him to slide away from the pressure and into an edge rusher’s arms, may have done more than the edge rusher who gets credited with the sack.

The greatest interior rusher of the modern era made the case that a defensive tackle could be the most important player on the field. Aaron Donald, a 6-foot-1 defensive tackle the Rams drafted 13th overall in 2014, won Defensive Player of the Year three times (2017, 2018 and 2020) and had 20.5 sacks in 2018, astonishing for a player who lined up inside and was double-teamed constantly.5 What made him special was not size. It was get-off, hand speed and leverage: he was often past the guard before the guard had finished his first step, the one-gap penetration of Defensive Line Play applied to the pass, and his hands (the club, the push-pull, the swim) were too quick for a double team to land cleanly.

Stunts: winning with geometry instead of muscle

Moves win one-on-one fights. A stunt, also called a pass-rush game or a twist, wins by making two blockers’ assignments collide. In a stunt, two or more rushers exchange gaps after the snap: one goes first to occupy blockers (the penetrator), and the other loops around behind him into the space he left (the looper). The defense still rushes four, still covers with seven; it just rearranges where the four go.

You saw the most common stunt from the offense’s side in Pass Protection, where the guard and tackle passed it off by switching. Here are the main families from the defense’s side.

The T-E stunt. In a T-E stunt (tackle first, end second) the defensive tackle drives into the offensive tackle’s inside half to pick him, then takes over contain, and the end loops around behind him into the inside gap he vacated. The penetrator’s aiming point is the near hip of the far blocker (the offensive tackle, not the guard in front of him): he has to collide with both blockers, a legal pick, so they cannot switch cleanly. A penetrator who just runs around the tackle has wasted the stunt. If the guard follows the 3-technique outside and the tackle waits for the end on the outside, both block the wrong man, and the end walks through the gap.

The E-T stunt. The E-T stunt is the mirror: the end crashes inside first, picking the guard, and the tackle loops around him to the outside, taking over the end’s rush lane. It sends the looper wide, toward an offensive tackle who has just watched the end leave and may have turned inside to chase him.

Which one, when? Both hunt the same seam; the difference is who arrives second, unblocked or against the wrong man. Defenses usually make the best rusher the looper, so a star end means T-E (he loops inside onto a guard, who is slower than a tackle and rarely sees an end) and a quick, rangy 3-technique means E-T (he loops outside onto a tackle who is turned the wrong way). The matchup matters too: aim the looper at the weaker blocker, and the pick at the one who switches poorly.

The interior twist. Two defensive tackles can do the same thing in the middle, a T-T twist. In Figure 39.5 the 3-technique crashes through the right guard’s inside shoulder into the A gap, picking the guard and the center, and the 1-technique wraps over the top behind him into the B gap he left. It attacks the guards and center, the three men who are supposed to hold firm, and it is especially effective against centers who have to make the protection calls and then find a man.

The words overlap: many staffs say “stunt” for run-down line movement and “game” or “twist” for a pass-rush exchange, the penetrator is also the crasher or picker, and the looper the wrapper. Each stunt gets a code word (“Tex” is a common one for T-E), and many defenses run three-man games too. Most stunts come in the coordinator’s call, but many lines also have games they can check into at the line when they see a particular protection.

Why stunts work, and what they cost

Stunts exist because they create a mismatch or a free rusher without blitzing. Against man protection, the blockers have to recognise the stunt and pass it off in a split second; if either one hesitates, the looper is free. Against slide protection, as you are about to see, a stunt aimed the wrong way is simply absorbed, but one aimed at the right spot can still choose who blocks whom.

Notice what the stunt did and did not do in Figure 39.6. It did not produce a free rusher: the slide protection handled it exactly as designed, because the right tackle owned a gap, not a man, and simply took whoever came into it. That is slide protection’s strength: gaps do not move. What the stunt did instead was choose the matchup. The protection had assigned the back to the end. The stunt swapped the end for a defensive tackle who outweighs the back by ninety pounds, and the back lost. No one was unblocked, and the quarterback was still pressured.

Run the same T-E into the slide side, where the center, guard and tackle are all stepping toward it, and it dies: the guard owns the B gap the end loops into, the tackle owns the C gap the 3-technique crashes into, and each blocker simply takes the man who arrives. So defenses aim stunts away from the slide, at the man side of a half-slide, or at a specific blocker: a slow-footed guard, a rookie tackle, a back who has to take a lineman.

How does the defense know which way the slide will go before the snap? It mostly doesn’t know; it bets on clues. Film shows how an offense protects from each formation and each back alignment, so the back’s spot is a clue to the offense’s habits. Before the snap the center points out the linebacker he is counting from, which hints at the slide’s direction. And many defenses simply shift or align late, so the slide is called against a picture that changes. Line coaches talk about stunts the way offensive coordinators talk about plays, as weapons chosen for a particular opponent.

Stunts have three costs, and they are why defenses do not stunt on every snap.

  • They take time. The looper runs a longer path than a straight rush, so a stunt usually arrives later, about two and a half seconds or more after the snap. Against a quick throw, a stunt is a defense that rushed two men late and two men sideways. Stunts belong on long-yardage downs, when the quarterback must hold the ball for routes to develop.
  • They give up the one-on-one. The penetrator’s job is to occupy blockers, not to win, so a great rusher who is asked to penetrate on a stunt is a great rusher not rushing. Defenses often build stunts so their best player is the looper.
  • They open lanes. Two rushers crossing paths leave, for a moment, a space where neither of them is, and a quarterback who can run will find it. That is the next subject.

Rush lanes and contain

A four-man rush is not four men running at the quarterback. It is four men each responsible for a lane, and the lanes together form a box around him.

A rush lane is the path to the quarterback that one rusher owns. The two ends usually own the outside lanes and have contain: they rush outside the tackles but must not run past the quarterback’s depth, because the space behind them would become a running lane. The two interior rushers own the middle and have the job of pushing the pocket straight back, so the quarterback cannot step up. If every rusher stays in his lane, the quarterback has nowhere to go: forward is the interior push, sideways is the edge rushers, backward is a longer throw. If one rusher leaves his lane, there is a hole.

The trade in Figure 39.7 is the oldest one in pass rushing. A rusher who stays disciplined in his lane gives the quarterback nowhere to go, but he also gives up some of his own chance to get home. A rusher who abandons his lane for the sack, whether by running too far upfield, spinning inside out of an outside lane, or looping on a stunt, may get there, or may open the door. Against a quarterback who will stand in the pocket, the gamble usually pays. Against one who runs, it often doesn’t. Defenses guard the contain on the quarterback’s throwing-arm side hardest (the offense’s right, for a right-hander), because he can escape that way and still throw on the run.

Stunts make the lanes move. On a T-E, the penetrator crashing outside takes over the outside lane and with it contain, while the end who usually has contain loops inside. If the penetrator does not get there, or gets there too deep, the edge is open. That is part of why stunts are rarer against running quarterbacks, and why the stunts defenses do run against them are often the ones that keep a fast player outside.

The mush rush: squeezing instead of racing

Keeping lanes is ordinary discipline: every good four-man rush does it, and every end in it is still trying to beat his tackle to the corner and then stop at the quarterback’s depth. Against a quarterback who is a real threat to run, defenses often go further and change what the rush is for. In a mush rush (some staffs say cage rush), the ends stop trying to win the corner at all. Each one stays on his tackle’s outside shoulder, in front of him, and pushes him straight back, with a bull rush or a long arm, until the tackle is standing beside the quarterback. The inside rushers do the same to the guards. The pocket is squeezed from all sides rather than raced into, and the goal is no longer to sack him quickly. It is to make sure that when the routes are covered and he looks for somewhere to run, there is nowhere to go.

The word “mush” sounds passive, and the mush rush is often criticised as timid when it fails. But watch it work against a great runner and it is anything but. The rushers are still fighting their blockers; they are fighting them for position rather than for a path around them. The quarterback ends up throwing from a shrinking space with no lanes, or running into a defender who was waiting for exactly that.

The cost is the clock. A mush rush produces fewer quick pressures and fewer sacks, so the coverage behind it has to hold for four or five seconds instead of two and a half. Against a quarterback who is both a great runner and a patient passer, that is a miserable choice, which is why such quarterbacks are so hard to defend. Some defenses add the spy you met in Linebackers, a defender assigned only to mirror the quarterback; it buys insurance at the cost of a man in coverage or in the rush.

CautionCommon misconception

“A mush rush means the defense isn’t trying to get to the quarterback.” It means the defense has decided where it wants the quarterback to be, and has accepted that he will get there later. Against a quarterback who runs for a first down every time a lane opens, a sack is less valuable than an escape is costly.

Measuring a pass rush: why pressure beats sacks

A sack, the quarterback tackled behind the line before he can throw, is the pass rush’s famous statistic, and for good reason: it is one of the most damaging plays in football. It costs yards, a down and often the drive, and sometimes the ball. The NFL has counted sacks officially only since 1982;6 the single-season record is the 23.0 that Myles Garrett set in 2025, breaking the 22.5 shared by Michael Strahan (2001) and T.J. Watt (2021).7

But as a way to judge a pass rush, sacks are a poor instrument, and the reason is simple arithmetic. A sack happens only when three things go right for the defense: a rusher gets to the quarterback (pressure), the quarterback fails to get rid of the ball or escape, and the rusher finishes the tackle. Only the first is mostly about the rush. So analysts split it in two:

\[\text{sacks} = \text{pressures} \times \text{the share of pressures that become sacks}\]

Pressure, as Pass Protection defined it, is any rusher forcing the quarterback to change what he wanted to do: a sack, a hit or a hurry. In FTN’s charting of the 2023–2025 regular seasons, quarterbacks were pressured on 30% of dropbacks and sacked on 6.6%, so only about 22% of pressures, roughly one in 5, became sacks.

Does a pressure that is not a sack matter? A great deal. Over the same seasons, a clean dropback was worth +0.23 expected points to the offense on average, a pressured dropback that did not end in a sack -0.07, and a sack -1.77. Quarterbacks completed 70% of their throws when clean and 47% when pressured.8 (Some of that gap is selection, since long-developing plays get pressured more often, but the direction is not in doubt.) The hurry that nobody counts turns a good play for the offense into a slightly bad one, on many more snaps than sacks ever do. So the question is what decides the conversion from pressure to sack, and whether it belongs to the rush at all.

CautionCommon misconception

“Sacks tell you who the best pass rushers are.” They tell you who finished plays, which is real but partial. Many sacks are clean-up sacks: an interior rusher collapses the pocket, the quarterback slides away from him into an edge rusher, and the edge rusher gets the credit. Others are coverage sacks, where the quarterback held the ball for four or five seconds because nobody was open, and whoever arrived first got the stat. And some of the best rushes in a game, like the one that opened this chapter, produce a throw a beat too early and no statistic at all. Watch for who caused the pressure, not just who was standing over the quarterback.

Pass rush win rate, get-off and time to pressure

If sacks are too noisy and pressure is better, can we do better still? Pressure has its own flaw: it depends on the clock. A rusher who beats his man cleanly in 1.5 seconds against a quarterback who throws in 1.3 gets no pressure, and a rusher who is stonewalled for four seconds and then wanders in while the quarterback pats the ball gets one. Tracking data allows measures that do not depend on what the quarterback does.

Pass rush win rate. ESPN’s pass rush win rate (PRWR), introduced in 2018 by ESPN Analytics and built on Next Gen Stats tracking data, asks one question of every rush: did the rusher beat his block within 2.5 seconds of the snap?11 Its companion, pass block win rate, asks the reverse of every blocker. The fixed 2.5-second clock is the clever part. It sits a little before the average throw, ESPN found it best separated the rushers and blockers already known to be good, and it takes the quarterback out of the number: a quarterback who holds the ball cannot hand rushers late “wins,” and one who throws quickly cannot hide a rusher who was about to get home. ESPN publishes it as leaderboards ranked within position groups, so edge rushers are compared with edge rushers and interior linemen with interior linemen, because their jobs are so different. For a scale: when ESPN introduced it, four weeks into the 2018 season, the average edge rusher won about 22% of his rushes and the leaders were in the 40s. A rusher who wins one rush in three is very good; one in five is ordinary. Double teams get special care: since 2019, beating one blocker of a double team no longer counts as a win, which is why an interior rusher who wins often against constant doubles is so impressive.

Its limits are worth knowing. A “win” is not a pressure; a rusher can beat his man and still be too late or too wide to matter. A bull rush that walks the tackle into the quarterback without ever shedding him is the kind of rep a “did he beat his block” measure can miss. The model does not know each rusher’s assignment, so a penetrator on a stunt, whose job is to occupy blockers, can look like a loser while doing exactly what was asked. And the play-by-play data is not public, only the leaderboards, so you cannot rebuild it yourself. Use it the way you would use a scout’s opinion: an excellent tiebreaker between players, not a final verdict.

Get-off and time to pressure. Two simpler tracking measures bracket the rush. Get-off, as above, is how quickly a rusher crosses the line after the snap. Time to pressure is the seconds from the snap to the first pressure on the play. Together they separate the rushes that won from the ones that were handed something. A pressure inside about two and a half seconds, before most throws are due, is almost always a rush win. One after three and a half seconds, when every route has finished, is usually the coverage, the quarterback holding the ball, or both. In between is shared credit: the routes are just finishing, and the rush and coverage are working together. Here is how each is computed from tracking data, on a simulated four-man rush.

Where the moves came from

The modern pass rusher’s ancestor is Deacon Jones, the Los Angeles Rams’ defensive end of the 1960s, a ferocious speed rusher who is credited with coining the word “sack” for what he did to quarterbacks. His signature weapon was the head slap: a blow to the side of the blocker’s helmet at the snap to stun him for an instant, then go around him. The NFL banned the head slap in 1977, and blows to the head have been illegal for rushers ever since.12 A year later the 1978 rule changes went the other way for blockers, letting them extend their arms and open their hands on pass blocks (The 1978 Liberation tells that story). Together, the two changes turned the pass rush into what it is now: a fight between the rusher’s hands and the blocker’s, aimed at arms and chests, where technique and timing matter as much as violence.

Then came the player who made offenses rebuild their protections around him: Lawrence Taylor, the Giants’ outside linebacker from 1981, rushing from a two-point stance on the edge of a 3-4 with a speed no blocker of the time could match (Pass Protection told how offenses answered him with extra blockers). In 1986 he had 20.5 sacks and was the league’s Most Valuable Player, the last defensive player to win the award,13 and the value of the left tackle, who blocks a right-handed quarterback’s blind side, rose with him. Defenses still like a great rusher on that side, but many of today’s best flip sides from week to week, or snap to snap, to hunt whichever tackle is weaker.

Reggie White, “the Minister of Defense,” showed what power and hands could do. After fifteen seasons, most of them with the Eagles and the Packers, he retired as the NFL’s career sack leader, with 198 (Bruce Smith later passed him, finishing with 200), and his signature, the “hump” move, was a club-and-lift: a forearm club up under the blocker’s arm that lifted and threw him aside, a rip and a bull rush in one violent motion.14 In the 2000s Dwight Freeney’s spin move made a smaller, quicker body type dangerous on the edge. And in the 2010s Aaron Donald proved that the most dangerous rusher on the field could line up inside.

Today’s edge rushers study tackles’ sets on film like quarterbacks study coverages, and the hand fight has become its own discipline: the long arm, the cross-chop, the ghost. T.J. Watt tied the sack record in 2021 and won Defensive Player of the Year; Myles Garrett broke it in 2025 and won the award unanimously.15 And Micah Parsons, the Cowboys’ first-round pick in 2021, made a different point: drafted as an off-ball linebacker, he became one of the league’s most feared rushers by lining up all over the front, over a guard one snap and wide on the edge the next, so that protections could never be sure where to slide. Dallas traded him to Green Bay just before the 2025 season, for two first-round picks and defensive tackle Kenny Clark.16

NoteFilm room: Super Bowls LIX and LX — pressure with four

The games. The last two Super Bowls were won by defenses that got to the quarterback without blitzing. In Super Bowl LIX (February 2025) the Eagles rushed four or fewer on 44 of Patrick Mahomes’s 45 dropbacks and sacked him 6 times, 6 of them with four or fewer rushers. In Super Bowl LX (February 2026) the Seahawks rushed four or fewer on 46 of Drake Maye’s 54 dropbacks and sacked him 6 times, 5 of them with four or fewer.19 Mike Macdonald’s Seattle defense has a reputation for four-man simulated pressures, in which a lineman drops and a linebacker or defensive back rushes in his place (Zone Blitzes and Simulated Pressure shows how). Macdonald himself pushed back on that in November 2025 (“We do not run a lot of simulated pressures”),20 and the Super Bowl bears him out: FTN’s charters counted a blitzer on only 4 of Seattle’s 46 four-man rushes. Mostly, its four were its front four, winning.

What to notice:

  • Count the rushers on every dropback. Almost always four. Then count the coverage behind them: seven men for at most five receivers.
  • Look for the stunts. On long downs, watch for two rushers crossing paths, and ask which blocker the looper was aimed at.
  • Time the pressure. Count from the snap to the moment the quarterback has to move, and see how many pressures arrive before the receivers have finished their routes.

Watch for it on Sunday

TipWatch for it

Before the snap.

  • Find the best rusher and his alignment. Wide outside the tackle (built for speed), tight over him (power and inside moves), or standing up over a guard (he is being moved around to beat the protection).
  • Count the rushers you expect and the blockers you see. Four rushers against five or six blockers means the defense is trusting its four; look at the coverage behind them.
  • Check the down and distance. Third-and-long is when stunts and patient rushes have time to work; on third-and-2 the ball will be out too fast for either.
  • Is the quarterback a runner? If so, expect the ends to rush with discipline, and maybe a spy.
  • Where is the help? Find the back and any tight end. One aligned near the best rusher probably means a chip, and at the snap the linemen’s first steps tell you slide or man (all five stepping the same way is a slide). Stunts and the best rusher’s moves are aimed away from the help.

After the snap: the stopwatch drill.

  • Count the seconds from the snap to the first pressure (“one-Mississippi”), or use a stopwatch on a replay. Under about two and a half seconds is a rush win. Past three and a half, the coverage or the quarterback owns most of it. In between, they share it.
  • Note whether the ball was already out. A rusher who arrives just after the throw has still done his job if he made the quarterback throw early; a rusher who arrives after a five-second hold was handed it.
  • Name the move. Around the corner (speed and rip), through the chest (bull or long arm), or back inside (swim, spin). In the fourth quarter, look for the counter to whatever worked in the first.
  • Watch for crossing paths. Two rushers swapping lanes is a stunt; find the looper, and see which blocker he was aimed at, and whether the blockers passed it off.
  • Check the lanes. If the quarterback escapes, freeze the replay and find the rusher who left his lane.

For your prediction log: for one quarter, write down on every dropback the number of rushers, your stopwatch time to pressure (or “none”), and whether the ball was out first. You will see which defenses win early and which only win late.

How do you expect the four to rush, which of them has the hardest job, and what is the defense giving up?

TipAnswer

Expect a mush rush. On third-and-6, a quarterback who runs does not need a receiver to get open; he needs one lane. So the defense’s first priority is that no lane opens, and its second is the sack.

The wide-9 end has the hardest job, because his alignment is built for the long, fast arc that runs an end past the quarterback and opens the escape lane in Figure 39.7, on the quarterback’s throwing-arm side at that. Expect him to tighten his alignment before the snap, or, if he stays wide, to rush through the tackle’s outside half rather than around him. The other side is less of a worry: the back, lined up on the quarterback’s left, will likely chip the left end or stay in to block, and the tight end beside the right tackle can chip the wide-9 on his way out, which is one more reason that end cannot win with speed alone. Expect fewer stunts, and don’t be surprised if a linebacker mirrors the quarterback instead of covering anyone: a spy.

The price is time: the coverage, with six or seven men, must hold for four seconds or more, and if anyone breaks contain, the play you will see is the one the defense feared most, a quarterback through the hole and past the line to gain.

What move is coming, and what can the offense do about it, this snap and the next?

TipAnswer

The speed rush, around the corner: a dip and rip. This is the rush plan paying off from the other direction. Twice the rusher won inside, so the tackle has stopped bailing: he is sitting short and square to protect the inside. But a tackle who has not gained depth has no time to get back to the corner, and the rusher meets him level before he can turn and run. The spin is the wrong answer now; that is exactly the move the tackle is sitting on.

The offense has fewer answers on this snap than you might think. The right guard cannot help outside, because the 3-technique is on him. The quarterback’s best move is to step up into the pocket, so the rusher’s arc carries him past behind. Next snaps: the tackle varies his sets (a jump set at the line, or a 45-degree set instead of sitting) so the rusher cannot read the set and pick the counter; the offense chips the rusher with a back or tight end on the way out; and the play caller calls quick throws, so the ball is gone before any move can finish. That is the conversation across a game in miniature: the rusher made the tackle afraid of one thing, and now the offense has to change the question.

Back to the replay

Return to the play that opened the chapter. Third-and-8, four rushers, an end who took three strides up the arc and then drove straight into the tackle: speed-to-power, set up by every speed rush he had shown before. The tackle, fearing the corner, had kicked back fast, so his weight was going backward when the end turned into him, and he could not anchor. The pocket collapsed from the side, the quarterback threw early, and the pass sailed.

The box score has nothing for the end. FTN’s charters would mark that dropback as pressured. Pass rush win rate might well count it as a win, though a bull rush that never sheds the blocker is the kind of rep a beat-the-block measure can miss. And a stopwatch would put the pressure at about two seconds, early enough that it was entirely the rush’s doing. That is the whole chapter in one snap: the moves, the plan that set them up, and the reason a coach values the hurry that nobody counts more than the sack total everybody does.

Takeaways

  • The rush is a race and a fight. A rusher must beat a bigger blocker before the ball is out, a little under three seconds on average. Get-off decides where they meet: a tenth of a second is most of a yard, and meeting the tackle level at the corner instead of in front of him is the difference between winning and losing.
  • Each move punishes a mistake. Speed beats a slow set; the bull rush and long arm beat a high, light or retreating one; the swim, rip and spin beat a blocker who has committed. Cross-chops and ghosts beat the punch itself.
  • The best rushers have a plan. They show one move until the tackle adjusts, then use the counter his adjustment opens; speed-to-power works because tackles fear speed.
  • Stunts win with geometry. T-E, E-T and interior twists exchange rushers to beat man blocking or choose matchups against slides, at the cost of time and of lanes.
  • Four rush lanes make a box. Ends keep contain, the middle pushes the pocket; against a quarterback who runs, the mush rush squeezes instead of racing and accepts a longer clock.
  • Count pressure, not sacks. Pressure carries over from season to season and predicts future sacks; the share of pressures that become sacks mostly does not, except as a trait of the quarterback. Pass rush win rate, get-off and time to pressure use tracking data to take the quarterback out of the measure.
