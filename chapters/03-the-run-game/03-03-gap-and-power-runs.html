PURPLE = style.SERIES[6]  # a ring colour for pullers: neither offense blue nor defense orange
OL_W = {"LT": -3.2, "LG": -1.6, "C": 0.0, "RG": 1.6, "RT": 3.2}


def pdf_build():
    return os.environ.get("RTG_FORMAT", "").lower() in ("pdf", "typst", "print")


def O(key, pos, d, w, label):
    return Player(key, pos, "off", d, w, label=label)


def D(key, pos, d, w, label):
    return Player(key, pos, "def", d, w, label=label)


def line():
    """The five offensive linemen, labelled LT, LG, C, RG, RT."""
    return [O(k, k, -0.7, w, k) for k, w in OL_W.items()]


def te(side=1, key=None, label=None, d=-0.7):
    """An in-line tight end: Y on the right, U on the left."""
    key = key or ("Y" if side > 0 else "U")
    return O(key, "TE", d, side * 4.8, label or key)


def T(t, side=1):
    """Width of technique t on the offense's right (side=+1) or left (side=-1)."""
    return technique(t, side)


def dl(key, t, side, label=None):
    """A defensive lineman in technique t, labelled with the number."""
    return D(key, "DL", 1.0, T(t, side), label or t)


def go_to(p, key, end, kind="block", delay=0.0, speed=None, via=()):
    """Move `key` through absolute (d, w) points `via` to `end`, as a block, pull, path or run."""
    start = np.array(p._cur(key))
    pts = [tuple(np.array(q) - start) for q in (*via, end)]
    if kind == "block":
        return p.block(key, pts=pts, delay=delay, speed=speed)
    if kind == "pull":
        return p.pull(key, pts, delay=delay, speed=speed)
    return {"path": p.path, "run": p.run}[kind](key, pts, delay=delay, speed=speed)


def _P(fld, p, d, w):
    return fld.xy(*p.to_abs(d, w))


def say(fld, p, d, w, text, color=style.INK_2, fontsize=7.2, ha="center", weight="normal"):
    """A short on-field label at (d, w)."""
    x, y = _P(fld, p, d, w)
    fld.ax.text(x, y, text, fontsize=fontsize, color=color, ha=ha, va="center", zorder=9, fontweight=weight,
                bbox=dict(boxstyle="round,pad=0.18", fc=style.SURFACE, ec="none", alpha=0.9))


def point_to(fld, p, text_at, target, text, color=style.INK_2, fontsize=7, shrink=7):
    """A label with a thin leader line to the player or spot it names."""
    (xt, yt), (xp, yp) = _P(fld, p, *text_at), _P(fld, p, *target)
    fld.ax.annotate(text, (float(xp), float(yp)), xytext=(float(xt), float(yt)), fontsize=fontsize,
                    color=color, ha="center", va="center", zorder=9,
                    arrowprops=dict(arrowstyle="-", color=color, lw=0.7, shrinkA=2, shrinkB=shrink),
                    bbox=dict(boxstyle="round,pad=0.18", fc=style.SURFACE, ec="none", alpha=0.9))


def pair_oval(fld, p, d, w, width, height=2.5, color=GREEN):
    """A faint oval around a double team (two blockers and their defender)."""
    x, y = _P(fld, p, d, w)
    fld.ax.add_patch(Ellipse((float(x), float(y)), width, height, fc=color, alpha=0.22, ec="none", zorder=1.5))
    fld.ax.add_patch(Ellipse((float(x), float(y)), width, height, fc="none", ec=color, lw=1.6,
                             ls=(0, (4, 2)), zorder=1.6))


def spot(fld, p, d, w, color=style.THIRD, ms=9):
    """Mark a point (the hole, the point of attack) with an X."""
    x, y = _P(fld, p, d, w)
    fld.ax.plot([x], [y], marker="X", ms=ms, color=color, mec="white", mew=0.8, zorder=9)


def shade(fld, p, w0, w1, d0=-0.2, d1=6.5, color=style.THIRD, alpha=0.08):
    """Shade a rectangle from width w0 to w1 and depth d0 to d1 (the box, the play side)."""
    (x0, y0), (x1, y1) = _P(fld, p, d0, w0), _P(fld, p, d1, w1)
    fld.ax.add_patch(Rectangle((min(x0, x1), min(y0, y1)), abs(x1 - x0), abs(y1 - y0), fc=color, alpha=alpha,
                               ec=color, lw=0.9, ls=(0, (3, 2)), zorder=0.8))


def dashed(fld, p, pts, color=style.THIRD, lw=1.6, dash=(0, (3, 2.2)), end="arrow"):
    """A dashed 'option' or 'intent' path through absolute (d, w) points, ending in an arrow or a block bar."""
    xs, ys = _P(fld, p, np.array([q[0] for q in pts]), np.array([q[1] for q in pts]))
    if end == "arrow":
        fld.ax.plot(xs[:-1], ys[:-1], color=color, lw=lw, ls=dash, zorder=6)
        fld.ax.add_patch(FancyArrowPatch((xs[-2], ys[-2]), (xs[-1], ys[-1]), arrowstyle="-|>",

    else:
        fld.ax.plot(xs, ys, color=color, lw=lw, ls=dash, zorder=6)
        v = np.array([xs[-1] - xs[-2], ys[-1] - ys[-2]])
        perp = np.array([-v[1], v[0]]) / np.linalg.norm(v) * 0.75
        fld.ax.plot([xs[-1] - perp[0], xs[-1] + perp[0]], [ys[-1] - perp[1], ys[-1] + perp[1]],

def clean(ax):
    """Box-zoomed diagrams: drop hash ticks and yard numbers, clip anything outside the window."""
    for ln in list(ax.lines):
        xs, ys = ln.get_data()
        if len(xs) == 2 and ln.get_linewidth() == 0.5 and np.hypot(xs[1] - xs[0], ys[1] - ys[0]) < 1.0:

def i_form(side=1):
    """I-formation, 21 personnel, with the tight end on `side` (wide receivers outside the picture)."""
    return line() + [O("QB", "QB", -1.9, 0, "Q"), O("FB", "FB", -4.9, 0, "F"), O("RB", "RB", -7.4, 0, "R"),
                     te(side)]


def pistol(y_side=0):
    """Pistol backfield (QB about 3.5 yards deep, RB behind him) with in-line tight ends:
    y_side=+1 the Y on the right, -1 the U on the left, 0 both."""
    off = line() + [O("QB", "QB", -3.5, 0, "Q"), O("RB", "RB", -7.0, 0, "R")]
    if y_side >= 0:
        off.append(te(+1))
    if y_side <= 0:
        off.append(te(-1))
    return off


def over_front(side=1, safeties=True):
    """A 4-3 Over (named in 05-02), strength toward `side`: on the line a 5, a 1, a 3 and a 5, with the
    Sam standing up in a wide 9 outside the tight end; the Mike over the strong A gap, the Will over the
    weak B gap; two deep safeties."""
    s = side
    out = [dl("WE", "5", -s), dl("NT", "1", -s), dl("DT", "3", s), dl("SE", "5", s),
           D("SAM", "SAM", 1.0, s * 7.0, "S"), D("MIKE", "MIKE", 4.6, s * 0.9, "M"),
           D("WILL", "WILL", 4.6, -s * 2.5, "W")]
    if safeties:
        out += [D("FS", "FS", 10.5, -s * 7.0, "FS"), D("SS", "SS", 10.5, s * 7.0, "SS")]
    return out


def four_three_base(sam=True):
    """A four-man even front, 5-1-3-5 from the offense's left, with the 3-technique to the right."""
    out = [dl("BE", "5", -1), dl("NT", "1", -1), dl("DT", "3", 1), dl("PE", "5", 1),
           D("WILL", "WILL", 4.6, -2.4, "W"), D("MIKE", "MIKE", 4.6, 0.9, "M")]
    if sam:
        out.append(D("SAM", "SAM", 4.2, 6.4, "S"))
    return out

MARK_R = 0.6


def tracking(play):
    df = play.to_tracking()
    df["_id"] = df["displayName"].astype(str)
    return df


def panel(ax, play, df, t, lateral, window, rings=()):
    """One moment of a simulated play on `ax`: markers, trails of on-screen players, rings."""
    tmap = df.groupby("frameId")["time"].first()
    fid = int((tmap - t).abs().idxmin())
    ylo, yhi = sorted((play.ball_y - lateral[0], play.ball_y - lateral[1]))
    fld = Field(ax=ax, orient="vertical", los=play.los, window=window, numbers=False, hashes=False,
                lateral=(ylo, yhi), to_go=play.to_go)
    cur = df[df["frameId"] == fid]
    shown = cur[cur["x"].between(play.los + window[0] + 0.8, play.los + window[1] - 0.8)
                & cur["y"].between(ylo + 0.8, yhi - 0.8)]              # only players whose marker fits whole
    ids = set(shown["_id"])
    trails = df[(df["frameId"] <= fid) & df["_id"].isin(ids) & (df["side"] != "ball")
                & df["x"].between(play.los + window[0], play.los + window[1])]
    for (_, side), g in trails.groupby(["_id", "side"]):
        xs, ys = fld.xy(g["x"].values, g["y"].values)
        ax.plot(xs, ys, color=style.OFFENSE if side == "off" else style.DEFENSE, lw=0.9, alpha=0.4, zorder=3)
    ppy = fld.pts_per_yard()
    for _, r in shown.iterrows():
        x, y = (float(v) for v in fld.xy(r["x"], r["y"]))
        if r["side"] == "ball":
            ax.add_patch(Ellipse((x, y + 0.45), 0.7, 0.42, fc=style.BALL, ec="white", lw=0.5, zorder=9))
            continue
        lab = str(r.get("label", "") or "")
        if r["side"] == "off":
            ax.add_patch(Circle((x, y), MARK_R, fc=style.OFFENSE, ec="white", lw=0.6, zorder=8))
            tc = "white"
        else:
            ax.add_patch(RegularPolygon((x, y), 4, radius=MARK_R * 1.3, orientation=np.pi / 4,
                                        fc=style.DEFENSE_TINT, ec=style.DEFENSE, lw=1.0, zorder=8))
            tc = style.INK
        fs = max(4.0, min(8.0, ppy * (0.85 if len(lab) <= 1 else 0.6)))
        ax.text(x, y, lab, ha="center", va="center", fontsize=fs, color=tc, fontweight="bold", zorder=10)
    for key, col in rings:
        r = cur[cur["displayName"] == key].iloc[0]
        x, y = fld.xy(r["x"], r["y"])
        ax.add_patch(Circle((float(x), float(y)), 0.95, fc="none", ec=col, lw=1.5, zorder=7))
    return fld


def strip(play, times, lateral, window, notes=None, rings=(), ncols=2, width_in=6.5):
    """Numbered frame strip with a fixed, box-sized window, so markers stay legible in print."""
    df = tracking(play)
    nrows = int(np.ceil(len(times) / ncols))
    pw = width_in / ncols
    span_w, span_d = lateral[1] - lateral[0], window[1] - window[0]
    fig, axes = plt.subplots(nrows, ncols, squeeze=False,
                             figsize=(width_in, nrows * (pw * span_d / span_w + (0.55 if notes else 0.25))))
    for ax in axes.flat[len(times):]:
        ax.axis("off")
    for i, (ax, t) in enumerate(zip(axes.flat, times)):
        panel(ax, play, df, t, lateral, window, rings)
        ax.set_title(f"{i + 1}.  " + ("snap" if abs(t) < 1e-6 else f"{t:+.2f} s".rstrip("0")),
                     fontsize=8.5, color=style.INK_2, loc="left")
        if notes:
            ax.text(0.5, -0.03, notes[i], transform=ax.transAxes, ha="center", va="top",
                    fontsize=7.4, color=style.INK)
    fig.tight_layout(h_pad=1.3 if notes else 0.6, w_pad=0.6)
    return fig


def video(play, times, lateral, window, notes=None, rings=(), width_in=5.4):
    """Web version of the strip: the same panel at every frame, with the strip's notes as captions."""
    import shutil
    from matplotlib import animation
    from IPython.display import HTML, display
    df = tracking(play)
    span_w, span_d = lateral[1] - lateral[0], window[1] - window[0]
    fig, ax = plt.subplots(figsize=(width_in, width_in * span_d / span_w + 0.4))
    tmap = df.groupby("frameId")["time"].first()

    def update(fid):
        ax.clear()
        t = float(tmap[fid])
        panel(ax, play, df, t, lateral, window, rings)
        txt = ("snap" if abs(t) < 0.05 else f"{t:+.1f} s")
        if notes:
            cur = [n for tt, n in zip(times, notes) if tt <= t + 1e-6]
            txt += ("   " + cur[-1]) if cur else ""
        ax.set_title(txt, fontsize=9, color=style.INK, loc="left")
        return []

    anim = animation.FuncAnimation(fig, update, frames=sorted(df["frameId"].unique()), interval=100)
    html = (anim.to_html5_video(embed_limit=40).replace("<video ", '<video style="max-width:100%;height:auto" ')
            if shutil.which("ffmpeg") else anim.to_jshtml(default_mode="once"))
    plt.close(fig)
    display(HTML(html))


def animation_or_strip(play, times, **kw):
    """Video on the website; a numbered, captioned frame strip in the PDF."""
    if pdf_build():
        strip(play, times, **kw)
        plt.show()
    else:
        kw.pop("ncols", None)
        kw.pop("width_in", None)
        video(play, times, **kw)

Third-and-1, the offense in a heavy formation with a fullback and a tight end, everyone in the stadium knowing it will run. At the snap the right side of the offensive line fires inward, at an angle, away from where the ball is going. At the same instant the left guard, who should be blocking the man in front of him, turns and sprints sideways behind his own line. The fullback heads for the edge. The running back waits a beat, then follows the guard into a hole just outside the tight end, and the guard turns up into it and meets a linebacker head-on. Four yards, first down. The announcer says the offense “just lined up and ran it right at them.” True, but not the interesting part. Everyone knew where the ball was going, so how did the offense get more blockers to that spot than the defense had defenders? The answer is a family of plays coaches call gap schemes, and the guard who ran sideways is the key to every one of them.

You’ll be able to:

  • Recognize power, counter, duo, trap, wham, iso, the toss and sweep, pin-and-pull, and the draw on a broadcast, and draw each against a front you are shown by applying its blocking rules.
  • Count the blockers and defenders at the point of attack and show how a gap scheme gains a man there, and what it pays for him.
  • Say when a coach is likely to call a gap run rather than a zone run (the front, the personnel, the offensive line, the situation).
  • Spot the pre-snap and first-step tells of a pulling lineman, and call the play’s direction from them.

How to read the diagrams. Offense is blue circles, defense orange squares, and the offense moves up the page. Left and right are always the offense’s left and right. LT, LG, C, RG and RT are the five offensive linemen; Q is the quarterback, R the running back (HB a halfback, in the 1960s sweep), F a fullback, and Y and U are tight ends on the right and left. H is the “move” player from Formation Rules: an H-back in some diagrams, a slot receiver in others; each caption says which. Defensive linemen carry their technique number; S, M and W are the Sam, Mike and Will linebackers; FS and SS the safeties; CB a cornerback; NB the nickel back. Lines ending in a short bar are blocks; plain arrows are paths; the thick dark arrow is the ball carrier. A purple ring marks a puller (or another extra blocker), the player this chapter wants you to follow. Most diagrams zoom in on the box, so the wide receivers (who keep the formation legal, with seven men on the line) are often outside the picture.

What makes a run a “gap” run?

In Zone Running the offensive line asked one question after the snap: who comes into my area? A gap scheme asks its question before the snap: which man do I block, and from which angle? That is why gap blocking is also called man or power blocking. A gap scheme (also man/power blocking) assigns each blocker a man or a gap in advance, picks those assignments to give blockers good angles, and brings extra blockers to a hole the coach chose before the snap.

That definition is the whole family resemblance, and it is worth reading twice, because it says nothing about pulling. What makes a run a gap run is that every blocker knows his man before the snap, and the hole is chosen in advance. Some gap runs pull two linemen; duo, iso and the draw pull nobody. They are still gap runs because their blockers are assigned to men, not areas.

The classic gap run (power, counter, the sweep) is assembled from three parts you met in 03-01, and the rest of the family drops or swaps one of them:

  1. Down blocks on the play side. The linemen at the point of attack block the defender inside them, hitting him from the side. A down block is the easiest block in football to win, because the blocker has an angle and the defender is looking straight ahead. Down blocks build the inside wall of the hole.
  2. A kick-out on the end man. Somebody (a fullback, an H-back, a pulling guard) hits the end man, the last defender on the line (the kick-out; Option Football gives the full rule for finding him), from the inside and drives him toward the sideline. That builds the outside wall.
  3. A puller through the hole. A lineman from the backside runs behind his own line, turns up between the two walls and blocks the first linebacker to show. He is the extra man.

Down blocks in, kick-out out, puller up through the middle. The hole is fixed and known to everyone on the offense, including the running back, who does not have to find it; he has to follow someone into it. Contrast zone, where the hole is wherever the wall cracks and the back’s read is half the play.

The lineman who leaves his spot is the puller: the guard (or tackle, or both) assigned to pull on this play. He is the most important player on the field for you to find, because a puller runs to the hole. The rest of this chapter is, in a sense, a catalogue of where pullers go.

ImportantWhy it exists

Gap schemes exist to win the numbers at the point of attack. A defense can always put as many men in the box as the offense has blockers, and usually one more (hat math). It cannot be equally strong everywhere, though, and it has to line up before it knows where the ball is going. A gap scheme picks one spot, then moves blockers there from elsewhere (a backside guard, a fullback) faster than the defense can move defenders there. It pays for those men by leaving someone away from the play unblocked, and by being more predictable: a pulling guard is a signpost. Every gap play in this chapter is a different answer to the same two questions: where do the extra blockers come from, and who is left free?

Power: the play the whole family is built on

Power (often called Power O) is the purest gap run, and the one to learn first. Here it is from the I-formation (a fullback behind the quarterback and the tailback behind him) against a 4-3 Over, a four-man front whose 3-technique lines up on the side of the tight end (fronts are named and explained in Even Fronts; here it is just a set of numbers to block). Read it from the right, the play side, to the left. The hole is off-tackle, the old name for “just outside the tackle”: here that is the D gap, outside where the tight end lined up (with no tight end on that side, the same play hits the C gap).

Walk through Figure 17.1 one man at a time, because each assignment exists for a reason.

The play-side wall: down blocks. The tight end blocks down on the 5-technique inside him, and the right tackle blocks down on the 3-technique inside him. Neither has to beat a defender lined up across from him; each hits a man from the side, which is why gap teams love the down block. The right guard joins the tackle on the 3-technique for a moment, a combo block, and then climbs to the backside linebacker, the Will. Coaches teach these linemen one short rule, usually some version of gap, down, backer: if a defender is in your inside gap, block him; if not, block down on the man over the lineman inside you; if nobody is there, go to a linebacker.1 The rule means a play-side lineman never has to reach a man lined up outside him.

The center blocks back. The left guard is about to leave. The defensive lineman who was across from the gap he leaves, here the 1-technique, would walk into the backfield, so the center blocks back on him, away from the play. Because the 1 is shaded on the center’s backside shoulder, it is another easy angle. This is not a coincidence: coaches like to call power toward a 3-technique and away from a 1-technique for exactly these angles.

The kick-out. The fullback runs at the end man (here the Sam, standing up on the line outside the tight end in a 9-technique) and hits him from the inside, driving him toward the sideline. The fullback does not need to win big. If the Sam stays outside, the hole is inside him.

The puller. The left guard opens his hips toward the play with a short step back and out (coaches call it a bucket step or a skip pull), sprints flat behind the center and the right side, staying tight to his teammates’ down blocks (“hug the down blocks”), and turns up into the hole, just outside the tight end’s down block and inside the kick-out, with his eyes inside for the first defender who appears. His man is the first linebacker who shows, almost always the play-side linebacker (the Mike here) coming to fill. The running back follows him, aiming his first steps at the tight end’s outside hip and staying a step behind the guard’s back, then cutting off the guard’s block.

The backside. With the guard gone, the left tackle hinges: he steps inside to close the B gap the guard left, then swings back outside like a door closing, to keep the backside end from running straight down the line. He does not try to drive the end anywhere. The end is the one defender power chooses not to block. That is the deal at the heart of the play.

Figure 17.2 is the reason gap schemes exist. Before the snap, the side the offense wants to run to has three blockers and four defenders. Run zone that way and somebody has to beat a man he is outnumbered against. Run power and two blockers who started on the other side of the ball arrive at the point of attack within a second. Now count the assignments, because that is where the extra man shows up:

  • the Y takes the 5, the fullback takes the Sam, the pulling guard takes the Mike;
  • the right tackle takes the 3, and the right guard helps him: that double team is the spare blocker.

Five blockers for four defenders means one man to spare, and power spends him in two installments: first as half of a double team that drives the 3-technique off the ball, then, once the 3 is moving, as the blocker who climbs off it to the backside linebacker, the Will. The defense cannot match that move, because its players learn where the ball is going only when the snap shows them, while the offense knew before the huddle broke.

Now the whole box. The offense has seven blockers (five linemen, the Y and the fullback) against seven defenders (four linemen and three linebackers). The center takes the 1, the right guard’s climb takes the Will, and the left tackle’s hinge is a hat spent on nobody: he only closes the gap the guard left. So one defender is left over, and power makes sure it is the backside end. Power does not add a blocker to the box. It moves one, and the backside end is the receipt. It is betting that a defender who lines up farthest from the hole, and has to run down the line behind a wall of bodies, will arrive too late. Most of the time he does. When the defense knows that and sends that end screaming down the line at the snap, the offense has answers (a quarterback who keeps the ball and runs where the end left, in Option Football; a bootleg pass behind him, in Play-Action, Boots and Screens).

The rest of the offense has jobs too, small but real. The quarterback opens, hands off and keeps carrying out a fake; on many teams he then sells a bootleg away from the play, which is what keeps the free backside end honest. The wide receivers, out of the picture here, block the men over them (usually a cornerback and a safety) so that the run does not end at the second level. And the defense’s usual answer to power, which Run Fits and Box Math builds in full, is to attack the pullers: an end or linebacker who meets the kick-out or the puller inside, so the ball has to bounce outside to a fast safety, plus that free backside end chasing from behind.

CautionCommon misconception

“Power works because the offensive line is bigger and stronger.” Strength helps, but the play is designed so that nobody has to win a fair fight. The down blocks hit men from the side; the kick-out hits the end man from inside, where he is not looking; the puller arrives as an extra man. What power needs is not the biggest line but a line that can pull: a guard who can run ten yards laterally and still hit a linebacker. That is why gap-scheme teams prize athletic guards as much as heavy ones.

The “O” in Power O is not explained consistently from one coaching tree to the next, so do not read much into it.

Walk the rules and the play draws itself. On the play side nothing changes in principle: the Y has a man in his inside gap (the 5), so he blocks down; the right guard has the 1 in his gap, so he blocks down on him; the right tackle has nobody in his gap and nobody over the guard beside him, so the third word of the rule sends him to a linebacker, the backside one. The problem is the backside. The center can block back on a 1-technique shaded on him, but a 3-technique sits outside the guard, two gaps away, and a center cannot get there. If the left guard pulled anyway, the 3 would be loose in the backfield, right in the path of the puller and the back. So the offense keeps the guard home and pulls the tackle, a version many coaches call dart or “tackle power.”2 The price is distance: the tackle has farther to run, so dart needs an athletic tackle. Other answers to the same front are to check to a different run or to run power the other way, at the 3.

Counter: the same blocks, a different first step

Power has one weakness that defenses learn quickly: the running back’s first step points at the hole, and so does the puller. A linebacker who reads both is moving toward the hole before the guard gets there. Counter attacks that reflex. It starts with the back stepping the wrong way, and it pulls two backside blockers the right way: classically the guard, who kicks out the end man, and the tackle, who wraps, swinging around the guard’s kick-out and up through the hole as the lead blocker. (The tackle is not throwing a log block here; “wrap” is a word coaches use loosely, as the log-block section below explains.) The guard-and-tackle version is called GT counter, GT for guard and tackle, and its most famous name is the counter trey of Joe Gibbs’s Washington teams. Where “trey” came from is not well documented; treat it as a name rather than a description.

Compare Figure 17.4 with Figure 17.1. The play side follows the same rules, though the picture differs because the front does: there is no tight end or Sam on the right here, so the right tackle blocks down on the 3-technique, the right guard climbs to the backside linebacker, the center blocks back on the 1, and the end man is the 5-technique himself. What changed is who supplies the extra men. Instead of a fullback for the kick-out, the backside guard does it; instead of the guard leading through the hole, the backside tackle does. Two linemen leave the backside, so something has to fill the space they left: here the U tight end blocks the backside end, and in other versions an H-back fills, or the end is simply left for the quarterback to read.

Count it the way you counted power. The defense has six in the box (four linemen, two linebackers) and keeps two safeties deep. The offense has six blockers: five linemen and the U. RT on the 3, the pulling guard on the 5, the pulling tackle on the Mike, the right guard on the Will, the center on the 1, the U on the backside end: a hat on every defender in the box. Against this two-high front, counter does not even leave the backside end free; the free men are the two safeties, ten yards away, and the back’s job is to make them miss.

And the running back’s first step goes away. That counter step, a single lateral jab in the wrong direction, is the whole trick. Linebackers are taught to read the back and the guards. On counter the back says left while the guards say right, and a linebacker who trusts his eyes takes one false step toward the back, which is all the pulling tackle needs to arrive first. In the simulation the Mike’s first steps go toward the back’s jab (frame 1), and by the time he corrects, the tackle is already in the hole.

Counter has a price too. Two pullers take longer to arrive than one, so the back’s counter step is partly a way to buy time: the ball cannot hit the hole before the blockers do. If a defensive lineman penetrates into the space the pullers must run through, he can wreck both pullers at once. And a defense that keys the guards (watches them for its cue) and ignores the back is not fooled at all, which is why offenses mix counter with power, zone, and fakes of all of them.

Counter comes in many flavors that keep the same skeleton, and playbooks name them by the pullers: GT counter pulls the guard and tackle; GH or GY counter pulls the guard and an H-back or tight end (Y), which spares the tackle the long trip and leaves him home to block the backside end.5 Some teams run it from the shotgun with the quarterback reading the backside end instead of blocking him (Option Football covers those reads); some run it as the quarterback’s own run. When you see two blockers pull the same way after the back has stepped the other way, think counter first.

New block: the log

Every kick-out has an assumption hidden in it: that the end man will stay outside, where there is room to drive him toward the sideline. Defenses do not always oblige, and Zone Running already gave you their two plans for meeting a blocker. With box (also called squeeze), the end squeezes down inside as the man beside him blocks down, but keeps his outside arm free, so the ball has to go back inside him. With spill, the end attacks the kick-out man’s inside half with his outside shoulder (coaches call this wrong-arming or wrong-shouldering the blocker), giving up the outside on purpose to pile up the hole, so the ball has to bounce outside to a teammate waiting there. Choosing between them is a big part of Run Fits and Box Math.

So the kick-out man reads the end as he arrives. He aims for the end’s inside number. If he can get his head inside the defender, he kicks him out and the ball goes inside. If the end has crashed so far inside (a hard squeeze, or a spill player wrong-arming) that the blocker cannot get his head inside, there is no “out” left to kick him toward, so he throws a log block: he takes the defender’s outside shoulder, gets his head and body outside him, seals him in, and the ball carrier bounces outside the block. Some line coaches call this wrapping or logging the end; others use wrap for any puller who swings around a block in front of him and turns up, like the counter tackle above. Either way, the defender decides where the hole is, and the blockers and the back read him.

Who reads him depends on the play. On power the kick-out man is the fullback or H-back, and the pulling guard turns up off whatever the fullback does; on counter the kick-out man is the pulling guard, and the tackle behind him does the same. The lead puller always goes to the hole the kick-out man leaves.

This is why gap runs are not as rigid as they look on a whiteboard. The assignments are fixed before the snap; the hole still moves a yard or two depending on what the end man does, and the back’s eyes are on the kick-out man’s helmet. The back runs to the side the helmet is on: head inside the defender (a kick-out), cut inside; head outside him (a log), bounce outside.

Duo: power without a puller

Not every gap run pulls anybody. Duo is the downhill, no-frills member of the family, and the one that manufactures nobody: two double teams (hence the name) drive the two defensive tackles straight back, every other blocker takes the man in front of him, and nobody moves sideways. It is still a gap run, because every blocker knows his man before the snap and the back is aimed at a hole chosen in advance. It looks like inside zone for the first instant, which is part of the point, but the footwork is different: zone linemen step sideways toward the play; duo linemen step forward, into the defenders, and the double teams stay on the defensive tackles longer, trying to move them back a yard or two before anyone climbs.

Where do duo’s extra blockers come from, if nobody pulls? From the defense’s own choice. Look at the box in Figure 17.6: six defenders against six blockers, with two safeties sitting 12 yards deep. Against a defense that keeps two safeties high to protect against the pass (a light box), the offense already has a hat for every hat, and it does not need to manufacture one. What it needs is movement, someone driven backward so the back can fall forward for four yards, and two double teams on the two biggest defenders are the most direct way to get it. Duo does not fool anyone. It is a bet that your five linemen and a tight end can move their six, and that your running back can make the last man miss.

The running back’s job is the subtle part. Since the linemen block the defensive linemen first and climb late, the linebackers are, for a beat, unaccounted for, and the back is the one who “blocks” them with his path. The double teams own them: the double on the 1 owns the Will, the double on the 3 owns the Mike, and whichever way a backer attacks decides which lineman comes off to meet him. The back presses the double teams, aiming at the A gap between them, and watches the Mike. If the Mike fills the A gap, the back bounces to the B gap off the double on the 3; if the Mike fills outside, the back stays in the A gap. It is a read, like zone’s, but a vertical one: the back is choosing which side of a double team to run past, not whether to cut back across the field.

That difference is also how you tell duo from inside zone on the broadcast, where the two look alike for a heartbeat (both start with double teams and no pullers). Watch the line’s first step from the high end-zone camera: on inside zone every lineman steps sideways toward the play together, so the whole line drifts; on duo the linemen step straight ahead and the double teams stay on the tackles longer, so the line surges without drifting. Then watch the back: on duo he runs straight downhill at the doubles; on inside zone his path angles toward the play side before he reads.

Duo’s trade is plain. It needs blockers who can move people straight back, and it does nothing to the numbers: against a seventh or eighth defender in the box, there is nobody to block him. That makes duo a two-high play, and its rise in the last decade tracks the rise of two-high defenses: when defenses take a safety out of the box to stop the pass, offenses answer by running straight at the six who are left. NFL Next Gen Stats measured split-safety coverage on 42% of plays in 2025, the most since its tracking began in 2018, which tells you how many light boxes NFL offenses now see.6

Trap: let him in, then hit him from the side

Trap is one of the oldest plays in football and still one of the most satisfying to watch, because it uses a defender’s aggression against him. The offense leaves a defensive lineman unblocked at the snap. He feels nobody in front of him and does what he is paid to do: he charges across the line. Then a pulling lineman, usually the backside guard on a short pull, hits him from the side, where he is not looking, and the ball carrier cuts just inside him.

Why would anyone invite a defender into the backfield? Because of what it does to the rest of the line. With the 3-technique left alone, the right guard does not have to block him and can go straight to the Mike, and the left tackle can go straight to the Will: in Figure 17.7 both linebackers are blocked within a second, which is hard to do on most runs. And the defender who was left alone is beaten by geometry rather than strength. A lineman charging upfield has his shoulders pointed at the quarterback; a guard arriving from the side hits him where he cannot anchor, and the harder he charged, the farther past the hole he ends up.

Trap is a counter-punch to penetration. It is most dangerous against an aggressive, one-gap defensive tackle, the kind of 3-technique who lives in the backfield, and against defenses that slant hard at the snap. It is least dangerous against a disciplined lineman who reads the guard: if the guard in front of him releases inside instead of blocking him, a well-coached tackle knows the trap is coming, squeezes down, and closes the hole before the trapper gets there. Coaches answer with constraint plays (plays designed to punish a defender for doing his job too well) such as the influence trap, in which the lineman in front of the trapped man pretends to pass-block, inviting the defender upfield, before the trap hits him.

The trap’s pedigree runs back to the earliest days of the T-formation. Paul Brown’s Cleveland teams of the late 1940s, with quarterback Otto Graham and fullback Marion Motley, were famous for two plays that turn a defense’s aggression against it: the trap and the draw (which closes this chapter).8

Iso’s logic is a clean version of hat math. Against a seven-man box (four linemen, three linebackers), the offense has seven blockers: five linemen, a tight end and a fullback. Every defender gets a hat, and the one matchup the play depends on is the fullback against the Mike. Where it hits is decided by the front: iso usually aims at a bubble, an open gap or an uncovered lineman where the nearest defender is a linebacker rather than a defensive lineman, so the fullback can get to him. In Figure 17.9 the strong A gap is open because the 1 is shaded to the other side of the center and the 3 lines up outside the right guard; the center and guard only have to turn their men away from it. Against this front the offense could also run iso weak, over the uncovered left guard, at the Will.

The trade is that iso gains nothing in the numbers and moves nobody sideways; it is a straight contest. Its virtues are speed and simplicity: the ball hits fast, and the back has one read (which side of the fullback’s block to take). Iso was a staple of the I-formation era, and it has faded with the fullback. In 2025 only about one NFL snap in ten came from personnel with two backs (a fullback or a second running back), so most teams now lead a tight end or H-back through the hole instead, either as a true iso from two- and three-tight-end groupings or as the insert block on inside zone (zone insert, where the line zone-blocks instead of blocking the man on it).11

Toss, sweep and pin-and-pull: power goes outside

Everything so far hit between the tackles or just outside them. The gap family also has outside runs, and they are built from the same parts: block down, pull, lead.

A toss is the quarterback pitching the ball underhand to a back who is already moving toward the sideline; a sweep is any outside run with blockers leading the back around the corner, whether the ball is pitched or handed off. The toss saves time: the back catches the ball two or three yards wider and already at speed, instead of waiting for a handoff and then turning the corner.

Guards Jerry Kramer and Fuzzy Thurston pulled in tandem, leading Paul Hornung or Jim Taylor around the end, and the play was the cornerstone of an offense that won five NFL championships in seven seasons (1961, 1962, 1965, 1966 and 1967, the last two including Super Bowls I and II).12 Lombardi’s own phrase for the back’s job became the title of his 1963 book, Run to Daylight: the back did not run to a spot drawn on the card; he read the tight end’s block on the outside linebacker and went wherever the defender was not. Lombardi drilled the sweep endlessly precisely because everyone knew it was coming, and the answer to “everyone knows” was perfect execution plus a back who could choose his hole late.

Look at Figure 17.10 with this chapter’s vocabulary and you can see it is power stretched sideways: down blocks inside, pullers leading, a fullback on the end, the backside left to chase. Why pitch or hand the ball wide at all, instead of running power off-tackle? Because some defenses are strongest inside, and the sweep makes every defender run. The price is time: the back is in the backfield longer, so a defender who beats a down block gets to him. The modern descendant is pin-and-pull.

Pin-and-pull

Outside zone (03-02) has a problem against wide fronts, defenses whose ends line up well outside the tackles and tight ends. To run outside zone, the play-side blockers must reach the defenders in front of them, getting their heads to the outside, and a defender lined up two yards outside you is almost impossible to reach. Defenses that play wide (a wide-9 end, an end in a wide 5, an outside linebacker walked out over the tight end) are partly built to kill the reach.

Pin-and-pull gives up on reaching. Each lineman, backside ones included, asks one question about the gap on his backside, the side away from the play:

  • Is a defender in it (or shaded into it)? Pin him. Block down on him, which is an easy angle because he is inside you.
  • Is it empty? Pull. Run around the blocks outside you and lead.

In Figure 17.11 that sorts the whole line. The left tackle has the wide 5 in his backside gap and the center has the 1 shaded on his backside, so both pin; the right tackle has the 3 in his, so he pins; both guards have nobody in theirs, so both pull. The tight end is the usual exception: in most versions he does not pull even with an empty backside gap (somebody has to stop the play-side linebacker from running straight to the ball), so he goes inside to the Mike.

The beauty of the rule is that it turns the defense’s alignment into the offense’s advantage. The wider the defenders line up, the more of them end up inside an offensive lineman, where he can pin them, and the more linemen are left uncovered, free to pull. Every block in Figure 17.11 is either a down block or a block by a puller with a running start, and nobody has to reach anyone. The rules adjust to the front without a new call, the way zone rules do, which is why some coaches describe pin-and-pull as a gap scheme that thinks like zone.13

The draw: a run disguised as a pass

The last member of the family barely looks like a run at all. On a draw play (or simply a draw), the offensive line sets up as if to pass-protect, the quarterback shows a drop, the pass rushers are allowed to run upfield, and the ball is handed to the back late, after the linebackers have turned their backs to run to their pass-coverage zones. The rush has drawn the defenders out of the play.

The draw belongs in a gap chapter because it is built on the same principle as the trap: let the defender go where he wants to go, then block him there. The guards in Figure 17.12 do not try to stop the 3-techniques from rushing; they let them come upfield and then turn them outward, away from the A gaps. The tackles ride the ends wide and deep, the way they would on any pass. The more eagerly the defense rushes the passer, the bigger the hole in the middle gets. And the linebackers, who on a normal run would be the problem, have spent the first second running away from the ball to cover receivers.

That tells you when to expect it: on passing downs, against a four-man rush with linebackers who drop. Third-and-long is the classic draw situation, and a defense with two deep safeties and two linebackers dropping into zones is the classic draw defense.

Be honest about what it buys, though, because you have the tools from Downs, Distance and Field Position. A six-yard gain on third-and-11 is a failed play by success rate: the offense punts anyway. In the 2023–2025 regular seasons, NFL designed runs on third-and-10 to -12 converted about 12% of the time and averaged about −0.29 EPA a play; dropbacks in the same spot converted about 27% and averaged about −0.11.15 Some of that gap is selection (offenses that have already given up on the down, late in a half or protecting a lead, call runs), but the point stands: a third-and-long draw is usually a deliberate conservative choice. Coaches call it to avoid a sack or an interception, to shorten the punt or set up a field-goal try, and to make the pass rush hesitate a half-step on the next third down. Against a defense that drops its linebackers it is also, now and then, a long run. It is rarely the best way to get eleven yards.

The trade: the draw is helpless against interior pressure. If a linebacker comes forward at the snap instead of dropping, or walks up into an A gap before the snap, he is standing exactly where the play needs to go, and nobody is assigned to him because the line is busy pretending to pass-block. A defense that sits a linebacker in the middle as a spy on a running quarterback ruins it for the same reason. Pressure off the edge is a different story: an outside blitzer runs himself past the play, which is exactly what the draw wants.

Paul Brown’s Browns made the draw famous in the late 1940s with Marion Motley, a fullback of about 232 to 238 pounds who could also pass-block, which made the fake believable. One story says it began by accident, with Otto Graham improvising a handoff on a broken pass play; another has Brown already experimenting with a delayed handoff to Motley in 1945. Either way it was a Cleveland staple by the time the Browns joined the NFL in 1950.16 Its descendants, the quarterback draw and the run-pass option, belong to Option Football and RPOs.

Gap or zone? How coaches choose

Almost every NFL team runs both families, so the real question is when each one is the better call. Five things push a coach one way or the other.

The front. Gap schemes need angles, so they like fronts that give them: a 3-technique to block down on, a 1-technique to block back on, an end man who can be kicked out. They are vulnerable to movement, since a defender who slants into the space a puller is running through can blow up two blockers at once. Zone, built to absorb movement, is the natural answer to slanting fronts. Wide fronts, as you saw, push teams toward pin-and-pull and kick-outs; a penetrating 3-technique invites the trap or the wham; linebackers who flow fast to stop outside zone invite counter. Defenses know this too, and some fronts are built partly to take the angles away: a lineman on a blocker’s inside shoulder (a 4i on the tackle, say) leaves that blocker no down block to make, and if he penetrates he is standing in the lane the pullers run through. Odd and Hybrid Fronts shows those fronts.

The defense’s shell. Against a light box (two deep safeties, six in the box), the offense already has its hats and wants movement: duo and inside zone. Against a seven-man box, power and counter win the point of attack by moving hats there. Against eight, even they leave a defender free (the backside end and the safety who came down), which is when offenses read one of them with the quarterback (Option Football) or stop running and throw, which is the point of loading the box.

The personnel. Power wants a kick-out blocker: a fullback, an H-back, a second tight end. Iso needs a fullback. Wham needs a tight end who can trap. A team with no fullback on the roster runs a different gap game from one that has an All-Pro at the position.

The offensive line. Zone wants linemen who can move laterally and stay on their feet, and has made a home for lighter, quicker linemen. Gap schemes want linemen who can drive a man off the ball on a down block and guards (or tackles) who can pull and hit on the move. Teams draft and coach their line for the run game they want, and their run game also bends toward the line they have.

The situation and the sequence. Short yardage favours the known hole and the extra hat of a gap scheme. Long yardage favours the draw. And every run sets up another: counter exists because power exists, the trap because 3-techniques penetrate, the bootleg because the backside end chases (Constraint Theory and Sequencing is the full version of that idea).

In 2019 Kyle Shanahan, the most famous wide-zone coach in football, surprised people by adding more power and counter to his 49ers’ run game; he told his team he had called such plays only about seven times in more than a decade as a play-caller.17 That is the modern consensus in one decision: a scheme is a menu, not an identity.

How do the two families compare league-wide? The public numbers come from charting services, since play-by-play data does not record the blocking scheme. Pro Football Focus’s charting had inside and outside zone together making up nearly half of all NFL running plays in 2022, with draws only about 3%. Over 2020–2022 it had power averaging 4.53 yards a carry, inside zone 3.90, counter 3.90, outside zone 3.78 and duo 3.71 (and draws, called against defenses set for a pass, 5.95).18 Treat those as rough. Yards per carry is a crude measure, and each play is called in the situations it suits: duo against light boxes on early downs, power in short yardage, the draw on third-and-long. No family dominates, which is why almost every team runs both. The Go Deeper box at the end shows how to count the concepts yourself in tracking data.

Spotting a puller before the snap

Because a puller runs to the hole, finding him early is the single most useful habit in watching gap runs. Defenses know this too: many linebackers are taught to read the guards, and a pulling guard is their signal to run with him. So both sides are looking for the same tells, and offenses work to hide them.

Before the snap, look for:

  • The backfield, and which guard to watch. The guard to watch is the backside one, the guard away from where the play is likely to go, so first guess the direction. In the shotgun, gap runs, like zone runs, usually send the back across the quarterback’s face, so a back offset to the left usually runs right, and the backside guard is the one on the same side as the back (in Figure 17.4, the back is offset left and the left guard pulls). Under center, a fullback or H-back offset to one side is often the kick-out man, and power usually goes toward him, so watch the guard away from him. With nothing to go on (an I-formation, a back directly behind the quarterback), watch both guards: on the high end-zone replay they are side by side, and the one that moves first tells you. A back set deeper than usual needs time, which counter and the sweep require (the pullers have to get there first).
  • The guards’ stance. A lineman about to pull often has less weight on his hand, his feet a little more staggered, his hips ready to turn. This is the heavy hand tell in reverse: a puller is going sideways, not forward, and many give it away with a light hand or a slight cheat backward off the ball, since depth gives him room to clear his teammates’ heels. The best pullers hide it, and some offenses deliberately line up their guards the same way on every snap.
  • The formation and the tendency. Certain personnel groups and formations are gap-heavy for every team (two tight ends with one off the ball, the I-formation, an H-back on the wing); the broadcast graphics and your prediction log will teach you which.

At the snap, the first step settles it. A pulling lineman’s first step is backward and sideways, opening his hips toward the play (the bucket step; some coaches call it a drop step); a zone lineman’s first step is sideways and forward; a pass-blocking lineman’s first step is straight back with his shoulders still square to the defense. One look at the backside guard tells you which. One exception to keep in mind: on dart (Figure 17.3) the guard stays home and the tackle pulls, so if the guard blocks but the run still looks like power, check the backside tackle.

Offenses exploit the defenders who read guards. A pulling guard on a play-action pass pulls linebackers toward the fake, and a “false pull” on a run sends them the wrong way; counter itself is a lie told with the back’s first step. Those plays exist precisely because the tell is so reliable. Trust the guard, but notice when the offense is using your trust against you.

The first half-second

  • Does a guard pull? If yes, call “gap” and say the direction out loud: the play is going where he is going.
  • Does a second blocker pull with him? Two pullers with the back stepping away first means counter. Two guards pulling wide with the back flowing the same way means a sweep or pin-and-pull. Any pull with the back stepping the other way means counter.
  • The guard blocks, but the backside tackle pulls? Dart: power with a tackle as the puller.
  • Nobody pulls, but the line fires straight ahead in double teams? Duo. The whole line steps the same way? Zone.
  • A defensive tackle comes through untouched? Look for the trap (a guard) or the wham (a tight end) arriving at his hip.
  • The line drops back as if to pass, and the back gets the ball late? Draw.

After the play (on the replay)

  • Find the kick-out. Did the kick-out man kick the end out or log him in, and did the back go to the side of his helmet?
  • Find the free defender: on power, the backside end. Did he make the tackle?

On third-and-11 this linebacker will be running backward at the snap, and a play built to catch a man running away from it is the draw.

Will it get 11 yards? Probably not, and the offense knows it: as the draw section showed, third-and-long runs rarely convert and usually lose expected points. A draw here is a conservative call (no sack, no interception, a shorter punt) and a warning shot to the pass rush.

What would ruin it: the linebacker coming forward at the snap, a blitz up the middle, or a “spy” assigned to sit in the middle and watch the back. Any defender who stays home in the A gaps is standing exactly where the draw needs to go, and nobody is assigned to block him.

TipAnswer

Log him and bounce, and the tackle follows the log outside. The end has taken away the inside: there is no room to kick him out, because he is already standing where the hole was supposed to be, and the guard cannot get his head inside him. So the guard takes his outside shoulder, gets his head and body outside him and seals him in (a log block). The pulling tackle, who was going to turn up inside the kick-out, now has no inside to turn up into, so he follows the log outside and takes the first defender to show, usually the Mike scraping over the top. The back, reading the guard’s helmet, bounces outside the log behind the tackle.

The defender left over is the strong safety, coming down outside, and he is the back’s to beat. That is not an accident: a defense that tells its end to spill the ball outside is counting on a fast defender outside to meet it there, the alley player (Run Fits and Box Math explains who that is and why). The offense’s counter-move is to block that man, often with a wide receiver’s crack block (a receiver cutting back inside to block a defender from the side, which Perimeter Runs and Misdirection covers).

Takeaways

  • A gap scheme blocks men, not areas, and aims at a hole chosen in advance. The core gap run is down blocks (the inside wall), a kick-out (the outside wall) and a puller through the middle; the rest of the family drops or swaps one part. The hole moves only as far as the end man forces it, and the back runs to the side of the kick-out man’s helmet (kick or log).
  • Power is the template. Block down, kick out with the fullback, lead with the backside guard, hinge the backside, and leave the backside end unblocked. It turns a play side of three blockers against four defenders into five against four, with the spare man doubling the 3 and then climbing. It does not add a blocker to the box; it moves one.
  • The rest of the family varies one thing. Counter changes the back’s first step and pulls two (guard kicks, tackle wraps). Duo pulls nobody and drives two double teams into a light box. Trap and wham leave a lineman unblocked and hit him from the side, with a guard or a tight end. Iso sends a fullback at one linebacker. Pin-and-pull and the sweep take the same blocks outside. The draw blocks the rush where it wants to go.
  • Every gap run names who is left free. Usually it is a defender far from the hole (the backside end); the offense bets he arrives late, or reads him with the quarterback.
  • Coaches choose by front, shell, personnel and line. Gap schemes like angles, seven-man boxes and big static fronts; zone likes movement; duo and zone like light boxes; most teams run both from the same looks. The rules travel from front to front, which is why a backside 3-technique turns power into dart.
  • Follow the guard. A puller runs to the hole. Find the backside guard on every run (in the shotgun, usually the one on the back’s side), and you will call the direction of most gap plays before the back gets there.
