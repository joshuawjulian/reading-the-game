def named_line(players):
    """Label the five offensive linemen so the reader can follow each one."""
    return [p.moved(label=OL_NAMES[p.key]) if p.key in OL_NAMES else p for p in players]


def ace_condensed():
    """12 personnel: two in-line tight ends (Y right, U left), both receivers off the ball in tight splits."""
    off = named_line(offense("ace"))
    return [q.moved(w=-9.0) if q.key == "X" else (q.moved(w=9.0) if q.key == "Z" else q) for q in off]


def D(key, pos, d, w, label):
    return Player(key, pos, "def", d, w, label=label)


def secondary(cb_w=15.0):
    """Two corners and two deep safeties (a two-high shell)."""
    sw = min(9.0, cb_w - 1.5)
    return [D("LCB", "CB", 6.5, -cb_w, "CB"), D("RCB", "CB", 6.5, cb_w, "CB"),
            D("FS", "FS", 11.5, -sw, "FS"), D("SS", "SS", 11.5, sw, "SS")]


def even_front(playside=1, sam_w=6.6, cb_w=15.0):
    """A generic four-man even front (named in 05-02), linemen labelled by technique number:
    5-technique, 1-technique, 3-technique, 5-technique, three linebackers, two deep safeties.
    playside=+1 puts the 1-technique on the play side (the offense's right)."""
    s = playside
    return [
        D("PE", "DE", 1.0, technique("5", s), "5"),     # play-side end
        D("PT", "DT", 1.0, technique("1", s), "1"),     # play-side tackle, shaded on the center
        D("BT", "DT", 1.0, technique("3", -s), "3"),    # backside 3-technique
        D("BE", "DE", 1.0, technique("5", -s), "5"),    # backside end
        D("SAM", "SAM", 4.0, s * sam_w, "S"),
        D("MIKE", "MIKE", 4.5, s * 1.6, "M"),
        D("WILL", "WILL", 4.5, -s * 2.6, "W"),
    ] + secondary(cb_w)


def front7(playside=1):
    return [q for q in even_front(playside) if q.key in ("PE", "PT", "BT", "BE", "SAM", "MIKE", "WILL")]


def go_to(p, key, end, kind="block", delay=0.0, speed=None, via=()):
    """Move `key` through absolute (d, w) points `via` to `end`, as a block, path or run."""
    start = np.array(p._cur(key))
    pts = [tuple(np.array(q) - start) for q in (*via, end)]
    if kind == "block":
        return p.block(key, pts=pts, delay=delay, speed=speed)
    return {"path": p.path, "run": p.run}[kind](key, pts, delay=delay, speed=speed)


def _P(fld, p, d, w):
    return fld.xy(*p.to_abs(d, w))


def branch(fld, p, pts, color=style.THIRD, label=None, label_at=None, lw=2.0, fontsize=8,
           dash=(0, (3, 2.2)), end="arrow"):
    """A dashed 'option' path through absolute (d, w) points, with an arrowhead (or block tee)."""
    xs, ys = _P(fld, p, np.array([q[0] for q in pts]), np.array([q[1] for q in pts]))
    if end == "arrow":
        fld.ax.plot(xs[:-1], ys[:-1], color=color, lw=lw, ls=dash, zorder=6, solid_capstyle="round")
        fld.ax.add_patch(FancyArrowPatch((xs[-2], ys[-2]), (xs[-1], ys[-1]), arrowstyle="-|>",
                                         mutation_scale=11, color=color, lw=lw, zorder=6))
    else:  # a block: dashed line ending in a T-bar
        fld.ax.plot(xs, ys, color=color, lw=lw, ls=dash, zorder=6)
        v = np.array([xs[-1] - xs[-2], ys[-1] - ys[-2]])
        perp = np.array([-v[1], v[0]]) / np.linalg.norm(v) * 0.75
        fld.ax.plot([xs[-1] - perp[0], xs[-1] + perp[0]], [ys[-1] - perp[1], ys[-1] + perp[1]],
                    color=color, lw=lw + 0.6, zorder=6)
    if label:
        x, y = _P(fld, p, *(label_at or pts[-1]))
        fld.ax.text(x, y, label, fontsize=fontsize, color=color, fontweight="bold", ha="center",
                    va="center", zorder=9,
                    bbox=dict(boxstyle="round,pad=0.2", fc=style.SURFACE, ec=color, lw=0.8))


def overtake(fld, p, key, via, end, color=ZONE_GREEN, lw=2.4):
    """An overtake block (dashed green, T-bar end) from `key`'s alignment through `via` to `end`."""
    q = p.players[key]
    branch(fld, p, [(q.d, q.w), *via, end], color=color, end="tee", lw=lw)

def clean(ax):
    """Box-zoomed diagrams: drop hash ticks and yard numbers, clip anything outside the window."""

MARK_R = 0.6      # frame-strip marker radius in yards: smaller than the static diagrams, so engaged
                  # blockers and defenders (about a yard apart) stay separate in print


def tracking(play):
    df = play.to_tracking()
    df["_id"] = df["displayName"].astype(str)
    return df


def panel(ax, play, df, t, lateral, window, rings=(), down=None):
    """One moment of a simulated play on `ax`: markers, trails of on-screen players, rings, cut fades."""
    tmap = df.groupby("frameId")["time"].first()
    fid = int((tmap - t).abs().idxmin())
    ylo, yhi = sorted((play.ball_y - lateral[0], play.ball_y - lateral[1]))
    fld = Field(ax=ax, orient="vertical", los=play.los, window=window, numbers=False, hashes=False,
                lateral=(ylo, yhi), to_go=play.to_go)
    cur = df[df["frameId"] == fid]
    shown = cur[cur["x"].between(play.los + window[0] + 0.4, play.los + window[1] - 0.4)
                & cur["y"].between(ylo + 0.4, yhi - 0.4)]
    ids = set(shown["_id"])
    trails = df[(df["frameId"] <= fid) & df["_id"].isin(ids) & (df["side"] != "ball")
                & df["x"].between(play.los + window[0], play.los + window[1])]
    for (_, side), g in trails.groupby(["_id", "side"]):        # trails only for players in the panel
        xs, ys = fld.xy(g["x"].values, g["y"].values)
        ax.plot(xs, ys, color=style.OFFENSE if side == "off" else style.DEFENSE, lw=0.9, alpha=0.4, zorder=3)
    ppy = fld.pts_per_yard()
    downed = [k for k, t_down in (down or {}).items() if t >= t_down]
    for _, r in shown.iterrows():
        x, y = (float(v) for v in fld.xy(r["x"], r["y"]))
        if r["side"] == "ball":
            ax.add_patch(Ellipse((x, y + 0.45), 0.7, 0.42, fc=style.BALL, ec="white", lw=0.5, zorder=9))
            continue
        lab = str(r.get("label", "") or "")
        if r["side"] == "off":
            ax.add_patch(Circle((x, y), MARK_R, fc=style.OFFENSE, ec="white", lw=0.6, zorder=8))
            tc = "white"
        else:
            gone = r["displayName"] in downed            # cut to the ground: faded, under the blockers
            ax.add_patch(RegularPolygon(
                (x, y), 4, radius=MARK_R * 1.3, orientation=np.pi / 4, fc=style.DEFENSE_TINT,
                ec=style.DEFENSE, lw=1.0, zorder=6 if gone else 8, alpha=0.4 if gone else 1.0))
            tc = style.MUTED if gone else style.INK
        fs = max(4.0, min(8.0, ppy * (0.8 if len(lab) <= 1 else 0.6)))
        ax.text(x, y, lab, ha="center", va="center", fontsize=fs, color=tc, fontweight="bold",
                zorder=10 if tc != style.MUTED else 6.5)
    for key, col in rings:
        r = cur[cur["displayName"] == key].iloc[0]
        x, y = fld.xy(r["x"], r["y"])
        ax.add_patch(Circle((float(x), float(y)), 1.05, fc="none", ec=col, lw=1.7, zorder=7))
    if downed:                                           # one "cut" tag beside them (backside)
        rr = cur[cur["displayName"].isin(downed)]
        x, y = fld.xy(rr["x"].mean(), rr["y"].max())
        ax.text(float(x) - 0.9, float(y), "cut", fontsize=7, color=style.DEFENSE_DARK, ha="right",
                va="center", fontweight="bold", zorder=12,
                bbox=dict(boxstyle="round,pad=0.1", fc=style.SURFACE, ec="none", alpha=0.85))
    return fld


def strip(play, times, lateral=(-12, 12), window=(-8, 13), ncols=3, notes=None, rings=(("RB", style.THIRD),),
          down=None, title=None, width_in=6.5):
    """Frame strip with a fixed, box-sized window so markers stay legible in print.
    rings: players to circle in every frame; down: {key: t} players cut to the ground from time t."""
    df = tracking(play)
    nrows = int(np.ceil(len(times) / ncols))
    pw = width_in / ncols
    span_w, span_d = lateral[1] - lateral[0], window[1] - window[0]
    fig, axes = plt.subplots(nrows, ncols, squeeze=False,
                             figsize=(width_in, nrows * (pw * span_d / span_w + (0.36 if notes else 0.2))))
    for ax in axes.flat[len(times):]:
        ax.axis("off")
    for i, (ax, t) in enumerate(zip(axes.flat, times)):
        panel(ax, play, df, t, lateral, window, rings, down)
        ax.set_title(f"{i + 1}.  " + ("snap" if abs(t) < 1e-6 else f"{t:+.2f} s".rstrip("0")),
                     fontsize=8.5, color=style.INK_2, loc="left")
        if notes:
            ax.text(0.5, -0.03, notes[i], transform=ax.transAxes, ha="center", va="top",
                    fontsize=7.5, color=style.INK)
    if title:
        fig.suptitle(title, x=0.01, ha="left", fontsize=11, fontweight="bold", color=style.INK)
    fig.tight_layout(h_pad=1.3 if notes else 0.6, w_pad=0.6)
    return fig


def video(play, times, notes=None, lateral=(-12, 12), window=(-8, 13), rings=(("RB", style.THIRD),),
          down=None, width_in=5.6):
    """Web version of the strip: the same panel drawn at every frame, with the strip's notes as captions
    (each note shows from its frame's time on), the rings and the cut fades."""
    import shutil
    from matplotlib import animation
    from IPython.display import HTML, display
    df = tracking(play)
    span_w, span_d = lateral[1] - lateral[0], window[1] - window[0]
    fig, ax = plt.subplots(figsize=(width_in, width_in * span_d / span_w + 0.4))
    tmap = df.groupby("frameId")["time"].first()

    def update(fid):
        ax.clear()
        t = float(tmap[fid])
        panel(ax, play, df, t, lateral, window, rings, down)
        txt = ("snap" if abs(t) < 0.05 else f"{t:+.1f} s")
        if notes:
            cur = [n for tt, n in zip(times, notes) if tt <= t + 1e-6]
            txt += ("   " + cur[-1]) if cur else ""
        ax.set_title(txt, fontsize=9, color=style.INK, loc="left")
        return []

    anim = animation.FuncAnimation(fig, update, frames=sorted(df["frameId"].unique()), interval=100)
    html = (anim.to_html5_video(embed_limit=40).replace("<video ", '<video style="max-width:100%;height:auto" ')
            if shutil.which("ffmpeg") else anim.to_jshtml(default_mode="once"))
    plt.close(fig)
    display(HTML(html))

Second-and-7, early in the second quarter. The offense lines up with two tight ends, and at the snap all five offensive linemen take the same step to their right, in unison, like a chorus line. The running back takes the handoff on a tight path toward the right tight end, and for a moment it looks like he is going to be swallowed by a pile of bodies. Then he plants his foot, cuts back to the left, and runs through a hole near the center that was not there half a second earlier. Twelve yards. The announcer says “great vision.” He is right, but it is not the whole story. That hole was not luck and the cut was not improvised. The play was built to produce it. This chapter shows you how.

Why block an area instead of a man?

Start with the problem zone blocking solves. On a classic “man” (or gap) blocking run, the coach draws the play against the front he expects: the right guard blocks that defensive tackle, the tight end blocks that end. That works beautifully until the defense moves. Defensive linemen do not stand still. They slant (step sideways into a different gap at the snap), they stunt (two linemen swap gaps), and linebackers blitz into gaps the play assumed would be empty. A blocker assigned to a man who has just left becomes a spectator, and the man who arrived in his place is unblocked.

Zone blocking flips the question. Instead of “which defender is mine?”, each lineman asks “who comes into my area?” Every lineman steps the same direction at the snap, and whoever enters the space he is responsible for is his man. If the defensive tackle slants away, the guard simply keeps moving and picks up whoever replaced him.

Notice what zone blocking is not. Linemen are not blocking “grass.” They block people, every time. What changes is how each lineman decides who that person is: by a rule applied after the snap, not by a name written on the play card.

That rule-based design has a second consequence that makes zone runs feel different to watch. Because nobody knows before the snap exactly where the hole will be, the running back has to find it. The line’s job is to create a moving wall and get the defense flowing sideways; the back’s job is to read where that wall cracks. A zone run is half a blocking scheme and half a running back’s decision. You will see both halves in this chapter.

ImportantWhy it exists

Zone blocking exists because defenses move. Assigning blockers to areas by rule means a slant, a stunt or a late shift does not leave a defender unblocked by surprise, the way it can when blockers are assigned to names. That is a promise about assignments, not about winning: a defender who moves fast enough can still beat his blocker physically, and you will see later which kinds of movement hurt zone most. The price of the scheme: zone blocks are rarely knock-down, drive-him-five-yards blocks. Zone teams accept “get in the way and keep moving” in exchange for never losing track of a defender. And because the hole is not fixed, the play depends on a running back who can read it.

Gap schemes (power, counter and their relatives) are the other great family of run plays and make the opposite trade, a fixed hole with extra blockers pulled to it. Most NFL teams run both; 03-03 covers when coaches choose which.

The three rules: step, covered or uncovered, climb

Every zone play, inside or outside, runs on the same three rules. Learn these and you can draw a zone play against any front.

Rule 1: everyone steps play-side. At the snap, all five linemen (and the tight ends) take their first step toward the side the play is going. On a broadcast this is the single clearest tell of a zone run: the whole line moves the same direction together. On a gap run, by contrast, linemen block in different directions and at least one usually pulls.

Rule 2: covered or uncovered decides your job. You met these words in 03-01; here is how zone teams count them. A lineman is covered if a defensive lineman is head-up on him or shaded to his play-side, that is, sitting in the gap he is stepping into. He blocks that man. A defender shaded to a lineman’s backside belongs to the neighbor on that side, not to him. A lineman with nobody head-up or play-side of him is uncovered. He does not stand around: his first step is still play-side, but his hands go to the defender in the gap on his backside, the man his covered neighbor is blocking, while his eyes find the linebacker behind that defender. Coaches count this in slightly different ways (some staffs call a lineman covered whenever anyone is shaded on him), but the pairs and the jobs come out the same.1

Rule 3: double-team to the second level. That help is the heart of zone blocking. When a covered and an uncovered lineman both have hands on the same defensive lineman, they form a combo block (a double team). Two men move one defensive lineman, and while they do it both keep their eyes on the linebacker behind him. When the linebacker commits to a gap, one of the two linemen comes off the double team and “climbs” to block him, leaving the other to finish the defensive lineman alone. The double team gets you movement; the climb gets you the second level (the linebackers).

Which of the two climbs? Usually the uncovered man, the one who started with his eyes on the linebacker; the covered man’s job is the defensive lineman, and he finishes him. When the climber leaves varies by coach: in one survey of line coaches, the largest group taught linemen to come off as soon as they saw the linebacker run over the top of the double team or underneath it.2 Come off too early and the defensive lineman splits the double team; too late and the linebacker is already in the hole.

Edge cases. Three situations the rules seem not to cover come up in this chapter’s pictures and drills:

  • Two uncovered linemen side by side. An uncovered lineman helps the covered man on his backside. If his backside neighbor is uncovered too, there is nobody to help, so he skips the double team and climbs straight to the nearest linebacker.
  • The last man on the play side, with a defender outside him (a tight end with an end lined up on his outside shoulder). He is covered, and there is no play-side neighbor to help him, so he has to block that man alone. That is the hardest block in zone, and it changes where the play hits.
  • The last man on the backside, with a defender outside him. Nobody needs to reach a backside defender; the job on the backside is to stop chasers. So the last backside blocker steps inside and protects the gap inside him (coaches call it hinging, swinging shut like a door), and the man outside him is left to the quarterback’s fake or read. That outside man is the “unblocked backside end” you will meet again and again.

Two things are worth noticing. First, the rules found three natural double teams (LT and LG on the 3, C and RG on the 1, RT and Y on the 5) without the coach drawing them; a different front would produce different pairs from the same rules, which is exactly the point. Line coaches give these pairs short code names so two linemen can tell each other “we’re together on this one” before the snap; the names vary by staff. Second, the defender who lines up farthest from the play on the backside, the left 5-technique, is not on anyone’s list. Zone plays almost always leave someone unblocked on the backside. Much of this chapter is about how the offense deals with him.

Who decides all this? Before every snap the center usually identifies the front out loud and makes a “Mike” call: he points out one linebacker and names him the Mike for this play, so all five linemen count defenders from the same starting point and nobody disagrees about which linebacker is whose (how lines use that call in pass protection is in 04-02). The linemen then confirm their pairs with one-word calls. When a pair gets it wrong, two linemen block the same man, or nobody blocks one, and you see a defender in the backfield untouched at the snap.

CautionCommon misconception

“Zone linemen just block whoever is in front of them.” Not quite. A lineman whose man is directly in front of him but shaded to the backside is uncovered under zone rules, because what matters is the gap he is stepping into. That is why two linemen facing the same front can get different jobs depending on which way the play goes, and why zone teams spend hours of practice on “who’s covered?” against every front they might see.

Inside zone: the downhill version

Inside zone is the zone play that attacks the middle of the line. The linemen take short, firm steps play-side (more forward than sideways), the double teams try to drive the defensive linemen backward, and the running back aims between the guard and tackle.

That target has a name. The aiming point is the spot the running back’s first steps point at. On inside zone it is usually the play-side guard’s outside hip, the B gap (some teams teach the A gap; it varies with the team and with whether the quarterback is under center or in the shotgun). The aiming point is not necessarily where the back will end up. It is where he starts, so that the defense has to respect the play-side gaps and the back can read the blocks from there.

Why build a run this way? Because the double teams are where inside zone wins. Two 300-pound men on one defensive lineman can move him, and moving the defensive tackles off the line of scrimmage is what opens everything else. The linebackers then face an impossible choice. If a linebacker fills the play-side B gap early, the climbing lineman meets him there and walls him toward the sideline, and the back cuts just inside that block, into the gap the linebacker left. If he waits, the double team is still moving his teammate backward and the back gets downhill before anyone arrives.

What about the unblocked backside end? Count the box in Figure 16.3: six blockers (five linemen and the tight end) against seven defenders. One defender is always left over, and inside zone chooses the one with the longest way to go to the ball: the end farthest from the play. That does not make him harmless. If he simply chases down the line, he can tackle a back who cuts back toward him. After the handoff the quarterback fakes a bootleg (a rollout the other way, which you will meet properly near the end of the chapter) to make him hesitate; that is the only help plain inside zone gives against him. The H standing on that side in the picture is a slot receiver, too far out to block him. Later sections show the ways teams deal with him: a second tight end to block him (the U in the two-tight-end pictures), a crossing tight end to kick him out (split zone), or the quarterback reading him and keeping the ball, the zone read of 03-04.

What inside zone gives up: it is a grind-it-out play. It needs linemen strong enough to move a defensive tackle with help, and when the defense wins those double teams (a big, penetrating defensive tackle who splits the double team) the play dies in the backfield. Its average gain is modest; its virtue is that it is hard to make go badly, and it sets up everything else.

Outside zone: get the defense running sideways

Outside zone, also called wide zone, has the same rules and a different intent. Inside zone wants to move the defense backward; outside zone wants to move it sideways. The linemen take a lateral first step (some coaches teach a “bucket step,” opening the hips and stepping slightly back to gain width) and run, trying to get their heads across the play-side shoulder of the defender in front of them. “Head across” is literal, and it is the whole idea of outside zone: a blocker whose helmet and shoulders are on the play-side of a defender is standing between that defender and the ball, so the defender can only go inside, away from where the back is heading. A blocker whose head ends up on the inside has only got in the defender’s way; the defender can still run around him toward the ball. The running back’s aiming point moves out to the outside leg of the tight end (or of the tackle, if there is no tight end on that side).

Two new ideas are hiding in that picture.

The receivers’ jobs. Z stalk-blocks his cornerback: he runs at him as if on a route, then breaks down, squares up and stays between the corner and the ball. X runs his cornerback off: he sprints straight downfield so the corner, who cannot know it is a run yet, has to turn and run with him, which takes him out of the play without a block.

Reach-and-overtake. On outside zone, a covered lineman wants to reach his defender: get his head and body to the play-side of the defender so the defender is sealed inside and the running back can get outside him. Physically, a reach is a long lateral first step to gain ground, then a second step aimed through the defender’s play-side leg, with the inside hand driving into his chest to stop him running.4 The trouble is that a defender sitting in your play-side gap is already a step closer to the play than you are, and he is moving the same way you are: he only has to take one step sideways to stay outside you. Reaching him alone is close to impossible. So zone teams do it in pairs: the reach-and-overtake. The pairs are the same ones the rules produced for inside zone; what changes is who touches the defender first. The uncovered man of the pair (here the Y beside the 5-technique, and the RG beside the 1-technique) is already outside the defender, so he gets his hands on him first, just enough to stop him running wide. The covered man (the RT, the C) races flat down the line and overtakes the block, getting in front of the defender from the inside. As soon as the overtake arrives, the first lineman lets go and climbs to the linebacker. The covered man still ends up with his own defender; his uncovered neighbor just got there first to hold him. It is the outside-zone cousin of the inside-zone double team: inside zone’s combo goes forward and splits; outside zone’s combo goes sideways and hands off.

The backside cut-off. Away from the play, the linemen are not trying to move anyone. Their job is the backside cut-off: step play-side and get in front of the defenders on their side, so those defenders cannot chase the play down from behind. A backside defender who is cut off is out of the play even if he is never knocked back an inch. Notice who cuts off the backside end here: the U. Against the even front in Figure 16.1, nobody had that man; a second tight end gives the backside a blocker for him, which is a big reason zone teams like two-tight-end sets. As you will see, the backside cut-off is also what makes the cutback possible.

Why outside zone exists, and what it gives up

Outside zone was built to beat a defense that is bigger than you. You cannot drive a 320-pound defensive tackle backward, but you can make him run, and a running defensive lineman is a defensive lineman who is not holding his gap. Stretch the whole front laterally, and gaps open between defenders who are moving at different speeds. Linebackers, flowing fast to stay level with the back, over-run the play. The back only has to find one crease.

It gives up a few things in return:

  • It needs a particular kind of lineman. Outside zone favors lighter, quicker linemen who can run laterally and reach, over bigger, stronger linemen who can drive straight ahead. Teams built around outside zone have often gone smaller on the line; the scheme shapes the roster.
  • Fast movement with the line hurts it. Zone keeps every defender accounted for when the defense moves, but it cannot make movement harmless. A defensive lineman who slants hard toward the play at the snap is moving the same way as his blocker with a head start, and the blocker may never get his head in front of him. One who knifes straight upfield into the gap a lateral-stepping blocker just left is in the backfield before the block exists, and the back’s track is so wide that he is tackled for a loss. (Figure 16.13, later, draws it.)
  • It is patient by design, which means it can lose yards. The back runs sideways for a couple of yards before he turns upfield. When the crease never appears, the result is a loss.
NoteMadden vs. real life

In Madden you choose “HB Stretch,” a lane lights up, and you steer the back into it. On a real zone run, nobody knows where the lane is when the play is called. The play call tells the line which way to step and the back where to aim; the hole is created by the defense’s reaction, and the back has to see it at full speed. Two identical calls of outside zone can hit three different gaps. That is not the play failing; that is the play working. The same goes for inside zone, which is not Madden’s “HB Dive”: a dive goes to one numbered hole, while inside zone’s hole is wherever the double teams and the back’s read put it.

Stretch: the widest track

The word stretch causes confusion because teams use it differently. In many playbooks “stretch” is simply another name for outside zone. When coaches distinguish them, stretch is the widest member of the family: the aiming point moves outside the tight end, toward the numbers, and the back’s first intent is to get the edge (the outside boundary of the defense, where the last defender on the line stands). The blocking is the same lateral, reach-first zone blocking; the difference is how hard the back works to get outside before he even considers cutting up.

Why have it? Because some fronts invite it. When the defensive end lines up inside the tight end (a 7-technique, on the tight end’s inside shoulder), the tight end is already outside him before the snap: half the reach is done. By the rules the end belongs to the right tackle (he sits in the tackle’s play-side gap), so the uncovered tight end gets his hands on him, the tackle overtakes, and the pair can pin him inside and win the race to the sideline. An end lined up outside the tight end, as in the odd front below, is the opposite case: nobody can get outside him easily, and stretch becomes a bad bet. The Indianapolis Colts of the Peyton Manning era made “the stretch” a signature under line coach Howard Mudd, running it with Edgerrin James and later Joseph Addai.5 The cost is the same as outside zone’s, only more so: the wider the track, the more yards the back can lose if the edge is not there.

In this book, when a distinction matters we say inside zone (aimed inside), outside zone (aimed at the tight end, ready to cut up), and stretch (aimed outside the tight end, edge first). On a broadcast you will usually hear “outside zone” and “stretch” used interchangeably. That is fine. Watch the aiming point and you will know which one you are seeing.

Same play, different front: the rules re-pair

The most important thing about zone blocking is that the same call works against different fronts, because the rules re-pair the blockers after the defense shows its hand. Here is the same outside zone against an odd front, one with a nose tackle head-up on the center (a 0-technique), two 4i-techniques (on the inside shoulder of each tackle) and two outside linebackers on the line, each in a 9-technique (on the tight end’s outside shoulder). You will learn its name and logic in 05-03; for now it is “a different front.”

Apply the rules again (Figure 16.7). The Y is now covered (a 9-technique sits in his play-side gap), so he has to reach that man by himself, a hard block. The right tackle is uncovered: the 4i lines up on his inside shoulder, which is his backside, so that defender belongs to the right guard, whose play-side gap he sits in. So the RT and RG form the new reach-and-overtake pair: the RT gets his hands on the 4i first, the RG overtakes, and the RT climbs to the Mike. The center is covered by the nose head-up on him. The left guard is uncovered and right next to him, so he helps: the center works to the nose’s play-side half while the left guard hits the nose’s backside hip, and once the center has him, the guard climbs to the Will. The left tackle cuts off the backside 4i. The U is the last man on the backside with a defender outside him, the third edge case from the rules: he hinges, stepping inside to wall off anything that crashes down inside him, and the 9-technique outside him is left alone, as the backside end was on inside zone.

ImportantWhy it exists

A zone play is really a set of rules, not a drawing. Rules let one play call adapt to every front the defense can show, which is why zone teams can run a handful of plays all game without being predictable in the way that matters: the defense knows the call, but not where the ball will hit. Defenses fight back with fronts and movement that put the offense’s weakest blocker in the hardest job, a story you will pick up in Part 5.

Who picks inside or outside zone, and which way? The play-caller, from the sideline, as with every call (01-05). But the line of scrimmage still gets a vote. Many offenses let the quarterback flip a run to the other side (a kill call), or check out of it into a pass, when the defense’s alignment makes the called side a bad bet, for example when it shows more defenders in the box than the offense can block. How much freedom he has varies a great deal by team; the pre-snap chapter covers how checks work.

The runner’s half: bang, bend, bounce

Now the other half of the play. The zone back’s job is to take the handoff, press the aiming point (run at it hard enough that the defense believes the ball is going there), read the blocks, and choose. Coaches compress the choice into three words: bang-bend-bounce.

  • Bang: hit the designed gap and go straight upfield. The blocks went as drawn.
  • Bend: cut back against the flow, behind the double teams, usually into the backside A or B gap. The defense flowed too fast, and the space is behind them.
  • Bounce: take it outside. The edge defender got pinned inside, and the corner is open.

The order matters, and it is different for the two plays.

On inside zone the back thinks bang, bend, bounce. He reads the first defensive lineman play-side of the center, the man the first double team is working on. Why that man? Because he sits right next to the aiming point, and what happens to him answers both questions at once: if the double team is moving him, the play-side gap beside him is open; if he is sliding toward the play to get in front of it, he is taking the space behind him out of play for the defense, and the linebackers behind him usually flow the same way. One defender tells the back where the defense is going. If that defender is walled off and there is daylight beside him, bang. If he (and the linebackers behind him) flowed hard to the play-side, there is space behind them: bend. Bounce is the last resort on inside zone, because running outside on a play built to go inside usually means running away from your blockers. It is the right answer only when the play-side end has crashed inside and nobody has stayed outside to take his place.

On outside zone the order flips to bounce, bang, bend, and the back makes two reads in order. His first read is the play-side end, the first defensive lineman outside the tackle: is he reached? Remember who ends up with him: the reach-and-overtake pair, with the right tackle taking him over from the tight end. If the end is slow, or plays inside, and the tackle gets his head across him and pins him inside, the edge is open: bounce. That is uncommon against a good end lined up outside the tackle, which is why most outside-zone runs end up as bang or bend. If the end fights hard to stay outside, the tackle rides him wider and the back moves to his second read, the next defensive lineman inside, and asks where the hole is relative to him. If that man has been walled off, there is a crease between him and the end: bang, up inside the end. If he and the linebackers are racing to the sideline, the space is back the other way: bend, which on outside zone often becomes a full cutback (next section). The second read is the one to watch if you want to see a cutback coming.

Coaches teach this read in slightly different words and keys (some read a linebacker rather than a lineman), and you should expect that variety.6 The principle under all of them is the same: press, read, then make one cut.

For your own log, keep the boundary simple: bang if the back crosses the line in the play-side gaps, bounce if he goes outside the play-side end, bend if he comes back against the flow. Every bend is a cutback; the big ones that carry him outside the backside tackle are what broadcasters mean by “he found the cutback.”

That last phrase of the principle describes a style. A one-cut runner presses the aiming point with patience, plants one foot, and goes straight upfield; he does not dance, stutter or reverse field behind the line. Zone coaches prize this style above raw speed, because a zone run’s creases are open for a fraction of a second and close quickly. A back who hesitates or takes two cuts arrives after the hole has closed. It is why zone teams have often won with backs drafted in late rounds or signed off other teams’ rosters: the scheme rewards decisiveness and vision more than it rewards a top-10 pick’s raw talent. (The 1990s Broncos are the classic example; see below.)

The cutback and the backside

Here is the play from the chapter’s opening: second-and-7, two tight ends, the line steps right in unison, the back runs at the right tight end, plants, and comes back through a hole near the center.

A cutback is a cut against the flow of the play, back toward the side the line stepped away from. On outside zone it is not a desperation move. It is one of the three answers the play was designed to produce. Here is why it works. When eleven defenders see five linemen and a running back all racing to their right, they race to their right too. Linebackers who are taught to stay level with the ball carrier take aggressive angles, and a fast defense can over-run the play: everyone ends up play-side of the ball. The space behind them is empty, as long as the backside defenders cannot get there first.

That condition is what the backside cut-off is for. The backside linemen do not need to move anyone; they need the defensive linemen on their side to be on the ground, turned around, or walled away from the cutback lane. When they succeed, the cutback is the most explosive play in the zone family, because the back runs into space that the defense vacated on purpose.

CautionCommon misconception

“The cutback is the running back freelancing.” It is the opposite. The back is following the play’s own logic: press the aiming point to pull the defense one way, then take the space the defense left. Zone coaches often say the back should expect to cut back on outside zone, and coach the backside linemen as if the ball will come their way. A back who never cuts back makes outside zone easy to defend, because the defense can then over-pursue for free.

New block: the cut block

To clear a cutback lane quickly, backside linemen often do not try to stand a defender up. They cut him. A cut block is a block aimed at a defender’s thighs or lower legs, from the front, to put him on the ground. A defender on the ground cannot chase, and for a lighter zone lineman facing a heavier defensive tackle who is trying to run down the line, a cut is often the most reliable way to take him out of the play. “From the front” matters, because the backside defender is running sideways, chasing the play: the blocker’s play-side step puts him in front of that path, so the defender runs into the block instead of being hit from behind or the side.

It is also the most controversial block in football, because a block at the knees can end a career. The NFL has narrowed it again and again: low blocks are now legal only near the ball, inside the “tight end box” around the line; never from behind or from the side while moving back toward your own goal line (since 2013); and never combined with a high block by a teammate, the chop block, banned in every situation since 2016.7 The details are in 09-03. Mike Shanahan’s Denver lines were criticized for years for backside cuts; in 2004, after a Broncos lineman’s cut block broke a Bengals defensive tackle’s ankle on Monday Night Football, opposing head coaches and the broadcast crew criticized the team publicly.8 Defenders call it dirty; zone coaches call it legal technique that lets small linemen block big ones. With the rules this much tighter, how often a modern line cuts on the backside depends on its line coach. When you do see a backside defensive tackle go down at the snap on a zone run, you have probably just seen one.

The line blocks exactly as it would on inside zone. The difference is one man: an H-back or tight end, lined up off the ball on the play-side, who at the snap runs across the formation, just behind the linemen’s heels and in front of the handoff, and kicks out the backside end. The name comes from the picture: the play “splits,” with the line and the back going one way and the H going the other.

Under center or in the shotgun. Every picture before this one put the quarterback under center, where he turns his back to the defense, hands off and fakes the other way; turning away hides the ball and puts him in position to carry out the fake. From the shotgun, where most NFL snaps happen, the back usually lines up beside the quarterback and on inside zone runs across the quarterback’s face at the handoff: a back offset to the left usually runs inside zone to the right. That gives him a running start toward the aiming point and keeps the exchange (the mesh, the moment the ball changes hands; not the pass concept of the same name in 04-04) moving downhill. Defenses know this tendency, so offenses break it on purpose with same-side zone (the back runs to the side he lined up on) and with the pistol, where the back stands directly behind the quarterback and his alignment says nothing about direction.

Why it exists. Split zone solves three problems at once. It blocks the one defender plain inside zone leaves free. It protects the cutback, so the back can bend without an end waiting for him. And the H’s path is misdirection: a linebacker who keys the H (follows his movement to find the play) takes a false step the wrong way. It also builds a family. The same H motion across the formation is the action for bootlegs and for passes to the H in the flat (the short area near the sideline, within a few yards of the line), so the defense cannot simply crash on the H’s path.

What it costs. The H is no longer available to block play-side, so the play-side is one blocker lighter than it could be. If the defense brings an extra defender to the play-side, split zone can be outnumbered there. Every zone variation is this kind of trade: put a blocker where the threat is, and accept that he is not somewhere else.

Insert. On insert (or “zone insert”), a fullback, H-back or tight end leads up through a gap and blocks a linebacker, usually the play-side one (some staffs insert on the backside linebacker instead). That frees the play-side double team from having to climb: it can stay on the defensive lineman longer and move him farther. In effect, insert adds a blocker at the point of attack, which is how a zone team answers a defense that keeps one more defender in the box than the offense has blockers. The cost is the usual one: the inserting player is not available as a receiver, and his alignment (an offset fullback, or a tight end off the ball) can tip the play.

The other two tags borrow an idea from option football, which gets its own chapter: instead of blocking the unblocked backside end, the quarterback reads him (watches what he does after the snap) and decides whether to hand the ball off or keep it. For now, all you need is that idea; 03-04 builds it properly.

Arc. Arc happens on the backside, away from the line’s flow. The line zones right and the back runs inside zone to the right, while on the left a tight end or H-back makes an arc block: he releases outside the unblocked backside end, runs past him without touching him, and blocks the next defender out, the linebacker or safety responsible for the alley between the end and the sideline (the alley player of Part 5). The quarterback reads the end. If the end stays home, the quarterback gives and the back runs inside zone. If the end chases the running back down the line, the quarterback keeps, runs around the end the other way, and finds the arc block already waiting for the next man. Arc is why you sometimes see a tight end run right past a defensive end without touching him: he was never supposed to.

Bash needs one more step of background. On the standard zone read (03-04 draws it), the back runs the inside-zone track and the quarterback, reading the backside end, keeps the ball around that end if the end chases the back. Defenses learned an answer: the end crashes down hard at the back, inviting the keep, while a linebacker loops around behind him to meet the quarterback where he was about to run (a scrape exchange: the end and the linebacker swap jobs).9 The defense’s plan is a bet that the back will be inside and the quarterback outside.

Bash. Bash is short for “back away”: the running back runs away from the direction the line is zoning, on a wide path around the read end, while the quarterback, if he keeps, runs the inside-zone track behind the line (Figure 16.12). It flips the usual roles: the running back threatens the edge and the quarterback becomes the inside runner. The read is the same end, with the answers reversed: if he squeezes (closes down the line toward the zone flow), the quarterback hands the ball to the back going around him; if he stays wide to take the back, the quarterback keeps and runs inside him.10 Against a scrape exchange, that matters: the end who crashes down to take the inside runner has crashed onto the quarterback’s fake, and the ball is going outside him with the running back. Bash is the offense’s reply to the defense’s reply, and the full back-and-forth belongs to option football.

All three tags share one logic: keep the zone rules for the line, and change one or two players to fix the specific problem the defense is creating. Watch for the extra player (the inserting back, the arcing tight end, the back running the “wrong” way) and you will usually know which tag you are watching.

Penetration and movement. A defensive lineman who gets upfield at the snap, or slants hard toward the play, can beat a lateral-stepping blocker to the spot. Defenses that want to stop zone often play their linemen this way, attacking a gap instead of reading the man in front of them; how defensive linemen are taught to do either is the subject of 05-01.

The cutback player. Because outside zone’s biggest gains come from the cutback, defenses assign someone to stay home for it: usually the backside linebacker, sometimes a safety rotating down. You saw the deep safety arrive late in Figure 16.9; a defense that gets a defender there early takes away the play’s third answer.

Spill or squeeze. When a blocker comes at a defender, the defense chooses whether that defender fights inside the block to force the ball to bounce wide toward pursuit (spill) or holds the edge and squeezes the play back inside (squeeze). Against zone, either plan can work if all eleven players agree on it, which is the subject of run fits in 05-05.11

And when the defense simply puts more defenders in the box than the offense has blockers, a zone team answers with the tools you have already seen: an extra blocker (insert), a read of the unblocked man (arc, bash, the zone read), or a pass that looks like the run.

Why wide zone and the bootleg are married

Outside zone’s greatest weapon is not a run at all. It is the pass that looks exactly like it.

Think about what outside zone teaches a defense. Every snap, the line steps hard to one side, the back runs at the tight end, and the linebackers learn that if they do not flow fast and hard, they lose. The backside end learns that if he does not chase, the cutback beats him. After a few of these, the defense is trained to run with the line’s first step.

The bootleg cashes that in. It is a play-action pass (a pass that begins with a fake handoff) in which the quarterback fakes outside zone one way and rolls out the other, away from where the line went, with receivers crossing the field to meet him. In the “naked” version nobody blocks for him at all; in the protected version a lineman or a back is assigned to the backside end. Both are built in 04-05; here, what matters is why they pair with zone.

The two plays look identical for the first second, which is the second that decides how the defense moves. (The “over” route is a deep crossing route: the receiver runs across the field 15 or more yards downfield, over the linebackers.) On a bootleg the backside end’s job is to contain the quarterback: stay outside him and force him back toward the pursuit. An end who chased the run fake (as outside zone has taught him to) is now behind a quarterback rolling away from him, and nobody is there to make the quarterback hurry. The linebackers who flowed with the line are a step or two the wrong way, and the crossing routes (receivers running across the field) come open behind them. The boot gives the quarterback three receivers at three depths on the same side: a short one in the flat, one crossing at linebacker depth, and one deep. That simple read is what makes the play so reliable.

Why the pair works. Each play protects the other. If the defense sits back and refuses to flow, outside zone gets the edge or the crease. If it flows hard, the bootleg punishes the flow. And because the two share the same first step, the defense cannot wait to find out which one it is facing. This is the backbone of the Shanahan-family offenses you will meet in 12-03: a run game and a play-action game built to be indistinguishable for as long as possible.

How to tell them apart after the snap. Three plays send a tight end or H-back at the backside end, and before the snap they can look the same. Watch what he does when he gets there:

  • He hits the end and drives him out: split zone.
  • He runs past the end without touching him and blocks the next man out (a linebacker or safety), while the quarterback rides the ball with the back and watches the end: arc.
  • He runs past the end into the flat and turns to look for the ball, while the quarterback pulls the ball and rolls the other way: bootleg.

The cost. A naked bootleg leaves the quarterback with nobody blocking the backside end. If that end does not bite on the fake (if he stays home), he is waiting for a quarterback who has turned his back to him. That is why the bootleg works only once outside zone has earned the defense’s respect.

Where zone came from

Running backs reading blocks is as old as football. Vince Lombardi’s Green Bay Packers of the 1960s ran their famous power sweep with a simple instruction to the ball carrier: run to daylight, wherever the blocks open it, a phrase that became the title of a 1963 book Lombardi wrote with W. C. Heinz.12 That sweep was a gap play with pulling guards, not a zone play, but its runner’s freedom to choose the hole is the same idea zone backs live by.

Who invented zone blocking is disputed, and nobody serious claims a single inventor. The best-documented claims put it in the NFL of the 1980s. Jim McNally, the Cincinnati Bengals’ offensive line coach, was one of the era’s most copied line coaches, and Cincinnati’s run game finished in the league’s top five every year from 1986 to 1990. McNally himself said, “I didn’t invent zone blocking, but we and the Cleveland Browns did it back in the ’80s,” and he described it as an answer to the rise of the 3-4 defense and its penetrating linemen. Cleveland’s line coach in those years was Howard Mudd, the same coach who later made the stretch a Colts signature.13 Coaches who credit older roots point to blocking ideas in earlier college offenses; treat any one-name “inventor” story with suspicion.

What is not in dispute is who made outside zone the identity of an NFL offense. Alex Gibbs, an offensive line coach with a famously blunt teaching style and a reputation as one of the best line coaches of his era, built the Denver Broncos’ running game under head coach Mike Shanahan from 1995.14 Denver had a specific problem to solve. Pittsburgh defensive coordinator Dom Capers had popularized the zone blitz, in which a linebacker blitzes and a lineman drops into coverage in his place, so the defenders a man-blocking line had assigned itself kept changing. “When you’re in a man scheme, you’re getting picked off and you look bad,” Gary Kubiak, then Denver’s offensive coordinator, later remembered.15 (In blocking, “picked off” means a blocker’s path is cut off by a defender he was not assigned, so his own man runs free: exactly the left side of Figure 16.2.) Blocking by area meant that whoever came was blocked.

Gibbs’s system was close to a religion: a small number of runs, repeated until the linemen could execute them against any front; light, quick linemen who could reach and run; backside cut blocks; and backs taught to press the aiming point and make one cut. His clinic tapes on zone running circulated widely among coaches and are still sold as coaching videos, and many of the phrases in this chapter descend from his teaching.16

Denver’s results became the advertisement. The Broncos won back-to-back Super Bowls after the 1997 and 1998 seasons, and running back Terrell Davis, a sixth-round draft pick, ran for 2,008 yards in 1998 and was named the league’s MVP.17 When injuries ended Davis’s peak, Denver kept producing 1,000-yard rushers with a string of other backs, several of them mid- or late-round picks, which convinced the league that the scheme was doing much of the work.18

The system spread through a coaching tree. Gary Kubiak carried it to Houston as head coach and back to Denver, where he won Super Bowl 50 after the 2015 season. Kyle Shanahan, Mike’s son, coached under Kubiak in Houston and under his father in Washington, then became the 49ers’ head coach in 2017. From Kyle Shanahan’s staffs came Sean McVay (Rams), Matt LaFleur (Packers) and others, and Kubiak’s son Klint called plays for the 2025 Seattle Seahawks before becoming the Raiders’ head coach for 2026.19 By late in the 2019 season, ESPN’s charting found zone blocking in use by all 32 teams and on at least half of the runs of most of them; LaFleur’s Packers led the league at 70.3% of runs (same ESPN study). The tree and its offense get a full case study in 12-03.

Does it work? Reading the numbers

You might expect a data source to tell us how well zone runs work. It cannot, directly, and the reason is worth understanding before looking at any chart.

The NFL’s play-by-play data (which nflverse packages for free) records where a run went, not what play was called. Each run gets a run location (left, middle or right) and, unless it went up the middle, a run gap (guard, tackle or end). These are charted by the league’s scorers from where the ball carrier crossed the line, and they are coarser than the gap letters of 03-01: roughly, “middle” is between the guards (the A gaps), “guard” is over or just outside a guard (the B gap), “tackle” is over or just outside a tackle (the C gap), and “end” is outside the last man on the line. There is no “zone” flag. So a zone run that cuts back is logged where the back ended up, not where the play was aimed. (Commercial charting services such as PFF, Pro Football Focus, do label run concepts by watching film. In the sample of their labels released with the 2025 Big Data Bowl (2022 season), outside zone was the single most common run concept.25 That is the most recent public count we know of; no league-wide zone share for 2025 has been published that we could check.)

So to study zone we need a proxy: a team we know runs a lot of zone. We use Kyle Shanahan’s 49ers. LaFleur’s Packers ran an even higher share of zone in ESPN’s 2019 count, and would also do; the 49ers are the root of the coaching tree and the subject of the 12-03 case study, and Shanahan was their head coach for every season in the chart. (The Packers’ number appears below too.)

One more thing before the chart: expect negative numbers. Across the league, the average running-back carry loses a little expected value, because a typical run gains less than the down and distance needs to keep the offense’s scoring chances where they were. So “good” on this chart means “less negative than the league,” and a run game is judged by how far above the league line it sits, not by whether it clears zero.

Only running back and fullback carries count, so quarterback runs and receivers’ jet sweeps and end-arounds, which are not zone runs, stay out.

Four things to read from Figure 16.15.

Runs that reach the end are the least bad, for everyone. League-wide, running back carries logged at the end averaged -0.022 EPA per rush, against -0.083 up the middle. That does not mean “always run outside.” It means that when a run gets outside, it is usually because the edge was open, and the runs that tried to get outside and failed were often logged somewhere else (or as losses at the tackle). This is a selection effect (backs only go outside when the edge is open, so the runs that get there are the lucky ones), and it is why no coordinator reads this chart as a play-calling instruction.

The 49ers were a little better everywhere, most of all at the two ends of the chart. Their EPA edge over the league was +0.031 per carry at the end and +0.025 up the middle, against +0.015 at guard and +0.011 at tackle. That fits outside zone’s two big answers, the bounce (logged at the end) and the cutback (often logged up the middle), but the gaps are small, a few hundredths of a point per carry, and the chart cannot prove the scheme caused them.

Success rate tells a slightly different story. San Francisco’s success rate beat the league’s by 2.6 points at guard and 2.5 up the middle, but only 0.2 at tackle. Success rate rewards steady, positive gains; EPA also rewards big ones. A team can be consistent inside and explosive outside, and the two panels let you see both.

Overall, good, not magic. Across all seven seasons the 49ers’ running back carries averaged -0.043 EPA per rush against a league average of -0.064, ranking 9th of 32 teams; only one team averaged above zero. LaFleur’s Packers, the other famous zone offense of the period, averaged -0.010 and ranked 2nd. So two of the period’s best-known zone offenses landed in quite different places, both above the league average. That spread is the lesson: zone is one input to a run game, alongside the backs, the linemen, the quarterback and what the passing game makes defenses fear. Part of the 49ers’ edge is surely talent rather than scheme (McCaffrey, the 2023 Offensive Player of the Year, carried for them), and run-location data cannot separate the two; that would take play-level scheme labels and a model of player quality. Teams still run, even though the average carry loses a little expected value, because a run game does things a single carry’s EPA does not capture: it keeps the defense honest, it shortens third downs, and above all it sets up the play-action passes that look like it. You will see that argument tested with data in 13-02.

CautionCommon misconception

“Run location tells you which runs a team called.” It tells you where the ball carrier went. On zone runs especially, those are different things: an outside zone that cuts back is logged up the middle. Treat run-location splits as a picture of outcomes, and be suspicious of anyone who ranks “best run plays” from them.

TipWatch for it

On Sunday: bang, bend, bounce. You cannot watch everything in two seconds, so watch in this order.

Before the snap (you have time here)

  • Count the tight ends and find the running back. Two tight ends and a back directly behind the quarterback (or in the pistol) is a natural zone look. In the shotgun, an inside-zone back usually runs away from the side he lines up on, crossing the quarterback’s face. (How deep the back stands varies too much by team to be a reliable tell on its own.)
  • Find the backside end, the defender farthest from where you think the play is going. Remember where he is; you will check on him after the play.

The first half-second: watch the line, nothing else

  • Do all five linemen step the same way? Zone. Does a guard pull? Gap scheme (03-03). Double teams but no zone steps, and the back aiming straight at a linebacker? Probably duo, also in 03-03.
  • Are the linemen’s helmets going up (inside zone) or across (outside zone)? The high end-zone replay angle shows this best (01-06); on the live sideline angle, look for the whole line drifting toward one sideline.

Then: watch the back’s feet

  • Where is he aiming: the guard, or the tight end? Wider than the tight end is stretch.
  • Watch him press, plant and cut. Call it as it happens: bang (play-side gaps), bend (back against the flow) or bounce (outside the end).

After the play (on the replay)

  • Did a tight end or H-back go at the backside end? Hit him: split zone. Ran past him to the next man: arc. Slipped into the flat looking for the ball: bootleg. Did someone lead up into the line? Insert.
  • Where was the backside end when the back made his cut? On the ground (cut), chasing (and how close), or blocked?
  • If the defense has been flowing hard all game, start watching for the bootleg off the same look.

For your prediction log (01-06): before each zone run write I (inside), O (outside) or S (split) from the alignment; afterwards write which of bang, bend or bounce the back chose.

Most likely to the right, toward the tight end. From the shotgun, inside zone usually has the back cross the quarterback’s face: he lines up on one side and runs to the other, which gives him a running start toward the aiming point and a downhill angle at the mesh. With the tight end on the right too, the right side is also where the offense has its extra blocker.

Does the front change it? Not the direction. Run the rules to the right: the RG is covered by the 3 and the RT by the 7, so the RG has no uncovered neighbor to help him and the 3-technique is the hardest block on the play side. That tells you where trouble is likely (a 3 who wins knocks the back off his track, and he bends), not which way the play is called. The back’s alignment is the stronger tell.

Why run here: two deep safeties and six in the box means six blockers for six defenders: “hat on a hat,” a fair fight up front, with both safeties starting 11–12 yards away. That is exactly the look zone teams want to run against, and it is a common look now: two-high shells rose from about a third of snaps in 2018 to 42% in 2025.26

The catch: good defenses know the “cross the face” tendency too, which is why offenses mix in same-side zone and the pistol. If you see a team do that a lot, that is why.

Takeaways

  • Zone blocking assigns blockers by rule, not by name: step play-side; block the man head-up on you or in your play-side gap; if nobody is, help the neighbor on your backside and climb to a linebacker. Because the rules are applied after the snap, slants and blitzes get picked up, and the same call re-pairs itself against every front.
  • Inside zone attacks the guard-tackle area with double teams that drive defenders backward; outside (wide) zone aims at the tight end’s outside leg and moves the defense sideways with reach-and-overtake blocks; stretch is the widest track, edge first; split zone adds a crossing H to kick the backside end. Tell them apart by the line’s first step (up or across) and the back’s aiming point.
  • The running back makes the play’s last decision: press the aiming point, then bang, bend or bounce, with one cut. On inside zone the order is bang-bend-bounce; on outside zone it is bounce-bang-bend.
  • The cutback is designed, not improvised: it works because the defense over-runs the play and the backside cut-off keeps the backside defenders out of the lane (in Gibbs’s Denver, often with a cut block, a legal but controversial low block that the rules have steadily narrowed).
  • Tags change one or two jobs to fix one problem: insert adds a lead blocker, arc runs past the unblocked end to block the next man while the quarterback reads the end, bash sends the back away and makes the quarterback the inside runner. Defenses fight back with penetration, a cutback player and coordinated run fits.
  • Outside zone’s biggest payoff is the bootleg that looks identical for the first second. The data shows zone running is efficient for a run game, not magic; its value is in what it sets up.

Names you will hear. Zone vocabulary is a dialect map. Alex Gibbs called inside zone tight zone, and many coaches still do. Some analysts describe a mid zone between inside and outside zone, in which the play-side tackle pushes the end outward instead of trying to reach him, so the back’s read moves to the defensive tackles; one coach-analyst used the name for Sean McVay’s Rams’ base run.27 Split zone is also called zone slice, and bash is sometimes spelled out as back away. And duo, a two-back or one-back run with double teams that a casual viewer can mistake for inside zone, is not a zone play at all: the linemen do not take zone steps and the back reads a linebacker. It belongs to the gap family in 03-03.
