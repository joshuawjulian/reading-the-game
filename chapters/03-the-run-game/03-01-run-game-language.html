GREEN = "#0f7a55"         # darker step of the aqua highlight, for text on the light field
PURPLE = style.SERIES[6]  # a ring colour that is neither offense blue nor defense orange
OL_KEYS = ("LT", "LG", "C", "RG", "RT")


def pdf_build():
    return os.environ.get("RTG_FORMAT", "").lower() in ("pdf", "typst", "print")


def O(key, pos, d, w, label):
    return Player(key, pos, "off", d, w, label=label)


def D(key, pos, d, w, label):
    return Player(key, pos, "def", d, w, label=label)


def line(labels="full"):
    """The five offensive linemen, labelled LT..RT ("full") or with one letter, T G C G T ("short")."""
    short = {"LT": "T", "LG": "G", "C": "C", "RG": "G", "RT": "T"}
    return [p.moved(label=(p.key if labels == "full" else short[p.key])) for p in offense("singleback")
            if p.key in OL_KEYS]


def te(side=1, key=None, label=None):
    """An in-line tight end: Y on the right, U on the left."""
    key = key or ("Y" if side > 0 else "U")
    return O(key, "TE", -0.7, side * 4.8, label or key)


def T(t, side=1):
    """Width of technique t on the offense's right (side=+1) or left (side=-1)."""
    return technique(t, side)


def go_to(p, key, end, kind="block", delay=0.0, speed=None, via=()):
    """Move `key` through absolute (d, w) points `via` to `end`, as a block, path or run."""
    start = np.array(p._cur(key))
    pts = [tuple(np.array(q) - start) for q in (*via, end)]
    if kind == "block":
        return p.block(key, pts=pts, delay=delay, speed=speed)
    if kind == "pull":
        return p.pull(key, pts, delay=delay, speed=speed)
    return {"path": p.path, "run": p.run}[kind](key, pts, delay=delay, speed=speed)


def _P(fld, p, d, w):
    return fld.xy(*p.to_abs(d, w))


def say(fld, p, d, w, text, color=style.INK_2, fontsize=7.5, ha="center", weight="normal"):
    """A short on-field label at (d, w)."""
    x, y = _P(fld, p, d, w)
    fld.ax.text(x, y, text, fontsize=fontsize, color=color, ha=ha, va="center", zorder=9, fontweight=weight,
                bbox=dict(boxstyle="round,pad=0.18", fc=style.SURFACE, ec="none", alpha=0.9))


def point_to(fld, p, text_at, target, text, color=style.INK_2, fontsize=7, shrink=7):
    """A label with a thin leader line to the player or spot it names."""
    (xt, yt), (xp, yp) = _P(fld, p, *text_at), _P(fld, p, *target)
    fld.ax.annotate(text, (float(xp), float(yp)), xytext=(float(xt), float(yt)), fontsize=fontsize,
                    color=color, ha="center", va="center", zorder=9,
                    arrowprops=dict(arrowstyle="-", color=color, lw=0.7, shrinkA=2, shrinkB=shrink),
                    bbox=dict(boxstyle="round,pad=0.18", fc=style.SURFACE, ec="none", alpha=0.9))


def gap_tag(fld, p, d, w, text, color=GREEN, fontsize=9):
    """A gap letter or hole number in a small rounded box."""
    x, y = _P(fld, p, d, w)
    fld.ax.text(x, y, text, fontsize=fontsize, color=color, ha="center", va="center", zorder=9,
                fontweight="bold", bbox=dict(boxstyle="round,pad=0.22", fc=style.SURFACE, ec=color, lw=1.0))


def pair_oval(fld, p, d, w, width, height=2.5, color=GREEN):
    """A faint oval around a double team (two blockers and their defender)."""
    x, y = _P(fld, p, d, w)
    fld.ax.add_patch(Ellipse((float(x), float(y)), width, height, fc=color, alpha=0.22, ec="none", zorder=1.5))
    fld.ax.add_patch(Ellipse((float(x), float(y)), width, height, fc="none", ec=color, lw=1.6,
                             ls=(0, (4, 2)), zorder=1.6))


def spot(fld, p, d, w, color=style.THIRD, ms=9):
    """Mark a point (the point of attack, an aiming point) with an X."""
    x, y = _P(fld, p, d, w)
    fld.ax.plot([x], [y], marker="X", ms=ms, color=color, mec="white", mew=0.8, zorder=9)


def shade_box(fld, p, w0, w1, d1=6.5, d0=-0.2, color=style.THIRD, alpha=0.08, label=None):
    """Shade the box (or any rectangle) from width w0 to w1 and depth d0 to d1."""
    (x0, y0), (x1, y1) = _P(fld, p, d0, w0), _P(fld, p, d1, w1)
    fld.ax.add_patch(Rectangle((min(x0, x1), min(y0, y1)), abs(x1 - x0), abs(y1 - y0), fc=color, alpha=alpha,
                               ec=color, lw=0.9, ls=(0, (3, 2)), zorder=0.8))


def dashed(fld, p, pts, color=style.THIRD, lw=1.6, dash=(0, (3, 2.2)), end="arrow"):
    """A dashed 'option' or 'intent' path through absolute (d, w) points."""
    xs, ys = _P(fld, p, np.array([q[0] for q in pts]), np.array([q[1] for q in pts]))
    fld.ax.plot(xs[:-1] if end == "arrow" else xs, ys[:-1] if end == "arrow" else ys, color=color, lw=lw,
                ls=dash, zorder=6)
    if end == "arrow":
        fld.ax.add_patch(FancyArrowPatch((xs[-2], ys[-2]), (xs[-1], ys[-1]), arrowstyle="-|>",
                                         mutation_scale=10, color=color, lw=lw, zorder=6))


def play_arrow(fld, p, d, w0, w1, text, fontsize=7.5):
    """A broad 'the play goes this way' arrow at depth d, from w0 to w1."""
    (x0, y0), (x1, y1) = _P(fld, p, d, w0), _P(fld, p, d, w1)
    fld.ax.add_patch(FancyArrowPatch((x0, y0), (x1, y1), arrowstyle="-|>", mutation_scale=14,
                                     color=style.OFFENSE, lw=2.0, zorder=3))
    xm, ym = _P(fld, p, d + 0.9, (w0 + w1) / 2)
    fld.ax.text(xm, ym, text, fontsize=fontsize, color=style.OFFENSE_DARK, ha="center", va="center", zorder=4,
                fontweight="bold")


def clean(ax):
    """Box-zoomed diagrams: drop hash ticks and yard numbers, clip anything outside the window."""
    for ln in list(ax.lines):
        xs, ys = ln.get_data()
        if len(xs) == 2 and ln.get_linewidth() == 0.5 and np.hypot(xs[1] - xs[0], ys[1] - ys[0]) < 1.0:
            ln.remove()
    for t in list(ax.texts):
        if t.get_text().isdigit() and t.get_alpha() == 0.55:   # yard numbers, not technique labels
            t.remove()
        else:
            t.set_clip_on(True)


def draw(p, width_in=5.8, ax=None, **kw):
    """Play.draw, sized for the page, cleaned for a box-sized window."""
    fld = p.draw(width_in=width_in, ax=ax, **kw)
    clean(fld.ax)
    return fld


# --- a front, labelled by technique number ---------------------------------------------------------
def front(spec, lbs=(), safeties=()):
    """Defensive linemen from (key, technique, side) triples, labelled with their technique number,
    plus (key, d, w, label) linebackers and safeties."""
    out = [D(k, "DL", 1.0, T(t, s), t) for k, t, s in spec]
    out += [D(k, "LB", d, w, lab) for k, d, w, lab in lbs]
    out += [D(k, "S", d, w, lab) for k, d, w, lab in safeties]
    return out


def cov_labels(fld, p, status, d=-2.0, fontsize=6.6):

        say(fld, p, d - (0.95 if v == "uncovered" else 0.0), p.players[k].w, v, fontsize=fontsize,
            color=style.DEFENSE_DARK if v == "covered" else GREEN)

MARK_R = 0.62


def tracking(play):
    df = play.to_tracking()
    df["_id"] = df["displayName"].astype(str)
    return df


def panel(ax, play, df, t, lateral, window, rings=(), enter=()):
    """One moment of a simulated play on `ax`: markers, trails of on-screen players, rings."""
    tmap = df.groupby("frameId")["time"].first()
    fid = int((tmap - t).abs().idxmin())
    ylo, yhi = sorted((play.ball_y - lateral[0], play.ball_y - lateral[1]))
    fld = Field(ax=ax, orient="vertical", los=play.los, window=window, numbers=False, hashes=False,
                lateral=(ylo, yhi), to_go=play.to_go)
    cur = df[df["frameId"] == fid]
    snap = df[df["frameId"] == int((tmap - 0.0).abs().idxmin())]
    first = snap[snap["x"].between(play.los + window[0] + 0.8, play.los + window[1] - 0.8)
                 & snap["y"].between(ylo + 0.8, yhi - 0.8)]       # players whose marker fits whole at the snap ...
    ids = set(first["_id"]) | {"football"}
    inside = cur[cur["x"].between(play.los + window[0], play.los + window[1]) & cur["y"].between(ylo, yhi)]
    ids |= set(inside["_id"]) & set(enter)                         # players allowed to walk into the picture
    shown = cur[cur["_id"].isin(ids) | (cur["side"] == "ball")]    # ... stay in every frame (clipped at the edge)
    trails = df[(df["frameId"] <= fid) & df["_id"].isin(ids) & (df["side"] != "ball")]
    for (_, side), g in trails.groupby(["_id", "side"]):
        xs, ys = fld.xy(g["x"].values, g["y"].values)
        ax.plot(xs, ys, color=style.OFFENSE if side == "off" else style.DEFENSE, lw=0.9, alpha=0.4, zorder=3)
    ppy = fld.pts_per_yard()
    for _, r in shown.iterrows():
        x, y = (float(v) for v in fld.xy(r["x"], r["y"]))
        if r["side"] == "ball":
            ax.add_patch(Ellipse((x, y + 0.45), 0.7, 0.42, fc=style.BALL, ec="white", lw=0.5, zorder=9))
            continue
        lab = str(r.get("label", "") or "")
        if r["side"] == "off":
            ax.add_patch(Circle((x, y), MARK_R, fc=style.OFFENSE, ec="white", lw=0.6, zorder=8))
            tc = "white"
        else:
            ax.add_patch(RegularPolygon((x, y), 4, radius=MARK_R * 1.3, orientation=np.pi / 4,
                                        fc=style.DEFENSE_TINT, ec=style.DEFENSE, lw=1.0, zorder=8))
            tc = style.INK
        fs = max(4.0, min(8.0, ppy * (0.85 if len(lab) <= 1 else 0.62)))
        ax.text(x, y, lab, ha="center", va="center", fontsize=fs, color=tc, fontweight="bold", zorder=10)
    for key, col in rings:
        r = cur[cur["displayName"] == key].iloc[0]
        x, y = fld.xy(r["x"], r["y"])
        ax.add_patch(Circle((float(x), float(y)), 0.95, fc="none", ec=col, lw=1.6, zorder=7))
    for patch in ax.patches:
        patch.set_clip_on(True)
    return fld


def strip(play, times, lateral, window, notes=None, rings=(), ncols=4, width_in=6.5, after=None, enter=()):
    """Numbered frame strip with a fixed, box-sized window, so markers stay legible in print."""
    df = tracking(play)
    nrows = int(np.ceil(len(times) / ncols))
    pw = width_in / ncols
    span_w, span_d = lateral[1] - lateral[0], window[1] - window[0]
    fig, axes = plt.subplots(nrows, ncols, squeeze=False,
                             figsize=(width_in, nrows * (pw * span_d / span_w + (0.55 if notes else 0.25))))
    for ax in axes.flat[len(times):]:
        ax.axis("off")
    for i, (ax, t) in enumerate(zip(axes.flat, times)):
        fld = panel(ax, play, df, t, lateral, window, rings, enter)
        if after:
            after(fld, t)
        ax.set_title(f"{i + 1}.  " + ("snap" if abs(t) < 1e-6 else f"{t:+.2f} s".rstrip("0")),
                     fontsize=8.5, color=style.INK_2, loc="left")
        if notes:
            ax.text(0.5, -0.03, notes[i], transform=ax.transAxes, ha="center", va="top",
                    fontsize=7.2, color=style.INK)
    fig.tight_layout(h_pad=1.3 if notes else 0.6, w_pad=0.5)
    return fig


def video(play, times, lateral, window, notes=None, rings=(), width_in=5.2, after=None, enter=()):
    """Web version of the strip: the same panel at every frame, with the strip's notes as captions."""
    import shutil
    from matplotlib import animation
    from IPython.display import HTML, display
    df = tracking(play)
    span_w, span_d = lateral[1] - lateral[0], window[1] - window[0]
    fig, ax = plt.subplots(figsize=(width_in, width_in * span_d / span_w + 0.4))
    tmap = df.groupby("frameId")["time"].first()

    def update(fid):
        ax.clear()
        t = float(tmap[fid])
        fld = panel(ax, play, df, t, lateral, window, rings, enter)
        if after:
            after(fld, t)
        txt = ("snap" if abs(t) < 0.05 else f"{t:+.1f} s")
        if notes:
            cur = [n for tt, n in zip(times, notes) if tt <= t + 1e-6]
            txt += ("   " + cur[-1]) if cur else ""
        ax.set_title(txt, fontsize=9, color=style.INK, loc="left")
        return []

    anim = animation.FuncAnimation(fig, update, frames=sorted(df["frameId"].unique()), interval=100)
    html = (anim.to_html5_video(embed_limit=40).replace("<video ", '<video style="max-width:100%;height:auto" ')
            if shutil.which("ffmpeg") else anim.to_jshtml(default_mode="once"))
    plt.close(fig)
    display(HTML(html))


def animation_or_strip(play, times, **kw):
    """Video on the website; a numbered, captioned frame strip in the PDF."""
    if pdf_build():
        strip(play, times, **kw)
        plt.show()
    else:
        kw.pop("ncols", None)
        video(play, times, **kw)

The replay rolls, and the analyst draws on the screen. “Look at this. They’ve got a 3-technique over the right guard, and they run right at him. The guard and tackle double-team him, one of them climbs to the Mike, the fullback kicks out the end, and the back hits the C gap untouched.” You nodded, because he sounded certain. But almost every noun in that sentence was a word nobody ever taught you: 3-technique, double team, climbs, kicks out, the end, the C gap. This chapter teaches them, and near the end you will read that sentence again and understand every word. It is the vocabulary the rest of the run game is written in, and it is smaller than it sounds: a handful of names for spaces, a numbering for where a defender stands, and eight verbs for what a blocker does. Each one exists because it solves a problem. Learn the problems and the words stick.

You’ll be able to:

  • Name every gap from A to D and read a numbered run call (“24”) on a diagram.
  • Look at a defensive line before the snap and describe it in technique numbers (“5, 1, 3, 9”), and say which offensive linemen are covered and which are uncovered.
  • Recognize the eight core blocks (drive, down, reach, cut-off, combo, pull, kick-out and lead) and say what each is for.
  • Count the box and do the “hat math”: who is left unblocked, and why the defense can almost always arrange to have one more man than the offense can block.
  • Explain why a quarterback who only hands off leaves the offense a defender short, and how a quarterback who can run gets that man back.
  • Say in two sentences how zone and gap blocking differ, and what each one trades away.

How to read the diagrams. Offense is blue circles, defense orange squares. Left and right always mean the offense’s left and right, and plays are drawn going up the page. LT, LG, C, RG and RT are the five offensive linemen (in the small block diagrams, just T, G and C); Q is the quarterback, R the running back, F a fullback, Y and U are tight ends on the right and left, and X, Z and H are wide receivers (H in the slot). Defensive linemen are labelled with their technique number, which this chapter teaches; M, W and S are the Mike, Will and Sam linebackers you met in The Twenty-Two; $ is the nickel back, CB a cornerback, and FS and SS the safeties, as in A First Look at Defense. Lines ending in a short bar are blocks; arrows are paths; the thick dark arrow is the ball carrier.

Why the run game needs its own words

A running play lasts about three seconds and involves twenty-two people colliding in a space about fifteen yards wide. The people inside it have to talk about it in two very different moments. Before the snap, a center has maybe ten seconds to tell four other linemen who is blocking whom against a defense that is still moving. (He usually starts by naming one linebacker as “the Mike,” the reference point every lineman counts from; that call, Mike identification, is taught in Pass Protection.) Later, in the film room, a line coach has to explain why a play that gained twelve yards last week lost two today. Both conversations need words that are short, unambiguous, and shared by everyone in the room.

So football built a vocabulary for exactly three things:

  1. The spaces in the line where a runner can go (gaps and holes).
  2. Where each defender stands relative to the blockers (technique numbers).
  3. What each blocker does (a short list of named blocks).

Nearly every sentence about the run game, from a high-school coach’s whistle to a network analyst’s telestrator, is built from those three sets of words. One more idea glues them together: counting. Run blocking is in the end an arithmetic problem, blockers against defenders, and the last third of this chapter is about that arithmetic.

Gaps: the spaces between blockers

Start with the line itself. When an offense breaks the huddle, its five linemen stand side by side, a little more than an arm’s length apart (the space between them is the lineman’s split). The spaces between them, and the space outside the last man, are where a running back can go. Each space has a letter.

A gap is the space between two adjacent players on the line of scrimmage, named by a letter counted outward from the center:

  • The A gaps are on either side of the center, between the center and each guard.
  • The B gaps are between each guard and the tackle beside him.
  • The C gaps are just outside each tackle (between the tackle and a tight end, if there is one).
  • The D gap is outside a tight end. It exists only on a side with a tight end on the line.

There is one A, one B and one C gap on each side, so an offense with no tight end has six gaps, and every tight end on the line adds one more. Some coaches keep going (an “E gap” outside a second tight end or a wing), but A to D covers almost everything you will hear on a broadcast.

Why letters, and why from the defense’s point of view? Gap letters are mostly the defense’s language. A run defense’s first job is to make sure that every gap has a defender responsible for it, so that wherever the ball carrier goes, somebody is already there. Coaches call that a defender’s gap responsibility: “you have the B gap.” Letters are fast to shout and the same on both sides of the ball, which is why a linebacker can say “A gap, A gap!” and every teammate knows he means the space next to the center. (How defenses divide up the gaps, and what happens when a defender leaves his, is Run Fits and Box Math.)

That point leads straight to the first piece of run-game arithmetic, which you first met in Formation Rules: every player the offense puts on the line creates a gap the defense has to fill. Add a tight end and there are seven gaps instead of six; add a second tight end and there are eight. A defense that wants a defender for every gap needs one more defender near the ball for every blocker the offense attaches to its line. That is a big part of why offenses use tight ends in the run game at all, and why the personnel an offense sends on the field changes how many defenders the defense has to keep close.

CautionCommon misconception

“The A gap is a spot on the field.” It is not. A gap is defined by the players on either side of it, so it moves when they move. If the center and right guard both step two yards to their right at the snap, the right A gap moves two yards to the right with them. That sounds pedantic, but it matters as soon as you watch a zone run in 03-02, where the whole line moves sideways together and the gaps slide along the line with it. Gaps also stretch and shrink: a lineman who takes a wider split before the snap makes the gaps on either side of him wider, and defenses watch splits for exactly that reason.

In hole numbering, every place a run can go gets a digit. In the most common version, the holes are counted outward from the center with even numbers on one side and odd numbers on the other (in Figure 35.7, even to the right). The backs get digits too: in one classic scheme, built for the four backs of the old T-formation, the quarterback is 1, the fullback 3, and the halfbacks 2 and 4 (2 on the left, 4 on the right; in an I-formation the tailback is the 2).1 Put the two together and a run call is just two digits: “24” means the 2-back carries the ball through the 4 hole. A coach can add a word for the blocking scheme (“24 Power,” “32 Trap”) and the whole play fits in one breath.

Why it exists. Before radio helmets and wristbands, a play had to be called in a huddle in a few seconds and remembered by eleven people. A numbering system lets one short code cover hundreds of plays: any back, any hole, any blocking scheme. Linemen learn rules for each scheme (“on 24 Power, the play-side tackle blocks down”), so the number tells them which rule applies and which way to apply it.

The catch. Numbering is a dialect, not a standard. Some teams number the gaps, as in the picture; others number the linemen, so that the “0 hole” is the center himself and the “2 hole” is wherever the right guard was. Some put the odd numbers on the right. And many modern offenses barely use hole numbers in their run calls at all, because their most common runs do not have one hole. On a zone run, as you will see in 03-02, the line steps together and the runner picks the hole after the snap; a call like “outside zone right” says where the play starts, not where it ends. That is why you will hear NFL coaches name runs by scheme and direction far more often than by number. (How whole playbooks name things is in Offensive Systems and Their Languages.)

NoteMadden vs. real life

Madden’s play art draws the running back’s path to one spot, and the game rewards you for hitting it. In a real NFL run the drawn path is only the plan: on zone runs, one of the two great families you will meet at the end of this chapter, the hole is decided after the snap by how the defense reacts. A hole number tells you where the play is designed to go. Whether it goes there is up to eleven defenders.

The play side (often written playside) is the half of the formation the run is aimed at; the backside is the other half, away from the play. The point of attack is the exact spot on the line where the play is designed to hit: in Figure 15.3, the right A gap. Broadcasters abbreviate it “the POA” or say “the point.”

Notice why the play aims there. The point of attack is chosen for who is standing near it and how he can be blocked. Here the right A gap has no defensive lineman in or beside it, so the nearest defender is a linebacker four or five yards away: coaches call that a bubble, and a lead play loves it, because the fullback can get to the linebacker before the linebacker gets to the hole. Often, though, the chosen spot has a defender standing right in it. Then the blocks are what empty it: the linemen move the man who owns that gap out of it, and the run goes into the space they made. On a designed run the hole is mostly made, not found.

Do not confuse the play side with the strong side. The strong side is a fact about the formation (the side with the tight end, or with more players), fixed before the snap. The play side is a fact about the play, and offenses run away from their strong side all the time, precisely because defenses load up toward it.

Why the distinction matters: blockers have completely different jobs on the two sides. On the play side their job is to make the hole: move defenders, wall them off, create space at the point of attack. On the backside their job is to stop the defenders on that side from chasing the play down from behind. Those backside defenders do not need to be knocked back; they only need to be kept out of the way for a second or two. As you are about to see, there are different blocks for each job, and the side a block is on tells you most of what it is trying to do.

Now turn to the defense. Before an offensive lineman can block anyone, he needs to know exactly where the defender in front of him is standing, because a defender on his left shoulder and one on his right shoulder are different problems. Saying “the tackle’s a bit outside the guard” is too slow and too vague. So football gave every spot along the line a number.

A technique (or alignment number) says where a defensive lineman lines up relative to the offensive linemen. You met two of its words, head-up and shaded, and the 3-technique in The Twenty-Two; here is the full system. It rests on two ideas:

  • A defender lined up directly across from a blocker, nose to nose, is head-up on him, and gets an even number: 0 head-up on the center, 2 on a guard, 4 on a tackle, 6 on a tight end.
  • A defender on one of a blocker’s shoulders is in a shade. On the outside shoulder he gets the odd number after that blocker’s even one: 1 on the center’s shoulder, 3 on the guard’s outside shoulder, 5 on the tackle’s. On an inside shoulder he gets the blocker’s even number plus an “i”: 2i on the guard’s inside shoulder, 4i on the tackle’s.

The numbers are mirrored: a 3-technique can be on the left guard or the right guard, and the caption or the context tells you which. The pattern breaks at the tight end: in the most common version, a defender on the tight end’s inside shoulder is a 7 (not a “6i”), and anything outside him is a 9. Systems disagree here more than anywhere else. Some swap the 6 and the 7, some use an 8 when two tight ends line up on the same side, and many use “wide 9” for an end lined up well outside the tight end.2

That table is the reason the system exists. A shaded defender is usually responsible for the gap he is shaded into. A 3-technique sits in the B gap and, in most defenses, owns it; a 1-technique owns the A gap he is sitting in. So the alignment number is not just a location: it is a strong hint about the defender’s job. Head-up defenders are different. A 0 or a 4 is often asked to control the blocker in front of him and then fill whichever gap the ball goes to, on either side. That style is called two-gapping, and it is taught in Defensive Line Play.

If a 3 and a 4i both sit in the B gap, why does anyone care which one it is? Because the number says whose shoulder he is on, and so which blocker has a man in his face. A 3 is on the guard; a 4i is on the tackle. Same gap, different blocker occupied, and as the next section shows, that decides who is free to help and who has to block him. That is also how to tell them apart in a game. The real distances are tiny: linemen usually leave a gap of about two feet between themselves and the next man,3 so a whole B gap is two or three feet wide and a 3 and a 4i stand about a foot apart. Don’t try to place a defender in the gap; ask which lineman’s shoulder he is lined up on. The angle that shows it best is the end-zone replay, shot from behind the offense straight down the line; the high sideline camera of a live broadcast flattens the line and hides it.

Where the numbers came from. The system is usually credited to Paul “Bear” Bryant, but the better-documented story gives it to Bum Phillips, who, by his hometown paper’s account, devised it while coaching at Nederland High School in Texas in the early 1950s, and who joined Bryant’s staff at Texas A&M in 1957. Bryant himself credited Phillips. Before it, every staff used its own code words for alignments, so a lineman who changed schools had to learn a new language.4 Phillips, the Stetson-wearing Texan who later coached the Houston Oilers, liked things simple, and the numbers stuck because they are simple: two rules, ten digits and an “i”, and a defensive coordinator can describe an entire line in a breath.

The numbers that became position names

A few technique numbers are used so often that they turned into names for positions:

  • The 3-technique is the defensive tackle on a guard’s outside shoulder. In a four-man line he is usually the best pass rusher on the inside, because his alignment puts him one-on-one with a guard in the B gap, with room to shoot past him. When analysts call a player “a 3-technique” they mean that job and that body type: quick, explosive, penetrating. Aaron Donald, whom you met in The Twenty-Two, was the archetype.
  • The 5-technique is on the tackle’s outside shoulder. In a 3-4 defense it became the name for the defensive end: a big, long player who can hold off the tackle and still close the C gap.
  • The 4i-technique sits on the tackle’s inside shoulder, in the B gap. It is a favorite alignment for the ends of modern three-man fronts. It squeezes the B gap from the inside, it is hard for the tackle to get outside of (he is already inside the tackle’s body), and because he can play off either the guard or the tackle, he muddles which of those two should block him and which should go to the linebacker.
  • The 0 and 1 are the nose. A 0 is head-up on the center; a 1, often called a shade nose, sits on one of the center’s shoulders. Both are usually the heaviest men on the field (the nose tackle of 01-03).

Once you know the numbers, a whole front becomes a string you can say out loud.

An offensive lineman is covered when a defensive lineman is lined up on him, head-up or on either shoulder. He is uncovered when nobody is, so that the nearest defender in front of him is a linebacker four or five yards away. In the top front of Figure 15.6, the 5 covers the left tackle, the 1 covers the center, the 3 covers the right guard and the 9 covers the tight end; the left guard and the right tackle are uncovered.

Why every blocking scheme starts here. A covered lineman has a man in his face at the snap and usually has to deal with him first. An uncovered lineman is the offense’s free resource. He can help a neighbor on a defensive lineman (a double team), or go straight to a linebacker at the second level, the linebackers’ depth. Which uncovered linemen exist, and where, shapes what any run play can do against that front.

Uncovered does not always mean free, though; you still have to count. Look at the bottom front. Both guards are uncovered, but they are not equally free. On the left, the 5 and the 4i bracket the left tackle, and one man cannot block two, so the left guard has to help with the 4i before he can think about a linebacker. On the right, the tackle has only the 4i, so the right guard can double-team the 4i with him and then climb: he is the natural climber. The center, covered by a 0, has the heaviest man on the field on his nose and little chance to help anyone. A defense that wants to stop a certain run chooses its front partly to change who is covered, and who has to stay home to help.

Covered and uncovered is a snapshot taken before the snap, and the defense knows it. Defensive linemen can slant at the snap, stepping sideways into the next gap instead of straight ahead (Defensive Line Play teaches it), and a lineman who was uncovered can suddenly have a man in his face. How a blocking scheme copes with that is one of the big differences between the two families at the end of this chapter.

One refinement, which you will meet in Zone Running: zone-blocking teams count “covered” only on the side the play is going. A defender shaded on a lineman’s backside shoulder is not his problem on a zone run, because he is stepping away from him. That rule is why the same lineman can be covered on a run to the right and uncovered on a run to the left.

Work through Figure 15.7 one panel at a time.

1. Drive. The drive block is the simplest block in football: the defender is lined up on you, and you block him straight ahead and try to move him backward, off the ball. NFL coaches and broadcasters often call it a base block. Every other block is a variation on it. Why it exists: moving a defensive lineman a yard or two backward collapses the space linebackers need to get to the hole. What it costs: it is a pure strength contest. Drive-blocking a 320-pound defensive tackle alone, head-on, is the hardest way to move him, and against a better player it simply fails.

2. Down. On a down block the blocker goes after a defender lined up inside him, toward the center, and hits him from the side. Why it exists: angles. A defender is built to take on force from the front; shove him from the side and he has much less to push back with, the way a parked car is easy to rock sideways and hard to push forward. A down block also seals: it pins the defender inside, building a wall on the inside edge of the hole. Gap-scheme runs are built out of down blocks. What it costs: the down-blocker leaves the man on his outside unblocked, so somebody else (a puller, a fullback) has to deal with that man.

3. Reach. A reach block is the opposite problem: the defender is on your play-side shoulder, a step closer to the play than you, and you have to get your head and body outside him so he cannot get to the ball, sealing him inside. Why it exists: runs to the outside need the defenders inside them to be kept inside. What it costs: it is the hardest block in football. The defender is already a step ahead and moving the way you want to go; he only has to take one step sideways to stay outside you. That is why the reach is usually done with help (the zone chapter shows the two-man version). And it has to last: the runner is coming toward this defender, so the seal must hold until the ball carrier is past him.

4. Cut-off. The cut-off block is a backside block. The blocker steps toward the play and gets his body between a defender and the ball carrier, usually a defender on his play-side shoulder, the one who could chase the play. Why it exists: backside defenders do not need to be moved, only kept from pursuing, and a body in their path for one second is enough. What it costs: little, which is the point; but if the defender beats it, he is the man who tackles a cutback runner from behind. When two or more backside linemen do it together, each stepping play-side and taking whoever shows up, coaches call it a scoop.5 (Do not confuse the cut-off with the cut block, a low block at the thighs, which is a different thing taught in 03-02.)

Three of these blocks are easy to mix up, because the footwork can look alike. The difference is the side of the play and the job:

  • A down block is on the play side. The man is inside you, and you hit him from the side and drive him inward, because he is the inside wall of the hole and the runner is coming right past him.
  • A reach block is also on the play side, but the man is on your play-side shoulder, so you must get around him and seal him in. The runner is coming toward him, so the seal must hold until the runner is past: that is why it is the hard one.
  • A cut-off block is on the backside. The play is running away from this defender, so you do not have to move him or seal him for long; you only have to get your body into his path toward the ball. On the backside, the direction of the play is toward the center, so the man you cut off is often inside you, exactly where a down block’s man would be (panel 4 shows one). Same spot, different job: the down-blocker drives his man; the cut-off blocker just gets in the way, and a moment’s delay is enough because the ball is going the other way.

5. Combo. A combo block, also called a double team, puts two adjacent linemen on one defensive lineman: usually one covered and one uncovered. They drive him together, and both keep their eyes on the linebacker behind him. Many playbooks name the pair: an ace is the center and a guard, usually on a 1, 2i or 2-technique, and a deuce is a guard and tackle on a 3-technique (the names vary by team).6 When the linebacker commits to a gap, one of the two comes off and climbs to the second level to block him; the other finishes the defensive lineman alone. Why it exists: it is the only way to get both things the run needs from one pair. Two men can move a defensive tackle off the ball when one man cannot, and the climb means the linebacker is still blocked. What it costs: timing. Come off too early and the defensive lineman splits the two; too late and the linebacker is already in the hole. The animation below shows it.

6. Pull. To pull is to leave your spot at the snap, step back off the line, and run sideways behind it to block somewhere else. It is mostly guards (and some centers and tackles) who pull. Why it exists: it adds a blocker where the play is going. A lineman lined up on the backside, where his block matters least, becomes an extra body at the point of attack, where it matters most. What it costs: the spot he left is now empty, so his neighbors must cover for it; pulling takes time, so a fast defender can beat the puller to the hole; and it is a tell. Linebackers are taught to read the guards, and a pulling guard tells them where the play is going. (Linebackers and the Second Level covers that read.)

7. Kick-out. A kick-out block hits the defense’s end man (short for the end man on the line of scrimmage, the last defender on the line on the play side) from the inside, and pushes him toward the sideline. The run then goes inside him. Why it exists: after the play-side linemen block down, the end man is the one defender left at the edge of the hole, and he is lined up outside everybody. Nobody on the line can reach him, so a blocker coming from the inside, a pulling guard or a fullback with a running start, takes him on with the angle in his favor. What it costs: an extra blocker, who has to come from somewhere (the backfield, or the backside of the line).

8. Lead. A lead block is thrown by a back, usually a fullback, who runs through the hole ahead of the ball carrier and blocks the first defender to show up, often a linebacker. Why it exists: the linemen can block the linemen, but somebody has to block the linebacker who is reading the play and running to the hole; the lead blocker arrives there at full speed, with the ball carrier right behind him. What it costs: a player who could otherwise run a route or carry the ball, which is one reason fullbacks became rare in the NFL. (The classic lead play, iso, is in 03-03.)

Notice a pattern in that list. Four blocks are about winning a matchup (drive, down, reach, cut-off): one blocker against one defender, with the block chosen to give the blocker the best angle. The other four are about changing the numbers (combo, pull, kick-out, lead): getting two blockers onto one defender, or getting a blocker to a spot where the defense did not expect one. That second group is where most run plays are won, and it is the subject of the rest of the chapter.

ImportantWhy it exists

Every block has a geometry. A defender is hard to move from the front and easy to move from the side, so offenses design runs so that as many blocks as possible are angle blocks (down blocks, kick-outs) rather than head-on drive blocks, and so that the hardest block, the reach, is done with help. When you watch a run and wonder why a lineman went sideways at the snap instead of straight ahead, the answer is almost always the angle.

Hat math: counting blockers and defenders

Now the arithmetic. Coaches call it hat math (sometimes “counting hats,” a hat being a helmet, so a player): count the blockers, count the defenders, and see who is left over. The phrase you will hear most is “a hat on a hat”: every defender who can get to the run has a blocker assigned to him. If the defense has one more man than the offense has blockers, one defender is unblocked, and on a run that defender is usually the one who makes the tackle.

The count happens in the box, the area near the ball where the run is defended: from just outside one end of the offensive line to just outside the other (including an attached tight end), and about five to seven yards deep. You learned to count it in A First Look at Defense, along with light and loaded boxes, and Figure 15.9 is the same picture you saw there. What is new is that the blocks are now drawn, so you can see exactly who blocks whom and who is left over.

Read the two pictures in Figure 15.9 as a negotiation.

  • Six against six (left). The offense has a blocker for every defender in the box. The run has a fair fight up front, and if it gets through the line, the next defenders are two safeties a dozen yards away. Offenses love to run against this look. It is common now, because defenses have leaned into two-deep safeties to stop the pass: NFL Next Gen Stats measured split-safety coverage on 42.0% of dropbacks in 2025, up from 32.9% in 2018.7
  • Seven against six (right). The defense has rolled a safety down. Now there is one more defender than blockers, and the unblocked man is in the box, not twelve yards away. Some defender will be unblocked whatever the blocking scheme does; the scheme only gets to choose which one. A good run play leaves the defender who is farthest from the play, or the one the ball carrier can most easily make miss.

Why doesn’t the defense just put everyone in the box? Because every man it adds there is a man taken out of pass coverage. In the right-hand picture there is now only one deep safety, and each receiver outside has a cornerback on him with less help over the top. That is the trade the box count measures. It is why quarterbacks count the box before the snap and switch from a run to a pass when the numbers are bad (the kill call in The Pre-Snap Phase), and why defenses try to disguise how many men will really be in the box at the snap. Defensive coaches have a name for having one more than the offense can block: being plus-one in the run game, which is the subject of 05-05.

The offense can add blockers, of course. Bring in a second tight end or a fullback and there are seven blockers. But every blocker you add is a receiver you took off the field, so the defense needs fewer men in coverage and can bring another one into the box. The defense can match each added blocker with an added defender, one for one, and keep its extra man. The next section explains why that extra man is almost always there.

Why a quarterback who only hands off leaves you a man short

Count everyone on the field, not just the box. Eleven offensive players, eleven defensive players. On a normal handoff:

  • The ball carrier does not block anyone; he is the man the defense is trying to tackle.
  • The quarterback hands the ball off and then, usually, does nothing that matters. Nobody has to cover him or tackle him. For the rest of the play he is a spectator.
  • The other nine players (five linemen, and four tight ends, backs and receivers) each block a defender or take one away by running a route that a defender has to cover. Even on a run, a cornerback has to cover his receiver at the snap, because he does not yet know it is a run; by the time he finds out, he is ten yards from the ball.

So nine offensive players occupy nine defenders, and two defenders are left free with nobody assigned to them. The defense gets to choose where those two free men stand. Put both deep (the two-high picture) and the run gets a fair fight in the box but runs into two safeties if it breaks through. Put one in the box (the one-high picture) and that man can tackle the runner near the line. Either way, the offense is playing ten men against eleven. The quarterback is the missing man.

Now give the quarterback a job. If he is a real threat to run, a defender has to account for him, and that defender cannot also tackle the running back. The cleanest version is the read: the offense deliberately leaves one defender unblocked and the quarterback watches him after the snap. If that defender plays the running back, the quarterback keeps the ball; if he plays the quarterback, the quarterback hands off. Either way the read has done a blocker’s work: the defender is “blocked” by the decision. The offense is back to eleven against eleven, and one of the defense’s free men has disappeared. That is the core idea of the zone read and every other option play, which Option Football builds out properly. (Read option is the umbrella name you will hear for this family; the zone read is its most common play.) Jim Harbaugh, whose 49ers ran the read option with Colin Kaepernick, put it plainly in the week before Super Bowl XLVII, crediting the thought to Lou Holtz: without an option attack “the offense is basically playing with 10,” but when the quarterback is a threat to run, “that gets the numbers back to even, back to 11-on-11.”8

That arithmetic is why running quarterbacks change a run game: the 2019 Ravens, an option-heavy offense built around Lamar Jackson, set the NFL record for team rushing yards (3,296), and he did not have to keep the ball every time to do it.9

There is a quieter version of the same idea that every offense uses: the pass threat. Each receiver who might run a route takes a defender with him, which is why defenses facing three or four receivers keep fewer men in the box. A quarterback who can make the defense pay for loading the box with an accurate throw is, in run-game arithmetic, worth a blocker too. And when the quarterback runs the ball himself on a designed run (a sneak, or the QB power of 03-04, a planned quarterback run behind a pulling lineman and a lead blocker), the ball carrier is the quarterback, so the running back becomes a blocker: the offense gains a hat. That is the arithmetic behind the tush push, too (Red Zone, Goal Line, and Short Yardage).

CautionCommon misconception

“Running quarterbacks are dangerous because they are fast.” Speed helps, but the bigger effect is arithmetic. A quarterback the defense has to account for on every run takes a defender away from the running back, even on the plays where he hands off. That is why an offense can get a better run game from a quarterback who keeps the ball only a few times a game: the threat does the work.

Zone blocking assigns blockers to areas. All the linemen step the same direction at the snap, each takes whoever enters his area, uncovered men double-team and climb, and the running back reads where the wall cracks. Its virtue is that it adapts: if the defense slants (the line movement from the covered-and-uncovered section) or blitzes, whoever arrives in a blocker’s area gets blocked. Its price is that blocks are rarely knock-down blocks, and the hole is not known in advance, so the play depends on a runner who can see it. (Zone Running is the full story.)

Gap blocking (also called man or power blocking) assigns blockers to men, chosen to give them angles, and moves extra blockers to a hole chosen in advance. Down blocks seal the inside, a kick-out walls off the outside, and a puller or a fullback leads through the space between them. Its virtue is numbers at the point of attack: it gets more blockers to the hole than the defense has defenders there. Its price is that it is easier to read (a pulling guard is a signpost) and more fragile against movement: each blocker was told which man to block, so if that man slants into the space a puller was supposed to run through, the play can die. (Gap Schemes covers power, counter, trap and their relatives.)

Neither is better. Almost every NFL team runs both, and good offenses build them to look alike for the first half-second so the defense cannot tell which is coming. The first thing to watch on any run is which family it belongs to, and you can usually tell from one question: does the whole line step the same way, or does somebody pull? The big exception is duo, a gap-family run with no puller that is built to look like inside zone (Gap Schemes); for that one, look for double teams driving straight at the linebackers rather than a line moving sideways together.

Where this language came from: the Hogs

Most of this vocabulary is older than anyone now coaching, but one team made it famous to a national audience, and its story explains why the blocks in this chapter are still the foundation.

When Joe Gibbs became Washington’s head coach in 1981, his offense was built to run behind a big offensive line. Line coach Joe Bugel started calling his linemen “the Hogs” in training camp in 1982 (“Okay, you hogs, let’s get running down there”), and the name stuck to a group of big, unglamorous blockers (center Jeff Bostic, guards Russ Grimm and Mark May, tackles Joe Jacoby and George Starke, among others) who became the most famous offensive line in football.10 Washington won three Super Bowls under Gibbs, after the 1982, 1987 and 1991 seasons, with three different starting quarterbacks, which tells you how much of the offense lived up front.

Gibbs’s teams are a good place to see the language in use because their signature runs were the gap-scheme plays this chapter just described: down blocks on the play side, guards and even tackles pulling around to kick out and lead, a fixed hole. Their counter play, with two backside linemen pulling (the left guard Russ Grimm and, unusually, the left tackle Joe Jacoby), became one of the most copied runs in football (it is taught in 03-03).11 And Gibbs’s answer to one defender, the Giants’ Lawrence Taylor, gave the run game a new blocker: Gibbs moved a back up near the line as an extra blocker for the edge, the position now called the H-back.12 That is hat math again. Taylor was a defender the existing blocking could not handle, so Gibbs added a hat.

The motion is what put him there. A cornerback who follows a man in motion across the formation is behaving like a man-coverage defender (the travel tell of Motion and Shifts), so when Didier came back to the left, McNeal came with him and ended up near the line on the play side, the one unblocked man between Riggins and the end zone. The modern echo of the Hogs is not one team but one position. As offenses spread out and defenses got faster, the center became the hub of the run game: the man who makes the line’s calls, who is usually uncovered and must climb, and who in some offenses pulls. Two recent lines show the vocabulary at its best.

TipWatch for it

On Sunday: find the hole. For one quarter, on every run, do this.

Before the snap

  • Count the box and the blockers (five linemen, plus each tight end and fullback near the line). Six against six? Seven against six? Who would be the free man?
  • Read the defensive line in numbers, left to right (“5, 1, 3, 9”). Which guard has a 3-technique on him? Which linemen are uncovered?

The first half-second

  • Watch the running back’s first step. It usually points at the play side and at the point of attack. Say the gap out loud before he gets there: “right B gap.” The exception is a counter play, where the back’s first step is a deliberate lie, away from the play (03-03).
  • Glance at the guards. Does one pull? (Gap play, and the puller is running to the hole.) Does the whole line step together? (Zone, and the hole is not decided yet.) Double teams driving straight ahead with nobody pulling? Think duo.

After the play (on the replay)

  • Name the gap he actually hit. Same as your guess? On a gap play it usually is; on a zone play it often is not, and that is the play working, not failing.
  • Find one block of each kind you can see: a down block, a double team with a climb, a kick-out, a lead.
  • Find the unblocked defender. Where was he, and did he make the tackle?

For your prediction log: before each run write the gap you expect (say, “R-B”); afterwards write the gap he hit and Z (zone) or G (gap). After a quarter you will know whether this offense’s first step tells the truth.

Back to the replay

Here is the analyst’s sentence from the start of the chapter: “They’ve got a 3-technique over the right guard, and they run right at him. The guard and tackle double-team him, one of them climbs to the Mike, the fullback kicks out the end, and the back hits the C gap untouched.” Every word now means something:

  • A 3-technique over the right guard: a defensive tackle on the guard’s outside shoulder, in the B gap. The guard is covered; the tackle beside him is not, so he is free to help.
  • Run right at him: the play side is right and the point of attack is next to the 3. He is standing near the hole, so the blocks will have to move him out of it.
  • Double-team him, one of them climbs to the Mike: a combo block (a deuce, guard and tackle). They move the 3 together until the Mike commits, then one comes off to block him.
  • The fullback kicks out the end: a kick-out on the end man, from the inside, pushing him toward the sideline.
  • The C gap: just outside the right tackle, between the double team and the kick-out.
  • Untouched: the hat math worked. Every defender who could reach the hole had a blocker, and the free men were somewhere the play was not.

Takeaways

  • Gaps and holes name the spaces. Gaps are lettered A, B, C, D outward from the center (the defense’s language; every gap needs a defender, and every blocker the offense attaches to its line adds one). Holes are the offense’s numbers, usually odd on one side and even on the other, and “24” means the 2-back through the 4 hole, though numbering is a dialect.
  • Technique numbers say where a defender stands: even numbers head-up (0 center, 2 guard, 4 tackle, 6 tight end), odd numbers on an outside shoulder (1, 3, 5, and 9 outside the tight end), “i” on an inside shoulder (2i, 4i), and 7 inside the tight end. A shaded defender usually owns the gap he is shaded into; the 3-technique, 4i and 5-technique have become names for positions.
  • Covered or uncovered is the first question every lineman asks. Covered linemen have a man; uncovered linemen are the free hats, the ones who double-team and climb, as long as the count says nobody needs their help.
  • Eight blocks describe almost every run. Four win a matchup with the best angle available (drive, down, reach, cut-off); four change the numbers (combo, pull, kick-out, lead). Play-side blocks make the hole; backside blocks stop pursuit.
  • Hat math decides runs before the snap. A handoff leaves the quarterback a spectator, so the offense plays ten against eleven and the defense has two free men to place, deep or in the box. A quarterback who can run, or a read that “blocks” a defender without touching him, gets one back.
  • Zone and gap are the two answers. Zone blocks areas and lets the runner find the hole; gap blocks men with angles and brings extra blockers to a hole chosen in advance. Watch the first step: together, or does someone pull? (Duo is the gap run that hides it.)

Hat math in the data, and why it is slippery. A First Look at Defense showed the cost of a crowded box: on first-and-10 in 2023–2025, runs into five or fewer defenders succeeded about 40% of the time and runs into eight or more about 31%. Hat math says the number that really matters is not the box count but the box count minus the blockers. Count it that way and, surprisingly, most of the cost disappears from the averages. On the same first-and-10 designed runs, success was 35% when the defense had fewer men in the box than the offense had blockers, 36% when the two were even, 33% when the defense was plus-one, and 35% when it was plus-two or more.19

That is not hat math failing on the field. It is both sides choosing. Defenses go plus-one when the formation and the offense’s tendencies say run and they expect to win the fit; offenses that see a bad count often check out of the run (the kill call) or call a run built to handle the free man, such as a read; and the personnel count treats every tight end as a blocker, even one split out wide. A simple average cannot separate the arithmetic from the decisions made around it. Run Fits and Box Math takes up the defense’s side of this with data, and Expected Points, EPA and Success Rate and Tendencies give you the tools to untangle it.
