def check_legal(players):
    """Seven on the line, and only the two end men of the line may be receivers."""
    off = [p for p in players if p.side == "off"]
    on = sorted([p for p in off if p.d > -1.0], key=lambda p: p.w)
    assert len(off) == 11 and len(on) == 7, f"{len(off)} players, {len(on)} on the line"
    for p in on[1:-1]:
        assert p.pos in ("LT", "LG", "C", "RG", "RT"), f"{p.key} is covered up but is a receiver"
    return players


def front(pe=("9", 5.7), pt=("3", 2.35), bt=("1", -0.7), be=("5", -4.0), lbs=(("WILL", 4.5, -2.4, "W"),

def go_to(p, key, end, kind="block", delay=0.0, speed=None, via=()):
    """Move `key` through absolute (d, w) points `via` to `end`, as a block, pull, path or run."""
    start = np.array(p._cur(key))
    pts = [tuple(np.array(q) - start) for q in (*via, end)]
    if kind == "block":
        return p.block(key, pts=pts, delay=delay, speed=speed)
    if kind == "pull":
        return p.pull(key, pts, delay=delay, speed=speed)
    if kind == "route":
        return p.route(key, pts, delay=delay, speed=speed, frame="field")
    return {"path": p.path, "run": p.run}[kind](key, pts, delay=delay, speed=speed)


def motion_to(p, key, via, end, speed=6.0):
    """Pre-snap motion through absolute points, ending where the player is when the ball is snapped."""
    start = np.array(p._cur(key))
    return p.motion(key, pts=[tuple(np.array(q) - start) for q in (*via, end)], speed=speed)

def clean(ax):
    """Box-zoomed diagrams: drop hash ticks and yard numbers, clip text to the window."""
    for ln in list(ax.lines):
        xs, ys = ln.get_data()
        if len(xs) == 2 and ln.get_linewidth() == 0.5 and np.hypot(xs[1] - xs[0], ys[1] - ys[0]) < 1.0:
            ln.remove()
    for t in list(ax.texts):
        if t.get_text().isdigit() and t.get_alpha() == 0.55:
            t.remove()
        else:
            t.set_clip_on(True)

def tracking(play):
    df = play.to_tracking()
    df["_id"] = df["displayName"].astype(str)
    return df


def panel(ax, play, df, t, lateral, window, rings=(), ghosts=None, decorate=None):
    """One moment of a simulated play: markers, faint trails, rings, and optional hollow 'ghosts'
    (players from a second version of the play, at the same moment)."""
    tmap = df.groupby("frameId")["time"].first()
    fid = int((tmap - t).abs().idxmin())
    ylo, yhi = sorted((play.ball_y - lateral[0], play.ball_y - lateral[1]))
    fld = Field(ax=ax, orient="vertical", los=play.los, window=window, numbers=False, hashes=False,
                lateral=(ylo, yhi), to_go=play.to_go)
    cur = df[df["frameId"] == fid]
    shown = cur[cur["x"].between(play.los + window[0] + 1.0, play.los + window[1] - 1.0)
                & cur["y"].between(ylo + 1.0, yhi - 1.0)]
    trails = df[(df["frameId"] <= fid) & df["_id"].isin(set(shown["_id"])) & (df["side"] != "ball")]
    for (_, side), g in trails.groupby(["_id", "side"]):
        xs, ys = fld.xy(g["x"].values, g["y"].values)
        ax.plot(xs, ys, color=style.OFFENSE if side == "off" else style.DEFENSE, lw=0.9, alpha=0.4, zorder=3)
    ppy = fld.pts_per_yard()
    if ghosts is not None and t > 0.05:
        gplay, gkeys = ghosts
        for k in gkeys:
            x, y = (float(v) for v in fld.xy(*play.to_abs(*gplay.position(k, t))))
            ax.add_patch(RegularPolygon((x, y), 4, radius=MARK_R * 1.3, orientation=np.pi / 4, fc="none",
                                        ec=style.DEFENSE, lw=1.1, ls=(0, (2, 1.2)), zorder=6))
            glab = str(gplay.players[k].label)
            ax.text(x, y, glab, ha="center", va="center", color=style.DEFENSE, fontweight="bold", zorder=6,
                    fontsize=max(4.5, min(8.0, ppy * (0.95 if len(glab) <= 1 else 0.75))))
    if decorate is not None:
        decorate(ax, fld, t)
    for _, r in shown.iterrows():
        x, y = (float(v) for v in fld.xy(r["x"], r["y"]))
        if r["side"] == "ball":
            ax.add_patch(Ellipse((x, y + 0.45), 0.7, 0.42, fc=style.BALL, ec="white", lw=0.5, zorder=11))
            continue
        lab = str(r.get("label", "") or "")
        if r["side"] == "off":
            ax.add_patch(Circle((x, y), MARK_R, fc=style.OFFENSE, ec="white", lw=0.6, zorder=8))
            tc = "white"
        else:
            ax.add_patch(RegularPolygon((x, y), 4, radius=MARK_R * 1.3, orientation=np.pi / 4,
                                        fc=style.DEFENSE_TINT, ec=style.DEFENSE, lw=1.0, zorder=8))
            tc = style.INK
        fs = max(4.5, min(8.5, ppy * (0.95 if len(lab) <= 1 else 0.75)))
        ax.text(x, y, lab, ha="center", va="center", fontsize=fs, color=tc, fontweight="bold", zorder=10)
    for key, col in rings:
        rr = cur[cur["displayName"] == key].iloc[0]
        x, y = fld.xy(rr["x"], rr["y"])
        ax.add_patch(Circle((float(x), float(y)), 1.3, fc="none", ec=col, lw=1.7, zorder=7))
    return fld


def strip(play, times, notes, lateral, window, ncols=2, rings=(), ghosts=None, width_in=6.5, decorate=None):
    """The print version of an animation: numbered panels with a caption under each."""
    df = tracking(play)
    nrows = int(np.ceil(len(times) / ncols))
    pw = width_in / ncols
    span_w, span_d = lateral[1] - lateral[0], window[1] - window[0]
    fig, axes = plt.subplots(nrows, ncols, squeeze=False, figsize=(width_in, nrows * (pw * span_d / span_w + 0.6)))
    for ax in axes.flat[len(times):]:
        ax.axis("off")
    for i, (ax, t) in enumerate(zip(axes.flat, times)):
        panel(ax, play, df, t, lateral, window, rings, ghosts, decorate)
        ax.set_title(f"{i + 1}.  " + ("snap" if abs(t) < 1e-6 else f"{t:+.1f} s"), fontsize=8.5,
                     color=style.INK_2, loc="left")
        ax.text(0.5, -0.03, notes[i], transform=ax.transAxes, ha="center", va="top", fontsize=7.5,
                color=style.INK, linespacing=1.15)
    fig.tight_layout(h_pad=2.4, w_pad=0.8)
    return fig

            if shutil.which("ffmpeg") else anim.to_jshtml(default_mode="once"))

Third quarter, second-and-8. For the fourth time today a receiver sprints across the formation in jet motion just as the ball is snapped. The first time he got the ball and ran around the left end for eleven yards. The next two times the quarterback stuck the ball in his belly and pulled it out, and both linebackers took three hard steps toward him anyway, while the running back went the other way. This time both linebackers sprint left again, the safety on that side races up to meet the receiver at the corner, and the quarterback hands the ball to the running back going right, into a hole where a linebacker stood a moment ago. Nobody on the left side of the field touched the ball, and every one of them helped make the play. This chapter is about why the defense keeps chasing a receiver who mostly doesn’t get the ball, and what happens to defenses that stop.

You’ll be able to:

  • Recognize the jet sweep, the end-around, the reverse, the crack-toss and the buck sweep, and say what each one punishes.
  • Spot the two perimeter blocks (the crack and the seal) and say why a receiver blocking back toward the ball is one of the most effective blocks in football.
  • Explain the Wing-T’s series logic (buck sweep, trap, waggle, counter, and the belly series) and point to its NFL descendants on a Sunday broadcast.
  • Watch the linebackers on jet motion and say whether the motion did its job, even when the ball went somewhere else.
  • Explain why a play that gains a lot when it’s run rarely can’t simply be run more often.

  • Gap Schemes: a pull is a lineman leaving his spot at the snap to block somewhere else; a puller either kicks out the end man (hits him from the inside and pushes him toward the sideline) or logs him (gets outside him and pins him in). The toss is a pitch to a back already running wide, and pin-and-pull is the modern outside run in which the linemen nearest the play block down (pin) and the ones farther away pull around them. Trap lets a defender in and hits him from the side with a pulling lineman; counter starts the backfield one way and pulls blockers back the other.
  • Motion and Shifts: jet motion is a receiver sprinting flat across the formation with the snap timed just before he reaches the quarterback, and fly motion is the Wing-T coaches’ name for the same thing (the fly sweep was born outside the Wing-T and adopted by it; more below). Against motion the defense can travel (one defender follows), bump (everyone slides a spot) or spin (a safety comes down toward the motion and the other goes to the middle).
  • From earlier chapters: play side and backside (the half of the formation the run is aimed at, and the other half), the technique numbers used to label defensive linemen (a 1 or 3 is on a guard’s shoulder, a 5 outside a tackle, a 7 on the tight end’s inside shoulder, a 9 outside him), the Wing-T formation (a wingback just outside and behind the tight end), and the condensed formation (receivers lined up close to the line, where they can block back inside).

The edge is a race

Everything you have seen in the run game so far has been a fight in a phone booth. On inside zone or power, blockers and defenders start a yard apart, and the run hits before the defense can move very far. Runs to the perimeter, the edge of the formation and the open grass outside it, are a different contest. The ball carrier has 10 or 15 yards of sideways running to do before he can turn upfield, and every defender on the field has time to see where he is going and run there too. An outside run is a race between the ball and the defense’s pursuit, the whole team running to the ball.

The defense organizes that race with two jobs. One defender on each side is the force player (more on him in Run Fits): his job is to get outside the ball and turn it back inside, toward everyone else. Everyone else pursues, each on an angle aimed at where the ball carrier will be, not where he is. If the force player holds and the pursuit is fast, an outside run dies for no gain or a loss, and losing yards behind the line is the outside run’s signature failure.

Who is the force player? The coverage decides, so it changes from snap to snap, but a rule of thumb gets you most of the way: he is usually the outermost defender standing near the line on that side. That is often an outside linebacker or a nickel back lined up outside the tight end or slot. If a safety has walked down to about seven or eight yards deep outside the formation, it’s probably him. If the cornerback is squatting at five yards with a safety deep behind him (the look of Cover 2), the corner is the force. You’ll hear a close cousin, contain. Both mean “nothing gets outside me.” Coaches say force for the defender who meets a run coming at him and turns it in, and contain for the man whose rule is to keep the ball, or a scrambling quarterback, from ever getting outside him. That matters most on the side the play is leaving, as you’ll see.

So the defense wants to pursue fast. And that is the opening for everything else in this chapter. A defense that flows hard to stop the sweep leaves the other side of the field empty and its linebackers a step out of place. The offense can punish that in two ways: run the ball back where the defense came from, or simply make the defense think the ball is going somewhere it isn’t.

Perimeter runs punish a defense that is slow to the edge. Misdirection, any play design that gets the defense to commit in one direction while the ball goes another, punishes a defense that is fast to it. The two families need each other: misdirection only works if the defense has a reason to run hard, and the only reason it has is that the perimeter run will beat it if it doesn’t.

The perimeter blocks: crack and seal

Outside runs need two kinds of block that you haven’t met yet, because inside runs rarely need them.

A crack block is a block thrown by a player lined up outside the play, usually a wide receiver or a wingback, who comes back toward the ball and hits a defender from the side. The usual target is a linebacker or safety who is watching the ball carrier and running to the edge, which is why the block is so effective: the defender’s eyes are on the ball, his momentum is going outside, and the blocker arrives from a direction he isn’t looking. You saw one animated in 02-04, where a condensed split put the receiver close enough to crack in a step or two.

A seal block is any block whose job is to wall a defender off from the ball’s path, rather than to move him. On an outside run that usually means pinning an inside defender inside: the tight end steps down onto an end lined up on his inside shoulder, gets his body between that end and the sideline, and the back runs outside the wall. A seal doesn’t need to drive the defender back a yard; it only needs to stay in the way while the ball goes by. Coaches use the word loosely (a log block by a puller who gets outside a defender and pins him in is a kind of seal), but the idea is always the same: you don’t move him, you shut him off.

The two blocks together make the classic outside run of the modern NFL, the crack-toss. 02-04’s animation showed the crack and the puller. Here are the parts it skipped: the seal, exactly who replaces whom, and (after the figure) the rule that governs the crack and the defense’s answer to it.

Read Figure 19.2 as a list of problems and answers. The Sam linebacker lined up outside the tight end is the defense’s force player: if he gets outside the ball, the toss is dead. Nobody on the line can reach him in time, but Z, lined up just outside him, can hit him from the side before he gets going. The end is lined up on the tight end’s inside shoulder (a 7-technique), so Y only has to keep him there: one step down, body between the end and the sideline, and hold. (If the end were lined up outside Y, in a 9, keeping him inside would take a reach, the hardest block a tight end is asked to make, and offenses usually build a different answer for that look.) The cornerback is now unblocked, so a lineman pulls around both blocks to get him. That’s crack-and-replace: the receiver leaves his usual man to crack inside, and a puller replaces him.

Who pulls varies, and this is why the 02-04 animation pulled the guard. Here the right tackle blocks down on the 3-technique and the right guard pulls behind him, the pin-and-pull logic of 03-03: the tackle has the better angle on the 3, the guard is free to go. Some teams pull the tackle instead, which is a shorter trip, as long as the guard can handle the 3 on his own; some pull both, and the second puller goes after the safety who is left over here. Every block on the play is an angle the defender wasn’t expecting.

The cost is that crack-toss is fragile in two specific ways. If the cracked defender sees it coming, he can duck under the crack and be in the backfield, and the puller has to choose between him and the corner. And the cornerback has an answer of his own: when he sees the receiver go inside, he yells “crack!” and comes up hard to take the force job himself. The puller still has him to block, but he now meets him sooner and tighter than planned, can only kick him out, and the back is forced to cut up inside, back toward the pursuit, instead of getting around the edge.

CautionCommon misconception

“A crack block is a cheap shot.” A legal crack is a hands-first block, from the side and chest-high, aimed at the defender’s near number or armpit on the side he’s facing. Receivers practise it the way linemen practise down blocks. What’s illegal is hitting him low, hitting him in the head or neck, or leading with the shoulder, and if all the blocker can see is the back of the defender’s jersey, it isn’t a crack at all but a block in the back (10 yards), so a well-coached receiver lets that man go and finds another.1

The rules are strict because a defender hit from the side by a player running at him can’t protect himself. The crackback rule (Rule 12, Section 2, Article 6) covers a player who lines up more than two yards outside the tackle, or starts in the backfield and is in motion or moves out that wide, and then blocks back toward the spot of the snap: within five yards of the line of scrimmage he may not hit a defender below the waist, nor in the head or neck (or in any other way banned against a defenseless player). It costs 15 yards.2 Since 2019 a separate rule has also banned any blindside block, a forcible hit with the helmet, forearm or shoulder thrown while the blocker’s path is toward or parallel to his own end line, outside the tight area around the snap,3 and from the 2026 season blindside blocks are one of the five fouls in the league’s three-strike suspension policy (three in a season means a one-game suspension).4 A crack runs parallel to the line, which is why the shoulder shot is gone and the hands-first block is the only kind you’ll see.

The jet sweep: a running start at the edge

The jet sweep is the purest perimeter play: a receiver in jet motion takes the ball from the quarterback without slowing down and races for the edge on the side he’s heading toward. You met the motion and the defense’s three answers to it (travel, bump, spin) in 02-06. This section is about the run itself, and about what it does to the linebackers.

Three things make it work:

  • The running start. The receiver is already at full speed when the ball is snapped; the defenders start from a standstill. The simple model in 02-06 put the gap at about three yards after one second. On a play that is a race to the corner, three yards is the race.
  • The angle. He takes the ball roughly in front of the quarterback, and within a stride or two he is past the far guard, already outside the box and heading for the edge. The defense’s force player has to cover more ground than he does.
  • What it does to everyone else. Linebackers are taught to read keys: usually a guard, the running back, or the backfield as a whole, depending on the defense (05-04 covers which). But no key-reading rule lets a linebacker ignore a man crossing the backfield at full speed who might have the ball, so his first steps tend to go with him. Whether or not the receiver gets the ball, the defense’s whole second level is now moving sideways, which is exactly what the rest of this chapter takes advantage of.

The mechanics vary. In some offenses the quarterback hands the ball off in front of himself; in others (Kansas City’s, for one) he flips it forward a foot or two to a receiver passing in front of him, a “pop” or shovel toss that counts as a forward pass in the record books. The flip has a hidden advantage: if the exchange is botched, the ball on the ground is an incomplete pass, not a fumble.5 (It also means those plays won’t appear in the run data later in this chapter.) Either way, the runner is gone before the defense has finished its first step.

Some offenses go one step further and make the jet an option. The quarterback reads a defender on the jet side, usually the end or whoever is responsible for the edge, exactly as 03-04’s read option reads an end: if that defender widens to take the jet, the quarterback hands the ball inside; if he sits inside, the jet man gets it. The motion man’s job is the same either way, which is why the defense can’t tell from his speed or his path.

What it gives up. The jet sweep is a high-variance play. When the force player holds the edge, the runner is caught going sideways behind the line, and the loss can be large because he started deep. The exchange happens at full speed between two players moving in different directions, which adds fumble risk (the forward flip is one answer). The risk is real but small: over the ten seasons 2016–2025, receivers lost a fumble on about 0.65% of their carries, against 0.46% for running backs, a difference of roughly one extra lost fumble every 500 carries (course calculation from nflverse play-by-play; the receivers’ figure rests on only 26 lost fumbles). And it uses a receiver as a runner, so every jet sweep is a play on which that receiver isn’t running a route. Coaches accept all of that because of what the motion does to the defense when the ball goes somewhere else, which is the subject of the rest of the chapter.

End-arounds and reverses: sending the ball back

If the defense is going to run hard one way, the obvious punishment is to give the ball to someone running the other way. There are two classic ways to do it, and broadcasters mix up their names all the time.

An end-around gives the ball to a receiver who comes back behind the line from his spot outside and runs toward the other side of the field. The quarterback hands it to him directly: one exchange. Usually the play starts with a fake the opposite way, so the defense flows toward the fake while the receiver runs away from it. (The name is older than the modern receiver: the “end” in end-around is the end of the line, the position that became the wide receiver and the tight end.)

A reverse has two exchanges. The ball first goes to a player running one way, usually a running back on a sweep, and he hands or pitches it to a teammate running the other way. The ball’s direction reverses behind the line. A double reverse does it again.

Both plays bet on the same thing: that the defense will chase the first action hard enough to leave the other side empty. Both are beaten by the same defender, the end on the backside (the side away from the first action). He is the contain player on that side, and his rule in most defenses is not to chase flat down the line after the fake. He shuffles along behind the line at about the depth of the ball, trailing the play, and checks the three things that can come back his way before he commits: a bootleg, a counter, a reverse.6 If he stays home like that, the reverse runs straight into him six yards behind the line of scrimmage. So the offense has to account for him. In Figure 19.4 the left tackle’s job is to get outside him and wall him inside, and the quarterback, once the ball is gone, leads up the field for the first pursuer who turns back. It is one of the few plays on which a quarterback is asked to block at all, and most coaches ask only that he get in the way of somebody.

They are also rare. The NFL’s play-by-play descriptions mention a reverse on only about 60 runs in the ten regular seasons 2016–2025, and on just one each in 2023 and 2024. Scorers don’t always note a reverse, so treat those counts as a floor, but the trend is real enough. Those flagged runs averaged about eight and a half yards (course calculation), which is the pattern this chapter ends on: a play that pays well because it’s called only when the defense has been caught chasing.

When you see one, ask two questions. Which defender was responsible for the backside, and did he chase? And how deep was the exchange? A reverse handed off seven yards behind the line has to gain seven yards just to get back to zero. It’s a play for a defense that has been caught chasing on film, not a play to run every week.

The Wing-T: an order of football

Most of the ideas in this chapter were organized into a system long before the NFL adopted them. The Wing-T, the formation with a wingback just outside and behind the tight end, was built by Dave Nelson at Maine and then the University of Delaware, where he coached from 1951 to 1965. His assistant Harold “Tubby” Raymond succeeded him and ran it for 36 more seasons, until 2001, winning 300 games, a total few coaches have reached at one school.7 Raymond, a Michigan graduate famous for painting a portrait of each of his senior players, and his assistant Ted Kempski wrote the book on it, and the title says what it is about: The Delaware Wing-T: An Order of Football.8

The problem Nelson is usually said to have faced is the one every underdog coach faces: his players were usually not bigger or faster than the other team’s. He couldn’t win by blocking every defender one-on-one, so he built an offense that made defenders block themselves, by running to the wrong place.

The buck sweep

The Wing-T’s signature play is the buck sweep. Its name tells you its two halves: the fullback fakes the buck (an old word for a straight plunge up the middle) and the halfback runs the sweep.

The buck sweep is power football, the same family as the gap schemes of 03-03: down blocks to wall off the inside, pullers to add blockers at the point of attack. It is also a close relative of the most famous run in NFL history, Vince Lombardi’s Packers sweep of the 1960s, which also pulled both guards around a down-blocking line, and an ancestor of today’s pin-and-pull.

What makes it a Wing-T play rather than just a sweep is everything that happens in the backfield before the ball goes outside. The quarterback reverse-pivots: instead of turning toward the sweep, he spins the other way, turning his back to the line of scrimmage, so for a moment nobody on the defense can see the ball. In that first second the defense sees the fullback plunge at the middle, the quarterback’s back, the halfback run across, and both guards pull. Then, a beat later, the quarterback rolls the other way without the ball. A linebacker who reads the fullback steps up; one who reads the guards flies outside; one who reads the quarterback drifts backside. Only one of them is right. That is what the buck sweep punishes: a defense that respects the fullback. Every linebacker who takes a step up to stop the buck is a step late to the edge, where two guards are about to arrive.

You’ll rarely hear “buck sweep” on an NFL broadcast, but you’ll see its shape. From the high sideline camera, look for two offensive linemen leaving the line at the snap and running sideways behind it toward one sideline while everyone in front of them blocks down toward the ball. When the back follows them around the end, that’s the buck sweep’s descendant, usually called a pull sweep or pin-and-pull in the modern game.

Series football: every play is the answer to the last one

That first second is the point. Raymond’s Wing-T was built as series football: groups of plays that begin with the same backfield action and end differently, each one designed to beat the defense’s reaction to another. Chris Brown, writing about Raymond’s influence, put the idea in one line: “each time a team tries to stop one thing, it opens itself up to something else.”9 The buck series is the classic example.

Read Figure 19.6 from left to right. If the linebackers learn to fly outside with the pulling guards, the fullback, who faked into the middle on the sweep, keeps the ball this time, and the middle they just vacated is where he runs: the buck trap, a trap in which the defensive lineman left unblocked at the snap gets hit from the side by a pulling guard. If the backside end chases the sweep, the quarterback keeps the ball and rolls out behind him. If the whole defense flows, the wingback comes back the other way on a counter or reverse. If a safety races up to make the tackle on the sweep, the receivers run past him on play-action (a fake handoff followed by a pass).

None of those plays is clever on its own. A trap is a trap. What makes the series work is that each play looks like the others for the first second, and that the defender’s correct reaction to one is his wrong reaction to another.

“Looks like” needs one honest qualification. The backfield is identical on the sweep and the trap, but the guards aren’t. On the sweep both guards pull toward the sweep side. On the buck trap the far guard (the left guard here) still pulls toward that side, only a short distance, to trap the first defensive lineman past the center, while the near guard blocks straight ahead to the linebacker. So a linebacker who reads the backfield, or who reads the far guard, sees the sweep starting and flies outside; one who reads the near guard sees him coming straight at him and knows. That is why the series has more than one answer. The trap beats linebackers who chase the backfield and the pulling guard. The waggle beats the backside end. The counter beats a whole defense that flows: the wingback, who starts toward the sweep side as if to block, turns back and takes the ball going the other way, against the flow. The defense can’t stop all of them by being fast or by being patient; it has to be right, play after play, about a fake it can’t see through. That is what “an order of football” means: plays called in a deliberate order, each one asking the defense a question the previous one made it answer.

The buck series is the most famous, not the only one. The Wing-T’s other base family is the belly series. The fullback, instead of faking up the middle, takes the ball off tackle behind down blocks, often with a guard pulling to kick out the end man (a close cousin of 03-03’s power). When the defense starts collapsing on the fullback, the quarterback fakes to him and keeps it, or runs an option to the outside off the same fake, and there is a waggle off belly action too, for teams that have learned to key the fullback. Buck and belly are two conversations running at once: one starts by threatening the middle and goes outside, the other starts off tackle and goes wider still.10

The waggle

One member of the series reaches straight into the modern NFL. The waggle is the Wing-T’s play-action bootleg: the quarterback fakes the buck sweep (or another run) one way and rolls out the other, behind both guards, who pull to that side to protect him, with receivers spread at three depths on that side. Notice which way the guards go: opposite the backfield’s flow. A linebacker who reads the backs runs one way while the blockers go the other.

The Wing-T’s children

You will almost never see a true Wing-T in the NFL. You will see its children every Sunday.

The jet sweep. Wing-T offenses already sent the wingback in motion, and the fly sweep, a handoff to a wingback or slot in full-speed motion behind an under-center quarterback, grew up beside them. Its exact origin is murky: many coaches credit Gene Beck’s Delano High School teams in California in the 1950s. In the 1980s and 1990s coaches such as Mark Speckman built whole offenses around it, with a trap, a bootleg and passes off the same motion, and spread it through high school clinics; Dan Hawkins carried it from small-college Willamette to Boise State, where it helped launch the Broncos’ run of winning offenses in 1999 and 2000.11 So the fly sweep is the Wing-T’s adopted child, not its own: Wing-T coaches, who already had the motion, took the play in, and the two are now usually taught together. Move the quarterback back to the shotgun, call the motion “jet,” and you have the NFL play. Pro Football Focus counted 174 jet sweep runs across the league in 2017 and 309 in 2018, when they averaged 6.1 yards a carry and Sean McVay’s Rams ran 48 of them, playoffs included.12

Gus Malzahn’s Auburn. The clearest line from Delaware to modern football runs through Gus Malzahn. As a young high school coach in Arkansas in the early 1990s, he taught himself offense from Raymond and Kempski’s book; by the late 1990s he had moved to an up-tempo shotgun spread, but kept the Wing-T’s plays and its series logic, and at Auburn he ran the buck sweep from the shotgun with pulling linemen and a receiver faking an end-around the other way.13 His 2013 team averaged more than 300 rushing yards a game, the first SEC team to do it in almost thirty years.

The NFL’s misdirection revival. When the McVay Rams and Kyle Shanahan’s 49ers made jet motion a weekly staple in the late 2010s, they were doing what Raymond did with his backfield: one picture, several plays. Shanahan went a step further with Deebo Samuel, a receiver he lined up in the backfield, put in motion and handed the ball on sweeps, reverses and runs between the tackles, so that “which player is the runner” became one more question the defense had to get right. Andy Reid’s Chiefs used jet and orbit motion to set up flips, shovels and screens back the other way. The formations are modern; the logic is the Wing-T’s.

Eye candy: half a second is a gap

Here is the payoff from Figure 19.3. Once the defense has learned to fly at jet motion, the motion itself becomes a weapon even when the ball goes somewhere else. Coaches call this eye candy: motion and fakes by players who won’t get the ball, designed to pull defenders’ eyes and first steps the wrong way.

How much can a fake buy? Not much, and that’s enough. A linebacker who takes one hard step toward a jet fake and then has to come back has lost about half a second. Linebackers move at something like 5 to 7 yards a second when they’re trying to fill a hole; half a second is three yards. A hole on an inside run is a yard or two wide and stays open for less than a second. A linebacker three yards out of place doesn’t fill it.

That is the trade eye candy makes. The receiver in motion isn’t blocking anybody, so the offense starts a man short at the point of attack, and the motion has to hold more than one defender for the play to come out ahead. It usually does only when the defense believes the motion. Eye candy is only as good as the last time the ball actually went there. Robert Woods, one of the 2018 Rams’ two main jet-sweep runners, carried the ball only 19 times in the regular season,14 but the Rams faked the jet constantly; the defense had to treat every fake as real because some of them weren’t.

How often is “sometimes”? Nobody publishes a count of jet fakes, so there is no clean ratio, but the shape is clear. In 2018 NFL offenses shifted or motioned before the snap on 43% of their plays, roughly 27 plays a game, while even the Rams, the heaviest jet-sweep team in the league, handed the ball off on a jet sweep about twice a game.15 Not every motion is jet motion, but the proportions tell you the real thing can be rare. It just can’t be absent.

Jet motion is the most common kind of eye candy, not the only one. An orbit motion behind the quarterback, a running back faking a sweep one way while the ball goes the other, a tight end running across the formation as though he’s going to block, the quarterback carrying out a bootleg fake after he hands the ball off: all of them are designed to make a linebacker’s eyes and feet do something for a fraction of a second that they shouldn’t.

CautionCommon misconception

“The jet motion was wasted; the guy didn’t even get the ball.” On most plays with jet motion the ball goes to someone else, by design. Judge the motion by what the defenders near it did. If two linebackers took a step toward the motion man and the run went through the gap one of them left, the motion was the most important “block” on the play.

How defenses fight back

None of this is free for the offense, because good defenses train against every trick in this chapter.

  • Keep the backside honest. The backside end and the backside cornerback have rules on outside runs, and the first rule is “don’t chase until the ball has crossed the formation”: check boot, counter, reverse first. That kills reverses and end-arounds and makes bootlegs and waggles harder, at the price of one fewer defender pursuing the play that really does go outside.
  • Answer the crack. When a receiver cracks inside, the cornerback over him yells “crack!” and steps up to replace the man who was cracked as the force player. The puller still has him to block, but meets him sooner and tighter than planned and can only kick him out, so the back has to cut inside, back toward the pursuit. This is a zone rule: a cornerback playing man coverage on that receiver follows him inside, so in man the defense needs someone else, usually a safety, to take over the force.
  • Spin to the motion. Rotating a safety down toward the jet side at the snap (the spin of 02-06) puts an extra defender at the edge, so the linebackers don’t have to chase. The offense’s counter is to run the other way, into the side the safety left.
  • Read your keys, not the motion. This is the hard-line version of the keys idea from the jet sweep section: tell the linebackers to read their guard (or the running back) and to leave the motion man entirely to the safeties and corners. A guard stepping right means the run is going right, whatever the jet man is doing. The problem is that the jet man is going very fast, and if nobody on the second level gives him a thought, the edge belongs to him.

The numbers in Figure 19.9 are striking. Across ten seasons, a running back’s carry was worth about −0.07 expected points on average and succeeded (gained expected points) 39% of the time. A receiver’s carry was worth about +0.10 and succeeded 51% of the time, and it gained ten or more yards on 22% of carries, more than twice the running backs’ rate. That gap, 0.17 expected points a play, is enormous. In 01-02’s terms, a tenth of a point per play is the difference between a good offense and an average one; the receiver’s edge over the back is nearly twice that.

But notice the other bar: receivers lose yards on 15% of their carries, against 10% for backs. That is the jet sweep and reverse caught behind the line. And notice the volume: well under one receiver carry per team per game, every season. Teams aren’t ignoring free yards. The yards come from the rarity.

Coaches call plays like these constraint plays, plays whose main job is to punish the defense for overplaying the offense’s base plays, the bread-and-butter runs and passes it calls most. The idea is developed fully in Constraint Theory, but the shape is already visible here. A jet sweep gains a lot when the defense isn’t ready for it, and the defense isn’t ready for it because the offense mostly runs something else. Run it twenty times a game and the defense would stop playing the inside run and start sitting on the edge, the jet sweep would turn into a two-yard loss, and the inside run would get worse too, because the linebackers would no longer be taking those hard first steps toward the motion. The value of a misdirection play lives partly in the plays it isn’t.

You may be thinking of the 2018 Rams, who ran the jet sweep twice as often as anyone and still got about +0.44 expected points per receiver carry (more in the Film room). Fair objection, and the data answers part of it. Across the ten seasons, the teams that handed the ball to receivers more often did not do worse with it: the correlation between a team’s receiver carries in a season and their average value is close to zero (about 0.07). But “more often” is relative. The Rams’ 35 receiver carries were 8% of their 437 runs, about two a game, and no team in those ten seasons went much beyond four a game. Nobody has tried twenty, so the data can’t show the collapse. What it shows is that every team, the Rams included, stayed in the range where the play still surprises. Their trick was keeping the motion constant and the carry rare.

The same caution applies when you read the chart. Some of the receivers’ edge is selection: coaches call these plays when the film says the defense is vulnerable to them, and the defense often isn’t expecting a receiver run at all. That is not a flaw in the numbers; it’s the point. A constraint play’s average looks great because it’s called at the right moments. And don’t read much into one season’s wiggle, such as 2024’s dip below zero: with only 300 to 500 receiver carries a season, any single season’s average is uncertain by about a tenth of a point either way.

ImportantWhy it exists

Misdirection exists because pursuit works. A defense that runs to the ball fast and together is the best answer to almost every run in football, so the offense needs plays that make running fast to the ball a mistake: plays that go back the other way (reverses, counters, waggles) and motion that makes the defense run to the wrong place (eye candy). The price is risk and blockers: misdirection plays lose yards more often, take longer to develop, and use players as decoys who could have been blocking. They work only in the right dose, which is why the best offenses build them into series, so the base play and its answers look the same until it’s too late.

Watch for it on Sunday

TipWatch for it

Watch the second level. On every play with jet or orbit motion:

Before the snap

  • Find the motion man and the side he’s heading to. Does anyone travel with him (man coverage), does everyone slide a spot (zone), or does a safety spin down toward his side?
  • Find the two linebackers and the backside end. These are the players whose reaction you’ll grade.

At the snap: forget the ball for one second

  • Watch the linebackers’ eyes and feet. Did they take a step toward the motion? Did they keep going?
  • Did the backside end chase flat down the line after the motion, or stay home, trailing at the depth of the ball and watching for a boot, counter or reverse?

After the play

  • Where did the ball go? If it went with the motion: did the force player hold the edge, or did the runner beat him? Was there a crack block on the edge?
  • If it went away from the motion: were the linebackers late because they chased? That’s eye candy working.
  • If the defense has been chasing all game, start expecting the play that goes back the other way: a counter, a reverse, a bootleg.

For your prediction log (01-06): on every jet motion, write G (the motion man gets it) or F (fake) before the snap, then C if the linebackers chased, H if they held. Count a forward flip to the motion man as G: to the defense it’s the same play. After a game you’ll have a small table of what this defense does with motion, which is exactly what offensive coordinators study during the week.

Takeaways

  • Outside runs are a race between the ball and the defense’s pursuit. The defense wins it with a force player (usually the outermost defender near the line) who turns the ball back inside and everyone else pursuing fast. Perimeter plays punish a slow defense; misdirection punishes a fast one, and each family needs the other.
  • Two blocks make outside runs work: the crack block, a receiver or wing hitting an inside defender from the side (above the waist, by rule), and the seal block, walling a defender off rather than moving him. Together with a puller replacing the cracking receiver, they make the crack-toss.
  • The jet sweep turns jet motion’s running start into a race the defense starts a step behind, and pulls the linebackers sideways even when the ball goes elsewhere. End-arounds (one exchange) and reverses (two) send the ball back against the flow and live or die on the backside end, whose rule is to stay home and check boot, counter, reverse.
  • The Wing-T organized all of this as series football: the buck sweep and the plays that look the same for one second and punish each reaction to it (the trap, the waggle, the counter), with the belly series as a second conversation. Its children are the NFL’s jet sweeps, pull sweeps and bootlegs.
  • Eye candy is motion or a fake that holds defenders without blocking them. Half a second is about three yards, which is a gap. It works only as long as the motion is sometimes real.
  • Receiver runs are rare (under one per team-game) and, on average, much more valuable than running back carries, with more lost-yardage plays. They are constraint plays: their value comes from the base plays they punish, which is why you can’t simply run them more.

import pandas as pd
from gridiron import bdb, show_animation

plays = pd.read_csv("data/bdb/plays.csv")
player_play = pd.read_csv("data/bdb/player_play.csv")
players = pd.read_csv("data/bdb/players.csv")

# ball carriers on designed runs, with their roster position
carriers = player_play[player_play["hadRushAttempt"] == 1].merge(players[["nflId", "position"]], on="nflId")

trk = bdb.load_tracking("data/bdb/tracking_week_*.csv")
