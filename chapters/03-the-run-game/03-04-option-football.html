def check_legal(players):
    """Seven offensive players on the line, and only the two at its ends may be eligible receivers."""
    off = [q for q in players if q.side == "off"]
    on = sorted([q for q in off if q.d > -1.0], key=lambda q: q.w)
    assert len(on) == 7, [q.key for q in on]
    for q in on[1:-1]:
        assert q.key in OL_NAMES, f"{q.key} is covered up but not a lineman"


def go_to(p, key, end, kind="block", delay=0.0, speed=None, via=()):
    """Move `key` through absolute (d, w) points `via` to `end`, as a block, path or run."""
    start = np.array(p._cur(key))
    pts = [tuple(np.array(q) - start) for q in (*via, end)]
    if kind == "block":
        return p.block(key, pts=pts, delay=delay, speed=speed)
    return {"path": p.path, "run": p.run, "pull": p.pull}[kind](key, pts, delay=delay, speed=speed)


def _P(fld, p, d, w):
    return fld.xy(*p.to_abs(d, w))

def clean(ax):
    """Box-zoomed diagrams: drop hash ticks and yard numbers, clip anything outside the window."""
    for ln in list(ax.lines):
        xs, ys = ln.get_data()
        if len(xs) == 2 and ln.get_linewidth() == 0.5 and np.hypot(xs[1] - xs[0], ys[1] - ys[0]) < 1.0:
            ln.remove()
    for t in list(ax.texts):
        if t.get_text().isdigit() and t.get_alpha() == 0.55:
            t.remove()
        else:
            t.set_clip_on(True)

MARK_R = 0.6   # frame-strip marker radius in yards (smaller than static diagrams, so engaged players stay apart)


def tracking(play):
    df = play.to_tracking()
    df["_id"] = df["displayName"].astype(str)
    return df


def carrier(play, t):
    """Who holds the ball at time t (QB until the first handoff or pitch arrives)."""
    holder = "QB"
    for kind, to, te in sorted(play.ball_events, key=lambda e: e[2]):
        if t >= te + (0.25 if kind == "pitch" else 0.0):
            holder = to
    return holder


def panel(ax, play, df, t, lateral, window, rings=()):
    """One moment of a simulated play: markers, trails, the ball, rings (and a ring on the ball carrier)."""
    tmap = df.groupby("frameId")["time"].first()
    fid = int((tmap - t).abs().idxmin())
    ylo, yhi = sorted((play.ball_y - lateral[0], play.ball_y - lateral[1]))
    fld = Field(ax=ax, orient="vertical", los=play.los, window=window, numbers=False, hashes=False,
                lateral=(ylo, yhi), to_go=play.to_go)
    cur = df[df["frameId"] == fid]
    shown = cur[cur["x"].between(play.los + window[0] + 0.4, play.los + window[1] - 0.4)
                & cur["y"].between(ylo + 0.4, yhi - 0.4)]
    ids = set(shown["_id"])
    trails = df[(df["frameId"] <= fid) & df["_id"].isin(ids) & (df["side"] != "ball")
                & df["x"].between(play.los + window[0], play.los + window[1])]
    for (_, side), g in trails.groupby(["_id", "side"]):
        xs, ys = fld.xy(g["x"].values, g["y"].values)
        ax.plot(xs, ys, color=style.OFFENSE if side == "off" else style.DEFENSE, lw=0.9, alpha=0.4, zorder=3)
    ppy = fld.pts_per_yard()
    for _, r in shown.iterrows():
        x, y = (float(v) for v in fld.xy(r["x"], r["y"]))
        if r["side"] == "ball":
            ax.add_patch(Ellipse((x, y + 0.5), 0.75, 0.45, fc=style.BALL, ec="white", lw=0.5, zorder=11))
            continue
        lab = str(r.get("label", "") or "")
        if r["side"] == "off":
            ax.add_patch(Circle((x, y), MARK_R, fc=style.OFFENSE, ec="white", lw=0.6, zorder=8))
            tc = "white"
        else:
            ax.add_patch(RegularPolygon((x, y), 4, radius=MARK_R * 1.3, orientation=np.pi / 4,
                                        fc=style.DEFENSE_TINT, ec=style.DEFENSE, lw=1.0, zorder=8))
            tc = style.INK
        fs = max(4.0, min(8.0, ppy * (0.8 if len(lab) <= 1 else 0.6)))
        ax.text(x, y, lab, ha="center", va="center", fontsize=fs, color=tc, fontweight="bold", zorder=10)
    for key, col in list(rings) + [(carrier(play, t), style.THIRD)]:
        rr = cur[cur["displayName"] == key]
        if len(rr):
            x, y = fld.xy(rr["x"].iloc[0], rr["y"].iloc[0])
            ax.add_patch(Circle((float(x), float(y)), 1.05, fc="none", ec=col, lw=1.7, zorder=7))
    return fld


def strip(plays, times, lateral, window, notes=None, rows=None, rings=(), width_in=6.5, ncols=None):
    """Frame strip: one row per play (an option's branches), or one play wrapped over rows."""
    dfs = [tracking(p) for p in plays]
    if len(plays) > 1:
        cells = [(i, t) for i in range(len(plays)) for t in times]
        ncols = len(times)
    else:
        cells = [(0, t) for t in times]
        ncols = ncols or min(3, len(times))
    nrows = int(np.ceil(len(cells) / ncols))
    pw = width_in / ncols
    span_w, span_d = lateral[1] - lateral[0], window[1] - window[0]
    fig, axes = plt.subplots(nrows, ncols, squeeze=False,
                             figsize=(width_in, nrows * (pw * span_d / span_w + (0.55 if notes else 0.25))))
    for ax in axes.flat[len(cells):]:
        ax.axis("off")
    for k, (ax, (i, t)) in enumerate(zip(axes.flat, cells)):
        panel(ax, plays[i], dfs[i], t, lateral, window, rings)
        head = f"{k % len(times) + 1 if len(plays) > 1 else k + 1}.  " + ("snap" if abs(t) < 1e-6 else f"{t:+.1f} s")
        if rows and k % ncols == 0:
            head = rows[i] + "   " + head
        ax.set_title(head, fontsize=8, color=style.INK_2 if not (rows and k % ncols == 0) else style.INK,
                     loc="left", fontweight="bold" if rows and k % ncols == 0 else "normal")
        if notes:
            ax.text(0.5, -0.03, notes[k], transform=ax.transAxes, ha="center", va="top",
                    fontsize=7.2, color=style.INK)
    fig.tight_layout(h_pad=1.4 if notes else 0.6, w_pad=0.5)
    return fig


def video(plays, times, lateral, window, notes=None, rows=None, rings=(), width_in=6.4):
    """Web version: the same panels drawn at every tenth of a second, side by side for branches."""
    import shutil
    from matplotlib import animation
    from IPython.display import HTML, display
    dfs = [tracking(p) for p in plays]
    t0 = min(float(df["time"].min()) for df in dfs)
    t1 = max(float(df["time"].max()) for df in dfs)
    grid = np.round(np.arange(t0, t1 + 1e-9, 0.1), 2)
    n = len(plays)
    span_w, span_d = lateral[1] - lateral[0], window[1] - window[0]
    pw = (width_in if n == 1 else width_in / n) * (0.85 if n == 1 else 1.0)
    fig, axes = plt.subplots(1, n, squeeze=False, figsize=(pw * n, pw * span_d / span_w + 0.6))
    axes = axes.flat

    def update(t):
        for i, ax in enumerate(axes):
            ax.clear()
            panel(ax, plays[i], dfs[i], t, lateral, window, rings)
            txt = "snap" if abs(t) < 0.05 else f"{t:+.1f} s"
            if notes:
                per = len(times)
                mine = notes[i * per:(i + 1) * per] if n > 1 else notes
                cur = [m for tt, m in zip(times, mine) if tt <= t + 1e-6]
                txt += ("\n" + cur[-1].replace("\n", " ")) if cur else ""
            head = (rows[i] + ": ") if rows else ""
            ax.set_title(head + txt, fontsize=7.5, color=style.INK, loc="left")
        return []

    anim = animation.FuncAnimation(fig, update, frames=list(grid), interval=100)
    html = (anim.to_html5_video(embed_limit=40).replace("<video ", '<video style="max-width:100%;height:auto" ')
            if shutil.which("ffmpeg") else anim.to_jshtml(default_mode="once"))
    plt.close(fig)
    display(HTML(html))

Second-and-6, midway through the first quarter. The offense is in the shotgun with the running back standing beside the quarterback. At the snap the line surges to the right, the quarterback puts the ball in the back’s stomach, and the two of them ride together toward the line for a heartbeat. The defensive end on the left side sees the handoff coming and crashes down the line to make the tackle. He gets there, too: he hits the running back square, a perfect form tackle. Except the back does not have the ball. The quarterback pulled it out at the last instant and is running around the left end, through the exact patch of grass the defensive end just left, and nobody is within ten yards of him. The replay shows something even stranger: nobody blocked that end. On purpose. This chapter explains why leaving a defender alone can be smarter than blocking him, and how to see the decision coming before the quarterback makes it.

The option principle: read one, block the rest

Start from the arithmetic of 03-01: on a normal handoff the quarterback becomes a spectator, so some defender is always free. The only question is which one.

An option answers that question in the offense’s favour. The offense chooses the free defender in advance and leaves him unblocked on purpose. Then the quarterback, holding the ball, watches what that defender does after the snap and hands the ball to whichever runner the defender is not playing. The defender is not blocked, but he is not free either: whatever he does, he is wrong. If he tackles the running back, the quarterback has the ball. If he takes the quarterback, the running back has it. The read did the work of a blocker, and the blocker the offense saved is now blocking someone else.

That is the whole idea, and everything else in this chapter is detail. Option coaches say it in one line: you don’t block him, you read him. The defender the quarterback watches is the read key (also called the read defender).

Which defender gets read?

Usually the end man on the line of scrimmage: the outermost defensive player lined up on the line on one side of the formation. Coaches shorten it to EMOLOS or EMOL, and you will see both in coaching material. “On the line” means within about a yard of the ball, in a lineman’s stance or standing up; a linebacker four yards deep is off the line and doesn’t count, however far outside he stands.

Why him? Because he is the cheapest defender to leave alone. He is at the edge of the formation, so a blocker would have to travel to reach him; he is usually a big, fast pass rusher who is hard to block anyway; and on most runs he is the man whose job is to chase the play from behind or to squeeze it from outside. If the offense can make his decision for him, the five linemen can spend their blocks inside, where the double teams are.

The end man is the usual read key, not the only one. Later in the chapter you will see plays that read a defensive tackle (the midline), the play-side end instead of the backside one (the inverted veer), and two defenders one after the other (the triple option). The principle is the same every time: pick a defender the offense would struggle to block, and make his choice the wrong one.

What the option costs

ImportantWhy it exists

Option football exists because an offense that blocks every defender it can reach still leaves the defense a free man, and the option turns that man into a non-factor without spending a blocker on him. It also attacks the defense where it is strongest: the best pass rushers usually line up on the edge, and the option makes the best of them guess. The trade is real, though. The quarterback has to be a genuine running threat, or the defender simply ignores him. He takes hits, often from big men, on plays where he keeps the ball, and sometimes on plays where he doesn’t. Every exchange between quarterback and back is a chance to fumble (the numbers section near the end puts a figure on the cost). And because the defense decides who gets the ball, the offense cannot always put the ball in its best player’s hands.

NoteMadden vs. real life

In a video game, an option play is a button press: you decide who gets the ball. In real football, the defense decides. The quarterback’s job is not to choose who gets the ball but to see, in a fraction of a second, which choice the defender has already made, and to take the other one. Good option quarterbacks talk about it this way: the defender tells them what to do. That is also why option plays are so hard to call against: a defensive coordinator who guesses right on the read defender only changes which runner gets the ball.

The mesh: deciding in half a second

On most read plays the quarterback and the running back meet just after the snap, the quarterback holding the ball in the back’s stomach while both move toward the line. That meeting is the mesh (or mesh point), which you met in 03-02. Don’t confuse it with the passing concept called mesh, which you will meet in 04-04; same word, different thing.

The mesh is a ride, not a handoff. The quarterback puts the ball in the back’s belly, the back closes his arms softly over it, and the quarterback “rides” him for a step or two, with his eyes not on the back but on the read key. At the end of the ride he either lets go (the give) or pulls the ball back out and runs (the keep). Those are the quarterback’s two answers on every read, and the vocabulary is simply give / keep (coaches also say pull for the keep).

The ride has a deadline. The ball meets the back at the quarterback’s back hip, and the ride carries it forward across his body; option coaches teach that the decision must be made before the ball passes his front hip. A quarterback who reads past that point, and the back who clamps down on the ball too early, cause most mesh fumbles: both men think they own it.1

What is the quarterback looking for? Not where the defender is, but what he can do next, which coaches teach him to judge from two things: the defender’s shoulders (turned down the line toward the back, or still square to the line of scrimmage) and his distance to the back:

  • If the read key can tackle the running back, because he is crashing down the line toward the mesh or squeezing inside, the quarterback keeps. The defender has taken himself out of the quarterback’s path.
  • If the read key can’t tackle the running back, because he stays wide, sits, or rushes upfield outside, the quarterback gives. The defender is in position to tackle the quarterback, so the quarterback is the wrong man to carry it.

Teams differ on which answer is the default. A give read says: hand it off unless the defender clearly takes the back away. A pull read says the opposite. Most teams teach the give as the default, because the running back is the designated ball carrier and a hesitant keep is the worst outcome of all: a quarterback who isn’t sure should give.

Why does the ride matter? Because the defender has to commit too. A read key who sees the handoff immediately can stop, wait, and play whoever comes out with the ball. The ride hides the ball and buys time: by the time the defender knows who has it, he is already moving toward one of them.

The zone read

The first option play most NFL fans saw, and still the most common, is the zone read, which broadcasters usually call the read option. You already know its pieces from 03-02. The line blocks inside zone to one side, every lineman stepping play-side and double-teaming to the linebackers. As in plain inside zone, the defender farthest from the play on the backside, usually the backside end, is not on anyone’s list. On plain inside zone he is a problem the quarterback can only fake at. On the zone read, he is the read key.

If the end crashes (top row of Figure 18.2), he has done exactly what he would do against plain inside zone: chase the running back from behind, down the line. Against a handoff, that is the play that catches the cutback. Against the zone read, it leaves the whole backside empty. The quarterback pulls the ball, the end tackles a back with no ball, and the quarterback runs around him.

If the end stays home (bottom row), he takes the quarterback away, so the quarterback gives. Now the end is a step or two wider than he would have been on a normal run, which means the back’s cutback lane, the best part of inside zone, is wider too. Either way the offense got what it wanted from the one defender it didn’t block.

Count the hats to see why this matters. In the picture the defense has seven in the box (four linemen, three linebackers) and the offense has six blockers (five linemen and the tight end). On a plain handoff, that’s one defender free, and inside zone picks the backside end to be him. On the zone read, the free man is “blocked” by the read, and the box is seven against seven. Remember 03-01’s lesson: the defense can always bring one more defender than the offense can block, and the read is the offense’s way of getting that man back. Expect the defense to answer by bringing yet another one, usually a safety, which is the plus-one idea that 05-05 builds.

What the zone read gives up. The keep only gains one man. The quarterback who pulls the ball runs into the next defender out. In Figure 18.2 the left guard has climbed to the Will, so that man is outside the picture: a slot defender (the nickel back lined up over a slot receiver) or a safety coming down. If nobody accounts for him, the quarterback is running into a free tackler. That is why zone-read teams add an arc block (a tight end running past the read end to block the next man) or have a receiver block the force defender (the defender whose job is to turn outside runs back inside), and why the quarterback has to be a runner who can make that man miss, and why he is exposed even on a give: he can be hit while carrying out his fake (see “Hit the quarterback,” under How defenses answer the option below).

CautionCommon misconception

“The quarterback decides whether to run.” On a true read, he doesn’t; the read key does. The quarterback’s skill is in seeing the decision early and riding the mesh long enough to make the defender commit. That is why the best option quarterbacks are not always the fastest ones: a quarterback who reads the end cleanly and keeps only when the end is truly gone gains more than a faster one who guesses.

The defense strikes back: the scrape exchange

If the offense reads one defender, the defense can answer by making that defender lie. The classic answer is the scrape exchange, which 03-02 previewed. The end crashes down hard on the running back, every time, showing the quarterback a “keep” read. Meanwhile a linebacker who would normally fill inside instead loops around behind the end (a scrape, in linebacker language: a sideways run behind the line to a gap outside; 05-04 teaches it) to the spot where the quarterback will be running. The end and the linebacker have exchanged jobs: the end takes the back, the linebacker takes the quarterback.

The scrape exchange is clever because it beats the quarterback while he does everything right. He read the end correctly; the end really could not tackle the quarterback. The defense simply put someone else there. A cousin of the same idea, the gap exchange, has two defenders swap gaps after the snap in other ways (a linebacker replacing a slanting lineman, for instance); it is built from the defense’s side in Run Fits, along with the other ways a defense “fits” an option play.

The defense pays for the exchange, too. The scraping linebacker is no longer filling his gap inside, so the give is now running at one fewer defender, and a running back who sees the crashing end can cut back into the space the linebacker left. And a defense that exchanges every time is predictable, which is exactly what the offense’s counters feed on.

The offense’s replies are the chapter in miniature, each one a different way of turning the exchange against itself:

  • Bash (03-02): swap the runners. The back runs wide around the read end and the quarterback, if he keeps, runs the inside-zone track. The read flips with them: if the end squeezes, give; if he widens with the back, keep. Here is why that hurts an exchange. The exchange is a plan made before the snap, built on where the runners usually go: the end crashes at the inside runner, and the Will races to the spot a yard or two outside the end where a zone-read keep arrives. Against bash, the end crashes as planned, which is now a squeeze, so the quarterback gives. The ball goes to the back on a much wider path, bending around the spot the Will is racing to rather than through it, and the Will has to turn and chase a running back in space from inside, instead of meeting a quarterback head-on. If the end refuses to crash and widens with the back, the quarterback keeps on the inside track the Will has just left. Bash doesn’t make the scraper vanish; it makes the defense’s plan point the wrong way.
  • Arc (03-02): send a tight end or H-back past the read end to block the scraper. The quarterback keeps into a lane with a blocker already in it.
  • Read someone else. Leave the end blocked and read the linebacker instead, or read a defensive tackle (the midline, below). The defense’s exchange only works if it knows who is being read.
  • Throw it. If the linebacker is scraping outside at the snap, the space he left behind him is open for a quick pass. Packaging a throw with the read is the run-pass option (RPO), which gets its own chapter.

The defense, in turn, has answers to those. Option football is a conversation, and in the eras this chapter visits (1960s Houston, the college spread offenses of the 2000s, 2010s Baltimore) it was usually the offense who spoke last.

Why build it this way? Three reasons, each a trade.

It puts the bigger runner inside. Many running quarterbacks are big, strong men (Cam Newton, Jalen Hurts, Josh Allen), better suited to running through a crease behind a pulling guard than to outrunning a safety on the edge. The inverted veer gives them the inside track and a lead blocker: power blocking, with the quarterback as the ball carrier.

It attacks the edge with the back. The running back, already moving sideways at full speed, gets the ball going outside a defender who has just stepped inside. Defenses are taught to “squeeze” plays inside-out with their end; the inverted veer makes that squeeze the wrong answer.

It saves the blocker the play needed most. On power as 03-03 drew it, the fullback or H-back kicks out the play-side end and the pulling guard leads through the hole for a linebacker. The inverted veer doesn’t block the end at all; it reads him. The guard still pulls for the linebacker, but the back who would have been the kick-out man is now a second runner, attacking the edge.

What it gives up: the back is running sideways, not downhill, so if the read is wrong (the quarterback gives when the end was actually wide), the back is running at a defender with no momentum and loses yards. And the quarterback is a ball carrier between the tackles, in traffic, on every keep.

Inverted veer or bash? Both put the back outside and the quarterback inside, so they are easy to merge. Two things separate them. Which end is read: bash reads the backside end, away from the line’s flow, while the inverted veer reads the play-side end, the man the play is aimed at. What the line does: bash is zone blocking with everyone stepping the same way; the inverted veer is power blocking, down blocks and a guard pulling across the formation. Look for the puller: if one crosses the center toward the read end, it is a power read. Many NFL teams also hand the outside role to a receiver instead of the back, a jet motion man sprinting across the formation at the snap (a “jet” power read); the read is the same, and 03-05 builds the jet sweep it borrows from.

The name comes from the older veer (next section), in which the back is the inside runner and the quarterback the outside one. Turn it inside out and you have the “inverted” version. Who invented the play is unclear; it was made famous by TCU in 2009 and by Gus Malzahn’s Auburn offense with Cam Newton in 2010, when Newton won the Heisman Trophy and Auburn the national championship, and in Malzahn’s circle it also went by dash read.2

Speed option: straight at the edge

The fastest of the option plays, and the simplest to watch, is the speed option. There is no handoff and no mesh. The quarterback takes the snap and runs straight at the end man on the line of scrimmage, with the running back trailing him, ready for a pitch: a one-handed, underhand toss, sideways or backward, to the trailing back (the pitch man). The back keeps a pitch relationship with the quarterback: typically about four or five yards outside him and a yard or so deeper, so that a toss can reach him without traveling forward (coaches differ on the exact spacing; the one fixed rule is that the end man must not be able to tackle both runners).3 The end man has two runners coming at him. If he takes the quarterback, the quarterback pitches; if he takes the back, the quarterback keeps and turns up inside him.

The speed option exists to get the ball to the edge fast, against a defense that is outnumbered there. Count it in Figure 18.5: on the right side the offense has the quarterback, the back and two blockers (H and Z) against the end, the nickel and the cornerback. The end is read, the other two are blocked, and the two runners face nobody until a linebacker or the safety arrives from inside. The play is over before the defense can rally.

Its costs are the pitch and the quarterback. Two rules shape the pitch. A toss that goes sideways or backward is a backward pass, and if nobody catches it, it is a live ball, like a fumble, that either team can recover. A toss that drifts forward is a forward pass, and that is worse, not safer: a forward pass is legal only from behind the line of scrimmage, and an option quarterback who has reached the end man is often at or past the line when he pitches. A forward pitch from there is an illegal forward pass, five yards and the loss of the down.4 So the pitch must go back, and a backward ball on the ground belongs to whoever falls on it. Option teams drill the pitch endlessly, and a quarterback who pitches under pressure, falling, with a defender hitting him, is the play’s weak point. That is the other cost: on the pitch branch, the end man has been invited to hit the quarterback, and he will, every time, often just after the ball is gone.

Veer and the triple option: two reads, three runners

Everything so far has had two possible ball carriers. The older option plays had three, and that is where the modern idea came from.

The veer came from Bill Yeoman’s University of Houston teams in the mid-1960s. Yeoman had a weak offensive line that could not block a dive play straight up; his solution, first used in the middle of a losing 1965 season that he expected to get him fired, was to stop blocking the first defensive lineman outside the play-side tackle and read him instead.5 The play-side tackle releases inside to a linebacker, which makes the down lineman a free man, and the quarterback either gives to a back diving just inside that defender or keeps the ball. Yeoman ran it from split backs (two running backs side by side behind the quarterback, the pro set’s backfield in 02-03): the play-side halfback was the dive, and the other halfback swung out behind the quarterback to trail him.6 Yeoman’s Houston teams ran it for two decades, and it spread across college football in the 1970s.

So be precise about the words. The veer is a blocking-and-read rule: release the play-side tackle, read the first man outside him, give to the dive or keep. Put a pitch man behind the quarterback, so a keep can become a pitch, and the veer becomes the triple option: the same first read with a second one bolted on. The triple is the play the wishbone (1968) and later the flexbone of 02-03 were built to run, and the version drawn here is the flexbone one. It has three runners:

  • the dive: the fullback (the flexbone’s B-back) driving straight at the play-side guard’s outside hip;
  • the quarterback, who rides the dive, then, if he keeps, runs down the line toward the edge;
  • the pitch man: an A-back who went in motion before the snap and is now trailing the quarterback in pitch relationship.

And it has two read keys. The first, the dive key, is the first defensive lineman on or outside the play-side tackle; the second, the pitch key, is the next defender outside him, often an outside linebacker or a safety who has come down. Everyone else is blocked. Option coaches turn that into a count, numbering the defenders from the inside out on the play side: #1 is the dive key and #2 the pitch key, both “blocked” by the quarterback’s reads; #3 is the run-support defender who shows up in the alley (the gap between the pitch key and the sideline), and the play-side A-back arcs to block him; #4 is the deep defender, usually the cornerback, and the receiver takes him.7 Count #1 to #4 in the next picture and every defender on that side is accounted for.

The quarterback’s two decisions come one after the other, both in less than two seconds:

  1. At the mesh, read the dive key. If he squeezes inside to tackle the dive, pull the ball. If he stays outside, or rushes upfield, give to the dive: it is going straight up the field just inside him.
  2. If he kept, run at the pitch key. If the pitch key attacks the quarterback, pitch. If he widens to take the pitch man, keep and turn up inside him.

Notice the order. The first read decides most snaps: if the dive key does not squeeze, the ball goes to the dive and the second read never happens. That is the triple’s most common result and much of its point. A fullback who gains four yards every time he is given the ball lets the offense run the play all day, and the only way to stop him is to squeeze the dive key inside, which opens the keep and the pitch.

The triple option is a machine for one thing: making a defense play eleven men against eleven with every assignment perfect, every snap. Against it, a defense needs someone for the dive, someone for the quarterback and someone for the pitch, on both sides of the formation, and it needs them to do their job and nothing else. That idea, “assignment football,” is how coaches describe defending it. A defender who does two jobs (who tackles the dive and tries to get the quarterback) is doing neither. One player out of place and the ball goes to the man he should have had.

Why isn’t it everywhere, then? Because its costs are steep, and steeper the higher the level:

  • The quarterback takes a hit on nearly every play, from the dive key on the pull and from the pitch key on the pitch. College option quarterbacks are built for that and are replaceable; NFL franchise quarterbacks are neither.
  • The classic flexbone version needs under-center snaps and three backs in the backfield, which tilts the offense toward the run and makes passing harder (though option teams hit big play-action passes because the safeties have to play the run). The dive needs the quarterback right behind the center, so he can open, put the ball in the B-back’s belly a step later and ride him downhill. Spread teams get a shotgun triple another way: a zone read with a pitch man added behind it, a slot in motion or a second back, so a keep can still become a pitch. Even Washington’s 2012 NFL offense mixed in what NFL.com called “double- and triple-option play calls,”8 and Navy itself has moved much of its option game into the shotgun.9
  • It is hard to come from behind with: it runs clock and gains yards in chunks, not in bursts of passing.
  • NFL defenders are fast enough to cover mistakes. A pitch key who is a step slow in college is a step fast in the NFL.

The triple option lives on where its trade still pays: at the service academies, in many high schools, and for eleven seasons (2008–2018) at Georgia Tech under Paul Johnson, whose flexbone teams 02-03 introduced. The full college story, from the split-T to the wishbone to the spread option, is told in The College Laboratory.

Midline: reading the man in the middle

The option doesn’t have to read someone on the edge. The midline option reads the first defensive lineman outside the center, often a 3-technique, the defensive tackle on the guard’s outside shoulder. The guard in front of him leaves him alone and climbs to a linebacker. The dive back runs at the A gap, right at the center’s hip, and the quarterback reads the tackle.

Why read a tackle? Two reasons. First, the 3-technique is often the defense’s best interior player, a penetrator who is very hard to block one-on-one (03-01 explains why the position became a star’s job). Leaving him unblocked and reading him neutralizes him the same way the zone read neutralizes an edge rusher. Second, the midline beats defenses that load up on the edge to stop the outside options: if the pitch and the keep outside are covered, the ball goes up the middle. Option teams use the midline as a change-up when defenses crowd the edges: it reads a different man and attacks the middle.

The cost: the read is in the middle of the line, among bodies, with almost no time. A dive back in the A gap is at the line in about a second, and a defensive tackle who reads the play can split the difference (step inside and tackle the quarterback a step later). The give is a chain-mover by design, a dive into the A gap for a few yards; the keep is different, because a clean read puts the quarterback through the line with the linebackers already blocked, and it can go the distance.

Designed quarterback runs: adding a blocker

Everything above reads a defender. The other way to make the quarterback count in the run game is simpler: give him the ball on purpose. On a designed quarterback run there is no read. The quarterback is the ball carrier from the start, and that changes the arithmetic in the offense’s favour, because the running back is free to block. On a normal run, the back is the ball carrier and the quarterback watches. On a quarterback run, the quarterback carries and the back blocks: the offense has gained a hat.

There are four designed quarterback runs you will see most weeks.

  • QB power (Figure 18.9): the power play, with the backside guard pulling through the hole and the running back as a lead or kick-out blocker. It is a short-yardage and red-zone favourite, because on third-and-2 the extra hat is worth more than anything a running back can do with the ball.
  • QB counter (Figure 18.10): the quarterback steps one way, as if running with the back’s flow, then follows two linemen pulling the other way: one kicks out the end, the other leads through the hole (03-03’s counter blocking). The back often runs the opposite way as a decoy or blocks the backside end the pulling tackle left.
  • QB draw (Figure 18.11): a run disguised as a pass. The line shows pass protection, the quarterback shows a drop, and only when the pass rushers have charged upfield and the linebackers have turned to cover receivers does he run up the middle, through the space they left. It is strongest against man coverage, where defenders have their backs to the quarterback.
  • QB sweep: the quarterback runs wide, with pulling linemen or the back leading him around the edge. It is the outside version of QB power, built for a quarterback with a running back’s speed.

QB power or power read? The names sit side by side, and a beginner (or a broadcaster) can merge them. The power read is an option: two runners, one unblocked end, and the quarterback decides after the snap who carries. QB power has no read at all: the quarterback carries every time, and the back, who on the power read was a runner, is now the man who blocks the end. Same blocking up front, a different job for the back.

Two more versions matter in the NFL. The first adds a blocker without a back: from empty (no back beside the quarterback, five receivers spread wide; 02-04), the defense has to send defenders out with the receivers, the box thins to five linemen against five or six defenders, and a quarterback draw or counter makes the quarterback the extra hat by himself. The numbers section below counts how often NFL designed runs come from empty. The second is the quarterback zone: the quarterback runs the inside-zone track himself with the back leading him, or, in the version coaches call dart, with the backside tackle pulling across to lead him through the hole.

The simplest designed quarterback run of all is the sneak, where the quarterback follows his center for a yard; its pushed-from-behind version, the tush push, has its own story in Red Zone, Goal Line, and Short Yardage.

ImportantWhy it exists

Designed quarterback runs exist because a quarterback carrying the ball turns the running back into an extra blocker: ten men block for the ball carrier instead of nine, which no handoff can do. On short yardage and near the goal line, where the defense crowds the box and every inch is contested, that extra hat is often the difference. The cost is the quarterback’s body, and NFL teams pay their quarterbacks too much to spend it twenty times a game. So designed quarterback runs are a spice, used when the extra blocker matters most and rationed the rest of the time.

How defenses answer the option

You have met the scrape exchange. Defenses have five broader answers, and you will see all of them on a Sunday when an option team is in town.

Assignment football. Every option has a fixed number of ball carriers, and the defense assigns one defender to each: someone for the dive (or the back), someone for the quarterback, someone for the pitch. The rule is discipline. The defender assigned to the quarterback takes the quarterback even when the ball was given, every time, and trusts that his teammate has the back.

Make the read hard. The read key does not have to commit. A defensive end can slow-play or feather the read: shuffle down the line with his shoulders square, inside enough to threaten the back and outside enough to threaten the quarterback, giving the quarterback a muddy picture until the last instant. A slow-playing end wants the quarterback to give, so the defense can tackle the back with the linebackers inside, or to hesitate, which kills the play.

So what does the quarterback do? The rule “if he can tackle the back, keep” doesn’t settle it, because a slow-player is built to sit exactly between can and can’t: he can reach either runner, a step late. This is where the shoulder-and-distance read from the mesh section earns its keep: shoulders turned down the line toward the back mean pull; shoulders still square and upfield mean give. Against a true slow-play, most offenses fall back on their default, the give, because a give into a slow-playing end gives the back a one-on-one with a man moving sideways, and a late, hesitant keep is the worst outcome on the menu. The offense’s longer answers are the ones you have met: an arc block, so a keep has a blocker in front of it; bash or the power read, so the end’s in-between spot is the wrong spot; or simply block the end and run the ball somewhere else.

Out-number it. Every read gains the offense a man (the triple option, with two reads, gains two). The defense can bring one more into the box, usually by rolling a safety down, and get its plus-one back. That is the box-count battle of 05-05, and it is why option offenses work best when the passing game keeps the safeties deep.

Spy the quarterback. Against a quarterback who runs, a defense can assign one defender, often a linebacker or a safety, to do nothing but shadow him: a spy (10-02 uses him on third down). The spy takes the quarterback on every keep and every designed run, which frees the rest of the defense to play the back, at the cost of one fewer man in coverage or in the box.

Hit the quarterback. The bluntest answer is also legal. Since 2013, NFL officials have treated a quarterback who is carrying out a read-option fake as a runner: as the league’s vice president of officiating, Dean Blandino, put it in a 2013 rules video, “the quarterback can be hit like a runner until he’s clearly out of the play.”11 A defense facing a read offense will tell its read key to hit the quarterback whenever he can, ball or no ball, to make every keep, and every fake, cost something.

Where option came from, briefly

Option football is older than the forward pass’s dominance, and almost all of it was invented in college, by coaches who had worse players than their opponents and needed arithmetic to win.

  • The split-T option, 1941. Missouri’s Don Faurot widened the T-formation’s line splits and built a series of plays (a dive, a quarterback keep and a “running pitchout”) that together worked like the modern option. Bud Wilkinson learned it from Faurot in the Navy during the war and took it to Oklahoma, whose split-T teams won a record 47 straight games from 1953 to 1957.12
  • The veer, 1965, Bill Yeoman at Houston, as above, and the wishbone, 1968, Emory Bellard at Texas (02-03): the triple option’s two great formations.
  • The zone read, early 1990s. At Glenville State, a small college in West Virginia, Rich Rodriguez’s quarterback Jed Drenning fumbled a handoff on a zone play in practice in 1991, picked the ball up, saw the end crashing inside, and ran where the end had been. Rodriguez saw that the mistake was a play.13 He took it to Tulane and West Virginia, and Urban Meyer’s spread-option offenses at Utah and Florida (shotgun, three or four receivers spread wide, and the quarterback’s legs as the run game’s extra threat) made it national.
  • The pistol, 2005, Chris Ault at Nevada (02-02): shotgun depth for the quarterback’s read, with the back directly behind him so the run can go either way. In 2009 Ault’s Nevada became the first team in NCAA history with three 1,000-yard rushers in one season; one of them was the quarterback, Colin Kaepernick.14

The whole lineage, from the single wing through the spread option and the RPO, is in The College Laboratory.

The NFL’s on-and-off romance with option

For most of NFL history, option football was a college thing. Professional quarterbacks were pocket passers, too valuable to be hit, and professional defenders were fast enough to punish a slow read. There were exceptions, like Michael Vick’s Falcons, and there were workarounds, like the 2008 Dolphins’ wildcat, which snapped the ball directly to a running back to get the option’s arithmetic without putting the quarterback at risk. But no NFL team built its run game on the read until 2012.

The 2012 wave. In one season, four young, fast quarterbacks, three of them from college spread-option or pistol offenses, arrived in, or took over, NFL offenses at once: Robert Griffin III in Washington, Russell Wilson in Seattle, Colin Kaepernick in San Francisco and, a year after his own rookie season, Cam Newton in Carolina. Washington’s coaches, Mike Shanahan and his son Kyle, the offensive coordinator, married their outside-zone run game to Griffin’s pistol zone read, and the result was the league’s top rushing attack: the rookie running back Alfred Morris ran for 1,613 yards and Griffin for 815, then a record for a rookie quarterback.15 In the playoffs, Kaepernick ran for 181 yards against Green Bay, a playoff record for a quarterback (the film room below has the details).16 The season is remembered as the read-option wave.

The fade. Then it cooled. Griffin, already playing on a sprained knee, tore the ACL and LCL in his right knee in the wild-card loss to Seattle on January 6, 2013, and never recaptured his rookie form.17 Defenses spent the 2013 offseason studying the read, the scrape exchange and the slow-play became standard, and the league told officials that read quarterbacks could be hit as runners. Most coaches concluded that the read option was a good change-up and a dangerous base offense, and through the mid-2010s it became a few plays a game.

The return. Lamar Jackson changed the conversation. In 2019 Baltimore built an entire offense around his reads, from the pistol and the gun, with zone read, inverted veer, speed option and designed quarterback runs, and set the NFL’s team rushing record while Jackson ran for 1,206 yards, a quarterback record, and won the MVP award unanimously.18 Jalen Hurts’s Eagles leaned on QB power, the read and the sneak; in Super Bowl LVII he ran for 70 yards and three touchdowns, both Super Bowl records for a quarterback.19 Josh Allen made the designed run a weapon for a pocket passer, especially near the goal line, and in 2024 the Commanders’ rookie Jayden Daniels broke Griffin’s rookie rushing record.20 The Ravens era is the subject of its own case study, 12-05.

What the numbers say

Can we see that history in data? Partly. The NFL’s play-by-play does not record the play call, so it cannot tell a zone read from inside zone. But it records who carried the ball, and it separately flags quarterback scrambles, the unplanned runs when a pass play breaks down. A quarterback run that is not a scramble, not a kneel-down and not a botched snap is, almost always, a designed quarterback run: a keep on a read, or a planned quarterback run. We also set aside likely sneaks (up the middle, two yards or less to go), because they are their own play with their own story. What is left is a fair proxy for the quarterback run game this chapter is about.

Three things to read from Figure 18.12.

The history is there. Designed quarterback runs nearly doubled from the late 2000s to 2012–2013, settled back, and then rose again, much higher, from 2018, peaking in 2022 at 1.4 per team per game before falling back in 2025. Two bars break the pattern, and both are about a few quarterbacks rather than the league. The high 2006 bar is mostly Michael Vick (66 designed runs, by this chapter’s count, in the season he set the quarterback rushing record) and the rookie Vince Young (30). The 2025 dip comes mostly from the heaviest users: Lamar Jackson went from 75 designed runs in 17 games in 2024 to 28 in 13, Jalen Hurts from 58 to 27, and Jayden Daniels, who played 7 games, from 63 to 14. The data shows who ran less, not why: missed games account for part of it, play-calling for the rest. Notice how small these numbers are: about one designed quarterback run a game for a typical team, even at the peak. (Washington in 2012 had 68 of them, more than four a game; Jackson alone had 113 in 2019.) Most of the option’s effect on a game is invisible in these counts, because the read’s main product is the give, a running back carry that looks like any other in the data. No public source flags a read play either: FTN’s charting marks run-pass options, but not whether a quarterback read an end on a handoff, so how often an NFL “gun handoff” is really a read is invisible in the numbers.

Per play, designed quarterback runs look far better than running back carries. Across the twenty seasons they averaged +0.15 EPA per play against -0.07 for running backs, and a 53% success rate against 38%. On first down, where the situations are most comparable, the gap is smaller but still there (+0.03 against -0.08). Read the early seasons of the right panel loosely: before 2012 the league ran only 191 to 303 designed quarterback runs a season, so one long run moves the average, while every season since 2018 has had at least 412.

But the comparison is not like for like, and the right panel cannot tell you to run your quarterback more. Three reasons:

  • Selection. On a read play, the quarterback keeps precisely when the defense has taken the back away, that is, when the keep is the open option. The keeps in this data are the reads that came out in the quarterback’s favour; the reads that said “give” are filed under the running back.
  • Situation. 28% of designed quarterback runs come on third or fourth down, against 10% of running back carries, often against defenses spread out to stop a pass. A run that converts third-and-4 is worth a lot of EPA.
  • Survivorship. The quarterbacks who run often are the ones who are good at it: Jackson, Allen, Hurts. A team without one cannot buy their numbers by calling the plays.

Some of the costs do show up. Designed quarterback runs end in a fumble far more often: on 3.1% of them across these seasons, against 0.9% of running back carries, roughly 3 times the rate, a number that bundles mesh exchanges, quarterbacks who carry the ball less securely, and the hits they take. The hits themselves are not in the play-by-play, and injuries are recorded by week, not by play, so the injury cost the chapter keeps mentioning cannot be measured cleanly from public data. One more thing the charting does show is how NFL teams build these runs: from 2022 to 2025, 11% of designed quarterback runs came from an empty backfield, with nobody beside the quarterback, the five-receiver version of adding a blocker.

The fairest reading is the one coaches reached on their own: a quarterback who is a real running threat makes the whole run game better, partly through his own carries and partly through the gives and the defenders he occupies. Baltimore’s running backs in 2019 averaged +0.043 EPA per carry, against a league average of -0.088 for running backs that season; this data cannot prove the reads caused that, but it is what the arithmetic predicts.21

NoteFilm room: Washington 2012

Washington’s 2012 offense is the cleanest NFL example of the zone read as a base play, not a gadget. The Shanahans ran their outside-zone system from the pistol, with Griffin about four yards deep and Morris directly behind him, so the back could run zone either way and the quarterback could read either end. The line’s first step looked like outside zone every time; the backside end had to decide whether to chase Morris or stay home for Griffin, and Griffin, a former college hurdler, was one of the fastest quarterbacks in the league.

What to notice on the broadcasts and the All-22 of that season: the backside end’s first step. Ends who chased gave up Griffin keeps around the edge; ends who stayed home left wide cutback lanes for Morris. Watch, too, how often Griffin carried out his fake after a give, running toward the sideline without the ball: that fake kept the end honest on the next play, and it was also where he took hits.

NoteFilm room: San Francisco 49ers, NFC divisional round, January 12, 2013

Kaepernick’s 181 rushing yards against Green Bay, in his eighth NFL start, are the read option’s most famous day in the NFL. San Francisco won 45–31. The 49ers ran their read plays from the pistol and the shotgun, and the Packers’ edge had no answer. In Robert Mays’s account for Grantland, midway through the first quarter, on third-and-2, outside linebacker Erik Walden chased Frank Gore on an option read and Kaepernick kept around the end for 13 yards; two quarters later the 49ers ran the same play, with fullback Bruce Miller sealing Walden (blocking him inside, away from the play), and Kaepernick went 56 yards for a touchdown.22

What to notice: the read key on Kaepernick’s keeps. An end who chases the back takes himself out of the play, an end who is sealed can’t recover, and the free safety is too deep to arrive in time. Then watch how defenses played the read the next season: the crash-and-scrape became standard, and long keeps got rarer. That is the defense learning.

NoteFilm room: Baltimore Ravens 2019

The 2019 Ravens are the read game as a complete offense: zone read, bash, inverted veer, speed option, QB power and the sneak, run from the pistol and the gun with tight ends and fullbacks as extra blockers. Jackson’s 113 designed runs that season (by this chapter’s count) averaged +0.32 EPA per play.

What to notice: how many Ravens runs had two threats in the backfield on every snap, and how often the same unblocked end had to defend a different blocking scheme underneath the read. The power read is this chapter’s inverted veer. A counter read uses 03-03’s counter blocking (two pullers) and reads the backside end: the back follows the pullers, and the quarterback keeps around the end if he crashes. Counter bash swaps the runners: the back runs away from the pullers, around the read end, and the quarterback, if he keeps, follows the pullers. All three leave an end alone, but each sends the runners to different places.23 Defenses fought back with the scrape exchange, with ends who kept outside leverage (staying outside the runner, so the only way past them is inside, toward help) to force the give, and with extra second-level defenders flowing to the ball, as Tennessee did in its January 2020 playoff upset with a 3-3 front (three down linemen and three linebackers, one more defender off the ball than usual);24 the full story, including how the offense changed when Derrick Henry arrived in 2024, is in the Ravens case study.

TipWatch for it

On Sunday: find the read key. Option plays happen fast, so do the work before the snap.

Before the snap

  • Where is the quarterback, and where is the back? Shotgun or pistol with a back beside or behind the quarterback is the read look, but most NFL snaps from that look are plain handoffs or passes, so treat it as “a read is possible,” not “a read is coming.” The more the quarterback has run this season, the more seriously to take it.
  • Which end? On the zone read, a back offset to one side usually runs to the other side (crossing the quarterback’s face), which makes the end on the back’s side, the backside end, the read key. The power read flips this: the back still crosses the quarterback’s face, but the read is the play-side end, the one the back is running toward. Look for a guard pulling toward that end to tell them apart. (Which end gets read is part of the play call; on some plays the quarterback can flip it at the line.)
  • Find the end man on the line of scrimmage on each side. Count from the outside in: who is the last defender standing on the line? Recheck after every shift; a linebacker walking up changes the answer.
  • Is anyone left unblocked on purpose? You can’t know until the snap, but look for a tight end or H-back lined up to the side away from the line’s likely flow: he may be the arc blocker, which means read.

At the snap

  • Real read or plain handoff? On a real read nobody touches the end, the quarterback’s head is turned toward him rather than toward the back, and the ball stays in the back’s belly for a long ride. On a plain handoff the quarterback looks at the back and the ball is gone in a step.
  • Watch the read key’s first step, not the ball. Crash or squeeze inside: predict keep. Sit, widen or rush upfield outside: predict give. Say it out loud before the ball comes out.
  • If the end crashes, find the linebacker behind him. Is he scraping outside? Then the keep is running into a scrape exchange.

After the play

  • On the replay, check your call. Did the quarterback do what the read key “told” him? If not, was it a mistake, or was the defender slow-playing?
  • On designed runs, count the hats: if the running back is blocking and the quarterback carries, the offense added a blocker. Did the defense answer with a safety in the box?

For your prediction log (01-06): on every read look, write the read key’s position and your prediction (G or K) from his first step; afterward, write what happened.

TipAnswer

Play A is the power read (inverted veer); play B is bash. In both, if the read key squeezes inside, the quarterback gives and the back has the ball outside the end. What differs is which end, and the line tells you.

  • Play A: the line blocks power. The right tackle blocks down, the right guard climbs, the center blocks back, and the left guard pulls across the center toward the right end. Nobody blocks that end, the one the play is aimed at: he is the read key, the play-side end. The back sweeps right across the quarterback’s face.
  • Play B: the line zones right, every lineman stepping the same way, with no puller. The back runs the other way, wide around the left end, the backside end, who is the read key. It is a zone read with the runners swapped.

The tell is the puller. A guard crossing the formation toward the unblocked end means power read; a line flowing away from the unblocked end means a zone-based read, and the back’s path tells you whether it is the plain zone read (back inside) or bash (back outside).

TipAnswer

On a handoff, six blockers (five linemen and the tight end) against seven in the box: one defender free, and on third-and-2 one free defender is usually enough.

Two kinds of run get a hat on every hat. A designed quarterback run makes the running back the seventh blocker: on QB power (Figure 18.9) the back kicks out the Sam, the guard pulls for the Mike, and seven blockers meet seven defenders. A read gets there the other way, by leaving one defender unblocked and reading him: a zone read or power read makes it six blockers plus a read against seven. A QB draw is tempting against man coverage too, because defenders turn their backs to chase receivers and the middle empties. (A sneak is the shortest version of the same idea, if the line to gain were a yard or less.)

But man coverage has a catch. In man, one of the box defenders, usually a linebacker, is assigned to the running back. When the back stays in to block, as on QB power or a draw, that defender has nobody to cover, and good defenses tell him to attack the run instead (coaches call him a hug or green dog player: he “hugs” the back, and when the back blocks, he comes). So the count stays an honest seven on seven, but none of the seven is busy with a pass, and on a QB draw the man who had the back is already looking at the quarterback. Power, which blocks him, suits this defense better than the draw.

The defense’s likely answer is to account for the quarterback: put a spy on him (from How defenses answer the option), or bring the deep safety down as an eighth man and live with no help deep. Either choice gives something back to the passing game, which is exactly the trade the quarterback’s legs create.

Takeaways

  • An option leaves one defender unblocked on purpose and reads him. Whatever he does, the ball goes to the runner he isn’t playing, so the read does a blocker’s work and the offense gets back the man a handoff leaves free.
  • The usual read key is the end man on the line of scrimmage; find him before the snap, and predict give or keep from his first step: if he can tackle the back, keep; if he can’t, give.
  • The families differ by who runs where and who is read: the zone read (back inside, quarterback outside, read the backside end), the inverted veer (back outside, quarterback inside behind a puller, read the play-side end; bash also puts the back outside but reads the backside end behind zone blocking), the speed option (quarterback and pitch man at the edge), the triple option (dive, keep, pitch, with two reads) and the midline (read a defensive tackle).
  • Designed quarterback runs (power, counter, draw, sweep) don’t read anyone; they turn the running back into a blocker. Their price, like every option’s, is hits on the quarterback.
  • Defenses answer with assignment football, the scrape exchange and gap exchanges, slow-playing the read, an extra man in the box, and hitting the quarterback; offenses answer back with bash, arc, new read keys and RPOs.
  • The NFL adopted the read in 2012, cooled on it, and returned to it with Jackson, Hurts and Allen. Designed quarterback runs look efficient per play, but selection and situation inflate that; their real value is what they do to the defense on every other run.
