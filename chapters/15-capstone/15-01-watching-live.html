def no_numbers(ax):
    """Remove the painted yard-line numbers (they sit where wide receivers and corners stand)."""

def snapshot(play, t, ax, lateral, window, notes=(), trails=True):
    """Everyone where he is at time t, with faint trails of where he has been and the ball in flight."""
    pos = {k: play.position(k, t) for k in play.players}
    moved = [pl.moved(d=float(pos[k][0]), w=float(pos[k][1])) for k, pl in play.players.items()]
    q = Play([m for m in moved if m.side == "off"], [m for m in moved if m.side == "def"],
             los=play.los, ball_y=play.ball_y, to_go=play.to_go)
    q.highlights = dict(play.highlights)
    for n in notes:
        q.note(*n)
    fld = q.draw(ax=ax, lateral=lateral, window=window, title=False)
    no_numbers(fld.ax)
    ppy = fld.pts_per_yard()
    P = lambda d, w: fld.xy(*play.to_abs(d, w))  # noqa: E731
    t0 = play._compile()[1]
    if trails:
        for k, pl in play.players.items():
            if pl.pos in ("LT", "LG", "C", "RG", "RT"):
                continue
            pts = np.array([play.position(k, tt) for tt in np.linspace(t0, t, 40)])
            if np.ptp(pts[:, 0]) + np.ptp(pts[:, 1]) < 0.4:
                continue
            xs, ys = P(pts[:, 0], pts[:, 1])
            col = style.OFFENSE if pl.side == "off" else style.DEFENSE
            ls = "-" if pl.side == "off" else (0, (3, 2))
            ax.plot(xs, ys, color=col, lw=1.0, alpha=0.5, ls=ls, zorder=3)
    bx, by = P(*play.ball_position(t))
    thrown = False
    for kind, _, t_rel in play.ball_events:
        if kind == "pass" and t > t_rel:
            thrown = True
            sx, sy = P(*play.ball_position(t_rel))
            ax.add_patch(FancyArrowPatch((float(sx), float(sy)), (float(bx), float(by)), arrowstyle="-|>",
                                         mutation_scale=8, color=style.BALL, lw=1.3, ls=(0, (2, 1.5)), zorder=9))
    if thrown:
        draw_marker(ax, float(bx), float(by), "ball", "", ppy)   # before the throw the QB has it: no icon on him
    elif t > 0:
        for who, tgt in play.reads:                               # the quarterback's eyes, until he throws
            (x0, y0), (x1, y1) = P(*play.position(who, t)), P(*play.position(tgt, t))
            ax.plot([float(x0), float(x1)], [float(y0), float(y1)], color=style.THIRD, lw=1.2, ls=(0, (4, 2)),
                    zorder=4)
    return fld


def strip(play, times, labels, lateral, window, notes_at=None, ncols=2, width_in=6.5):
    """The print version of an animation: one panel per moment, each titled with what to see."""
    n = len(times)
    nrows = int(np.ceil(n / ncols))
    pw = width_in / ncols
    h = nrows * (pw * (window[1] - window[0]) / abs(lateral[1] - lateral[0]) + 0.36)
    fig, axes = plt.subplots(nrows, ncols, figsize=(width_in, h), squeeze=False)
    for ax in axes.flat[n:]:
        ax.axis("off")
    for i, (ax, t, lab) in enumerate(zip(axes.flat, times, labels)):
        snapshot(play, t, ax, lateral, window, notes=(notes_at or {}).get(i, ()))
        ax.set_title(lab, fontsize=8.2, color=style.INK, loc="left")
    fig.tight_layout(h_pad=0.5, w_pad=0.4)
    return fig


def animate_or_strip(play, times, labels, lateral, window, notes_at=None):
    """Video on the web, a titled frame strip in print."""
    if PDF:
        strip(play, times, labels, lateral, window, notes_at=notes_at)

def protect(p, linemen=("LT", "LG", "C", "RG", "RT")):

    """Rushers who run into the protection (enough to show a pass rush, no more)."""

    d = pbp([season], columns=PCOLS)

    f["play_id"] = f.play_id.astype(float)
    f = f[["game_id", "play_id", "starting_hash", "qb_location", "n_offense_backfield", "is_motion", "is_no_huddle",
           "is_play_action", "is_screen_pass", "n_pass_rushers", "n_blitzers", "read_thrown",
           "is_qb_out_of_pocket"]]
    p = nfl().load_participation([season]).to_pandas()
    p = p.rename(columns={"nflverse_game_id": "game_id"})
    p["play_id"] = p.play_id.astype(float)
    p = p[["game_id", "play_id", "offense_personnel", "defense_personnel", "defenders_in_box",
           "defense_coverage_type", "defense_man_zone_type", "time_to_throw", "was_pressure", "route"]]
    d["play_id"] = d.play_id.astype(float)
    return d.merge(f, on=["game_id", "play_id"], how="left").merge(p, on=["game_id", "play_id"], how="left")


def count_pos(s, keys):
    if not isinstance(s, str):
        return np.nan
    c = {k: int(v) for v, k in re.findall(r"(\d+) ([A-Z]+)", s)}
    return sum(c.get(k, 0) for k in keys)


def pers_code(s):
    """FTN-era personnel string ('1 C, 2 G, 1 QB, 1 RB, 2 T, 1 TE, 3 WR') -> '11'."""
    if not isinstance(s, str):
        return "na"
    c = {k: int(v) for v, k in re.findall(r"(\d+) ([A-Z]+)", s)}
    code = f"{c.get('RB', 0) + c.get('FB', 0)}{c.get('TE', 0)}"
    return code if code in ("10", "11", "12", "13", "21", "22", "01", "02") else "other"


def features(d):
    d = d.copy()
    dist = pd.cut(d.ydstogo, [0, 1, 2, 3, 6, 10, 99], labels=["1", "2", "3", "4-6", "7-10", "11+"]).astype(str)
    d["dd"] = d.down.astype(int).astype(str) + "_" + dist
    d["zone"] = pd.cut(d.yardline_100, [0, 5, 10, 20, 40, 60, 80, 100],
                       labels=["g5", "g10", "rz", "o40", "mid", "own40", "own20"]).astype(str)
    d["score"] = pd.cut(d.score_differential, [-99, -9, -1, 0, 8, 99],
                        labels=["d9+", "d1-8", "tie", "u1-8", "u9+"]).astype(str)
    d["late"] = np.where(d.half_seconds_remaining <= 120, "late", "no")
    d["q4score"] = np.where(d.qtr >= 4, "q4", "no") + d["score"]
    d["pers"] = d.offense_personnel.map(pers_code)
    backs = d.n_offense_backfield.fillna(-1).clip(upper=2).astype(int).astype(str)
    d["formation"] = d.qb_location.fillna("na").astype(str) + backs
    d["motion"] = d.is_motion.fillna(False).astype(bool).astype(str)
    d["tempo"] = d.is_no_huddle.fillna(False).astype(bool).astype(str)
    d["box"] = pd.cut(d.defenders_in_box.fillna(-1), [-2, 0, 5, 6, 7, 99],
                      labels=["na", "5-", "6", "7", "8+"]).astype(str)
    d["dbs"] = d.defense_personnel.map(lambda s: count_pos(s, ("CB", "FS", "SS", "S", "DB")))
    return d

def design(d, cats, levels=None):
    if levels is None:
        levels = {c: sorted(d[c].unique()) for c in cats}
    cols = [np.ones(len(d))]
    for c in cats:
        for lv in levels[c][1:]:
            cols.append((d[c].values == lv).astype(float))
    return np.column_stack(cols), levels


def fit(X, y, lam=1.0, iters=30):
    """Ridge-penalised logistic regression by Newton's method (numpy only)."""
    w = np.zeros(X.shape[1])
    for _ in range(iters):
        p = 1 / (1 + np.exp(-X @ w))
        H = X.T @ (X * (p * (1 - p))[:, None]) + lam * np.eye(X.shape[1])
        w -= np.linalg.solve(H, X.T @ (p - y) + lam * w)
    return w


def nested(train):
    models, cats = [], []
    for name, cs in STEPS:
        cats = cats + cs
        X, lv = design(train, cats)
        models.append((name, list(cats), lv, fit(X, train["pass"].values.astype(float))))
    return models


def predict(model, d):
    _, cats, lv, w = model
    X, _ = design(d, cats, lv)
    return 1 / (1 + np.exp(-X @ w))

Twenty seconds, a hundred clues

Start with the problem the checklist solves: there’s far more to see before a snap than anyone can see, and very little time. The play clock is 40 seconds, starting the moment the previous play is whistled dead (25 seconds after some stoppages, such as a timeout or a change of possession).1 On a broadcast the first part of that is usually spent on replays of the play that just ended. The offense huddles while its play-caller talks to the quarterback through the helmet radio, which goes silent with 15 seconds left on the play clock.2 Then the players walk to the line, set, maybe move someone in motion, and the ball is snapped. A team that huddles spends most of what is left of the play clock at the line; a team in tempo can snap within a few seconds of setting.

So your budget is roughly twenty seconds, from the moment the scorebug updates to the snap, and the camera won’t show you all of it. Within those twenty seconds, information doesn’t arrive all at once. It arrives in a fixed order, because the game produces it in a fixed order:

That ordering is the first reason the checklist has the shape it has: each step is asked when its answer is visible. There is no point hunting for the safeties while the offense is still in the huddle; they haven’t settled. There is every point in reading the scorebug during the replay, because the situation is already fixed and nothing else is.

The second reason is less obvious and more important: each answer changes what the next clue means. A safety walking down to eight yards on third-and-9 is a defense guarding the sticks; the same move on first-and-10 against two tight ends is an eighth man for the run. A receiver tightening his split means one thing from a team that runs on 70% of its snaps in that look and another from a team that throws on 70%. Clues aren’t independent bits of evidence to be added up. They are answers to questions, and the questions come in an order. Read the box before you know the personnel, and you’ll miscount it. Read the shell before you know the down, and you won’t know what you’re looking at.

ImportantWhy it exists: a checklist, of all things

A checklist sounds like the opposite of expertise. Surely the expert just sees it? But the professions that have to perform under time pressure without missing anything (pilots running a pre-flight check, surgical teams before an incision) settled long ago on short, ordered checklists, and not because their people were beginners. A checklist offloads the remembering so that attention is free for the judging.3 Yours is the kind Gawande calls do-confirm: you run it from memory, then glance at the card to make sure you skipped nothing. A quarterback’s pre-snap routine (count the safeties, find the Mike, know where the hot throw goes) is a checklist too. Yours fights the same enemy: the eye goes to the most interesting thing on the screen (usually the quarterback) and stays there, while the clue that decided the play happens forty yards away.

What it gives up: a fixed routine is slower than a glance, and on the snaps where the offense hurries you’ll have to cut it short. That’s why the card has a short version.

Each tell sits in the step where you’d see it (alignment tells in step 3, depth and leverage in step 7, pressure in step 8). The rest of this section walks the nine steps in order: what you’re looking at, what it tells you, and why it sits where it does.

Step 1: situation, the base rate

Down, distance, field zone, clock, score, timeouts. You learned to read the scorebug in five seconds in How to Watch a Broadcast, and the situational chapters (Part 10, from Third Down to The Clock) taught you what each situation does to the play-caller. The output of step 1 is a number: the base rate, how often offenses throw in this situation before anyone has lined up. That is what xpass measures, and it’s the anchor every later step moves.

Add one more thing a coach reads first: where the ball is spotted. On a hash, one side of the formation has far more room (the field side) than the other (the boundary side). Defenses set their strength, rotations and blitzes by it, and offenses put their formations and perimeter plays into the room: in 2025, screens thrown from a hash went to the wide side 63% of the time.4

You don’t need a model in your head, just a few landmarks from the 2025 regular season:5

In the two-minute drill and the four-minute offense the clock calls half the plays, and The Clock gave you a separate set of questions for them (in bounds or out, timeout or spike, kneel or not). Second, does this team break the base rate? That is PROE. Before kickoff, copy three numbers per team from the one-page scouting report in Tendencies onto your sheet: its PROE, its pass rate from under center, and its play-action rate. A team with a high PROE moves your 50% first down toward 55 or 60 before you’ve seen a thing.

Why first: it’s the only clue complete during the replay, and the biggest single clue there is.

Step 2: personnel, and the match

Count the backs and tight ends as the offense runs on, and watch who runs off. The two digits (11, 12, 21, 13) were the first thing Personnel Groupings taught, because personnel is chosen before the formation: the play-caller sends in a group, and the group limits what the formation can be. Then notice which players came on, not only how many. Defenses substitute for players as well as for numbers: a star receiver on the field can draw a nickel even when the digits say two tight ends (you’ll see exactly that on the drive below). A defense in nickel against 12 personnel has a defensive back where a linebacker usually stands, and the offense will often run at him; a defense in base against 11 personnel has a linebacker who may have to cover a slot receiver, and the offense will often throw at him. This is also where you notice tempo: an offense that hurries to the line without substituting after a big gain, or right after the defense subbed into a run-stopping group, is usually hunting a mismatch the substitution freeze has locked in.

Why second: personnel is visible before the formation is, and it’s the context for what the formation shows. On its own it adds surprisingly little to a run-or-pass call; its value is in what it sets up.

Step 3: formation, strength and the offensive tells

This is the richest step. Where is the quarterback: under center, in the shotgun, in the pistol? Where is the back, how deep, offset to which side? Where is the tight end, on the line or off it, in a three-point stance or standing up? How wide are the receivers? Which side is the formation strength?

Strength matters because both sides organise around it. The defense sets its front and usually its safety rotation to the strong side, and the offense’s runs and its best route combinations usually go there too. But “strong” is a dialect, not a fact: a defense may declare it to the tight end, to the side with more receivers, or to the field, and its choice decides which way the front and the rotation set (Formation Rules and Vocabulary showed all three). When the tight end is on one side and three receivers are on the other, watch which way the linebackers shift; that tells you which dialect this defense speaks.

The vocabulary came from Formation Rules and Vocabulary, the families of plays each look goes with from the run-game and passing-game parts, and the tells catalogue below collects every alignment clue in one place.

Why third: the offense has to show its formation before the defense commits to its final picture, and the defense’s last-second picture is a response to it. Read the formation first and you’ll understand what the defense does next.

Step 4: motion response

When a player moves before the snap, watch the defense, not the man in motion. A defender who travels across the formation with him suggests man coverage; defenders who bump a spot over suggest zone; a safety who spins down turns two-high into one-high. That’s the man/zone indicator from Motion and Shifts as Weapons. Then ask the offensive question: is he still moving when the ball is snapped (motion at the snap)? A jet motion at full speed is a run threat, or a decoy for one, to that side.

You will get to use this step on most snaps. PFF’s charting had a shift or motion on 63.9% of 2025 snaps, the highest in its series, and every team but the Giants used one on at least half its plays.6

Treat the answer as a lean, not a read. Defenses know this indicator as well as you do: a man defense can bump over and swap assignments (an exchange), and a pattern-matching zone can send a defender across with the motion, which is how Motion and Shifts showed the indicator lying. File the motion response next to leverage and the shell; when all three agree, believe it.

Why fourth: motion only happens after the offense sets, and its whole point is to make the defense answer a question before the snap. The answer is a clue for steps 5 to 7.

Step 5: the box

Count the defenders in the box and compare them with the blockers (box count). Six blockers against six defenders is a fair fight for a run; six against eight means two defenders will be unblocked, which is why an offense that sees eight usually throws or checks to a run away from the extra man (Run Fits and Box Math). A light box (six or fewer) against a shotgun offense is the classic invitation for a run or an RPO.

Why fifth: the box isn’t final until motion has settled it. Count too early and you’ll count a safety who is about to leave, or miss one who is about to arrive.

Step 6: the shell

One-high or two-high? Then the details: each safety’s depth (13 yards is deep, 8 is down), his width (on the hash or inside it, or out wide), and any creep in the last seconds of the play clock. This was the spine of the coverage chapters, and Disguise, Rotation, and Split-Field Coverage was about why the shell you see is often not the coverage you get. The numbers show the gap. After the snap, NFL Next Gen Stats measured two-high coverage on 42.0% of dropbacks in 2025, the highest since it began tracking in 2018.7 Before the snap, defenses show two-high far more often than that: preparing for Seattle’s defense before Super Bowl LX, the Patriots’ offensive coordinator, Josh McDaniels, said 65 of 70 snaps would “look the same before the ball’s snapped”8 (Fangio’s Two-High Revolution). The distance between the shell shown and the coverage played is the disguise. So two-high before the snap is not a tell by itself; the tell is which of the two safeties is shallower, narrower or moving.

Why sixth: safeties are the last defenders to settle and the most likely to be lying, so you read them as late as you can.

Step 7: leverage and depth

Look at the corners: pressed at the line or off at seven yards; on the receiver’s inside shoulder or his outside. Leverage tells you what the defender is protecting and so where his help is: a corner playing outside leverage is giving up the inside because someone (a safety in the middle, a linebacker underneath) is there to take it. Then the linebackers’ depth: at four and a half to five and a half yards they are playing the run and the short middle; creeping to three, they may be coming; backing out to eight before the snap on a passing down, they are getting a head start into zone drops.

Why seventh: leverage is the last thing a defense adjusts, and it only makes sense after you know the shell. An inside-leverage corner with two deep safeties means something different from the same corner with one.

Step 8: pressure tells

How many defenders are at or near the line, and could any of them drop? Linebackers mugging the A gaps, a safety or nickel back walked up beside the end, a defensive lineman sitting back on his heels: these are the pressure look tells from the pressure chapters. Then the count that caps everything: how many players are deep? Two deep safeties and two off corners leave seven to rush and cover underneath, and a defense almost never rushes six with only one underneath. Six-man pressure is real, but it comes as Cover 0, man-to-man with nobody deep, and that needs the safeties to walk down first; if they’re still at 12 yards at the snap, six isn’t coming.

Finish with the offense’s side, which is the easiest pressure tell to see on a broadcast: the offense reacting. The center pointing and calling out the middle linebacker, the quarterback stepping up to reset the protection or turning to the sideline, a back sliding across to the other side of the quarterback: each says the offense sees a pressure and has changed who blocks whom. Then ask which way the line will slide and where the hot route goes.

Why eighth: pressure looks are deliberately last-second. Defenses walk up late so the offense has as little time as possible to change its protection, and you get the same late look the quarterback does.

Step 9: call it, with a number

Say it out loud before the snap: run or pass, and how sure you are. Then, if you have a second left, one more layer: the coverage family, how many will rush, or where the ball should go. This is the step this chapter adds, and it gets its own section below. The rule is simple: the call is made before the snap, or it doesn’t count.

Three lessons hide in that chart.9

The situation is most of the call. Down, distance, field zone, score and clock take you from 60% (“always pass”) to 69%. That is the step a casual fan skips because it feels too obvious to count.

The formation is the one big jump after it. Adding the quarterback’s alignment and the number of backs lifts the model to 75%, and, as important, raises the share of snaps it can call at 80% confidence or better from 28% to 44%. Personnel alone added little on top of the situation (about 0.5 points), because most of what the personnel tells you, the formation tells you again, more precisely.

Motion, tempo and the box add almost nothing to run or pass. Together they move the needle by 0.3 points. Not because they’re useless: they answer other questions. Motion identifies the coverage; the box says whether a run, if it comes, has a chance (and partly repeats the formation, since the defense set it in reply). The same is likely true of the steps nobody charts: safety depth and corner leverage say little about whether the offense will throw, and a great deal about where and against what.

That changes how you should spend your twenty seconds. By the end of step 3 your run-or-pass call is essentially made. Steps 4 to 8 are for the second layer: the coverage, the pressure, where the ball should go, which way the run will hit. A reader who spends all twenty seconds agonising over run or pass is using the expensive steps on the cheap question.

A few of these deserve a sentence more than the table can give them.

The back’s offset is a direction tell, not a run-or-pass tell. Shotgun runs most often send the back across the quarterback’s face, because the mesh (the moment the ball is handed off) happens while the back is moving across. A back offset to the right usually runs left. But the same offset is where a back sets up to pass-protect against a rusher from that side, so it tells you which way a run goes far more than whether there will be one. (No public charting records the back’s offset, so this one you’ll have to calibrate in your own log.)

The run’s scheme has one pre-snap tell, and it’s on a guard. A gap scheme needs a puller, and a guard who is about to pull sideways often sets up for it: less weight on his hand, a few inches deeper than his neighbours, his heel turned toward where he’s going. Zone has no such tell before the snap; you confirm it at the snap, when the whole line steps the same way together (Zone Running).

The split tell is a where, not a what. A reduced split helps three different jobs (a crack block, a shorter crossing route, room to break outside), so it can’t say which is coming. But when a receiver tightens his split on a run look, watch the defender inside him: that’s the man he’s coming to block.

Leverage points at the help. A defender usually lines up to take away the side where he has no help, and funnels the receiver toward the side where he has some. So a corner with outside leverage and a single safety in the middle is telling you the safety is his help. Inside leverage is the man-coverage look, and it comes in two kinds. With nobody in the middle (Cover 0, much press Cover 1), the corner takes away the quick inside throw himself and uses the sideline as his help. With two safeties deep, he can sit inside and trail the receiver (trail technique) because a safety is over the top: that’s 2-Man. Read leverage as an arrow pointing at the help, and the coverage family usually follows.

Safety width is the most underused tell. Everyone counts deep safeties; few look at how far apart they are. NFL hash marks are only about three yards either side of the middle of the field, so use the numbers as your ruler. A two-high safety at normal width, eight to ten yards from the middle (about halfway between the hash and the top of the numbers), has a deep half or deep quarter to cover and can’t reach the deep middle in time. A safety pinched to the hash or inside it, within three to five yards of the middle, can, and the one standing shallower than his partner can come down. That asymmetry, not the two-high picture itself, is how you see a rotation coming.

CautionCommon misconception

“NFL players are too good to have tells.” Every player has tendencies, because the best way to do a job quickly is to start in the position the job needs. What separates good teams from bad ones isn’t the absence of tells, it’s what they do about them. Every staff self-scouts its own film looking for the clue an opponent could read, and then decides whether to break it, hide it, or keep it as bait (Constraint Theory and Sequencing). The Rams’ under-center passing in 2025 is the textbook bait: they lined up under center on 59% of their snaps (the league: 34%), threw on 40% of those (the league: 31%), and made 81% of those throws play-action. The alignment says run, so the linebackers step up, and the pass goes behind them. A tell that is right 70% of the time is still worth using. You just shouldn’t be 95% sure of it.

Stated confidence: putting a number on it

Step 9 asks for a number, and this is where the course’s most important habit gets its name. A stated confidence is the probability you attach to a prediction before the snap, said or written as a number rather than a feeling: not “probably a pass” but “pass, 75%.” The prediction log has had a confidence column since Part 1. From here on, it is not optional.

Why a number, rather than just the call?

  1. It separates a read from a guess, over many calls. A single miss tells you almost nothing: one 90% call in ten should miss. A pile of them is different. If your 90% calls miss one time in three, you’re misreading something; if your 55% calls miss half the time, you were honestly unsure, and you said so. Without the number, your log can’t tell those two patterns apart.
  2. It forces you to use the base rate. You can’t say “80%” for a pass on first-and-10 without asking what moved you from 50. If nothing did, the honest number is 50 to 55.
  3. It can be scored. A forecaster who says “70% chance of rain” isn’t wrong when it stays dry; she’s wrong if, across all her 70% days, it rains on nine in ten, or on three in ten. Your calls work the same way: a 70% call should come true about seven times in ten. The next chapter, Film Study, scores the season’s log that way, with a Brier score (the average squared gap between your stated confidence and what happened) and a calibration plot (do your 70% calls come true about 70% of the time?). Researchers who study forecasters have found that saying the number, rather than a word like “probably”, is itself one of the habits that makes forecasters better.10

How do you choose the number? Start from the base rate of step 1, move it with the strong clues (the formation is the big one), and then respect two limits:

  • Never below 50. Your call is whichever side you think is more likely, so its confidence is at least 50%. A 50% call is perfectly honest: it says “I have nothing”, and logging it is better than skipping the snap.
  • Rarely above 90. Offenses choose when to break their tendencies, and the snaps that break them are the ones built to fool you. Even the most lopsided situations leave room for a surprise: third-and-7 to 10 is a pass 94% of the time, not 100%.

That chart is the target for your own log. When the model says 70 to 80%, it’s right 71% of the time; when it says 90% or more, it’s right 94% of the time. Nobody expects a person to match a model learned on 33,000 snaps, but the shape is the goal: confident where the situation is lopsided, modest everywhere else, and right about as often as you claimed.11

But look at the top band: on a fifth of snaps the model was right 94% of the time. Doesn’t a cap at 90 make you underconfident there? A little, and that’s the point: the two mistakes don’t cost the same. On a call that comes true 94% of the time, the Brier score the next chapter uses (lower is better) is 0.056 per call if you say exactly that, 0.057 if you say 90%, and 0.058 if you say 99%. Saying 90 gives up almost nothing; saying 99 costs more even when the situation really is that lopsided, and far more when it isn’t, which is the usual human error: in study after study, people’s “90% sure” answers come true well under 90% of the time.12 So 90 is a ceiling for beginners, not a law. When Film Study shows that your 90% calls come true more than nine times in ten, raise your ceiling to 95 for the empty-backfield and third-and-long snaps that earned it.

A small vocabulary helps you say the number fast:

Say Means Typical use
“Coin flip” 50% Nothing to go on: the formation was off screen, or the clues pull both ways
“Lean” 55–60% First-and-10 from a balanced look; one weak clue
“Probably” 65–70% The formation agrees with the situation
“Likely” 75–80% A strong formation clue and a situation that points the same way
“Near certain” 85–90% Third-and-long, an empty backfield, a two-minute drill when trailing

Layered predictions

Run or pass is the first layer, and after step 3 it’s mostly decided. A good reader makes a second call on the snaps where there’s time, and states a confidence for it too:

  • On a likely pass: the coverage family (man or zone; one-high or two-high after the snap), the rush (four, or five or more), or where the ball should go (the quick throw, the sticks, a shot).
  • On a likely run: the direction (toward the strength, away from a shotgun back’s offset, at the light side of the box) or the scheme family (gap, if a guard is cheating back to pull; zone is confirmed only at the snap, when the line steps together). The hash won’t call the direction for you: runs from a hash went to the field and to the boundary about equally in 2025.

Second-layer calls are harder, and your confidence should show it. Their defaults come from base rates too. In 2025 defenses sent five or more rushers on only 27% of dropbacks (39% on third-and-7 to 10), so “four rush” is the default until a pressure tell moves you off it. Zone was the coverage on about 69% of charted dropbacks, so “zone” is the default, except near the goal line: man rose from 28% of dropbacks outside the red zone to 46% inside it, because a short field leaves zone defenders little deep space to sink into, and tight coverage on every receiver is the natural answer to short throws. (FTN’s man share jumps around between seasons in a way that looks like charting rather than football, so treat these as rough.13) A second-layer call at 60–70% is a good call; anyone who claims 90% on coverages before the snap is watching a very predictable defense or not keeping score.

NoteMadden vs. real life

In Madden you always know one of the two calls, your own. Watching a real game you know neither, and neither does anyone on the field: the quarterback’s pre-snap read is this checklist, and he gets it wrong often enough that offenses build hot routes and post-snap reads for the times he does. So a probability isn’t a hedge. It is the most accurate description of what anyone, including the players, knows before the snap.

CautionCommon misconception

“My 90% call missed, so I was wrong.” Not necessarily: one in ten 90% calls should miss. You can’t judge a single probabilistic call by its outcome, only a pile of them. That’s why the log matters, and why Film Study scores a season of calls rather than a play. The mistakes to look for are patterns: 90% calls that miss one time in three (overconfidence), or a log full of 55% calls that come true 75% of the time (you know more than you’re admitting).

Between snaps: keeping the read up to date

The best clue about the next snap is often the last twenty, and two habits keep your read current while the broadcast does something else. After every snap, during the replay, write the log line (card line 12), and don’t skip the clue column: after a few games it tells you which tells are working for you.

At every stoppage (a TV timeout, the end of a quarter, a change of possession), update the picture of this team, today:

  1. What are they doing from each look? “Three runs from 12 personnel under center on first down” is a game tendency. It is more current than any season number, and play-callers know it too, so expect the tendency breaker when the streak gets long: not because a pass is owed, but because the play-caller is watching the same tally you are.
  2. What did they just change? A new protection against a rusher who is winning, a coverage rolled toward a receiver who is hurting them, a run that worked coming back with a twist (In-Game Adjustments and Tempo).
  3. What does the clock want? The score and the time left change the base rate of step 1 (a team ahead late runs, a team behind throws). And when fourth down comes, make it its own prediction before the offense decides: go, kick or punt, with a confidence, logged in the second-layer columns (l2 go, l2conf 0.6).

When the offense goes fast, or the camera won’t show you

Against tempo the twenty seconds shrink to five or ten, and the full checklist won’t fit. Run the short one: 1, 3, 6, 9. Situation (you already have it from the last snap, plus the new down and distance), formation (the single biggest clue after the situation), shell (the single most useful second-layer clue), and the call. Drop personnel (it usually hasn’t changed; that’s the point of tempo), motion (tempo offenses use less of it), and the box and leverage counts you can’t finish anyway. The measured value of what you drop is small for run or pass; the cost is mostly in your second layer, so lower those confidences when the offense is hurrying.

The same applies when the broadcast is still on a replay at the snap, which happens more than it should. Call it from the situation alone, write “sit” as the clue, and move on. A 60% call from the scorebug is still a call.

The commonest problem is milder: the snap is live, but the shot is tight, and the safeties and corners are off the top and sides of the screen. Steps 1 to 5 still work (the formation and the box are in the picture); steps 6 and 7 don’t. Write ? in the shell column rather than guessing, make your second-layer call from the base rate alone, and keep its confidence at 60% or below. Then use the replay: the end-zone and high angles usually show the whole secondary, so fill in the shell afterwards and note that you saw it late. Coverage calls you couldn’t see live get checked after the game, on the All-22, as Film Study shows.

TipWatch for it on Sunday: the live run

For one half of one game, watching one offense (one team’s play-caller, so the between-snaps tally is about one set of habits; switch teams the next time):

  • First, a warm-up. Before kickoff, pause a recorded game the moment an offense sets, give yourself ten seconds, and say steps 1 to 9 out loud. Do it on five snaps. The order should feel automatic before the clock is real.

What is the offense hunting, and what’s your call?

TipAnswer

A lean to pass, about 60%, aimed at the linebacker on the slot. Situation: second-and-2 is a run down overall (only 31% passes in 2025), but from the shotgun with 11 personnel it was 57%, and with no huddle 64% (a small sample, 131 snaps, so don’t lean on the decimal). Second-and-short is also the down where offenses take shots, because a failure still leaves third-and-short.

Personnel and tempo: this is the substitution freeze at work. Because the offense didn’t substitute, the defense had no guaranteed chance to swap a linebacker for a nickel back, so its strongside linebacker is now standing over a slot receiver in space. That is the matchup the offense sprinted to the line to get: a quick throw or a crossing route at the linebacker, or a deeper shot if the safeties cheat up for the run.

Why only 60%: walking the Sam out leaves the frozen base defense only six in the box, and with the tight end attached the offense has six blockers for them, a fair fight for a run. A team that just gained eight on the ground may simply run it again. Tempo tells you the offense likes something about this picture; it doesn’t tell you which of the two things it likes. A good log line: call P, conf 0.60, clue tempo, l2 “target the slot”, 0.55.

Takeaways

  • Order beats speed. The master checklist asks nine questions in the order the game answers them (situation, personnel, formation, motion response, box, shell, leverage, pressure, call), and each answer changes what the next clue means.
  • Run or pass is mostly settled by step 3. In 2024–2025 data the situation and the formation did almost all the work; motion, tempo and the box added under a point. Spend the later steps on the second layer: coverage, rush, where the ball goes.
  • Tells are leans with reasons. Every tell in the catalogue leaks because of physics or assignment, and every one is broken on purpose by somebody. Trust them in proportion to how often they’re right, and expect the break where it’s cheapest for the offense.
  • Say a number. A stated confidence between 50% and 90%, made before the snap, turns watching into something you can score. Start from the base rate, move with strong clues, and rarely go past 90.
  • Score the read, not the play. A correct call on a failed play is a correct call, and a 65% call that misses on a tendency-breaker is honest reading.
