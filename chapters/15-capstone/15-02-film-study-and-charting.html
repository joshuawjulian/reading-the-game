def snapshot(play, t, ax, lateral, window, notes=(), trails=True):
    """Everyone where he is at time t, with faint trails of where he has been and the ball in flight."""
    pos = {k: play.position(k, t) for k in play.players}
    moved = [pl.moved(d=float(pos[k][0]), w=float(pos[k][1])) for k, pl in play.players.items()]
    q = Play([m for m in moved if m.side == "off"], [m for m in moved if m.side == "def"],
             los=play.los, ball_y=play.ball_y, to_go=play.to_go)
    q.highlights = dict(play.highlights)
    for n in notes:
        q.note(*n)
    fld = q.draw(ax=ax, lateral=lateral, window=window, title=False)
    no_numbers(fld.ax)
    P = lambda d, w: fld.xy(*play.to_abs(d, w))  # noqa: E731
    t0 = play._compile()[1]
    if trails:
        for k, pl in play.players.items():
            if pl.pos in ("LT", "LG", "C", "RG", "RT"):
                continue
            pts = np.array([play.position(k, tt) for tt in np.linspace(t0, t, 40)])
            if np.ptp(pts[:, 0]) + np.ptp(pts[:, 1]) < 0.4:
                continue
            xs, ys = P(pts[:, 0], pts[:, 1])
            col = style.OFFENSE if pl.side == "off" else style.DEFENSE
            ls = "-" if pl.side == "off" else (0, (3, 2))
            ax.plot(xs, ys, color=col, lw=1.0, alpha=0.5, ls=ls, zorder=3)
    if t > 0:
        for who, tgt in play.reads:                               # the quarterback's eyes
            (x0, y0), (x1, y1) = P(*play.position(who, t)), P(*play.position(tgt, t))
            ax.plot([float(x0), float(x1)], [float(y0), float(y1)], color=style.THIRD, lw=1.2, ls=(0, (4, 2)),
                    zorder=4)
    return fld


def strip(play, times, labels, lateral, window, ncols=2, width_in=6.5):
    """The print version of an animation: one panel per moment, each titled with what to see."""
    n = len(times)
    nrows = int(np.ceil(n / ncols))
    pw = width_in / ncols
    h = nrows * (pw * (window[1] - window[0]) / abs(lateral[1] - lateral[0]) + 0.36)
    fig, axes = plt.subplots(nrows, ncols, figsize=(width_in, h), squeeze=False)
    for ax in axes.flat[n:]:
        ax.axis("off")
    for ax, t, lab in zip(axes.flat, times, labels):
        snapshot(play, t, ax, lateral, window)
        ax.set_title(lab, fontsize=8.2, color=style.INK, loc="left")
    fig.tight_layout(h_pad=0.5, w_pad=0.4)
    return fig

def wash(fld, play, t, keys, reads=False, alpha=0.66, half=()):
    """Fade the whole picture, then redraw only `keys` (with their trails) on top: one pass of the film.

    ax = fld.ax
    (x0, x1), (y0, y1) = ax.get_xlim(), ax.get_ylim()
    ax.add_patch(Rectangle((min(x0, x1), min(y0, y1)), abs(x1 - x0), abs(y1 - y0), fc=style.SURFACE, ec="none",

    ppy = fld.pts_per_yard()
    P = lambda d, w: fld.xy(*play.to_abs(d, w))  # noqa: E731

from gridiron.play import _interp  # noqa: E402  (keyframe interpolation, used by the trail-technique play below)


class TrailPlay(Play):
    """A Play whose man defenders play trail technique: from their alignment at the snap they settle, within
    about a second, a yard behind and a yard inside their receiver ("on his inside hip, underneath").
    gridiron's own p.man() closes the cushion to almost nothing, which draws the defender on top of the
    receiver and hides him; this keeps both visible, and it's the technique 2-Man actually asks for."""

    def __init__(self, *a, **k):
        super().__init__(*a, **k)
        self.trail = {}

    def trail_man(self, key, target, behind=1.0, inside=1.1, lag=0.15, settle=1.0):
        self.trail[key] = (target, behind, inside, lag, settle)
        self._compiled = None
        return self

    def _compile(self):
        if self._compiled is not None:
            return self._compiled
        kf, t_start, t_end = super()._compile()
        for key, (tgt, behind, inside, lag, settle) in self.trail.items():
            own, rec = kf[key], kf[tgt]
            s0, r0 = _interp(own, 0.0), _interp(rec, 0.0)
            off0 = s0 - r0
            want = np.array([-behind, -np.sign(r0[1]) * inside])          # behind him, toward the ball
            rows = [row for row in own if row[0] < 0]
            for t in np.linspace(0.0, t_end + 0.6, 70):
                f = np.clip(t / settle, 0, 1)
                f = f * f * (3 - 2 * f)                                     # ease in, ease out
                rows.append((t, *(_interp(rec, max(t - lag, 0.0)) + off0 * (1 - f) + want * f)))
            kf[key] = np.array(rows, dtype=float)
        self._compiled = (kf, t_start, t_end)
        return self._compiled

def load_season(season):
    """Runs and dropbacks, with FTN charting and participation attached."""
    d = pbp([season], columns=PCOLS)
    d = d[d.play_type.isin(["pass", "run"]) & d.down.notna()].copy()
    d["play_id"] = d.play_id.astype(float)
    f = nfl().load_ftn_charting([season]).to_pandas()
    f = f.rename(columns={"nflverse_game_id": "game_id", "nflverse_play_id": "play_id"})
    f["play_id"] = f.play_id.astype(float)
    p = nfl().load_participation([season]).to_pandas().rename(columns={"nflverse_game_id": "game_id"})
    p["play_id"] = p.play_id.astype(float)
    return (d.merge(f[FTN_COLS], on=["game_id", "play_id"], how="left")
             .merge(p[PART_COLS], on=["game_id", "play_id"], how="left"))


def count_pos(s, keys):
    if not isinstance(s, str):
        return np.nan
    c = {k: int(v) for v, k in re.findall(r"(\d+) ([A-Z]+)", s)}
    return sum(c.get(k, 0) for k in keys)


def pers_code(s):
    """FTN-era personnel string ('1 C, 2 G, 1 QB, 1 RB, 2 T, 1 TE, 3 WR') -> '11'."""
    if not isinstance(s, str):
        return "?"
    c = {k: int(v) for v, k in re.findall(r"(\d+) ([A-Z]+)", s)}
    return f"{c.get('RB', 0) + c.get('FB', 0)}{c.get('TE', 0)}"

def chart_codes(d):
    """The public charting, written in the chart sheet's short codes."""
    d = d.copy()
    d["pers"] = d.offense_personnel.map(pers_code)
    nb = d.n_offense_backfield
    d["align"] = d.qb_location.fillna("?") + nb.map(lambda v: "?" if pd.isna(v) else str(int(min(v, 2))))
    d["mot"] = np.where(d.is_motion.fillna(False).astype(bool), "M", "–")
    d["ndb"] = d.defense_personnel.map(lambda s: count_pos(s, ("CB", "FS", "SS", "S", "DB")))
    d["dpers"] = d.ndb.map(lambda v: {4: "base", 5: "nickel", 6: "dime", 7: "quarter"}.get(int(v), "?")
                           if pd.notna(v) else "?")
    d["box"] = d.defenders_in_box.map(lambda v: "?" if pd.isna(v) else str(int(v)))
    d["cv"] = d.defense_coverage_type.map(COV)
    d["rush"] = d.n_pass_rushers
    kind = np.where(d.qb_dropback == 1, "P", "R")
    kind = np.where(d.is_play_action.fillna(False).astype(bool) & (kind == "P"), "PA", kind)
    kind = np.where(d.is_screen_pass.fillna(False).astype(bool), "SCR", kind)
    d["kind"] = kind
    d["is_pass"] = (d.qb_dropback == 1).astype(int)

    def res(row):
        if row.sack == 1:
            return "sk FUM" if row.fumble_lost == 1 else f"sk {int(row.yards_gained):+d}".replace("-", "−")
        if row.interception == 1:
            return "INT"
        if row.fumble_lost == 1:
            return "FUM"
        if row.touchdown == 1:
            return "TD"
        return f"{int(row.yards_gained):+d}".replace("-", "−")
    d["res"] = d.apply(res, axis=1)
    d["hash"] = d.starting_hash.where(d.starting_hash.isin(["L", "M", "R"]), "?")   # FTN writes 0 when unknown
    return d

def fit(X, y, lam=1.0, iters=30):
    """Ridge-penalised logistic regression by Newton's method (numpy only)."""
    w = np.zeros(X.shape[1])
    for _ in range(iters):
        p = 1 / (1 + np.exp(-X @ w))
        H = X.T @ (X * (p * (1 - p))[:, None]) + lam * np.eye(X.shape[1])
        w -= np.linalg.solve(H, X.T @ (p - y) + lam * w)
    return w

def murphy(f, o):
    """Brier = reliability − resolution + uncertainty (Murphy 1973), grouping by the distinct forecast values."""
    t = pd.DataFrame({"f": f, "o": o})

    return rel, res, float(ob * (1 - ob))

What Sunday can’t tell you

Live watching, done well, is prediction. You read the situation and the formation, make a call with a number on it, and the snap tells you whether the call was right. But the snap only answers the first layer. It tells you pass; it doesn’t tell you whether the coverage you called was the coverage that was played, because the camera was on the quarterback when the safeties turned. It tells you sack; it doesn’t tell you whose fault the sack was. It tells you seven-yard run; it doesn’t tell you whether the run went where it was designed to go or bounced somewhere else because a block failed. And after a season of logging, a nagging question: you’ve been writing “70%” next to calls for months. Do your 70% calls come true seven times in ten?

Those are three different gaps, and this chapter closes each with its own tool:

  1. The film answers what happened, for every player, on every snap. It’s the only source that shows all 22 at once for the whole play.
  2. The chart turns what happened into rows: one per snap, the same fields every time, so a game can be counted instead of remembered. It is also the answer key for your log’s second layer.
  3. The score turns your log into a number that rewards both being right and being honest about how sure you were, and a plot that tells you which way to adjust.

Coaches have done the first two for as long as there has been film. In an NFL building, Monday morning belongs to the last game: every player graded on every snap, the staff deciding what to fix, and the quality control coaches’ tags on every snap feeding the reports the coordinators read (The Game Plan: A Week Inside an NFL Building). The third is borrowed from weather forecasters, who have been scored this way since 1950. You’ll do a fan’s version of all three, with free data, an affordable film subscription and a few dozen lines of Python.

ImportantWhy it exists: watching the same play four times

Why not just watch the replay carefully once? Because attention is narrower than it feels. In a famous experiment, viewers asked to count basketball passes in a short video missed a person in a gorilla suit walking through the middle of the scene; about half of them never saw it.1 Football film is that experiment on every snap: watch the quarterback and you won’t see the safety rotate; watch the safety and you won’t see which lineman lost. Coaches solved this long ago by dividing the film: the offensive line coach watches the line, the secondary coach the secondary. A fan has no staff, so you divide the film by time instead, watching the same snap several times, a different group each time.

What it gives up: time. Four passes on every snap of a game is hours of work, which is why the protocol below comes with a short version.

Getting the film

You met coaches film in How to Watch a Broadcast: every snap shown twice, once from the high wide sideline angle that keeps all 22 players in frame, and once from a high camera behind the end zone. Fans have been able to buy it since 2012. Today it comes with NFL+ Premium, the league’s own subscription: it was part of Premium for the 2026 season, and when the league described the package in 2024 its archive went back to the 2022 season.2 It isn’t posted the moment the game ends: the league’s help pages say to expect it 24 to 36 hours after the final whistle, so a Sunday game is ready by Monday night or Tuesday. Packages and prices change often, so check what’s included before you pay.

If you don’t have it, you can still do a lot with what’s free. The broadcast’s own replays include the skycam and end-zone angles on many snaps. Teams post highlight clips, and several film-breakdown writers post All-22 clips of the plays they analyse (there’s a list at the end of the chapter). None of it is every snap from every angle, which is what charting a whole game needs, but it’s plenty for studying the snaps that mattered. Setting up. Three things, side by side if you have the screen for it:

  • The film, in a player that can pause cleanly on a single frame, step a frame at a time, and slow down to half speed. Most of what you’ll do is pause on the snap, play a second and a half, and go back. If the subscription’s own player won’t step frames on your device, try it on a computer’s browser before you give up on it.
  • The chart, a spreadsheet with one row per snap (the template is below), already filled with everything the play-by-play knows, so you only type what the film adds.
  • Your live log, the sheet you kept on Sunday. You’ll check it against the film, but you won’t change it: it’s a record of what you knew before the snap, and correcting it after watching the film would make your score meaningless. Film notes go in their own column, marked as written after the fact.

Keeping the three lined up. Coaches film has no scorebug, so it can’t tell you which snap you’re on; the play-by-play can. Every row carries the quarter, the clock at the snap, the down and distance and the ball’s spot (qtr, time, down, ydstogo and yrdln in nflverse), and the film shows the snaps in the order they happened. Number the snaps in the chart, keep a finger on the down and distance and the yard line (both visible on the field), and the film, the chart and your log stay on the same row. Your log already has the quarter and clock from Sunday, which is how the two get joined later.

Timing a snap. Several passes below ask you to count seconds from the snap. A phone stopwatch started on the snap is good to a few tenths, which is enough to tell 2 seconds from 3.5. For better, step frames: video usually runs at 30 or 60 frames a second, so at 30, two and a half seconds is 75 frames.

Knowing who’s who. Jersey numbers are small in the wide shot. Keep both teams’ rosters open, identify players by where they line up first (the left tackle is the left tackle whatever his number), and confirm a number on the end-zone angle when it matters.

The four-pass film protocol

Here is the routine. It’s the course’s film-study protocol: a fixed way to study each snap on film, the same way every time. You freeze the picture at the snap and check your pre-snap read; then you watch the play four times, once for each group (the line, the quarterback, the coverage, the receivers and backs); then you write one line saying why the play worked or failed. Before any of that, watch the snap once at full speed, to see the play whole and where the ball went.

Each step uses the angle that shows its group best. The All-22 is the only one with every player in frame, so it’s the angle for the freeze, the coverage and the routes. The end-zone camera looks straight down the line, so it’s the angle for the blocks and for the quarterback’s feet and eyes.

Each step deserves its reasons.

Step 0: freeze at the snap

Watch the snap once at full speed first: you want the play as a whole, and the result, before you take it apart. Then go back to the frame where the ball is snapped and grade your Sunday read. The pre-snap columns of your log (personnel, alignment, box, shell) were your best guesses from a tight broadcast shot; the All-22 shows the truth. Was it really two-high, or was the second safety already at eight yards and creeping? Was the box six or seven? Which way did the defense set its strength, and did a linebacker shift when the motion went? Write what you missed in a notes column. Don’t overwrite the log.

Be honest with yourself here, because you now know how the play turned out. The test for each note is whether it was visible at the snap: “the second safety was at eight yards” is; “it was obviously Cover 3” usually isn’t, if the proof only came after the snap.

This step does two jobs. It grades the inputs of your call, which your score can’t: a 70% pass call that came true is a good result, but if you’d miscounted the backfield it was a lucky one. And it trains your live eye. After a few games you’ll find you’re consistently wrong about one thing (a safety’s depth is a common one, because the broadcast angle flattens the field), and you’ll start correcting for it on Sunday.

Pass 1: the line

Use the end-zone angle and watch only the ten big men: the five offensive linemen and whoever lines up across from them, plus any back or tight end who blocks. The first thing to see happens in the first half-second: did the line step forward or back? Coaches say it as “low hats, run; high hats, pass”: linemen who fire out low are blocking a run; linemen who rise and sit back are protecting a pass. Two exceptions. On play-action they fake the first, so look at how far they go: in the NFL a lineman may be only a yard past the line when a pass is thrown, so a line that surges and then stops is often a pass. And on an RPO the line run-blocks even when the ball is thrown; there the tell is the quarterback and the ball, not the line.

On a run, ask what scheme it was. If every lineman steps the same direction together, it’s zone blocking (Zone Running). If they drive straight ahead in double-teams with no lateral step and no puller, it’s duo. If linemen down-block toward the middle while someone pulls across behind them, it’s a gap scheme (Gap and Power Runs). Then find the defender nobody blocked. On most runs one is left alone on purpose (the backside end on power, the read key on an option, and often the backside end on inside zone too, for the quarterback to hold with a fake or for a tight end to cut across and block), and knowing which one tells you what the play was meant to do. Finally, where did the hole open, and was that where the back ran?

On a pass, count. How many rushed, how many blocked? Which way did the line slide, and did a back or tight end stay in? Then the most useful habit in pass-protection film: start counting when the ball is snapped. If a rusher gets to the quarterback in under two and a half seconds, the protection lost. If the quarterback still has the ball at three and a half seconds, something else went wrong first, and pass 2 and pass 3 will tell you what. In between is shared ground: the protection held long enough for a normal drop, but not forever, so the sack belongs partly to the line and partly to whatever kept the ball in the quarterback’s hand, and passes 2 to 4 decide the split. (Those thresholds are rules of thumb, not laws; a seven-step drop and a quick game have different clocks, so judge against the drop.) And when the offense has more blockers than the defense has rushers, ask where the spares went: a center with nobody to block and a back who helped nobody are part of the sack too.

Why first: the line sets the clock for everyone else. A quarterback with 1.8 seconds and one with 3.5 seconds are playing different games, and a run’s whole design is visible in the line’s first two steps.

Pass 2: the quarterback

Now watch only the quarterback, starting from the end-zone angle. His feet: a three- or five-step drop from the shotgun (roughly the timing of a five- or seven-step drop from under center, since he starts five yards back), a run fake, a rollout. His launch point, the spot he throws from. His eyes: where does he look first, and does he move them? A quarterback working a progression turns his head from one receiver to the next in order; one throwing on rhythm hitches once and lets it go at the top of his drop. Count when the ball comes out, and switch to the All-22 to see what he was looking at.

The question that matters most is a comparison: the throw he made against the throw that was there. You’ll only be able to answer it fully after pass 4, but start it here, at the moment he chose. And on runs, the quarterback still has a job: the mesh with the back, any read he made (on a zone read, did he keep or give, and was it right?), and whether his fake sold anything.

Why second: you can only judge what he did with the time you measured in pass 1. Sacks that were the protection’s fault get blamed on quarterbacks every week, and the reverse.

Pass 3: the coverage

Switch to the All-22 and start at the snap with the safeties. Their first step tells you more than anything else in the secondary: a safety who pedals back and widens is playing a deep half; one who sits flat-footed, reading the slot receiver, is playing a quarter (and will come downhill if it’s a run); one who runs toward the middle is becoming the single high safety; one who drops down toward the line is rotating into a short zone or taking a man. A second later, ask the question from the coverage chapters: is the middle of the field closed (one deep safety in the middle) or open (two deep, the middle between them)? That single answer splits the coverages into families (Disguise, Rotation, and Split-Field Coverage).

Then the corners: did each one bail (turn and run to a deep third), squat (sit on a short zone near the sideline), or trail (follow his receiver underneath, with help over the top)? Who traveled with the motion before the snap? Who blitzed, and who dropped into the spot he left? Defenders looking at the quarterback are usually in zone; defenders looking at a receiver are usually in man. Then write the coverage family, and a confidence. You never see the defense’s call sheet, so coverage read from film is still an inference, as Tendencies warned about professional charters’ labels: match coverages turn from zone into man by rule, a cornerback who turned and ran with a receiver might be in man or simply beaten, and a busted assignment can look like a design.

Why third: coverage is the hardest read on the film and deserves a clean pass with nothing else to look at. And by now you know where the ball went, so you can tell whether the coverage was beaten, or whether the ball went somewhere the coverage had already given away.

Pass 4: the skill players

Last, the receivers, tight ends and backs. On a pass, name the routes as one concept (Concepts versus Coverages): smash, flood, mesh, a four-verticals look. Then go back to the moment pass 2 found, when the quarterback looked or threw, and ask who was open then. “Open” at the moment of the throw is the only kind that matters; a receiver who comes open a second after the ball is gone wasn’t an option. Notice separation, how the receiver beat his man or found the hole in the zone, and what happened after the catch.

On a run, watch the back: where he aimed, where he cut, how many yards he gained before anyone touched him and how many after. Then the blockers in space: did a receiver’s crack block seal the edge, did the tight end reach his man?

Why last: routes only make sense against the coverage you identified in pass 3, and “open” only makes sense at the moment you found in pass 2.

The “because” line

Finish every snap with one line that says why it worked or failed, and whose play it was. There are six candidates, and a disciplined charter picks one first and adds a second only if he must:

  • The offense’s call. The play was wrong for the defense it met: a run into an eight-man box, a deep concept against a coverage built to stop it. Nobody failed; the play-caller lost the guess.
  • The defense’s call. The mirror image: a pressure or a coverage that beat the protection or the concept by design, such as a simulated pressure that left a blocker with nobody to block while a defender came free.
  • The line. A block lost and the play died there.
  • The quarterback. He had time and an open man, and he did something else.
  • The coverage. On defense: a rotation took the throw away, or a defender lost his man.
  • The skill players. A drop, a missed cut, a broken tackle, or a great catch.

“Pass, incomplete, because the boundary safety rotated down and took away the out; QB stayed with it too long” is a good line. “Bad play” isn’t. If you can’t write the line, you haven’t seen the play yet: watch it again.

Why four passes, and not more

Why four, and not one per position? Because a fan’s eye can hold about one group at a time, and these four groups answer different questions in an order where each pass gives context to the next (watch the receivers first and you’ll call a receiver “open” who the quarterback never had time to look at). More passes add little: by pass 4 you’ve seen every player, and what’s left is re-watching the moment that matters, which you should do as often as you need. A snap takes about two minutes this way once you’re practised, more at first.

A full game, in the time you have

About 60 to 65 offensive snaps a side, at two minutes each, is more than four hours for both teams. Few people have that on a Monday. So run the protocol in layers:

  • Every snap: step 0 (freeze and check the pre-snap columns) and pass 3 (the coverage), because those are the two things the broadcast hid from you and your log’s second layer needs.
  • Every third down, red-zone snap, sack, turnover and explosive play: the full protocol. These are the snaps that decided the game, and there are usually 20 to 30 of them. (For an “explosive” use the analysts’ version from Expected Points and Success Rate, a run of 10 yards or more or a pass of 20 or more; staffs vary, and many count runs of 12 or more and passes of 16 or more.)
  • Everything else: at real speed once, then the chart row.

That’s two or three hours for one team’s offense against the other’s defense, which is a realistic weekly habit. (Coaches chart the kicking game too, every punt and kickoff; this chapter sticks to the snaps from scrimmage.)

Now run the protocol on it, the way you would on the real film.

Step 0, the freeze. Third-and-6 near midfield, down 12, sixteen seconds left in the third quarter: a pass, near certain. The offense is in 11 personnel in the shotgun; the defense in dime with six in the box and two safeties at about 12 yards. The motion is the useful clue: when the tight end crossed the formation, his defender went with him. That’s the man/zone indicator from Motion and Shifts voting man. A vote, not a verdict: Macdonald’s defenses disguise heavily and play plenty of match coverage (Fangio’s Two-High Revolution and Macdonald’s Answer), in which a defender can travel with a receiver too; and the other answer to watch for is a bump, the defenders sliding over a man each, which says zone or match. The two deep safeties say two-high. Put them together and the pre-snap read is 2-Man or a disguised rotation; on third-and-6 2-Man is a common answer, because it puts a defender on every receiver short of the sticks and two safeties over the top of them.

Pass 1, the line. Four rushers against six blockers (five linemen and the back). Ask where the two spares went: here the center and left guard double the tackle, and the back checks the gap in front of him for a blitzing linebacker, finds none, and stays to help inside. The right side holds one-on-one. That leaves the left tackle alone on Hall, which is normal: a tackle is expected to handle an edge rusher by himself for a few seconds. He does, riding Hall wide and deep past the quarterback’s launch point; the problem is that Maye climbs, and Hall comes back under. Count from the snap: in the reconstruction Hall arrives at about 3.3 seconds, in the shared zone between 2.5 and 3.5, where the line held long enough for the drop and the rest of the answer is in passes 2 to 4. The charting doesn’t record that time; on the real film it’s the first thing you’d measure.

Pass 2, the quarterback. Maye takes his drop, looks right, hitches up into the pocket and still doesn’t throw. That’s a quarterback with time and no answer, or one who missed an answer. Which, pass 4 will say.

Pass 3, the coverage. Both safeties pedal back and widen into deep halves. Underneath, every receiver has a defender a yard behind him and a yard inside, in trail technique, so anything short and inside is contested and anything deep has a safety coming. The middle linebacker has the back, and when the back stays in to block, he has a choice his coach gave him: add himself to the rush, or sit in the short middle as an extra defender. Here he sits, taking away the throw a quarterback looks for against man (the crossing route underneath) and, just as important, watching Maye. 2-Man’s great weakness is the quarterback scramble: every man defender has his back to the quarterback, so against a mobile one the linebacker in the low hole and the rushers’ lanes (who keeps him from escaping outside) are part of the coverage. Seven defenders in coverage against four receivers is the whole idea of rushing four: the rush doesn’t have to win quickly, because the coverage is designed to hold out.

Pass 4, the skill players. Against 2-Man, receivers need to win on their own: the route that beats it is the one that gets away from a trailing defender toward the sideline or back toward the quarterback, before the safety arrives. The drawn concept tries exactly that on the left, an out at the sticks over a flat underneath, with a dig on the right as the backside throw. At the moment Maye hitched, nobody had won. That’s the question to answer on the real film, frame by frame: was there a window at 2.5 seconds he passed up?

The “because” line. In this version: Sack because of the coverage: 2-Man took away everything inside and short of the sticks, the protection held for a normal drop, and the four-man rush won late, at 3.3 seconds, when the quarterback climbed into the edge rusher’s second move. If on the real film Hall had arrived in two seconds, it would read because the left tackle lost, and if a receiver had been open at 2.5 seconds, because the quarterback held it. Same play in the box score, three different lessons. That’s what the film is for.

CautionCommon misconception

“A sack is a pass-protection failure.” Often it isn’t. A sack is credited to the rusher who gets there, and the offensive line usually takes the blame, but a sack is the end of a chain: protection, quarterback, coverage, routes. A defense that rushes four and covers with seven is betting that its coverage will hold long enough for four rushers to win eventually, and when that bet pays off it’s a coverage sack. In Super Bowl LX Seattle rushed four on 45 of New England’s 53 dropbacks and got 5 of its 6 sacks that way. You can’t tell from the box score which of those were the line’s fault and which were the coverage’s. The film can. (And against a mobile quarterback the bet has a second half: the four rushers must keep their lanes, so the quarterback can’t escape the pocket while the coverage holds.)

Charting a game

The film tells you what happened on one snap. To learn what a team does, you need every snap in the same form, and that’s charting. Game charting means recording every snap of a game as one row of the same fixed fields (situation, personnel, formation, motion, the defense’s look, what happened after the snap, the result), coded the same way every time, so a game becomes data you can count. Everyone calls it charting for short.

You’ve been using other people’s charting since Part 13. When the participation file says a snap was Cover 6 against 11 personnel, somebody watched that snap and wrote those codes. NFL staffs do it too: their quality control coaches tag every snap of every opponent, and the tags become the tendency report (The Game Plan). Charting your own game puts you in that chair.

Why chart, when FTN and nflverse already do? Three reasons.

  1. Timing. The public charting doesn’t all arrive at once. The play-by-play for a game is usually available within hours; FTN’s charting (alignment, motion, play-action, rushers) arrives game by game during the season; but personnel and coverage labels, in the participation file, arrive only after the season ends, around February.5 If you want to know what coverage Seattle played against 12 personnel in this season, the public data can’t tell you until the season is over. Your chart can tell you tonight.
  2. What nobody publishes. The public files don’t record the pre-snap shell (only the coverage after the snap), the corners’ leverage, the formation’s strength, which way the run went relative to that strength, the run scheme, the pass concept, the routes of receivers who weren’t targeted, or who lost the block on a sack. Those are exactly the fields that make a tendency useful, and exactly what step 0 and the four passes record.
  3. The answer key. Your live log’s second layer (the coverage family, the rush, the direction of the run) needs something to be scored against. On a 2026 game, until the participation file arrives, your chart is the only answer key there is.

And here is the codebook that goes with it. The codes matter less than the rule behind them: decide them before you start, write them on the sheet, and use them the same way on every snap. A chart in which “Cover 3” meant one thing in the first quarter and another in the fourth can’t be counted.

Three rules keep a chart honest:

  • ? is a code. If the broadcast cut away, or you can’t tell Cover 1 from Cover 3 on a late rotation, write ?. A guessed label looks exactly like a real one in the totals and quietly poisons them.
  • Code what you see, not what you expect. Charting a team you’ve studied, you’ll be tempted to write its favourite coverage before you’ve watched the safeties. That’s how a tendency becomes a self-fulfilling one in your chart.
  • Never edit the live log. The chart is what happened; the log is what you knew before it happened. Keep them in separate columns and score one against the other.

def chart_skeleton(game_id, team):
    """One row per snap of `team`'s offense, with the play-by-play columns filled and the film columns empty."""
    season = int(game_id[:4])
    d = pbp([season])
    d = d[(d.game_id == game_id) & (d.posteam == team) & d.play_type.isin(["pass", "run"])
          & d.down.notna()]               # a two-point try has no down; it isn't a charted snap
    sheet = pd.DataFrame({
        "play_id": d.play_id.astype(int), "qtr": d.qtr.astype(int), "clock": d.time,
        "down": d.down.astype(int), "togo": d.ydstogo.astype(int), "ball_on": d.yrdln,

        "yds": d.yards_gained.astype(int), "success": d.success.astype(int), "epa": d.epa.round(2)})
    for c in FILM_COLUMNS:
        sheet[c] = ""                     # yours to fill from the film
    return sheet.reset_index(drop=True)

Is your chart any good?

Professional charters disagree with each other, and you’ll disagree with yourself. The cheap test is to chart one quarter twice, a week apart, without looking at the first attempt, and compare. For each film column, count how often the two attempts agree. Personnel and the box should agree almost every time; if they don’t, your definition is fuzzy (where exactly does “the box” end?). Coverage will agree less often, and that’s normal: it’s the hardest column, and the place to use ? more freely. If you want a single number that allows for agreement by chance, Cohen’s kappa is the standard one.6 You don’t need it to learn something: the columns where you disagree with yourself are the ones to stop trusting in your totals, and the ones to practise.

Scoring yourself

Now the log. Since Downs, Distance, and Field Position you’ve been calling run or pass before the snap. Since How to Watch a Broadcast you’ve written a confidence next to each call, and since Watching Live that confidence has been a stated number between 50% and 90%. This section scores it.

To make that concrete, the rest of the chapter uses an example log: every Panthers offensive snap of the 2025 season, 1054 calls across 18 games (17 regular-season games and the wild-card loss to the Rams). The situations and the results are real. The calls are simulated: a reader who sees roughly what a model using the situation and the formation sees, through some noise, and who starts the season overconfident and settles down as it goes on.9 It’s a stand-in for your log, built so that its mistakes are the common ones. Its clue column uses just two codes, so it can be counted: align on snaps where the formation moved the call away from what the situation alone said (where the situation-and-formation model and xpass differ by more than ten points), and sit where the situation alone decided it. (In Watching Live the clue column was free text; settling on a few codes is what lets you score it.) On the website, the folded code at the top of the chapter builds this log; LOGDF.to_csv("prediction_log_2025.csv", index=False) writes it out if you want to run what follows.

Why hit rate isn’t enough

The first number is the one you’ve been computing since Part 1: the share of calls that came true. The example reader was right on 74% of calls. Against the bars from How to Watch a Broadcast, that’s good: on the same Panthers snaps, “always pass” would have been right 59% of the time, the down-and-distance rule from your first tally 60%, and nflverse’s situation-only xpass model 66%.

But hit rate throws away the confidence column, and the confidence column is where most of a reader’s skill shows. Two readers can both be right 70% of the time: one who said 70% on every call and meant it, and one who said 95% on half his calls and 55% on the rest and was wrong on a pile of the 95s. The second reader is making a mistake the hit rate can’t see. You need a score that rewards being right and being honest about how sure you were.

The Brier score

That score is the Brier score: the average, over all your calls, of the squared gap between your stated confidence and what happened, counting what happened as 1 if the call came true and 0 if it didn’t.

\[ \text{Brier} = \frac{1}{N}\sum_{i=1}^{N} \left(c_i - o_i\right)^2, \qquad c_i = \text{stated confidence},\quad o_i = \begin{cases}1 & \text{call came true}\\ 0 & \text{it didn't}\end{cases} \]

Lower is better. Three calls make the arithmetic plain:

Three calls, scored. The confident miss costs twenty times what the confident hit earned back.
Call Confidence Came true? Gap Squared
Pass 0.80 yes (1) 0.20 0.04
Run 0.60 no (0) 0.60 0.36
Pass 0.90 no (0) 0.90 0.81
Brier (0.04 + 0.36 + 0.81) / 3 = 0.40

Read the table as a set of incentives. A confident call that comes true costs almost nothing (0.04). A timid call that misses costs a moderate amount (0.36). A confident call that misses costs a lot (0.81). And a reader who says 50% on every snap, never committing to anything, scores exactly 0.25, whatever happens, because every gap is one half. So 0.25 is the “I have nothing” line: a log that scores worse than 0.25 has done harm by stating its confidences.

Glenn Brier, a statistician and meteorologist at the U.S. Weather Bureau, proposed the score in 1950 to grade forecasts like “70% chance of rain”, and it’s still the standard way to grade probability forecasts in weather, in medicine, and in the forecasting tournaments that taught Watching Live its habit of stating numbers.10 It doesn’t matter which way round you write your calls: scoring “pass, 0.70” against whether it was a pass gives the same number as scoring “my call, 0.70” against whether your call came true. Your log’s conf and hit columns are all it needs.

Why squared? A score you can’t game

Why square the gap, rather than just averaging it, or awarding points for confidence on the calls you got right? Because of what each rule rewards you for saying. Suppose that on a certain kind of snap, a call comes true 75% of the time, and you know it. What should you say?

That property has a name: the Brier score is a proper scoring rule. The curves also explain the advice in Watching Live to rarely go above 90%: near the bottom each curve is flat, and away from it it climbs fast. Calibration

The Brier score says how good. The calibration plot says which way to adjust. Calibration is how closely your stated confidences match how often your calls come true: a well-calibrated reader’s 70% calls come true about 70% of the time, his 90% calls about 90%. To see it, group your calls by the confidence you stated, and for each group plot the stated confidence against the share that actually came true. Perfect calibration is the diagonal.

def bins(conf):
    return pd.cut(conf, [0.49, 0.575, 0.675, 0.775, 0.875, 1.0], labels=False)


def wilson(k, n, z=1.96):
    if n == 0:
        return np.nan, np.nan
    p = k / n
    den = 1 + z ** 2 / n
    mid = (p + z ** 2 / (2 * n)) / den
    half = z * np.sqrt(p * (1 - p) / n + z ** 2 / (4 * n ** 2)) / den
    return mid - half, mid + half


def calib_table(conf, hit):
    t = pd.DataFrame({"c": conf, "h": hit, "b": bins(np.asarray(conf))})
    g = t.groupby("b").agg(n=("h", "size"), k=("h", "sum"), c=("c", "mean"), h=("h", "mean")).reset_index()
    lohi = np.array([wilson(k, n) for k, n in zip(g.k, g.n)])
    g["lo"], g["hi"] = lohi[:, 0], lohi[:, 1]
    return g

Three things to read off a calibration plot, in this order.

Where are the dots relative to the diagonal? In the first half of the season the example reader’s dots sit well below it: his 70–75% calls came true 60% of the time, and his 90%-plus calls 85%. That’s overconfidence, the commonest pattern there is: in study after study, people’s “90% sure” comes true well short of 90% of the time (Watching Live cited the classic review). The fix is mechanical. Pull the confidences in: when you feel 90%, write 80% for a few weeks, and watch whether the dots climb toward the line.

How wide are the bars? A dot from 30 calls can sit ten points off the diagonal by chance alone; a dot from 300 can’t. The vertical bars are 95% intervals for that wobble.11 Where a bar crosses the diagonal, the gap may be noise. This is why one game can’t tell you whether you’re calibrated: sixty calls spread over five groups is about a dozen per group, and every bar would span more than thirty points. A season can. As a rule of thumb, wait for about 50 calls in a group before you trust its dot: at 50 calls, a group that comes true 75% of the time has a bar about 12 points either side. With five groups that’s roughly 300 calls, five or six games of every snap.

How spread out are the dots? This is the one people forget. A reader who says 60% on every snap of a team that throws 60% of the time is perfectly calibrated: his single dot sits right on the diagonal. He’s also useless; he never tells you anything the team’s pass rate didn’t. What makes a reader good is resolution: stating different confidences on different snaps, and having those differences come true. On a calibration plot, resolution is the dots being spread from the bottom-left to the top-right, with the 90% calls really coming true more often than the 60% calls. The Brier score rewards both. Calibration is knowing how sure to be; resolution is knowing more.

A Brier score you can read off the plot. Each dot is enough to estimate what its group costs. If a group was stated at \(c\) and came true a share \(h\) of the time, each call in it costs, on average,

\[ h\,(1-c)^2 + (1-h)\,c^2 . \]

A group stated at 90% that came true 80% of the time costs 0.8 × 0.01 + 0.2 × 0.81 = 0.17 a call; the same calls stated at 80% would have cost 0.8 × 0.04 + 0.2 × 0.64 = 0.16. Weight each group by its number of calls (the bars under the plot) and you have the whole log’s score, near enough.

NoteGo deeper: the Brier score in three pieces

Allan Murphy showed in 1973 that the Brier score splits exactly into three parts:12

\[ \text{Brier} = \underbrace{\text{reliability}}_{\text{calibration error}} \;-\; \underbrace{\text{resolution}}_{\text{knowing more}} \;+\; \underbrace{\text{uncertainty}}_{\text{the game's own}} \]

Write each call as a chance of a pass (a “run, 70%” call is a 30% chance of a pass) and group the calls by that chance. Reliability is the average squared gap between each group’s stated chance and its actual pass rate: calibration error, lower is better. Resolution is how far the groups’ actual pass rates spread from the overall pass rate: higher is better, and it’s the only term that rewards reading. Uncertainty is \(p(1-p)\) for the overall pass rate \(p\): the part nobody can remove, the same for every reader of the same snaps.

For the example log, games 1 to 8: reliability 0.020, resolution 0.075, uncertainty 0.244. Games 9 to 18: reliability 0.009, resolution 0.064, uncertainty 0.238. The calibration error halved: that’s the reader getting honest, and it’s the cheap kind of improvement, because it takes no new knowledge. Resolution slipped a little, partly because the later games were a different mix of snaps, which is why the total Brier score moved less than the calibration did. Raising resolution takes reading more, which is what the rest of this course was for.

Scoring the log in Python

Here’s the scoring routine. It reads a log in the final form from Watching Live (you need at least the call, conf and result columns, and clue and game if you have them) and prints everything this section has discussed. It’s run here on the example log, saved as prediction_log_2025.csv.

import numpy as np
import pandas as pd


def score_log(log, base_rate=None):
    """Hit rate, Brier score, skill against the base rate, and the calibration table of a prediction log."""
    hit = (log["call"] == log["result"]).astype(float)
    conf = log["conf"].astype(float)
    brier = ((conf - hit) ** 2).mean()
    p = (log["result"] == "P").mean() if base_rate is None else base_rate
    ref = p * (1 - p)                                   # Brier score of always saying the base rate
    print(f"{len(log)} calls: right {hit.mean():.1%}, average stated {conf.mean():.1%}")
    print(f"Brier {brier:.3f} (always 50%: 0.250; base rate: {ref:.3f}); "
          f"skill vs base rate {1 - brier / ref:+.2f}")
    groups = pd.cut(conf, [0.49, 0.575, 0.675, 0.775, 0.875, 1.0],
                    labels=["50-55%", "60-65%", "70-75%", "80-85%", "90%+"])
    table = (pd.DataFrame({"stated": conf, "right": hit})
             .groupby(groups, observed=True).agg(calls=("right", "size"), stated=("stated", "mean"),
                                                 right=("right", "mean")))
    table["gap"] = table.right - table.stated           # negative = overconfident
    print(table.round(3).to_string())
    return brier

Scoring the second layer against your chart

Your run-or-pass calls can be scored from the play-by-play alone. The second layer can’t: “zone, 65%” needs to know what coverage was played, “five or more rushers, 60%” needs the rush count, “the run goes left” needs the direction. That’s what the chart is for. Join the log to the chart on the snap (your log has the quarter and the clock from Sunday, and so does the chart), and score each second-layer call exactly as you scored the first: its confidence against whether it came true.

The calls come from a scouting card, Seattle’s 2025 regular-season two-high rate on dropbacks by down, which is the kind of call a fan could make from Tendencies before the game. NoteGo deeper: Brier score or log loss?

Tendencies scored xpass with log loss, the average of \(-\ln(\text{probability you gave to what happened})\). It’s also proper, and it’s what most machine-learning models are trained on. The difference is in how it treats confident misses. A 95% call that misses costs 0.90 in Brier and 3.0 in log loss; a 99% call that misses costs 0.98 in Brier and 4.6 in log loss; a 100% call that misses costs 1 in Brier and infinity in log loss. For a model, that’s a useful alarm. For a human log, where one careless 99% would swamp a season, the Brier score’s cap is kinder and easier to read, and it’s what the best-known forecasting tournament for people, the one behind Superforecasting, used.

Single-high after the snap, about 60–65%, most likely Cover 3 with the boundary safety rotating down; watch the field safety’s first step. He’s already close enough to drop into a short zone on the boundary side at the snap (a buzz rotation if he goes to the hook area, a sky rotation if he goes to the flat), and the field safety, already wide, would then run to the deep middle. Look at the slot on the short side, too: nobody is aligned over H. The pinched safety is the only defender who can get to that area quickly, another sign that he’s the one coming down. The corners fit: off, outside leverage, eyes inside is the alignment of a corner who will bail to a deep third and keep everything in front of him and outside him (The Cover 3 Family), and the seven-yard cushion is the zone-ish part of that clue.

The field safety’s width argues the other way: he’s well outside the middle of the field, and to become the single-high safety he has to cover a lot of ground at the snap. What will pass 1 show at the snap, which way will the run go, and which defender will nobody block?

TipAnswer

A gap run to the right, most likely power: the left guard pulls, the line blocks down, and two defenders go unblocked: the backside end by design and the walked-down safety by arithmetic. Run about 75%; right, about 65%. Start with run or pass: second-and-4 from 21 personnel under center is a run look, though eight in the box is the defense inviting play-action, so not 90%.

The guard is the scheme tell from Watching Live’s catalogue: a lineman about to pull sideways sets up for it, with less weight forward and a little depth. A pulling left guard runs to the right, toward the tight end, so this looks like power (Gap and Power Runs). What pass 1 should show, if that’s right, against this front: the center blocks back on the tackle shaded toward the pulling guard’s side; the right guard and right tackle double-team the tackle outside the guard and one of them climbs to the backside linebacker; the tight end blocks down on the end over him; the fullback runs at the Sam, the last man on the line, and kicks him out; and the left guard pulls behind the line and turns up through the hole the kick-out and the down blocks create, looking for the middle linebacker.

Who’s unblocked? Count it. Power brings seven blockers to this front (the five linemen, the tight end and the fullback). The left tackle spends his on the gap the guard left, so six of them block a defender, and eight defenders are in the box: two are free. One is the defensive end on the left, the backside, left alone on purpose: he’s too far from the play to matter if it’s fast enough. That’s the trade power makes: it spends the backside guard to get an extra blocker at the point of attack, and it accepts one free defender chasing from behind. The other is the strong safety, walked down outside the Sam. Nobody has him; he’s the defense’s extra man, and the back has to make him miss. That’s why runs into eight-man boxes are so often checked to play-action, or sent with a receiver cracking down on the safety. If the film shows the backside end making the tackle, the “because” line is about speed (the play was too slow, or the back hesitated); if the safety makes it two yards downfield, the call lost to the eight-man box.

And if the line instead steps together, all to one side, with no puller, the guard’s stance was a false tell: it’s zone, and you note it in your clue column.

Which reader has the better Brier score? What’s each one’s mistake, and what should each change?

TipAnswer

Reader B scores better (0.187 against 0.195), though both are making a fixable mistake in opposite directions. Both were right on about 75% of calls, so the whole difference is in the confidence column.

Reader A is overconfident. His dots sit below the diagonal all the way up, and his 90%-plus calls, the bulk of his log, came true only 81% of the time. Each of those misses costs between 0.8 and 1 in Brier score. With the back-of-envelope rule from the calibration section: that group, 66% of his calls, was stated at 97% on average and came true 81% of the time, so it cost about 0.18 a call; stated at its true rate it would have cost 0.16. B’s big group, the 70% of his calls at 65% or less, was stated at 58% and came true 68%: about 0.23 a call. The comparison is closer than the pictures suggest, which is why the answer has to come from the numbers rather than the look of the plot. The fix for A: pull everything in. His calls already sort snaps well (his dots climb from left to right, which is resolution); his numbers are just too big.

Reader B is underconfident. His dots sit above the diagonal: his 70% calls came true far more often than 70%. He knows more than he admits, and he never stated more than 85% on any snap. The fix: push the confidences out where the clues agree, toward 90% on the empty-backfield, third-and-long snaps.

How much is at stake? If each reader replaced every stated confidence with the rate at which those calls actually came true (perfect calibration, the same calls), A’s Brier score would fall to 0.173 and B’s to 0.172. Same reading, better numbers. That is the cheapest improvement in the whole course, and the calibration plot is how you find it.

Takeaways

  • Film answers what Sunday can’t. The All-22 shows all 22 players for the whole play; the coaches film’s end-zone angle shows the line. Use them to check your reads, not to rewrite them.
  • Four passes, in order. Freeze at the snap and grade your read; then the line, the quarterback, the coverage, the skill players, from what sets the clock to what uses it; then one “because” line that says whose play it was.
  • A chart turns a game into rows. Same fields, same codes, every snap, with ? allowed. Let the play-by-play fill a third of it; the film columns are where your chart knows more than any public file.
  • Score with a proper rule. The Brier score rewards being right and being honest about how sure you were; 0.25 is “I have nothing”, and the base rate and xpass are the bars to beat.
  • Calibration is how sure; resolution is how much. Read the calibration plot for which way to adjust, the spread of the dots for whether you’re learning anything, and never judge either from one game.
