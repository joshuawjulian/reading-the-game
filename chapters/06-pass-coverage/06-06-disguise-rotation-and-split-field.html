def nickel_vs_2x2(shell="two_high", safety_w=7.5):
    """11 personnel 2x2 against a nickel (4-2-5) defense. The Will apexes the left slot, the
    corners play 7 yards off, and two-high safeties sit 13 deep a few yards outside the hashes."""
    off = offense("gun_2x2")
    dfn = defense("nickel", shell, off)
    out = []
    for p in dfn:
        if p.key == "WILL":
            p = p.moved(d=5.0, w=-6.0)
        elif p.pos == "CB":
            p = p.moved(d=7.0, w=p.w + (0.8 if p.w > 0 else -0.8), label="CB")
        elif p.key in ("FS", "SS") and shell == "two_high":
            p = p.moved(d=13.0, w=safety_w if p.w > 0 else -safety_w)
        out.append(p)
    return off, out


def protect(p, rushers=("SDE", "SDT", "WDT", "WDE")):
    """Five linemen set, four rushers engage them: enough to show it is a pass, no more."""
    for k in ("LT", "LG", "C", "RG", "RT"):
        p.block(k, pts=[(-1.2, 0.0)])
    for k in rushers:
        p.rush(k, pts=[(-1.6, 0.0)], speed=2.0)


def trips(detached=False):
    """11 personnel 3x1: X alone to the left; Z (#1), H (#2) and the tight end Y (#3) to the right.
    attached: Y in-line as the end on the line (a "trey" set). detached: Y flexed off the ball as a
    third slot and Z steps up to be the end on the line, so seven are still on the line."""
    off = offense("gun_trips")
    if detached:   # a real third slot: about 4 yards outside the tackle, a step off the ball
        off = [p.moved(d=-1.5, w=7.0) if p.key == "Y" else p.moved(d=-0.7) if p.key == "Z" else p
               for p in off]
    return off


def trips_defense(off, fs_w=-6.5):
    """Two-high nickel against 3x1. The backside corner presses X with a one-yard inside shade."""
    dfn = defense("nickel", "two_high", off)
    moves = {"FS": (12.0, fs_w), "SS": (12.0, 9.0), "LCB": (1.5, -14.0), "RCB": (7.0, 17.5),
             "NB": (5.0, 10.5), "MIKE": (4.5, 3.0), "WILL": (4.5, -2.0)}
    dfn = [p.moved(d=moves[p.key][0], w=moves[p.key][1]) if p.key in moves else p for p in dfn]
    return [p.moved(label="CB") if p.pos == "CB" else p for p in dfn]


def number_receivers(p, d=-3.2, extra=None):
    """Label the trips receivers #1-#3 (outside in); `extra` adds a second label below, e.g. {"Y": "detached"}."""
    for key, txt in (("Z", "#1"), ("H", "#2"), ("Y", "#3")):
        pl = p.players[key]
        p.note(txt, at=(pl.d + d, pl.w))
        if extra and key in extra:
            p.note(extra[key], at=(pl.d + d - 1.8, pl.w - 1.0))


def no_numbers(ax):
    """Remove the painted yard-line numbers (they sit where wide receivers and safeties stand)."""
    for t in list(ax.texts):
        if t.get_alpha() == 0.55:
            t.remove()

def ruler(fld, play, widths=False, fs=6.5):
    """Depth labels (yards past the line of scrimmage) down the left edge of the field window, so
    the faint 5-yard lines can be read as a scale. widths=True also marks where the hash marks and
    the painted numbers are, along the bottom edge. Drawn on the Field's ax: gridiron has no ruler."""
    ax = fld.ax
    x_left = FIELD_W - fld.y1 + 0.35                     # plot x of the window's left edge
    for d in range(5, 41, 5):
        y = play.los + d
        if fld.x0 + 0.8 <= y <= fld.x1 - 0.8:
            ax.text(x_left, y, f"{d}" + (" yd" if d == 5 else ""), fontsize=fs, color=style.INK_2,
                    ha="left", va="center", zorder=12,
                    bbox=dict(boxstyle="round,pad=0.12", fc=style.SURFACE, ec="none", alpha=0.85))
    if widths:
        y = fld.x0 + 0.9
        for yb, txt in ((23.58, "hash"), (FIELD_W - 23.58, "hash"), (13.0, "numbers"), (FIELD_W - 13.0, "numbers")):
            if fld.y0 + 1.5 <= yb <= fld.y1 - 1.5:
                ax.text(FIELD_W - yb, y, txt, fontsize=fs, color=style.INK_2, ha="center", va="center",
                        zorder=12, style="italic",

def snapshot(play, t, ax=None, ghosts=(), ghost_t=None, lateral="full", window=(-8.5, 19.5),
             width_in=6.5, notes=(), show_ruler=False):
    """Everyone where he is at time t; `ghosts` also keep a hollow marker where they stood at
    `ghost_t` and a dashed trail from there, so the pre-snap picture survives in print."""
    pos = {k: play.position(k, t) for k in play.players}
    moved = [pl.moved(d=float(pos[k][0]), w=float(pos[k][1])) for k, pl in play.players.items()]
    q = Play([m for m in moved if m.side == "off"], [m for m in moved if m.side == "def"],
             los=play.los, ball_y=play.ball_y, to_go=play.to_go)
    q.highlights = dict(play.highlights)
    for n in notes:
        q.note(*n)
    fld = q.draw(ax=ax, lateral=lateral, window=window, width_in=width_in, title=False)
    ax, ppy = fld.ax, fld.pts_per_yard()
    no_numbers(ax)
    if show_ruler:
        ruler(fld, play, fs=5.5)
    P = lambda d, w: fld.xy(*play.to_abs(d, w))  # noqa: E731
    start = ghost_t if ghost_t is not None else play._compile()[1]
    for k, pl in play.players.items():
        if pl.side == "off" and pl.pos not in ("WR", "TE", "RB"):
            continue
        pts = np.array([play.position(k, tt) for tt in np.linspace(start, t, 30)])
        if np.ptp(pts[:, 0]) + np.ptp(pts[:, 1]) < 0.3:
            continue
        xs, ys = P(pts[:, 0], pts[:, 1])
        if pl.side == "off":
            ax.plot(xs, ys, color=style.OFFENSE, lw=1.0, alpha=0.45, zorder=3)
        else:
            ax.plot(xs, ys, color=style.DEFENSE, lw=1.3, ls=(0, (3, 2)), zorder=3)
    if ghost_t is not None and t > 0.05:
        for k in ghosts:
            x0, y0 = P(*play.position(k, ghost_t))
            ghost_marker(ax, float(x0), float(y0), play.players[k].label, ppy)
    bx, by = P(*play.ball_position(t))
    for kind, _, t_rel in getattr(play, "ball_events", []):
        if kind == "pass" and t > t_rel:          # a thrown ball: draw its flight so far
            sx, sy = P(*play.ball_position(t_rel))
            ax.add_patch(FancyArrowPatch((float(sx), float(sy)), (float(bx), float(by)), arrowstyle="-|>",
                                         mutation_scale=8, color=style.BALL, lw=1.4, ls=(0, (2, 1.5)), zorder=9))
    draw_marker(ax, float(bx), float(by), "ball", "", ppy)
    return fld


def strip(play, times, labels, ghosts=(), ghost_t=None, lateral=(-21, 21), window=(-8.5, 19.5),
          notes_at=None, width_in=6.5, ncols=2):
    """The print version of an animation: panels at chosen moments, ghosts kept on every
    post-snap panel so 'what was shown' and 'what was played' sit side by side."""
    n = len(times)
    nrows = int(np.ceil(n / ncols))
    pw = width_in / ncols
    h = nrows * (pw * (window[1] - window[0]) / abs(lateral[1] - lateral[0]) + 0.32)
    fig, axes = plt.subplots(nrows, ncols, figsize=(width_in, h), squeeze=False)
    for ax in axes.flat[n:]:
        ax.axis("off")
    for i, (ax, t, lab) in enumerate(zip(axes.flat, times, labels)):
        snapshot(play, t, ax=ax, ghosts=ghosts, ghost_t=ghost_t, lateral=lateral, window=window,
                 notes=(notes_at or {}).get(i, ()), show_ruler=(i == 0))
        ax.set_title(lab, fontsize=8.5, color=style.INK, loc="left")
    fig.tight_layout(h_pad=0.6, w_pad=0.4)
    h = [Patch(facecolor="none", edgecolor=style.DEFENSE, lw=1.2, ls=(0, (2, 1.2)),
               label="Dashed outline: where he stood before the snap (the disguise)"),
         Patch(facecolor=style.DEFENSE_TINT, edgecolor=style.DEFENSE, lw=1.3, label="Solid, filled: where he is now"),
         Line2D([], [], color=style.DEFENSE, lw=1.3, ls=(0, (3, 2)), label="Defender's path"),
         Line2D([], [], color=style.OFFENSE, lw=1.0, alpha=0.6, label="Receiver's path")]
    fig.legend(handles=h, loc="upper center", bbox_to_anchor=(0.5, 0.0), ncol=2, fontsize=7.5, frameon=False)
    return fig

Third-and-7 from midfield. As the offense breaks the huddle, the broadcast’s high camera shows two safeties standing 13 yards deep, one on each side of the field, with a wide-open patch of grass between them. The quarterback sees it too. Two deep safeties usually mean the deep middle is empty, so he plans to throw a deep in-cut right into that patch. He takes the snap, sets his feet, and lets it go, and a safety who had been standing 13 yards deep and 7 yards to the side is suddenly sitting at 11 yards, right in the throwing lane. Interception. On the replay the analyst circles both safeties and says “the coverage rotated.” What does that mean? Why did one safety sprint to the middle while the other one came forward? And if defenses can do that, why does anyone bother to read the defense before the snap? This chapter is about the gap between what a defense shows and what it plays, and about the last half-second in which that gap closes.

You’ll be able to…

  • Say what a defense is doing when it disguises, and name the price it pays for it.
  • Spot the common tells (safety depth and width, corner technique and leverage, weight and eyes, the timing of movement) and say before the snap which way a shell is likely to rotate.
  • Call the two big rotations on the replay (and, with practice, live from the wide shot): two-high turning into one-high (often with a robber) and one-high turning into Cover 2.
  • Recognise split-field coverage, read a Cover 6 call by its two halves, and explain how a defense answers a 3x1 formation with poach, solo or stubbie.
  • Explain how offenses fight disguise with motion, tempo and hard counts, and why late rotation forces the quarterback to read after the snap.

NoteYou’ll need

This chapter sits on top of the whole coverage part and a good deal before it. One line each:

  • Shells and the two families (A First Look at Defense): the shell is the safeties’ picture before the snap: one-high (one deep safety in the middle) or two-high (two deep safeties, one per side). Man means each defender has a receiver; zone means each defender has an area.
  • Formations (Spread Formations): 2x2 is two receivers on each side; 3x1 (trips) is three to one side and one to the other. Receivers are numbered from the sideline in: #1 is the widest on his side, then #2, then #3.
  • Motion (Motion and Shifts as Weapons): offenses move a player before the snap partly to make the defense react. A defender who travels with him suggests man; defenders who bump over suggest zone; a safety who spins down suggests one-high. That is the man/zone indicator.
  • Cadence and checks (The Pre-Snap Phase): the quarterback can change the play at the line, and a hard count (a fake snap call) tries to make the defense move early.
  • Zones and the shell read (Coverage Fundamentals): deep zones are thirds, halves or quarters. Underneath, from the sideline in, are the flat (near the sideline, 0–6 yards), the curl (10–12 yards, between the numbers and the hash), the hook (inside the curl, around the hash) and the hole (the middle, 10–15 yards, in front of the deep safety). A defender’s landmark is the spot his zone drop aims for. Press vs off corners and bail technique; MOFC (middle of the field closed, a deep safety in the middle) vs MOFO (open, no one there); pattern matching (zone defenders who pick up receivers who enter their area). The deep-middle spot is called the post (it is where a post route aims), so the safety who ends up there is the post safety, and “one safety in the post” means middle of the field closed. A corner in trail technique runs just behind and underneath his man, trusting a safety over the top.
  • Man coverage (Man Coverage): Cover 1 is man with one deep safety; Cover 0 is man with none; a robber is a safety who sits in the hole 10–12 yards deep to undercut in-breaking routes; a bracket puts two defenders on one receiver, one on each side.
  • Cover 3 (Cover 3): three deep, four under. It is usually reached by rotating a safety down: sky (safety to the flat), buzz (safety to the curl) or cloud (corner to the flat). The defender who covers both the curl and the flat on one side is the curl-flat defender; Cover 3 match is Cover 3 whose zone players pick up receivers by pattern.
  • Cover 2 (Cover 2 and Tampa 2): two deep halves, five under, with hard corners who squat in the flats; 2-Man is two deep with man underneath.
  • Quarters (Quarters and the Two-High Match Family): quarters gives four defenders a deep quarter each, with safeties who read #2 and fit the run; Cover 6 is quarters on one side and Cover 2 on the other; Palms is a two-high check in which the corner and safety on one side trade jobs on #1 and #2; MEG means a corner plays his man everywhere he goes.
  • Routes and leverage (Passing Fundamentals, The Quick Game): a seam runs straight up between the hash and the numbers, a post breaks deep toward the middle, a dig breaks flat inside at 10–15 yards, an out breaks to the sideline, a hitch stops and turns back, and a double move fakes one break and runs another. A defender’s leverage is which side of the receiver he stands on: inside leverage protects the middle, outside the sideline.
  • Pass concepts and the quarterback (Dropback Concepts, Quarterback Play): concepts like dagger are built to beat a particular coverage, so the quarterback picks a side and a progression (the order in which he looks at his receivers) partly from what he sees before the snap. When nothing is open he throws a short checkdown to a back.
  • Run fits (Run Fits and Box Math): the number of defenders near the line versus the number of blockers decides whether a run has a chance, and a rotating safety is often the extra run defender.
  • Field landmarks (The Game in Fifteen Minutes): the NFL hash marks sit only about 3 yards either side of the middle of the field, and the painted yard-line numbers start 12 yards in from each sideline, roughly 13 to 15 yards from the middle (sources in the split-field section). So “outside the hash” means 3 or more yards from the middle, and “on the numbers” means about 14 yards out, closer to the sideline than to the ball.
  • Who’s who (The Twenty-Two, Formation Rules and Vocabulary): the Mike is the middle linebacker, the Will the other (weak-side) linebacker in a nickel defense. On offense, X is the wide receiver on the line, Z the wide receiver off it, Y the tight end, and H the third wide receiver, usually in the slot. A dropback is a play on which the quarterback drops back to pass (sacks and scrambles included); coverage statistics are counted per dropback.

Why a defense would lie

Start with what the quarterback is trying to do before the snap. In the 20 or so seconds he spends at the line, he decides a lot: which way his linemen slide in pass protection, which defender he must throw away from if that defender blitzes, which side of the field his progression starts on, and whether the called play is any good against what he sees. If it isn’t, he can change it (The Pre-Snap Phase). Almost all of that depends on one question: what coverage is this? If he knows the coverage before the snap, he knows roughly where the open grass will be before anyone has moved. Then the play is half over before it starts.

The defense, meanwhile, has to put eleven people somewhere. And where they stand leaks information, because of plain physics. A safety can only cover ground so fast. If he is going to guard the deep middle of the field, he has to stand somewhere from which he can get there before the ball does. If a cornerback is going to cover one receiver man-to-man, he tends to line up facing that receiver, close to him, on the side he wants to protect. A defense that simply lined up where its call works best would be handing the quarterback its call sheet.

Disguise is the answer: lining up in a way that hides or misrepresents the coverage the defense will actually play. It comes in three flavours, and good defenses use all of them:

  1. Show one thing, play another. Line up in a two-high shell and play a one-high coverage, or the reverse. This is the dramatic kind, and the next three sections are about it.
  2. Show the same thing every time. Line up in one shell for nearly everything, and play many coverages out of it. The quarterback isn’t fooled so much as left without information. This is the quieter kind, and it is how the two-high defenses that spread through the NFL after 2018 live. Case Study: Fangio’s Two-High Revolution and Macdonald’s Answer tells that story in full.
  3. Move at the right moment. Hold a picture until the last instant and change it at the snap, so there’s no time to react. We’ll call that late rotation below.
ImportantWhy it exists

Disguise is about time. A quarterback who knows the coverage before the snap can decide where to throw before the pass rush has started. A quarterback who must work it out after the snap spends part of his roughly 2.5 to 3 seconds of protection just finding out what he is looking at. The defense isn’t trying to make the quarterback wrong so much as late. A late quarterback holds the ball, takes sacks, throws into windows that close, or settles for a short checkdown.

There’s an asymmetry in the rules that makes this possible. The offense must be set: after a shift, all eleven players have to be still for a full second before the snap, and only one player may be in motion when the ball is snapped, and not toward the line. The defense has no such rule. Defenders may move, shift, walk up and back out as much as they like before the snap, as long as they stay out of the neutral zone (the strip, one ball long, between the two teams) and don’t use words or acts meant to disconcert the offense at the snap, such as imitating its snap count.1 The offense has to hold still and be read, while the defense can keep moving.

What disguise costs

  • The real job starts from the wrong spot. A safety who shows two-high but is really the robber has to travel to his real landmark; one who shows one-high in the box and is really a deep-half player has a long way to run. Every yard of travel is time out of position.
  • The run defense gets worse. A nickel defense usually has six or seven players near the line (four linemen, two linebackers, sometimes the nickel), the box. A safety who comes down makes it seven or eight, but a safety standing 13 yards deep who is supposed to be that extra run defender arrives late. A safety walked down to look like a run defender who is really bailing to a deep half has left the box one man short. Offenses that run the ball well punish disguise by simply handing it off (Run Fits and Box Math).
  • Communication and mistakes. Every rotation is a little play inside the play, with its own timing and its own way to go wrong. When one safety rotates and the other doesn’t, the defense ends up with two players in one zone and nobody in another: the “bust” you see when a receiver runs untouched through the middle of the field.
  • Practice time. A defense that can play five coverages from one look has to rehearse all five from that look, and every extra starting look multiplies the reps. Something has to give. Some defenses keep one look and many coverages (flavour 2 above); others keep a short list of coverages and practise reaching each one from several looks (flavours 1 and 3). Few can afford both.
NoteMadden vs. real life

In a video game the defense picks a play from a menu, and each defender walks to the spot that play’s diagram gives him. What you see is close to what was called. In the NFL, alignment is a separate decision from the call. Coordinators design their calls so that several coverages start from the same picture, and coach players to hold that picture until a set moment. Real quarterbacks are taught to treat the pre-snap shell as a hypothesis, a best guess they expect to confirm or overturn in the first second after the snap.

The physics of a disguise: what leaks before the snap

If you can’t hide where you’re going, you can at least make it hard to see. But the distances involved put limits on what a defense can fake, and those limits are exactly what a good quarterback (and now you) can read.

Some rough numbers. A deep pass is usually released around 3 seconds after the snap (in 2025 the middle half of throws travelling 20 or more yards left between 2.6 and 3.6 seconds).2 But the safety’s real deadline is earlier: he has to be on top of the receiver before the receiver runs past him, and in this chapter’s simulated plays a slot receiver running up the seam is 15 yards downfield about two seconds after the snap. The safety also can’t sprint at the snap. His first step or two are a read (is it a pass? where are the quarterback’s eyes?), and a backpedal covers ground noticeably more slowly than turning and running. The fastest safeties top out around 22 miles an hour, roughly 10 yards a second, but a safety reading the quarterback is rarely anywhere near that.3

So give him a working speed of 5 to 6.5 yards a second once he goes (the figure the simulated plays use) and a few tenths of a second to read. That leaves about 1.6 seconds of running before the seam receiver arrives: roughly 8 to 10 yards. A safety 13 yards deep and 7.5 yards from the middle (a few yards outside the hash) is about 8 yards from the deep-middle landmark at 16 or 17 yards: inside the budget. One standing on the numbers, about 14 yards from the middle, is about 14 yards away: far outside it.

Each defender’s real job therefore draws a circle around his real landmark. He can line up anywhere inside it, and the disguise has to fit inside every defender’s circle at once. Within the circle there is still a trade. A safety who will really be a robber at 10 or 11 yards can line up at 13 and still get there, which keeps the two-high picture intact, but the trip costs him time. Lining up at 11 makes his job easy, and it is also a tell. Every disguise is that bargain, made defender by defender: the closer to the real job, the easier the job and the louder the tell. (These are model numbers for teaching the idea, not measurements; with the real tracking data in Tracking Data and the Big Data Bowl you can measure the circles.)

That is why disguises leak. Here is what to look at, and why each one tells you something. The figure shows the textbook two-high spots as faint outlines so you can see how far each defender has cheated from them.

  • Safety depth. Deep-zone safeties in a two-high shell usually stand 12–14 yards off the ball. A safety at 8–10 yards is probably either coming down (into the box, the flat or the hole as a robber) or about to sprint backward, and the second is hard to do well. Why it tells: the deeper he is, the fewer underneath jobs he can reach in time; the shallower, the fewer deep ones.
  • Safety width. A safety within about 8 yards of the middle (on the hash or a few yards outside it) can reach the deep middle. One splitting the hash and the numbers, 9 to 11 yards out, is probably staying on his half of the field. When one safety pinches toward the middle and the other widens or creeps down, the shell is leaning toward one-high. Why: the deep middle is a destination, and the safety who will go there likes to start close to it.
  • Corner technique, then leverage. Two different questions. Technique (stance, eyes, depth) says man or zone: a man corner is square to his receiver with his eyes on him; a zone corner is turned toward the quarterback, seven or eight yards off and ready to backpedal or turn and run to a deep third, or five or six yards off and flat-footed (a squat Cover 2 corner). Leverage says where his help is. A corner who gives up the inside expects a safety or robber there (Cover 1, Cover 3). One who takes away the inside expects help over the top or has none (2-Man, Cover 0). Why it tells: the corner’s technique has to match his job, and technique is hard to fake. Corners can disguise too: a corner pressed in a two-high look who bails at the snap, one who shows deep-third depth and comes up into the flat (the cloud rotation), or a corner and safety who trade jobs (an invert: the safety comes down to the flat and the corner bails to the deep half, as Palms does on one side). They have less room to lie than safeties do, but an invert means the deep player can be a corner, so on the replay watch the two deepest defenders, whoever they are.
  • Eyes and weight. Zone defenders look at the quarterback. Man defenders look at their man. A safety leaning forward on his toes is thinking about coming downhill; one sitting on his heels is thinking about getting depth. This is the hardest tell to see on a broadcast, but the All-22 film coaches use (a high, wide camera showing all 22 players) shows it.
  • Timing. When does anyone move? A safety who drifts toward his real spot during the cadence (coaches say he creeps or walks down) is giving away his rotation; one who stands still until the ball moves is not. That discipline has a name, late rotation, and its own section below.
  • Situation and tendency. A defense that always rotates to the field on third-and-long has a habit opponents will have found on film (Tendencies). The best disguises also break the defense’s own patterns.
CautionCommon misconception

“Two high safeties means Cover 2.” It doesn’t. Two-high is a shell, a picture, not a coverage. From the same two-high look a defense can play Cover 2, 2-Man, quarters, Cover 6 or Palms, or rotate into Cover 3 or Cover 1 at the snap. What the shell tells you is which coverages are easy from here, and so which ones are likely. To name the coverage, look again one second after the snap.

Post-snap rotation, part 1: two-high becomes one-high

Post-snap rotation is defenders (almost always the safeties) moving at the snap from the shell they showed into a different one. The most common version starts two-high and ends one-high: one safety races to the deep middle and the other comes down. Where the down safety lands decides the coverage. If he goes to the flat, it’s a sky rotation into Cover 3. To the curl area, a buzz rotation. To the hole in the middle of the field at 10 or 11 yards, it’s Cover 3 or Cover 1 with a robber. In Cover 3 robber the robber is one of the four underneath defenders: the two curl-flat players and the hook player keep their zones and the robber takes the hole in front of the deep safety. (You’ll also hear rat: some staffs use it for any robber, others only for a linebacker sitting in the low hole at 5 to 8 yards, as in “Cover 1 rat”.) (Cover 3 built all three from a one-high start; here they start from two.)

Coaches describe all this movement with a pile of short verbs, and some of them mean different things in different mouths. This chapter uses them like this:

Follow the play from the quarterback’s side. Before the snap he sees two-high and an empty deep middle. Against two-high, the space between the safeties and behind the linebackers is often where the ball should go, and dagger (a seam route that pulls the underneath and deep defenders away, with a deep dig coming in behind it; see Dropback Concepts) is built to hit it. Here Y’s seam bends slightly inside to push the middle open. At the snap, the defense changes the answer. The free safety’s sprint to the middle puts a defender over the top of the seam. The strong safety’s short trip down and in puts him exactly where the dig is going to arrive. The quarterback who decided before the snap, and throws on time without looking, throws it to the robber.

Notice the distances. The free safety travels about 8 yards to the deep middle, inside the 8-to-10 yard circle from the previous section. That’s possible only because he lined up 7 or 8 yards from the middle, not on the numbers. The strong safety travels about 4 yards. Neither had far to go, which is the whole design. The shell was two-high, but each safety was standing where his real job was still within reach.

The same rotation can land in man coverage instead: Cover 1 robber. The safeties move exactly as they do here, so the difference is in the corners. In Cover 3 they turn and run to deep thirds with their eyes on the quarterback; in Cover 1 they stay on their receivers, pressing or trailing them wherever they go. One second after the snap, the corners tell you which.

What beats the robber, once the quarterback sees him? The throws he left behind. He has come down from one side, so the seam and the curl on that side are thinner, and his spot in the hole is in front of the deep safety, so a route that gets behind him (a post or deep crosser) is open if the post safety is late. The rest of Cover 3’s weaknesses are still there too: the seams between the deep middle and each deep third, and four underneath defenders who can’t cover every short zone, so stress the curl-flat defender on the far side with a route in front of him and one behind. Concepts Versus Coverages builds the quarterback’s side of this.

ImportantWhy it exists: two-high to one-high

What it solves. It gets an extra defender near the line against the run, or a robber against in-breaking routes, without announcing either before the snap. Offenses facing two-high tend to run the ball, because with both safeties 12 or more yards deep only six or seven defenders are near the line to face the offense’s blockers, and to throw between the safeties. The rotation punishes both: the down safety adds a run defender or sits in the throwing lane.

What it gives up. The rotating safety starts late and from deep, so he arrives in the box after the run has started. The safety spinning to the middle has a long run, so a quick deep throw outside him (a post or a deep corner route released before he gets there) can beat it. And the robber looks only at the quarterback, so a quarterback who looks him off with his eyes and shoulders, then throws elsewhere, can move him out of the play (Concepts Versus Coverages).

Which safety comes down? A rule, not a whim

A defense doesn’t decide on the field which safety rotates. The call says so, usually by tying the rotation to something the offense shows: rotate to the passing strength (the side with more receivers), to the field (the wide side), toward motion, or the other way, weak or to the boundary. Staffs build names for these out of a direction and a landing spot (“sky strong,” “buzz weak” and the like); one of Nick Saban’s trips calls, for example, drops the safety down to the passing strength.4 Before the snap, then, your best guess about direction is:

  • Toward the strength or the motion is the default. That is where the extra receivers are, so that is where an extra underneath defender helps most. Quarterbacks are taught to expect it.
  • Away from the strength is itself a disguise. A defense that knows the quarterback expects the strong safety to come down can spin the weak safety down instead, so the robber appears on the side the quarterback thought was clean.
  • Look for the tells on one safety, not both. The safety who is shallower, narrower or leaning forward is the likely down safety; the other one is the likely post safety. Combine that with the strength of the formation and you can usually call the direction.

The corners, instead of bailing to deep thirds, hold their depth and widen into the flats, the Cover 2 corner’s job. With a safety in the deep middle and the corners backing off, the outside is where the quarterback looks: quick outs and hitches in front of the bailing corners, curls under them, back-shoulder throws to receivers facing one-on-one coverage. What opens instead is the deep middle, between the two safeties, which is exactly what the quarterback ruled out before the snap. A quarterback who reads the safeties after the snap sees them split and throws between them, up the seam or on a post (the throw that Cover 2 and Tampa 2 shows the Tampa 2 Mike running to carry). ImportantWhy it exists: one-high to two-high

What it solves. It lets a defense look like it is loading up against the run, with the down safety near the box, and still play two deep defenders against the pass. Offenses that see one-high often check to throws outside or to deep shots against single (one-on-one) coverage. The bail punishes both.

What it gives up. The down safety was a run defender in the picture, and now he’s running away from the line, so a run against this look faces one fewer defender than it seemed to. He also has the longest trip on the field, 8 or more yards of depth plus width, so a fast vertical route outside can reach his area before he does.

Late rotation: winning the last half-second

The two plays above have something in common. In both, nothing moves until the snap. That is the difference between a rotation the quarterback can see and one he can’t.

Late rotation means timing the rotation to start at the snap or just after it, rather than during the cadence. A defense that rotates early (safeties drifting into place while the quarterback calls signals) gives him a free look: he sees the real coverage, still has time to change the play, and the disguise has bought nothing. A defense that rotates late shows him the real coverage only after the play has begun. He has to recognise it while dropping back, with the pass rush coming.

It is also why late rotation is risky. A safety who starts moving at the snap is a step behind one who was already there, so the rotation has to be fast and the distances have to be short. Every late rotation is a race between the safety and the first throw.

The clearest way to see a rotation is to track just the two safeties, how deep each one is and how far from the middle, frame by frame from three seconds before the snap to two seconds after. Player-tracking data records exactly that: every player’s position ten times a second, from chips in the shoulder pads. def _seg(p, key, start, end, tau, speed):
    """Move `key` from start to end at `speed`, starting at time tau (negative = before the snap)."""
    start, end = np.array(start, float), np.array(end, float)
    L = float(np.linalg.norm(end - start))
    if tau < 0:                                  # part of the trip happens before the snap
        pre = min(L, speed * -tau)
        mid = start + (end - start) * (pre / L if L else 0)
        p.motion(key, to=tuple(mid), speed=max(pre / -tau, 0.05))
        if L - pre > 0.05:
            p.drop(key, tuple(end), speed=speed)
    else:
        p.drop(key, tuple(end), speed=speed, delay=tau)


def sim_play(kind, rng, tau=None, play_id=0):
    """One simulated dropback, returned as Big Data Bowl-style tracking rows (10 frames a second).

    kind: 'stay2' (two-high shown and played), 'rot1' (two-high shown, rotated to one-high),
          'stay1' (one-high shown and played), 'spin2' (one-high shown, bailed to two-high).
    tau : when the safeties start their post-snap move (seconds; negative = before the snap).
    """
    off, dfn = nickel_vs_2x2()
    side = rng.choice([-1, 1])
    if kind in ("stay2", "rot1"):
        d1, d2 = rng.uniform(11.5, 14.0, 2)
        w1, w2 = rng.uniform(5.5, 8.5, 2)
        fs0, ss0 = (d1, -side * w1), (d2, side * w2)
    else:
        fs0 = (rng.uniform(13.0, 15.5), rng.uniform(-1.5, 1.5))
        ss0 = (rng.uniform(7.0, 10.0), side * rng.uniform(4.0, 7.0))
    dfn = [q.moved(d=fs0[0], w=fs0[1]) if q.key == "FS" else q for q in dfn]
    dfn = [q.moved(d=ss0[0], w=ss0[1]) if q.key == "SS" else q for q in dfn]
    p = Play(off, dfn, los=35, to_go=10)
    p.motion("MIKE", to=(4.5, 1.5), speed=0.1)   # a slow 3-second shuffle so frames start 3 s early
    p.route("X", r.go(22)); p.route("Z", r.go(22)); p.route("H", r.seam(20)); p.route("Y", r.seam(20))
    p.dropback(2.5)
    tau = rng.uniform(-0.9, 0.3) if tau is None else tau
    v = rng.uniform(5.0, 6.5)
    if kind == "stay2":
        _seg(p, "FS", fs0, (fs0[0] + rng.uniform(2, 4), fs0[1] - side * rng.uniform(0.5, 2.5)), max(tau, 0), v * .6)
        _seg(p, "SS", ss0, (ss0[0] + rng.uniform(2, 4), ss0[1] + side * rng.uniform(0.5, 2.5)), max(tau, 0), v * .6)
    elif kind == "rot1":
        _seg(p, "FS", fs0, (rng.uniform(15.5, 18.0), rng.uniform(-1.0, 1.0)), tau, v)
        _seg(p, "SS", ss0, (rng.uniform(8.0, 10.5), side * rng.uniform(3.0, 6.0)), tau, v * 0.8)
    elif kind == "stay1":
        _seg(p, "FS", fs0, (fs0[0] + rng.uniform(2, 4), fs0[1]), max(tau, 0), v * 0.6)
        _seg(p, "SS", ss0, (ss0[0] + rng.uniform(-1, 1), ss0[1] + side * rng.uniform(1, 4)), max(tau, 0), v * 0.6)
    else:
        _seg(p, "FS", fs0, (rng.uniform(15.5, 18.0), -side * rng.uniform(9.0, 12.0)), tau, v)
        _seg(p, "SS", ss0, (rng.uniform(15.5, 18.0), side * rng.uniform(9.0, 12.0)), tau, v)
    trk = p.to_tracking(fps=FPS, play_id=play_id)   # columns: gameId, playId, nflId, frameId, x, y, s, a, ...
    trk["kind"], trk["los"], trk["ball_y"] = kind, p.los, p.ball_y
    return trk


def shell_at(frame, deep=10.0, split=3.5):
    """Classify one frame of any BDB-style play: 'two-high' if two or more defenders are 10+ yards
    deep and none of them is within 3.5 yards (sideways) of the ball; otherwise 'one-high'.
    Meant for pre-snap frames only: after the snap, a corner bailing past 10 yards would count."""
    dfn = frame[frame["side"] == "def"]
    d = dfn["x"] - frame["los"].iloc[0]
    w = (frame["ball_y"].iloc[0] - dfn["y"]).abs()
    deep_w = w[d >= deep]
    return "two-high" if len(deep_w) >= 2 and deep_w.min() >= split else "one-high"

Read it as a picture of the idea, not as evidence: the simulation only plays back the timings we gave it. What it shows is the shape to look for. A rotation is a divergence: before it starts, the rotated plays and the honest plays are indistinguishable, and within about ten frames (one second) the safeties are in completely different places. When the divergence starts before the snap, even by half a second, the quarterback still has time to use it: he can change the protection, check to another play, or decide which side his progression starts on. A defense coached for late rotation pushes that starting point to zero or past it. Split-field coverage: two halves, two answers

So far the whole defense has rotated together. A different idea, and one of the most important in modern defense, is to not treat the field as one problem at all.

Split-field coverage means playing a different coverage on each half of the field, with each half following its own rules for the receivers on its side. The split is usually made one of two ways: by formation (the side with more receivers, the strong side, against the side with fewer) or by field position (the field side, the wider side of the field when the ball sits on a hash mark, against the boundary side, the shorter one).

You have already met the most common split-field coverage: Cover 6, quarters on one half and Cover 2 on the other. Here it is with the split made by field position: the ball is on the left hash, so the field (wide) side is to the offense’s right.

Why split the field? Because the two halves really are different problems.

  • The field and the boundary are different sizes, a little. NFL hash marks sit 70 feet 9 inches from each sideline, so with the ball on a hash there are about 23.6 yards to the near (boundary) sideline and 29.7 to the far (field) one, a difference of about 6 yards. In college the hashes are 60 feet from the sidelines, so the split is about 20 yards against 33.5 A boundary corner has the sideline as an extra defender; he can squat in the flat because a throw over his head has little room to land. That difference is big in college, which is where field/boundary quarters grew up. In the NFL it is modest, which is one reason pro split-field calls are often keyed to the formation’s strength (the trips side, the tight-end side) rather than to the field.
  • The two halves face different receivers. A tight end and a slot to one side, a lone wideout to the other: those sides threaten different things, and one coverage rule can’t be ideal for both.
  • Each player’s job is simpler. Each player only has to know his own half: who is #1 and #2 on my side, and what is my rule against them? The defense gets flexibility from combining simple halves, not from teaching eleven people one complicated call. The defenders in the middle, the Mike and the hook players, belong to whichever half the call assigns them; in the figure above the Mike takes the hook in the middle, the Will the curl for the Cover 2 half and the nickel the curl-flat for the quarters half.
  • But the communication is harder. Both halves have to agree where the split is: by formation, by field, or, with the ball in the middle and a balanced formation, by a default the staff sets in advance. A safety or the Mike typically makes that strength call by shout and hand signal. When a receiver motions across the formation, the strength can flip, and both halves must re-declare in the last second before the snap. That is a big reason offenses motion so much against split-field defenses (see below).

Split-field coverage is also a disguise machine, and that’s why it belongs in this chapter. From one two-high alignment the defense can play quarters on both halves, Cover 2 on both, Cover 6 to either side, Palms on one side, or rotate to one-high. The quarterback now has to read two coverages, one per half. A pre-snap guess that is right about the left half can be wrong about the right.

How do you tell Cover 6 from its neighbours after the snap? Count the deep safeties first, then watch the corners. Cover 3: one safety in the deep middle, both corners bail. Cover 2: two safeties split the field, both corners sit down in the flats. Cover 6: two safeties split the field, and the corners disagree: one sits in the flat (the Cover 2 half), the other backpedals with his eyes on the quarterback and plays a deep quarter. One squatting corner and one bailing corner under two deep safeties is the Cover 6 picture.

One naming warning before the history. Coverage numbers are dialect, not law (Coverage Fundamentals and the dialect table in the coverage atlas), and “Cover 7” is the clearest example. In Nick Saban’s vocabulary, Cover 7 names a whole family: a two-high, quarters-based pattern-match coverage in which each side of the formation gets its own small call, so the full call is a split-field call assembled from parts.6 Other staffs use “Cover 7” for a man-match coverage, a bracket on one receiver, or nothing at all.

A short history: from college laboratories to the NFL

Split-field quarters grew up in college football in the 1990s and 2000s, and the problem it solved is easy to state. College offenses spread the field and ran option and play-action from the same looks, on wide college hashes. A defense that put a safety in the box to stop the run gave up the deep ball; one that kept both safeties deep got run over. Quarters let it have both: safeties at 10 to 12 yards who read the receivers in front of them and fit the run if it came, but stayed deep if it was a pass. TCU’s Gary Patterson, one of the coaches most identified with the approach, summed up the safeties’ job at a 2015 coaches’ clinic: “Don’t go till you know.” His safeties didn’t backpedal at the snap; they shuffled flat-footed and read, because, as he told the same clinic audience, “we want to take away the short game, we want to take away combination routes, and we want to stop the run.”7

Saban’s pattern-matching quarters, which goes back to his years as Bill Belichick’s defensive coordinator in Cleveland (1991 to 1994), went with him to Michigan State, LSU and Alabama. Michigan State under Mark Dantonio and Pat Narduzzi, Patterson’s TCU and Iowa’s long-running quarters defenses became reference points that other coaches studied at clinics.8

The NFL caught up later, and fast. NFL Next Gen Stats (NGS, the league’s tracking-data arm) measures how often defenses play a split-safety (two-high) coverage after the snap: 32.9% of dropbacks in 2018, 37.8% in 2021, 40.4% in 2024 and 42.0% in 2025, the highest since tracking began. In 2025, Cover 6 alone was 9.2% of coverages.9 Brandon Staley’s 2020 Rams (below) are one of the teams that made the approach fashionable in the league. Staley came from the staff of Vic Fangio, the defensive coordinator most associated with the NFL’s two-high wave, and coaches from Staley’s staff and Fangio’s wider tree, such as Ejiro Evero (the Rams’ safeties coach in 2020, then Denver’s defensive coordinator in 2022 and Carolina’s since 2023), have leaned on rotation from a two-high shell: by one count, Carolina stayed two-high and rotated to one-high in almost equal measure.10 The full story of the two-high spread, Vic Fangio and the counterpunch that followed, is told in Case Study: Fangio’s Two-High Revolution and Macdonald’s Answer and The Modern Era.

Trips: the formation that forces a decision

Split-field rules work beautifully against 2x2: two receivers and two or three defenders per side. A 3x1 formation (trips) breaks the symmetry on purpose. Three receivers go to one side and one to the other, and now the count is ugly.

Count only the players who can run with a receiver 20 yards downfield. On the trips side the defense has the corner, the safety and the nickel against three receivers who can all go deep. On the backside it has a corner and a safety against one receiver. (The backside linebacker, the Will, is there too, but his jobs are the run and a short zone; he can’t chase a receiver up the field.) If the defense just plays quarters as usual, the trouble is the third receiver (#3, the one closest to the ball). Quarters rules are built for two receivers per side: the corner takes #1 deep and the safety reads #2. Most systems do give #3 to someone underneath, the hook or apex player who walls him off (stays in his path and keeps him from running free across the middle). But if #2 and #3 both run vertical, the safety can carry only one of them, and the underneath defender can’t carry a fast receiver 20 yards deep. Meanwhile two defenders on the backside guard one man. Offenses love this: send #2 and #3 vertical and see who covers the second one. The problem is sharpest when #3 is a detached receiver, a slot or a tight end split off the line, who can get up the field fast; an in-line tight end is slower to release, and many quarters teams just play their normal rules against him.

So every two-high defense carries a set of 3x1 checks: pre-planned adjustments that tell each player what to do when trips appears. They fall into a few families. All of them answer the same question: who takes #3 if he goes deep, and who gets left alone because of it?

Poach: the backside safety comes across

Poach is the most aggressive answer. The backside safety ignores the lone receiver, leaves him to the cornerback in pure man coverage, and works across the field to take #3 if #3 runs deep. The trips side can then play its normal quarters rules on #1 and #2.

Poach is a read, not a man assignment. If #3 goes vertical, the poaching safety carries him. If #3 stays short or breaks outside, the safety doesn’t stand around: he robs inside, jumping #2’s in-breaking route or the backside dig, or works back to help on X. That is what separates poach from a backside safety simply locked on #3 in man coverage.

The trade is stated in the picture. Poach turns the trips side into a fair fight and gives the defense a deep player over #3, but the corner on the backside is now on an island. Offenses who expect poach build “poach beaters”: they put their best receiver alone on the backside (often called an X-iso look) and attack him knowing no safety is coming, or they attack the poaching safety himself with a route that breaks behind him as he runs to #3.11 The poaching safety’s alignment is also a tell: a backside safety who starts closer to the middle than usual, or who leans toward the trips side, is probably not staying home.

Solo and stubbie: other ways to solve the count

The 3x1 check (solo / stubbie) family covers the other common answers. The names are shared across many playbooks but the details aren’t, so treat these as the typical versions rather than the only ones.

Solo. A warning first: if you search for “solo coverage”, most of what you’ll find, from pro and big-college quarters writers, uses Solo for the backside corner alone on X while the backside safety takes #3 vertical. That is the call this chapter has just shown you as poach. The version here comes from other playbooks, and we keep it under the name solo because it is a genuinely different answer, with a different defender on #3: the backside corner plays the single receiver by himself, and a linebacker, usually the Mike, takes #3 if he runs deep. That frees the backside safety to be an extra run defender against trips runs and run-pass options.12 The trade: a linebacker running with a tight end or slot up the seam is a mismatch the offense will look for, which is why NFL defenses rarely ask a linebacker to carry a detached #3 deep. Solo shows up mostly against tight-end trips, where #3 is in-line and slower to get going.

Rotate. The defense can also give up on two-high against trips and rotate to a one-high, three-deep Cover 3 match coverage, with the trips-side safety coming down toward the passing strength and the backside safety becoming the deep-middle player.

Why does this matter to a viewer? Because trips is how offenses force the defense to declare. The 3x1 check the defense uses usually shows up in the backside safety’s alignment before the snap: home and wide (stubbie), cheated in toward the middle (poach, or a rotation in which he becomes the post safety), or creeping down (solo). Poach and rotation can look the same before the snap; the difference shows in the first second after it. The poaching safety keeps reading #3 and goes where #3 goes; the rotating safety runs to the deep middle and stays there, while the trips-side safety comes down. And offenses use motion to create trips at the last moment, so the defense has to make that decision with a second left on the play clock. Usually a safety or the Mike shouts and signals the check. A defender who doesn’t hear it plays the old call, and that is how the busts described at the start of the chapter happen: two defenders on one receiver, nobody on #3.

The same question comes up against other three-receiver sides. An empty 3x2 set has a trips side too, and the defense answers it with the same families. A bunch, three receivers packed tightly together, adds a different problem, receivers crossing each other at the line, which Man Coverage handles with its banjo calls.

How offenses fight back

Offenses’ counters share one idea: make the defense show its hand earlier, or make the hand not matter.

Motion. Moving a receiver before the snap forces defenders to adjust, and adjustments are information (Motion and Shifts as Weapons). A cornerback who travels across the formation with the motion man hints at man coverage. A safety who spins down hints at one-high. Motion also changes the receiver count: turning 2x2 into 3x1 forces the defense into its trips check with no time to disguise it. The league has gone all in. PFF (Pro Football Focus, a charting and grading company) counted pre-snap shifts or motion on 63.9% of plays in 2025, up from 43.0% in 2018.13

Defenses answer motion in three ways:

  • Match rules. In a pattern-match zone, defenders hand receivers off by rule, so a defender may follow the motion man a few steps (looking like man) and then pass him to the next zone. The motion read says “man” and the coverage is zone, or the reverse.
  • Fake spins. A safety spins down with the motion on purpose and goes back to his old job at the snap.
  • Many starting points. The defense rehearses every rotation from more than one pre-snap picture, so no single picture means one coverage.

The offense’s counter to all three is motion at the snap: the receiver is still moving when the ball is snapped, so the defense must react to him and rotate at the same instant, with no time for a second move.

Tempo. Snapping the ball quickly gives the defense less time to line up, communicate and disguise. Under tempo (the pace at which an offense gets to the line and snaps the ball; see In-Game Adjustments and Tempo), defenses often fall back on one simple call, and simple calls are easier to read.

Hard counts. A hard count isn’t only fishing for an offside flag. Safeties drilled to move “on the snap” sometimes move on what sounds like the snap, and a safety who flinches at a fake reveals where he’s going. The defense’s answer is discipline: key the ball, not the voice.

Read it after the snap. The deepest answer is for the quarterback to stop relying on the pre-snap picture. Modern quarterbacks are taught to form a pre-snap guess, then confirm it in the first steps of the drop by looking at the safeties: is the middle of the field closed or open now? Which safety moved, and where? Pass concepts are built to have an answer against either rotation. The classic example is a slot receiver on a “bender” in four verticals: if the middle opens (two safeties split), he bends his seam into a post between them (see four verticals), as Y does in the Cover 2 figure above; if the middle closes (one safety in the post), he stays on his seam and the ball goes away from the post safety. The route changes after the snap, on the same read the quarterback makes. Eye manipulation, looking off a robber to move him, is the advanced version (Concepts Versus Coverages).

Play-action and RPOs. The safety who rotates down is usually the defense’s extra run defender, and that makes him a target. A run-pass option can read him: if he comes down to fit the run, throw where he left; if he stays deep, hand off into a box that is one defender short. Play-action attacks the other safety: a run fake holds the post safety’s eyes for a beat, and his long trip to the middle is exactly the kind of run a deep shot behind him is built to beat. Rotation buys the defense a run defender it never announced; a good run fake makes it pay for him.

Throw before it matters. The quick game (throws out in about two seconds or less; The Quick Game) doesn’t care much about rotations, because the ball is gone before the safeties have finished moving. Against heavy disguise, a steady diet of quick throws is a way of declining the contest.

Measuring the lie

Can you count how often a defense lies? Broadcasters and charting services label the coverage after the snap, and the nflverse data used throughout this course carries those labels (2018–2025). What they don’t record is the picture the defense showed before the snap. For that you need tracking data: the safeties’ positions in the frames before the snap.

That is exactly what the 2025 Big Data Bowl asked for. Its theme was pre-snap behaviour, and it was the first edition whose tracking files included full pre-snap frames, from the first nine weeks of the 2022 season. The winning entry, by NYU students Smit Bajaj and Vishakh Sandwar, was called “Exposing Coverage Tells in the Pre-Snap.”14 NFL Next Gen Stats then built a pre-snap model of its own, crediting that project as inspiration. It classifies the shell a quarterback sees before the snap and compares it with the coverage after. In its words: “If a defense shows split-safety zone and rotates to single-high man (or spins to Cover 0), we flag it as disguised.”15 The article that introduced the model gave no league-wide disguise rate, we found no other published one, and the Big Data Bowl files cover only half of one season. So “how often does an NFL defense lie?” has no public league-wide answer yet (one analyst’s count for a single team is in the Film room below).

Here is how you would get one. First fix what “the shell” means: the picture once the offense is set and the quarterback starts his cadence, say two seconds before the snap. Then classify that picture with a simple rule (two or more defenders at least 10 yards deep and none of them near the middle means two-high), do the same at the snap, and compare both with the coverage the defense actually played. A defense that rotates during the cadence has still disguised, but the quarterback had time to see the truth; one that rotates at the snap did not give him that time.

NoteFilm room: Rams 2020 — Brandon Staley’s first year

Brandon Staley arrived in Los Angeles in 2020 as a first-time NFL defensive coordinator, from Vic Fangio’s staff in Denver, and took over a defense with two great individual players: Aaron Donald inside and Jalen Ramsey at cornerback. He built it around a two-high shell. Most snaps looked alike before the snap, and the coverage was decided at and after it: quarters and Cover 6 to the field or the boundary, rotations down into Cover 3 or a robber, and Ramsey moved around the formation, from outside corner to the nickel (“star”) spot, to follow the matchup.18 Three things made the light-box, two-high approach affordable. Donald’s four-man pass rush meant the Rams didn’t need to blitz more than an average defense (they sent five or more rushers on about a quarter of dropbacks, close to the league median). The front, with Donald inside, could hold up against the run with fewer defenders in the box, while the safeties fitted the run from depth (Run Fits and Box Math). And the defense was disciplined enough to rotate late.

What to notice: on a 2020 Rams passing down, find Ramsey first (outside or in the slot), then the two safeties: level and deep while the quarterback calls signals, with the rotation, if there is one, coming only as the ball moves. The results: the 2020 Rams finished the regular season first in the NFL in points allowed (18.6 a game), total yards allowed (281.9) and passing yards allowed (190.7).19 Staley was hired as the Chargers’ head coach after one season, and the Rams’ approach spread through his coaching tree and the league. Its parent, Fangio’s system, is the subject of Case Study: Fangio’s Two-High Revolution and Macdonald’s Answer.

NoteFilm room: Panthers under Ejiro Evero — the coin-flip rotation

Ejiro Evero, a safeties coach on Staley’s 2020 Rams staff, has run Carolina’s defense since 2023. His Panthers are the cleanest public example of this chapter’s main move: show two-high, then decide at the snap whether to stay there or rotate a safety down to a single-high, “closed-post” coverage. In Cody Alexander’s charting of the Panthers’ defense, published in April 2026, they stayed two-high 397 times and rotated 375 times, what he called “an almost perfect 1:1 ratio.” Cornerback Jaycee Horn described what that felt like from inside: “Everything is really out of a shell, not many tells or pre(-snap) things. It was just on the snap; everything happened.”20

What to notice: on a Carolina passing down, call the shell as the quarterback starts his cadence; it will almost always be two-high. Then, on the replay, watch the two safeties at the snap. Roughly half the time one of them comes down and the other runs to the middle; the rest of the time both stay deep. Nothing before the snap tells you which, and that is the design. It also gives you a benchmark for your own log below: one charted NFL defense built around disguise rotated out of its two-high shell on about half the snaps that started there.

TipWatch for it on Sunday: call it twice

For one quarter (two if you’re keen), on every obvious passing down, make two calls out loud and write them down.

  • Call 1, as the quarterback starts his cadence: the shell. One-high or two-high? Then the details that could change it: how deep is each safety, is one narrower or shallower, are the corners pressed or off? Which side is the formation’s strength?
  • Call 2, one second after the snap: what it became. Is the middle of the field closed (one safety in the post) or open (two safeties splitting the field)? Closed: Cover 1 if the corners turn and run with receivers, Cover 3 if they bail into deep thirds. Open: then the corners tell you which, squatting in the flats for Cover 2 (or that side of Cover 6), backpedalling with eyes on the quarterback for quarters, trailing their man for 2-Man. No safety deep at all: Cover 0. If the two safeties went to two different jobs, notice which one came down and where he landed.
  • Log the lie. Mark each play S (same) or R (rotated). By the end of the quarter you’ll have a disguise rate for that defense. For scale, the Panthers in the film room above rotated out of two-high on about half their two-high snaps. Also note whether the rotation started before the snap (you could see it coming) or at the snap (you couldn’t). Both count as a lie; only the second one worked.
  • Bonus: trips. When the offense shows 3x1, watch the backside safety. Did he stay home (stubbie), come across to #3 (poach), run to the deep middle (rotate), or come down (solo)? Then watch where the ball went.

Live, Call 1 needs a shot that shows the safeties: the wide pre-snap view some broadcasts use while the quarterback surveys the defense, or a high camera behind the offense. The standard live angle sits at the line and often cuts the safeties out of frame at the snap, so on many plays Call 1 has to wait for the replay. The high, wide replays (the closest a broadcast gets to the All-22 film coaches use) are the best place to check yourself: use the first replay to see the shell and the second to check the rotation.

Takeaways

  • The shell is a hypothesis, not an answer. Defenses choose their alignment separately from their coverage, and the best ones show the same picture for many calls. Name the shell before the snap, and expect to check it one second after.
  • Disguise buys time, and it costs something. Every defender must line up where his real job is still reachable, so disguises leak: safety depth and width, corner technique and leverage, eyes, and above all the timing of movement. Late rotation hides the most and risks the most. Which safety rotates is set by the call, most often toward the strength or the motion.
  • Rotations run both ways. Two-high can become Cover 3 or a robber (taking away the middle throws two-high invites); one-high can bail into Cover 2 (taking away the outside throws one-high invites). Each rotation opens the area the quarterback ruled out.
  • Split-field coverage plays two coverages at once, one per half, keyed to strong and weak or (more in college, with its wide hashes) field and boundary. Each player’s job is simpler; the communication is harder, and motion across the split tests it. It is a big part of why two-high defenses spread through the NFL after 2018.
  • Trips forces a decision, and the decision is about #3: the backside safety comes across (poach), a linebacker takes him (solo), the trips side handles it four-on-three (stubbie), or the defense rotates. Whoever is left alone is where the offense will attack.
  • Offenses answer by forcing the defense to show earlier or by making the picture not matter: motion, tempo, hard counts, post-snap reads, play-action and RPOs that target the rotating safety, and quick throws.
