Quarters and the Two-High Match Family: Cover 4, Cover 6, Palms

Two safeties who are deep defenders and run stoppers at the same time, the corner rules that make it work, the half-and-half and trap versions, and why the NFL turned to it after 2018.

def pro_2x2():
    """11 personnel, shotgun 2x2 with the tight end attached: X (on the line) and H (slot, off the
    ball) to the left; Y in-line on the right and Z wide right, off the ball. Seven on the line:
    five linemen, X and Y. On the right #1 is Z and #2 is Y; on the left #1 is X and #2 is H."""
    out = []
    for p in offense("gun_2x2"):
        if p.key == "Y":
            p = p.moved(d=-0.7, w=4.8)
        elif p.key == "Z":
            p = p.moved(d=-1.5, w=15.0)
        out.append(p)
    return out


def spread_2x2():
    """11 personnel 2x2 with the tight end detached as the right slot (gridiron's gun_2x2)."""
    return offense("gun_2x2")


QUARTERS = {"LCB": (7.5, -15.6), "RCB": (7.5, 15.6), "FS": (9.5, -6.5), "SS": (9.5, 6.5),
            "NB": (4.8, -6.2), "MIKE": (4.8, 1.6), "WILL": (4.8, -2.4)}


def quarters_vs(off, moves=None, sde_9=True):
    """Nickel (4-2-5) quarters shell: corners 7-8 yards off with a slight outside shade; safeties
    9-10 yards deep, a few yards inside a slot or over/just outside an attached tight end
    (shallower and narrower than Cover 2 safeties); the nickel ($) at an apex over the left slot;
    the Mike and Will in the box. sde_9 puts the strong-side end outside an attached tight end
    (a 9 technique) so he can set the edge."""
    m = dict(QUARTERS)
    m.update(moves or {})
    out = []
    for p in defense("nickel", "two_high", off):
        if p.key in m:
            p = p.moved(d=m[p.key][0], w=m[p.key][1])
        if p.key == "SDE" and sde_9:
            p = p.moved(w=5.9)
        if p.pos == "CB":
            p = p.moved(label="CB")
        out.append(p)
    return out


def protect(p, rushers=("SDE", "SDT", "WDT", "WDE")):
    """Five linemen set, four rushers engage them: enough to show it is a pass, no more."""
    for k in ("LT", "LG", "C", "RG", "RT"):
        p.block(k, pts=[(-1.2, 0.0)])
    for k in rushers:
        p.rush(k, pts=[(-1.6, 0.0)], speed=2.0)


def no_numbers(ax):
    """Remove the painted yard-line numbers (they sit where wide receivers and safeties stand)."""
    for t in list(ax.texts):
        if t.get_alpha() == 0.55:
            t.remove()


def ruler(fld, play, widths=False, fs=6.5):
    """Depth labels (yards past the line of scrimmage) down the left edge of the field window;
    widths=True also marks the hash marks and the painted numbers along the bottom edge.
    Drawn on the Field's axes: gridiron has no ruler."""
    ax = fld.ax
    x_left = FIELD_W - fld.y1 + 0.35
    for d in range(5, 41, 5):
        y = play.los + d
        if fld.x0 + 0.8 <= y <= fld.x1 - 0.8:
            ax.text(x_left, y, f"{d}" + (" yd" if d == 5 else ""), fontsize=fs, color=style.INK_2,
                    ha="left", va="center", zorder=12,
                    bbox=dict(boxstyle="round,pad=0.12", fc=style.SURFACE, ec="none", alpha=0.85))
    if widths:
        y = fld.x0 + 0.9
        for yb, txt in ((23.58, "hash"), (FIELD_W - 23.58, "hash"), (13.0, "numbers"), (FIELD_W - 13.0, "numbers")):
            if fld.y0 + 2.5 <= yb <= fld.y1 - 2.5:
                ax.text(FIELD_W - yb, y, txt, fontsize=fs, color=style.INK_2, ha="center", va="center",
                        zorder=12, style="italic",
                        bbox=dict(boxstyle="round,pad=0.12", fc=style.SURFACE, ec="none", alpha=0.85))


def say(fld, play, text, at, fs=7.0, color=None, ha="center", weight="normal"):
    """A small boxed on-field label at (d, w), sized independently of Play.note."""
    x, y = fld.xy(*play.to_abs(*at))
    fld.ax.text(float(x), float(y), text, fontsize=fs, color=color or style.INK, ha=ha, va="center",
                zorder=12, fontweight=weight,
                bbox=dict(boxstyle="round,pad=0.2", fc=style.SURFACE, ec=style.BASELINE, lw=0.5))


def clip(fld):
    """Crop the axes to the drawn field window, so players outside a zoomed window don't
    stretch the picture past the turf. (gridiron draws every player; this trims the view.)"""
    fld.ax.set_xlim(FIELD_W - fld.y1, FIELD_W - fld.y0)
    fld.ax.set_ylim(fld.x0, fld.x1)
    for a in list(fld.ax.texts) + list(fld.ax.patches) + list(fld.ax.lines):
        a.set_clip_on(True)
    return fld

def snapshot(play, t, ax=None, ghosts=(), lateral=(-21, 21), window=(-8.5, 19.5), notes=(), show_ruler=False):
    """Everyone where he is at time t, with faint trails since the snap and dashed outlines
    (`ghosts`) where the chosen defenders stood at the snap."""
    pos = {k: play.position(k, t) for k in play.players}
    moved = [pl.moved(d=float(pos[k][0]), w=float(pos[k][1])) for k, pl in play.players.items()]
    q = Play([m for m in moved if m.side == "off"], [m for m in moved if m.side == "def"],
             los=play.los, ball_y=play.ball_y, to_go=play.to_go)
    q.highlights = dict(play.highlights)
    for n in notes:
        q.note(*n)
    fld = clip(q.draw(ax=ax, lateral=lateral, window=window, title=False))
    ax, ppy = fld.ax, fld.pts_per_yard()
    no_numbers(ax)
    if show_ruler:
        ruler(fld, play, fs=5.5)
    P = lambda d, w: fld.xy(*play.to_abs(d, w))  # noqa: E731
    for k, pl in play.players.items():
        if pl.side == "off" and pl.pos not in ("WR", "TE", "RB"):
            continue
        pts = np.array([play.position(k, tt) for tt in np.linspace(0, t, 30)])
        if np.ptp(pts[:, 0]) + np.ptp(pts[:, 1]) < 0.3:
            continue
        xs, ys = P(pts[:, 0], pts[:, 1])
        if pl.side == "off":
            ax.plot(xs, ys, color=style.OFFENSE, lw=1.0, alpha=0.45, zorder=3)
        else:
            ax.plot(xs, ys, color=style.DEFENSE, lw=1.3, ls=(0, (3, 2)), zorder=3)
    if t > 0.05:
        for k in ghosts:
            x0, y0 = P(*play.position(k, 0.0))
            ghost_marker(ax, float(x0), float(y0), play.players[k].label, ppy)
    bx, by = P(*play.ball_position(t))
    bd = np.array(play.ball_position(t))
    in_flight = any(k == "pass" and t > tr and np.hypot(*(bd - np.array(play.position(to, t)))) > 1.2
                    for k, to, tr in play.ball_events)
    if not in_flight:                             # tuck a held ball beside its holder so his letter stays readable
        bx, by = float(bx) + 1.0, float(by) + 0.35
    for kind, _, t_rel in play.ball_events:
        if kind == "pass" and t > t_rel:          # a thrown ball: draw its flight so far
            sx, sy = P(*play.ball_position(t_rel))
            ax.add_patch(FancyArrowPatch((float(sx), float(sy)), (float(bx), float(by)), arrowstyle="-|>",
                                         mutation_scale=8, color=style.BALL, lw=1.4, ls=(0, (2, 1.5)), zorder=9))
    draw_marker(ax, float(bx), float(by), "ball", "", ppy)
    return fld


def strip(play, times, labels, ghosts=(), lateral=(-21, 21), window=(-8.5, 19.5), notes_at=None,
          width_in=6.5, ncols=2):
    """Panels at chosen moments of one play, ghosts kept on every post-snap panel."""
    n = len(times)
    nrows = int(np.ceil(n / ncols))
    pw = width_in / ncols
    h = nrows * (pw * (window[1] - window[0]) / abs(lateral[1] - lateral[0]) + 0.32)
    fig, axes = plt.subplots(nrows, ncols, figsize=(width_in, h), squeeze=False)
    for ax in axes.flat[n:]:
        ax.axis("off")
    for i, (ax, t, lab) in enumerate(zip(axes.flat, times, labels)):
        snapshot(play, t, ax=ax, ghosts=ghosts, lateral=lateral, window=window,
                 notes=(notes_at or {}).get(i, ()), show_ruler=(i == 0))
        ax.set_title(lab, fontsize=8.5, color=style.INK, loc="left")
    fig.tight_layout(h_pad=0.6, w_pad=0.4)
    h = [Patch(facecolor="none", edgecolor=style.DEFENSE, lw=1.2, ls=(0, (2, 1.2)),
               label="Dashed outline: where he stood at the snap"),
         Line2D([], [], color=style.DEFENSE, lw=1.3, ls=(0, (3, 2)), label="Defender's path"),
         Line2D([], [], color=style.OFFENSE, lw=1.0, alpha=0.6, label="Offensive player's path")]
    fig.legend(handles=h, loc="upper center", bbox_to_anchor=(0.5, 0.0), ncol=3, fontsize=7.2, frameon=False)
    return fig

Second-and-8 near midfield. The offense comes out with three wide receivers and a tight end, two receivers on each side. Before the snap the broadcast’s high camera shows both safeties standing only about 9 or 10 yards deep, one over each slot receiver, and both cornerbacks backed off 7 or 8 yards. It looks soft. The offense hands the ball off on an inside run, and the first defender to hit the running back, four yards downfield, is one of those safeties, arriving at full speed from a spot that looked like pass coverage. Two plays later the offense fakes the same run and throws deep, and the same safety is running stride for stride with the tight end up the seam. How can one player be a deep pass defender and a run stopper on the same snap? And why did NFL defenses, which spent the 2010s crowding an eighth man into the box, spend the next six years taking him back out? The answer is a coverage with a misleading name: quarters.

This is a level 4 chapter. One line each on what it builds on:

  • Shells and the box (A First Look at Defense): a two-high shell has two safeties deep, one on each side of the middle; one-high has one in the deep middle. The box is the area near the ball where the run is defended; a light box has no more defenders in it than the offense has blockers. A defender at an apex stands halfway between the last man in the box and a slot receiver, so he can play either.
  • Formations (Spread Formations): 2x2 is two receivers on each side; 3x1 (trips) is three to one side. Receivers are numbered from the sideline in on each side: #1 is the widest, then #2, then #3.
  • Zones and matching (Coverage Fundamentals): the deep field splits into thirds, halves or deep quarters (about 13 yards wide each). The flat is the short area near the sideline, the curl the area between the numbers and the hash, the hook the area inside it. Pattern matching is zone coverage whose defenders pick up receivers man-to-man by rule once the routes declare. A press corner lines up within a yard of his receiver; an off corner, 5 to 10 yards away.
  • Cover 2 (Cover 2 and Tampa 2): two safeties each guard a deep half and the hard corner squats in the flat.
  • Run fits (Run Fits and Box Math): the force defender turns a run back inside; the alley player fills the lane between the force and the box; the plus-one is the one more run defender than the offense has blockers. Defenders read blockers’ “hats”: a blocker firing out low means run, one popping up tall to protect means pass.
  • Routes and play-action (Passing Fundamentals, Play-Action, Boots and Screens): a seam runs straight upfield between the hash and the numbers; an out breaks flat toward the sideline; a slant breaks sharply inside after a step or two; a dig breaks flat inside at 10 to 15 yards; a wheel starts toward the flat and turns up the sideline. Four verticals sends four receivers deep; a flood puts three receivers at three depths on one side. Play-action fakes a run before a pass; on a bootleg the quarterback fakes one way and rolls out the other.
  • Who’s who: on offense X is the receiver on the line, Z the receiver off it, Y the tight end, U a second tight end and H the third receiver, usually in the slot. In a nickel defense the Mike and Will are the two linebackers and the nickel back is the fifth defensive back; a base 4-3 adds a third linebacker, the Sam.

Quarters in one picture: four deep, but not really

Start with the name, because it misleads almost everyone. Quarters, also called Cover 4, is a two-high coverage in which four defenders, the two cornerbacks and the two safeties, each start with a deep quarter of the field, a lane about 13 yards wide (hence the “4” in the numbering system). If that were the whole story, quarters would be a passive coverage: four defenders guarding deep grass, three underneath, plenty thrown in front of them. But each of those four deep defenders has a rule that tells him whom to cover once the receivers start running, and two of those rules point the safeties forward, not back. Here are the alignment and the rules against the NFL’s most common formation, a 2x2 set with the tight end attached.

Go around the picture one position at a time.

  • The corners line up 7 or 8 yards off the outside receiver (#1), square or with a slight outside shade, eyes on him. Their rule, MOD, gets its own section below. The short version: if #1 runs deep, the corner has him man-to-man; if #1 runs a short route, the corner lets him go and drops into his deep quarter.
  • The safeties are the heart of the coverage. They line up 8 to 10 yards deep (some teams 10 to 12): a few yards inside a slot receiver, or over or just outside an attached tight end, who is already inside the box. That’s much shallower and narrower than a Cover 2 safety, who has to reach the sideline and so stands at 12 to 14 yards, outside the hash. The quarters safety doesn’t have a sideline to reach. His real job is to read the inside receiver.
  • The apex defender, here the nickel back on the slot side, stands between the tackle and the slot, 4 or 5 yards deep. Against a pass his first job is to wall #2: get in the slot receiver’s path from the inside and keep him from running freely up the seam, so the safety behind has time to read him. Then he takes the first receiver who comes to the flat, the running back included.
  • The linebackers read run first. Against a pass they drop to the hook zones and take the running back if he runs up the middle. On the tight end’s side there is no apex defender, so the strong-side linebacker (here the Mike) is the first man to the flat if the tight end releases outside.
  • The front rushes four. The end outside the tight end is the edge (contain) player against runs to his side, and he can jam the tight end on his way out.

Now count the defenders against the two receivers on one side. On the slot side: the corner, the safety, the apex defender and, a step inside, a linebacker. On the tight end’s side: the corner, the safety, the Mike and the end. Four players against two receivers, with the back as a possible third. Coaches like to say quarters is really “four over two” on each side, not four deep, and that everyone deep is also, depending on what the receivers do, a man defender or a run defender. That is what makes quarters a pattern-matching coverage, and the purest one in common use.

NoteMadden vs. real life

In a video game’s play menu, “Cover 4” usually draws four blue deep zones and three short ones, and the defenders drop to them like spot-drop zone players. If the real coverage worked that way it would be a prevent defense (the soft, everyone-deep look teams use to protect a late lead, covered in Clock and Game Management): four defenders guarding grass 15 yards deep while the offense completes everything underneath. Real quarters is a read coverage. The deep quarters are starting points; the receivers’ first steps decide who covers whom, and the two safeties spend the first second of every snap deciding whether they are deep defenders at all. If you remember one thing from this chapter, make it this: quarters safeties play the inside receiver, not a zone.

ImportantWhy it exists

What it solves. A defense wants two things that normally conflict: two safeties deep, so the offense can’t hit an explosive pass over the top, and an extra run defender, so it can’t run the ball down its throat. Quarters gets both from the same two players. Standing at 9 or 10 yards instead of 13, a safety is close enough to the line to fit a run in time and deep enough to carry a receiver up the seam.

What it gives up. The shallow, inside-leaning safeties are reading, and anything that makes them read wrong hurts: a run fake, a receiver whose release looks like a block, a route that starts one way and ends another. With only three defenders underneath, the short outside throws (the quick out, the hitch, the flat) belong to the offense unless a corner or an apex defender takes them away. Quarters trades a few easy short completions for no deep ones, and asks its safeties to be right.

The read safety: one key, three answers

The quarters safety is the clearest example of a read safety: a safety whose job is decided after the snap by a single key receiver, usually #2 on his side. Coaches teach it as three answers to one question, what did #2 do?

  1. #2 goes vertical: carry him. If the inside receiver runs up the seam, the safety covers him, man-to-man, as deep as he goes. With the corner on #1, every vertical route on that side has a defender on top of it.
  2. #2 goes short or out: help on #1. If #2 runs to the flat or breaks off a short route, the safety has nobody to cover, so he turns to the outside receiver and sits inside him. Now #1 has the corner outside and over the top and the safety inside: a bracket, two defenders on one receiver. The dig and the post, the routes that hurt quarters corners most because they break away from the corner’s outside position, run straight into the safety. (If #2 runs inside, across the field, many systems tell the safety to pass him to the linebackers and look for work: #1 breaking in, or a crossing route coming from the other side.)
  3. #2 blocks: it’s a run, so fill. If #2 is a tight end who blocks, or a slot receiver who steps up to block the apex defender, and the linemen in front of the safety fire out low, the safety reads run and comes downhill. In the language of Run Fits and Box Math, he becomes the alley player: he fills the lane between the edge and the box, inside-out, and he is the defense’s plus-one.

What counts as “vertical”? Most staffs draw the line where the corner’s rule does, at about 5 yards. A #2 who breaks out or in before that is short; a #2 still pushing upfield past it is vertical, and the safety stays on top of him until he breaks. If he then curls or digs at 10 or 12 yards, the safety sits over it with a hook defender underneath. Exact depths vary by staff.

Why does one receiver make such a good key? Because #2’s release tells the safety, almost instantly, both run-or-pass and where the danger is. A receiver who blocks is the strongest run signal there is; one who releases vertical is the danger; one who runs to the flat isn’t a deep threat, so the deep threat must be the outside man. Good safeties check it against the linemen in front of them: low hats confirm the run, but a #2 who “blocks” while the linemen pop up tall to pass-protect is selling a fake, a chip before a release or a play-action block. That check takes a beat, and it is the read’s weak spot, as the bootleg section will show.

That also explains where the safety stands. He has to be shallow enough to fill the alley before the running back gets through it, and deep enough to turn and run with a tight end up the seam without being beaten over the top. Nine or ten yards is the compromise. TCU’s Gary Patterson, one of the coaches who made quarters famous in college football, taught his safeties not to backpedal at the snap at all but to shuffle flat-footed and read, because a safety rocking backward can’t come forward in time.1 Watch one quarters safety for a whole game on the All-22 film (the high, wide camera coaches use) and you’ll be surprised how rarely he ends the play deep.

The corner’s rule: MOD and MEG

The safety’s rule only works if the corner’s rule fits with it. Quarters teams use two, and the names come from the playbook vocabulary that spread out of Nick Saban’s defenses (more on that history below).

MOD, short for man only deep, is the standard quarters corner rule. The corner covers #1 man-to-man if #1 runs a deep route. If #1 breaks his route off short, about 5 yards or less (a hitch, a quick out, a slant, a shallow cross), the corner doesn’t chase him. He passes him to the underneath defenders and drops into his deep quarter, looking for the next threat to come into it. “Deep” here means anything past that 5-yard line, so a 12-yard dig or a curl counts: the corner stays on it. That is nearly word for word the rule Cameron Soran decoded from Saban’s playbooks, which also give the other two clauses of the same call. The safety: take all of #2 vertical; if #2 isn’t vertical, bracket #1. The apex defender: take the first man to the flat, rerouting #2 first if he starts vertical.2

MEG, short for man everywhere he goes, is the other rule. The corner covers #1 on every route, short or deep, no exceptions. You met MEG in Cover 3 as the backside corner’s rule in a trips check; it shows up wherever a defense needs one corner to take care of his man alone so that the defenders around him can do something else.

The trade is clear in the two panels.

  • MOD keeps the corner deep and gives the defense extra help on the dangerous routes. Short routes by #1 are left to the defenders already in front of them (the apex defender, the linebackers), who can rally and tackle for a short gain. The corner, freed from the short route, becomes a second defender on whatever goes deep on his side, often #2 breaking outside, as here. The cost: the short throw to #1 is there if the underneath defenders are slow to it, and a receiver who fakes short and goes deep (a stop-and-go, a slant-and-go) can catch a MOD corner who passed him off too early.
  • MEG locks the corner on his man and frees someone else. In the figure the freed player is the nickel; in the trips checks of Disguise, Rotation, and Split-Field Coverage it is a whole safety, who leaves his side to help elsewhere. The cost: the corner is on an island with no inside help, and every route that breaks away from him, especially a slant or a dig against an off corner, is his problem alone. MEG needs a corner the defense trusts.

A defense picks between them by situation and by personnel. A team with one great corner might play him MEG on the opponent’s best receiver and MOD on the other side. Against a lone receiver in a trips formation, MEG on the backside corner is the standard way to free the backside safety. And in the red zone MEG becomes more common, because MOD’s escape hatch is gone: with the end line 10 or 15 yards away there is no deep quarter to retreat into, so the short routes are the dangerous ones, and a corner who passes them off has nothing left to guard.

NoteGo deeper: rules that agree with each other

The rules fit together like a contract. MOD says “I’ll let #1 go if he’s short”; the apex and hook defenders say “we’ll take what comes short, outside and inside”; the safety says “if #2 isn’t vertical, I’ll bracket #1,” covering the one thing MOD gives up deep. Change one clause, from MOD to MEG for example, and every other clause shifts with it. That is why match defenses talk so much before the snap.

Quarters against the run: the safeties fit from depth

Now the part of quarters that made it a run defense. Look back at the alignment: four defensive linemen and two linebackers in the box, six defenders, against five offensive linemen and a tight end, six blockers. By the box count of A First Look at Defense that is a light box: every run defender can be blocked and nobody is left over. A defense playing one-high would usually have walked its strong safety down to make it seven against six.

Quarters gets its seventh (and sometimes eighth) run defender from 9 or 10 yards away: when #2 blocks and the linemen fire out low, the safety comes downhill into the alley. College coaches with a 4-3 front call it a “nine-man front,” seven defenders near the line plus two safeties who arrive when the play is a run.

Three things in that play are worth slowing down for.

The trigger is a read, not a call. The strong safety didn’t know before the snap that this would be a run. He found out from Y’s block in the first half-second, and he went. That is the difference between quarters and the rotation of Disguise, Rotation, and Split-Field Coverage, where a safety is told before the snap to come down at the snap, read or no read. The called rotation arrives sooner, and it is wrong whenever the guess is wrong. The read arrives later, and an honest pass never pulls it forward, because #2 running a route is a pass read. Its weak spot is the dishonest run look, a play-action fake and a #2 who pretends to block, which is exactly what the bootleg flood below attacks.

He meets the back past the line of scrimmage. From 9 or 10 yards the safety is a beat later than a linebacker at 4 or 5, so he typically makes the play three to five yards downfield, roughly the bargain Run Fits and Box Math describes for the two-high light box: give up a four-yard run, never a forty-yard pass.

The other safety went with his receiver. The free safety’s #2, H, released up the field, a pass read, so the free safety stayed with him and was out of the run fit. Here that cost the defense nothing, because the run went toward Y and the safety who filled was the play-side one. Now flip it: run the same play to the left, toward H. H releases, the free safety (now the play-side safety) goes with him, and the only safety left to fill is the strong safety, on the backside, arriving late through traffic. That is the offense’s main weapon against quarters’ run defense, and it costs the offense little, one fewer blocker on the perimeter (the nickel is left unblocked, so the back has to beat him). Offenses with a good slot receiver run the ball away from the tight end for exactly this reason, and RPOs go one step further by letting the quarterback read that safety and throw to the receiver if he comes down anyway.

Had H blocked instead, the free safety still wouldn’t have raced to the ball. With the run going away from him he is the cutback player, filling late and inside-out in case the back reverses course, and the first defender who has to see a bootleg or a reverse coming back his way. Outside the safeties, the end sets the edge on the tight end’s side and the corner comes up once Z blocks him; on the slot side the nickel is both the force player and the first man to the flat, the overhang defender job (the defender outside the box, over the slot). Names vary by staff; the geometry, an outside defender who forces and a safety who fills inside him, doesn’t.

ImportantWhy it exists: run support from the secondary

What it solves. Most of the time a defense that wants an extra run defender has to show him: a safety walked down to 7 or 8 yards before the snap, which tells the offense to throw. Quarters hides the extra defender inside a two-high picture and lets the offense’s own blocking call him into the fit. Every running play is met by six defenders in the box plus one or two who arrive late from depth, and every pass that looks like a pass is met by two safeties who never left.

What it gives up. Time and certainty. The safety arrives after the back has crossed the line, so quarters gives up more yards per run than an eight-man box, and he must tackle in space with nobody behind him. And anything that disguises the read attacks the coverage at its heart.

Cover 6: quarters on one side, Cover 2 on the other

Quarters works beautifully when each side of the formation has two receivers. Formations don’t always cooperate. Against trips, one side has three receivers and the other a lone receiver, and on the lone receiver’s side there is no #2 at all: a read-#2 safety has nothing to read. The most common fix is to split the field. Cover 6 plays quarters on one half of the field and Cover 2 on the other, which is why coaches also call it quarter-quarter-half (the corner and safety each take a quarter on one side; one safety takes the whole half on the other). The number is dialect, not law: in Vic Fangio’s system the split-field family is Cover 6 and Cover 8, and his Cover 8 flips the halves, putting the Cover 2 half toward the passing strength (Coverage Fundamentals has the dialect notes).

Disguise, Rotation, and Split-Field Coverage shows Cover 6 in its most common NFL use, against a 2x2 set split by field position (quarters to the wide side, Cover 2 to the short side). Here it is against a 3x1 trips set, one of the formations that calls for it, with the split made by formation: quarters to the three receivers, Cover 2 to the lone receiver.

Why put Cover 2 on the lone receiver? Look at what the two defenders on that side are doing to X. The squat corner sits in front of him, taking away the quick out, the hitch and the flat; the half-field safety sits over the top of him, taking away the go route. X is bracketed under and over by two defenders, which is more help than most single receivers ever see. That matters because offenses put their best receiver alone on the backside of trips precisely to get him a one-on-one matchup. Cover 6 says no.

The cost is on the trips side, where four defenders (corner, safety, nickel and Mike) handle three receivers who can all run deep. Nobody is uncovered: the corner has #1, the safety #2 and the Mike #3 if he runs up the middle. The problem is the matchup. If #2 and #3 both go vertical, the safety takes #2 and the Mike is left running down the seam with a tight end or a slot receiver, a race most linebackers lose. That is the trips problem in its purest form, and two-high defenses carry other answers to it, such as poach (the backside safety leaves the lone receiver to his corner and comes across to help on #3), which Disguise, Rotation, and Split-Field Coverage covers in detail. Cover 6 is the simplest answer: give the lone receiver to two defenders, and trust four on three.

Cover 6 is also how Cover 2 survived: too easy to attack as a whole defense (Cover 2 and Tampa 2), it still earns its keep as half of one, aimed at the short side or a lone receiver. By NFL Next Gen Stats’ count, Cover 6 was 9.2% of coverages in 2025.3 After the snap the corners give it away, one squatting and one bailing; the Watch-for-it box at the end has the full sequence.

Palms: the corner who traps the out

Quarters has one obvious soft spot: the flat, in the first second after the snap. The apex defender’s rule makes him the owner of the flat, but he starts inside #2, walling him off the seam, 4 or 5 yards from where a quick out breaks. On a throw that comes out in about a second he can’t widen in time. The corner is 8 yards deep playing #1, and the safety is reading #2 from 9. So a quick throw to #2 breaking toward the sideline lands in front of everyone. Offenses learned to throw it all day: the slot receiver’s quick out, the “now” throw to the flat (a cousin of the now screen), the out by the slot in a smash-type concept.

Palms is the answer. The same coverage goes by 2-read and, because of what the corner does, trap coverage. Different staffs use different names and slightly different rules, but the core is one adjustment on one side of the field: the corner reads #2 instead of #1.

  • If #2 breaks out quickly (inside the first 5 yards or so), the corner leaves #1 and jumps #2’s out or flat route. He was 6 or 7 yards deep, square or a little outside #1, with his eyes turned in to #2, so he comes forward and outside-in onto the throw instead of chasing it. That is the trap.
  • The safety, seeing #2 go out, takes #1 deep: he’s now the deep defender over the outside receiver whom the corner just left. In Saban’s version the safety’s rule reads: take #1 vertical if #2 is out in the first 5 yards; otherwise, take all of #2 vertical, and if #2 isn’t vertical, bracket #1.4
  • If #2 goes vertical, the safety carries #2 and the corner plays #1 man to man, in most versions without the MOD release, so a short route by #1 runs into a corner who is already shallow and squeezing.

Watch the timing. The out route is a rhythm throw: the quarterback decides before the snap, takes his drop and lets it go as #2 breaks, about a second after the snap. He threw it because the corner looked like a quarters corner, off and playing the outside receiver. Why would he fall for it if there is a tell? Because the tell is a yard of depth and a turn of the head, often disguised until the snap, and a rhythm throw is decided before the snap: the quarterback isn’t re-reading the corner as he lets it go. A Palms corner’s eyes were on #2 from the start, and he started forward the moment #2 broke. Against a throw that travels 15 or more yards across the field, a defender coming downhill onto the route gets to the catch point first. That’s how quick outs become interceptions returned the other way.

Palms also cleans up two routes that trouble Cover 2. Against smash (a hitch by #1, a corner route by #2), #2’s corner route starts vertical, so the safety carries it, and the hitch runs into the corner, who is manned up on #1 and sitting on it. Against a slot’s flat route paired with a curl by #1, the corner jumps the flat and the safety, who no longer has a vertical #2, drives on the curl from inside.

ImportantWhy it exists: taking away the throw quarters gives up

What it solves. Quarters concedes quick outside throws to #2. Palms puts a defender on them without adding one: the corner, who would have been 8 yards deep over a #1 who isn’t going deep, trades jobs with the safety, who would have had nothing to do once #2 went to the flat.

What it gives up. The trap is a guess about the route, and good offenses bait it. A #2 who breaks out and then turns up the sideline (an out-and-up, or a wheel) runs right past the jumping corner, and the safety is busy with #1. A #1 who runs a double move against a safety coming from inside can win too. And the corner who reads #2 is, for a moment, not watching #1 at all, so a fast #1 on a quick slant inside him can be open in the first second. Defenses mix Palms with plain quarters for exactly this reason: a corner who always traps gets beaten by the play designed for him.

How do you spot it? Before the snap, barely: the Palms corner is often a yard or two shallower than the other corner, with his shoulders and eyes turned in toward the slot. After the snap it’s unmistakable: the corner comes forward toward #2 while the safety on that side runs outward and deep over #1, and the two cross like a pair of scissors. Palms is usually played to one side, often the short side of the field or a 2x2 side, while the other half plays its own call, which makes it another piece of the split-field toolbox.

Who decides: calls, checks and motion

Three coverages that look alike before the snap raise an obvious question: how do eleven defenders know which one they’re in? Usually in two layers. The sideline sends in a call for the family, and the players on each side check within it once they see the formation. The corner and safety on a 2x2 side might play quarters or Palms depending on the receivers’ splits; a 3x1 set might turn the call into Cover 6 or a trips check such as poach. A safety or the Mike makes the strength call, by shout and hand signal, so both halves agree where the split is.

Motion tests all of that, because motion changes who is #2. Receivers are recounted from the sideline after every move. If H motions from the left slot across to the right, the free safety’s side suddenly has one receiver and no #2 to read, and the right side has three. The defense has a second or two to re-declare: the strength flips, the free safety becomes a lone-receiver-side player (a Cover 2 half, or a poach player watching #3), and the apex defender usually slides over with the motion, a bump that keeps everyone in his zone. Get the re-declaration wrong and two defenders take the same receiver while the new #2 runs free. That is a big reason offenses motion so much against split-field defenses (Disguise, Rotation, and Split-Field Coverage has the offense’s side, Motion and Shifts as Weapons the defense’s answers).

Why two-high match took over the NFL, 2018 to 2025

None of this was new in 2018. What changed is how much NFL defenses used it. NFL Next Gen Stats (the league’s tracking-data arm) measures how often defenses play a split-safety (two-high) coverage after the snap: 32.9% of dropbacks in 2018, 37.8% in 2021, 40.4% in 2024 and 42.0% in 2025, the highest since its tracking began. Quarters was the biggest single piece: Cover 4 alone was 17.2% of 2025 coverages by Next Gen Stats’ count, against a 2018–2024 average of 13.9%, with Cover 2 at 13.9% and Cover 6 at 9.2%.5

Why the rise? No single cause, but five reasons show up again and again in coaches’ and analysts’ explanations, and they reinforce each other.

  1. Explosive passes became the thing to stop. The plays that decide games are the long completions (Expected Points and Success Rate shows why a 40-yard pass is worth so much more than a 4-yard run), and by the late 2010s passing was more efficient than ever. The 2018 season was the loudest version: Patrick Mahomes threw 50 touchdown passes in his first year as a starter, and the Rams beat his Chiefs 54–51 on a Monday night in November.7 Two deep safeties are the most direct protection against the explosive pass, and quarters keeps two deep without giving up the run.
  2. Match rules answered the spread passing game. Offenses built on putting two receivers in one zone defender’s area ran into zones that followed the receivers.
  3. The extra run defender could come from depth. A defense that played one-high with an eighth man in the box was guessing run every snap, and run-pass options punished that guess (RPOs). Quarters’ read safeties are harder to put in conflict before the snap, because their run job is decided by the play itself.
  4. Two-high pictures disguise everything. From a two-high shell a defense can play quarters, Cover 6, Palms, Cover 2 or 2-Man, or rotate into a one-high coverage at the snap. The more of those it plays, the less the quarterback learns before the snap (Disguise, Rotation, and Split-Field Coverage).
  5. Coaching trees and a college pipeline. Coaches who learned split-field quarters in college (more below) reached NFL staffs, and one tree in particular spread it: that of Vic Fangio, whose two-high defenses in Chicago (as coordinator) and Denver (as head coach, 2019–2021) became the model. His story is a case study: Fangio’s Two-High Revolution and Macdonald’s Answer.

There is a sixth condition that isn’t a cause so much as a price of admission: a four-man rush that can win. A light box and two deep safeties work only if four rushers can pressure the quarterback without help, because nobody else is coming. Quarters teams can blitz, usually with an apex defender or a linebacker, but every extra rusher comes out of the four-over-two, so much of their pressure is still four rushers chosen in a surprising way, a simulated pressure. The defenses that lived in two-high were also the ones with dominant interior rushers, a pattern you’ll see in the film room.

And offenses answered. If quarters invites the run against a light box, run it; if it reads #2, make #2 a blocker. The 2024 and 2025 seasons brought more two- and three-tight-end personnel and more snaps under center, the subject of The Modern Era.

A short history: from Cleveland to East Lansing and Iowa City

Four defenders deep is an old idea; prevent defenses have used it for decades. Quarters as a read coverage whose safeties fit the run is mostly a product of 1990s and 2000s college football, with one important root in the NFL.

That root is Cleveland. Nick Saban was Bill Belichick’s defensive coordinator with the Browns from 1991 to 1994, and the pattern-matching rules the two developed there, first for Cover 3 and then for the quarters family, went with Saban to Michigan State as head coach in 1995 and later to LSU and Alabama.8 Saban’s vocabulary is the one this chapter has been using (MOD, MEG, Palms), plus “Cover 7” for his quarters-based split-field family (Disguise, Rotation, and Split-Field Coverage) and “Rip” and “Liz” in his match Cover 3 (Cover 3). One set of match rules, several coverages: that’s the Saban-Belichick inheritance. Like most who-invented-it stories, this one has rivals; other staffs were building match rules at the same time, and the coverage numbers themselves meant different things in different buildings.

Michigan State kept the quarters thread alive after Saban left. Under head coach Mark Dantonio, who had coached Saban’s secondary in East Lansing, and defensive coordinator Pat Narduzzi (2007–2014), the Spartans’ 4-3 quarters became one of the most studied defenses in the country (the film room has its best season). Narduzzi came to Cover 4 the hard way: in 2002, as a Northern Illinois assistant, he watched Miami (Ohio)’s Ben Roethlisberger throw for 525 yards against his defense’s two-deep look, and when he became Miami’s defensive coordinator the next year, by his own telling at a 2014 clinic, his answer was Cover 4.9

Iowa is the other long-running reference point. Norm Parker installed his 4-3 defense when Kirk Ferentz took over the program at the end of the 1990s, and Phil Parker (no relation), the defensive coordinator since 2012, kept it: a front that holds up against the run while the safeties play split-field quarters behind it, year after year.10 TCU’s Gary Patterson built a 4-2-5 (four linemen, two linebackers, five defensive backs) around the same idea. When NFL defenses went looking for a two-high answer to the spread offenses of the late 2010s, this is where the coaching knowledge was.

Where quarters breaks

The holes of quarters follow directly from its rules. An offensive coordinator facing a quarters team starts with three of them.

Light-box runs

You’ve already seen the first, in the run-fit section: six in the box against six blockers, with the safeties arriving late, and an offense that releases #2 or reads the safety with an RPO to take one of them away. The other lever is simpler: add a blocker. Two-tight-end and fullback sets make #2 a blocker who can block the safety, or put a seventh blocker against a six-man box so the box is outnumbered before anyone arrives from depth; the heavier personnel of 2024 and 2025 was partly this. The defense’s counters are a front that holds its gaps without help (which is why two-high defenses prize big interior linemen), safeties who tackle in space, and a willingness to give up four-yard runs. Run Fits and Box Math has the data on what runs against light boxes actually gain.

The bootleg flood

Play-action attacks any coverage whose defenders read the run. Quarters is special because its deep defenders read the run.

Walk through it from the defense’s side. The corner saw #1 go deep and went with him: correct, MOD. The Will stepped up on the fake, then chased U to the flat: correct. The free safety read #2, saw a block and came forward: correct by his rule, and wrong, because U’s block was a fake. An honest pass never fools a quarters safety; a pass dressed as a run can, because for a beat the fake and the real thing look the same. The over route, coming from the other side of the field, entered the space between the corner (deep with X) and the safety (up for the run) where nobody’s rule put a defender: three receivers at three depths on one side, against defenders whose first read sent them the wrong way. The fake has to hold one more defender. On a bootleg the end on the side the quarterback rolls to is usually left unblocked, so the quarterback is running toward a free rusher who must bite on the run fake too.

It isn’t that play-action is magic against quarters in particular. In the nflverse charting for 2023–2025, play-action raised the average result of a dropback against quarters by about as much as it did against Cover 3 or Cover 1.11 The point is which play-action. A run fake that pulls the safeties forward, paired with a deep crosser from the far side, attacks the exact gap between quarters’ corner rule and its safety rule. The defense’s counter is discipline in the read (safeties taught to key through the blocker to the linemen and the quarterback, not just #2’s first step) and calls that tell the safety on the bootleg side to stay deep no matter what, at the cost of the run fit.

Each rule is reasonable on its own. Together, the routes in front of the corner and the safety pulled both of them inside, and the back’s route started as a flat route (the apex defender’s) and became a deep route down the sideline (nobody’s). The defender left with him is a nickel back or a linebacker, chasing a running back from behind, with an empty deep quarter in front of them. The wheel is the classic match-coverage beater because it changes category in the middle of the play: short when the rules assign it, deep when it’s caught.

Defenses answer by teaching the apex player to “carry” a flat route that turns up, and by telling the corner, when #1 goes inside, to look for something coming outside behind him; each answer loosens something else. The offense’s side of all this, the full menu of concepts aimed at quarters’ rules, is the job of Concepts Versus Coverages.

NoteFilm room: Michigan State 2013 — Narduzzi’s quarters

The college reference point. TipWatch for it on Sunday: two-high, now what?

When the defense shows two deep safeties, don’t stop at “two-high.” Use the next three looks. On the broadcast angle the safeties are usually in the frame before the snap and the wide corners often aren’t, so read the safeties live and save the corners for the replay.

  • Before the snap: depth and width. Safeties at 12 to 14 yards, outside the hashes: think Cover 2 or a half-field player. Safeties at 8 to 10 yards over the slot or the tight end: think quarters. One deep and wide, one shallow and inside: think Cover 6, with the half on the deep safety’s side. Then the corners, if you can see them: both off 7 or 8 yards, eyes on their receivers, suggests quarters; one squatting at 5 with outside leverage suggests a Cover 2 half; one a yard or two shallower than the other, with his shoulders turned in to the slot, suggests Palms.
  • At the snap: the safeties’ first steps. Watch one safety for one second. Backpedal or turn and run for the sideline: a half-field player (Cover 2, or the Cover 2 side of Cover 6). Downhill toward the line: a quarters safety reading run, or a called rotation (the other safety will tell you which: if he runs to the middle, it’s a rotation to one-high). Flat-footed, then with #2 up the seam: quarters, carrying #2.
  • One second later: the corners. Both bailing with eyes on the quarterback: quarters. Both squatting: Cover 2. One of each: Cover 6. A corner coming forward toward the slot’s out route while the safety on that side runs deep and wide over the outside receiver: Palms.
  • Then watch the ball. On runs, find the first defender to the ball. If it’s a safety arriving three to five yards downfield, you’ve just watched quarters’ run fit. On passes, note whether the completion was short and outside (what quarters concedes) or deep (what it exists to stop).

TipAnswer

Quarters, or something that will look like it for a beat. Two safeties at 9 yards, inside the slot on the left and just outside the tight end on the right, are reading safeties, not half-field players: a Cover 2 safety would be at 12 to 14 and outside the hash, because he has to reach the sideline. Off corners on both sides fit quarters’ MOD rule. Six defenders in the box against six blockers is the quarters light box. (A defense can still rotate out of this at the snap, as Disguise, Rotation, and Split-Field Coverage shows.)

The situation matters. On second-and-12 the defense is happy to give up a four-yard run or a five-yard out and play third-and-long. The offense wants a chunk, so attack the safety’s read rather than take what quarters concedes.

The run: run away from the tight end, toward the slot side, with H releasing up the field so the free safety has to go with him; or call an RPO that reads that safety.

The pass: play-action with a deep crosser (the safeties step up on the run fake), or scissors, which makes the safety and corner cross. The quick out by #2 is there unless the corner is playing Palms, but it only gets you to third-and-7. Four verticals is the one call this defense was built for.

The quarterback sees an off corner and thinks the quick out to H is free. What is the danger, and what would you call the next time you see this corner?

TipAnswer

The danger is Palms: the trap. The left corner is a yard or so shallower than the right one, square to X but turned in toward the slot: he is reading #2. When H breaks out, the corner leaves X and drives forward, outside-in, onto the out, and the free safety, seeing #2 go out, takes X deep. X’s go route was supposed to clear the corner away; instead it runs into the safety. The situation is part of the tell: on third-and-2 the quick out to #2 is the obvious conversion throw, which is exactly when a defense calls the trap. Best case for the offense, an incompletion and a punt; worst case, an interception returned the other way.

Next time, bait the trap. Call an out-and-up for H (or a wheel): the same first 4 yards, then up the sideline behind the jumping corner, with the safety busy with X. Or attack the moment the corner looks inside: a quick slant by X, breaking inside a corner whose eyes are on H. The deeper lesson is the trade every match rule makes. Palms takes away the quick out, and in doing so tells you which route will beat it.

Takeaways

  • Quarters is a read coverage, not four deep zones. Corners and safeties start with deep quarters, but the receivers’ releases decide who covers whom: the corner takes #1 deep (MOD), the safety reads #2. It is the purest pattern-matching coverage in common use.
  • The read safety is the whole design. One key, #2, gives three answers: carry him if he goes vertical, bracket #1 if he goes out or short, fill the alley if he blocks. Standing 8 to 10 yards deep lets one player be a deep defender and a run defender on the same snap.
  • MOD and MEG are a trade between help and freedom. MOD lets short routes go to the underneath defenders and doubles the deep ones; MEG locks a trusted corner on his man and frees a teammate for another job.
  • Cover 6 and Palms fix quarters’ soft spots without adding a player. Cover 6 puts a Cover 2 half on the lone receiver (under-over on him) and quarters on the trips side; Palms swaps the corner and safety so the corner can trap #2’s quick out.
  • Two-high match spread because it stops explosive passes without surrendering the run, answers spread passing concepts, and disguises well. Split-safety coverage rose from 32.9% of dropbacks in 2018 to 42.0% in 2025 by NGS’s count; the charting data agree on the total but split it differently across the 2022/2023 source break.
  • Its holes follow from its rules: light-box runs when receivers take the safeties away, the bootleg flood that turns the safeties’ run read against them, and the wheel that changes from a flat route to a deep one after the rules have assigned it.
