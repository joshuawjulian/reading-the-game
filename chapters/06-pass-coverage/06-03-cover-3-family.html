Cover 3: Sky, Buzz, Cloud, and Seattle’s Match Version

Three deep, four under: the NFL’s most common zone, the three ways a defense rotates into it, the holes every offense knows about, and how Seattle taught it to carry the seams.

# ---------------------------------------------------------------- players and alignments
def O(key, pos, d, w, label=None):
    return Player(key, pos, "off", d, w, label=key if label is None else label)

def D(key, pos, d, w, label):
    return Player(key, pos, "def", d, w, label=label)

def line5():
    """The five offensive linemen, 1.6 yards apart (only the center is lettered)."""
    return [O(k, k, -0.7, w, "C" if k == "C" else "") for k, w in
            (("LT", -3.2), ("LG", -1.6), ("C", 0.0), ("RG", 1.6), ("RT", 3.2))]

def spread(rb_w=-1.5):
    """11 personnel, shotgun 2x2: X and Z on the line (with the five linemen: seven on the line),
    H in the left slot and the tight end Y flexed into the right slot, both a step off the ball."""
    return line5() + [O("QB", "QB", -5.0, 0.0, "Q"), O("RB", "RB", -5.0, rb_w, "R"),
                      O("X", "WR", -0.7, -15.0), O("H", "WR", -1.5, -9.0),
                      O("Y", "TE", -1.5, 9.0), O("Z", "WR", -0.7, 15.0)]

def front4():
    return [D("WDE", "DE", 1.0, -5.6, "E"), D("WDT", "DT", 1.0, -0.8, "T"),
            D("SDT", "DT", 1.0, 2.4, "T"), D("SDE", "DE", 1.0, 5.6, "E")]

LAB = {"MIKE": ("MIKE", "M"), "WILL": ("WILL", "W"), "NB": ("NB", "$"), "LCB": ("CB", "CB"),
       "RCB": ("CB", "CB"), "FS": ("FS", "FS"), "SS": ("SS", "SS")}

C3_POST = {"MIKE": (9.0, 4.2), "WILL": (9.0, -4.2), "NB": (9.5, -12.5), "LCB": (18.0, -16.8),

def nickel(spots):
    """4-2-5 nickel: the front four plus seven droppers at `spots` = {key: (d, w)}."""
    return front4() + [D(k, LAB[k][0], *spots[k], LAB[k][1]) for k in spots]

def protect(p, rushers=("WDE", "WDT", "SDT", "SDE"), linemen=("LT", "LG", "C", "RG", "RT")):
    """Linemen set, four rushers engage them: enough to say 'pass', no more."""
    for k in linemen:
        if k in p.players:
            p.block(k, pts=[(-1.2, 0.0)])
    for k in rushers:
        if k in p.players:
            p.rush(k, pts=[(-1.6, 0.0)], speed=2.0)

# ---------------------------------------------------------------- drawing helpers
def P(fld, p, d, w):
    return fld.xy(*p.to_abs(d, w))

def clean(ax):
    """Drop the painted yard numbers (they sit where receivers stand); clip text to the window."""
    for t in list(ax.texts):
        if t.get_alpha() == 0.55:
            t.remove()
        else:
            t.set_clip_on(True)

def ruler(fld, p, fs=6.0, step=5, top=40):
    """Depth labels (yards past the line of scrimmage) just inside the left edge of the window."""
    xl = (160 / 3) - fld.y1 + 0.4
    for d in range(step, top + 1, step):
        y = p.los + d
        if fld.x0 + 1.5 <= y <= fld.x1 - 1.5:
            fld.ax.text(xl, y, f"{d}" + (" yd" if d == step else ""), fontsize=fs, color=style.INK_2,
                        ha="left", va="center", zorder=12,
                        bbox=dict(boxstyle="round,pad=0.12", fc=style.SURFACE, ec="none", alpha=0.85))

def say(fld, p, d, w, text, color=style.INK_2, fontsize=7.0, ha="center", weight="normal", box=True):
    x, y = P(fld, p, d, w)
    fld.ax.text(float(x), float(y), text, fontsize=fontsize, color=color, ha=ha, va="center", zorder=13,
                fontweight=weight, linespacing=1.15,
                bbox=dict(boxstyle="round,pad=0.18", fc=style.SURFACE, ec="none", alpha=0.9) if box else None)

def point_to(fld, p, text_at, target, text, color=style.INK_2, fontsize=7.0, ha="center"):
    (xt, yt), (xp, yp) = P(fld, p, *text_at), P(fld, p, *target)
    fld.ax.annotate(text, (float(xp), float(yp)), xytext=(float(xt), float(yt)), fontsize=fontsize,
                    color=color, ha=ha, va="center", zorder=13, linespacing=1.15,
                    arrowprops=dict(arrowstyle="-", color=color, lw=0.7, shrinkA=2, shrinkB=7),
                    bbox=dict(boxstyle="round,pad=0.18", fc=style.SURFACE, ec="none", alpha=0.9))

def zone_box(fld, p, d0, d1, w0, w1, label="", color=style.DEFENSE, alpha=0.12, fs=6.8, ls=(0, (3, 2)),
             label_d=None):
    """A rounded, shaded rectangle from depth d0 to d1 and width w0 to w1 (yards), labelled near its top."""
    (x0, y0), (x1, y1) = P(fld, p, d0, w0), P(fld, p, d1, w1)
    lo_x, lo_y = float(min(x0, x1)), float(min(y0, y1))
    for fc, ec, a in ((color, "none", alpha), ("none", color, 1.0)):
        fld.ax.add_patch(FancyBboxPatch((lo_x, lo_y), float(abs(x1 - x0)), float(abs(y1 - y0)),
                                        boxstyle="round,pad=0,rounding_size=1.2", fc=fc, alpha=a if fc != "none" else 0.8,
                                        ec=ec, lw=0.8, ls=ls, zorder=1.5))
    if label:
        say(fld, p, label_d if label_d is not None else d1 - 1.0, (w0 + w1) / 2, label, color=style.INK,
            fontsize=fs, box=False)

def path(fld, p, pts, color=style.DEFENSE, lw=1.4, dash=(0, (3, 2)), z=5, shrink=1.15):
    """A dashed arrow through absolute (d, w) points, stopping just short of the last one (the marker)."""
    pts = [tuple(map(float, q)) for q in pts]
    (a, b), (c, e) = pts[-2], pts[-1]
    n = float(np.hypot(c - a, e - b))
    if n > shrink + 0.2:
        pts[-1] = (c - (c - a) / n * shrink, e - (e - b) / n * shrink)
    xs, ys = P(fld, p, np.array([q[0] for q in pts]), np.array([q[1] for q in pts]))
    fld.ax.plot(xs, ys, color=color, lw=lw, ls=dash, zorder=z, solid_capstyle="round")
    fld.ax.add_patch(FancyArrowPatch((xs[-2], ys[-2]), (xs[-1], ys[-1]), arrowstyle="-|>",
                                     mutation_scale=9, color=color, lw=lw, zorder=z))

def route_line(fld, p, pts, alpha=0.9, lw=1.5, color=style.OFFENSE, ls="-", z=4):
    """A receiver's route through absolute (d, w) points, with an arrowhead (alpha < 1 for 'faint')."""
    xs, ys = P(fld, p, np.array([float(q[0]) for q in pts]), np.array([float(q[1]) for q in pts]))
    fld.ax.plot(xs, ys, color=color, lw=lw, alpha=alpha, zorder=z, ls=ls, solid_capstyle="round")
    fld.ax.add_patch(FancyArrowPatch((xs[-2], ys[-2]), (xs[-1], ys[-1]), arrowstyle="-|>", mutation_scale=9,
                                     color=color, lw=lw, alpha=alpha, zorder=z))

def ghost(fld, p, d, w, label="", color=style.DEFENSE):
    """A hollow, dashed defender: where he stood before he moved."""
    x, y = P(fld, p, d, w)
    ppy = fld.pts_per_yard()
    fld.ax.add_patch(RegularPolygon((float(x), float(y)), 4, radius=gplay.MARKER_R * 1.25, orientation=np.pi / 4,
                                    fc="none", ec=color, lw=1.1, ls=(0, (2, 1.2)), zorder=6))
    if label:
        fld.ax.text(float(x), float(y), label, ha="center", va="center", color=color, fontweight="bold",
                    zorder=6, fontsize=max(4.0, min(8.5, ppy * 0.75)))

def hole(fld, p, d, w, wd, dd, num, text=None, text_at=None, fs=6.6):

    x, y = P(fld, p, d, w)
    fld.ax.add_patch(Ellipse((float(x), float(y)), wd, dd, fc=style.THIRD, alpha=0.16, ec=style.THIRD,

def panel_title(ax, text, fs=8.5):

def draw(p, ax=None, lateral="full", window=(-8.5, 22), width_in=6.5, show_ruler=True, rfs=6.0):

    clean(fld.ax)
    if show_ruler:
        ruler(fld, p, fs=rfs)
    return fld

def two_state(off, pre, post, via=None, rings=(), ax=None, lateral="full", window=(-8.5, 27.0),
              width_in=6.5, rfs=6.0, rush=True, keep=()):
    """The diagram style for drops and rotations: every defender drawn where he ends up, a dashed
    outline where he lined up (the pre-snap picture) and a dashed path between them.
    via = {key: [(d, w), ...]} bends a path through intermediate points."""
    p = Play(off, nickel(post), los=LOS, to_go=None)
    if rush:
        protect(p)
    p.highlight(*rings)
    fld = draw(p, ax=ax, lateral=lateral, window=window, width_in=width_in, rfs=rfs)
    for k, a in pre.items():
        b = post.get(k, a)
        if np.hypot(b[0] - a[0], b[1] - a[1]) > 0.8 or k in keep:
            ghost(fld, p, *a, LAB[k][1])
            path(fld, p, [a] + list((via or {}).get(k, [])) + [b])
    return fld, p

def deep_thirds(fld, p, d0=15.0, d1=26.5, labels=("deep ⅓", "deep middle ⅓", "deep ⅓"), label_d=None,
                fs=6.8):
    edges = [-HALF, -THIRD_W, THIRD_W, HALF]
    for i, (a, b) in enumerate(zip(edges[:-1], edges[1:])):
        zone_box(fld, p, d0, d1, a + 0.6, b - 0.6, labels[i], alpha=0.07 if i % 2 == 0 else 0.14, fs=fs,
                 label_d=label_d)

# ---------------------------------------------------------------- moments of a simulated play
def set_track(p, key, rows):
    """Hand-made keyframes [(t, d, w), ...] for one player (zone reads and pattern-match carries
    that gridiron's straight-line drop() and man() can't express). Call after every other action."""
    kf, t0, t1 = p._compile()
    kf = dict(kf)
    q = p.players[key]
    rows = [(min(t0, -1.0), q.d, q.w), (0.0, q.d, q.w)] + [tuple(x) for x in rows]
    kf[key] = np.array(rows, dtype=float)
    p._compiled = (kf, t0, max(t1, rows[-1][0]))

def shadow(p, key, target, off, lag=0.3, t_on=0.9, t_from=0.0, start=None, t_end=3.6):
    """Keyframes for a defender who leaves `start` (or his alignment) at t_from and settles
    `off` = (down, right) yards from where `target` was `lag` seconds earlier, fully by t_on."""
    q = p.players[key]
    s = np.array(start if start is not None else (q.d, q.w), dtype=float)
    rows = []
    for tt in np.round(np.arange(t_from + 0.05, t_end + 1e-9, 0.05), 3):
        f = min(1.0, max(0.0, (tt - t_from) / max(1e-6, t_on - t_from)))
        f = f * f * (3 - 2 * f)
        tgt = p.position(target, max(0.0, tt - lag)) + np.array(off)
        rows.append((tt, *(s * (1 - f) + tgt * f)))
    return rows

def moment(ax, p, t, lateral, window, rings=(), throw_t=None):
    """Everyone at time t, with faint trails of where they have been and the ball's flight so far."""
    ylo, yhi = sorted((p.ball_y - lateral[0], p.ball_y - lateral[1]))
    fld = Field(ax=ax, orient="vertical", los=p.los, window=window, numbers=False, lateral=(ylo, yhi),
                to_go=None)
    ppy = fld.pts_per_yard()
    inside = lambda d, w: (window[0] + 0.6 <= d <= window[1] - 0.6) and (lateral[0] + 0.6 <= w <= lateral[1] - 0.6)  # noqa: E731
    for k, q in p.players.items():
        if q.pos in ("LT", "LG", "C", "RG", "RT", "DE", "DT"):
            continue
        pts = np.array([p.position(k, tt) for tt in np.linspace(0.0, max(t, 0.0), 40)])
        if t > 0 and np.ptp(pts[:, 0]) + np.ptp(pts[:, 1]) > 0.3:
            xs, ys = P(fld, p, pts[:, 0], pts[:, 1])
            ax.plot(xs, ys, color=style.OFFENSE if q.side == "off" else style.DEFENSE,
                    lw=1.0, alpha=0.55, zorder=3, ls="-" if q.side == "off" else (0, (3, 2)))
    if throw_t is not None and t > throw_t:
        b = np.array([p.ball_position(tt) for tt in np.linspace(throw_t, t, 25)])
        xs, ys = P(fld, p, b[:, 0], b[:, 1])
        ax.plot(xs, ys, color=style.BALL, lw=1.3, ls=(0, (2, 1.5)), zorder=9)
    for k, q in p.players.items():
        d, w = p.position(k, t)
        if not inside(d, w):
            continue
        x, y = P(fld, p, d, w)
        draw_marker(ax, float(x), float(y), q.side, q.label, ppy, ring=style.THIRD if k in rings else None)
    if t > 0.05:
        bd, bw = p.ball_position(t)
        if inside(bd, bw):
            bx, by = P(fld, p, *((bd, bw - 1.0) if t <= (throw_t or 99) else (bd + 1.2, bw)))
            ax.add_patch(Ellipse((float(bx), float(by)), 1.0, 0.62, fc=style.BALL, ec="white", lw=0.5, zorder=11))
    return fld

def arrival(p, key, t_throw):
    """When the thrown ball reaches its receiver (first moment it is within half a yard of him)."""
    for tt in np.arange(t_throw + 0.05, t_throw + 4.0, 0.02):
        if np.linalg.norm(p.ball_position(tt) - p.position(key, tt)) < 0.5:
            return float(round(tt, 2))
    return t_throw + 1.5

def tag(fld, text, fs=6.6, loc="bl"):
    """A caption tucked into a corner of a frame panel."""
    x, ha = (0.03, "left") if loc[1] == "l" else (0.97, "right")
    y, va = (0.03, "bottom") if loc[0] == "b" else (0.97, "top")
    fld.ax.text(x, y, text, transform=fld.ax.transAxes, ha=ha, va=va, fontsize=fs, color=style.INK, zorder=15,

First-and-10 near midfield. Before the snap the broadcast’s high camera shows one safety, alone, 15 yards deep in the middle of the field. The other safety has walked down to eight yards, just outside the tight end, and stands there like an extra linebacker. The offense hands the ball to its running back, who runs into a wall: eight defenders near the line against seven blockers, and the one nobody blocked makes the tackle for a yard. Two plays later, same formation, same defense. This time it’s a fake handoff, the eight defenders lean forward, and the tight end runs straight up the field past the linebackers. The quarterback drops it in behind them for 28 yards. Same coverage both times, and the analyst says, “That’s the deal with Cover 3.” What deal? Why would a defense take a safety out of the deep part of the field, where the big plays happen, and leave one man in the middle? Why does the same structure stop the run so well and leak a seam route? And why, for more than a decade, was the most feared defense in the NFL built almost entirely on this one plain-looking coverage? This chapter takes Cover 3 apart: what each defender does, the three ways a defense rotates into it, where the holes are, and the version that taught it to close them.

Cover 3 in one picture: three deep, four under

Cover 3 is a zone coverage with three deep defenders, each guarding a third of the field’s width, and four underneath defenders, each guarding an area in front of them. With four rushers, that’s all eleven. The two cornerbacks take the outside thirds, one safety takes the middle third, and the other safety comes down to join three linebackers or nickel backs underneath. That last move, a safety coming down, is what the rest of this chapter is about, and it has a name. Here is the picture with the most common version of it, the sky rotation.

Read the picture from the top down, because that’s how the coverage is built.

The deep three. Each cornerback has the deep third on his side, from a few yards inside the numbers out to the sideline, and the free safety has the middle third, from hash to hash and a few yards beyond. Their rule is the oldest one in pass defense: nobody gets behind me. A Cover 3 corner lines up 7 to 9 yards off his receiver with outside leverage and, at the snap, opens his hips toward the sideline and runs back, keeping the receiver in front of him and his eyes, as much as he can, on the quarterback. He’ll give up a five-yard hitch all day. He will not give up the deep ball over his head. The free safety starts 13 to 16 yards deep in the middle and drifts back to 17 or 18, reading the quarterback, ready to break toward whichever side the ball is thrown. He is the post safety: the man in the post, where post routes go, and the reason the middle of the field is “closed.”

Why three, and why these three? Because a third of the field, about 18 yards, is ground a deep defender can cover comfortably from a good starting spot while the ball is in the air, and because the corners are the defense’s best pure cover men and the outside thirds are where a single receiver runs a go route along the sideline. Two deep defenders can split the field too (that’s Cover 2), but then each has 27 yards and nobody stands in the deep middle. Three is the fewest that keeps a defender in the post and a defender over each sideline.

The underneath four. That leaves four players for the short and intermediate areas, and there are seven named zones underneath (two flats, two curls, two hooks and the hole). Cover 3 solves the arithmetic the way all zones do, by giving each underneath defender two zones and an order to play them in:

  • The two curl-flat defenders, one on each side, take the outside underneath areas. Here they are the strong safety on the right and the nickel on the left. A curl-flat defender drops toward the curl area, about 10 to 12 yards deep under the outside receiver, and breaks to the flat when a receiver or the ball goes there. He’s the busiest man in the coverage, and the next section is about him.
  • The two hook defenders, the Mike and Will linebackers here, take the inside areas: they drop to 8 to 10 yards around the hashes, wall off anything crossing the middle, and keep an eye on the running back, who may slip out late. Some staffs call them hook-curl players, because when a receiver runs a curl inside the curl-flat defender, the hook defender widens to take it.

The four rushers don’t cover anything, but they’re part of the deal. Cover 3 is a four-man-rush coverage. If the defense wants to send a fifth rusher and still play three deep, it has to take one of the underneath four away, and then it’s a different call, a three-deep, three-under fire zone, which Zone Blitzes, Fire Zones, and Simulated Pressure builds.

ImportantWhy it exists

Cover 3 exists because it is the cheapest way to guard the whole deep field. Three deep defenders, a third each, means nothing goes over the top, and yet one safety is left over to come down and help against the run or underneath. Two-deep coverages (Cover 2, quarters) need two safeties deep; man coverage needs a defender for every receiver. Cover 3 is the middle path: deep security with only three, an extra body near the line, and every underneath defender facing the quarterback, so the ball is always thrown in front of somebody who can see it and tackle (and somebody who sees a quarterback scramble the moment it starts, which man defenders, turned to their receivers, often don’t).

What it gives up. Four underneath defenders for seven underneath zones, so there are always soft spots, and three deep defenders against an offense that can send four receivers deep. Every hole this chapter draws comes from one of those two numbers.

The curl-flat defender, and why he became a seam-flat defender

If you want to understand Cover 3, watch the curl-flat defender. He has the hardest job in the coverage because he has two zones, the curl and the flat, and the offense knows it.

Here is his problem. He lines up a few yards inside the slot receiver, in an apex between the slot and the tackle, because he also has a job against the run. At the snap the outside receiver (#1) is somewhere outside and in front of him, the slot (#2) right in front of him. If #1 runs a curl at 12 yards and #2 runs to the flat at 3 yards, one defender has to cover both: if he sits under the curl, the flat is open; if he jumps the flat, the curl is open behind him. That’s the whole idea of the offense’s curl-flat concept, and of a dozen others (Dropback Concepts and The Quick Game draw them from the quarterback’s side: he reads this one defender and throws away from him).

Coaches handle the problem in two different ways, and the difference is one of the most important in this chapter.

The spot-drop answer: play the quarterback. The classic curl-flat defender drops to his landmark, about 10 to 12 yards deep, a few yards inside the outside receiver, and reads the quarterback’s eyes. If the quarterback looks at the curl, he’s already under it; if the quarterback looks at the flat, he breaks down on it as the ball is thrown, and since he’s facing the throw, a quick receiver in the flat should gain only a few yards before he arrives. The coaching phrase is “sink and drive”: sink under the curl, drive on the flat. It’s simple, it’s fast to teach, and it keeps the defender’s eyes where they can see a scramble or a throw. Its weakness is everything that runs past him. The slot receiver’s seam route, straight up the field between the hash and the numbers, is not his business in spot-drop: he lets it go by, and the deep defenders have to deal with it.

The match answer: play the slot receiver first. The other approach changes the order of his jobs. A seam-flat defender keys the number 2 receiver from the snap. If #2 goes vertical, the defender walls him (gets his body into the receiver’s inside path so #2 has to run around him; any contact has to come within 5 yards of the line) and then carries him up the seam, running with him like a man defender until the deep help is over the top. If #2 goes to the flat, the defender goes with him. If #2 goes inside, the defender passes him to the hook defender and only then settles under the outside receiver’s curl. Seam first, flat second, curl last.

Why would a defense choose one over the other? Because they lose to different things. The curl-flat defender is beaten by routes that go past him (the seam) or that put two receivers on his two zones at once. The seam-flat defender closes the seam, but he is now following a receiver, so he’s beaten by routes that drag him away from his area (a vertical release that pulls him deep while the outside receiver runs a quick out underneath) and by switches, crossing releases in which #1 and #2 trade places at the snap so the defenders’ rules have to trade them back at full speed. The seam-flat label is also what you’ll hear in fire zones (three under, three deep), where the two outside underneath defenders are called seam-flat players for exactly this reason: with only three underneath, nobody else can protect the seams.

Most NFL Cover 3 today is closer to the seam-flat version, at least against vertical releases. Hold that thought: it’s the heart of the four-verticals section below.

Why defenses love it: the eighth man

Now the payoff. Cover 3 brings one of the two safeties down. Against a passing formation he becomes a curl-flat defender. Against a running formation, he’s something more valuable: an eighth man in the box.

The logic is simple arithmetic, and it’s the arithmetic of every run game in this book (Run Fits and Box Math builds it properly). An offense has at most five linemen plus whatever tight ends and backs stay in to block. If the defense has more men near the line than the offense has blockers, somebody is unblocked, and an unblocked defender usually makes the tackle. That spare man is the plus-one. Two-high coverages keep both safeties 10 to 14 yards deep. They can still fit the run (quarters safeties, in particular, are taught to fill the lane just outside the box as soon as they read run), but they arrive a beat later, from depth, so at the snap the box is seven against seven. Cover 3 puts the eighth man there from the start, and still has three deep.

That down safety also gives the defense a clean answer to the hardest question in run defense: who stops the ball from getting to the outside? In a sky rotation the safety is the force player, the defender whose job is to turn a wide run back inside toward the pursuit. In a buzz rotation he’s the alley player, filling the lane between the force player and the box from the inside out. Behind him the linebackers fit the gaps inside, each knowing that the safety owns the edge or the lane next to it (Run Fits and Box Math draws the full fits). Either way, the coverage and the run fit come from the same call. That’s why Cover 3, and its man-coverage cousin Cover 1, were for decades the default on first-and-10: one call that defends the run with eight and still guards the deep ball with three.

The data shows the pattern. On dropbacks charted from 2018 to 2022, Cover 3 was the call on 38 percent of first downs but only 19 percent of third-and-4-to-6 and 19 percent of snaps in the red zone, against 34 percent elsewhere on the field. Defenses reach for it on early downs, when they have to respect the run, and turn to man coverage when the field is short or the down demands a tight cover: Cover 1 or Cover 0 was the call on about half of those third-and-medium and red-zone dropbacks, against 28 percent elsewhere.1

Where’s the eighth man, then? An average of 6.3 isn’t eight, because this chart is all dropbacks, most of them against three-receiver sets, where the down safety is out over a slot as a curl-flat defender rather than in the box. The eighth man shows up against run formations. On the same early downs, when the offense had two or fewer wide receivers on the field, Cover 3 had eight or more defenders in the box on about a quarter of its dropbacks; Cover 2 did on 1 in 20 and quarters almost never.2 The point isn’t that Cover 3 always has eight; it’s that it can, at no cost to its deep coverage, whenever the formation calls for it.

Cover 3 is still a zone with every underneath defender facing the ball, so completions come at about the two-high rate, mostly in front of the defense. Cover 1 allows far fewer completions (about 60 percent) but more 20-yard plays: man coverage contests every catch, but when a defender is beaten there’s nobody behind him except one safety. Cover 3 gives up more 20-yard plays than either two-high zone, about one dropback in ten against fewer than one in eleven for Cover 2 and quarters. Those extra explosive plays are the seams and the single safety, the holes the rest of this chapter is about. (These are observational numbers: defenses choose coverages for situations, so this shows what each coverage looks like when teams call it, not what would happen if they swapped one for another.)

CautionCommon misconception

“Three deep defenders means Cover 3 is safe against the deep ball.” It’s safe against a deep ball along the sideline and down the middle, where the three deep defenders start. It’s thinnest exactly between them, up the seams (hole 1 below). Four receivers going deep against three deep defenders is the oldest Cover 3 problem there is, and the deep defenders can’t solve it alone. The underneath defenders have to help, and how they help is the difference between the 1980s version of the coverage and the modern one.

Sky, buzz and cloud: three ways to get to Cover 3

Very little Cover 3 in the NFL is simply lined up and played. The defense usually arrives at it, because the shell it shows before the snap is part of the plan: it wants the quarterback guessing, and it wants its run fits to match the formation. The three classic ways to arrive have names that describe which defender drops into the extra underneath spot, and they come with a handy memory aid: sky is the safety to the flat, cloud is the corner to the flat, and buzz is the safety to the hook, buzzing around inside.

All three end with the same skeleton, three deep and four under. What changes is who plays which zone, and that matters because players aren’t interchangeable: a safety coming from depth, a linebacker already outside, and a cornerback already pressing a receiver are each good at different things. (Disguise, Rotation, and Split-Field Coverage shows these rotations launched from a two-high shell at the last instant; here we start each one from the alignment where it is most natural, so you can see what it’s for.)

Sky: the safety takes the flat

The sky rotation is the one in the first figure. The safety on the rotation side comes down and out at the snap to become the curl-flat defender. The corner on that side bails to his deep third as usual, and the free safety goes to the middle.

Which side is the rotation side? Whatever the call’s rule says. The most common rule is “rotate to the strength,” and for a coverage, strength usually means the passing strength, the side with more receivers (Coverage Fundamentals makes the distinction from formation strength, which counts the tight end). In a 2x2 like the first figure the receiver count is even, so the defense needs a tiebreaker, and staffs pick different ones: the tight end’s side (as drawn), the field side or the boundary side, or the side the running back sets to. Some calls rotate away from the strength on purpose (a weak rotation; more on the names in the box below). Because the rule is tied to the formation, motion that changes the formation can change the rotation, and that is how an offense finds out which way it goes.

Why sky? Because it puts a safety, a player who is both a good tackler and fast enough to cover, at the point of the field where the run game and the quick passing game meet. Against a run toward him he’s the force player, coming downhill from outside in. Against a pass he’s the curl-flat defender, reading #2 and #1. Teams with a big, physical strong safety (Kam Chancellor in Seattle is the archetype, below) build their defense around this.

What sky gives up. The safety has a long way to go if he starts deep, so the quick game to his side (a hitch, a bubble screen to the slot, a quick out) can be thrown before he arrives, which is why teams that sky from a deep alignment often “cheat” him down before the snap. That’s a tell. And with the strong safety committed to the flat, the free safety is alone in the middle, with no help on either seam.

In the buzz rotation the safety comes down inside, to the hook-curl area, and the player already lined up outside on that side (a nickel or an outside linebacker) takes the curl-flat. Compared with sky, the two defenders on the rotation side trade jobs: the safety goes inside, the outside man goes to the flat. The other side doesn’t change. That’s also why the nickel has moved in the figure. Defenses put the nickel, a defensive back, over the slot receiver and let a linebacker take the tight end; in the first figure the receiver slot was on the left, here it is on the right, and buzz needs that outside defender standing on the rotation side.

Why buzz? Three reasons. First, the flat defender is already standing near the flat, so the quick game to that side is covered from the snap instead of from three seconds later. Second, a safety coming down from depth and inside is perfectly placed to wall the slot receiver’s seam: he arrives on the inside of it, facing it, which makes buzz a good answer to the seam route that haunts Cover 3. Third, against the run he is the alley player, filling inside the force defender rather than outside him, which suits a safety who is a better downhill tackler than a cover man in space.

What buzz gives up. The defender on the edge is now the nickel or linebacker, usually smaller or slower than the safety, so the force against outside runs is weaker on that side. The hook defender who slides across has ground to cover if the offense runs away from the rotation. And the safety’s landing spot, the hook-curl area, is close to where a robber would sit, which makes buzz and Cover 3 robber easy to confuse on a broadcast. The difference: a robber sits in the middle of the field and jumps in-breaking routes, a buzz safety walls the seam on his side and widens to the curl (Man Coverage covers the robber).

The cloud rotation, also called a corner rotation, flips the roles of the corner and the safety on one side. The corner stays shallow and plays the flat; the safety on that side takes the deep outside third. On the other side it’s ordinary Cover 3, and the other safety still goes to the middle.

Why cloud? Look at the picture before the snap: a corner squatting at 5 yards with a safety deep behind him is exactly what a Cover 2 side looks like, so a cloud rotation is a natural disguise, a two-high shell that becomes three-deep. It also puts the defense’s best cover man, already near the receiver, on the quick throws to that side: he can re-route #1 (from 5 yards off he gets in the receiver’s path rather than jamming him; teams that want a real jam press the cloud corner at a yard or two) and sit on the hitch, the quick out and the flat. And it makes him the force player against outside runs, so a defense can cloud to the side where it expects the offense to throw quick or run wide. Defenses like it to the boundary (the short side of the field), where the deep third is a shorter run for a safety. Every diagram in this chapter puts the ball in the middle; with the ball on an NFL hash, the boundary is about 23.6 yards from the ball to the sideline against 29.8 on the field side, so a deep third measured from the ball is about 3 yards narrower on the boundary. That’s a modest gift in the NFL, whose hashes sit only 3 yards off the middle, and a large one in college and high school, whose hashes are much wider (The Game in Fifteen Minutes draws the difference).

What cloud gives up. The deep outside third now belongs to a safety, who is usually not as fast or as practised at running with a wide receiver as the corner. A go route by a fast #1 against a safety who started 12 yards deep and several yards inside him is the classic cloud beater, and the smash corner route (a route that breaks toward the sideline behind a squatting corner) attacks exactly the spot between the cloud corner and the safety, just as it does against Cover 2.

ImportantWhy it exists: three rotations for one coverage

A defense wants to play Cover 3 against many formations without telling the quarterback which defender is the extra underneath player, and it wants that player to be the right kind of player for the job. Each rotation moves a defender who is good at one thing into a job he’s merely adequate at, and each leaks a hint before the snap.

NoteMadden vs. real life

In Madden, “Cover 3 Sky,” “Cover 3 Buzz” and “Cover 3 Cloud” are separate plays on the menu, and you pick one. Real defenses rarely call them that way. The call is usually “three” plus a rule for the rotation (“rotate to the strength,” “cloud the boundary,” “buzz to the tight end”), and the players work out the rest from the formation: which side is strong, which is the boundary, where the motion went. That’s why the same call can be sky on one snap and buzz on the next, and why a defensive back’s first job before every play is to say out loud which way the rotation goes. In Nick Saban’s system the rotating safety’s declaration is literally a word: “Rip” or “Liz,” for the side he is coming down to.3

The names are dialects, too. “Sky” and “cloud” are also used for the two sides of split-field coverages (the “cloud side” of a Cover 6, a coverage that plays quarters on one half of the field and Cover 2 on the other, is the side where the corner sits in the flat), and a Cover 3 whose rotation goes away from the strength, a weak rotation, is Saban’s “6” and some NFL staffs’ Cover 9, as Coverage Fundamentals explained. When you hear a coverage name, ask which rule it describes, not which picture.

1. The seams. The boundaries between the thirds sit about 9 yards either side of the middle of the field, between the hashes and the numbers, which is precisely where a slot receiver or a tight end runs a seam route. A seam thrown 18 to 25 yards downfield lands between a corner who is guarding the sideline and a free safety who is guarding the middle, and if both receivers run seams, the free safety can take only one. This is four verticals’ target, and the next section draws it. Against match: mostly closed, because the seam-flat defenders carry the seams; the cost moves underneath, to the flat they leave.

2. The curl-flat stretch. The curl-flat defender guards two zones, so two receivers, one in each, force him to choose: the curl-flat concept, slant-flat, stick, snag. The quarterback reads him and throws to the receiver he didn’t take. The hook defenders help on the curl only if they’re not needed elsewhere, which is why offenses add a third route that holds them inside (a check-down by the back or a crossing route). Against match: smaller. The seam-flat defender goes with #2 to the flat by rule, and the corner, seeing #1 break short, sinks and then drives on the curl from above, so both routes have a defender; the curl is still the softer of the two, because the corner arrives from depth.

3. The flood. Put three receivers on one side at three depths (a go route that takes the corner deep, a sail at 12 to 15 yards, a flat route) and there are two defenders to cover them, the corner and the curl-flat defender. One is always open. The sail, landing behind the curl-flat defender and in front of the bailing corner, is the money throw against spot-drop Cover 3, and some version of flood is in every NFL playbook. Against match: still open. Matching changes who is stretched, not how many defenders are on that side (the third drill at the end of the chapter works through it).

4. In front of the corners. A Cover 3 corner’s first rule is to stay deeper than the deepest receiver in his third, so he bails at the snap. A hitch at 5 or 6 yards, or a quick out, is caught in front of him. The curl-flat defender is the nearest help, and in the first figure his zone seems to cover the spot, so why is it open? Timing. The hitch is thrown before the receiver turns around, at a spot, while the curl-flat defender is still doing his first job: sinking toward the curl with his eyes on the quarterback. He can’t stop, turn and close 5 yards in the time the ball is in the air, so the best he and the corner can do is arrive with the catch and tackle. It’s not a big play, but it’s a completion every time the defense lets it be, which is why offenses throw it to “keep the defense honest” and why a corner who cheats up to stop it opens the deep ball. There’s a fifth, quieter soft spot you’ll hear analysts mention: the window between the hook defender and the curl-flat defender at 10 to 14 yards, where a dig sits down. And there’s the run game’s answer: a play-action fake that pulls the eight-man front forward and throws behind it, as in this chapter’s opening. Cover 3 brought a safety down to stop the run, and play-action punishes exactly that.

Spot-drop Cover 3 against four verticals is a math problem the defense loses: four receivers past 15 yards, three defenders back there, and the one in the middle can be moved by the quarterback’s eyes. Or the coverage can stop being a spot-drop zone and start matching the routes.

Cover 3 match: carrying the seams

Cover 3 match keeps the Cover 3 shell, three deep and four under, but gives every defender a rule about which receiver to take once the routes declare. It is the pattern-matching idea from Coverage Fundamentals applied to Cover 3, and it’s the version most NFL teams play today. Nick Saban, building it with Bill Belichick in Cleveland in the early 1990s, is the coach most often credited with turning Cover 3 into a match coverage, and his later college defenses spread the idea; like most origin stories it’s disputed, and other staffs built similar rules.5 The rules differ from staff to staff, but the core of most versions fits in a table.

Defender Reads Rule (simplified)
Cornerbacks #1 #1 goes vertical: carry him, on top, like man. #1 breaks short: sink, look for #2 coming outside, then drive on #1.
Seam-flat defenders ($, SS) #2, then #1 #2 vertical: wall him, carry him up the seam until the post safety is over the top. #2 out: take him in the flat. #2 in: pass him inside, sink under #1.
Hook defenders (M, W) #3 (the back), then the crossers The back releases: take him. No release: wall the crossers and sink into the hole.
Free safety the #2 away from the rotation (in many versions), or both #2s; the quarterback Stay in the middle third; play over the top of the seam he’s assigned or the quarterback looks at, or the deepest route in the middle.

The key line is the seam-flat defender’s: #2 vertical, carry him. That single rule turns four verticals from a math problem into a matchup. Staffs split on the far seam, though. Some carry both #2s, as drawn below. Many, Seattle among them, carry only the seam on the rotation side and give the other one to the post safety, who leans toward it before the ball is snapped. That keeps the other curl-flat defender home for the flat and the curl, at the price of asking more of the free safety.

ImportantWhy it exists: Cover 3 match

The problem it solves. Spot-drop Cover 3 had a structural problem that no amount of effort fixed: four verticals, and more generally any concept that puts two receivers into one deep defender’s area. Matching the routes gives every vertical receiver a defender, without giving up what makes Cover 3 attractive: three deep, the extra man near the line before the snap, and defenders who start out facing the quarterback.

What it gives up. Simplicity, and the empty spaces the carries leave behind. A seam-flat defender running up the field with #2 is not in the flat, so a quick out by #1, or the back releasing to the flat, is open underneath him. And any route combination that makes two defenders’ rules collide becomes a target: post-wheel (the corner carries the post inside while the wheel goes up the sideline behind the seam-flat defender), switch releases that cross #1 and #2 at the snap, and motion that changes who is #2 at the last moment. The match coverage’s rules close the seams; offensive coaches now design plays to open the gaps between the rules.

A switch release crosses #1 and #2 at the snap, the outside receiver running inside and the slot running outside, so the corner and the seam-flat defender, each reading “his” number, have to decide in the first step who is now #1 and who is #2; if they hesitate or both take the same man, the other one is free. The trips problem

Match rules work cleanly against 2x2, two receivers each side. Against 3x1 (trips: three receivers to one side), Cover 3 has a counting problem: four underneath defenders, but three receivers on one side whose routes can all go vertical. If #3, the innermost trips receiver, runs straight up the middle, who carries him? In spot-drop it’s the free safety, and he can’t be on #3 and over #2’s seam at the same time, so the trips side’s seam falls to the corner and the curl-flat defender, who are guarding other things. Match teams answer with a trips check, a pre-planned adjustment (the two-high versions are the 3x1 checks of Disguise, Rotation, and Split-Field Coverage). The check doesn’t add a defender; it moves one. In one common version the backside corner takes the lone receiver on the other side by himself, man to man wherever he goes (Saban’s playbook calls this MEG, “man everywhere he goes”). That frees the defenders who would have helped on the backside to slide toward the trips, where the trips-side corner and the free safety stay the deep players and one of the freed underneath defenders (which one varies by staff) takes #3 if he goes vertical, so the free safety can stay over #2. Saban’s system names its trips checks (one is “Mable,” in which the safety comes down toward the trips), and every NFL staff has its own.6 The tell for a viewer: against trips, look at the backside corner. If he’s pressing the single receiver, alone, with no safety leaning his way, the defense has probably checked its Cover 3 to put its help on the trips side. (Disguise, Rotation, and Split-Field Coverage covers the two-high answers to trips, including rotating to Cover 3 match.)

Seattle’s version: Cover 3 as an identity

For most of the 2000s, Cover 3 was something defenses played on early downs between the more fashionable calls. Then Pete Carroll’s Seattle Seahawks made it the most feared defense in football, mostly by showing one pre-snap picture, a single high safety and corners pressed on the receivers, and playing two coverages behind it: Cover 3 and its man twin, Cover 1 (“man free”).7 Coverage Fundamentals met this defense in its film room; here is why it worked as a scheme.

Carroll had spent decades as a defensive backs coach, coordinator and head coach before Seattle hired him in 2010; Gus Bradley ran the defense from 2009 to 2012 and Dan Quinn in 2013 and 2014.8 Carroll’s idea, laid out in his book Win Forever, was that a program needs one clearly defined philosophy and should try to do things “better than they have ever been done before.”9 On defense, that meant a 4-3 front with a one-high safety and five distinctive pieces:

  1. A free safety with rare range. Earl Thomas, the 14th pick of the 2010 draft, covered the deep middle third alone, and fast enough to arrive at either seam. Cover 3 lives or dies with its post safety; Seattle’s could erase mistakes in front of him.
  2. A strong safety built like a linebacker. Kam Chancellor, a fifth-round pick that same year, was the eighth man in the box: the sky or buzz player who came down at the snap, the force or alley defender against the run, and a seam-flat defender who could carry a tight end up the field.
  3. Tall corners who pressed and carried. Richard Sherman and Brandon Browner (both 6 feet 3 or taller, a fifth-round pick and a Canadian Football League signing)10 lined up on receivers’ noses, jammed them inside the 5-yard limit with Carroll’s step-kick technique, and if the receiver went deep, ran with him into the corner’s own deep third. That press looked exactly like man coverage, which is the point: before the snap the quarterback couldn’t tell Cover 3 from Cover 1. The difference shows only after it, when a receiver breaks inside. In Cover 1 the corner follows him across the field; in Cover 3 he lets him go and sinks.
  4. Match rules on the verticals. The corners carried #1’s vertical release like man defenders, the down safety matched a seam route from his underneath spot, and the free safety leaned toward the other seam (the “both versions” note in the match section), so four verticals met four defenders.11
  5. Repetition. With few calls, the players practised the same reads thousands of times, and rarely blew an assignment. The scheme was simple enough to play fast.

The results were extraordinary. Seattle allowed the fewest points in the NFL four seasons running, 2012 to 2015, the first team to do so since the Cleveland Browns of the 1950s.12

The tree

Carroll’s assistants took the scheme with them, and for a while Seattle-style Cover 3 was the NFL’s fastest-spreading defense. Bradley went on to coach or coordinate four more defenses (Jacksonville, the Chargers, the Raiders and the Colts); Quinn took Atlanta to Super Bowl LI as head coach and is Washington’s head coach today.14 Robert Saleh, a junior quality-control assistant in Seattle (the entry-level staffer who breaks down film), built his first San Francisco defenses on the same eight-man-front Cover 3 before mixing in more coverages.15 The tree is still growing: Saleh, the Tennessee Titans’ head coach in 2026, hired Bradley as his defensive coordinator.

That is the Carroll philosophy, and it has a modern cost: an offense that knows the coverage before the snap can call the concept that beats it. In the same years, Vic Fangio’s two-high defenses were showing quarterbacks a different picture on every snap (Coverage Fundamentals tells that story), and the league’s split-safety rate climbed every year. The coach who made it famous has moved on too. Carroll coached Seattle through the 2023 season and the Raiders in 2025, when they went 3–14 and dismissed him after one year. Seattle won Super Bowl LX in February 2026 under Mike Macdonald, with a split-safety defense from a different school (from Week 14 through the Super Bowl, two-high coverage on 54.8 percent of pass plays) (Fangio’s Two-High Revolution and Macdonald’s Answer).18 But nearly every NFL defense still plays Cover 3 on some early downs, usually as a match coverage, usually arrived at by one of the three rotations above. The Seattle version didn’t disappear; it became part of everyone’s playbook.

Recognising Cover 3: before and after the snap

Everything above adds up to a two-step read you can make from your couch: a hypothesis before the snap, a test one second after it. Here is the reasoning behind each check; the checklist at the end of the section is the short version.

Before the snap: the Cover 3 look.

  • One safety deep in the middle, 13 to 16 yards, roughly over the ball. Why it matters: the middle of the field is closed, which narrows the menu to the one-high coverages, Cover 3 and Cover 1.
  • Corners off and turned in, 7 to 9 yards from their receivers with outside leverage, shoulders angled toward the quarterback. Why: a corner who must stay deeper than everyone in his third starts deep and watches the ball; a man corner is more often pressed and square to his receiver. Seattle-style corners who press and then carry are the exception, and you check them after the snap.
  • A safety down near the box, 7 to 11 yards deep. Why: he is the fourth underneath defender and the eighth man against the run. Where he stands hints at the rotation: outside the slot or tight end suggests sky; inside, over the slot, with a nickel outside him, suggests buzz.
  • Underneath defenders in apex alignments, a few yards inside the slot receivers, not tight on them. Why: zone defenders split the difference between a run fit and a pass drop; man defenders line up on their man.
  • The cloud tell: a two-high look in which only one corner squats at 5 yards and the other is off, often with the free safety shaded toward the middle. Why: a real Cover 2 usually squats both corners; the off corner is already in a deep-third alignment. (Asymmetry is also how Cover 6 looks, so check after the snap.)
  • Motion. When a receiver motions across, watch the safeties. If they spin with him, one coming down on the new strong side and the other heading for the middle, the call is a single-high rotation tied to the strength; if a defender runs across with the motion man, think man coverage.

After the snap: count the deep defenders, at one second. About one second after the snap, look at the deep part of the field and count the defenders backpedalling or running away from the line, spread across the width. Three means a Cover 3 family call (Cover 3 or a three-deep fire zone); two means Cover 2 or a relative; four means quarters; one means Cover 1 (with or without a robber, it’s still Cover 1). Two refinements make the count reliable:

  • Count early. In Cover 3 match, the seam-flat defenders who carry slot receivers up the seams are 15 to 20 yards deep two seconds later and look like extra deep players (count the last panel of the match strip and you get five). The count means something only before the carries.
  • Check who the three are. In Cover 3, two of them are the corners, deep and wide. If the corners are sitting in the flats and the middle man is a linebacker running down the middle of the field, it’s Tampa 2, which also shows three deep at one second.

Then look underneath. Corners who keep their eyes inside and let a receiver who breaks short or across go are zone; corners who turn and follow him across the field are man. A safety into the flat is sky, a safety into the hook-curl is buzz, a corner sitting in the flat with a safety running over him is cloud. Slot defenders who run away from a vertical release to a spot are spot-drop; slot defenders who run with him up the seam are match. Four rushers and three deep is Cover 3; five rushers and three deep, with a defensive lineman dropping into coverage, is a fire zone (Zone Blitzes, Fire Zones, and Simulated Pressure). And for how often a one-high picture before the snap is honest, and how to keep score of a defense that lies, see Disguise, Rotation, and Split-Field Coverage.

TipWatch for it on Sunday: count to three

For one quarter, on every passing down:

  • Before the snap: one high safety, corners off and turned in, a safety near the box? Say “Cover 3” and guess sky (safety outside) or buzz (safety inside, nickel outside). Two-high with one squatting corner? Say “cloud?”

Who’s the quarterback’s read, what has that defender done, and where does the ball go?

TipAnswer

The read is the strong safety, the curl-flat defender, and he has sunk under the sail route, so the ball goes to the back in the flat. This is the flood: three receivers at three depths on the right (the go, the sail, the flat) against two defenders, the corner and the curl-flat defender. The corner did his job, staying on top of Z’s go route, which leaves the safety alone with two receivers. He’s at about 10 yards, wide, in the window where the sail will arrive, so the back, running to the flat underneath him with nobody within 7 yards, is the throw. Had the safety jumped down toward the flat, the sail behind him would have been open instead. (Notice the Mike in the frame, too: with #2 going outside and the back heading for the flat, he has widened from his hook toward the curl, as he’s taught, and he still can’t reach the flat in time.)

If this were Cover 3 match, the picture would differ only a little: with #2 (Y) releasing vertically the safety would have carried him at first, and when Y broke out to the sail he’d stay with him, while the Mike, who matches the back (#3), would have a long way to run to the flat. The flat would still be the throw. Flood beats both versions for the same reason. There are only so many defenders on one side of the field.

Takeaways

  • Cover 3 is three deep and four under. The corners take the outside thirds and a safety the middle; two curl-flat defenders and two hook defenders take the underneath areas; four rush. It guards the whole deep field with only three players, which is why it’s the NFL’s most common zone.
  • Its bargain is the eighth man. Bringing the second safety down gives the defense an extra run defender (and its force or alley player) while still playing three deep. The price is the seams, the soft spots in front of the corners, and an underneath defender with two zones.
  • Sky, buzz and cloud are three ways to arrive at the same coverage. Sky sends a safety to the curl-flat, buzz sends him inside to the hook-curl while an outside defender takes the flat, and cloud puts the corner in the flat with a safety over him. Each fits a different player to the job and leaks a different pre-snap tell.
  • The curl-flat defender is the key, and the match version changed his job. In spot-drop he guards a place and lets the seam go by; as a seam-flat defender in Cover 3 match he carries #2 up the seam first, which turns four verticals from an unsolvable count into four-on-four.
  • Seattle made it an identity. Carroll’s 2010s Seahawks showed one picture (a single high safety, pressed corners) and played Cover 3 or its man twin Cover 1 behind it, with a rangy free safety, a box strong safety and long corners who pressed and carried receivers into their thirds. They led the league in scoring defense four years running and sent coordinators across the league; Bradley’s defenses still topped the Cover 3 charts years later.
  • Recognise it by counting. One high safety, corners off and turned in, a safety near the box before the snap; three deep defenders one second after it, two of them the corners (count before the match carries, and don’t mistake Tampa 2’s running Mike for a third deep defender).
