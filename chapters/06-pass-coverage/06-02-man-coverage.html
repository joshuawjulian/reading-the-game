def O(key, pos, d, w, label=None):

def line5():
    """Five offensive linemen, 1.6 yards apart (only the center is lettered)."""
    return [O(k, k, LINE_D, w, "C" if k == "C" else "") for k, w in
            (("LT", -3.2), ("LG", -1.6), ("C", 0.0), ("RG", 1.6), ("RT", 3.2))]


def spread(rb_w=2.0, x_w=-15.0, h_w=-9.0, y_w=9.0, z_w=15.0):
    """11 personnel, shotgun 2x2: X and Z on the line (with the five linemen, seven on the line);
    H in the left slot and Y, a tight end split out, in the right slot, both a step off the ball."""
    return line5() + [O("QB", "QB", GUN, 0.0, "Q"), O("RB", "RB", GUN, rb_w, "R"),
                      O("X", "WR", LINE_D, x_w), O("H", "WR", OFF_BALL, h_w),
                      O("Y", "TE", OFF_BALL, y_w), O("Z", "WR", LINE_D, z_w)]


def dline():
    """Four down linemen (a four-man rush), strength to the offense's right."""
    return [D("WDE", "DE", 1.0, -5.4, "E"), D("WDT", "DT", 1.0, -0.8, "T"),
            D("SDT", "DT", 1.0, 2.4, "T"), D("SDE", "DE", 1.0, 5.4, "E")]


def moved(players, extra=None):
    extra = extra or {}
    return [q.moved(d=extra[q.key][0], w=extra[q.key][1]) if q.key in extra else q for q in players]


def cover1_def(extra=None):
    """Nickel Cover 1 against 2x2: corners pressed with a slight outside shade (their help is
    inside), the nickel ($) on H, the strong safety on the tight end, the Will on the back, the
    Mike free, the free safety alone in the deep middle."""
    return moved(dline() + [D("MIKE", "MIKE", 4.5, -1.8, "M"), D("WILL", "WILL", 4.5, 2.4, "W"),
                            D("NB", "NB", 4.0, -9.8, "$"), D("SS", "SS", 5.0, 9.8, "SS"),
                            D("LCB", "CB", 1.2, -15.9, "CB"), D("RCB", "CB", 1.2, 15.9, "CB"),
                            D("FS", "FS", 14.0, 0.0, "FS")], extra)


def zero_front():
    """Six at the line for an all-out pressure: four linemen widened a little, and both linebackers
    walked up into the A gaps beside the center (a 'double A-gap mug')."""
    return [D("WDE", "DE", 1.0, -5.8, "E"), D("WDT", "DT", 1.0, -3.0, "T"), D("SDT", "DT", 1.0, 3.0, "T"),
            D("SDE", "DE", 1.0, 5.8, "E"), D("MIKE", "MIKE", 1.4, -0.95, "M"), D("WILL", "WILL", 1.4, 0.95, "W")]


def protect(p, rushers=("WDE", "WDT", "SDT", "SDE"), linemen=("LT", "LG", "C", "RG", "RT")):
    """Linemen set, rushers engage them: enough to say 'pass', no more."""
    for k in linemen:
        p.block(k, pts=[(-1.2, 0.0)])
    for k in rushers:
        p.rush(k, pts=[(-1.6, 0.0)], speed=2.0)


def route_to(p, key, *pts, delay=0.0, speed=None):
    """A route through absolute (d, w) points."""
    start = np.array(p._cur(key))
    return p.route(key, [tuple(np.array(q) - start) for q in pts], frame="field", delay=delay, speed=speed)

def P(fld, p, d, w):
    return fld.xy(*p.to_abs(d, w))


def clean(ax):
    """Drop the painted yard numbers (they sit where receivers stand); clip all text to the window."""
    for t in list(ax.texts):
        if t.get_text().isdigit() and t.get_alpha() == 0.55:
            t.remove()
        else:
            t.set_clip_on(True)

    """Depth labels (yards past the line of scrimmage) just inside the left edge of the window."""
    xl = FW - fld.y1 + 0.4
    for d in range(step, top + 1, step):
        y = p.los + d
        if fld.x0 + 1.5 <= y <= fld.x1 - 1.5:
            fld.ax.text(xl, y, f"{d}" + (" yd" if d == step else ""), fontsize=fs, color=style.INK_2,
                        ha="left", va="center", zorder=12,

# ---------------------------------------------------------------- motion: keyframes, frame strips, video
def set_tracks(p, tracks, t_end=None):
    """Hand-made keyframes {key: [(t, d, w), ...]} for some players, after all routes are added.
    gridiron's man() keeps a shrinking cushion that ends on top of the receiver (which hides him in
    print) and cannot switch receivers, so man defenders here are keyframed by hand."""
    kf, t0, t1 = p._compile()
    kf = dict(kf)
    for key, rows in tracks.items():
        q = p.players[key]
        rows = [(min(t0, -1.0), q.d, q.w), (0.0, q.d, q.w)] + [tuple(r_) for r_ in rows]
        kf[key] = np.array(rows, dtype=float)
    p._compiled = (kf, t0, max(t1, t_end or 0.0, *[rows[-1][0] for rows in tracks.values()]))


def chase(p, key, target, off, lag=0.25, t_on=0.7, t_end=3.6, t_start=0.1):
    """Keyframes for a man defender who settles `off` = (down, toward the offense's right) yards
    from where his receiver was `lag` seconds earlier."""
    q = p.players[key]
    rows = []
    for tt in np.round(np.arange(t_start, t_end, 0.05), 3):
        f = min(1.0, max(0.0, (tt - t_start) / (t_on - t_start)))
        f = f * f * (3 - 2 * f)
        tgt = p.position(target, max(0.0, tt - lag)) + np.array(off)
        rows.append((tt, *(np.array([q.d, q.w]) * (1 - f) + tgt * f)))
    return rows


def ease(points, step=0.05):
    """Smooth keyframes through (t, d, w) waypoints (ease in and out on each leg)."""
    rows = []
    for (ta, da, wa), (tb, db, wb) in zip(points[:-1], points[1:]):
        for tt in np.arange(ta, tb, step):
            f = (tt - ta) / (tb - ta)
            f = f * f * (3 - 2 * f)
            rows.append((round(float(tt), 3), da + (db - da) * f, wa + (wb - wa) * f))
    rows.append(tuple(points[-1]))
    return rows


def frame(ax, p, t, lateral, window, rings=(), ltg=True):
    """Everyone at time t, with faint trails of where they have been and the ball's flight so far."""
    ylo, yhi = sorted((p.ball_y - lateral[0], p.ball_y - lateral[1]))
    fld = Field(ax=ax, orient="vertical", los=p.los, window=window, numbers=False, lateral=(ylo, yhi),
                to_go=p.to_go if ltg else None)
    ppy = fld.pts_per_yard()
    wl, wr = sorted(lateral)
    ring = dict(rings)
    for k, q in p.players.items():
        d, w = p.position(k, t)
        if not (window[0] + 0.4 < d < window[1] - 0.4 and wl + 0.4 < w < wr - 0.4):
            continue
        if t > 0:
            pts = np.array([p.position(k, tt) for tt in np.linspace(0.0, t, 40)])
            if np.ptp(pts[:, 0]) + np.ptp(pts[:, 1]) > 0.3:
                xs, ys = P(fld, p, pts[:, 0], pts[:, 1])
                ax.plot(xs, ys, color=style.OFFENSE if q.side == "off" else style.DEFENSE, lw=1.0,
                        alpha=0.5, zorder=3)
        x, y = P(fld, p, d, w)
        draw_marker(ax, float(x), float(y), q.side, q.label, ppy, ring=ring.get(k))
    for kind, to, te in p.ball_events:
        if kind == "pass" and t > te:
            ts = np.linspace(te, t, 20)
            b = np.array([p.ball_position(tt) for tt in ts])
            xs, ys = P(fld, p, b[:, 0], b[:, 1])
            ax.plot(xs, ys, color=style.BALL, lw=1.4, ls=(0, (2, 1.5)), zorder=9)
            ax.add_patch(Ellipse((float(xs[-1]) - 0.9, float(ys[-1])), 1.0, 0.62, fc=style.BALL, ec="white",
                                 lw=0.5, zorder=11))
    clean(ax)
    return fld


def strip_grid(p, times, notes, lateral, window, rings=(), ncols=2, width_in=6.5, deco=None, fs=6.9):
    """Print stand-in for an animation: numbered panels, each with a sentence underneath."""
    nrows = int(np.ceil(len(times) / ncols))
    pw = width_in / ncols
    ph = pw * (window[1] - window[0]) / abs(lateral[1] - lateral[0])
    fig, axes = plt.subplots(nrows, ncols, squeeze=False, figsize=(width_in, nrows * (ph + 0.75)))
    for ax in axes.flat[len(times):]:
        ax.axis("off")
    for i, (ax, t) in enumerate(zip(axes.flat, times)):
        fld = frame(ax, p, t, lateral, window, rings)
        if deco:
            deco(i, fld, p)
        panel_title(ax, f"{i + 1}.  " + ("snap" if abs(t) < 1e-6 else f"+{t:.1f} s"))
        under_note(ax, notes[i], fs=fs)
    fig.tight_layout(h_pad=3.2, w_pad=0.8)
    return fig


def strip_rows(plays, names, times, notes, lateral, window, rings=(), width_in=6.5, fs=6.6, deco=None):
    """One row per play, one column per moment: two versions of a play on the same clock."""
    nc = len(times)
    pw = width_in / nc
    ph = pw * (window[1] - window[0]) / abs(lateral[1] - lateral[0])
    fig, axes = plt.subplots(len(plays), nc, squeeze=False, figsize=(width_in, len(plays) * (ph + 0.8)))
    for i, (p, nm) in enumerate(zip(plays, names)):
        for j, t in enumerate(times):
            fld = frame(axes[i, j], p, t, lateral, window, rings)
            if deco:
                deco(i, j, fld, p)
            panel_title(axes[i, j], f"{nm}: " + ("snap" if abs(t) < 1e-6 else f"+{t:.1f} s"), fs=8)
            under_note(axes[i, j], notes[i][j], fs=fs)
    fig.tight_layout(h_pad=3.0, w_pad=0.6)
    return fig


def video(plays, names, lateral, window, t_end, rings=(), width_in=6.0, t_start=-0.4):
    """Web version: the plays side by side, on one clock."""
    from IPython.display import HTML, display
    n = len(plays)
    span_w, span_d = abs(lateral[1] - lateral[0]), window[1] - window[0]
    fig, axes = plt.subplots(1, n, figsize=(width_in, width_in / n * span_d / span_w + 0.5), squeeze=False)
    ts = np.round(np.arange(t_start, t_end + 1e-9, 0.1), 2)

    def update(i):
        for ax, p, nm in zip(axes[0], plays, names):
            ax.clear()
            frame(ax, p, max(0.0, float(ts[i])), lateral, window, rings)
            panel_title(ax, (nm + ": " if nm else "") + ("snap" if ts[i] <= 0 else f"+{ts[i]:.1f} s"), fs=9)
        return []

    fig.tight_layout()
    anim = animation.FuncAnimation(fig, update, frames=len(ts), interval=100, blit=False)
    html = (anim.to_html5_video(embed_limit=40).replace("<video ", '<video style="max-width:100%;height:auto" ')
            if shutil.which("ffmpeg") else anim.to_jshtml(default_mode="once"))
    plt.close(fig)
    display(HTML(html))

What man coverage asks of a defender

Every pass defense has to answer one question before the snap: when the receivers start running, how does each defender know what to do? Zone coverage answers with places: you have the deep third, you have the flat, guard it and take whoever comes. Man coverage answers with people: you have number 11, go where he goes. Everything in this chapter grows from that one choice, so it is worth slowing down on what it buys and what it costs.

What it buys is tightness and certainty. A zone defender guards grass, so a receiver can sit down in the soft spot between two zones and catch the ball with nobody within three yards of him. A man defender guards a body. If he is good enough, there is no soft spot, because the soft spot moves with the receiver and so does the defender. There is also no confusion about who takes the receiver who wanders between two areas, which is where zone coverages break down. And man coverage lets a defense choose its matchups: its best cornerback on their best receiver, its fastest linebacker on their running back.

What it costs comes from where the man defender is looking. A zone defender watches the quarterback and can break on the ball the moment it’s thrown. A man defender watches his receiver, usually his hips and belt, because that is where a route’s direction shows first. So he often has his back to the quarterback. He doesn’t see the ball leave the hand; he reacts to the receiver’s hands and eyes going up. He can’t help a teammate, because his own man would be left alone. If the quarterback escapes the pocket and runs, half the defense is running away from him, following receivers downfield. And if a receiver beats his defender, by a quick release, a sharp break, or a route that drags the defender into traffic, there is nobody nearby to fix it.

That last problem is the one man coverages are built around. The pure version, every defender on a receiver and nobody else, is all risk: one lost footrace is a touchdown. So almost every man coverage keeps one or two defenders free, with no receiver of their own, to help the others. Where the free defenders stand, how many there are, and what they do give us the family names: Cover 1, Cover 0, the robber and rat versions, brackets. The free defender is the whole game.

ImportantWhy it exists

Man coverage exists because a defender who covers a person never has to decide whose person he is. Against short, precise throws (the slant, the stick, the quick out) a zone defender is often a step too far away, because he started in the middle of an area. A man defender starts on the receiver and stays there, and that tightness is worth most where throws are short and timing is everything: third-and-short, the red zone, any down where the defense is also sending extra rushers. The price is eyes: man defenders watch receivers, not the ball, and need help somewhere behind them.

A little history: the bump-and-run, and the rules that answered it

Man coverage is as old as the forward pass, but the modern, aggressive form grew up in the 1960s American Football League. Willie Brown, a cornerback who had been a linebacker in college, began jamming receivers right at the line in 1963, and with Al Davis’s Raiders he became the face of bump-and-run: line up in the receiver’s face, hit him at the snap, then run with him. (Davis modelled the Raiders’ version on the full-court press of John Wooden’s UCLA basketball teams.) Brown played 12 seasons in Oakland and is in the Hall of Fame.1

The rules answered twice (the 1978 “Mel Blount rule” and the 2004 “Ty Law rule”, both told in Coverage Fundamentals; the Ty Law game is in this chapter’s film room). For man coverage the result is the shape of the modern craft. Inside five yards it is a fight of hands and feet; beyond five yards, until the ball is thrown, the defender may not initiate contact that impedes the receiver, so everything depends on position: being on the right side of the receiver, at the right distance, before he breaks.2 Physical man coverage followed by a rules response is a pattern you’ll see again in Contact Rules and Penalties.

Cover 1: man-free

The most common man coverage in football is Cover 1, which coaches also call man-free. The name is the coverage: man coverage on every eligible receiver, plus one free safety who has no man at all and plays the deep middle of the field. The “1” is the one deep defender.

The free safety’s own job sounds impossible at first: one man “helping” on five receivers. He doesn’t try to cover them all. He starts in the middle, 13 to 15 yards deep, reads the quarterback’s eyes and shoulders, and works toward the deepest route breaking inside or straight up the middle, the throws a corner with outside leverage can’t reach. Routes to the sideline are the corners’ problem. Two deep routes into the middle at once (two posts, or a post and a seam) are his nightmare, for the same reason a high-low is the robber’s below: one man, two targets. How common is all this? In the public charting, Cover 1 was the label on roughly a quarter to a third of charted NFL dropbacks in every season from 2018 to 2025, by far the most of any man coverage.3

Who takes whom

A man coverage is only as good as its matching rules, because the offense chooses the formation and the defense has to sort itself out in the 15 or so seconds before the snap. A typical set of rules, which varies by team:

  • Corners take the widest receiver on each side (#1). They are the best man-to-man athletes on the field and the outside receivers are usually the fastest.
  • The nickel takes the slot receiver (#2 to his side); in a base defense, with no nickel back, that job goes to a safety or a linebacker.
  • A safety or linebacker takes the tight end, matched for size.
  • A linebacker takes the running back, and if the back stays in to block, that linebacker is free to rush the quarterback (a “hug” or “green-dog” rule) or drop into the middle.
  • The free safety takes nobody and plays the deep middle.

These are starting rules, and the call can rearrange them. If the plan is for the strong safety to come down and rob (the next big section), he can’t also cover the tight end, so a linebacker takes the tight end instead, and the spare defender moves up from the linebacker level to the safety level. The count never changes: five receivers need five men, and whoever is left over is the free defender.

The rules have to cope with whatever the offense shows. When a receiver motions across the formation, his defender either travels with him, the classic tell you met in Motion and Shifts as Weapons, or the defenders exchange, each sliding over a spot and swapping receivers. When the offense shows three receivers to one side, someone from the other side has to be matched across. When the offense lines up in a formation the defense didn’t prepare for, the defense has a short list of simple “checks” to fall back on. Much of what looks like a cornerback getting “beat” on a broadcast is actually two defenders who disagreed about a rule.

The help is inside, so the leverage is outside

Look again at the cornerbacks in the figure: each is a step outside his receiver. That isn’t a style choice; it follows from where the help is. In Cover 1 the free safety lives in the deep middle, and often a second free defender sits underneath in the middle too. That help is inside help: defenders positioned toward the middle of the field from the man defender, who can get to routes that break inside (posts, digs, slants) faster than they can get to the sideline.

So the man defender takes away the side where help can’t reach him. A cornerback with inside help plays outside leverage: he protects the sideline routes (the fade, the out, the comeback) and, in coaches’ language, funnels the receiver inside, toward the help. If the receiver runs a post, the corner chases and the free safety arrives from the middle. If he runs a slant or a dig, the corner trails and the help underneath is waiting. The general rule, which works for every man coverage in this chapter: play away from your help. Where there is help, you can give ground; where there isn’t, you can’t.

That is the starting rule, not a law, and coaches bend it in three common ways:

  • Split. A receiver who lines up close to the sideline has very little room outside him, so the sideline itself is a kind of help, and his defender often plays inside leverage instead. A receiver with a tight split, near the formation, has room both ways and gets the outside shade.
  • Down and distance. On third-and-short many Cover 1 teams press head-up or with an inside shade to take away the slant, the fastest way to the sticks, and accept the fade, a slower, harder throw the free safety couldn’t reach anyway.
  • The slot. A slot receiver has room to break either way (a “two-way go”), and his defender often shades inside: the inside break is the quick throw, while a break outside means a longer throw toward the sideline that gives him time to recover.
ImportantWhy it exists: Cover 1

Cover 1 is the compromise that makes man coverage affordable. Pure man everywhere leaves every defender alone against a deep route. One deep safety turns every one-on-one into a one-on-one with help on the deep, inside throw, and the coverage still has a defender left over to rush or to rob. It keeps the box strong, too. Only one safety lives deep; the other plays close to the line of scrimmage, on the tight end or as the spare, so with the linebackers the defense still has seven or eight players near the line to stop the run. (Against a run, a man defender whose receiver blocks him becomes a run defender; who takes what in the run game is the subject of Run Fits and Box Math.) What it gives up: the middle of the field underneath (unless the extra defender sits there), the receiver who simply wins his matchup to the outside, where there is no help, and the throws that make man defenders run through traffic.

Press-man is the bump-and-run’s descendant. The defender lines up within a yard or two of the receiver, gets his hands on him at the snap (legal inside five yards), and tries to knock him off his path and onto the defender’s terms. If the jam lands, the route is late, and a late route in a timing offense is a dead route: the quarterback’s feet have finished their drop and the receiver isn’t where he’s supposed to be. If the jam misses, the defender has to turn and run with a receiver who now has a step on him. Press-man is the purest test of athletes in football, and the reason teams pay elite cornerbacks: hips that turn fast, long arms, and the recovery speed to survive a miss. You met the receiver’s side of this fight in The Quick Game.

Off-man gives the receiver room. The defender starts 5 to 8 yards away, often with his weight back, and backpedals or shuffles as the receiver comes toward him, keeping that cushion while he reads the route. Off-man trades the jam for vision: he sees the break coming and is rarely beaten deep. The cost is the short throw. A receiver who turns around at six yards against a defender who started at seven is open, and good quarterbacks take that throw all day. Off-man is common against very fast receivers, against receivers in the slot (where a jam is harder because the receiver can release either way), and on long-yardage downs where a six-yard catch doesn’t hurt.

Trail technique is what man defenders do when they have help over the top. The defender runs a step behind the receiver and slightly underneath him, on his hip, taking away anything thrown low or short, and trusts a safety to take the ball thrown high and deep. It is the technique of 2-Man, two deep safeties with man underneath (Cover 2 and Tampa 2), and of any Cover 1 defender whose man runs into the free safety’s area. Trailing looks like losing on a broadcast (the receiver is ahead), but the trailing defender is exactly where he wants to be: under the throw. To tell trailing from being beaten on a replay, look for the safety. If one is arriving over the top, the corner is trailing on purpose; if nobody is coming, he has simply been beaten.

Whatever the technique, everything ends at the catch point, and there the key idea is phase. A defender who is in phase is on the receiver’s hip, stride for stride, close enough to turn his head, find the ball and play it as if he were the receiver. A defender who is out of phase, a step or more behind, can’t afford to turn and look, because he’d lose more ground. So he watches the receiver instead and plays his hands: when the receiver’s eyes widen and his hands come up, the ball is arriving, and the defender rips through the hands as they close. That is what “he reacts to the receiver’s hands” meant earlier. It is also where pass interference lives. Once the ball is thrown, a defender who contacts a receiver more than a yard past the line and significantly hinders his chance to catch it commits pass interference, unless he is playing the ball himself. Not turning around is not a foul in itself, but a defender who isn’t looking for the ball is not “playing the ball”, so any contact before it arrives is likely to draw the flag.4 The details are in Contact Rules and Penalties.

Teams rarely pick one technique for a whole game. They pick per matchup and per call: press on a slow receiver, off on a fast one; press on third-and-2, off on third-and-14. The quarterback’s pre-snap read is partly a read of these techniques, which is why defenses mix them and sometimes show press and bail out at the snap.

CautionCommon misconception

“Press means man, off means zone.” As Coverage Fundamentals warned, alignment alone doesn’t tell you; plenty of man calls are played off, and plenty of zone corners press and bail. The man-coverage tell to add is the defender’s stance and angle before the snap: a man defender is usually squared up to his receiver and slightly turned toward the sideline; a zone corner is usually turned in so he can see the quarterback.

  • The robber is the free defender sitting in the middle of the field, usually 8 to 12 yards deep, driving on in-breaking throws: digs, deep crossers, the slant that gets past its man. He doesn’t just stare at the quarterback. In most teaching he keys a receiver, usually #2 or #3 to the passing strength, and “robs the first in-breaker”, while feeling the quarterback’s eyes to know when to go. Classically he is a safety who lines up deep (often disguised as half of a two-high shell, the subject of Disguise, Rotation, and Split-Field Coverage) and comes down at the snap. His name says what he is for: he steals the throws man coverage is worst at defending, the ones where receivers run away from their defenders across the field. When the ball goes outside instead, he becomes a pursuit player: the first tackler coming from the inside.
  • The rat is, in many playbooks, simply another name for the same job. When coaches distinguish them, the rat is lower and usually a linebacker, the man left over when the back he would have covered stays in to block, or the Mike in the figure above, sitting 5 to 8 yards deep, low in the hole. (That is why the hole is sometimes called the rat zone, and the rat a hole player.) His prey is the shallow cross and the quarterback scramble up the middle.
  • Lurk describes the same idea with disguise built in. The lurker lines up somewhere that says something else, often walked up to the line as if blitzing, sometimes deep as a safety, and drops at the snap into the shallow middle. Pressure defenses love it, because the quarterback who sees a blitz coming throws the quick hot route to the middle, and the lurker is standing in it. Some staffs use “lurk” only for a player who drops into the hole off a pressure look, sometimes even a defensive lineman.

Whatever it’s called, the free underneath defender changes the math of Cover 1. Against pure man, a receiver who wins inside is open. Against Cover 1 with a robber, a receiver who wins inside runs into a second defender who has been watching the quarterback the whole time and is moving before the ball is. Here it is in motion.

What makes the robber work is that the dig is a man-beater by design. The receiver runs straight at his defender, then breaks flat across the field, away from a corner who was protecting the outside. Against pure man that corner is chasing, and a dig thrown on time is a completion with room to run. The robber doesn’t need to cover Z. He only needs to be in the lane when the ball comes, which he can do because he is free to read the whole play (his key receiver and the quarterback) instead of chasing one man. That is the key idea of modern man coverage: man coverage on the receivers, zone eyes on the quarterback from the one defender who can afford them.

The robber has his own weakness, and offenses know it. He is one man in the middle guarding a lane, so put two receivers in his area at different depths, a shallow cross under a dig, and he can only take one. That is a high-low on the robber, exactly the read Dropback Concepts built the drive and shallow-cross concepts around. And a quarterback who looks the robber off, staring at the middle to pull him in before throwing outside, turns the robber’s best habit against him.

NoteMadden vs. real life

In Madden, “Cover 1” is a play on a menu, and the game decides who covers whom. On a real field the call is a family of rules, and the most important ones are about the defender with no man. The same Cover 1 call might have the leftover linebacker rushing if the back stays in to block, sitting in the hole if the back releases, or spying a mobile quarterback; the safety might show two-high and rob, or line up in the middle and stay there. The defensive coordinator also decides technique and leverage player by player and receiver by receiver, often on the sideline between series. Two Cover 1 calls can look nothing alike.

Cover 0: everybody’s man, nobody’s deep

Now remove the safety. Cover 0 is man coverage on every eligible receiver with no deep defender at all. The “0” is the number of deep safeties. Do the count again. The two defenders who were free in Cover 1 (the deep safety and the spare man) have no receiver and no deep job, so they become two more rushers. Usually that doesn’t mean the safeties themselves rush: the safeties come down to cover the tight end and the back, and the linebackers who used to cover them rush instead. So Cover 0 usually travels with a big blitz, six or seven rushers. It is the all-out blitz, the “house call,” the coverage behind the moment when the defense decides that the quarterback won’t have time to find the open man. (Not always, though: some zero calls rush only four or five and keep one spare defender sitting low in the middle, a “zero hole” or “zero lurk,” especially near the goal line.)

The logic is arithmetic. The offense has five linemen. If the running back and tight end go out on routes, the five linemen are the only blockers, and six rushers beat five blockers: one is free. If the offense keeps a back in to block, his man defender has nobody to cover. In most zero calls he joins the rush (the “hug” rule again), so it’s seven against six, still one free; in some he just sits, as the safety does in the Corn Dog figure in Motion and Shifts as Weapons. Cover 0 is built so that someone comes unblocked, and the quarterback has to get rid of the ball before he arrives. In the charted seasons, Cover 0 was played with five or more rushers about three times in four, and the median throw against it came out about three tenths of a second sooner than against any other coverage.5 Three tenths is roughly the difference between throwing on the last step of the drop and taking one more hitch step to wait for a route to come open.

Because no one is behind them, Cover 0 defenders play inside leverage, the mirror of Cover 1. With the linebackers rushing, nobody is left in the middle of the field, so a slant caught there has open grass in front of it and nobody to tackle the receiver but the man chasing him. So they take the inside throw away and dare the quarterback to throw outside, a longer, slower throw with the sideline as an extra defender. Some teams play it off instead of pressed, five to seven yards deep and inside (“zero off”), so the defenders can see the quick throw, break on it and tackle, without the risk of being beaten clean at the line. The other danger is obvious from the picture: a receiver who beats his man on a deep route has nothing in front of him but grass. That is the bargain.

ImportantWhy it exists: Cover 0

Cover 0 exists to win a race. Every rusher it can spare is sent, so the quarterback must throw now, often before his receivers have finished their routes, and every receiver has a man on him so there is no easy hot throw. It shines where the field is short and the throw has to be quick anyway: third-and-short, fourth down, inside the 10. What it gives up: any help at all. One missed tackle, one blown assignment or one receiver who beats his man deep is a touchdown, and a quarterback who escapes the free rusher finds a field full of defenders with their backs to him.

A handful of modern coordinators have leaned on the call harder than anyone. In nflverse charting the league leader in Cover 0 rate was Wink Martindale’s Baltimore defense in 2019 and his Giants in 2022 and 2023, and Brian Flores’s Miami Dolphins in 2020 and 2021.6 The most famous is Steve Spagnuolo. His 2007 Giants beat the 18–0 Patriots in Super Bowl XLII mostly by rushing four and blitzing only now and then, so his Cover 0 reputation comes from Kansas City, where he has been defensive coordinator since 2019: the Chiefs ranked in the top five in Cover 0 rate in every season from 2019 to 2025, and first in 2025.7 He is known less for how often he calls it than for when: at the moments offenses least expect it, which is part of what makes it work, because a quarterback who knows Cover 0 is coming can call a protection and a route to beat it.

Stacks and bunches: banjo and combo coverage

Man coverage has a structural weak point, and offenses found it long ago: receivers who line up close together. When two receivers stand a yard apart, or one directly behind the other in a stack, their defenders are close together too, and the receivers can release in directions that make the defenders cross. A defender chasing his man has to go around the other receiver, or around his own teammate, and loses a step. That is the rub from Dropback Concepts, and stacks and bunches are formations built to create it. (A rub has to look like an accident. A receiver more than a yard past the line who deliberately blocks a defender before the pass is thrown commits offensive pass interference, so offenses design the two paths to cross close by, not to collide.8)

The defense’s answer is to stop deciding before the snap. In a banjo, two defenders take two receivers together, and decide who has whom by how the routes declare: the outside defender takes the first receiver who breaks outside, the inside defender the first who breaks inside. Nobody has to cross anybody. (Don’t confuse it with the in-out bracket later in this chapter, which puts two defenders on one receiver, or with the exchange on motion from Motion and Shifts as Weapons, which swaps receivers before the snap.)

Notice where the banjo defenders stand: about five yards off, not pressed. They need a moment to see which way each receiver is going, and from the line of scrimmage they would have no time to read before having to commit. The in/out split is the most common; some teams split them low/high instead, so the first receiver to break short goes to one defender and the first to go deep to the other. Every banjo also needs a rule for the release that doesn’t declare neatly, both receivers going the same way or both running straight upfield; usually the defenders fall back on their pre-snap match, or the outside man takes #1.

The banjo solves the rub by trading a little of man coverage’s certainty for zone-like flexibility. It is a small zone rule inside a man call: for the first yard or two, each defender guards a side rather than a person, then locks on to whoever comes to that side. (It runs the opposite way from the pattern matching you met in Coverage Fundamentals, which is a zone call whose defenders become man defenders once the routes declare. A banjo starts from man and borrows a moment of zone.) That makes it the simplest example of a combo coverage, the general name for coverages in which two or more defenders share receivers and decide who takes whom by how the routes declare. Against a three-receiver bunch, a common version is a “point” call: one defender presses the receiver at the point of the bunch (the one nearest the line), and the other two play first-out and first-in on the other two. Bigger combos play man on one side of the formation and zone on the other.

The price is communication and a new weakness. Both defenders must call it, both must read the same release, and both must commit at the same instant. Offenses go after banjos with routes that make the release ambiguous. The clearest is to send both receivers the same way, one a beat after the other.

  • Under-over (top-bottom). The cornerback presses head-up or a step inside and plays underneath, in trail technique, taking away anything short or intermediate (the slant and the dig are his); a safety plays over the top, taking away anything deep, the fade included. The receiver can’t run past the safety and can’t break in front of the corner. It is the classic way to take a deep threat out of a game.
  • In-out. The cornerback plays outside leverage and owns the sideline; a second defender (a robber, a linebacker, a nickel) sits inside and owns everything that breaks toward the middle. It suits a receiver who wins on slants and digs more than on go routes. (Same “in” and “out” words as a banjo, different job: the banjo is two defenders on two receivers, the bracket two on one.)

In today’s NFL the receiver being bracketed is often a tight end: a linebacker or the nickel underneath him, a safety over the top.

Before the snap, an under-over bracket can look like a two-high shell, two safeties deep, which Coverage Fundamentals taught you to read as Cover 2, quarters or 2-Man. The tell is where the safety stands. A two-high safety has to cover half the field, from the middle to the sideline, so he lines up roughly halfway between them. A bracket safety has one man to cover, so he cheats out over that man: only a few yards inside the receiver, far from the middle, while the other safety sits over the rest of the formation.

Brackets are expensive, and the bill always arrives somewhere else. Eleven defenders minus four rushers leaves seven for five receivers. Spend two on one receiver and the other four have five defenders between them: one is still free, but now he is usually the only deep player, or there is no robber, or one man defender has no help at all. In the 3x1 picture, the bracket on X is paid for by an empty hole in the middle of the field. Bracket the star and the offense’s job becomes simple: throw to the others, especially the receiver now covered by a linebacker.

That is a perfectly good outcome for many defenses. The idea is associated above all with Bill Belichick, whose game plans often started from one question: what does this offense most need to do, and how do we take it away? If the answer was a receiver, the answer was often a bracket, and the offense’s second and third options were invited to beat the Patriots instead. (How that habit became a philosophy is the story of Case Study: Belichick and Game-Plan Football.) A defense can also bracket only on some downs, or only in certain areas of the field, so the quarterback can’t assume where the second defender will be.

Shadow corners

There is a cheaper way to deal with a great receiver, if you have the player for it: give him your own great player, everywhere. A shadow corner is a cornerback assigned to one receiver for the whole game. Wherever that receiver lines up (left, right, in the slot, in motion) the corner goes with him.

The reason most teams don’t do this is that it’s harder than it looks, for the corner and for the defense. A cornerback who has played the left side his whole career has to learn the right side, where his footwork is reversed, and the slot, where a receiver can release in either direction and the help is in different places. The rest of the defense has to adjust every snap around wherever the shadow went. And the corner has to be good enough that the defense can treat one receiver as solved, which is the whole point: a true shadow corner turns a bracket’s two defenders into one, and gives the defense its extra defender back. That is also why even man-heavy defenses with a great corner often don’t travel him every snap: a corner who is superb on the outside may be ordinary in the slot, and keeping him on one side and bracketing when needed is simpler for everyone else.

The modern archetype is Darrelle Revis. With Rex Ryan’s 2009 Jets he followed the opponent’s best receiver week after week, and the list of receivers he held quiet that season reads like an All-Pro team: Randy Moss twice, Andre Johnson, Terrell Owens twice, Reggie Wayne, Chad Ochocinco (no catches at all in their regular-season meeting). The press called it “Revis Island,” the place where receivers went to disappear, and the Jets allowed the fewest points in the NFL that year. Revis went into the Hall of Fame in his first year of eligibility, in 2023.9 More recently, Denver’s Patrick Surtain II, the 2024 Defensive Player of the Year, played the role for Vance Joseph’s defense: in the 2024 opener he lined up across from Seattle’s DK Metcalf on 24 of Metcalf’s 25 routes, by Next Gen Stats’ count.10 By this book’s count from nflverse charting, the Broncos played man coverage on a larger share of dropbacks than any other defense in both 2024 and 2025 (details in the section on when defenses choose man), a reminder that teams play the coverage their best players are built for.

The contrast is instructive. Richard Sherman, one of the best cornerbacks of the 2010s, mostly did not shadow. Seattle’s defense was built on single-high coverage, mostly Cover 3 (a zone) with plenty of Cover 1, and in both Sherman usually lined up on the defense’s left side and stayed there, letting the offense decide who would face him. Seattle sometimes moved him to follow a particular receiver, as against Atlanta’s Julio Jones in October 2016, but it was the exception.11 Revis was a travelling corner in a defense built around his matchup; Sherman was a fixed-side corner in a defense built around its structure. Both were great; the system decided which kind of great it needed.

Three situations stand out, and each has a reason.

Third down, short and medium. On third-and-3 to third-and-6, defenses played man on about 60% of charted dropbacks in both charting eras, far more than on first down (about 25 to 36%).12 The defense knows where the ball has to go: just past the line to gain. Man coverage contests the throw at the sticks instead of rallying to tackle after a catch short of them. It also pairs naturally with a blitz, and third down invites one: the offense almost has to throw, so the defense can send extra rushers without much fear of a run, and a quick incompletion ends the drive. Notice what happens on third-and-11 or more: man coverage drops to its lowest rate of any situation. With that much to go, the defense is happy to let the receiver catch the ball in front of it and tackle him short, which is a zone’s job. Third-and-long is the zone defense’s down; third-and-medium is man’s. And first down leans zone for the opposite reason: the offense might run, or fake a run and throw, and man defenders with their backs to the backfield are the worst-placed players on the field against both. Zone defenders face the ball and can come up to tackle a runner.

The red zone and the goal line. As the field shrinks, man coverage takes over. Inside the defense’s 5-yard line, charters labelled 78 to 88% of dropbacks as man coverage, and Cover 0 on 41 to 62% of them, depending on the source. Near the goal line the biggest weakness of man coverage nearly disappears. A receiver who beats his man deep runs out of field: the back line of the end zone, the end line, acts like a safety that never misses. Zones, meanwhile, lose their main tool: there is no depth to drop into, so zone defenders end up standing on the end line while quick, sharp-breaking routes are completed in front of them. Man coverage contests the catch itself. So defenses spend their deep defenders on the rush or on more man coverage. Read the goal-line Cover 0 numbers with care, though: near the goal line “Cover 0” often means “nobody deep, because there is no deep,” not necessarily an all-out blitz. Of charted Cover 0 dropbacks inside the 5, only about 60 to 70% had five or more rushers.13 (The red zone gets its own chapter: Red Zone, Goal Line, and Short Yardage.)

Talent. Man coverage is only as good as the worst matchup in it, so teams with great cornerbacks play more of it. In nflverse charting, the Patriots played man on a larger share of dropbacks than any other defense in 2018 and 2019 (about 61% and 68%), the seasons around Stephon Gilmore’s Defensive Player of the Year award; Miami, coached from 2019 to 2021 by Brian Flores, a long-time Belichick assistant, led the league in 2020 and 2021; Denver, with Surtain, led it in 2024 and 2025.14 The reverse is true too. A defense with a weak link in the secondary, or facing a quarterback who punishes defenders with their backs turned by running, leans toward zone, where everyone faces the ball.

CautionCommon misconception

“NFL defenses are playing more and more man coverage.” You may see charts that say so. Be careful. The public coverage labels come from two different charting sources (NGS through 2022, FTN from 2023), and the man share jumps from about 29% of charted dropbacks in 2022 to about 44% in 2023 and 50% in 2024, then falls back to 31% in 2025. Defenses didn’t change that much that fast; the labelling did. Comparisons within a source (one season, or one era, as in the chart above) are safe; trend lines across the 2022/2023 break are not. (The season-by-season figures are in the note to the third-down paragraph above.)

NoteFilm room: Patriots 2003 and 2019 — Ty Law, Stephon Gilmore and the man-coverage habit

Two Patriots seasons, sixteen years apart, bookend Belichick’s man-coverage defenses. In the 2003 AFC Championship Game against Indianapolis, Ty Law intercepted Peyton Manning three times in a 24–14 win, playing press-man against the Colts’ receivers and, in his words afterward, the simplest game plan imaginable: “Just beat them up.” The rules response that followed, the 2004 crackdown on illegal contact, became known as the “Ty Law Rule.”15

In 2019 the Patriots led the NFL in points allowed (14.1 per game) and total defense, and Stephon Gilmore became the first Patriot to be named Defensive Player of the Year, tying for the league lead with six interceptions and returning two of them for touchdowns.16 That was the season New England led the league in man coverage (the talent paragraph above has the number), and the habit around Gilmore was the Belichick one: man coverage on most of the field, and the spare defenders spent on whatever the opponent most wanted to do, often its best receiver or tight end.

What to notice: on a 2019 Patriots third down, find the free defenders first: usually a safety deep in the middle and a second defender in the hole or bracketing the offense’s best receiver. The pattern repeats from 2003 to 2019 with different players: man coverage on the people the defense trusts, and the free defenders spent on the player or the area it fears.

NoteFilm room: Chiefs, Super Bowl LVIII (February 11, 2024) — Spagnuolo brings the house

TipWatch for it on Sunday

Who’s facing the quarterback? It is the fastest man-or-zone test there is.

  • Before the snap: the corners. Squared up to the receiver, close, weight forward, eyes on the receiver’s belt: lean man. Turned inside, deeper, eyes on the quarterback: lean zone.
  • Before the snap: motion. Does somebody run across the formation with the man in motion? That is man. Do the defenders shift over and swap? Could be either: watch the next clue.
  • Before the snap: the safeties. One deep safety in the middle and corners pressed: think Cover 1. Nobody deep and six or seven players on or near the line on third down or near the goal line: think Cover 0 and expect a quick throw.
  • At the snap: the eyes. One second after the snap, look at the defenders underneath. Backs turned to the quarterback, running with receivers: man. Facing the quarterback, settling into areas: zone. The one defender in the middle facing the quarterback when everyone else is running with someone is the rat or robber. Find him and you’ll know where not to throw.
  • Close formations. When receivers stack or bunch, watch the two or three defenders over them at the release. Do they cross each other (locked man) or trade (banjo)?
  • The star. Where is the offense’s best receiver, and who is on him? The same corner every snap, on both sides of the field, is a shadow. A safety 10 to 12 yards deep and only a few yards inside him, far from the middle, is a bracket.

What are the two defenders over Y doing, and where should the ball go?

TipAnswer

Cover 1 with a bracket on the tight end. Count: four rushers, and five receivers with a man each (the corners on X and Z, the nickel on H, the Will on the back, the Mike on Y). That leaves two spare defenders. One, the free safety, is deep in the middle: Cover 1. The other, the strong safety, has nobody to cover and yet stands only a step and a half inside Y, 10 yards deep, which no robber would do (a robber sits in the middle). He is the “over” half of an under-over bracket on Y, and the Mike, tight and inside, is the “under” half.

Throw somewhere else. The bracket’s bill is paid in the middle: with the strong safety spent on Y, nobody sits in the hole. A crosser from H or X has room once it clears the Mike’s area, and the running back on the Will (an angle route) is a classic answer. X and Z are one-on-one with only the free safety behind them. Throwing to Y into two defenders on third-and-7 is what the defense is daring the quarterback to do. A defense can disguise this, of course, by standing the safety there and playing something else; and the same lean against a 3x1 formation can also belong to a split-field coverage check (see Disguise, Rotation, and Split-Field Coverage).

Takeaways

  • Man coverage covers people, not places. It is tight, certain about who covers whom, and lets the defense pick matchups. Its price is that man defenders watch receivers, not the quarterback, and need help somewhere.
  • The free defenders define the family. Cover 1 keeps one safety deep and usually one more defender to rush, rob or spy; Cover 0 keeps nobody deep and usually sends six or seven. The rat, the robber and the lurker are the same idea: one defender with zone eyes, in the middle, where man coverage is weakest.
  • Leverage follows the help. With inside help (Cover 1), corners usually play outside and funnel receivers inside (split, down and distance can change it); with no help (Cover 0), they play inside and protect the quick throw. Technique (press, off, trail) is chosen per matchup and per down.
  • Close formations and great receivers get special rules. Banjo and combo calls stop stacks and bunches from creating rubs; brackets spend two defenders on one star; a shadow corner spends one.
  • Man coverage is a situational weapon. In the charted NFL seasons it was most common on third and short-to-medium and near the goal line (where Cover 0 is often the most common call), and least common on third-and-long. The public man-rate trend breaks at 2022/2023 because the charting source changed.
  • Every man-beater makes staying with a receiver harder: crossers and mesh, rubs, slants against leverage, double moves, motion at the snap, a back on a linebacker, a scrambling quarterback. The defense’s counters are the free defender, the banjo and the bracket.
