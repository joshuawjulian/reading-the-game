Cover 2 and Tampa 2

Two deep, five under: the zone that keeps everything in front of it, the cornerbacks who refuse to run, the middle linebacker who does, and the holes every offense knows by heart.

def c2_shell(off, corner_d=5.5, corner_out=1.3, safety_d=13.0, safety_w=9.5, press=False):
    """Nickel (4-2-5) two-high shell against `off`. Corners squat 5-7 yards off, a step outside
    the widest receiver on each side (outside leverage); safeties deep and between the hash and
    the numbers; the Will and the nickel ($) split the difference between the tackles and the
    slots; the Mike sits over the ball. press=True walks the corners up to a yard off."""
    wide = [p for p in off if p.pos in ("WR", "TE") and abs(p.w) > 6]
    wl = min(p.w for p in wide if p.w < 0)
    wr = max(p.w for p in wide if p.w > 0)
    cd = 1.2 if press else corner_d
    moves = {"LCB": (cd, wl - corner_out), "RCB": (cd, wr + corner_out),
             "NB": (5.0, 7.0), "WILL": (5.0, -6.5), "MIKE": (4.5, 0.5),
             "FS": (safety_d, -safety_w), "SS": (safety_d, safety_w)}
    out = []
    for p in defense("nickel", "two_high", off):
        if p.key in moves:
            p = p.moved(d=moves[p.key][0], w=moves[p.key][1])
        if p.pos == "CB":
            p = p.moved(label="CB")
        out.append(p)
    return out


def protect(p, rushers=("SDE", "SDT", "WDT", "WDE")):
    """Five linemen set, four rushers engage them: enough to show it is a pass, no more."""
    for k in ("LT", "LG", "C", "RG", "RT"):
        p.block(k, pts=[(-1.2, 0.0)])
    for k in rushers:
        p.rush(k, pts=[(-1.6, 0.0)], speed=2.0)


def no_numbers(ax):
    """Remove the painted yard-line numbers (they sit where wide receivers and corners stand)."""
    for t in list(ax.texts):
        if t.get_alpha() == 0.55:
            t.remove()


def ruler(fld, play, widths=False, fs=6.5, side="left"):

def trim(ax):
    """Drop text labels whose anchor lies outside the drawn window (players cropped out of view)."""
    x0, x1 = sorted(ax.get_xlim())
    y0, y1 = sorted(ax.get_ylim())
    for t in list(ax.texts):
        x, y = t.get_position()
        if t.get_transform() == ax.transData and not (x0 <= x <= x1 and y0 <= y <= y1):
            t.remove()

def snapshot(play, t, ax=None, lateral="full", window=(-8.5, 24.0), width_in=6.5, notes=(), show_ruler=False,
             ruler_side="left"):
    """Everyone where he is at time t, with faint trails of where he has been and the ball's flight."""
    pos = {k: play.position(k, t) for k in play.players}
    moved = [pl.moved(d=float(pos[k][0]), w=float(pos[k][1])) for k, pl in play.players.items()]
    q = Play([m for m in moved if m.side == "off"], [m for m in moved if m.side == "def"],
             los=play.los, ball_y=play.ball_y, to_go=play.to_go)
    q.highlights = dict(play.highlights)
    fld = q.draw(ax=ax, lateral=lateral, window=window, width_in=width_in, title=False)
    ax, ppy = fld.ax, fld.pts_per_yard()
    no_numbers(ax)
    if show_ruler:
        ruler(fld, play, fs=5.5, side=ruler_side)
    P = lambda d, w: fld.xy(*play.to_abs(d, w))  # noqa: E731
    start = play._compile()[1]
    for k, pl in play.players.items():
        if pl.side == "off" and pl.pos not in ("WR", "TE", "RB"):
            continue
        pts = np.array([play.position(k, tt) for tt in np.linspace(start, t, 30)])
        if np.ptp(pts[:, 0]) + np.ptp(pts[:, 1]) < 0.3:
            continue
        xs, ys = P(pts[:, 0], pts[:, 1])
        if pl.side == "off":
            ax.plot(xs, ys, color=style.OFFENSE, lw=1.0, alpha=0.45, zorder=3)
        else:
            ax.plot(xs, ys, color=style.DEFENSE, lw=1.3, ls=(0, (3, 2)), zorder=3)
    bx, by = P(*play.ball_position(t))
    for kind, _, t_rel in play.ball_events:
        if kind == "pass" and t > t_rel:
            sx, sy = P(*play.ball_position(t_rel))
            ax.add_patch(FancyArrowPatch((float(sx), float(sy)), (float(bx), float(by)), arrowstyle="-|>",
                                         mutation_scale=8, color=style.BALL, lw=1.4, ls=(0, (2, 1.5)), zorder=9))
    passes = [t_rel for kind, _, t_rel in play.ball_events if kind == "pass"]
    if not passes or t > passes[0]:          # before the throw the ball is in the QB's hands: don't hide him
        draw_marker(ax, float(bx), float(by), "ball", "", ppy)
    for text, at in notes:
        say(fld, play, text, at, fs=6.4)
    trim(ax)
    return fld


def strip(play, times, labels, lateral=(-HALF, HALF), window=(-8.5, 24.0), notes_at=None, width_in=6.5, ncols=2,
          ruler_side="left", extras=None):
    """The print version of an animation: one panel per chosen moment. `extras` maps a panel index to a
    function f(fld) that draws anything else on that panel (a shaded spot, an arrow)."""
    n = len(times)
    nrows = int(np.ceil(n / ncols))
    pw = width_in / ncols
    h = nrows * (pw * (window[1] - window[0]) / abs(lateral[1] - lateral[0]) + 0.32)
    fig, axes = plt.subplots(nrows, ncols, figsize=(width_in, h), squeeze=False)
    for ax in axes.flat[n:]:
        ax.axis("off")
    for i, (ax, t, lab) in enumerate(zip(axes.flat, times, labels)):
        fld = snapshot(play, t, ax=ax, lateral=lateral, window=window, notes=(notes_at or {}).get(i, ()),
                       show_ruler=(i == 0), ruler_side=ruler_side)
        if extras and i in extras:
            extras[i](fld)
        ax.set_title(lab, fontsize=8.5, color=style.INK, loc="left")
    fig.tight_layout(h_pad=0.6, w_pad=0.4)

Third-and-12 near midfield, a minute left in the first half. Both cornerbacks stand five yards off the wide receivers, a step outside them, flat-footed. Both safeties are deep, one on each side of the field. The quarterback throws a quick out to the sideline; the corner closes, tackles, gain of eight, and the offense punts. The analyst sounds almost pleased: “That’s Cover 2 doing its job. Keep everything in front, make the tackle.” Two series later the same corner stands in the same spot, the receiver runs straight past him, and the ball drops into his hands 22 yards downfield, between the corner and the safety, before the safety can get there. “And that’s the hole in Cover 2,” says the analyst. So which is it? Why would a defense build a coverage around a hole it knows is there, and why did one version of that coverage, Tampa 2, become the identity of a generation of NFL defenses and two Super Bowl champions? This chapter takes Cover 2 apart: who stands where, why, what it buys, where it bleeds, and what became of it.

This chapter builds on the first three coverage chapters. One line each:

  • Shells, man and zone (A First Look at Defense): the shell is the safeties’ picture before the snap, one-high or two-high. Man coverage gives each defender a receiver, zone coverage an area. A nickel defense has five defensive backs.
  • Zones (Coverage Fundamentals): deep, the field splits into thirds, halves or quarters. Underneath, from the sideline in: the flat (0–6 yards deep), the curl (10–12 yards), the hook (over the tackles) and the hole (the middle, 8–15 yards). Each defender drops to a landmark, either as a spot-drop player who reads the quarterback or a pattern-matching one who picks up receivers by rule. Press is lining up on the receiver; off is several yards away.
  • Man techniques (Man Coverage): in trail technique a defender runs a step behind and underneath his man, trusting help over the top; a bracket puts two defenders on one receiver.
  • Cover 3 (The Cover 3 Family): three deep, four under; its corners bail (turn and run) to deep thirds. In a cloud rotation one corner stays short and takes the flat, the Cover 2 corner’s job borrowed for one side. This chapter calls the area between the hook and the curl the hook-curl zone, as that chapter did.
  • Receivers and routes (Spread Formations, Dropback Concepts): defenses number receivers from the sideline in, #1 widest, then #2 and #3. A seam runs straight up between the hash and the numbers, a hitch stops and turns back, a corner route breaks deep toward the sideline. Smash pairs an outside hitch with an inside corner route; four verticals sends four receivers deep. A high-low read puts one defender between a deep route and a short one, and a defender’s leverage is which side of the receiver he stands on.
  • The field: the NFL hash marks are about 3 yards either side of the middle, the painted numbers about 13 to 15 yards, each sideline about 26.7.

Reading the diagrams: offense is blue circles (X, Y, Z, H receivers; Q quarterback; R running back); defense is outlined squares (E and T ends and tackles; M the Mike, W the Will, $ the nickel back, CB corners, FS and SS safeties). Faint lines are 5 yards apart from the line of scrimmage (dark blue); small numbers on the edge give the depth; the orange line is the line to gain. Dashed arrows are defenders’ movements, solid arrows routes, dotted lines man assignments, shaded ovals zones.

Two deep, five under

Every coverage is an answer to the same arithmetic. The defense has eleven players. If four of them rush the passer, seven are left to cover. Those seven can be split between deep defenders, who stop the ball being thrown over everyone’s head, and underneath defenders, who stop the short and medium throws. Cover 3 splits them three deep and four under. Cover 2 splits them two deep and five under: two safeties each take a deep half of the field, and five defenders spread across the short zones in front of them.

Here is each job, and the reason behind it.

  • The two safeties: a deep half each. Each safety owns everything deep on his side of the field, from the middle of the field to the sideline: about 27 yards of width. His rule, in its simplest form, is nothing gets behind me: as deep as the deepest receiver in his half, as wide as the widest. He reads the inside receiver (#2) first, because #2’s seam is the threat closest to the middle, then the outside one (#1), and breaks on the ball when it’s thrown. He lines up 12 to 14 yards deep, between the hash and the numbers, roughly 8 to 10 yards from the middle of the field, because from there he can reach both ends of his half; by the time the quarterback finishes his drop he has backpedalled to about 18 to 20 yards, near the top of the numbers. A quarters safety (Quarters and the Two-High Match Family) usually stands shallower and closer to the hash, because he has only a quarter of the field; being deeper and wider is the price of owning a half.
  • The two corners: the flats. A Cover 2 cornerback is not a deep defender at all. He lines up close, outside the wide receiver, and at the snap he jams or redirects him and then stays short, at the top of the flat (6 to 9 yards deep, a little deeper than the 0–6 of the textbook flat, so he can also slow down anything going deep). This is the hard corner, and the next section is about him.
  • The Will and the nickel: the hook-curl zones. Between the flat and the middle, each takes the curl and hook area on his side, 10 to 12 yards deep, under the inside receiver’s routes and in the lane of the outside receiver’s curls and digs. One more job matters a lot: if the slot receiver (#2) runs straight up the seam, the hook-curl defender gets in his way (coaches say he walls him) and rides him to 10 or 12 yards before letting the safety take over. Every step that costs the receiver is a step the safety gains. In a base 4-3 defense these two are the outside linebackers, the Sam and the Will; in nickel, the usual passing-down defense, the nickel back plays the Sam’s part.
  • The Mike: the hole. The middle linebacker drops straight back to the middle of the field, 10 to 12 yards deep, to take away throws over the ball: digs, crossing routes, a tight end sitting down in the middle. (Hold on to this job. Tampa 2 changes it.) A word trap: the hole is the name of this zone, a defended one, from Coverage Fundamentals. When announcers say “a hole in the coverage” they mean the opposite, an empty spot. This chapter calls those empty spots thin spots; the one exception is the name hole shot, which the whole league uses.
  • The four rushers. In a pure Cover 2 nobody else is spare. The coverage depends on four linemen getting to the quarterback without help, because seven defenders are all spoken for.
ImportantWhy it exists

What Cover 2 buys. Five underneath defenders is the most any common zone puts there, so every short throw has a defender within a few yards of it, and the quarterback has to fit the ball into small windows or settle for short gains. The corners in the flats take away the quick throws to the outside that beat off-coverage corners (quick outs, hitches, bubble screens), and they squeeze the outside receivers’ releases, which disrupts the timing of everything else. Two deep safeties mean no single deep defender is ever alone against two vertical routes in the middle of the field, as Cover 3’s lone safety can be. And the corners never have to run step for step with the fastest receivers on the field, which lets a defense survive with corners who are physical and smart rather than fast.

What it gives up. Only two defenders guard the whole deep field, so each safety has a huge area, and wherever two receivers go deep in one half, one safety is in a two-on-one. The corners stay short, so the sideline behind them is empty until a safety gets there. With both safeties deep, only six or seven defenders are near the line of scrimmage against the run (the box), one fewer than when a safety comes down. Cover 2 is a bet that the offense can’t complete enough of the hard throws to make up for all the easy ones it takes away.

Cover 2 also has a run-defense job built into it, and it explains the corners’ alignment as much as the passing game does. Against a run to his side, the Cover 2 corner is the force player: he comes up hard from the outside, meets the ball carrier at the edge and turns him back inside, toward the linebackers. The safety on that side becomes the alley player, filling the gap between the corner and the box. A corner standing five yards off, outside the receiver, with his eyes inside, is in exactly the right spot to force, which is one more reason he stands there.

Count the box against 11 personnel and you see the cost. The offense has six blockers: five linemen and the tight end. A nickel Cover 2 has six defenders in the box: four linemen, the Mike and the Will (the nickel is out over the slot). Six against six leaves no unblocked defender at the line, so the spare men are the two safeties, and they arrive as alley players from 12 or more yards away. A run that gets past the linebackers against Cover 2 tends to go for 8 or 10 yards, not 4. (Run Fits and Box Math builds the run fits and the counting.)

NoteMadden vs. real life

On a video-game play-call screen, Cover 2 looks like one play: pick it and the same seven zones appear every time. On an NFL defense, “Cover 2” is a family with dozens of members. Some teams press the corners and some play them off; some spot-drop and some pattern-match; some play it with three linebackers and some with a nickel or dime; one coach’s safeties line up on the hashes and another’s on the numbers. Many staffs never call it “Cover 2” at all: the numbers on a call sheet are a team’s own dialect. What stays the same is the skeleton, two deep halves and five underneath, with the corners in the flats. That is what to look for.

The hard corner

The single most distinctive player in Cover 2 is the cornerback who doesn’t go deep. A hard corner (also called a squat corner) lines up 5 to 7 yards off the outside receiver, or right on him in press, a step outside him with his shoulders turned slightly in so he can see through the receiver to the quarterback. At the snap he does three things in order:

  1. Collision and redirect. He meets the receiver at the line or as he comes off it, gets his hands on him, and makes him release inside. A receiver who wants to run straight up the sideline gets pushed toward the middle of the field, where the safety is. Every step the receiver loses getting past the corner is a step the safety gains. All of this has to happen within five yards of the line of scrimmage: after that, contact before the ball is thrown is a foul, as long as the quarterback is still in the pocket with the ball.1
  2. Read #2, and sink. Before the snap he looks at #1. At the snap his eyes go to the inside receiver (#2), because #2 is the one who can threaten his flat. If #2 goes straight up the field, or nobody comes toward the flat at all, he doesn’t stand still: he gives ground, backing up and widening under #1, often to 10 or 12 yards, to shrink the gap between himself and the safety. Coaches teach it as a rule: sink until something threatens the flat.
  3. Squat. When #2 (or a running back) breaks toward the flat, or the quarterback’s eyes go there, he stops, plants and comes forward to make the tackle. He keeps outside leverage throughout, so anything that goes inside of him goes to the linebackers and the safety, and nothing escapes to the sideline.

“Squat” does double duty, and it’s worth saying once: a squat corner is the alignment (short and outside), and squatting on a route is step 3. Same idea both times: this corner lives low.

Why build a corner this way? Three reasons, and each is about time.

The flats are where quick throws go. A corner 8 yards off and backing up concedes the 5-yard out and the hitch: by the time he plants and drives on the throw, the receiver has caught it and turned upfield. A squat corner is already there. Against offenses that live on quick timing throws to the outside (The Quick Game), that matters on every down.

The jam buys the safety time. The weakness of Cover 2 is the ground the safeties have to cover, so anything that delays the outside receiver helps. A receiver who has been knocked off his line and forced inside arrives at 15 yards a half-second late and closer to the middle, which is exactly where the safety is coming from.

Outside leverage turns the sideline into a defender. Because the corner refuses to let anything escape outside, every receiver on his side is funnelled toward the middle of the field, into the zones of the linebackers and the safety. The offense has to throw into the defense.

The price is plain from the picture: once the receiver is past him, the corner is no longer a deep defender. Everything behind him and outside the safety is grass, and it stays grass until the safety covers the 10 or 15 yards to get there.

Where else you’ll see him. The squat corner is not only a Cover 2 player. Many staffs call any side where the corner takes the flat a cloud side (C for corner) and a side where a safety comes down to the flat a sky side (S for safety): the same words as the Cover 3 rotations, because it’s the same job. The cloud corner in Cover 3 and the Cover 2 side of Cover 6 (Quarters and the Two-High Match Family) are both squat corners. And in today’s NFL the squat often comes after the snap: a corner lines up 7 or 8 yards off, looking like a deep defender, and drives down to the flat once the ball is snapped. Some defenses even flip the jobs, in what coaches call Cover 2 invert: the corners bail to the deep halves and the safeties come down to the flats.

NoteGo deeper: when the rules changed the corner’s job

Until the mid-1970s, defenders could keep chucking a receiver (hitting or shoving him with their hands as he ran his route) all the way down the field until the ball was in the air. The league chipped at that in 1974 and 1977, limiting how many times a defender could jam the same receiver. Pittsburgh’s Mel Blount, a 6-foot-3 cornerback on the Steelers defenses of the 1970s, was so good at manhandling receivers that the bigger 1978 change is still called the “Mel Blount rule”: from that season, defenders could make contact with a receiver only within 5 yards of the line of scrimmage.2 The Cover 2 corner’s jam survived because it happens inside those 5 yards. What disappeared was a defense’s ability to keep rerouting receivers with linebackers and safeties downfield. The full story of the chuck rule and its effect on passing is in 1978 and the Passing Revolution, and today’s contact rules are in Contact Rules and Penalties as Strategy.

Where Cover 2 is thin

Every coverage has thin spots, because seven defenders can’t cover a field 53 yards wide and as deep as a quarterback can throw. Cover 2’s are unusually well known, because the coverage is so old and so common that every offense carries a list of its “Cover 2 beaters.” There are four classic attacks, aimed at three places: two of them, the hole shot and the smash corner, land in the same grass along the sideline, but they get there from different directions and put a different defender in a bind.

① The hole shot. The most famous one. The corner is sitting in the flat, 6 to 10 yards deep. The safety is on his half, starting 13 yards deep and 8 to 10 yards from the middle. Between them, along the sideline from about 15 to 25 yards deep, is a strip of grass that nobody is standing on at the snap. A hole shot is a ball thrown into that strip: usually the outside receiver’s go route or a fade (a go route that bends toward the sideline, away from the safety, and is thrown over the receiver’s outside shoulder; it gets its full treatment in the red-zone chapter), released past the squatting corner and thrown on time, before the safety can get across. The arithmetic is a race. From his alignment, the safety is roughly 14 yards from the spot where a fade is caught 20 or 22 yards downfield, 5 yards from the sideline. Opening and gaining depth while he reads, then running at 6 yards a second, he needs close to 3 seconds to get there. The throw leaves the quarterback’s hand a little after 2 seconds and hangs in the air for most of a second, so it lands at about the same moment. And a tie goes to the offense: to break up the pass the safety has to arrive a step before the ball, not with it. So anything that holds him for a beat (a second receiver running up the seam inside him, or the quarterback’s eyes looking that way) turns a tie into a completion. That is why offenses pair the outside go route with a seam inside it: two vertical routes in one half put one safety in a two-on-one. Four verticals does it on both sides at once.

② The smash corner. Smash reaches the same grass from the inside, and it tests a different defender. The outside receiver runs a hitch at 5 or 6 yards, which holds the hard corner in the flat: his rules tell him to sit on it. The inside receiver runs about 10 yards up the field and breaks toward the back corner of the field, over the corner’s head and under the safety. The corner can’t play both: if he sinks under the corner route, the hitch is open; if he squats on the hitch, the corner route is. It is a high-low read of one defender, and the next section plays it out frame by frame.

③ The deep middle. With two safeties each guarding a half, the line where the halves meet is the farthest point from both of them. Nobody is standing in the middle of the field 18 to 25 yards deep, and the Mike, who has the hole at 10 to 12 yards, can’t run with a receiver who goes past him. A seam route that splits the safeties, a post, or a tight end running straight up the middle puts both safeties in a bind: whichever one comes inside to help has left his own sideline. This is the hole Tampa 2 was built to close.

④ The sideline hole. Underneath the safety and above the corner there is also an intermediate gap, about 12 to 18 yards deep, outside the hook-curl defender. A deep out or a comeback from the outside receiver lands there; so does the middle route of a flood concept, which sends one receiver deep to hold the safety and one to the flat to hold the corner, then throws to the one in between. Coaches name these spots differently; some call the whole strip from 12 to 25 yards “the Cover 2 hole” (an empty spot, in the announcers’ sense) and some reserve the term for the deep end. The geometry is the same either way: the corner is low, the safety is high and wide, and the sideline between them is the offense’s.

CautionCommon misconception

“Cover 2 is soft.” It’s an easy thing to believe when a defense gives up 8-yard completions on third-and-12 and the announcer says the defense is “playing soft.” But Cover 2’s corners are the most aggressive in football against short throws, and five underneath defenders make the short middle crowded. What Cover 2 gives up is not the short ball but specific intermediate and deep windows, and a defense that plays it well makes the offense find those windows with a well-timed, accurate throw, over and over. A third-and-12 completion for 8 is the coverage working.

Look at the corner across the panels and you see a player doing everything right and losing anyway. His rules say: jam the outside receiver, sink, and squat on anything short in the flat. The hitch is short and in the flat, so he squats on it. Meanwhile, the corner route breaks behind him into the strip of grass he just left. The safety’s rules say: stay deeper than anything in your half. He does: at 1.9 seconds he is 19 yards deep, deeper than Y, but 8 yards inside the spot the corner route is heading for, and he can’t start running at it until he’s sure that’s where the ball is going. By the time he is sure, the ball is in the air. The throw beats him by a step.

That is the general shape of beating any zone: find one defender with two jobs and make him choose. In smash the cornerback is that defender. And the quarterback’s read is mercifully simple: he watches only the corner. Corner sinks, throw the hitch. Corner squats, throw the corner route.

How does the defense fight back? Not by playing Cover 2 harder, but by changing what the corner and safety do when they see this route combination.

  • Wall the slot. Watch the nickel ($) in the figure: Y runs straight past him untouched. His rules say he should have got in Y’s way as Y released up the field and ridden him for a few yards. A collision at 6 yards delays the corner route by a beat, and a beat is all the safety needs. What the nickel can’t do is follow Y toward the sideline: that would empty the hook-curl zone behind him for the next route.
  • Teach the corner to sink further. This is the corner’s #2 key from the last section. When #2 goes vertical and #1 runs something short, many Cover 2 coaches tell the corner to keep sinking under the vertical instead of jumping the hitch: give up the 6-yard hitch, which the hook-curl defender can rally to, and take away the 18-yard corner route.
  • Widen the safety. A safety who aligns wider, or who reads the inside receiver’s release and goes straight to the corner route, gets there sooner, at the cost of the seam and the deep middle.
  • Change the coverage to that side. Against a team that loves smash, a defense can play Cover 2 to one side and quarters to the other (Cover 6), or play Palms (also called 2-Read), where the corner lines up off and stays on the outside receiver while the safety reads the slot and takes him when he goes vertical: the corner route runs into the safety and the hitch runs into the corner. Both are in Quarters and the Two-High Match Family. Or it can play 2-Man, where the corner ignores the zones and follows his man, and the corner route runs into a trailing slot defender and a safety over the top.

Smash also shows why play-calling is a guessing game. Smash is a great call against Cover 2 and a mediocre one against Cover 3, whose corners drop deep into the corner route’s path. Which is why defenses that play a lot of Cover 2 try hard to make it look like something else before the snap (Disguise, Rotation, and Split-Field Coverage), and why What Beats What puts smash versus Cover 2 in the first row of its matrix.

Tampa 2: the Mike takes the middle

Go back to thin spot ③, the deep middle. Cover 2 has nobody there, and offenses learned to send a receiver straight up it: a tight end, a slot receiver, a running back out of the backfield. The Mike, sitting in the hole at 10 to 12 yards, watches him run past. Each safety is busy with his own half. The ball goes over the Mike and between the safeties.

Tampa 2 answers that with one change. At the snap, if it’s a pass, the Mike doesn’t stop at 10 yards. He opens (usually toward the side with more receivers), reads the quarterback’s eyes and whichever inside receiver goes vertical, and runs down the middle of the field to 15 or 20 yards. He stays on top of and inside that receiver and carries him, and he bumps anything crossing his path on the way. Coaches call his landing spot the “pipe” or the “middle of the field”: the space between the two deep safeties. Everything else is Cover 2: two safeties on the deep halves, squat corners in the flats, two outside linebackers (or a linebacker and the nickel) in the hook-curl zones.

How does he know it’s a pass in time? The same way every linebacker does: he reads the offensive linemen in front of him. Linemen who stand up and step back into pass protection (“high hats,” in coaching shorthand) mean pass; linemen firing forward mean run. He has to make that call in his first step.

The safeties are free to play wider, because the middle is covered, which also shrinks the thin spot along the sideline. Now do the arithmetic, because it is the whole point. At the snap there are five underneath defenders, as in any Cover 2, so the quick game still runs into a crowd. About a second later the Mike has run past the short routes, and the coverage is four underneath and three deep. Tampa 2 gives up its fifth underneath defender only once the throws he was there to stop have mostly come and gone.

What does it give up? The Mike’s departure opens a new thin spot of its own: the middle of the field at about 10 to 15 yards, in front of the running Mike and behind the hook-curl defenders (the green patch in the last panel). A dig or a curl by the inside receiver, timed to sit down where the Mike was, is the classic Tampa 2 answer, and offenses have named concepts built for it: dagger sends the slot up the seam to drag the Mike deep and throws a deep dig behind him, and levels and drive stack two in-breaking routes at two depths under him (Dropback Concepts). The Mike also has to make a decision on every snap: run, or stop because it’s a run play, a screen, or a draw. A Mike who runs on a draw has left the middle of the field to the ball carrier.

What it solves. Cover 2’s worst thin spot, the deep middle, without moving a safety. It keeps five defenders underneath against the quick game and gives one up only once the Mike has run past the short routes. It also turns a two-high picture into a three-deep coverage after the snap, so the quarterback who saw “middle of the field open” and decided to throw the seam finds it closed.

What it gives up. The intermediate middle behind the hook-curl defenders, and a lot of the Mike. He has to read run or pass instantly, then run 15 to 20 yards with a tight end or a slot receiver: a linebacker’s job and a safety’s job in one body.

Cover 2 against trips

Everything so far has been against two receivers on each side. Put three on one side (trips, or 3x1) and plain Cover 2’s rules leave a gap. On the trips side, the corner sits on #1’s flat, the nickel walls #2, and the safety has a whole half with three receivers in it. If #3, the innermost, runs straight up the middle, he is past the Mike at 12 yards and in the deep middle with nobody near him. Meanwhile, on the other side, the backside safety has a half with only one receiver in it, X, and the backside corner is already sitting short on X’s side.

Cover 2 teams don’t just accept that. Against 3x1 they adjust, and the common answers are:

  • Bring the backside safety’s eyes inside. He has only X, and the corner is already on X’s side, so he is told to help on #3 going up the middle and leave X to the corner and the sideline.
  • Give #3 to the Mike. This is the Tampa 2 rule exactly: the Mike carries #3 vertical up the pipe. Trips is the formation where the Tampa Mike earns his money.
  • Get out of Cover 2. Many defenses simply check to a different coverage against trips, one built for three receivers on a side; those 3x1 checks are in Disguise, Rotation, and Split-Field Coverage.

Which answer a team uses is a big part of its identity, and the offense’s game plan is often built around finding out.

Why it existed: a defense for fast, smart, ordinary-sized players

The coverage didn’t come first. The players and the philosophy did.

Its roots are in Pittsburgh. Bud Carson, the Steelers’ defensive backs coach from 1972 and their defensive coordinator from 1973 to 1977, had started building a two-deep zone at Georgia Tech, and in Pittsburgh he refined it into the coverage behind the “Steel Curtain,” the nickname of the Steelers’ 1970s defense (and especially its four-man line). Its signature was a middle linebacker who dropped into coverage: when the Steelers got Jack Lambert, a rangy, fast middle linebacker, in 1974, Carson could ask him to run deep down the middle. The 1976 Steelers allowed 28 points in total during a nine-game winning streak at the end of the season.4 One of the safeties on the 1977 and 1978 Steelers was an undrafted rookie from Minnesota named Tony Dungy. After a season with the 49ers and a year coaching defensive backs at his old college, he returned to Pittsburgh as an assistant in the early 1980s and was its defensive coordinator from 1984 to 1988.5 Years later he described his whole approach in one sentence: “My philosophy is really out of the 1975 Pittsburgh Steelers playbook.”6

When Dungy became the Tampa Bay Buccaneers’ head coach in 1996, he brought that playbook and a defensive coordinator, Monte Kiffin, who had been on Dennis Green’s Minnesota Vikings staff with him in the early 1990s. Kiffin coordinated Tampa Bay’s defense for thirteen seasons, 1996 to 2008, and the two of them turned the old Steelers idea into an identity, so tightly linked to the franchise that the coverage took the city’s name.7

Their version was built on a few stubborn ideas:

  • Simplicity. A small number of calls, practised until every player could play them without thinking. Players who aren’t thinking play fast.
  • Speed over size. Lighter, faster players at every level: a quick, penetrating 3-technique defensive tackle (Warren Sapp) who penetrated rather than held his ground, linebackers who could run like safeties, and a 4-3 front in which every lineman attacked one gap (Even Fronts).
  • Four rushers, seven in coverage. No exotic blitzes. The four-man rush was expected to get home on its own, which is why the scheme prized pass-rushing linemen: if four can’t win, seven defenders can’t cover forever.
  • Everything in front, then tackle. Spot-drop zones, every underneath defender’s eyes on the quarterback, and a sprint to the ball the moment it’s thrown. Let the offense complete short passes, tackle immediately, and wait: a 12-play drive has a dozen chances for a penalty, a sack, a dropped pass or a turnover.

The Mike’s deep drop fits all four. It needs a fast linebacker rather than an extra safety, it uses no extra rusher, and it removes the one throw that turns a long drive into a short one.

NoteMadden vs. real life

In a video game you can call Tampa 2 with any eleven players and the Mike will run his drop. On a real field the coverage is only as good as the man running down the middle. Tampa 2 Mikes have to diagnose run or pass in a step, then turn and run 15 to 20 yards with a receiver while reading the quarterback. The players who made it famous were unusual athletes: Chicago’s Brian Urlacher, for example, had played a hybrid linebacker-safety position in college at New Mexico.8 A Tampa 2 with an ordinary middle linebacker is a Cover 2 with an extra hole.

2-Man: two deep, man underneath

Keep the two deep safeties, but change what the five underneath defenders do. Instead of dropping to zones, each one takes a receiver, man-to-man. That’s 2-Man, called Cover 5 in many playbooks and “Cover 2 Man” in video games and on many broadcasts. Don’t look for logic in the 5: unlike Cover 1, 2 and 3, it doesn’t count deep defenders. It is simply the number many systems gave this call, and some systems use 5 for something else entirely.

The logic of 2-Man is a cleaner, more aggressive version of the logic of Cover 2. Each man defender knows that anything deep belongs to the safety over the top, so he can play underneath and inside his receiver, in trail technique: a step behind him, on his hip, ready to undercut any route that breaks inside or stops short. The receiver’s only open space is deep, and deep is where the safety is waiting. In effect every outside receiver is in a bracket: the man defender underneath, the safety on top.

When it’s called. 2-Man is a long-yardage and end-of-half coverage. On third-and-8 the offense has to throw past the line to gain, and it knows where: receivers run to 8 or 9 yards and sit down in the seams between zone defenders, right at the sticks. Against Cover 2 zone, a hook-curl defender may be 3 yards away when that ball arrives, close enough to tackle but not to stop the first down. In 2-Man the receiver at 8 yards has a defender on his hip, so he has to be beaten, not just found. 2-Man contests every short and medium throw with a defender on the receiver’s hip and still keeps two players deep, so the offense can’t throw over the top. What 2-Man gives up. Everything man coverage gives up, plus the things two-deep gives up.

  • The quarterback’s legs. Five defenders have their backs turned to the quarterback, chasing receivers, and only four are rushing. If the pocket opens, there is nobody watching the quarterback, and a scramble up the middle can go for 15 yards. The usual counter uses the one defender who may be free: the man assigned to the running back. If the back stays in to block, that defender either rushes too (a “hug” or “green-dog” rule, from Man Coverage) or sits and mirrors the quarterback as a spy. Even so, 2-Man against a running quarterback is a risk every coordinator weighs.
  • The man-beaters. Rub routes and crossing routes like mesh make trailing defenders run through traffic, and routes that break outside (an out, a corner, a fade) go away from the defenders’ inside leverage (Man Coverage has the list).
  • The run. Two safeties deep and five defenders turned to their men: a draw on third-and-8 is a classic call against it.

Telling Cover 2 from 2-Man. Before the snap you can sometimes get a lean: corners pressed tight on the receivers’ inside shoulders, eyes on the receivers, on third-and-long, lean 2-Man; corners 5 yards off and a step outside, lean Cover 2. But it’s only a lean. Defenses know the tell, and plenty of 2-Man is played from the Cover 2 alignment, with corners off and outside until the snap. The answer comes after the snap, in the first second, and the cue is hips and eyes. A zone corner stays square or opens toward the quarterback, his eyes on the ball, while the receiver runs past him. A man corner turns his hips and runs with the receiver, on his inside hip, eyes on the receiver and back half-turned to the quarterback. (Depth alone can fool you: a Cover 2 corner who sinks because nobody threatens his flat can still be near the receiver at 12 yards, but he’ll be looking at the quarterback.)

Why classic Tampa 2 faded, and what survived

In 2005 the Buccaneers had the NFL’s best total defense and the Bears the second-best.16 By 2006 a long list of teams (Buffalo, Minnesota, Kansas City, Detroit and others) had imported the system, often by hiring coaches from Dungy’s and Kiffin’s staffs. A decade later almost none of them played it as a base defense. What happened?

The rules leaned against it. The coverage depended on squeezing receivers’ releases and on underneath defenders who could disrupt routes as they crossed their zones. After the 2003 AFC Championship Game, in which New England’s physical defensive backs (Ty Law above all) took apart Peyton Manning’s Colts, the league made downfield contact past 5 yards a “point of emphasis” for the 2004 season (a rule the league tells officials to enforce more strictly that year), and the flags for illegal contact more than doubled, from 79 in 2003 to 191 in 2004.17 Physical zone coverage became expensive. (There’s an irony in it: the Colts’ complaints drove the change, and Dungy was the Colts’ head coach.) Dynasties and Counterpunches tells the story of that rule and the passing boom that followed.

Offenses learned to live in the seam. A Tampa 2 Mike carrying a tight end up the middle was a fair fight in 1999. Against a 6-foot-6 tight end who ran like a receiver, or a slot receiver lined up as the inside man in a trips formation, it wasn’t. Offenses also learned to attack the hole the Mike vacates, with digs and curls that sit down 12 to 15 yards deep in front of him, and to flood the sideline holes the squat corners leave. Peyton Manning, who practised against Dungy’s Tampa 2 every day in Indianapolis, became one of the best at taking it apart (that, too, is in Dynasties and Counterpunches).

The Mikes were rare. A middle linebacker who can stop the run between the tackles and also run 20 yards with a receiver is one of the rarest athletes in the sport. Teams that copied the scheme without one found out quickly what Cover 2 without a deep middle defender looks like.

Light boxes met better run games. A two-deep, four-rush defense lives with six or seven defenders near the line. That was survivable when offenses lined up in two-back formations and ran into the teeth of it; it was less survivable against spread formations that thinned the box further and against run games designed to exploit it.

Then the league went one-high, and then two-high again, but differently. Through the early 2010s the fashionable answer to everything was Pete Carroll’s Seattle Cover 3, a single-high defense with press corners and an eighth defender near the line (The Cover 3 Family). (Gus Bradley, Seattle’s defensive coordinator from 2009 to 2012, had coached linebackers under Kiffin in Tampa.) When two-high defense came back after 2018, it came back as quarters and match coverage: two safeties who read the receivers in front of them and fit the run, not two safeties guarding halves while the Mike runs the middle (Quarters and the Two-High Match Family).

What survived

A great deal, just not under its old name.

  • Cover 2 itself. It never left. Next Gen Stats counted 13.9 percent of 2025 dropbacks as Cover 2 (the source is in the note on 2-Man’s situations), and the FTN charting in nflverse labels even more; it remains a standard call on long yardage, at the end of halves, and as a change-up for defenses whose base is something else.
  • The Mike who carries the seam. “The middle defender carries #3 vertical” (if the third receiver from the sideline runs straight up the field, the middle linebacker runs with him) is now a rule in Cover 3 match and quarters, the descendants of spot-drop zone. Any time you see a linebacker turn and run with a tight end up the middle of the field in a modern zone, that’s the Tampa 2 Mike’s job inside a newer coverage (Cover 3 match).
  • Two-high pictures that become three-deep. The Tampa 2’s real trick was showing two deep safeties and playing with three deep defenders after the snap. Modern defenses do the same thing by rotating a safety instead of running the Mike, and that is the whole subject of Disguise, Rotation, and Split-Field Coverage.
  • Cover 2 as half a coverage. Cover 6 plays quarters to one side and Cover 2 to the other, squat corner and all; split-field defenses use the Cover 2 half to the short side of the field or against a formation’s lone receiver (Quarters and the Two-High Match Family).
  • 2-Man on third down, as the chart above shows.
  • The people. The Tampa staffs of 1996–2008 seeded a generation of NFL coaches. Lovie Smith and Rod Marinelli took the system to Chicago and Detroit; Mike Tomlin, Tampa Bay’s defensive backs coach from 2001 to 2005, and Raheem Morris, who worked his way up the same secondary, became head coaches.18 The approach (simple calls, fast players, eyes on the quarterback, rally to the ball) outlived the coverage, and a branch of the tree, Matt Eberflus, ran some of the most Cover 2-heavy defenses of the charting era in Indianapolis and Chicago.19
TipWatch for it on Sunday: the squat-corner spotter

On every passing down for one quarter, look at the cornerbacks first.

  • Before the snap. Corners 5 to 7 yards off, a step outside the wide receiver, flat-footed, shoulders turned in toward the ball? Two safeties deep (12 to 14 yards) and wide (outside the hashes)? Think Cover 2. Corners pressed on the receiver’s inside shoulder on third-and-long? Lean 2-Man, but confirm it after the snap. Corners 7 or 8 off with two safeties deep? It may be quarters (Quarters and the Two-High Match Family), or a Cover 2 that squats only at the snap (Disguise, Rotation, and Split-Field Coverage).
  • At the snap. Does the corner punch the receiver and then stop, hips square, eyes on the quarterback, while the receiver runs past? Cover 2. Does he turn his hips and run with him, eyes on the receiver? 2-Man (or a disguise into something else).
  • One second later. Find the Mike. Did he stop at 10 yards, eyes on the quarterback? Cover 2. Is he still running straight down the middle at 15? Tampa 2.
  • When the ball is thrown. Where did it go? A sideline throw between the corner and the safety, or a seam between the safeties, is the offense attacking exactly the holes this chapter mapped. A completed 6-yard out on third-and-12, followed by an immediate tackle, is the coverage winning.

That is Tampa 2: the Mike carries #3 up the pipe, which is exactly the throw it was built to take away. Then the question becomes who is running up the middle. A big, slow tight end against a fast linebacker is a fair fight; a fast tight end or a slot receiver against a linebacker 15 yards downfield is not. And the dig by H, sitting down at 12 to 14 yards where the Mike used to be, is open underneath him.

(c) Look at the backside safety, the one on X’s side. He has only X in his half, and his corner is already sitting short on X’s side, so the most common adjustment is to give him #3: he cheats a couple of yards toward the middle before the snap, or he drifts there at the snap, and Y’s seam runs into him. Other defenses hand #3 to the Mike every snap, or check out of Cover 2 entirely against trips (Disguise, Rotation, and Split-Field Coverage). A backside safety narrower than his partner is your pre-snap hint.

Cover 2 or 2-Man? Which is more likely, what single thing would you watch in the first second, and how would you design the pass so the quarterback has an answer either way?

TipAnswer

You can’t be sure, and that is the point of the alignment. Third-and-8 is where 2-Man peaks, but even there Cover 2 is called about twice as often: on third or fourth and 7 to 10, the charting behind the down-and-distance chart has Cover 2 on roughly 12 to 17 percent of dropbacks and 2-Man on 6 to 10. And the corners are showing Cover 2’s alignment, off and outside, not the press-inside look of a 2-Man lean. So the base rate says Cover 2, with 2-Man very much alive.

Watch the corners’ hips and eyes. If a corner opens his hips and runs with his receiver, eyes on him, it’s 2-Man. If he stays square, eyes on the quarterback, and lets the receiver go, it’s Cover 2.

Design for both. Offenses facing this picture often put a zone-beater on one side and a man-beater on the other and let the quarterback pick after the snap: smash to one side (it beats the squat corner in zone), and a rub or an out-breaking route to the other (it beats a man defender playing inside leverage). And if it is 2-Man and nobody is open, the quarterback’s legs are the answer: five defenders have their backs to him, unless the one on the running back has been told to rush or spy. Throwing deep is the one thing not to do: two safeties are waiting there either way.

Takeaways

  • Cover 2 is two deep and five under. The safeties each guard a deep half; the corners guard the flats; three linebackers (or two and a nickel) guard the hook-curl zones and the hole. It takes away the short throws and everything over the top, and it costs a defender in the box.
  • The hard (squat) corner is the tell. Five to seven yards off, outside leverage, eyes inside; at the snap he jams, sinks and sits in the flat. When you see two of them with two deep safeties, think Cover 2, then confirm it after the snap: a corner whose hips turn to run with the receiver is playing 2-Man.
  • Cover 2’s thin spots are known by heart: the hole shot along the sideline, the smash corner over the squat corner (the same grass, reached from inside), the deep middle between the safeties, and the intermediate sideline hole. Deep passes go to the middle against Cover 2 about one time in three, as often as or more often than against any other coverage in the charting data.
  • Tampa 2 sends the Mike down the middle: five underneath at the snap, then four under and three deep. It closes the deep middle, needs a rare linebacker, opens the intermediate middle (dagger and levels live there), and is Cover 2’s best answer to trips. It was the identity of Dungy and Kiffin’s Buccaneers, Dungy’s Colts and Lovie Smith’s Bears.
  • 2-Man keeps the two deep safeties and plays man underneath, in trail technique with inside leverage. It’s a long-yardage and end-of-half call, vulnerable to scrambles, rubs and out-breaking routes.
  • Classic Tampa 2 faded (contact rules, seam-stretching offenses, rare Mikes, light boxes, the move to Cover 3 and then to quarters), but its pieces survive: Cover 2 as a call, the seam-carrying middle defender, two-high pictures that become three-deep, and a large coaching tree.
