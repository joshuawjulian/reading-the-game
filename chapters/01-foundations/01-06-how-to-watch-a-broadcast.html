# --- the broadcast camera's footprint on the field ---------------------------------------------
# The main camera sits high on one sideline near the 50; here, the offense's LEFT sideline (the
# left edge of every diagram). Seen from above, what it shows is a trapezoid: narrow on the near
# side, wider on the far side, because the far side is farther from the lens.
TV_WIDE = (8.5, 11.0, 12.0, 13.0)   # half-length along the field at the near / far edge; near / far reach
TV_TIGHT = (6.0, 7.5, 6.5, 7.5)


def tv_footprint(cd, cw, along_near, along_far, near_off, far_off):
    """(d, w) corners of the camera's footprint centred on (cd, cw). Near side = offense's left (w < 0)."""
    return [(cd - along_near, cw - near_off), (cd + along_near, cw - near_off),
            (cd + along_far, cw + far_off), (cd - along_far, cw + far_off)]


def _clip(poly, x0, x1, y0, y1):
    """Clip a convex polygon (BDB x, y) to a rectangle (Sutherland-Hodgman)."""
    def clip(pts, inside, inter):
        out = []
        for i in range(len(pts)):
            cur, prev = pts[i], pts[i - 1]
            if inside(cur):
                if not inside(prev):
                    out.append(inter(prev, cur))
                out.append(cur)
            elif inside(prev):
                out.append(inter(prev, cur))
        return out
    ix = lambda v: (lambda p, q: (v, p[1] + (q[1] - p[1]) * (v - p[0]) / (q[0] - p[0])))  # noqa: E731
    iy = lambda v: (lambda p, q: (p[0] + (q[0] - p[0]) * (v - p[1]) / (q[1] - p[1]), v))  # noqa: E731
    pts = [tuple(p) for p in poly]
    for inside, inter in [(lambda p: p[0] >= x0, ix(x0)), (lambda p: p[0] <= x1, ix(x1)),
                          (lambda p: p[1] >= y0, iy(y0)), (lambda p: p[1] <= y1, iy(y1))]:
        pts = clip(pts, inside, inter)
        if not pts:
            break
    return pts


def _signed_area(v):
    v = np.asarray(v)
    return 0.5 * np.sum(v[:, 0] * np.roll(v[:, 1], -1) - np.roll(v[:, 0], -1) * v[:, 1])


def shade_outside(fld, play, poly_dw, alpha=0.78, outline=True, ls="-"):
    """Grey out everything in the field window except the camera footprint. Returns the artists."""
    xs, ys = play.to_abs(np.array([p[0] for p in poly_dw]), np.array([p[1] for p in poly_dw]))
    poly = _clip(list(zip(xs, ys)), fld.x0, fld.x1, fld.y0, fld.y1)
    inner = [tuple(map(float, fld.xy(x, y))) for x, y in poly]
    outer = [tuple(map(float, fld.xy(x, y))) for x, y in
             [(fld.x0, fld.y0), (fld.x1, fld.y0), (fld.x1, fld.y1), (fld.x0, fld.y1)]]
    if _signed_area(outer) < 0:
        outer = outer[::-1]
    if _signed_area(inner) > 0:
        inner = inner[::-1]
    verts = outer + [outer[0]] + inner + [inner[0]]
    codes = ([MPath.MOVETO] + [MPath.LINETO] * 3 + [MPath.CLOSEPOLY]
             + [MPath.MOVETO] + [MPath.LINETO] * (len(inner) - 1) + [MPath.CLOSEPOLY])
    arts = [fld.ax.add_patch(PathPatch(MPath(verts, codes), fc=MASK, ec="none", alpha=alpha, zorder=12))]
    if outline:
        arts.append(fld.ax.add_patch(Polygon(inner, closed=True, fill=False, ec=style.INK, lw=1.5, ls=ls,
                                             zorder=13)))
    return arts


# --- the chapter's one play: a deep shot down the far sideline --------------------------------
def base_play(to_go=6, los=76.0):
    """Gun 2x2, 11 personnel, against a nickel defense with two deep safeties.
    Default spot: third-and-6 at the opponent's 34 (BDB x = 76), the moment from the opening."""
    off = offense("gun_2x2")
    dfn = defense("nickel", "two_high", off)
    # safeties 13 deep outside the hashes; the WILL apexed between the left tackle and the slot (H)
    adj = {"FS": dict(d=13.0, w=-9.5), "SS": dict(d=13.0, w=9.5), "WILL": dict(d=5.0, w=-5.5)}
    dfn = [p.moved(**adj[p.key]) if p.key in adj else p for p in dfn]
    return Play(off, dfn, los=los, to_go=to_go)


def shot_play():
    """Cover 2 against four receivers. The SS (deep half, offense's right) jumps Y's curl, and Z's
    go route runs past the squatting corner into the space he left. Ball to Z at 2.5 s."""
    p = base_play()
    p.route("X", r.go(24)); p.route("H", r.seam(18)); p.route("Y", r.curl(11)); p.route("Z", r.go(33))
    p.dropback(2.5)
    for k, dw in [("LT", -0.6), ("LG", -0.2), ("C", 0.0), ("RG", 0.2), ("RT", 0.6)]:
        p.block(k, pts=[(-1.6, dw)])
    p.block("RB", pts=[(0.6, 1.8)])
    for k in ("WDE", "WDT", "SDT", "SDE"):
        p.rush(k, pts=[(-1.4, 0.0)], speed=1.8)
    p.drop("MIKE", (8.0, 3.0)); p.drop("WILL", (8.0, -6.0)); p.drop("NB", (8.0, 11.5))
    p.path("LCB", [(9.0, -15.8)], speed=4.0, absolute=True)       # Cover 2 corners sink, stay under
    p.path("RCB", [(9.0, 15.8)], speed=4.0, absolute=True)
    p.drop("FS", (24.5, -12.0), speed=6.0)                         # deep half, on top of X
    p.path("SS", [(11.4, 8.4)], speed=1.5, absolute=True)          # bites forward on the curl...
    p.path("SS", [(29.0, 14.2)], speed=7.4, delay=0.15, absolute=True)   # ...then turns and chases
    p.pass_to("Z", t=2.5)
    return p


def catch_time(p, key, t_throw):
    for t in np.arange(t_throw, t_throw + 4.0, 0.02):
        if np.linalg.norm(p.ball_position(t) - p.position(key, t)) < 0.35:
            return float(t)
    return t_throw + 1.5


def tv_at(p, t, t_throw, t_catch, pre=(1.5, 0.0)):
    """Where the broadcast camera is pointed at time t: wide on the formation until the throw,
    then panning with the ball and tightening until it is on the receiver at the catch."""
    b = p.ball_position(t)
    s = float(np.clip((t - t_throw) / (t_catch - t_throw), 0, 1))
    s = 1 - (1 - s) ** 2          # the operator reacts fast, then settles on the catch
    cd = (1 - s) * pre[0] + s * (b[0] + 2.0 * (1 - s))
    cw = (1 - s) * pre[1] + s * b[1]
    ext = [(1 - s) * a + s * c for a, c in zip(TV_WIDE, TV_TIGHT)]
    return tv_footprint(cd, cw, *ext)


def tracking_with_id(p):
    trk = p.to_tracking()
    trk["_id"] = trk["displayName"]
    return trk


def draw_frame_ball_aside(fld, cur, trails, ppy):
    """gridiron's draw_frame, but the football is drawn smaller and nudged half a yard upfield and
    to the side whenever a player holds it, so it never hides the QB's or the receiver's marker."""
    ball = cur[cur.side == "ball"]
    arts = draw_frame(fld, cur[cur.side != "ball"], trails, ppy)
    for _, b in ball.iterrows():
        bx, by = float(b.x), float(b.y)
        pl = cur[cur.side != "ball"]
        dist = np.hypot(pl.x - bx, pl.y - by).min() if len(pl) else 9.0
        if dist < 1.3:
            bx, by = bx + 0.6, by - 1.15
        px, py = fld.xy(bx, by)
        e = Ellipse((float(px), float(py)), 0.85, 0.55, fc=style.BALL, ec="white", lw=0.6, zorder=11)
        fld.ax.add_patch(e)
        arts.append(e)
    return arts


def frame_at(trk, t):
    times = trk.groupby("frameId")["time"].first()
    return int((times - t).abs().idxmin())

Third-and-6 at the opponent’s 34, 2:41 to play, down by three. The broadcast shows you the quarterback barking at his line, cuts to a tight shot of the center’s hands, and the ball is snapped. For two seconds you watch linemen shove each other. Then the camera whips upfield and a receiver who was not on your screen a moment ago catches the ball at the 2-yard line, with a safety you never saw arriving a step too late. The analyst says, “The safety jumped the short route, and that’s the easiest throw he’ll make all day.” Which safety? Jumped what? You were watching the whole time, and you didn’t see any of it. That is not your fault. A broadcast is built to show you the ball, and most of what decides a play happens away from it. This chapter is about watching around the broadcast: what the camera leaves out, where the other angles come in, what to read in the few seconds before the snap, and how to keep score of your own guesses so that, by the end of this course, you can tell whether you are actually getting better at reading the game.

You’ll be able to…

  • Say what the main TV camera hides (usually the safeties and the outside receivers) and which replay angle (end zone, skycam, the wide All-22) gives back each missing piece.
  • Use the pause before each snap to read down, distance, field position, score, clock and timeouts from the scorebug in under five seconds, and turn them into a run-or-pass lean.
  • Pick one thing to watch on each snap and on each replay, chosen by the situation, instead of following the ball.
  • Read the win-probability bar and the Next Gen Stats numbers a broadcast puts on screen as plain-English estimates (the models behind them come later).
  • Spot four starter tells (the linemen’s stance, the running back’s depth and offset, receiver splits, safety depth) and treat each as a clue that shifts the odds, not a promise.
  • Keep a prediction log on paper or in a dozen lines of pandas, score it, and compare it with “always guess pass” and with what the down and distance alone would have earned you.

The camera is built for the ball

Almost everything you have ever seen of an NFL game came through one camera. The sideline camera is the broadcast’s main camera: it sits high on one side of the stadium, usually near the 50-yard line on the press-box side, and it follows the ball. Before the snap the director frames the formation fairly tightly, so the quarterback, the linemen and the players near them fill the screen. At the snap the operator keeps the ball in the middle of the picture, and when the ball is thrown he swings with it.

That is the right choice for television. The ball is the story, the faces are the drama, and a tight shot reads well on a phone. The cost is everything else. Here is roughly what a typical pre-snap shot covers, drawn from above:

Count what made the screen: the five linemen, the quarterback, the running back, the two inside receivers, the four defensive linemen, two linebackers and the slot defender. Sixteen players. The six you can’t see are the two safeties (the deepest defenders, usually 10 to 15 yards off the ball; see The Twenty-Two), the two wide receivers split out near the sidelines, and the two cornerbacks covering them. If you want to know whether a pass will work, those are exactly the six you most want to see. The safeties tell you how the defense is covering the deep part of the field; the outside receivers and corners are where most long throws go. Notice that the receiver on the camera’s own side is hidden too: the near side is the narrow end of the trapezoid, so the bottom of your screen cuts him off just as the far edge cuts off his partner. Which team’s sideline the camera sits on varies by stadium (it is usually the press-box side), but the shape of what you lose doesn’t.

def n_on_screen(t):

ImportantWhy it exists: a picture made for the ball

Why don’t networks just show the wide shot? Because the broadcast is made for a much larger audience than the people who want to study coverage. A wide shot makes the players tiny, and the things most viewers care about (who has the ball, the hit, the catch, the celebration) get lost in it. So the main camera trades context for clarity: it shows you the ball clearly and the structure of the play hardly at all. The good news is that the broadcast also shows the same play three or four more times, from other angles, a few seconds later. The skill is to use those replays on purpose.

The other angles: replays, the end zone and the skycam

Between plays, the director cuts to replays. On an ordinary play you might get one; on a big play you’ll typically get a short sequence, and each angle answers a different question:

  1. Live, from the sideline camera. The ball, framed tightly.
  2. A tight replay on the key player, often from a handheld camera at field level: the catch, the hit, the run. Great for drama, useless for structure.
  3. The end zone. The end-zone camera sits behind one end zone and looks straight down the field. It is the best angle there is for the line: which linemen moved forward and which backward, where the gaps opened, which defender the running back cut off. Coaches film includes a high version of it.
  4. The skycam. The skycam is a camera hung from cables above the field that can fly anywhere over it. On replays it usually sits a few yards behind and above the offense, roughly where the quarterback’s eyes are, and shows the defense the way he saw it: where the safeties went at the snap, who dropped into coverage, which receiver came open. You’ll recognise it by the picture: the backs of the offensive linemen along the bottom of the screen, the field stretching away from you, and both safeties small near the top.
  5. Super slow motion on the catch or the tackle, especially if a review is possible.

Directors vary the order and the mix, and some games use the skycam as a live angle for a few plays. But the logic is constant: the live shot follows the ball, and the replays fill in the rest, if you know which question each one answers.

The skycam has a nice history of being too early. It was invented by Garrett Brown, the cameraman-engineer who had already invented the Steadicam, and first used at an NFL preseason game between the Chargers and the 49ers in San Diego in August 1984. It then mostly gathered dust: it was expensive to fly, and the computers and motors of the day made it slow and finicky. It became a regular part of NFL broadcasts only after the XFL, a short-lived spring league, leaned on it heavily in 2001.1 Seventeen years from invention to routine: the angle that best shows you the coverage was there all along, waiting for the broadcast to decide coverage was worth showing.

NoteFilm room: Prime Vision on Amazon’s Thursday Night Football (2022–)

The closest thing to watching like a coach live is an alternate stream. When Amazon took over Thursday Night Football in 2022 it added a second feed, Prime Vision with Next Gen Stats, which uses a higher, wider camera during each play, so every player stays in view, and draws the receivers’ routes as they run.2 In 2023 it added “Defensive Alerts,” which uses the players’ pre-snap movement to mark the defenders most likely to rush the quarterback, a computer doing the blitz-spotting you will learn to do by eye in A First Look at Defense.3 Later seasons added features such as coverage identification (man or zone, called in real time), also computed from player tracking (the data source is explained below).

What to notice if you try it: how much more of the defense you can see during the play, and how often the defenders it marks as likely rushers are not the ones who rush. The model is reading the same kind of clues you are about to learn, and it is right more often than not, not every time. Alternate streams come and go by season and by network; if one is offered for a game you’re watching, it is worth a quarter of your time.

All-22: the coaches’ view

Coaches don’t study the broadcast. They study coaches film: every snap shown twice, once from a high, wide sideline camera and once from a high end-zone camera, with no replays, no close-ups and no graphics. The sideline half of it is the All-22, named for the one thing it promises: the camera is zoomed out far enough that all 22 players stay in the frame from the snap until the whistle. It looks dull on first viewing (tiny players, no drama), and it is the single most useful football picture there is.

For decades it was available only to teams and a few media analysts. In 2012 the NFL began selling coaches film to fans, both the “All-22” and the “High End Zone” views, through its Game Rewind service.4 Today it comes with the league’s NFL+ Premium subscription, which still offers it for the 2026 season; when the league described the package in 2024, the archive went back to the 2022 season.5 Some teams’ own sites and several film-breakdown writers post All-22 clips too. You don’t need it for this chapter, but if you ever want to settle a question the broadcast can’t answer (“was the receiver open?”), that is where you go. Film Study: All-22, Charting a Game, and Scoring Yourself builds a full film-study routine around it.

Here is an odd fact: Madden gives you a better view of football than the TV does. The game’s default camera sits high behind the offense and shows all eleven defenders, which is closer to the skycam or the All-22 than to anything on a broadcast. That is part of why a game-literate viewer can find real broadcasts confusing: in Madden you always know where the safeties are, and on television you almost never do until the replay. If you want the Madden picture of a real play, you need coaches film or the skycam replay. And don’t expect to “see the play” live by watching harder: the analysts who seem to see everything have usually seen the replay or the wider monitor in the booth. Your aim on the live shot is narrower: read the situation and the formation before the snap, watch one thing during the play, and use the replays for the rest.

The pre-snap pause: five things in five seconds

Every play has a pause. The offense breaks the huddle (or turns to look at the sideline, if it doesn’t huddle; see What a Play Really Is), walks to the line, and gets set. From the moment they are set to the snap there are usually a few seconds, during which the broadcast shows the formation and the scorebug. That pause is the most valuable time you have as a viewer, because it is the last moment you can make a guess that counts. Spend the first part of it on the scorebug.

Two warnings from real Sundays. The broadcast often stays on a replay, a sideline shot or a graphic until a second or two before the snap, and a no-huddle offense may snap the ball almost as soon as it is set. When that happens you won’t see the formation at all. Make the call from the scorebug alone and say so in your log (the clue is “situation”); that is a real prediction, just a less informed one.

Read it in the same order every time, and ask one question of each item:

  1. Down and distance (Downs, Distance, and Field Position): How many yards do they need, and how many tries do they have? Bucket it: short (1–3 yards), medium (4–6), long (7 or more; 01-02 split this into long and very long, but one bucket is enough here). On some downs this decides almost everything: in the 2025 regular season, teams passed on about nine in ten third-and-medium snaps and on fewer than one in four third-and-1s.6 On the most common snap of all, first-and-10, it tells you almost nothing, because that is a coin flip. So down and distance swings the odds hardest where they are lopsided, but across a whole game it lifts your hit rate less than you’d think; the chart at the end of this chapter measures exactly how much.
  2. Field position: Where are they? Backed up near their own goal line, offenses play it safe; inside the opponent’s 20 (the red zone) the field shrinks, because the back of the end zone caps how deep a receiver can run, and the playbook changes; around the opponent’s 35 a field goal becomes possible, which changes what fourth down means.
  3. Score: Who needs what? A team behind by two scores late has to pass. A team ahead wants to run and keep the clock moving.
  4. Clock: How much time, and is it running? (The clock rules are in The Game in Fifteen Minutes.) Inside the last two minutes of a half, everything tilts toward passes and the sideline. In 2025, teams trailing by one score in the last two minutes of the fourth quarter passed on 87% of snaps.7 Above two minutes, remember the two-minute warning: the clock stops for free when it passes 2:00, which acts like an extra timeout.
  5. Timeouts: Can they stop the clock? A trailing team with no timeouts can’t afford a run that ends in bounds; a team with three can.

The play clock (the :18 at the right of the bug) isn’t one of the five, but give it a glance as the pause runs down. Late movement, such as a safety walking toward the line or a quarterback changing the play, tends to happen in the last eight or ten seconds, because whoever moves last gives the other side the least time to answer. And when a no-huddle offense snaps with twenty seconds still showing, it is usually trying to deny the defense exactly that time.

Put it together for the example. Third-and-6 is medium on third down, a passing down almost every time. The ball is at the opponent’s 34, so a long field goal (about 51 yards: 34 to the goal line, about seven more back to the holder, and ten for the end zone) is the fallback if the pass fails, which makes a short, safe completion attractive. Down three with 2:41 left and one timeout, they need points, but they don’t need a touchdown, and they can’t waste a down. The two-minute warning will stop the clock soon whatever happens, so a run wouldn’t be crazy, but third-and-6 is still a passing down. Your lean: pass, probably short to medium, to someone near the first-down marker. (If it fails, a modern coach might even go for it on fourth-and-short here rather than try the 51-yarder; the fourth-down math comes in Win Probability and Decision Analysis.) As the opening showed, the offense took the deep shot instead, which is the point of a lean: it is a probability, not a script. That took longer to write than it should take to think. With a few weeks of practice it becomes a glance.

Glossed numbers: the win-probability bar and Next Gen Stats

Broadcasts now put model outputs on screen. You don’t need the models yet, but you do need to know what the numbers mean, so here is a plain-English reading of the common ones. Each is built properly later.

  • Win probability (the bar under the scorebug above). An estimate of how often teams in exactly this situation (score, time left, field position, down and distance, timeouts) have gone on to win, based on many past seasons of games. “41%” means “teams in this spot win about four times in ten.” It is mostly about the situation, not these two teams: many models also start from the pregame betting line, so they know a little about which team is better, but most of what moves the bar is the score, the clock and the field.8 That is why it swings on every big play. The models are built in Win Probability and Decision Analysis.
  • Next Gen Stats overlays. Every player wears small radio chips in his shoulder pads, and sensors mounted around the stadium record where each one is about ten times a second.9 The league turns those positions into numbers that broadcasts quote:
    • Speed (“he hit 21.4 miles per hour”): the ball carrier’s top speed on the play. Fun, and mostly about the open field he had, not just how fast he is.
    • Completion probability (“a 23% throw”): how often a pass with this distance, this receiver separation and this much pressure (how close the pass rushers were to the passer when he threw) is completed, judged from thousands of past throws. A low number on a completion means a hard throw and catch; a high number on an incompletion means an easy one missed.
    • Separation: the gap, in yards, between the receiver and the nearest defender when the ball arrives (the catch or the incompletion). The typical regular receiver averages about three yards over a season; a yard or less is tight coverage.10
    • Time to throw: seconds from the snap to the throw. The typical NFL pass is out in about two and a half seconds,11 which is why everything you want to see before the throw has to happen fast.
    • Expected rushing yards: how far an average runner would have gained from the spot where the runner took the handoff, given where the blockers and defenders were; the difference is rush yards over expected.
    How tracking data is collected, what it can and can’t measure, and how to work with it yourself is in Tracking Data and the Big Data Bowl.
CautionCommon misconception

“The bar said 90% and they lost, so the model was wrong.” A 90% estimate is supposed to be wrong about one time in ten. The fair test of a probability is whether, across many games where it said 90%, the team won about 90% of them. That idea, called calibration, is how you will score your own predictions at the end of the course.

Watch one thing

The scorebug gives you a lean. The next question is where to put your eyes, because you can’t watch 22 players at once, and if you try you’ll end up watching the ball like everyone else. The fix is a simple rule with two halves. Before the snap, scan: the scorebug, the backfield, and the safeties if they are on your screen. That takes a few seconds and several glances. From the snap on, watch one thing, chosen by your lean, and stay on it until the play is over, even when the ball goes somewhere else. Then change it on each replay.

Here is how to choose your one thing:

That one-yard limit is a rule, not a habit, which is why play-action linemen stop short at the line.12 Still, the line’s first step is a strong read, not a perfect one. On a run-pass option the line run-blocks while the quarterback decides, after the snap, whether to hand off or throw, and on a screen the linemen let the rushers through on purpose before releasing to block downfield. Both are built to fool exactly this read, which is one more reason every tell in this chapter comes with a “usually.”

Why does narrowing your eyes help you see more? Because a football play is several smaller contests happening at once (the line against the rush, the receivers against their defenders, the quarterback against the safeties), and each one is readable only if you follow it from the start. A glance at the safeties after the ball is thrown tells you where they ended up; watching them from the snap tells you what they were doing, and that is what explains the play. The broadcast will show you the ball whether you watch it or not.

Tells: clues you can see before the snap

Players can’t hide everything. A lineman about to fire forward stands a little differently from one about to step back; a running back who needs a running start lines up deeper; a defense that wants an extra man near the line has to put him there before the snap. Any such detail that leaks what’s coming is a tell: a clue that shifts the odds, never a guarantee. Coaches hunt for their opponents’ tells all week, and spend just as long hunting for their own, so they can break them (it’s called self-scouting; see Constraint Theory and Sequencing). That arms race is why every tell in this section comes with a “usually.”

Stance: where the weight is

A player’s stance is how he stands just before the snap. In a two-point stance only his two feet are on the ground: for a lineman that is still a crouch, knees bent and hands resting on his thighs, not standing up straight. In a three-point stance one hand is also down (a four-point stance, both hands down, shows up at the goal line). A hand on the ground lets you start low and drive forward, so interior linemen on both sides (the guards, the center, the defensive tackles) mostly use three-point stances. Elsewhere it varies: in offenses that play mostly from the shotgun, the offensive tackles often stand in a two-point stance on most snaps, because a pass blocker’s first step is backward and an upright start makes it quicker; and many edge rushers stand up too.

The tell is in the hand. A lineman with a heavy hand (also called a heavy stance) has his shoulders out over his down hand and a lot of his weight on it; sometimes you can see his knuckles whiten. He is coiled to go forward, which usually means a run block. A lineman whose hand barely touches the grass, hips back and weight on his heels, is ready to step back into a pass set (Pass Protection). Coaches studying film look for a few refinements of the same idea:

  • A tackle’s stagger. A tackle whose outside foot is set well back of the inside one is ready to kick backward into his pass set (the kick slide).
  • A guard tipping a pull. Weight leaning, or eyes peeking, toward one side often means he is about to pull that way to lead a run.
  • A team that puts its hands down only for runs. If a line stands up on passing plays and crouches into three-point stances for runs, the stance itself is the tell.
  • The defensive line, in reverse. A defensive lineman sitting back on his heels may be about to loop sideways on a stunt (two rushers swapping lanes) or even drop off the line into coverage, which some defenses do to confuse the blockers (Zone Blitzes).

Why does a professional give this away? Because physics doesn’t negotiate. A 320-pound man can’t explode forward from his heels or backward from his fingertips, and a tenth of a second lost at the snap is a lost block. Good coaches teach a balanced stance that looks the same for run and pass, and the best linemen manage it, but on ordinary downs, when the play could be either, the weight tends to creep to where the job is. On short yardage it goes the other way: everyone is heavy on every snap, run or play-action, so the tell goes quiet just when you might expect it to shout.

Be realistic about seeing it. The broadcast shows a close-up of the line before the snap only now and then, so treat stance as a replay clue first (the end-zone angle is best) and a live clue when you happen to get the shot.

The running back: depth and offset

Where the running back lines up is a clue to both whether the play is a run and which way it goes.13 Three common setups:

  • Shotgun with the back beside the quarterback, offset to one side. This is mostly a passing look: 79% of shotgun snaps were passes in 2025. When they do run from it, the most common runs send the back across the quarterback’s face, so a back offset to the quarterback’s right usually runs to the left. The reason is the handoff itself: the back takes the ball while moving across in front of the quarterback, so his path already points to the other side. (Zone Running shows the whole play.)
  • Pistol. The quarterback is about four yards back with the back directly behind him, six or seven yards deep (pistol). The back’s central spot hides the direction of the run, and the depth gives him a running start. Only 29% of pistol snaps were passes.
  • Under center, with the back usually six to seven yards deep. Also mostly a run look, 31% passes, but this is the alignment the play fake is built for, as the film room below shows.

Depth matters within each. A back who creeps closer than usual may be about to block or run a quick route; one who sets deeper than usual wants room to read the blocks before he picks a hole.

Receiver splits: how far out they stand

A receiver’s split is how far from the ball (or the nearest lineman) he lines up. The painted numbers are a handy ruler: most outside receivers line up near them. Departures from that default carry information, because a receiver chooses his split to make his job easier:

  • A reduced split (well inside the numbers) puts him close to the formation. That suits a crack block (blocking a linebacker or safety from the outside on a run toward him), a route that cuts across the middle underneath (the trip is shorter), or a route that breaks outside, toward the sideline, and needs room to get there.
  • A wide split (outside the numbers, near the sideline) isolates him. That suits a deep route, a route that breaks inside and wants space to do it, or a quick throw to him; or it simply means his job is to drag his defender out of the way. A route that breaks toward the sideline is cramped from out there, so a wide split makes one less likely.

Splits are a weaker clue than stance or the backfield, since a reduced split can mean three different things. They mainly tell you where to look after the snap. Remember too that the ball’s position on the hash marks (The Game in Fifteen Minutes) changes what “normal” looks like: on the short side of the field, everything is squeezed.

Safety depth: one high or two

The safeties’ depth is the defense’s biggest tell, because the safeties are the defense’s flexible pieces. Two safeties deep, each 10 to 15 yards off the ball on his own side, is a two-high picture: the deep field is protected on both sides, and there are fewer defenders near the line to stop a run. One safety in the deep middle with the other walked down to seven or eight yards is a one-high picture: the defense has added a man near the line (in the box, the area between and just outside the offensive tackles), which helps against the run and makes extra pass rushers possible. A safety who walks down late, with only a few seconds on the play clock, has either seen something he wants to stop or is trying to fool the quarterback. Defenses disguise this on purpose, and Disguise, Rotation, and Split-Field Coverage teaches how to see through it.

Here is the catch for a TV viewer: Figure 8.1 showed that both safeties are usually off your screen at the snap. What you can do live is limited but real. Watch the top edge of the picture, because a safety walking down appears there. Count the defenders near the line you can see: one more than a moment ago is often a safety arriving. Listen to the analyst, who has a wider monitor (“they’re rotating a safety down”). And take the occasional wide pre-snap shot, or an alternate stream like Prime Vision, when you get one. Otherwise, for now, safety depth is mostly a replay habit: check it on the skycam replay, every time, and you will get faster at spotting it live.

NoteFilm room: Rams 2025 — when the tell is the bait

Sean McVay’s Rams lined up under center on 59% of their 2025 snaps, more than any other team, and threw from under center 40% of the time, also the most.14 That is not an accident. Defenses read “under center” as “run”, so the linebackers step up at the snap; the Rams fake the handoff and throw behind them. That is play-action, the pass designed to look like a run, and it is the reason the tell exists and the reason it’s unreliable. League-wide in 2025, about 82% of passes thrown from under center used a play fake, against about 10% from the shotgun. (Matthew Stafford, the Rams’ quarterback, was the league’s MVP that season.15)

What to notice: watch the linebackers’ first step at the fake. If they step forward, they bit, and the skycam replay will show you the receiver running into the space behind them; then look for the safety, who is the last line behind that space. Play-callers often reach for the fake after a good run, but don’t treat that as a law: analysts testing it with play-by-play data found that play-action works about as well whether or not a team has been running well.16 What sells the fake is the look (under center, the same footwork as the real run), not the rushing stats.

CautionCommon misconception

“Under center means run.” On average, yes: only 31% of under-center snaps were passes in 2025. But those passes are the most dangerous ones, because they are built to exploit exactly that belief. Every tell in this chapter works the same way. A clue that is never wrong would be fixed by the coaches within a week; the ones that survive are right usually, and the offense chooses when to make them wrong. So log your guess with a confidence, not a certainty.

The prediction log

You now have a lean from the scorebug and a few clues from the formation. The habit that turns those into a skill is writing them down. A prediction log is your running record of pre-snap calls: for each snap, the situation, your guess, how sure you were, the clue you leaned on, and what happened. You started a rough version in Downs, Distance, and Field Position, guessing run or pass from the down and distance alone and keeping a tally. That tally was your baseline. This is the formal log you’ll keep for the rest of the course.

Why write it down rather than just watch and nod? Three reasons.

  1. Memory flatters you. After the play, the clue that pointed the right way seems obvious, and the guess you got wrong quietly disappears. On paper, it doesn’t.
  2. It tells you which clues work. If you note the clue behind each call, after a few games you can see whether “heavy hands” is making you right or wrong, for you, on the teams you watch.
  3. It makes improvement visible. Reading a game gets better slowly. A number that moves from 62% to 70% over a season is the only reliable way to know it’s happening.

A few rules keep the log honest and quick:

  • Call it before the snap; write it after. Say the call and the confidence out loud during the pause, then jot the line while the replays run. Abbreviate everything. If you have a pause button, use it; if not, fill in only the shaded columns and catch up at the next timeout.
  • Log what the quarterback did with the ball. If he dropped back to pass, it’s P, even if he was sacked, scrambled, threw it away or threw a screen; a run-pass option counts as whatever he actually did. If he handed off or ran a designed run (a sneak included), it’s R. Skip kneel-downs, spikes and plays wiped out by a penalty.
  • Pick one team for the half, ideally the offense you know better; which one doesn’t matter, as long as you don’t switch.
  • No formation on screen? Call it from the scorebug, write “sit” as the clue, and move on.

Three lessons hide in that chart.17 First, as the scorebug section warned, down and distance alone barely beat “always pass.” It is decisive on the downs where it’s lopsided, but first-and-10, the most common situation by far, is close to a coin flip (about 50% pass in 2025), so a rule built only on down and distance guesses “pass” there and gets half of them wrong. Second, across a whole game the score and the clock add more than the down does: late in a half, and when a team is behind, the run-pass mix swings hard. Third, one glance at the formation is worth almost as much as everything on the scorebug combined. The coaches know this too; it is why the best offenses work so hard to run and pass from the same looks.

So the targets for your own log are clear. If you are under 60%, you’re doing worse than a guesser who never looks up. Around 65–68%, you’re reading the scorebug well. Above 70%, the formation and the tells are working for you. Don’t be surprised if your first log lands in the low 60s; that is what the scorebug alone is worth before you have practised reading it.

NoteGo deeper: why the confidence column?

“Right or wrong” throws away information. A 55% call that misses and a 90% call that misses are very different mistakes, and a good reader knows the difference before the snap. That’s why the log has a confidence column even though we score only hits and misses for now: it costs one number per snap. Film Study: All-22, Charting a Game, and Scoring Yourself scores the season’s log the way forecasters are scored, by whether your 70% calls come true about 70% of the time. Until then, the only thing to log beyond run or pass is nothing: one prediction per snap, made before the snap, is plenty. As the course goes on, chapters add columns (which side the run goes, which receiver gets the ball, whether the safeties rotate, whether the defense blitzes), and by the end your log is a full pre-snap read.

Run or pass? How confident are you, and what is the one thing to watch at the snap?

TipAnswer

Run, about 85%. Third-and-1 is the most run-heavy situation in football: in 2025, teams ran on about three in four third-and-1s, and from under center they ran on about 84%.18 A good share of those runs never reach the tailback: about three in ten of the under-center snaps on third-and-1 were quarterback sneaks (the push-from-behind version, the tush push, is in Red Zone, Goal Line, and Short Yardage). The defense agrees with you, which is why nine defenders are crowded near the line.

Why not higher, when every clue points the same way? Because the clues aren’t independent. Two tight ends, a fullback, under center and a deep tailback travel together: that is simply what a short-yardage formation looks like, and the 84% already counts all of them. Adding them up again would be counting the same evidence three times. The heavy hands add even less here: on short yardage every lineman is heavy on every snap, run or play-action, so in this picture the stance tells you nothing new. Stance earns its keep on ordinary downs.

Your one thing: the line of scrimmage (the short-yardage row of Table 8.2), from the end-zone angle if you get it on the replay. Whose linemen move forward first and farther? On short yardage that yard of push decides the play before the ball carrier arrives.

And the other 15%? This is exactly the picture play-action is built for. With nine defenders charging the line and one safety deep, a fake handoff and a pass to a tight end sneaking out behind the linebackers is the classic counterpunch. A good log line here is “R, 85%, align.” If it turns out to be a play fake, that is not a bad prediction; it is the 15%.

On your TV, what tells you this happened, and how should it change your run-or-pass lean?

TipAnswer

What you’d see: not the walk itself, which starts off screen, but its arrival. A defender appears at the top edge of the picture where nobody stood a moment ago and settles a few yards off the ball. Other on-screen hints: count the defenders near the line, which goes from six to seven; linebackers may shift over to make room; the quarterback may point, shout or change the call (The Pre-Snap Phase); and the analyst, who has a wider monitor, often says it (“they’re rotating a safety down”). The other safety sliding to the middle you won’t see live; that is the skycam replay’s job.

What it means: the defense went from two-high to one-high, and now has seven defenders in the box against six blockers (five linemen and the back). That is the picture from 01-05’s second drill, and the logic is the same: against seven in the box a run has a free defender, so many quarterbacks check out of a run into a pass, and run-pass options throw into the space behind the safety who came down.

Your lean: nudge it toward pass, but only a little. First-and-10 from the shotgun was already a passing look (72% pass in 2025).19 Two things keep you from going much higher. With eight seconds left the quarterback may not have time to check, and the walk-down may be a bluff: some safeties race back to the deep middle at the snap, the disguise game of Disguise, Rotation, and Split-Field Coverage. Five clues: second-and-6, shotgun, the back’s offset, X’s reduced split, two deep safeties. Run or pass, and what does each clue tell you?

TipAnswer

Pass, about 70–75%. Weigh the clues in order of strength, broad ones first:

  1. Second-and-6 on its own is a coin flip: about 52% pass in 2025 for second-and-medium.
  2. Shotgun moves it a long way. In 2025, teams passed on 76% of their shotgun snaps on second-and-5 to -7. That is your call. (The two-point stances agree, but standing linemen are common in the shotgun, so they add little.)
  3. Two deep safeties pull you back a notch. Two-high leaves six defenders near the line against six blockers, so a run is a fair fight, and offenses like to run against two-high and throw against one-high. It doesn’t flip the call; it is why this isn’t 80%.
  4. The back’s offset doesn’t change run-or-pass much. It tells you which way a run would likely go (from the left, across the quarterback’s face, to the right, toward the trips) and which side he would protect on a pass.
  5. X’s reduced split tells you about X, not about the whole play. Run through its three meanings. A crack block fits a run toward him, but the back’s offset says a run would go the other way, so that one is unlikely. That leaves a route underneath across the middle (a dig, which runs ten or so yards and cuts flat inside, or a shallow crosser) whose trip the tight split shortens, or a route breaking outside, like a corner route toward the sideline, that wants the room he has given himself. A slant, the quick cut inside, is the least likely: it needs space inside, and he has just given it away.

So: “P, 75%, align.” Then pick your one thing: X and the cornerback over him, from the snap. If X cuts inside underneath or breaks for the sideline and the ball goes there, the reduced split was the tell. If the ball goes elsewhere, check on the replay what X did anyway. That is how small clues become reliable ones: by checking them, snap after snap, in your log.

Takeaways

  • The broadcast is built for the ball. The main sideline camera keeps the formation tight before the snap and chases the ball after it, so the safeties, the outside receivers and most of the coverage are off your screen when it matters. Replays exist to give them back: the end zone shows the line, the skycam shows the coverage, and the All-22 shows everyone.
  • Use the pause. Read the scorebug in a fixed order (down and distance, field position, score, clock, timeouts) and turn it into a lean before the snap. Down and distance decides the lopsided downs; on first-and-10 the formation tells you more.
  • Scan before the snap, then watch one thing, chosen by your lean: the safeties (or the quarterback’s first look) when you expect a pass, a guard when you expect a run, the line on short yardage, a different group on each replay. The safeties are mostly a replay habit for now, because the live shot rarely shows them.
  • Tells are clues, not promises. Stance, the back’s depth and offset, receiver splits and safety depth all shift the odds, and the quarterback’s alignment shifts them the most. Clues that travel together aren’t separate evidence. The offense knows its tells too, and play-action lives on breaking them.
  • Keep a prediction log, and score it against the bar: “always pass” gets about 60%, down and distance alone about 62%, the whole scorebug about 68%, the scorebug plus the formation about 75%.
